Multi-phase optimizer combining FISTA (softmax-smoothed), LP-guided direction finding,
exact subgradient descent, and coordinate descent for minimizing the step function
spectral metric: 2n * max(a*a) / (sum(a))^2.

Strategy:
- Phase 1: FISTA with adaptive beta at parent size (n=4096)
- Phase 2: LP-guided direction updates interspersed with FISTA
- Phase 3: Exact subgradient + projected steps
- Phase 4: Interpolate to n=8192 and optimize
- Phase 5: Multi-restart exploration
- Phase 6: Coordinate descent final polish

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_conv_at_lag(a, k):
    """
    Gradient of (a*a)_k with respect to a.
    g[j] = 2 * a[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(n - 1, k)
    if j_min <= j_max:
        g[j_min:j_max+1] = 2.0 * a[k - j_min:k - j_max+1][::-1]
    return g


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA with softmax-smoothed max of convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta  # Lipschitz constant estimate
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient of smoothed conv
        y_rev = y[::-1]
        wc = _convolve(w, y_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]
        
        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _lp_direction(f, M, top_k=200):
    """
    Solve the LP to find the best direction g.
    Maximize sum(g) s.t. g >= 0, (f*g)_k <= M for all k.
    Returns the normalized direction or None if infeasible.
    """
    n = len(f)
    # Build constraint matrix (sparse)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # We only include the top_k most binding constraints
    
    # First, compute which constraints are most binding
    ff = _convolve(f, f)
    # Sort lags by ff value (descending) - most binding are the largest
    k_order = np.argsort(ff)[::-1]
    k_selected = k_order[:top_k]
    
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_selected):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(k_selected), n))
    b = np.array([M] * len(k_selected))
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
        return None
    except Exception:
        return None


def _exact_subgradient_step(a, step_size, deadline, max_iter):
    """
    Exact subgradient descent on the max of the convolution.
    At each step, find the lag that maximizes the convolution,
    compute the subgradient, and take a projected step.
    """
    n = len(a)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    # Polyak step size: we need to know the optimal value (lower bound)
    # Use a running lower bound: 2n * (average conv value) = 2n * 1/(2n-1) ~ 1
    # But this is a weak bound. Let's use adaptive step size instead.
    
    # Use backtracking line search
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a_curr, a_curr)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        
        # Subgradient
        g = _grad_conv_at_lag(a_curr, k_star)
        
        # Try decreasing step sizes
        improved = False
        for t in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1]:
            a_trial = _proj_simplex(a_curr - t * g)
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                improved = True
                a_curr = a_trial.copy()
                break
        
        if not improved:
            # No improvement with any step size; reduce step size
            step_size *= 0.8
            if step_size < 1e-10:
                break
    
    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=100):
    """
    Coordinate descent: for each coordinate, find the best value by trying candidates.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _convolve(a, a)
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Process coordinates in order (or random order for large n)
        if n > 4096:
            rng = np.random.default_rng(42 + r)
            order = rng.permutation(n)[:min(2048, n)]
        else:
            order = np.arange(n)
        
        for j in order:
            if time.monotonic() > deadline:
                    break
            
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting to small positive value
                candidates = [old_val + 1e-6, old_val + 1e-5, old_val + 1e-4]
            else:
                # Try several multiplicative factors
                candidates = [old_val * 0.8, old_val * 0.9, old_val * 1.1, old_val * 1.2, old_val * 0.7]
            
            best_candidate = old_val
            best_local_sc = best_sc
            
            for x_new in candidates:
                if x_new < 0:
                    continue
                a[j] = x_new
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_candidate = x_new
            
            a[j] = best_candidate
            a = a / a.sum()
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = a.copy()
                improved = True
                c = _convolve(a, a)  # Update conv
        
        if not improved and r > 5:
            break
    
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
