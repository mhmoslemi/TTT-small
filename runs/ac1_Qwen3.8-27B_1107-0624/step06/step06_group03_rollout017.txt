def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _proj_simplex(v):
    n = len(v)

Refined optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive FISTA, long fine-grained Adam in log-space with aggressive beta scheduling,
dense coordinate descent with golden-section refinement, and KKT-based targeted polishing.
Key improvements over previous: much longer fine-tuning phase, adaptive top-K, 
and precise 1D subproblem solving in coordinate descent.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -80, 80))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr)
    max_c = float(c_init.max())
    L = 16.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y)
        w = _topk_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_fine(a_init, deadline, beta_schedule, lr_init, top_k_start, top_k_end, max_iter_per_beta):
    """Long Adam in log-space with aggressive beta schedule and adaptive top-K."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    n_betas = len(beta_schedule)
    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break

        # Adaptive top-K: linearly interpolate from top_k_start to top_k_end
        frac = bi / max(1, n_betas - 1)
        top_k = max(5, int(top_k_start + (top_k_end - top_k_start) * frac))

        # Learning rate decays slightly within each beta phase
        phase_lr = current_lr * (1.0 - 0.3 * frac)

        it_in_phase = 0
        while it_in_phase < max_iter_per_beta:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a)
            w = _topk_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad = 2.0 * n * a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = phase_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        # Decay learning rate between phases
        current_lr *= 0.85
        # Partially reset momentum
        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_dense(a, n_passes, top_k_w, deadline, grid_size=50):
    """Coordinate descent with dense grid evaluation and golden section refinement."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)
    inv_gn = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            c = _full_convolve(a)
            S0 = 1.0 - a_i_old

            # Direction vector
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = 2.0 * a[valid_j - i]
            # Special case: j = 2i is quadratic
            if 2 * i < 2 * n - 1:
                d[2 * i] = 2.0 * a_i_old  # will handle separately

            c_base = c - a_i_old * d
            # Fix the 2i position (quadratic term)
            c_base[2 * i] = c[2 * i] - a_i_old * a_i_old

            # Evaluate on a grid
            t_lo = 0.0
            t_hi = min(5.0 * max(a_i_old, 0.01), S0 * 5.0)
            if a_i_old < 1e-10:
                t_hi = 0.05
            t_grid = np.linspace(t_lo, t_hi, grid_size)

            def _eval_t(tc):
                # Linear part (excluding 2i)
                c_trial = c_base + d * tc
                c_trial[2 * i] = c_base[2 * i] + tc * tc  # quadratic fix
                max_c = float(c_trial.max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    return np.inf
                return 2.0 * n * max_c / denom

            # Coarse grid search
            vals = [_eval_t(tc) for tc in t_grid]
            vals_arr = np.array(vals)
            idx_best = int(np.argmin(vals_arr))
            best_t = t_grid[idx_best]
            best_val = vals_arr[idx_best]

            # Golden section refinement in the neighborhood
            if 0 < idx_best < grid_size - 1:
                lo = t_grid[max(0, idx_best - 2)]
                hi = t_grid[min(grid_size - 1, idx_best + 2)]
                # Golden section search
                a_g = lo
                b_g = hi
                c_g = b_g - inv_gn * (b_g - a_g)
                d_g = a_g + inv_gn * (b_g - a_g)
                fc = _eval_t(c_g)
                fd = _eval_t(d_g)
                for _ in range(40):
                    if fc < fd:
                        b_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = b_g - inv_gn * (b_g - a_g)
                        fc = _eval_t(c_g)
                    else:
                        a_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = a_g + inv_gn * (b_g - a_g)
                        fd = _eval_t(d_g)
                t_gold = (a_g + b_g) / 2.0
                val_gold = _eval_t(t_gold)
                if val_gold < best_val:
                    best_t = t_gold
                    best_val = val_gold

            if best_t != a_i_old and best_val < np.inf:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1

            if improved % 200 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _kkt_refine(a, deadline, n_iter=200):
    """Targeted refinement using KKT-like conditions.
    Identifies active peaks and tries to equalize them while reducing the max."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    best_a = a.copy()

    rng = np.random.default_rng(123)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        max_c = float(c.max())
        # Find active set: positions where c is within 1% of max
        active_thresh = max_c * 0.99
        active = np.where(c >= active_thresh)[0]

        if len(active) == 0:
            break

        # Compute gradient direction to equalize active peaks
        # For each active position j, we want to reduce c_j
        # Direction: negative gradient weighted by how far above average each active peak is
        active_vals = c[active]
        mean_active = float(active_vals.mean())
        weights = np.zeros(2 * n - 1)
        for idx, j in enumerate(active):
            excess = active_vals[idx] - mean_active
            if excess > 0:
                weights[j] = excess

        if np.all(weights == 0):
            # All active peaks are already equal; try random perturbation
            pert = 0.001 * rng.standard_normal(n)
            a_new = np.maximum(a + pert, 0)
            a_new = a_new / a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            continue

        w = _topk_weights(c, max_c * 0.001, len(active) + 10)
        grad = _compute_grad_conv(a, w)
        mean_grad = float(np.dot(grad, a))
        grad_z = 2.0 * n * a * (grad - mean_grad)

        # Take a small step
        step_size = 0.01 / (np.linalg.norm(grad_z) + 1e-10)
        a_new = np.maximum(a - step_size * grad / (np.linalg.norm(grad) + 1e-10), 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Backtrack
            a_new = a * (1.0 - 0.001) + 0.001 * rng.dirichlet(np.ones(n))
            a = a_new / a_new.sum()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()

                parent = parent_raw / s

    if parent is not None:

        a = np.maximum(a, 0.01)
        parent = a / a.sum()
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Work on the parent length (4096)
    n = len(parent)
    a_work = parent.copy()

    # Phase 1: Quick FISTA to ensure good neighborhood (120s)

    # Phase 2: Long Adam with very fine beta (main workhorse, ~1100s)

        sc_full = _score(a_work)

        max_cc2 = float(c_curr.max())
        betas_p2b = []

                                    max_iter_per_beta=20000)

                                   betas_ult, lr_init=0.002,

    # Phase 5: KKT refinement (50s)
    if time.monotonic() < deadline - 10:

            print(f"  KKT: score={sc_kkt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
