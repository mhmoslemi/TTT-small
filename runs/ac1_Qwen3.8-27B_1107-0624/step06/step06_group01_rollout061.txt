"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses structured initialization exploiting alternating patterns, FISTA with adaptive smoothed max,
Adam in log-space with careful scheduling, block-level local search, and multi-length exploration.
Key improvements: better structured inits, aggressive beta reduction, block operations,
and trying larger sequence lengths (up to 16384).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    if not np.isfinite(step) or step <= 0:
        step = 1.0 / (4.0 * n * max_c / beta + 1e-15)

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    stale_count = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        if not np.all(np.isfinite(grad)):
            break
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_next.copy()
                stale_count = 0
            else:
                stale_count += 1
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L_new = 4.0 * n * max_c_new / beta
            if L_new > L * 10:
                L = L * 2
            elif L_new < L * 0.1:
                L = L * 0.7
            step = 1.0 / L

            if stale_count > 20:
                beta_effective = beta * 0.5
                if beta_effective < 1e-10 * max_c:
                    break

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, lr_init, max_iter_total, beta_frac=0.001):

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / (beta + 1e-30)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    stale_count = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        if not np.all(np.isfinite(grad)):
            break
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_next = _proj_simplex(y - step * grad)

        if it % 80 == 0:
            sc = _score_norm(a_next)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_next.copy()
                stale_count = 0
            else:
                stale_count += 1

            if it > 0 and it % 400 == 0:
                c_new = _full_convolve(a_next, a_next)
                max_c_new = float(c_new.max())
                L_new = 4.0 * n * max_c_new / (beta + 1e-30)
                if L_new > L * 5:
                    L = max(L * 1.5, L_new)
                elif L_new < L * 0.2:
                    L = L * 0.7
                step = 1.0 / L

            if stale_count > 30:
                break

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace_multibeta(a_init, deadline, lr_init, max_iter_total, beta_fracs):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    iters_per_beta = max(100, max_iter_total // len(beta_fracs))

    for beta_frac in beta_fracs:
        if time.monotonic() > deadline:
            break
        if t >= max_iter_total:
            break

        for _ in range(iters_per_beta):
            if time.monotonic() > deadline:
                break
            if t >= max_iter_total:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            beta = beta_frac * cm
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 15 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 5000 == 0:
                current_lr = max(0.0001, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_alternating_init(n, ratio=2.5):
    a = np.ones(n)
    for i in range(n):
        a[i] = ratio ** (0.5 * ((i % 4) // 2))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiscale_init(n):
    a = np.ones(n)
    scale = 2
    amp = 0.6
    while scale <= n:
        for i in range(n):
            if (i // scale) % 2 == 0:
                a[i] *= (1.0 + amp)
            else:
                a[i] *= (1.0 - amp * 0.4)
        scale *= 2
        amp *= 0.65
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sinusoidal_init(n, freq=2.0, amp=0.4):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_init(n, power=0.3):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_uniform_init(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _mirror_blend(a, alpha=0.8):
    a_rev = a[::-1].copy()
    blended = alpha * a + (1.0 - alpha) * a_rev
    blended = np.maximum(blended, 1e-15)
    return blended / blended.sum()


def _block_coord_descent(a, deadline, n_steps=3000, rng=None):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    if rng is None:
        rng = np.random.default_rng(42)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        if step % 40 == 0 and n > 512:
            block_size = int(rng.integers(max(2, n // 500), max(3, n // 30)))
            start = int(rng.integers(0, max(1, n - block_size)))
            a_try = best_a.copy()
            block = a_try[start:start + block_size]
            reassign = block.sum() / block_size
            a_try[start:start + block_size] = reassign
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    continue

        i = int(rng.integers(0, n))
        improved = False
        for factor in [0.85, 0.92, 1.08, 1.15, 0.78, 1.25]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    improved = True
                    break

        if not improved:
            j = int(rng.integers(0, n))
            for swap_frac in [0.2, 0.4, 0.6]:
                a_try = best_a.copy()
                transfer = (a_try[i] - a_try[j]) * swap_frac
                a_try[i] -= transfer
                a_try[j] += transfer
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_try.copy()
                        break

    return best_a, best_score
