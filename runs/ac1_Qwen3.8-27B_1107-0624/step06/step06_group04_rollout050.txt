def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

    n = len(a)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _proj_simplex(v):

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements over previous: peak-directed subgradient polishing, cosine-annealed Adam,
spectral shaping initialization, and adaptive multi-restart fine-tuning.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _peak_shave(a_init, deadline, n_steps=10000, top_k=50):
    """Direct subgradient method targeting the top peaks of auto-convolution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    step = 0.05
    mean_c = 1.0 / (2 * n - 1)

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())

        top_idx = np.argsort(c)[::-1][:top_k]
        top_vals = c[top_idx]
        excess = np.maximum(top_vals - mean_c, 0.0)
        weights = excess ** 3 + 1e-15
        w = np.zeros(len(c))
        w[top_idx] = weights / weights.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        improved = False
        for trial in range(15):
            a_new = _proj_simplex(a - step * grad)
            sc_new = _score_norm(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                step = min(step * 1.02, 0.5)
                improved = True
                break
            else:
                step *= 0.85
        if not improved:
            step = max(0.001, step * 0.9)

def _adam_fine(a_init, deadline, betas, lr_init, max_iter, lr_schedule='cosine'):

    s = a_norm.sum()

        a_norm = a_norm / s

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())

                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0

                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]

            vh = v_buf / (1.0 - b2 ** t)

            if lr_schedule == 'cosine':
                progress = min(1.0, t / max_iter)
                lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            else:

            update_norm = np.linalg.norm(update)

        best_a = a_final.copy()
    return best_a, best_score


def _spectral_shaping(a_init, n_iter=300, alpha=0.2, deadline=None):
    """Iteratively shape the magnitude spectrum to reduce auto-conv peak."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        mean_c = 1.0 / (2 * n - 1)

        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute target: we want the circular autocorrelation to be flat
        # circ_autocorr[k] = ifft(|A|^2)[k]
        # We want circ_autocorr to be close to a delta at k=0 (which means flat |A|)
        # But we actually want LINEAR autocorrelation to be flat, which is harder.
        # As a heuristic: reduce |A| at frequencies where the circular autocorrelation
        # has high off-peak values.

        circ_acf = np.fft.irfft(np.abs(A) ** 2, N_fft)
        # circ_acf[0] is the "peak" (energy), we want the rest to be small relative to peak
        # But that's the opposite of what we want for flat linear autocorrelation...
        # Actually for flat linear autocorrelation, we want the "energy" spread out,
        # which means we want |A|^2 to be FLAT (uniform magnitude).

        # So: flatten the magnitude spectrum
        avg_mag = mag.mean()
        new_mag = (1.0 - alpha) * mag + alpha * avg_mag
        # Add a small regularization to avoid zeroing out DC
        new_mag[0] = max(new_mag[0], mag[0] * 0.99)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}", flush=True)
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}", flush=True)
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: Spectral shaping for better initialization ===
    print("Phase 1: Spectral shaping", flush=True)
    phase1_deadline = min(deadline, t_start + 45.0)
    a_ss, sc_ss = _spectral_shaping(a0, n_iter=200, alpha=0.15, deadline=phase1_deadline)
    sc_ss_full = _score(a_ss)
    if sc_ss_full < best_score:
        best_score = sc_ss_full
        best_seq = a_ss.copy()
        print(f"  Spectral shaping: score={sc_ss_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 2: FISTA with fine beta continuation ===
    print("Phase 2: FISTA beta-continuation", flush=True)
    phase2_deadline = min(deadline, t_start + 280.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    betas = []
    b = 0.005 * max_cw
    while b > 2e-7 * max_cw:
        betas.append(b)
        b *= 0.5
    betas = betas[:12]

    for i, b in enumerate(betas):
        if time.monotonic() > phase2_deadline - 5:
            break
        time_left_per = max(15.0, (phase2_deadline - time.monotonic() - 5) / (len(betas) - i))
        a_work, sc = _fista_smoothed(a_work, b, time.monotonic() + time_left_per, 30000)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={b:.6e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 3: Peak-directed shaving ===
    print("Phase 3: Peak shaving", flush=True)
    phase3_deadline = min(deadline, t_start + 550.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    a_shaved, sc_shaved = _peak_shave(a_work, deadline=phase3_deadline, n_steps=30000, top_k=80)
    sc_shaved_full = _score(a_shaved)
    if sc_shaved_full < best_score:
        best_score = sc_shaved_full
        best_seq = a_shaved.copy()
        print(f"  Peak shave: score={sc_shaved_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 4: Fine Adam with cosine schedule ===
    print("Phase 4: Fine Adam", flush=True)
    phase4_deadline = min(deadline, t_start + 1100.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas_fine = [0.002 * max_cw, 0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw, 0.00005 * max_cw]

    time_left = phase4_deadline - time.monotonic() - 10
    if time_left > 30:
        a_opt, sc_opt = _adam_fine(a_work, time.monotonic() + time_left, betas_fine,
                                   lr_init=0.008, max_iter=600000, lr_schedule='cosine')
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 5: Peak shave again on the Adam result ===
    print("Phase 5: Second peak shave", flush=True)
    phase5_deadline = min(deadline, t_start + 1400.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    time_left = phase5_deadline - time.monotonic() - 5
    if time_left > 20:
        a_shaved2, sc_shaved2 = _peak_shave(a_work, deadline=phase5_deadline, n_steps=15000, top_k=200)
        sc_shaved2_full = _score(a_shaved2)
        if sc_shaved2_full < best_score:
            best_score = sc_shaved2_full
            best_seq = a_shaved2.copy()
            print(f"  Peak shave 2: score={sc_shaved2_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Multi-restart with tiny perturbations ===
    print("Phase 6: Multi-restart fine-tune", flush=True)
    phase6_deadline = min(deadline, t_start + 1650.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    n_restarts = 8
    for i in range(n_restarts):
        if time.monotonic() > phase6_deadline - 40:
            break

        sigma = 0.003 + 0.001 * i
        p_start = _perturb(a_work, sigma, rng)

        c_s = _full_convolve(p_start, p_start)
        max_cs = float(c_s.max())

        p_opt, _ = _fista_smoothed(p_start, 0.001 * max_cs,
                                   time.monotonic() + 10.0, 2000)
        p_opt = p_opt / p_opt.sum()

        if time.monotonic() < phase6_deadline - 50:
            betas_r = [0.001 * max_cs, 0.0003 * max_cs]
            p_opt, sc_r = _adam_fine(p_opt, time.monotonic() + 50.0, betas_r,
                                     lr_init=0.004, max_iter=30000, lr_schedule='cosine')
            sc_r_full = _score(p_opt)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = p_opt.copy()
                a_work = p_opt.copy()
                a_work = a_work / a_work.sum()
                print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 7: Ultra-fine final polish ===
    print("Phase 7: Ultra-fine polish", flush=True)
    phase7_deadline = deadline
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas_ult = [0.00002 * max_cw, 0.00001 * max_cw, 0.000005 * max_cw]

    time_left = phase7_deadline - time.monotonic() - 15
    if time_left > 20:
        a_pol, sc_pol = _adam_fine(a_work, time.monotonic() + time_left, betas_ult,
                                   lr_init=0.001, max_iter=200000, lr_schedule='cosine')
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

        # One more peak shave at the very end
        if time.monotonic() < deadline - 30:
            a_work = best_seq.copy()
            a_work = a_work / a_work.sum()
            a_final_ps, sc_fps = _peak_shave(a_work, deadline=deadline - 20, n_steps=5000, top_k=100)
            sc_fps_full = _score(a_final_ps)
            if sc_fps_full < best_score:
                best_score = sc_fps_full
                best_seq = a_final_ps.copy()
                print(f"  Final peak shave: score={sc_fps_full:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            s = final.sum()
            final = final / s

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"Eval error: {ex}", flush=True)
        return [float(x) for x in final]

    fallback = np.ones(512) / 512
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
