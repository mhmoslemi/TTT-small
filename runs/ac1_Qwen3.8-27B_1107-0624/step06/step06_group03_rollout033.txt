    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _solve_lp(f, n):
    """Solve the LP: min t s.t. conv(f,g)[k]<=t, sum(g)=1, g>=0"""
    # Build constraint matrix
    # Row k: sum_i A[k,i]*g[i] - t <= 0
    # A[k,i] = f[k-i] if 0 <= k-i < n else 0
    # plus A[k,n] = -1 for the t variable
    
    # Use sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n, k+1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(f[k-i])
        # t column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
    b_ub = np.zeros(2*n-1)

    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    bounds = [(0, None)] * n + [(None, None)]

                     bounds=bounds, method='highs')
    if result.success:
        return result.x[:n]
    else:

    """Alternating LP optimization."""
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        g = _solve_lp(f, len(f))
        g = np.maximum(g, 0)
        s_g = g.sum()
        if s_g > 1e-15:

            best_f = f_new.copy()
        
        f = f_new
    
    return best_f, best_score

"""
LP-initialized gradient optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based alternating method for initialization with Adam/FISTA polishing
and aggressive coordinate descent for final refinement.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 50 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.96)
        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(99999)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                a[i] = 1e-5
                a = a / a.sum()
                continue
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old
            idx_top = np.argsort(c)[-top_k_w:]
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5, a_i_old * 2.0,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2, a_i_old * 0.1]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 10.0 * max(a_i_old, 0.01):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.8))
                        t_candidates.append(t_star * 1.2)
            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]
            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4, 5]:
            a += 0.2 / freq * np.cos(2.0 * np.pi * freq * np.arange(n) / n + freq * 0.7)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(777)
        a = rng.random(n)
        kern = np.exp(-0.5 * (np.arange(-10, 11) / 3.0) ** 2)
        a = np.convolve(a, kern, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lp_alternate_init(f_init, deadline, n_lp_iters=200, t_param=0.3):
    """LP-based alternating method for initialization."""
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    best_f = f.copy()
    best_score = _score_norm(f)

    # Precompute LP structure
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows_list.append(k)
            cols_list.append(i)
        rows_list.append(k)
        cols_list.append(n)

    rows_arr = np.array(rows_list, dtype=np.int32)
    cols_arr = np.array(cols_list, dtype=np.int32)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, None)] * n + [(None, None)]
    b_ub = np.zeros(2 * n - 1)

    for it in range(n_lp_iters):
        if time.monotonic() > deadline:
            break

        # Build values for LP
        vals = np.empty(len(rows_list))
        idx = 0
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                vals[idx] = f[k - i]
                idx += 1
            vals[idx] = -1.0
            idx += 1

        A_ub = sp.csr_matrix((vals, (rows_arr, cols_arr)), shape=(2 * n - 1, n + 1))

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                            bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            break

        g = result.x[:n].copy()
        g = np.maximum(g, 0)
        s_g = g.sum()
        if s_g < 1e-15:
            break
        g = g / s_g

        # Update
        t_eff = t_param
        f_new = (1.0 - t_eff) * f + t_eff * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new

        sc = _score_norm(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()

        f = f_new

        # Reduce step size over time
        if it % 50 == 0:
            t_param = max(0.05, t_param * 0.7)

    sc_final = _score_norm(f)
    if sc_final < best_score:
        best_score = sc_final
        best_f = f.copy()
    return best_f, best_score
