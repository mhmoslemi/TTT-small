"""
Hybrid optimizer combining exact coordinate descent, Adam in z-space with
adaptive temperature annealing, and multi-scale exploration to minimize
2*n*max(conv(a,a))/sum(a)^2. Key innovation: O(n) coordinate updates using
incremental convolution tracking (no per-coordinate FFT).
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _coordinate_descent_sweep(a, b, S, deadline, ratios=None, verbose=False):
    """
    One full coordinate descent sweep using incremental convolution tracking.
    a: current sequence (modified in place)
    b: current convolution (modified in place)
    S: current sum
    Returns updated a, b, S and the number of coordinates updated.
    """
    n = len(a)
    if ratios is None:
        ratios = np.array([0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0,
                           1.15, 1.3, 1.5, 2.0, 3.0, 5.0, 10.0])
    n_r = 2 * n - 1
    n_updates = 0
    best_local = np.inf

    for i in range(n):
        if time.monotonic() > deadline:
            break
        ai = a[i]
        S0 = S - ai

        # Build shift vector: shift_a[k] = a[k-i] if 0<=k-i<n else 0
        # Valid range for k: [i, i+n-1]
        k_lo = i
        k_hi = i + n - 1

        # Compute d vector (coefficient of (x-ai) in b_k change, for k != 2i)
        # d[k] = 2*a[k-i] for valid k, d[2i] = 0 (handled separately)
        # We only need d for k in [0, n_r-1]
        # Instead of allocating full d each time, compute on the fly
        # For evaluating b_k(x): b_k + (x-ai)*2*a_{k-i} for k != 2i
        #                       b_{2i} + x^2 - ai^2 for k = 2i

        # Precompute base = b (current convolution)
        # For candidate x: vals[k] = b[k] + (x-ai)*2*a_{k-i} (k in [i, i+n-1], k != 2i)
        #                   vals[2i] = b[2i] + x^2 - ai^2
        #                   vals[k] = b[k] (elsewhere)

        # To find max efficiently, note that only O(n) positions change.
        # max of b is known. The new max is either the old max (if it's outside
        # the changed region) or in the changed region.
        # For simplicity, just compute the full max over the changed region.

        # Extract the relevant portion of a for the shift
        # a_shifted[k] = a[k-i] for k in [i, i+n-1]
        a_seg = a  # we use a directly since a[k-i] for k in [i,i+n-1] is a[0..n-1]

        # Build the "affected" indices
        aff_start = i
        aff_end = i + n  # exclusive

        # For each candidate x, compute:
        #   max(b[aff_start:aff_end] + (x-ai)*2*a_seg_adjusted, b[2i] + x^2 - ai^2, max(b[unaffected]))
        # where a_seg_adjusted accounts for the fact that a[i] itself might be in the segment

        # Actually, let me just do it simply:
        # vals_changed = b[i:i+n] + (x-ai)*2*a  (since b[i+k] += (x-ai)*2*a[k] for k=0..n-1)
        # But we need to fix position 2i within this range:
        #   if 0 <= 2i-i < n, i.e., 0 <= i < n (always true), then position 2i-i = i in the segment
        #   At that position: instead of + (x-ai)*2*a[i], we have x^2 - ai^2
        #   So: vals[i] should be b[2i] + x^2 - ai^2, not b[2i] + (x-ai)*2*ai

        # The max of unaffected positions:
        # Compute max of b outside [i, i+n-1]
        # Positions 0..i-1 and i+n..2n-2
        if i > 0:
            max_before = float(b[:i].max()) if i > 0 else -np.inf
        else:
            max_before = -np.inf
        if i + n < n_r:
            max_after = float(b[i+n:].max())
        else:
            max_after = -np.inf
        max_unaffected = max(max_before, max_after)

        # Now for candidates:
        # max_val(x) = max(max(b[i:i+n] + (x-ai)*2*a, with correction at pos i),
        #               b[2i] + x^2 - ai^2 if 2i in [0, n_r-1],
        #               max_unaffected)
        # Note: position 2i in the segment [i, i+n-1] is at offset 2i-i = i.
        # So b[i:i+n][i] = b[2i]. The correction: instead of b[2i] + (x-ai)*2*a[i],
        # we use b[2i] + x^2 - ai^2.

        b_seg = b[i:i+n]  # length n
        # b_seg[k] corresponds to b[i+k] = b at position i+k
        # The "expected" new value: b_seg[k] + (x-ai)*2*a[k]
        # Correction at k=i: use b_seg[i] + x^2 - ai^2 instead of b_seg[i] + (x-ai)*2*a[i]

        best_x = ai
        best_score = 2.0 * n * float(b.max()) / (S * S)

        candidates = ai * ratios
        candidates = np.unique(np.concatenate([candidates, [0.0, ai]]))
        candidates = candidates[(candidates >= 0) & (candidates < 1000.0)]

        for x in candidates:
            dx = x - ai
            if abs(dx) < 1e-15:
                max_b = float(b.max())
            else:
                # Compute max in affected region
                # b_seg + dx*2*a, then correct position i
                affected = b_seg + dx * 2.0 * a
                # Correct position i (which is b[2i]):
                affected[i] = b_seg[i] + x * x - ai * ai
                max_affected = float(affected.max())
                max_b = max(max_affected, max_unaffected,
                            b[2 * i] + x * x - ai * ai if 2 * i < n_r else -np.inf)

            S_new = S0 + x
            if S_new < 0.01:
                continue
            score = 2.0 * n * max_b / (S_new * S_new)
            if score < best_score - 1e-15:
                best_score = score
                best_x = x

        if best_x != ai:
            # Update a and b
            dx = best_x - ai
            a[i] = best_x
            # Update b: b[i:i+n] += dx * 2*a, then correct b[2i]
            b[i:i+n] += dx * 2.0 * a
            if 2 * i < n_r:
                # b[2i] was updated by dx*2*a[i] (old ai), but should be x^2 - ai^2
                # So: b[2i] -= dx*2*ai + (x^2 - ai^2) ... wait
                # Current b[2i] = old_b[2i] + dx*2*a[i] = old_b[2i] + dx*2*ai (since a[i] was ai at the time of update)
                # But we want: new_b[2i] = old_b[2i] + x^2 - ai^2
                # So: b[2i] should be old_b[2i] + x^2 - ai^2
                # Currently it's old_b[2i] + dx*2*ai = old_b[2i] + (x-ai)*2*ai
                # Difference: (x^2 - ai^2) - (x-ai)*2*ai = (x-ai)(x+ai) - 2*ai*(x-ai) = (x-ai)(x+ai-2*ai) = (x-ai)(x-ai) = dx^2
                # So: b[2i] += dx^2
                b[2 * i] += dx * dx
            S = S + dx
            n_updates += 1

    return a, b, S, n_updates


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, use_active_set=False, active_k=5,
                  eval_interval=300, clip_norm=0.8):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if use_active_set and T < T_start * 0.01:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_soft_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.1:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_step(a, deadline):
    """LP-based step: maximize sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        time_left = deadline - time.monotonic()
        tl = min(15.0, max(2.0, time_left * 0.3))
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': tl})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            if n_upd == 0:
                break
        print(f"Phase 4 done: {cd_sweep} sweeps, best={best_score:.10f}")

    # ============ Phase 5: Adam z-space, active gradient (1100s - 1400s) ============
    phase5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase5_end:
        print(f"\n=== Phase 5: Adam z-space active (until t={phase5_end - t_start:.0f}s) ===")
        a = best_a.copy()
        c_best = _fft_convolve(a, a)
        max_c = float(c_best.max())
        a_active, sc_active = _adam_z_space(a, phase5_end - 5,
                                             T_start=0.001 * max_c, T_end=0.000005 * max_c,
                                             lr_start=0.02, lr_end=0.001,
                                             max_iter=500000,
                                             use_active_set=True, active_k=3,
                                             eval_interval=150)
        full_sc = _score(a_active)
        print(f"  Adam-active: score={full_sc:.10f}")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_active.copy()

    # ============ Phase 6: Multi-scale exploration (1400s - 1650s) ============
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end:
        print(f"\n=== Phase 6: Multi-scale (until t={phase6_end - t_start:.0f}s) ===")
        n_best = len(best_a)

        # Try n=8192
        n_large = 8192
        if n_best != n_large:
            a_large = _interp_to_length(best_a, n_large)
            c_large = _fft_convolve(a_large, a_large)
            max_cl = float(c_large.max())
            sub_deadline = min(phase6_end - 5, t_start + 1550.0)
            a_large_opt, _ = _adam_z_space(a_large, sub_deadline,
                                            T_start=0.003 * max_cl, T_end=0.00001 * max_cl,
                                            lr_start=0.03, lr_end=0.002,
                                            max_iter=150000,
                                            use_active_set=True, active_k=5,
                                            eval_interval=200)
            full_sc = _score(a_large_opt)
            print(f"  n=8192 Adam: score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_large_opt.copy()

            # Quick CD on the large solution
            if time.monotonic() < phase6_end - 30:
                a_cd = a_large_opt.copy()
                b_cd = _fft_convolve(a_cd, a_cd)
                S_cd = float(a_cd.sum())
                a_cd, b_cd, S_cd, _ = _coordinate_descent_sweep(a_cd, b_cd, S_cd, phase6_end - 5)
                sc_cd = _score(a_cd)
                print(f"  n=8192 CD: score={sc_cd:.10f}")
                if np.isfinite(sc_cd) and sc_cd < best_score:
                    best_score = sc_cd
                    best_a = a_cd.copy()

        # Try n=2048
        n_small = 2048
        if n_best != n_small and n_best != 2048:
            a_small = _interp_to_length(best_a, n_small)
            c_small = _fft_convolve(a_small, a_small)
            max_cs = float(c_small.max())
            sub_deadline = min(phase6_end - 5, t_start + 1600.0)
            a_small_opt, _ = _adam_z_space(a_small, sub_deadline,
                                            T_start=0.002 * max_cs, T_end=0.00001 * max_cs,
                                            lr_start=0.03, lr_end=0.002,
                                            max_iter=100000,
                                            use_active_set=True, active_k=3,
                                            eval_interval=150)
            full_sc = _score(a_small_opt)
            print(f"  n=2048 Adam: score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_small_opt.copy()

    # ============ Phase 7: Final polish - interleaved CD + Adam (1650s - deadline) ============
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 7: Final Polish (until deadline) ===")
        a = best_a.copy()
        n_fin = len(a)

        # Short CD pass
        b_fin = _fft_convolve(a, a)
        S_fin = float(a.sum())
        cd_deadline = min(deadline - 30, time.monotonic() + 60)
        a, b_fin, S_fin, _ = _coordinate_descent_sweep(a, b_fin, S_fin, cd_deadline)
        sc_cd = _score(a)
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_score = sc_cd
            best_a = a.copy()
        print(f"  Final CD: score={sc_cd:.10f}")

        # Very fine Adam polish
        if time.monotonic() < deadline - 10:
            a = best_a.copy()
            c_fin = _fft_convolve(a, a)
            max_cf = float(c_fin.max())
            a_final, _ = _adam_z_space(a, deadline - 5,
                                        T_start=0.00005 * max_cf, T_end=0.000001 * max_cf,
                                        lr_start=0.003, lr_end=0.0002,
                                        max_iter=200000,
                                        use_active_set=True, active_k=3,
                                        eval_interval=100)
            full_sc = _score(a_final)
            print(f"  Final Adam: score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_final.copy()

        # One more CD at the very end
        if time.monotonic() < deadline - 5:
            a = best_a.copy()
            b = _fft_convolve(a, a)
            S = float(a.sum())
            a, b, S, _ = _coordinate_descent_sweep(a, b, S, deadline - 3)
            sc = _score(a)
            print(f"  Last CD: score={sc:.10f}")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    # --- Final validation and return ---
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
