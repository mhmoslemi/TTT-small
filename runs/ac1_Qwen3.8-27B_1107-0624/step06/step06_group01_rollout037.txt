"""
Optimized step function search using gradient-based methods in log-space.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences a.
Uses Adam optimizer with adaptive softmax-smoothed max, curriculum learning,
and multi-phase local search.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Compute the full score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, c, T, eps=1e-15):
    """
    Compute gradient of smoothed max with respect to z (log-space).
    F_T = T * log(sum exp(c_k / T))
    dF/dz_m = 2 * a_m * [(w * a)_m - (w · c)]
    where w = softmax(c/T)
    """
    n = len(a)
    c_max = np.max(c)
    w = np.exp((c - c_max) / T)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fall back to one-hot at argmax
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    
    # (w ⊛ a)_m = sum_j w[m+j] * a[j] for m = 0, ..., n-1
    # This is np.correlate(w, a, mode='valid')
    cross = np.correlate(w, a, mode='valid')
    # cross[m] = sum_j w[m+j] * a[j]
    
    wc = float(np.dot(w, c))
    grad_z = 2.0 * a * (cross - wc)
    return grad_z, w

def _adam_step(z, grad, m_buf, v_buf, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam optimizer step in log-space."""
    m_buf = b1 * m_buf + (1 - b1) * grad
    v_buf = b2 * v_buf + (1 - b2) * grad * grad
    mh = m_buf / (1 - b1**t)
    vh = v_buf / (1 - b2**t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    # Center z for numerical stability
    z_new = z_new - np.mean(z_new)
    # Clip to avoid overflow
    z_new = np.clip(z_new, -50.0, 50.0)
    return z_new, m_buf, v_buf

def _optimize_adam(a_init, deadline, T_schedule, lr_init, lr_decay, max_iter):

    Main Adam optimization loop in log-space.
    a_init: initial normalized sequence (sum=1)
    T_schedule: list of temperature values (decreasing)
    """

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_score = _score_norm(a_init)

            grad, w = _compute_gradient(a, c, T)

                sc = _score_norm(a_chk)

    sc_final = _score_norm(a_final)

def _spectral_init(n, rng):
    """
    Initialize using spectral method: create a sequence with approximately flat spectrum.
    """
    N = _next_pow2(2 * n)
    # Create random phases
    phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    # Magnitude: try to make |FFT(a)|^2 approximately constant
    mag = np.ones(N // 2 + 1)
    spectrum = mag * np.exp(1j * phases)
    # Inverse FFT
    a = np.fft.irfft(spectrum, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    else:
        a = np.ones(n) / n
    return a

    a = np.exp(-np.linspace(0, 3, n))

"""
Optimized step function search using gradient-based methods in log-space with adaptive
softmax-smoothed max, curriculum learning, exact subgradient switching, and multi-phase
local search. Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_and_weights(a, c, T):
    n = len(a)
    c_max = np.max(c)
    shifted = (c - c_max) / T
    w = np.exp(shifted)
    w_sum = float(w.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    cross = np.correlate(w, a, mode='valid')
    wc = float(np.dot(w, c))
    grad_z = 2.0 * a * (cross - wc)
    return grad_z, w


def _exact_subgrad(a, c):
    n = len(a)
    top_val = c.max()
    active = (c >= top_val - 1e-12)
    if active.sum() == 0:
        active = np.zeros_like(c)
        active[int(np.argmax(c))] = 1.0
    w = active / active.sum()
    cross = np.correlate(w, a, mode='valid')
    wc = float(np.dot(w, c))
    grad_z = 2.0 * a * (cross - wc)
    return grad_z


def _adam_optimize(a_init, deadline, T_schedule, lr_init, lr_decay_per_phase,
                   iters_per_phase, momentum_reset_every=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    lr = lr_init

    b1, b2, eps = 0.9, 0.999, 1e-8

    for T in T_schedule:
        if time.monotonic() > deadline:
            break

        phase_iters = iters_per_phase
        for it in range(phase_iters):
            if time.monotonic() > deadline:
                break
            t += 1

            if t % momentum_reset_every == 0:
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

            a = _softmax(z)
            c = _full_convolve(a, a)

            if T < 1e-10:
                grad = _exact_subgrad(a, c)
            else:
                grad, _ = _compute_grad_and_weights(a, c, T)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_step = 2.0
            if update_norm > max_step:
                update = update * (max_step / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        lr = max(1e-5, lr * lr_decay_per_phase)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _local_search(a_init, deadline, n_trials=10000, rng=None):
    """Coordinate-wise and multi-coordinate local search."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        n_coords = rng.choice([1, 1, 1, 1, 2, 2, 3])
        idx = rng.choice(n, size=n_coords, replace=False)

        for attempt in range(4):
            a_try = best_a.copy()
            for j in idx:
                if a_try[j] > 1e-12:
                    factor = np.exp(rng.uniform(-0.05, 0.05))
                    a_try[j] *= factor
                else:
                    a_try[j] = rng.uniform(0, 0.0005)

            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try /= s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    break

    return best_a, best_score
