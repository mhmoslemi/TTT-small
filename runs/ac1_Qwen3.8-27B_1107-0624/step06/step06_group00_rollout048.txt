"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: multi-temperature subgradient, backtracking line search,
Hessian-aware step correction, diverse structured initializations,
and adaptive switching between z-space and a-space optimization.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient_from_p(a, p):
    """Compute g[j] = sum_k p[k]*a[k-j] via FFT cross-correlation."""
    n = len(a)
    h_flip = p[::-1]
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    H = np.fft.rfft(h_flip, N)
    conv = np.fft.irfft(A * H, N)[:3 * n - 2]
    g = conv[n - 1:2 * n - 1][::-1]
    return g[:n]


def _soft_max_weights(c, T):
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    return p


def _multi_temp_gradient(a, c, T1, T2, w1=0.5, w2=0.5):
    """Multi-temperature subgradient for better approximation of true subgradient."""
    p1 = _soft_max_weights(c, T1)
    p2 = _soft_max_weights(c, T2)
    p = w1 * p1 + w2 * p2
    g = _subgradient_from_p(a, p)
    return g


def _active_gradient(a, c, top_k=5):
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    g = _subgradient_from_p(a, p)
    return g


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, use_active=False, active_k=5,
                   eval_interval=200, clip_norm=0.5, mult_temp=False,
                   backtracking=True):
    """Adam in z-space with adaptive features."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_active and T < T_start * 0.002:
            g = _active_gradient(a, c, active_k)
        elif mult_temp:
            T_low = T * 0.1
            g = _multi_temp_gradient(a, c, T, T_low, 0.3, 0.7)
        else:
            p = _soft_max_weights(c, T)
            g = _subgradient_from_p(a, p)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if backtracking and t % 10 == 0:
            a_try = _softmax_stable(z_new)
            sc_try = _score_norm(a_try)
            if sc_try > best_score * 1.01 and t > 100:
                # Step was too large, shrink
                lr_mult = max(0.1, lr_mult * 0.5)
                z_new = z - 0.1 * update
                z_new = np.clip(z_new, -60.0, 60.0)
                z_new = z_new - z_new.mean()

        z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                if lr_mult < 1.0:
                    lr_mult = min(1.0, lr_mult * 1.3)
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fista_optimize(a_init, deadline, T_start, T_end, max_iter=100000):
    """FISTA with simplex projection for a-space optimization."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    L = 4.0 * n
    step = 1.0 / L
    eval_interval = 500

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        c = _fft_convolve(y, y)
        p = _soft_max_weights(c, T)
        g = _subgradient_from_p(y, p)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            step = step * min(2.0, max(0.5, 1.0 / (step * g_norm + 1e-10)))
        step = min(step, 2.0 / L)
        step = max(step, 1e-12)
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_structured_init(n, pattern_idx, rng):
    """Create diverse structured initializations."""
    a = np.ones(n) / n
    if pattern_idx == 0:
        # Uniform
        pass
    elif pattern_idx == 1:
        # Cosine bump

        a = np.cos(np.pi * j / n) ** 2 + 0.1
        a = a / a.sum()
    elif pattern_idx == 2:

        a = np.zeros(n)

        a += 0.05
        a = a / a.sum()

    elif pattern_idx == 4:

        decay = 0.999 + rng.random() * 0.001
        a = decay ** j
        a = a / a.sum()
    elif pattern_idx == 5:

        a = np.ones(n)
        for _ in range(3):

        a = a / a.sum()
    else:

        a = rng.random(n) + 0.1
        a = a / a.sum()
