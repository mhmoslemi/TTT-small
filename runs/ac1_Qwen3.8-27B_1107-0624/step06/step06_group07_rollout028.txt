def _make_raised_cosine(n, bandwidth=0.3):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5) / bandwidth))

def _peak_shaving(a_init, deadline, top_k=5, max_rounds=500):
    """Directly shave the top convolution peaks using per-peak gradients."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.002

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        top_indices = np.argsort(c)[-top_k:]

        improved = False
        for pk in top_indices:
            if time.monotonic() > deadline:
                break
            grad = _gradient_single_peak(a, c, n, int(pk))
            grad = grad - grad.mean()
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                continue

            # Try a few step sizes
            for s_mult in [1.0, 0.5, 0.25, 0.1]:
                a_try = a - step * s_mult * grad / gn
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_a = a_try.copy()
                    best_sc = sc_try
                    a = a_try.copy()
                    c = _full_convolve(a, a)
                    improved = True
                    break
                elif sc_try < _score(a):
                    a = a_try.copy()

        if not improved:
            step *= 0.8
            if step < 1e-8:
                break

    return best_a, best_sc

            break

        c = _full_convolve(a, a)
        grad = _gradient_quantile(a, c, n, quantile)
        grad = grad - grad.mean()
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        # Armijo line search
        accepted = False
        s = step
        for backtrack in range(15):
            a_try = a - s * grad / gn
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                s *= 0.5
                continue
            a_try = a_try / st
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-15:
                best_sc = sc_try
                best_a = a_try.copy()

            if sc_try < _score(a) - 1e-16 * s * gn:
                a = a_try.copy()
                accepted = True
                step = min(s * 1.1, 0.05)
                break
            else:
                s *= 0.5

        if not accepted:
            step *= 0.7
            if step < 1e-12:
                break

        step *= (1.0 - 5e-6)

    return best_a, best_sc


def _perturb_structured(a, rng, mode="fourier"):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    if mode == "fourier_low":
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 16)))
            amp = 0.008 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            z = z + amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif mode == "fourier_mid":
        for _ in range(5):
            freq = int(rng.integers(n // 16, max(n // 16 + 1, n // 4)))
            amp = 0.004 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            z = z + amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif mode == "local":
        for _ in range(8):
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 32, max(n // 16 + 1, n // 8)))
            amp = rng.uniform(-0.01, 0.01)
            t = np.arange(n)
            z = z + amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    elif mode == "mirror":
        # Perturb first half, mirror to second half (preserves symmetry)
        half = n // 2
        pert = 0.005 * rng.standard_normal(half)
        z[:half] += pert
        z[half:] += pert[::-1] if n % 2 == 0 else pert[-1::-1]
    elif mode == "gauss":
        z = z + 0.01 * rng.standard_normal(n)
    else:
        z = z + 0.005 * rng.standard_normal(n)

    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _fast_optimize(a_start, deadline, min_remaining=30.0):
    """Quick optimization: short L-BFGS + subgradient + peak shaving."""
    n = len(a_start)
    c = _full_convolve(a_start, a_start)
    mc = float(c.max())

    # Short L-BFGS
    if time.monotonic() + min_remaining < deadline:
        betas = mc * np.array([0.02, 0.005, 0.001, 0.0002])
        current = a_start.copy()
        for beta in betas:
            if time.monotonic() + min_remaining > deadline:
                break
            remaining = deadline - min_remaining - time.monotonic()
            max_iter = min(3000, max(100, int(remaining / 0.005)))
            current, _ = _lbfgs_run(current, beta, n, max_iter)

    # Subgradient
    if time.monotonic() + min_remaining < deadline:
        current, _ = _subgrad_quantile(current, deadline - min_remaining,
                                       quantile=0.999, max_steps=5000, lr_init=0.02)

    # Peak shaving
    if time.monotonic() + min_remaining < deadline:
        current, _ = _peak_shaving(current, deadline - min_remaining, top_k=3, max_rounds=200)

    return current


def propose_candidate(seed=42, budget_s=1800):
    """
    Ultra-fine optimizer combining adaptive L-BFGS, quantile subgradient,
    peak shaving, structured restarts, and length sweeping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"delta={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n0 = 4096
        a0 = _make_uniform(n0)
        print(f"No parent. Uniform n={n0}, score={_score(a0):.10f}")
        try_update(a0, "init")

    n_parent = len(best_seq)

    # === Phase 1: Try diverse initial structures at parent length ===
    print(f"\n=== Phase 1: Diverse initializations (t={time.monotonic()-t_start:.1f}s) ===")
    init_candidates = {
        "raised_cos_03": _make_raised_cosine(n_parent, 0.3),
        "raised_cos_05": _make_raised_cosine(n_parent, 0.5),
        "raised_cos_07": _make_raised_cosine(n_parent, 0.7),
        "gauss_03": _make_gaussian(n_parent, 0.3),
        "gauss_04": _make_gaussian(n_parent, 0.4),
        "bump_2": _make_bump(n_parent, 2.0),
        "bump_4": _make_bump(n_parent, 4.0),
        "cos_mod_1_05": _make_cosine_mod(n_parent, 1, 0.05),
        "cos_mod_2_03": _make_cosine_mod(n_parent, 2, 0.03),
    }

    for name, init_a in init_candidates.items():
        if time.monotonic() > deadline - 600:
            break
        sc = _score(init_a)
        print(f"  Init [{name}]: score={sc:.10f}")
        if sc < best_score_val:
            try_update(init_a, f"init_{name}")

    # === Phase 2: Main adaptive L-BFGS ===
    print(f"\n=== Phase 2: Adaptive L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    a_main = _adaptive_lbfgs(best_seq, deadline, min_remaining=500.0)
    try_update(a_main, "phase2_adaptive_lbfgs")

    # === Phase 3: Quantile subgradient polishing ===
    print(f"\n=== Phase 3: Quantile subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_sg, sc_sg = _subgrad_quantile(best_seq, deadline - 350,
                                        quantile=0.9995, max_steps=15000, lr_init=0.015)
        try_update(a_sg, "phase3_quantile_sg")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak shaving (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_ps, sc_ps = _peak_shaving(best_seq, deadline - 250, top_k=5, max_rounds=800)
        try_update(a_ps, "phase4_peak_shave")

    # === Phase 5: Direct subgradient with Armijo ===
    print(f"\n=== Phase 5: Direct subgradient Armijo (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_ds, sc_ds = _direct_subgrad_armijo(best_seq, deadline - 200,
                                             quantile=0.999, max_steps=5000)
        try_update(a_ds, "phase5_direct_armijo")

    # === Phase 6: Structured restarts ===
    print(f"\n=== Phase 6: Structured restarts (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        restart_modes = ["fourier_low", "fourier_mid", "local", "mirror", "gauss"]
        for mode in restart_modes:
            if time.monotonic() > deadline - 100:
                break
            p_start = _perturb_structured(best_seq, rng, mode)
            sc_start = _score(p_start)
            print(f"  Restart [{mode}]: start={sc_start:.10f}")

            a_r = _fast_optimize(p_start, deadline, min_remaining=50.0)
            try_update(a_r, f"restart_{mode}")

    # === Phase 7: Length sweeping ===
    print(f"\n=== Phase 7: Length sweeping (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        current_n = len(best_seq)
        target_lengths = []
        for tn in [1024, 2048, 3072, 4096, 5120, 6144, 8192]:
            if tn != current_n and tn >= 512:
                target_lengths.append(tn)

        for tn in target_lengths:
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 30:
                break
            print(f"  Trying n={tn} (remaining={remaining:.0f}s)...")
            a_interp = _interpolate_to_n(best_seq, tn)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            if sc_interp < best_score_val + 0.05:
                a_opt = _fast_optimize(a_interp, deadline, min_remaining=30.0)
                try_update(a_opt, f"len_{tn}")

    # === Phase 8: Final ultra-fine polish ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        for rnd in range(5):
            if time.monotonic() > deadline - 10:
                break

            # Ultra-tight L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-8
            remaining = deadline - 10 - time.monotonic()
            if remaining > 20:
                max_iter = min(5000, max(200, int(remaining / 0.003)))
                a_f, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight peak shaving
            if time.monotonic() < deadline - 10:
                a_ps, _ = _peak_shaving(best_seq, deadline - 10, top_k=3, max_rounds=300)
                try_update(a_ps, f"final_ps_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 10:
                a_sg, _ = _subgrad_quantile(best_seq, deadline - 10,
                                            quantile=0.9999, max_steps=5000, lr_init=0.008)
                try_update(a_sg, f"final_sg_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")

        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
