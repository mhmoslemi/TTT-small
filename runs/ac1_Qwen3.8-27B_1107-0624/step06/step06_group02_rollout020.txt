"""
Optimized step function search using log-space FISTA with aggressive beta continuation,
Adam polishing, multi-restart, and coordinate descent. Targets sub-1.5027 score.
Key improvements: exact gradient exploitation when max is unique, adaptive step sizes,
better initialization, and n-size exploration.

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _convolve(a, b):
    n_out = len(a) + len(b) - 1
    N = _next_pow2(n_out)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n_out]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed(a, w):
    """Gradient of smoothed max w.r.t. a (on simplex)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    return 2.0 * wc[n - 1: 2 * n - 1]


def _grad_exact(a, k_star):
    """Exact gradient of c[k_star] w.r.t. a. O(n) - no FFT needed."""
    n = len(a)
    g = np.zeros(n)
    # dc[k*]/da[m] = 2*a[k*-m] for 0 <= k*-m < n
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for m in range(lo, hi + 1):
        g[m] = 2.0 * a[k_star - m]
    return g


def _fista_logspace(z_init, beta, deadline, max_iter, lr_scale=1.0):
    """FISTA in log-space with smoothed max objective."""
    n = len(z_init)
    z_curr = z_init.copy()
    z_prev = z_curr.copy()
    t_val = 1.0
    
    # Estimate Lipschitz constant via backtracking
    L = 4.0 * n * lr_scale
    step = 1.0 / L
    
    best_z = z_curr.copy()
    a_b = _softmax(z_curr)
    best_sc = _score_norm(a_b)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = z_curr + ((t_val - 1.0) / t_next) * (z_curr - z_prev)
        
        a_y = _softmax(y)
        c_y = _convolve(a_y, a_y)
        cm = float(c_y.max())
        
        if beta < 1e-10 * cm:
            # Use exact gradient (effectively beta -> 0)
            k_star = int(np.argmax(c_y))
            G = _grad_exact(a_y, k_star)
        else:
            shifted = (c_y - cm) / beta
            np.clip(shifted, -500, 500, out=shifted)
            e = np.exp(shifted)
            ws = float(e.sum())
            if ws < 1e-300:
                w = np.zeros_like(c_y)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = e / ws
            G = _grad_smoothed(a_y, w)
        
        # Log-space gradient: dL/dz[i] = a[i]*(G[i] - dot(a,G))
        W = float(np.dot(a_y, G))
        grad_z = a_y * (G - W)
        
        # Backtracking line search (simplified: just use fixed step with occasional adjustment)
        z_next = y - step * grad_z
        z_next = np.clip(z_next, -50.0, 50.0)
        
        # Periodic score check
        if it % 50 == 0:
            a_chk = _softmax(z_next)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_z = z_next.copy()
        
        z_prev = z_curr
        z_curr = z_next
        t_val = t_next
    
    a_final = _softmax(z_curr)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_z = z_curr.copy()
    
    return best_z, _softmax(best_z), best_sc


def _adam_logspace(z_init, deadline, betas, lr_init, max_iter_total):
    """Adam optimizer in log-space with beta continuation."""
    n = len(z_init)
    z = z_init.copy()
    best_z = z.copy()
    a_b = _softmax(z)
    best_sc = _score_norm(a_b)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            
            if beta < 1e-10 * cm:
                k_star = int(np.argmax(c))
                G = _grad_exact(a, k_star)
            else:
                shifted = (c - cm) / beta
                np.clip(shifted, -500, 500, out=shifted)
                e = np.exp(shifted)
                ws = float(e.sum())
                if ws < 1e-300:
                    w = np.zeros_like(c)
                    w[k_star] = 1.0
                else:
                    w = e / ws
                G = _grad_smoothed(a, w)
            
            W = float(np.dot(a, G))
            grad_z = a * (G - W)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 30 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_z = z.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_z = z.copy()
    
    return best_z, _softmax(best_z), best_sc


def _coord_descent(a_init, deadline, max_rounds=300):
    """Coordinate descent in log-space for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    rng = np.random.default_rng(99)
    
    stall = 0
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        if n > 4096:
            indices = rng.choice(n, size=min(2048, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        
        improved = False
        a = _softmax(z)
        
        for j in indices:
            if time.monotonic() > deadline:
                    break
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                z[j] += delta
                a_try = _softmax(z)
                sc = _score_norm(a_try)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_try.copy()
                    improved = True
                    a = a_try
                    break
                else:
                    z[j] -= delta
        
        if not improved:
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0
    
    return best_a, best_sc


def _interp_to_size(a, new_n):
    """Interpolate sequence to new size, preserving the profile shape."""
    n0 = len(a)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_n)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _block_shift(a, rng):
    """Random block circular shift for diverse restarts."""
    n = len(a)
    block = max(1, n // 8)
    shift = rng.integers(1, block)
    a_new = a.copy()
    # Shift first half
    half = n // 2
    a_new[:half] = np.roll(a[:half], shift)
    a_new[half:] = np.roll(a[half:], -shift)
    return a_new / a_new.sum()

            if time.monotonic() < phase4_end - 15:
                c_bo = _convolve(a_big, a_big)
                max_cbo = float(c_bo.max())
                t_sub = phase4_end - time.monotonic() - 5
                if t_sub > 15:
                    z_big, a_big, sc_big = _adam_logspace(
                        z_big, time.monotonic() + t_sub,
                        [0.0005 * max_cbo, 0.0002 * max_cbo,
                         0.0001 * max_cbo, 0.00003 * max_cbo, 0.00001 * max_cbo],
                        lr_init=0.004, max_iter_total=200000
                    )
                    sc_full = _score(a_big)
                    if sc_full < best_score:
                        best_score = sc_full
                        best_seq = a_big.copy()
                        print(f"  Optimized n=8192: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # Also try n=2048 (sometimes smaller is better due to optimization)
    if time.monotonic() < phase4_end - 20 and best_seq is not None and len(best_seq) > 2048:
        n_small = 2048
        a_small = _interp_to_size(best_seq, n_small)
        sc_small = _score(a_small)
        if sc_small < best_score:
            best_score = sc_small
            best_seq = a_small.copy()
            print(f"  Smaller n=2048: score={sc_small:.10f}")
    
    # === Phase 5: Final ultra-fine Adam polish (1500 - 1700s) ===
    print("Phase 5: Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        
        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        
        z_pol = np.log(np.maximum(a_pol, 1e-15))
        z_pol = z_pol - z_pol.mean()
        
        betas_ult = [0.00001 * max_cp, 0.000003 * max_cp, 0.000001 * max_cp,
                     0.0000003 * max_cp, 1e-8 * max_cp]
        
        per_t = min(200.0, deadline - time.monotonic() - 50)
        if per_t > 20:
            z_pol, a_pol, sc_pol = _adam_logspace(
                z_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.0015, max_iter_total=1000000
            )
            sc_pol_full = _score(a_pol)
            print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
    
    # === Phase 6: Coordinate descent (last 100s) ===
    print("Phase 6: Coordinate descent")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = deadline - 5
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=500)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
    
    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 1.0
        # Scale to ensure sum >= 0.01
        if s < 1.0:
            final = final * (1.0 / s)
            s = 1.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
