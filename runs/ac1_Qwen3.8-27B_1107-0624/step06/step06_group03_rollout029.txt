def _next_pow2(n):
    k = 1
    while k < n:

    n, m = len(a), len(b)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smoothed gradient (Adam) with top-K softmax, precise coordinate descent
with incremental convolution updates, and alternating refinement phases.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1 (i.e., 2n*max(conv))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_weights(c, beta, top_k=None):
    """Compute softmax weights over top-k elements of c."""
    n_c = len(c)
    if top_k is not None and top_k < n_c:
        thresh = np.partition(c, n_c - top_k)[n_c - top_k]
        mask = c >= thresh
    else:
        mask = np.ones(n_c, dtype=bool)
    
    c_m = c[mask]
    c_max = float(c_m.max())
    exp_v = np.exp((c_m - c_max) / beta)
    w_sum = float(exp_v.sum())
    
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[int(np.argmax(c))] = 1.0
    else:
        w[mask] = exp_v / w_sum
    return w


def _grad_on_simplex(a, beta, top_k=200):
    """
    Gradient of 2n*beta*logsumexp_topk(conv(a,a)/beta) projected to simplex tangent.
    a is assumed to be on the simplex (sum=1, non-negative).
    """
    n = len(a)
    c = _full_convolve(a, a)
    w = _softmax_weights(c, beta, top_k)
    
    # cross[i] = sum_k w[i+k] * a[k] for k=0..n-1
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    
    raw = 4.0 * n * cross
    # Project to tangent space of simplex (sum of gradient = 0)
    proj = raw - float(np.dot(raw, a))
    return proj


def _adam_phase(a_init, deadline, betas, lr=0.01, top_k=200):
    """Adam optimizer on the simplex with decreasing beta schedule."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        m_buf *= 0.3
        v_buf *= 0.3
        count = 0
        while time.monotonic() < deadline:
            t += 1
            count += 1
            grad = _grad_on_simplex(a, beta, top_k)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1**t)
            vh = v_buf / (1 - b2**t)
            
            a_new = a - lr * mh / (np.sqrt(vh) + eps)
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = best_a.copy()
            else:
                a_new = a_new / s
            
            if count % 200 == 0:
                sc = _score_norm(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
            
            a = a_new
    
    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _coord_descend(a, deadline, verbose=True):
    """
    Precise coordinate descent with incremental convolution updates.
    For each coordinate i, find the best t >= 0 by evaluating candidates.
    """
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    c = _full_convolve(a, a)
    best_score = _score(a)
    
    # Precompute: for each i, the affected range in c is [i, i+n-1] (clipped to [0, 2n-2])
    # We'll iterate over all coordinates multiple passes
    
    pass_count = 0
    stale_passes = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        pass_count += 1
        improved = 0
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i = a[i]
            S_i = S - a_i
            j_start = i
            j_end = min(i + n, 2 * n - 1)
            j_len = j_end - j_start
            
            if j_len <= 0:
                continue
            
            # The direction vector for positions in [j_start, j_end)
            d = a[0:j_len]  # d[k] = a[k] corresponds to c position j_start+k = i+k
            
            # Build candidate t values
            max_c = float(c.max())
            candidates = [0.0, a_i]
            
            if a_i > 1e-15:
                for mult in [0.1, 0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0, 5.0]:
                    candidates.append(a_i * mult)
            
            # Add candidates from active constraints
            # Focus on positions where c[j] is large AND j is in [j_start, j_end)
            c_aff = c[j_start:j_end]
            if c_aff.max() > 0.3 * max_c:
                # Find top positions in affected range
                n_top = min(8, j_len)
                if j_len > n_top:
                    top_idx = np.argpartition(c_aff, -n_top)[-n_top:]
                else:
                    top_idx = np.arange(j_len)
                
                for k_idx in top_idx:
                    j = j_start + k_idx
                    d_val = d[k_idx]
                    if d_val > 1e-10:
                        # Critical point where f_j(t) is at local max:
                        # t_crit = d_val*(S_i + 2*a_i) - c[j]
                        t_crit = d_val * (S_i + 2.0 * a_i) - float(c[j])
                        if t_crit > 0:
                            candidates.append(t_crit)
                            candidates.append(t_crit * 0.8)
                            candidates.append(t_crit * 1.2)
                        # Also where f_j(t) = current score
                        # (c[j] + 2*(t-a_i)*d_val) / (S_i+t)^2 = best_score/(2n)
                        # This is a quadratic in t, skip for now
            
            # Deduplicate and filter
            candidates = [max(0.0, t) for t in candidates]
            candidates = list(set([round(t, 12) for t in candidates]))
            if not candidates:
                candidates = [a_i]
            
            # Evaluate each candidate
            best_t = a_i
            best_f = 2.0 * n * max_c / (S * S)
            
            for t in candidates:
                if abs(t - a_i) < 1e-12 and t == a_i:
                    continue
                k_val = t - a_i
                if abs(k_val) < 1e-14:
                    continue
                
                # New c inside affected range: c[j] + 2*k_val*d[k]
                c_new_aff = c_aff + 2.0 * k_val * d
                # Fix quadratic at j=2i if in range
                if 2 * i >= j_start and 2 * i < j_end:
                    k_in_range = 2 * i - j_start
                    c_new_aff[k_in_range] = c[2 * i] - a_i * a_i + t * t
                
                inside_max = float(c_new_aff.max())
                
                # Outside max: max of c outside [j_start, j_end)
                # Fast path: if global max is outside, it's unchanged
                gj = int(np.argmax(c))
                if gj < j_start or gj >= j_end:
                    outside_max = max_c
                else:
                    # Need to compute max outside
                    # Use a slice approach
                    if j_start > 0:
                        left_max = float(c[:j_start].max())
                    else:
                        left_max = 0.0
                    if j_end < 2 * n - 1:
                        right_max = float(c[j_end:].max())
                    else:
                        right_max = 0.0
                    outside_max = max(left_max, right_max)
                
                new_max = max(inside_max, outside_max)
                new_S = S_i + t
                if new_S < 1e-15:
                    continue
                f_val = 2.0 * n * new_max / (new_S * new_S)
                if f_val < best_f:
                    best_f = f_val
                    best_t = t
            
            if best_t != a_i:
                # Apply update
                k_val = best_t - a_i
                a[i] = best_t
                S = S_i + best_t
                c[j_start:j_end] += 2.0 * k_val * d
                if 2 * i >= j_start and 2 * i < j_end:
                    k_in_range = 2 * i - j_start
                    c[j_start + k_in_range] += k_val * k_val
                improved += 1
        
        # Periodic score check
        if pass_count % 20 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                if verbose:
                    print(f"  CD pass {pass_count}: score={sc:.10f}, improved={improved}")
        
        if improved == 0 and pass_count > 5:
            stale_passes += 1
            if stale_passes >= 3:
                break
        else:
            stale_passes = 0
    
    sc = _score(a)
    if sc < best_score:
        best_score = sc
    return a, best_score
