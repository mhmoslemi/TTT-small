The LP is:

At the optimum, at most n constraints are active (by Carathéodory's theorem, since we have n variables). So the active set has at most n constraints.

2. Solve the LP with only the active constraints (this is a small LP with at most n constraints)
3. Check if the solution violates any non-active constraints
4. If yes, add the most violated constraint to the active set
5. If the active set has more than n constraints, remove the least "important" one
6. Repeat until no violation

This is the revised simplex method for LPs. In practice, it converges in a few iterations.

    """
    t0 = time.monotonic()
    # Start with the peak constraint
    active = [n - 1]
    g = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        # Build sub-matrix for active constraints
        m_active = len(active)
        A_sub = np.zeros((m_active, n))
        for i, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                A_sub[i, j] = a[k - j]

        c_obj = -np.ones(n)
        b_vec = M * np.ones(m_active)

        try:
            res = linprog(c_obj, A_ub=A_sub, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': max(1.0, time_limit - (time.monotonic() - t0))})
        except Exception:
            break

        if res.status != 0:
            break

        g_new = res.x
        g_sum = float(g_new.sum())
        if g_sum < 1e-12:
            break
        g = g_new

        # Check all constraints
        ag = _full_convolve(a, g)
        max_violation = float(ag.max() - M)
        if max_violation < 1e-9:
            break  # Feasible

        # Add most violated constraint
        k_new = int(np.argmax(ag))
        if k_new not in active:
            active.append(k_new)

        # Control active set size
        max_active = max(50, n // 2)
        if len(active) > max_active:
            # Remove constraint with most slack
            slacks = M - ag[active]
            remove_idx = int(np.argmax(slacks))
            active.pop(remove_idx)

    return g


def _frank_wolfe_step(a, t, g):
    """Blend a with g (rescaled to same sum)."""
    g_sum = float(g.sum())
    if g_sum < 1e-15:
        return a.copy()
    a_sum = float(a.sum())
    g_scaled = g * (a_sum / g_sum)
    return (1.0 - t) * a + t * g_scaled


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init.copy()

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.008)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004, 0.00015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        scale = max(1.0, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (0.005 * scale * 5))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _nesterov_polish(a_init, deadline, max_steps=50000, lr_init=0.008,
                     momentum_init=0.85, momentum_end=0.995, tol_frac=0.02):
    """Nesterov-accelerated subgradient with adaptive active set width."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    beta_mom = momentum_init
    best_step = 0
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=tol_frac)

        # Adaptive momentum
        progress = min(1.0, (i + 1) / 2000.0)
        beta_mom = momentum_init + (momentum_end - momentum_init) * progress

        # Adaptive learning rate (decay)
        lr = lr_init / (1.0 + 0.0001 * i)

        v = beta_mom * v - lr * grad
        z = z + v
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _fw_iterations(a, deadline, max_iters=8, t_init=0.5, t_decay=0.9):
    """Run Frank-Wolfe iterations with LP direction finding."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    t = t_init

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        # Solve LP for direction
        lp_time = min(30.0, max(5.0, (deadline - time.monotonic()) / 3.0))
        g = _active_set_lp(a, M, n, time_limit=lp_time)

        if g is None or g.sum() < 1e-10:
            break

        g_sum = float(g.sum())
        if g_sum < float(a.sum()) * (1.0 + 1e-6):
            # No improvement possible
            t *= t_decay
            continue

        # Try different step sizes via line search
        best_local = a.copy()
        best_local_sc = best_sc
        for t_try in [t, t * 0.7, t * 0.5, t * 0.3, 0.1]:
            a_try = (1.0 - t_try) * a + t_try * (g * a.sum() / g_sum)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_local = a_try.copy()

        if best_local_sc < best_sc:
            best_sc = best_local_sc
            best_a = best_local.copy()
            a = best_local.copy()
            print(f"    FW iter {it+1}: score={best_sc:.10f}, t={t:.3f}, g_sum={g_sum:.4f}")
        else:
            t *= t_decay
            if t < 0.01:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss":
        t = np.linspace(-3, 3, n)
        a = np.exp(-t**2 / 2.0)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(-3, 3, n)
        a = np.exp(-((t - 1.2)**2) / 0.8) + np.exp(-((t + 1.2)**2) / 0.8)
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        # Slight taper at edges
        edge = int(n * 0.1)
        a[:edge] = np.linspace(0.5, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.5, edge)
        return a / a.sum()
    elif kind == "quarter_wave":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * t / 2.0) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_frequencies(a, rng, n_modes=5, amp=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        a_amp = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + a_amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_blocks(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i+block_size].copy()
        z[i:i+block_size] = z[j:j+block_size]
        z[j:j+block_size] = tmp
    return _softmax(z)


def _smoothed_continuation(a_init, deadline, min_remaining=20.0):
    """Run a continuation on beta with Nesterov momentum between L-BFGS calls."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_max
    base_beta = max(gap, c_max * 0.01)

    # Coarser beta schedule for this phase
    betas = base_beta * np.array([0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    beta_mom = 0.9
    lr = 0.005

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break

        # L-BFGS phase
        max_iter = min(1000, max(30, int(remaining * 0.4 / (0.005 * max(1, n/2048.0)))))
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                         'maxcor': 40, 'maxls': 50}
            )
            z = result.x
        except Exception:
            pass

        # Nesterov momentum phase
        n_steps = min(2000, max(100, int(remaining * 0.3 / (0.003 * max(1, n/2048.0)))))
        for i in range(n_steps):
            if time.monotonic() + min_remaining > deadline:
                break
            a_cur = _softmax(z)
            c_cur = _full_convolve(a_cur, a_cur)
            grad = _gradient_active(a_cur, c_cur, n, tol_frac=0.01)

            v = beta_mom * v - lr * grad
            z = z + v
            z = z - z.mean()

            if i % 50 == 0:
                sc = _score(a_cur)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_cur.copy()

        beta_mom = min(0.995, beta_mom * 1.01)
        lr *= 0.7

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: LP Frank-Wolfe + Nesterov subgradient +
    smoothed continuation + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 2048
        parent = _make_profile(n_parent, "uniform")
        update_best(parent, _score(parent), "uniform_init")

    # --- Phase 1: Initial L-BFGS on parent ---
    print(f"\n[Phase 1] Initial L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a1 = _run_lbfgs(parent, deadline, min_time_remaining=600.0)
    update_best(a1, _score(a1), "lbfgs_init")

    # --- Phase 2: LP-based Frank-Wolfe ---
    print(f"\n[Phase 2] Frank-Wolfe with LP (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        fw_deadline = min(deadline - 400, time.monotonic() + 300)
        a_fw, sc_fw = _fw_iterations(best_seq, fw_deadline, max_iters=6, t_init=0.4, t_decay=0.85)
        update_best(a_fw, sc_fw, "fw_lp")

        # Second round of FW with smaller step
        if sc_fw < best_score and time.monotonic() < deadline - 350:
            fw_deadline2 = min(deadline - 350, time.monotonic() + 120)
            a_fw2, sc_fw2 = _fw_iterations(a_fw, fw_deadline2, max_iters=4, t_init=0.2, t_decay=0.8)
            update_best(a_fw2, sc_fw2, "fw_lp_2")

    # --- Phase 3: Nesterov subgradient polish ---
    print(f"\n[Phase 3] Nesterov subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        ns_deadline = min(deadline - 200, time.monotonic() + 180)
        a_ns, sc_ns = _nesterov_polish(best_seq, ns_deadline, max_steps=30000,
                                       lr_init=0.006, tol_frac=0.015)
        update_best(a_ns, sc_ns, "nesterov_1")

    # --- Phase 4: Smoothed continuation from best ---
    print(f"\n[Phase 4] Smoothed continuation (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        sc_deadline = min(deadline - 150, time.monotonic() + 120)
        a_sc, sc_sc = _smoothed_continuation(best_seq, sc_deadline, min_remaining=30.0)
        update_best(a_sc, sc_sc, "smoothed_cont")

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 4096, 6144, 8192]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break

        # Initialize
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        sc_init = _score(a_init)

        # Quick L-BFGS
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick Nesterov
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _nesterov_polish(a_opt, min(deadline - 30, time.monotonic() + 60),
                                               max_steps=5000, lr_init=0.004, tol_frac=0.01)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_ns")

    # --- Phase 6: Diverse restarts from best ---
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        strategies = [
            ("freq_perturb", lambda: _perturb_frequencies(best_seq, rng, n_modes=4, amp=0.04)),
            ("block_swap", lambda: _perturb_blocks(best_seq, rng, n_blocks=5)),
            ("freq_large", lambda: _perturb_frequencies(best_seq, rng, n_modes=7, amp=0.08)),
        ]
        for label, make_perturbed in strategies:
            if time.monotonic() > deadline - 20:
                break
            a_pert = make_perturbed()
            a_r = _run_lbfgs(a_pert, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, label)

    # --- Phase 7: Final fine polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 2:
        # Very fine L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 2 - time.monotonic()
        if remaining > 3:
            max_iter = min(1000, max(30, int(remaining / 0.02)))
            try:
                result = minimize(
                    _smoobj, z, args=(beta_final, n),
                    jac=_smograd,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                             'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final Nesterov
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _nesterov_polish(best_seq, deadline - 1, max_steps=1000,
                                              lr_init=0.003, tol_frac=0.005)
            update_best(a_fin2, sc_fin2, "final_ns")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
