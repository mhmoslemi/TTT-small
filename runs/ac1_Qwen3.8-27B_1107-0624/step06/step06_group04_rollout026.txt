"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the state-of-the-art LP direction-finding approach combined with
FISTA smoothing, Adam fine-tuning, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f, k_indices):
    """Build the constraint matrix rows for specified lag indices.
    A[i, j] = f[k_i - j] if 0 <= k_i - j <= len(f)-1 else 0"""
    n = len(f)
    A = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        # A[i, j] = f[k - j] if 0 <= k - j <= n-1
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[i, j_min:j_max+1] = f[k - j_max:k - j_min:-1]
    return A

def _solve_lp_direction(f, M, max_rounds=30, initial_frac=0.05):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b)[k] <= M for all k.
    Uses delayed constraint (cutting plane) approach."""
    n = len(f)
    
    # Initial active set: lags where conv(f,f) is in top fraction
    ff = _full_convolve(f, f)
    thresh = np.percentile(ff, 100 * (1 - initial_frac))
    active_k = np.where(ff >= thresh)[0]
    
    b_sol = None
    for round_idx in range(max_rounds):
        # Build constraint matrix
        A = _build_conv_matrix(f, active_k)
        b_vec = np.full(len(active_k), M)
        
        # Solve: minimize -sum(b) s.t. A @ b <= b_vec, b >= 0
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            # Try with fewer constraints
            if len(active_k) > 10:
                active_k = active_k[:len(active_k)//2]
                continue
            break
        
        b_sol = res.x
        
        # Check for violated constraints
        fb = _full_convolve(f, b_sol)
        violations = np.where(fb > M * (1 + 1e-6))[0]
        
        if len(violations) == 0:
            break
        
        # Add violated constraints (limit to top violators)
        if len(violations) > 100:
            viol_vals = fb[violations]
            top_viol = violations[np.argsort(viol_vals)[::-1][:100]]
            active_k = np.unique(np.concatenate([active_k, top_viol]))
        else:
            active_k = np.unique(np.concatenate([active_k, violations]))
    
    if b_sol is None or b_sol.sum() < 1e-15:
        return np.ones(n) / n
    
    b_sol = np.maximum(b_sol, 0)
    s = b_sol.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return b_sol / s

def _lp_iteration(f, t_mix=0.3):
    """One LP-guided iteration: find direction b, mix with f."""
    n = len(f)
    ff = _full_convolve(f, f)
    M = float(ff.max())
    
    b_dir = _solve_lp_direction(f, M)
    b_dir = b_dir / b_dir.sum()
    
    # Mix
    f_new = (1 - t_mix) * f + t_mix * b_dir
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-15:
        return f
    f_new = f_new / s
    
    return f_new

def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        # Gradient: 2 * conv(w, y_reversed)[n-1:2n-1]
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w
            
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1**t)
            vh = v_buf / (1.0 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            
            if t % 5000 == 0:
                current_lr = max(0.001, current_lr * 0.95)
            if no_improve > 100 and t > 500:
                current_lr = max(0.0005, current_lr * 0.9)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _coord_descent(a_init, n_sweeps=5, deadline=None):
    """Coordinate descent: for each coordinate, find optimal value via 1D search."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute current convolution
        c = _full_convolve(a, a)
        
        # Process coordinates in random order (or reverse to break symmetry)
        order = np.arange(n)
        np.random.shuffle(order)
        
        for i in order:
            if deadline is not None and time.monotonic() > deadline:
                break
            
            a_old_i = a[i]
            # Effect of changing a[i] to a[i] + delta on convolution:
            # c[2i] changes by 2*a[i]*delta + delta^2
            # c[i+j] (j != i) changes by 2*delta*a[j] (for i+j < 2n-1)
            # c[j-i] ... actually let's just compute directly
            
            # Try a few candidate values
            # The optimal a[i] in [0, 2*a_old_i] (heuristic range)
            # Actually, let's just try 0 and the current value and a few perturbations
            
            # Quick approach: try setting a[i] to 0, and to current, and see which is better
            # More principled: the max of conv as a function of a[i] is convex,
            # so we can use ternary search on [0, 2*a_old_i + 0.1/n]
            
            a_max_candidate = max(2 * a_old_i + 1.0/n, 1.0/n)
            
            # Ternary search
            lo = 0.0
            hi = a_max_candidate
            for _ in range(12):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                
                a[i] = m1
                c1 = _full_convolve(a, a)
                s1 = a.sum()
                sc1 = 2.0 * n * c1.max() / (s1 * s1) if s1 > 1e-15 else np.inf
                
                a[i] = m2
                c2 = _full_convolve(a, a)
                s2 = a.sum()
                sc2 = 2.0 * n * c2.max() / (s2 * s2) if s2 > 1e-15 else np.inf
                
                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1
            
            # Best in the interval
            a[i] = (lo + hi) / 2
            c_best = _full_convolve(a, a)
            s_best = a.sum()
            sc_best = 2.0 * n * c_best.max() / (s_best * s_best) if s_best > 1e-15 else np.inf
            
            if sc_best < best_score * 1.001:
                best_score = min(best_score, sc_best)
                best_a = a.copy()
            else:
                a[i] = a_old_i  # revert
        
        # Renormalize
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score

"""
LP-guided optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines the state-of-the-art LP direction-finding with FISTA smoothing,
Adam fine-tuning, and coordinate descent polishing for ultimate refinement.
Key innovation: uses LP to find optimal mixing direction considering all
convolution constraints simultaneously, not just the active one.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_rows(f, k_indices):
    """Build rows of convolution constraint matrix for given lag indices.
    A[i, j] = f[k_i - j] if 0 <= k_i - j <= len(f)-1 else 0"""
    n = len(f)
    A = np.zeros((len(k_indices), n))
    for i, k in enumerate(k_indices):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            src = k - j_max
            A[i, j_min:j_max + 1] = f[src:k - j_min + 1][::-1]
    return A


def _solve_lp_direction(f, M, max_rounds=20, initial_top_frac=0.03):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k.
    Uses delayed constraint (cutting plane) approach for speed."""
    n = len(f)

    # Initial active set: lags where conv(f,f) is in top fraction
    ff = _full_convolve(f, f)
    n_lags = len(ff)
    n_initial = max(20, int(n_lags * initial_top_frac))
    top_idx = np.argsort(ff)[::-1][:n_initial]
    active_k = sorted(top_idx.tolist())

    b_sol = None
    for round_idx in range(max_rounds):
        A = _build_conv_rows(f, active_k)
        b_vec = np.full(len(active_k), M)

        c_obj = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        except Exception:
            break

        if not res.success:
            if len(active_k) > 10:
                active_k = active_k[:len(active_k) // 2]
                continue
            break

        b_sol = np.maximum(res.x, 0.0)

        # Check for violated constraints
        fb = _full_convolve(f, b_sol)
        violations = np.where(fb > M * (1.0 + 1e-7))[0]

        if len(violations) == 0:
            break

        # Add top violators (limit number)
        if len(violations) > 50:
            viol_vals = fb[violations]
            top_v = violations[np.argsort(viol_vals)[::-1][:50]]
        else:
            top_v = violations
        active_k = sorted(set(active_k) | set(top_v.tolist()))

    if b_sol is None or b_sol.sum() < 1e-15:
        return np.ones(n) / n

    b_sol = b_sol / b_sol.sum()
    return b_sol


def _lp_one_iteration(f, t_mix=0.3):
    """One LP-guided mixing iteration."""
    n = len(f)
    ff = _full_convolve(f, f)
    M = float(ff.max())

    b_dir = _solve_lp_direction(f, M)
    b_dir = b_dir / b_dir.sum()

    f_new = (1.0 - t_mix) * f + t_mix * b_dir
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    f_new = f_new / s
    return f_new


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)
            if no_improve > 100 and t > 500:
                current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descent_subset(a_init, n_coords=200, n_sweeps=3, deadline=None, seed=0):
    """Coordinate descent on a random subset of coordinates."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Pick random subset of coordinates to optimize
        if n_coords < n:
            indices = rng.choice(n, size=n_coords, replace=False)
        else:
            indices = np.arange(n)

        for i in indices:
            if deadline is not None and time.monotonic() > deadline:
                break

            a_old = a[i]
            s_old = a.sum()

            # Try candidate values via ternary search
            lo = 0.0
            hi = max(3.0 * a_old + 2.0 / n, 2.0 / n)

            for _ in range(10):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                a[i] = m1
                s1 = s_old - a_old + m1
                if s1 < 1e-15:
                    sc1 = np.inf
                else:
                    c1 = _full_convolve(a, a)
                    sc1 = 2.0 * n * float(c1.max()) / (s1 * s1)

                a[i] = m2
                s2 = s_old - a_old + m2
                if s2 < 1e-15:
                    sc2 = np.inf
                else:
                    c2 = _full_convolve(a, a)
                    sc2 = 2.0 * n * float(c2.max()) / (s2 * s2)

                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1

            a[i] = (lo + hi) / 2.0
            s_new = a.sum()
            if s_new > 1e-15:
                sc_try = _score(a)
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a.copy()
                else:
                    a[i] = a_old

    return best_a, best_score

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Parent loaded: n={len(parent)}")
    except Exception:
        parent = None

    if parent is None:
        n = 4096
        parent = np.ones(n) / n
        print(f"No parent, using uniform n={n}")

    best_seq = parent.copy()
    best_score = _score(best_seq)
    n_main = len(best_seq)
    print(f"Initial score: {best_score:.10f}, n={n_main}")

    # === Phase 1: LP-guided iterations (main phase) ===
    print("Phase 1: LP-guided iterations")
    phase1_end = min(deadline, t_start + 500.0)

    f = best_seq.copy()
    t_mix = 0.4
    consecutive_fail = 0

    for it in range(60):
        if time.monotonic() > phase1_end - 15:
            break

        t0 = time.monotonic()
        f_new = _lp_one_iteration(f, t_mix=t_mix)
        dt = time.monotonic() - t0

        sc_new = _score_norm(f_new)
        sc_old = _score_norm(f)

        if sc_new < sc_old * (1.0 - 1e-9):
            f = f_new
            consecutive_fail = 0
            # Slightly increase mixing if improving well
            if sc_new < sc_old * 0.99:
                t_mix = min(0.5, t_mix * 1.1)
            sc_full = _score(f)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = f.copy()
            print(f"  LP it={it}: score={sc_full:.10f} t_mix={t_mix:.3f} dt={dt:.1f}s")
        else:
            consecutive_fail += 1
            t_mix = max(0.05, t_mix * 0.85)
            if consecutive_fail >= 4:
                # Perturb to escape local minimum
                z = np.log(np.maximum(f, 1e-15)) + 0.03 * rng.standard_normal(n_main)
                z = z - z.mean()
                f = _softmax_compute(z)
                consecutive_fail = 0
                t_mix = 0.25
                print(f"  LP it={it}: perturbed, t_mix={t_mix:.3f}")

    sc_f = _score(f)
    if sc_f < best_score:
        best_score = sc_f
        best_seq = f.copy()
    print(f"End Phase 1: best={best_score:.10f} elapsed={time.monotonic()-t_start:.0f}s")

    # === Phase 2: FISTA with beta continuation ===
    print("Phase 2: FISTA fine-tuning")
    phase2_end = min(deadline, t_start + 1000.0)

    f = best_seq.copy() / best_seq.sum()
    c_f = _full_convolve(f, f)
    max_cf = float(c_f.max())

    betas = []
    beta = 0.003 * max_cf
    while beta > 0.000001 * max_cf:
        betas.append(beta)
        beta *= 0.5

    for i, b in enumerate(betas[:6]):
        if time.monotonic() > phase2_end - 10:
            break
        per_t = phase2_end - time.monotonic() - 5
        if per_t < 5:
            break
        f, sc_f = _fista_smoothed(f, b, time.monotonic() + min(25.0, per_t), 10000)
        f = f / f.sum()
        sc_full = _score(f)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = f.copy()
            print(f"  FISTA beta={b:.8f}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: More LP with smaller mixing ===
    print("Phase 3: LP with small t_mix")
    phase3_end = min(deadline, t_start + 1300.0)

    f = best_seq.copy() / best_seq.sum()
    t_mix = 0.15

    for it in range(30):
        if time.monotonic() > phase3_end - 10:
            break

        f_new = _lp_one_iteration(f, t_mix=t_mix)
        sc_new = _score_norm(f_new)
        sc_old = _score_norm(f)

        if sc_new < sc_old * (1.0 - 1e-9):
            f = f_new
            sc_full = _score(f)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = f.copy()
                print(f"  LP3 it={it}: score={sc_full:.10f}")
        else:
            t_mix = max(0.03, t_mix * 0.8)
            if t_mix <= 0.03 and it > 5:
                break

    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Adam log-space ===
    print("Phase 4: Adam fine-tuning")
    phase4_end = min(deadline, t_start + 1550.0)

    f = best_seq.copy() / best_seq.sum()
    c_f = _full_convolve(f, f)
    max_cf = float(c_f.max())

    betas_fine = [0.0005 * max_cf, 0.0002 * max_cf, 0.0001 * max_cf,
                  0.00005 * max_cf, 0.00002 * max_cf, 0.00001 * max_cf]

    per_t = phase4_end - time.monotonic() - 10
    if per_t > 20:
        f_opt, sc_opt = _adam_logspace(f, time.monotonic() + per_t, betas_fine,
                                       lr_init=0.004, max_iter_total=400000)
        sc_full = _score(f_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = f_opt.copy()
            print(f"  Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: Final LP polish ===
    print("Phase 5: Final LP polish")
    phase5_end = min(deadline, t_start + 1700.0)

    f = best_seq.copy() / best_seq.sum()
    t_mix = 0.08

    for it in range(15):
        if time.monotonic() > phase5_end - 10:
            break
        f_new = _lp_one_iteration(f, t_mix=t_mix)
        sc_new = _score_norm(f_new)
        sc_old = _score_norm(f)
        if sc_new < sc_old * (1.0 - 1e-9):
            f = f_new
            sc_full = _score(f)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = f.copy()
                print(f"  LP5 it={it}: score={sc_full:.10f}")
        else:
            t_mix = max(0.02, t_mix * 0.7)

    print(f"End Phase 5: best={best_score:.10f}")

    # === Phase 6: Coordinate descent on subset ===
    print("Phase 6: Coordinate descent polish")
    phase6_end = min(deadline, t_start + 1770.0)
    per_t = phase6_end - time.monotonic() - 5
    if per_t > 30:
        f = best_seq.copy() / best_seq.sum()
        a_cd, sc_cd = _coord_descent_subset(f, n_coords=150, n_sweeps=2,
                                            deadline=phase6_end, seed=int(seed))
        sc_full = _score(a_cd)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_cd.copy()
            print(f"  CoordDescent: score={sc_full:.10f}")

    print(f"End Phase 6: best={best_score:.10f}")

    # === Final validation ===
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
        s = final.sum()
        final = final / s

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
