def _next_pow2(n):
    k = 1

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Adam in log-space with top-K softmax, coordinate descent with 1D numerical optimization,
and subgradient polishing. Tries multiple sequence lengths (4096, 8192, 16384).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _adam_logspace(a_init, deadline, betas, lr_init, top_k, max_iter_per_beta):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_beta:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 80 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.96)

        m_buf *= 0.4
        v_buf *= 0.4

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_1d_eval(a, i, t, c_base, d_vec, S0, n):
    c_trial = c_base + d_vec * t
    max_c = float(c_trial.max())
    denom = (S0 + t) ** 2
    if denom < 1e-30:
        return np.inf
    return 2.0 * n * max_c / denom


def _coord_descend(a_init, deadline, n_passes=5, max_inner_iter=30):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_score = _score_norm(a)
    rng = np.random.default_rng(2024)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            S = a.sum()
            S0 = S - a_i_old

            c = _full_convolve(a, a)

            # Compute delta: (a'*a')[k] - (a*a)[k] when a[i] changes from a_i_old to a_i_old + t
            # For k != 2i and 0<=k-i<n: delta = 2*t*a[k-i]
            # For k = 2i: delta = 2*a_i_old*t + t^2
            # So (a'*a')[k] = c[k] + 2*t*a[k-i] (for k!=2i, valid)
            #                 = c[k] + 2*a_i_old*t + t^2 (for k=2i)
            #                 = c[k] (otherwise)
            # This is linear in t except at k=2i.
            # For the purpose of finding the best t, we approximate by treating k=2i specially.

            # Build d_vec for the linear part: d_vec[k] = 2*a[k-i] if 0<=k-i<n, else 0
            d_vec = np.zeros(2 * n - 1)
            lo = max(0, i - (2 * n - 2))  # k-i >= 0 => k >= i
            hi = min(2 * n - 2, i + n - 1)  # k-i < n => k < i+n
            d_vec[lo:hi + 1] = 2.0 * a[0:n]  # a[k-i] for k in [i, i+n-1]

            # The c_base for the linear approximation (ignoring the t^2 term at k=2i)
            c_base = c.copy()

            # Candidate t values
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5]
            if a_i_old < 1e-10:
                t_candidates = [0.0, 0.001, 0.005, 0.01]
            else:
                t_candidates.extend([a_i_old * 0.25, a_i_old * 0.75, a_i_old * 2.0, a_i_old * 0.1])

            # Find critical points from top-m active constraints
            top_m = min(20, 2 * n - 1)
            idx_top = np.argsort(c)[-top_m:]
            for j in idx_top:
                dj = d_vec[j]
                if abs(dj) > 1e-15:
                    # f(t) = 2n*(c_base[j] + dj*t) / (S0+t)^2
                    # f'(t) = 0 => 2n*dj*(S0+t)^2 - 2n*(c_base[j]+dj*t)*2*(S0+t) = 0
                    # => dj*(S0+t) - 2*(c_base[j]+dj*t) = 0
                    # => dj*S0 + dj*t - 2*c_base[j] - 2*dj*t = 0
                    # => dj*S0 - dj*t - 2*c_base[j] = 0
                    # => t = (dj*S0 - 2*c_base[j]) / dj = S0 - 2*c_base[j]/dj
                    t_star = S0 - 2.0 * c_base[j] / dj
                    if 0 <= t_star <= max(1.0, 10.0 * a_i_old):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.7))
                        t_candidates.append(t_star * 1.3)

            t_candidates = [max(0.0, min(S, tc)) for tc in t_candidates]
            # Deduplicate
            t_candidates = sorted(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = _coord_1d_eval(a, i, a_i_old, c_base, d_vec, S0, n)

            for tc in t_candidates:
                f_val = _coord_1d_eval(a, i, tc, c_base, d_vec, S0, n)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            # Local refinement around best_t using golden section
            if 0 < best_t < S:
                span = max(0.01 * S, 0.1 * a_i_old + 1e-6)
                lo_t = max(0.0, best_t - span)
                hi_t = min(S, best_t + span)
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _iter in range(max_inner_iter):
                    if time.monotonic() > deadline:
                        break
                    m1 = hi_t - gr * (hi_t - lo_t)
                    m2 = lo_t + gr * (hi_t - lo_t)
                    f1 = _coord_1d_eval(a, i, m1, c_base, d_vec, S0, n)
                    f2 = _coord_1d_eval(a, i, m2, c_base, d_vec, S0, n)
                    if f1 < f2:
                        hi_t = m2
                    else:
                        lo_t = m1
                    if hi_t - lo_t < 1e-10 * S:
                        break
                t_mid = (lo_t + hi_t) / 2.0
                f_mid = _coord_1d_eval(a, i, t_mid, c_base, d_vec, S0, n)
                if f_mid < best_f:
                    best_f = f_mid
                    best_t = t_mid

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    return a, best_score


def _subgrad_nesterov(a_init, deadline, max_iter=20000):
    """Nesterov-accelerated subgradient method on the actual max."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score(a)
    t_val = 1.0
    lr_base = 0.1

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - v)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y < 1e-15:
            break
        y = y / s_y

        c = _full_convolve(y, y)
        max_pos = int(np.argmax(c))
        max_val = c[max_pos]
        s = y.sum()

        # Subgradient of (a*a)[max_pos] wrt a:
        # sg[i] = 2*a[max_pos - i] if 0 <= max_pos - i < n, else 0
        sg = np.zeros(n)
        lo = max(0, max_pos - n + 1)
        hi = min(n - 1, max_pos)
        if hi >= lo:
            sg[lo:hi + 1] = 2.0 * y[max_pos - hi:max_pos - lo + 1][::-1]

        # Gradient of score = 2n*max_val/s^2 wrt a:
        # = 2n * (sg * s - 2*max_val) / s^3
        lr = lr_base / (t_next ** 0.7)
        a_new = y - lr * (sg * s - 2.0 * max_val) / (s * s)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / s_new

        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        v = a
        a = a_new
        t_val = t_next

    return best_a, best_score


def _interpolate_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            a4, sc4 = _subgrad_nesterov(a3, phase4_end, max_iter=50000)
            sc4_full = _score(a4)
            print(f"  SubgradNesterov: score={sc4_full:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc4_full < best_score:
                best_score = sc4_full
                best_seq = a4.copy()

        # Phase 5: Another round of coord descent from best
        if time.monotonic() < deadline - 20:
            phase5_end = min(deadline - 20, time.monotonic() + 60.0)
            a5, sc5 = _coord_descend(best_seq, phase5_end, n_passes=2)
            sc5_full = _score(a5)
            print(f"  CoordDescent2: score={sc5_full:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc5_full < best_score:
                best_score = sc5_full
                best_seq = a5.copy()

    # Final diversity restarts
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Diversity restarts (remaining: {deadline - time.monotonic():.0f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        rng = np.random.default_rng(999)

        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 20) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            if ri == 0:
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 512))
                p_start = np.roll(a_base, shift).copy()
            else:
                freq = rng.uniform(1.0, 5.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.005 * max_cs, 0.002 * max_cs, 0.001 * max_cs, 0.0005 * max_cs]

            p_best, _ = _adam_logspace(p_start, time.monotonic() + max(10, per_restart * 0.6),
                                       betas_r, lr_init=0.006, top_k=80, max_iter_per_beta=50000)
            # Quick coord descent polish
            if time.monotonic() < deadline - 20:
                p_pol, _ = _coord_descend(p_best, time.monotonic() + max(5, per_restart * 0.3), n_passes=1)
                sc_r = _score(p_pol)
            else:
                sc_r = _score(p_best)
                p_pol = p_best

            print(f"  Restart {ri}: score={sc_r:.10f} t={time.monotonic()-t_start:.0f}s")
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_pol.copy()

    # Final output
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
