"""

cosine-annealed learning rates, adaptive softmax smoothing, block coordinate

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_logspace(a_init, deadline, beta_list, lr_initial, lr_final, nesterov_momentum=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    # Nesterov velocity in log-space
    v_nest = np.zeros(n)

    for beta in beta_list:
        if time.monotonic() > deadline:

        phase_start = time.monotonic()

        max_iter_phase = max(100, min(max_iter_phase, 500000))

        total_phases = len(beta_list)
        phase_idx = beta_list.index(beta) if beta in beta_list else 0

        for it in range(max_iter_phase):
            if time.monotonic() > deadline:
                break

            t += 1
            # Cosine annealing within phase
            progress = min(1.0, it / max(1, max_iter_phase))
            lr = lr_final + 0.5 * (lr_initial - lr_final) * (1 + np.cos(np.pi * progress))

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            c_shifted = (c - c_max) / beta
            e = np.exp(c_shifted)

            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):

            c_look_max = float(c_look.max())
            c_look_shifted = (c_look - c_look_max) / beta

            grad_look = _compute_grad_logspace(a_look, c_look, w_look, n)

            mh = m_buf / (1.0 - b1 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)

            v_nest = nesterov_momentum * v_nest - update

            z = np.clip(z, -60.0, 60.0)

def _block_perturb_and_optimize(a, deadline, rng, n_blocks=8, block_frac=0.02):
    """Perturb random blocks of the sequence and re-optimize locally."""
    n = len(a)
    block_size = max(8, int(n * block_frac))
    best_a = a.copy()
    best_score = _score_norm(a)

    for _ in range(n_blocks):
        if time.monotonic() > deadline:
            break
        # Pick random block
        start = rng.integers(0, n - block_size)
        end = start + block_size

        # Create perturbation in log-space
        z = np.log(np.maximum(best_a, 1e-15))
        z_pert = z.copy()
        z_pert[start:end] += rng.standard_normal(end - start) * 0.1
        z_pert = z_pert - z_pert.mean()
        a_pert = _softmax_compute(z_pert)
        sc_pert = _score_norm(a_pert)

        # If perturbation improved, do a few Adam steps from there
        if sc_pert < best_score:
            # Fine-tune from perturbed point
            small_betas = [1e-5, 5e-6, 2e-6]
            a_opt, sc_opt = _adam_logspace(a_pert, min(deadline, time.monotonic() + 30),
                                           small_betas, lr_initial=0.003, lr_final=0.001,
                                           nesterov_momentum=0.8)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"    Block perturb improved: {best_score:.8f}")
        else:
            # Try a smaller perturbation
            z_pert2 = z.copy()
            z_pert2[start:end] += rng.standard_normal(end - start) * 0.05
            z_pert2 = z_pert2 - z_pert2.mean()

    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient of smooth objective
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * n * 2.0 * h  # d/d(a) of 2*n*softmax_dot

        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            # Adaptive step size
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 8.0 * n * max_c_new / (beta * max_c_new + 1e-30)
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _cosine_blend(a, b, t):
    """Spherical cosine blend in log-space."""
    za = np.log(np.maximum(a, 1e-15)) - np.log(np.maximum(a, 1e-15)).mean()
    zb = np.log(np.maximum(b, 1e-15)) - np.log(np.maximum(b, 1e-15)).mean()
    z = (1 - t) * za + t * zb
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        sc_parent = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    n = len(parent)
    print(f"Starting optimization for n={n}, target < 1.5027")

    # === Phase 1: Main Adam optimization from parent ===
    if time.monotonic() < deadline - 100:
        print(f"\n=== Phase 1: Main Adam optimization ===")
        a_work = parent.copy()

        # Compute initial max for beta scaling
        c0 = _full_convolve(a_work, a_work)
        max_c0 = float(c0.max())

        # Beta schedule: start moderately fine, go very fine
        beta_schedule = [
            0.005 * max_c0,
            0.002 * max_c0,
            0.001 * max_c0,
            0.0005 * max_c0,
            0.0002 * max_c0,
            0.0001 * max_c0,
            0.00005 * max_c0,
        ]

        phase1_deadline = min(deadline - 200, t_start + 900.0)
        a_opt, sc_opt = _adam_logspace(
            a_work, phase1_deadline, beta_schedule,
            lr_initial=0.01, lr_final=0.002, nesterov_momentum=0.9
        )
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Phase 1 result: score={sc_opt_full:.10f}")
        else:
            print(f"  Phase 1 did not improve: {sc_opt_full:.10f} vs {best_score:.10f}")

    # === Phase 2: Multi-restart diversification ===
    if best_seq is not None and time.monotonic() < deadline - 80:
        print(f"\n=== Phase 2: Multi-restart diversification ===")
        a_best = best_seq.copy()
        a_best = a_best / a_best.sum()
        n_b = len(a_best)

        n_restarts = 5
        per_restart_time = (deadline - 80 - time.monotonic()) / n_restarts

        restart_strategies = [
            ("small_perturb", 0.01),
            ("medium_perturb", 0.03),
            ("roll_shift", None),
            ("blend_tapered", None),
            ("large_perturb", 0.06),
        ]

        for i, (name, param) in enumerate(restart_strategies):
            if time.monotonic() > deadline - 50:
                break

            if name == "small_perturb":
                z = np.log(np.maximum(a_best, 1e-15))
                z += param * rng.standard_normal(n_b)
                z -= z.mean()
                p_start = _softmax_compute(z)
            elif name == "medium_perturb":
                z = np.log(np.maximum(a_best, 1e-15))
                z += param * rng.standard_normal(n_b)
                z -= z.mean()
                p_start = _softmax_compute(z)
            elif name == "roll_shift":
                sz = rng.integers(1, max(2, n_b // 512))
                p_start = np.roll(a_best, sz).copy()
                p_start = p_start / p_start.sum()
            elif name == "blend_tapered":
                t = np.linspace(0, 1, n_b)
                tapered = np.sin(np.pi * t) ** 0.5 + 0.1
                tapered = tapered / tapered.sum()
                p_start = 0.8 * a_best + 0.2 * tapered
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif name == "large_perturb":
                z = np.log(np.maximum(a_best, 1e-15))
                z += param * rng.standard_normal(n_b)
                z -= z.mean()
                p_start = _softmax_compute(z)
            else:
                p_start = a_best.copy()

            sc_start = _score(p_start)
            print(f"  Restart {i} ({name}): start_score={sc_start:.10f}")

            # Fine-tune from perturbed start
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            fine_betas = [0.001 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs, 0.00003 * max_cs]

            restart_deadline = min(deadline - 50, time.monotonic() + max(5.0, per_restart_time))
            a_r, sc_r = _adam_logspace(
                p_start, restart_deadline, fine_betas,
                lr_initial=0.005, lr_final=0.001, nesterov_momentum=0.85
            )
            sc_r_full = _score(a_r)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = a_r.copy()
                print(f"   Improved! score={sc_r_full:.10f}")
            else:
                print(f"    No improvement: {sc_r_full:.10f}")

    # === Phase 3: Block coordinate perturbations ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Phase 3: Block coordinate perturbations ===")
        a_bl = best_seq.copy()
        a_bl = a_bl / a_bl.sum()
        block_deadline = min(deadline - 20, time.monotonic() + 120.0)
        a_bl, sc_bl = _block_perturb_and_optimize(a_bl, block_deadline, rng, n_blocks=12, block_frac=0.03)
        sc_bl_full = _score(a_bl)
        if sc_bl_full < best_score:
            best_score = sc_bl_full
            best_seq = a_bl.copy()
            print(f"  Block perturb improved: {sc_bl_full:.10f}")
        else:
            print(f"  Block perturb no improvement: {sc_bl_full:.10f}")

    # === Phase 4: Ultra-fine final polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Ultra-fine final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        ultra_betas = [
            0.00002 * max_cp,
            0.00001 * max_cp,
            0.000003 * max_cp,
            0.000001 * max_cp,
        ]

        polish_deadline = min(deadline - 5, time.monotonic() + 150.0)
        a_final, sc_final = _adam_logspace(
            a_pol, polish_deadline, ultra_betas,
            lr_initial=0.001, lr_final=0.0003, nesterov_momentum=0.7
        )
        sc_final_full = _score(a_final)
        if sc_final_full < best_score:
            best_score = sc_final_full
            best_seq = a_final.copy()
            print(f"  Final polish improved: {sc_final_full:.10f}")
        else:
            print(f"  Final polish no improvement: {sc_final_full:.10f}")

    # === Phase 5: Try a different length (8192) if time allows ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Try n=8192 ===")
        n_new = 8192
        # Interpolate parent to new length
        x0 = np.linspace(0, 1, len(best_seq))
        x1 = np.linspace(0, 1, n_new)
        a_interp = np.interp(x1, x0, best_seq)
        a_interp = np.maximum(a_interp, 1e-12)
        a_interp = a_interp / a_interp.sum()

        c_int = _full_convolve(a_interp, a_interp)
        max_ci = float(c_int.max())
        interp_betas = [0.001 * max_ci, 0.0003 * max_ci, 0.0001 * max_ci, 0.00003 * max_ci]

        interp_deadline = min(deadline - 5, time.monotonic() + 60.0)
        a_int, sc_int = _adam_logspace(
            a_interp, interp_deadline, interp_betas,
            lr_initial=0.003, lr_final=0.001, nesterov_momentum=0.85
        )
        sc_int_full = _score(a_int)
        if sc_int_full < best_score:
            best_score = sc_int_full
            best_seq = a_int.copy()
            print(f"  n=8192 improved! score={sc_int_full:.10f}")
        else:
            print(f"  n=8192 no improvement: {sc_int_full:.10f} vs {best_score:.10f}")

    # === Final validation ===
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096) / 4096
        # Scale to use full range (helps with the 1000 cap)
        final = final / final.sum()
        sc_final_check = _score(final)
        print(f"Final internal score: {sc_final_check:.10f}, n={len(final)}")
        print(f"Time used: {time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
