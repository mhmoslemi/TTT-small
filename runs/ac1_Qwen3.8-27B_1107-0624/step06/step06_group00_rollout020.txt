import numpy as np
import time
from scipy.signal import fftconvolve

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve_full(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve_full(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve_full(a, a)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _game_optimize(a_init, deadline, initial_eta=0.5, eta_decay=0.999,
                   lr_start=0.01, lr_end=0.001, max_iter=500000,
                   eval_interval=500, verbose=True):

    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    # Initialize q as uniform
    q_len = 2 * n - 1
    q = np.ones(q_len) / q_len
    
    # Log-space parameterization
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    # Adam buffers
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # a_rev for cross-correlation computation
    a_rev_full = np.zeros(3 * n - 1)  # We'll recompute each iteration since a changes
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        eta = initial_eta * (eta_decay ** t)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a = _softmax_stable(z)
        
        # Compute convolution
        c = _fft_convolve_full(a, a)  # length 2n-1
        
        # Compute w[j] = sum_k q[k] * a[k-j]
        # w = fftconvolve(q, a[::-1])[:n][::-1]
        a_rev = a[::-1]
        conv_q_a = _fft_convolve_full(q, a_rev)  # length (2n-1) + n - 1 = 3n-2
        w = conv_q_a[:n][::-1].copy()
        
        # C = sum(a * w)
        C = float(np.dot(a, w))
        
        # Gradient of C w.r.t. a (on the simplex, use centered gradient)
        g = 2.0 * (w - C)  # Since sum(a)=1, the Lagrange multiplier makes this centered
        # Actually, the gradient of C on the simplex is 2*w (the constraint sum(a)=1 
        # just means we project). But for the softmax parameterization, we use:
        # z_grad = a * (g - dot(a, g))
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # q[k] proportional to prod over time of exp(eta_t * c[k])
        # Equivalently: q[k] = exp(eta * c[k]) / sum(exp(eta * c)))
        # But this is a "fresh start" each iteration. For the multiplicative weights
        # approach, we should accumulate: log_q += eta * c, then normalize.
        
        # Actually, for the game-theoretic approach, the standard update is:

        # But this requires tracking log_q. Let's use the "exponential weights" form:
        # log_q[k] += eta * c[k]
        # q = exp(log_q) / sum(exp(log_q))
        
        # For efficiency, let's just recompute q from scratch each time:
        # q[k] proportional to exp(eta * c[k])
        eta_c = eta * c
        eta_c = eta_c - eta_c.max()
        q = np.exp(eta_c)
        q_sum = q.sum()
        if q_sum > 0:

        else:
            q = np.ones(q_len) / q_len

                best_score = sc
                best_a = a.copy()
                if verbose:
                    print(f"  iter {t}: score={sc:.10f}, eta={eta:.4f}")

    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

"""
Game-theoretic optimizer using multiplicative weights for minimizing
2*n*max(conv(a,a))/sum(a)^2. Combines adaptive-temperature soft gradients,
active-set polishing, coordinate descent, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.signal import fftconvolve


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve_full(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_raw(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve_full(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve_full(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _compute_conv(a):
    return _fft_convolve_full(a, a)


def _compute_soft_grad(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(np.clip(c_shifted, -700, 700))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    a_rev = a[::-1]
    conv_q_a = _fft_convolve_full(p, a_rev)
    w = conv_q_a[:n][::-1].copy()
    C = float(np.dot(a, w))
    g = 2.0 * (w - C)
    return g


def _compute_active_grad(a, c, top_k):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    a_rev = a[::-1]
    conv_q_a = _fft_convolve_full(p, a_rev)
    w = conv_q_a[:n][::-1].copy()
    C = float(np.dot(a, w))
    g = 2.0 * (w - C)
    return g


def _game_optimize(a_init, deadline, eta_start, eta_end, lr_start, lr_end,
                   max_iter, eval_interval=200, use_active=False, active_k=5,
                   clip_norm=1.0, verbose=True, label=""):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience = best_score
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        eta = eta_start * (max(eta_end, 1e-10) / eta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        T = 1.0 / max(eta, 1e-10)

        a = _softmax_stable(z)
        c = _compute_conv(a)

        if use_active and T < 1.0 / (eta_start * 0.1):
            g = _compute_active_grad(a, c, active_k)
        else:
            g = _compute_soft_grad(a, c, T)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > clip_norm:
            update = update * (clip_norm / un)

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                if verbose:
                    print(f"    {label} iter {t}: {sc:.10f} eta={eta:.4f}")
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=50, verbose=True):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        indices = rng.permutation(n)
        for i in indices:
            if time.monotonic() > deadline:
                break
            a_orig = a[i]
            if a_orig < 1e-15:
                continue
            # Try reducing this coordinate (it often helps reduce the peak)
            # Binary search for best value in [0, a_orig * 2]
            best_val = a_orig
            best_local = _score_norm(a)
            for frac in [0.0, 0.1, 0.3, 0.5, 0.7, 0.9, 1.2, 1.5, 2.0]:
                trial = a_orig * (1.0 + frac - 1.0)  # frac is multiplier
                # Actually let's try: a[i] * factor for factor in a set
                pass
            # Simpler: try a few multipliers
            for factor in [0.0, 0.3, 0.5, 0.7, 1.3, 1.7]:
                a[i] = a_orig * factor
                sc = _score_norm(a)
                if sc < best_local:
                    best_local = sc
                    best_val = a_orig * factor
            a[i] = best_val
            if best_local < _score_norm(best) - 1e-12:
                a = a / a.sum()
                sc2 = _score_norm(a)
                if sc2 < best_sc:
                    best_sc = sc2
                    best = a.copy()
                    improved = True
            else:
                a[i] = a_orig
        if not improved:
            break
        if verbose:
            print(f"    CD round {rnd}: {best_sc:.10f}")

    return best, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
