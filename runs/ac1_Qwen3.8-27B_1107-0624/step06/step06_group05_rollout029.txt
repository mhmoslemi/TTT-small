def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_val = rho[np.where(cond)[0][-1]]
    theta = (cssv[np.where(cond)[0][-1]] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: non-palindromic sequences avoid the Cauchy-Schwarz lower bound
on the center of self-convolution, allowing max(conv) < 1/n.
Uses log-sum-exp smoothing with adaptive temperature, FISTA, Adam, and L-BFGS.
Tries multiple initializations and sequence lengths.
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_arr(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_grad(a, beta):
    """Compute (obj, gradient, weights) for log-sum-exp of conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = cm
    else:
        w = e / s
        obj = cm + beta * np.log(s)
    a_rev = a[::-1]
    aw = _full_convolve(a_rev, w)
    grad = 2.0 * aw[n - 1:2 * n - 1]
    return obj, grad, w


def _fista_opt(a_init, beta_start, beta_end, deadline, max_iter):
    """FISTA on simplex with log-sum-exp, adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-30:
        a = a / s
    a_prev = a.copy()
    t = 1.0
    L = 6.0 * n
    step = 1.0 / L
    c0 = _full_convolve(a, a)
    best_score = 2.0 * n * float(c0.max())
    best_a = a.copy()
    max_c0 = float(c0.max())

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, max_iter * 0.7))
        beta = beta_start * (beta_end / beta_start) ** frac
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n
        obj, grad, w = _lse_grad(y, beta)
        a_next = _proj_simplex(y - step * grad)
        if it % 150 == 0:
            c = _full_convolve(a_next, a_next)
            sc = 2.0 * n * float(c.max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a.copy()
        a = a_next
        t = t_next

    c = _full_convolve(a, a)
    sc = 2.0 * n * float(c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta_start, beta_end, lr, max_iter):
    """Adam in log-space (softmax parameterization) with log-sum-exp."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = 2.0 * n * float(_full_convolve(a_init, a_init).max())
    current_lr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1
        a = _softmax_arr(z)
        frac = min(1.0, t_count / max(1, max_iter * 0.85))
        beta = beta_start * (beta_end / beta_start) ** frac
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        aw = _full_convolve(a_rev, w)
        grad_a = 2.0 * aw[n - 1:2 * n - 1]
        correction = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - correction)
        if not np.all(np.isfinite(grad_z)):
            break
        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_count)
        vh = v / (1.0 - b2 ** t_count)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 0.4:
            update = update * (0.4 / un)
        z = z - update
        z = np.clip(z, -38.0, 38.0)
        z = z - z.mean()
        if t_count % 80 == 0:
            a_chk = _softmax_arr(z)
            sc = 2.0 * n * float(_full_convolve(a_chk, a_chk).max())
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0 and it > 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_arr(z)
    sc = 2.0 * n * float(_full_convolve(a_final, a_final).max())
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_polish(a_init, deadline, beta):
    """L-BFGS-B in log-space."""
    from scipy.optimize import minimize
    n = len(a_init)

    def _obj_fn(z):
        a = _softmax_arr(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def _grad_fn(z):
        a = _softmax_arr(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        aw = _full_convolve(a_rev, w)
        grad_a = 2.0 * aw[n - 1:2 * n - 1]
        correction = float(np.dot(a, grad_a))
        return 2.0 * n * a * (grad_a - correction)

    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    max_time = deadline - time.monotonic() - 1.0
    if max_time <= 0.5:
        a_chk = a_init / a_init.sum()
        return a_chk, 2.0 * n * float(_full_convolve(a_chk, a_chk).max())
    try:
        result = minimize(_obj_fn, z0, jac=_grad_fn, method='L-BFGS-B',
                          options={'maxiter': max(30, int(max_time * 25)),
                                   'gtol': 1e-14, 'maxls': 25})
        z_opt = result.x
        a_opt = _softmax_arr(z_opt)
        sc = 2.0 * n * float(_full_convolve(a_opt, a_opt).max())
        return a_opt, sc
    except Exception:
        a_chk = a_init / a_init.sum()
        return a_chk, 2.0 * n * float(_full_convolve(a_chk, a_chk).max())

    elif kind == "sine_perturb":

def _subgradient_step(a, step):
    """One subgradient step on the simplex for max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    tol = 1e-6 * cm + 1e-15
    active = np.where(c >= cm - tol)[0]
    w = np.zeros_like(c)
    w[active] = 1.0 / len(active)
    a_rev = a[::-1]
    aw = _full_convolve(a_rev, w)
    grad = 2.0 * aw[n - 1:2 * n - 1]
    a_new = _proj_simplex(a - step * grad)
    return a_new, cm

            best_score = sc_f

            print(f"    FISTA: {sc_f:.10f}")

        # Phase B: Adam refinement
        if time.monotonic() < c_end * 0.9:
            c_ad = _full_convolve(a_f, a_f)
            max_cd = float(c_ad.max())
            b_s = 0.003 * max_cd
            b_e = 0.0001 * max_cd
            a_ad, _ = _adam_opt(a_f, c_end * 0.85, b_s, b_e, lr=0.004, max_iter=200000)
            sc_ad = _score(a_ad)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
                print(f"    Adam: {sc_ad:.10f}")
            a_c = a_ad
        else:
            a_c = a_f

        # Phase C: Subgradient polish (exact max)
        if time.monotonic() < c_end - 2:
            a_sg, _ = _subgradient_opt(a_c, c_end - 2, max_iter=50000)
            sc_sg = _score(a_sg)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"    Subgrad: {sc_sg:.10f}")
            a_c = a_sg

    # Final global polish on best found
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nFinal polish phase, current best: {best_score:.10f}")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        n_pol = len(a_pol)
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 5.0

        # Adam with very small beta
        if remaining > 30:
            b_s = 0.0005 * max_cp
            b_e = 0.00001 * max_cp
            a_ad, _ = _adam_opt(a_pol, time.monotonic() + remaining * 0.4,
                                b_s, b_e, lr=0.002, max_iter=300000)
            sc_ad = _score(a_ad)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
                print(f"  Final Adam: {sc_ad:.10f}")
            a_pol = a_ad

        # L-BFGS polish
        if time.monotonic() < deadline - 10:
            c_lb = _full_convolve(a_pol, a_pol)
            b_lb = 0.00005 * float(c_lb.max())
            a_lb, sc_lb = _lbfgs_polish(a_pol, deadline - 5.0, b_lb)
            if sc_lb < best_score:
                best_score = sc_lb
                best_seq = a_lb.copy()
                print(f"  L-BFGS: {sc_lb:.10f}")

        # Active set refinement
        if time.monotonic() < deadline - 8:
            a_as, sc_as = _refine_active_set(best_seq, deadline - 3.0, max_iter=20000)
            if sc_as < best_score:
                best_score = sc_as
                best_seq = a_as.copy()
                print(f"  Active set: {sc_as:.10f}")

    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_init(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_init(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
