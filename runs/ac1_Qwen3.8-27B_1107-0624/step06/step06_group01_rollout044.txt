Optimized multi-strategy search for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_hot(a, hot_indices, hot_weights):
    """Compute gradient of weighted sum of c[k] for k in hot_indices."""
    n = len(a)
    a_rev = a[::-1]
    # Build a vector w of length 2n-1 with weights at hot positions
    w = np.zeros(2 * n - 1)
    for idx, wt in zip(hot_indices, hot_weights):
        w[idx] = wt
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _nesterov_logspace(a_init, deadline, max_iter=200000):
    """Nesterov-accelerated optimization in log-space with dynamic hot-set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Nesterov momentum
    z_prev = z.copy()
    t_val = 1.0

    # Adam buffers
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    # Hot-set parameters
    c_curr = _full_convolve(a_curr if 'a_curr' in dir() else a_init, a_init)
    max_c = float(c_curr.max())

    # Adaptive step size
    step_size = 1.0 / max_c
    patience = 0
    best_patience_score = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = z + ((t_val - 1.0) / t_next) * (z - z_prev)
        a_y = _softmax(y)

        # Compute convolution and hot set
        c_y = _full_convolve(a_y, a_y)
        c_max = float(c_y.max())

        # Dynamic hot set: top-k lags adaptively
        # Start with broader set, narrow over time
        progress = min(1.0, it / 30000.0)
        if progress < 0.3:
            threshold = 0.7 * c_max
        elif progress < 0.6:
            threshold = 0.85 * c_max
        elif progress < 0.8:
            threshold = 0.95 * c_max
        else:
            threshold = 0.99 * c_max

        hot_mask = c_y >= threshold
        hot_indices = np.where(hot_mask)[0]
        if len(hot_indices) == 0:
            hot_indices = np.array([int(np.argmax(c_y))])
        # Cap hot set size for efficiency
        if len(hot_indices) > 256:
            top_vals = c_y[hot_indices]
            top_order = np.argsort(top_vals)[::-1][:256]
            hot_indices = hot_indices[top_order]

        # Normalize hot weights
        hot_vals = c_y[hot_indices]
        hot_weights = hot_vals / hot_vals.sum()

        # Compute gradient
        grad_a = _compute_grad_hot(a_y, hot_indices, hot_weights)

        # Convert to log-space gradient: grad_z = (diag(a) - outer(a,a)) @ grad_a
        grad_z = a_y * (grad_a - np.dot(a_y, grad_a))

        # Adam update
        t_adam += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_adam)
        vh = v_buf / (1.0 - b2 ** t_adam)

        # Adaptive learning rate
        lr = 0.5 * step_size / (np.sqrt(vh) + eps)
        lr = np.minimum(lr, 0.1)  # cap per-coordinate step
        update = -lr * mh
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        new_z = y + update
        new_z = np.clip(new_z, -30.0, 30.0)
        new_z = new_z - new_z.mean()

        # Evaluate
        a_new = _softmax(new_z)
        sc_new = _score_norm(a_new)

        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            # Backtrack: reduce step size
            if patience > 500:
                step_size *= 0.9
                patience = 0

        # Periodically increase step size if making steady progress
        if it % 5000 == 0 and patience == 0:
            step_size = min(step_size * 1.05, 5.0 / max_c)

        z_prev = z.copy()
        z = new_z
        t_val = t_next

    return best_a, best_score


def _multiplicative_weights(a_init, deadline, max_iter=100000):
    """Multiplicative weights / mirror descent for the zero-sum game formulation."""
    n = len(a_init)
    log_a = np.log(np.maximum(a_init, 1e-15))
    log_a = log_a - log_a.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Distribution over lags (the "dueling" strategy)
    w = np.ones(2 * n - 1) / (2 * n - 1)

    eta_a = 1.0
    eta_w = 0.5
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(log_a)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Update w (maximizer): concentrate on high c values
        log_w = np.log(np.maximum(w, 1e-300))
        log_w = log_w + eta_w * (c - c_max)
        log_w = log_w - log_w.max()
        w_new = np.exp(log_w)
        ws = w_new.sum()
        if ws < 1e-300:
            w_new = np.zeros_like(c)
            w_new[int(np.argmax(c))] = 1.0
        else:
            w_new = w_new / ws
        w = w_new

        # Update a (minimizer): reduce coordinates that contribute to high-c lags
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]

        log_a = log_a - eta_a * h
        log_a = np.clip(log_a, -30.0, 30.0)
        log_a = log_a - log_a.mean()

        # Evaluate periodically
        if it % 100 == 0:
            a_chk = _softmax(log_a)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                patience = 0
            else:
                patience += 1
                if patience > 50:
                    eta_a *= 0.9
                    eta_w *= 0.9
                    patience = 0

        # Gradually increase learning rates
        if it % 5000 == 0:
            eta_a = min(eta_a * 1.02, 3.0 / c_max)
            eta_w = min(eta_w * 1.02, 2.0 / c_max)

    a_final = _softmax(log_a)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_fine(a_init, deadline, betas=None):
    """Fine-grained Adam optimization with very small smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-10

    if betas is None:
        c0 = _full_convolve(a_init, a_init)
        max_c0 = float(c0.max())
        betas = [0.001 * max_c0, 0.0003 * max_c0, 0.0001 * max_c0]

    current_lr = 0.01

    for beta in betas:
        if time.monotonic() > deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)

        while time.monotonic() < phase_deadline:
            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            # Gradient
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 10000 == 0:
                current_lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _block_descent_polish(a_init, deadline, n_blocks=None, block_size=None):
    """Block coordinate descent: optimize subsets of coordinates jointly."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    rng = np.random.default_rng(123)

    if n_blocks is None:
        n_blocks = max(10, n // 64)
    if block_size is None:
        block_size = max(8, n // n_blocks)

    for iteration in range(10000):
        if time.monotonic() > deadline:
            break

        # Pick a random block
        start = rng.integers(0, n - block_size)
        end = start + block_size
        idx = slice(start, end)

        # Try several multiplicative adjustments to the block
        a_try = best_a.copy()
        for factor in [0.7, 0.8, 0.9, 1.1, 1.2, 1.3]:
            if time.monotonic() > deadline:
                break
            a_try = best_a.copy()
            a_try[idx] *= factor
            a_try = np.maximum(a_try, 1e-30)
            a_try = a_try / a_try.sum()
            sc = _score_norm(a_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()
                break

        # Also try shifting probability within the block
        if time.monotonic() > deadline:
            break
        a_try = best_a.copy()
        block_sum = a_try[idx].sum()
        if block_sum > 1e-15:
            # Try making block more uniform
            a_try[idx] = block_sum / block_size
            a_try = a_try / a_try.sum()
            sc = _score_norm(a_try)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_try.copy()

    return best_a, best_score


def _subgradient_polish(a_init, deadline, n_steps=5000):
    """Direct subgradient method targeting the max lag specifically."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    t = 0
    step_base = 1.0
    momentum = np.zeros(n)
    beta_mom = 0.9

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        c_max = c[k_star]

        # Subgradient of c[k_star] w.r.t. a
        # dc[k]/da[j] = a[k-j] if 0 <= k-j < n
        grad_a = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad_a[j] = 2.0 * a[k_j]

        # Also subtract the normalization constraint multiplier
        # For sum(a) = 1 constraint: subtract (grad_a . a) / n from each
        dot = np.dot(grad_a, a)
        grad_a = grad_a - dot / n

        # Convert to log-space
        grad_z = a * (grad_a - np.dot(a, grad_a))

        # Adaptive step size (Polak-Ribiere type)
        t += 1
        step = step_base / (t ** 0.5)
        step = min(step, 0.5 / max(np.max(np.abs(grad_z)), 1e-10))

        # Momentum
        momentum = beta_mom * momentum - step * grad_z

        z = z + momentum
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()

        if it % 50 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # Determine starting sequences
    candidates = []

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
        candidates.append((parent.copy(), n_parent, "parent"))

        # Interpolated to different lengths
        for n_try in [8192, 2048]:
            if n_try != n_parent:
                p_interp = _interp_profile(parent, n_try)
                sc_interp = _score(p_interp)
                print(f"  Interpolated n={n_try}: score={sc_interp:.10f}")
                candidates.append((p_interp, n_try, "interp"))
                if sc_interp < best_score:
                    best_score = sc_interp
                    best_seq = p_interp.copy()
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent. Uniform n={n_main}: score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
        candidates.append((a0, n_main, "uniform"))

    # Allocate time to strategies
    remaining = deadline - t_start
    print(f"\nTotal time budget: {budget_s}s, remaining: {remaining:.1f}s")

    # Strategy 1: Nesterov log-space on parent (40% of time)
    if time.monotonic() < deadline - 60:
        print(f"\n--- Strategy 1: Nesterov log-space on n={len(parent) if parent is not None else 4096} ---")
        a_work = (parent if parent is not None else best_seq).copy()
        a_work = a_work / a_work.sum()
        phase_deadline = t_start + remaining * 0.35
        a_opt, sc_opt = _nesterov_logspace(a_work, phase_deadline, max_iter=200000)
        sc_full = _score(a_opt)
        print(f"  Nesterov result: score={sc_full:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()

    # Strategy 2: Multiplicative weights (15% of time)
    if time.monotonic() < deadline - 45:
        print(f"\n--- Strategy 2: Multiplicative weights ---")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        phase_deadline = t_start + remaining * 0.50
        a_opt, sc_opt = _multiplicative_weights(a_work, phase_deadline, max_iter=100000)
        sc_full = _score(a_opt)
        print(f"  MW result: score={sc_full:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()

    # Strategy 3: Try n=8192 (20% of time)
    if time.monotonic() < deadline - 40:
        print(f"\n--- Strategy 3: n=8192 optimization ---")
        if parent is not None:
            a_8k = _interp_profile(parent, 8192)
        else:
            a_8k = np.ones(8192) / 8192
        sc_8k = _score(a_8k)
        print(f"  Start n=8192: score={sc_8k:.10f}")

        phase_deadline = min(deadline - 60, time.monotonic() + (remaining * 0.20))
        a_opt, sc_opt = _nesterov_logspace(a_8k, phase_deadline, max_iter=100000)
        sc_full = _score(a_opt)
        print(f"  n=8192 Nesterov: score={sc_full:.10f}, time={time.monotonic()-t_start:.1f}s")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()

    # Strategy 4: Fine Adam polish on best (15% of time)
    if time.monotonic() < deadline - 30:
        print(f"\n--- Strategy 4: Fine Adam polish ---")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        phase_deadline = min(deadline - 30, time.monotonic() + (remaining * 0.15))
        a_opt, sc_opt = _adam_fine(a_work, phase_deadline)
        sc_full = _score(a_opt)
        print(f"  Adam fine: score={sc_full:.10f}, time={time.monotonic()-t_start:.1f}s")

            best_seq = a_opt.copy()

    if time.monotonic() < deadline - 15:
        print(f"\n--- Strategy 5: Subgradient + Block descent polish ---")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        # Subgradient polish
        sg_deadline = min(deadline - 10, time.monotonic() + 30)
        a_opt, sc_opt = _subgradient_polish(a_work, sg_deadline, n_steps=5000)
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Subgradient: score={sc_full:.10f}")

        # Block descent
        if time.monotonic() < deadline - 5:
            bd_deadline = min(deadline - 5, time.monotonic() + 20)
            a_opt, sc_opt = _block_descent_polish(best_seq, bd_deadline)
            sc_full = _score(a_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_opt.copy()
                print(f"  Block descent: score={sc_full:.10f}")

    # Strategy 6: Multi-restart perturbation on best found
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n--- Strategy 6: Multi-restart perturbation ---")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        n_best = len(a_work)

        n_restarts = 3
        per_restart = max(5.0, (deadline - time.monotonic() - 10) / n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            if i == 0:
                p_start = _perturb(a_work, 0.01, rng)
            elif i == 1:
                # Shift
                shift = rng.integers(1, max(2, n_best // 2048))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb(a_work, 0.03, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas = [0.0005 * max_cs, 0.0001 * max_cs, 0.00003 * max_cs]
            a_opt, sc_opt = _adam_fine(p_start, time.monotonic() + per_restart, betas=betas)
            sc_full = _score(a_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_opt.copy()
                print(f"  Restart {i}: score={sc_full:.10f}")

    # Final validation and output
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        # Scale to a reasonable range (the score is scale-invariant,
        # but evaluator checks sum >= 0.01)
        if s > 1000.0:
            final = final * (500.0 / s)
        elif s < 1.0:
            final = final * (1.0 / s)

        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
