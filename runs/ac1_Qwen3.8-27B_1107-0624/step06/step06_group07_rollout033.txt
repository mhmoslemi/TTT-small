def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    n = len(f)

    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    A = scipy.linalg.toeplitz(first_col, first_row)
    return A

    A = _build_conv_matrix(f)

        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        return res.x

        return f, M
    b = np.maximum(b, 0)

    if s < 1e-15:

    return f_new, M
