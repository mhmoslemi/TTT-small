"""
Optimized step-function sequence search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact coordinate descent with analytical 1D critical points, Adam in log-space
with adaptive beta schedule, and multi-phase optimization. Focuses on n=4096 primary
with exploration of other lengths.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _exact_coord_descent(a, c, S, top_k, n_steps, deadline, rng, verbose=True):
    """
    Exact-ish coordinate descent. For each coordinate, finds the optimal t by
    checking analytical critical points of the top-k convolution entries.
    """
    n = len(a)
    a = a.copy()
    c = c.copy()
    best_score = _score_norm(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Get top-k indices of current convolution
        if step % 20 == 0 or step == 0:
            top_idx = np.argsort(c)[-top_k:]
        else:
            top_idx = np.argsort(c)[-top_k:]

        # Random ordering of coordinates
        order = rng.permutation(n)
        num_coords = min(n, 512 + 128 * (step % 8))

        improved = 0
        for ii in range(num_coords):
            i = order[ii % n]
            a_i = a[i]

            if a_i < 1e-15:
                continue

            # Compute d[j] = a[j-i] for valid j (0 <= j-i < n)
            # d has length 2n-1
            # Valid j range: max(0, i) to min(2n-2, i+n-1)
            j_start = max(0, i)
            j_end = min(2 * n - 1, i + n - 1)
            # d[j] = a[j - i] for j in [j_start, j_end]
            d = np.zeros(2 * n - 1)
            d[j_start:j_end + 1] = a[j_start - i:j_end - i + 1]

            # c'[j] = c[j] - a_i * d[j]
            c_prime = c - a_i * d

            S0 = S - a_i

            # Generate candidate t values
            candidates = [0.0, a_i, a_i * 0.5, a_i * 1.5, a_i * 0.25]

            # Critical points from top-k
            for j in top_idx:
                dj = d[j]
                if dj > 1e-15:
                    t_star = S0 - 2.0 * c_prime[j] / dj
                    if t_star > 0 and t_star < 50.0 * max(S0, 0.1):
                        candidates.append(t_star)
                        candidates.append(t_star * 0.8)
                        candidates.append(t_star * 1.2)
                        candidates.append(t_star * 0.5)

            # Filter and deduplicate
            candidates = [max(0.0, min(5.0 * S, tc)) for tc in candidates]
            candidates = sorted(set([round(tc, 12) for tc in candidates]))

            if len(candidates) > 100:
                # Keep the most promising: 0, a_i, and closest to a_i
                candidates = [0.0, a_i] + [tc for tc in candidates if tc != 0.0 and tc != a_i][:98]

            # Evaluate score at each candidate
            # Score(t) = 2n * max_j (c'[j] + t*d[j]) / (S0+t)^2
            # For efficiency, only check top_k + a few others
            check_idx = top_idx
            # Also add some random indices
            extra = rng.integers(0, 2 * n - 1, size=min(20, 2 * n - 1 - len(check_idx)))
            check_idx = np.concatenate([check_idx, extra])

            best_t = a_i
            best_val = np.inf

            for tc in candidates:
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                vals = (c_prime[check_idx] + tc * d[check_idx]) / denom
                val = 2.0 * n * float(vals.max())
                if val < best_val:
                    best_val = val
                    best_t = tc

            if abs(best_t - a_i) > 1e-14:
                # Update a[i], c, S
                delta = best_t - a_i
                a[i] = best_t
                S += delta
                c += delta * d
                # Prevent negative c (numerical)
                # c = np.maximum(c, 0)  # skip for speed

                if best_val < best_score:
                    best_score = best_val
                    improved += 1

        if verbose and step % 50 == 0:
            actual_score = _score_norm(a)
            print(f"    CD step {step}: approx_score={best_score:.10f}, actual={actual_score:.10f}")
            if actual_score < best_score:
                best_score = actual_score

    # Recompute exact c
    c = _full_convolve(a, a)
    S = float(a.sum())
    return a, c, S, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter, restart_every=5000):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

            # Periodic restart
            if it_in_phase % restart_every == 0:
                m_buf *= 0.3
                v_buf *= 0.3

        # Reset at phase boundaries
        m_buf *= 0.3
        v_buf *= 0.3
        current_lr = max(0.001, current_lr * 1.2)

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    """FISTA on simplex with top-K softened max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    elif kind == "gaussian_centered":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        betas_fine = betas_fine[:10]

    a_opt, sc_opt = _adam_logspace(a_work, p3_end, betas_fine,
                                   lr_init=0.008, top_k=80,
                                   max_iter=100000, restart_every=4000)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 4: More CD on the Adam result
    if time.monotonic() < deadline - 60:
        p4_end = min(deadline - 60, time.monotonic() + 120)
        print(f"\n  Phase 4: CD polish (budget: {p4_end - time.monotonic():.1f}s)")
        c_work = _full_convolve(a_opt, a_opt)
        S_work = float(a_opt.sum())
        a_cd2, c_work, S_work, _ = _exact_coord_descent(
            a_opt, c_work, S_work, top_k_cd + 20,
            n_steps=10000, deadline=p4_end, rng=rng, verbose=True
        )
        a_cd2 = a_cd2 / a_cd2.sum()
        sc_cd2 = _score(a_cd2)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_seq = a_cd2.copy()
            print(f"  CD polish: score={sc_cd2:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Ultra-fine Adam
    if time.monotonic() < deadline - 30:
        p5_end = min(deadline - 20, time.monotonic() + 60)
        print(f"\n  Phase 5: Ultra-fine Adam (budget: {p5_end - time.monotonic():.1f}s)")
        c_work = _full_convolve(a_opt, a_opt)
        max_cw = float(c_work.max())
        betas_ult = [0.0003 * max_cw, 0.0001 * max_cw, 0.00005 * max_cw]
        a_ult, sc_ult = _adam_logspace(a_opt, p5_end, betas_ult,
                                       lr_init=0.002, top_k=50,
                                       max_iter=50000, restart_every=3000)
        sc_ult_full = _score(a_ult)
        if sc_ult_full < best_score:
            best_score = sc_ult_full
            best_seq = a_ult.copy()
            print(f"  Ultra-fine: score={sc_ult_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Try interpolated/other lengths
    if time.monotonic() < deadline - 90:
        print(f"\n=== Phase 6: Other lengths (remaining: {deadline - time.monotonic():.1f}s) ===")
        other_lengths = [2048, 8192]
        n_best = len(best_seq) if best_seq is not None else 4096
        for ol in other_lengths:
            if time.monotonic() > deadline - 60:
                break
            if ol == n_best:
                continue
            a_ol = _interp_profile(best_seq, ol)
            sc_ol = _score(a_ol)
            print(f"  n={ol} interp: score={sc_ol:.10f}")
            if sc_ol < best_score:
                best_score = sc_ol
                best_seq = a_ol.copy()

            # Quick CD on this length
            c_ol = _full_convolve(a_ol, a_ol)
            S_ol = float(a_ol.sum())
            ol_end = min(time.monotonic() + min(60, (deadline - time.monotonic()) - 30),
                         deadline - 30)
            a_ol, c_ol, S_ol, _ = _exact_coord_descent(
                a_ol, c_ol, S_ol, 60,
                n_steps=5000, deadline=ol_end, rng=rng, verbose=False
            )
            a_ol = a_ol / a_ol.sum()
            sc_ol2 = _score(a_ol)
            if sc_ol2 < best_score:
                best_score = sc_ol2
                best_seq = a_ol.copy()
                print(f"  n={ol} after CD: score={sc_ol2:.10f}")

    # Phase 7: Final restarts from best
    if time.monotonic() < deadline - 40 and best_seq is not None:
        print(f"\n=== Phase 7: Restarts (remaining: {deadline - time.monotonic():.1f}s) ===")
        a_base = best_seq.copy() / best_seq.sum()
        n_best = len(a_base)
        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 20) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            # Perturb base
            noise_scale = 0.01 + 0.005 * ri
            p_start = a_base + noise_scale * rng.standard_normal(n_best)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs]

            p_end = time.monotonic() + max(5, per_restart - 5)
            p_best, sc_best = _adam_logspace(p_start, p_end, betas_r,
                                             lr_init=0.005, top_k=60,
                                             max_iter=30000, restart_every=3000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Also validate with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence score: {ev_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
