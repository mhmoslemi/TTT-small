import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _convolve_complex(a, N):
    """Zero-pad a to length N, compute FFT, square it, IFFT to get convolution."""
    a_pad = np.zeros(N, dtype=np.complex128)
    a_pad[:len(a)] = a
    A = np.fft.fft(a_pad)
    P = A * A
    return np.fft.ifft(P).real, A

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _spectral_flatten_step(a, N, alpha):
    """One step of spectral flattening."""
    n = len(a)
    # Compute convolution via FFT
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)
    P = A * A
    c = np.fft.ifft(P).real
    c = c[:2*n-1]  # Linear convolution part
    c = np.maximum(c, 0)  # Should be non-negative for non-negative a
    
    c_max = c.max()
    if c_max < 1e-30:
        return a.copy()
    
    # Power compression
    c_norm = c / c_max
    c_shaped = c_max * (c_norm ** alpha)
    
    # Compute new spectrum
    c_shaped_pad = np.zeros(N, dtype=np.float64)
    c_shaped_pad[:2*n-1] = c_shaped
    P_new = np.fft.fft(c_shaped_pad)
    
    # Square root
    A_new = np.sqrt(P_new + 1e-30 * np.ones(N, dtype=np.complex128))
    
    # New time-domain
    a_new = np.fft.ifft(A_new).real[:n]
    
    # Project to non-negative
    a_new = np.maximum(a_new, 0)
    
    # Normalize
    s = a_new.sum()
    if s > 1e-30:
        a_new = a_new / s
    
    return a_new

def _spectral_flatten(a, alphas, iterations_per_alpha, deadline):
    """Multi-stage spectral flattening with decreasing alpha."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_s = _score_norm(a_curr)
    
    for alpha in alphas:
        for it in range(iterations_per_alpha):
            if time.monotonic() > deadline:
                break
            a_new = _spectral_flatten_step(a_curr, N, alpha)
            s = a_new.sum()
            if s < 1e-30:
                break
            sc = _score_norm(a_new)
            if sc < best_s:
                best_s = sc
                best_a = a_new.copy()
            # Use a_curr (not best) for the next iteration to explore
            a_curr = a_new
        if time.monotonic() > deadline:
            break
    
    return best_a, best_s

"""
Spectral flattening + gradient optimization for minimizing the peak
self-convolution ratio of non-negative sequences.
Core innovation: Gerchberg-Saxton style iterative spectral shaping to
flatten the autocorrelation, combined with FISTA/Adam gradient methods.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming a is normalized to sum=1."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _convolve_fft(a, N):
    """Compute linear convolution a*a using FFT of length N."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)
    P = A * A
    c = np.fft.ifft(P).real[:2*n-1]
    return c, A


def _spectral_step(a, N, alpha, lam=1.0):
    """One GS-style spectral flattening step.
    
    Parameters:
    - a: non-negative sequence, sum = 1
    - N: FFT length >= 2*len(a)-1
    - alpha: power exponent for shaping (0 < alpha <= 1)
    - lam: relaxation parameter (0 < lam <= 1)
    
    Returns: new non-negative normalized sequence
    """
    n = len(a)
    c, A = _convolve_fft(a, N)
    c = np.maximum(c, 0.0)
    
    c_max = float(c.max())
    if c_max < 1e-30:
        return a.copy()
    
    # Power compression: raise small values, keep peak
    c_norm = c / c_max
    c_shaped = c_max * np.power(c_norm, alpha)
    
    # Compute new spectrum via FFT of shaped convolution
    c_shaped_pad = np.zeros(N)
    c_shaped_pad[:2*n-1] = c_shaped
    P_new = np.fft.fft(c_shaped_pad)
    
    # Square root (principal branch)
    # Add small regularization to avoid exact zeros
    P_reg = P_new + 1e-30 * np.ones(N, dtype=np.complex128)
    A_new = np.sqrt(P_reg)
    
    # Inverse FFT to get time domain
    a_candidate = np.fft.ifft(A_new).real[:n]
    
    # Relaxation
    if lam < 1.0:
        a_candidate = (1.0 - lam) * a + lam * a_candidate
    
    # Project to non-negative
    a_new = np.maximum(a_candidate, 0.0)
    
    # Normalize
    s = a_new.sum()
    if s > 1e-30:
        a_new = a_new / s
    else:
        a_new = a.copy()
    
    return a_new


def _spectral_flatten(a, deadline, alphas, iters_per_alpha, lam=0.8):
    """Run spectral flattening with a schedule of alpha values."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_sc = _score_normalized(a_curr)
    
    for alpha in alphas:
        for it in range(iters_per_alpha):
            if time.monotonic() > deadline:
                break
            a_new = _spectral_step(a_curr, N, alpha, lam)
            sc = _score_normalized(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a_curr = a_new
        if time.monotonic() > deadline:
            break
    
    return best_a, best_sc


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
