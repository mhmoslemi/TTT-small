import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max_conv."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    # Standard projection onto simplex {x: x >= 0, sum(x) = target_sum}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.ones(n) * (target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

"""
Optimized step function searcher using FISTA with smoothed max, 
active-set gradient refinement, and QP-based polishing.
Starts from parent sequence and tries to improve the score.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_phase(a_init, beta, deadline, max_iter=50000):
    """FISTA with softmax-smoothed max for a fixed beta."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    best_a = a_curr.copy()
    best_score = float(2.0 * n * max_c)
    no_improve = 0
    patience = 2000
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_curr + ((t - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        max_cy = float(c_y.max())
        shifted = (c_y - max_cy) / beta
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _compute_grad(y, w)
        
        # Ensure gradient is finite
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        
        a_next = _proj_simplex(y - step * grad)
        
        # Periodic check
        if it % 500 == 0 or it < 5:
            sc = float(2.0 * n * _full_convolve(a_next, a_next).max())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_next.copy()
                no_improve = 0
            else:
                no_improve += 500
                if no_improve > patience:
                    # Reduce step and reset
                    step *= 0.5
                    t = 1.0
                    a_prev = a_curr.copy()
                    no_improve = 0
        
        # Update L periodically
        if it % 1000 == 0:
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())

    if sc_final < best_score:

        best_a = a_curr.copy()
    
    return best_a, best_score

def _interp_to_length(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
