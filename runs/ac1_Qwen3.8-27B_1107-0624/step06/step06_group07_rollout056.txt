import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = c.max()

    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

        if t % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                best_z = z.copy()

        if t % 2000 == 0 and t > 0:
            lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=20000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad
        z = np.clip(z, -85.0, 85.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.75
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        g = _gradient_active(a, c, n, tol_frac)
        g = g - g.mean()

        if np.linalg.norm(g) < 1e-15:
            break

        a_try = a - step_size * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s

        sc_cur = _score(a)
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        if sc_try < sc_cur - 1e-15:
            a = a_try
            step_size = min(step_size * 1.003, 0.05)
        else:
            stall += 1
            step_size *= 0.9

        step_size *= (1.0 - 5e-6)
        step_size = max(step_size, 1e-12)

        if stall > 500:
            break

    return best_a, best_sc


def _perturb_small(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 6)))
        amp = 0.003 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(8, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.3, 1.5)))
        z[start:end] += rng.uniform(-0.01, 0.01)
    z = z - z.mean()
    return _softmax(z)


def _peak_targeted_search(a, deadline, max_rounds=2000):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(best_a, best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = c.max()
        peak_idx = int(np.argmax(c))

        grad = _gradient_active(best_a, c, n, tol_frac=1e-5)
        grad_centered = grad - grad.mean()

        for alpha in [0.002, 0.005, 0.01]:
            a_try = best_a - alpha * grad_centered
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try
                best_sc = sc_try
                break

    return best_a, best_sc


def _local_search(a, deadline, max_rounds=3000):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.003
    rng = np.random.default_rng(777)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        for delta_frac in [0.2, 0.4, 0.6, 0.8]:
            s_pair = best_a[idx] + best_a[jdx]
            if s_pair < 1e-15:
                continue
            new_val_i = s_pair * (1.0 - delta_frac) + step
            new_val_j = s_pair * delta_frac - step
            if new_val_i < 0 or new_val_j < 0:
                continue

            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True
                break

        if not improved and rnd > 100:
            step *= 0.8
            if step < 1e-7:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(123)
        a = np.abs(rng.standard_normal(n))
        a = a / a.sum()
        return a
    elif kind == "random_smooth":
        rng = np.random.default_rng(456)
        a = np.abs(rng.standard_normal(n))
        kernel = np.ones(16) / 16
        a = np.convolve(a, kernel, 'same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer combining L-BFGS with aggressive beta-continuation,
    Adam polishing, subgradient methods, peak-targeted search, and diverse
    multi-restart to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Main L-BFGS with aggressive beta schedule ===
    print(f"\n=== Phase 1: Main L-BFGS aggressive (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _adam_lbfgs_schedule(best_seq, deadline, min_time_remaining=400.0, label="main")
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Adam optimization ===
    print(f"\n=== Phase 2: Adam optimization (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_ad, sc_ad = _adam_optimize(best_seq, deadline - 200, max_steps=30000, lr_init=0.005)
        try_update(a_ad, "phase2_adam")

    # === Phase 3: Subgradient z-space ===
    print(f"\n=== Phase 3: Subgradient z-space (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, _ = _subgrad_z_polish(best_seq, deadline - 150,
                                    tol_frac=1e-4, max_steps=15000, lr_init=0.03)
        try_update(a_sg, "phase3_sg_z")

    # === Phase 4: Direct subgradient ===
    print(f"\n=== Phase 4: Direct subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_sg2, _ = _subgrad_direct_polish(best_seq, deadline - 100,
                                          tol_frac=1e-5, max_steps=40000)
        try_update(a_sg2, "phase4_sg_direct")

    # === Phase 5: Peak-targeted search ===
    print(f"\n=== Phase 5: Peak-targeted (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_pt, _ = _peak_targeted_search(best_seq, deadline - 80, max_rounds=3000)
        try_update(a_pt, "phase5_peak")

    # === Phase 6: Local search ===
    print(f"\n=== Phase 6: Local search (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_ls, _ = _local_search(best_seq, deadline - 60, max_rounds=5000)
        try_update(a_ls, "phase6_local")

    # === Phase 7: Fine L-BFGS ===
    print(f"\n=== Phase 7: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 30:
                break
            remaining = deadline - 30 - time.monotonic()
            if remaining < 15:
                break
            max_iter = min(8000, max(500, int(remaining / 0.003)))
            a_new, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 8: Multi-restart ===
    print(f"\n=== Phase 8: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("g_001", "gauss", 0.001, 0),
            ("g_003", "gauss", 0.003, 0),
            ("g_010", "gauss", 0.010, 0),
            ("f_2", "fourier", 0, 2),
            ("f_5", "fourier", 0, 5),
            ("b_4", "block", 0, 4),
            ("b_12", "block", 0, 12),
            ("mix1", "mix", 0.005, 3),
        ]

        for ri, (name, ptype, sigma, modes) in enumerate(restart_configs):
            if time.monotonic() > deadline - 25:
                break

            r = np.random.default_rng(int(seed) * 100 + ri * 7)
            if ptype == "gauss":
                p_start = _perturb_small(best_seq, sigma, r)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, r, modes)
            elif ptype == "block":
                p_start = _perturb_block(best_seq, r, modes)
            else:
                p_start = _perturb_small(best_seq, sigma, r)
                p_start = _perturb_fourier(p_start, r, modes)

            sc_start = _score(p_start)
            if sc_start > best_score_val + 0.02:
                continue
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            a_r = _adam_lbfgs_schedule(p_start, deadline, min_time_remaining=25.0,
                                       label=name)
            try_update(a_r, f"restart_{name}")

            if time.monotonic() < deadline - 25 and _score(a_r) < best_score_val + 0.003:
                a_r2, _ = _subgrad_z_polish(a_r, deadline - 25,
                                            tol_frac=1e-4, max_steps=5000, lr_init=0.02)
                try_update(a_r2, f"restart_{name}_sg")

    # === Phase 9: Try different lengths ===
    print(f"\n=== Phase 9: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        current_n = len(best_seq)
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 18:
                break
            if target_n == current_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _adam_lbfgs_schedule(a_interp, deadline, min_time_remaining=18.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 10: Final ultra-fine polish ===
    print(f"\n=== Phase 10: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline:
        for rnd in range(5):
            if time.monotonic() > deadline - 1:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            a_f, _ = _lbfgs_run(best_seq, beta, n, 5000)
            try_update(a_f, f"final_fine_{rnd}")

            if time.monotonic() < deadline - 1:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 1,
                                           tol_frac=1e-5, max_steps=8000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

            if time.monotonic() < deadline - 1:
                a_p, _ = _peak_targeted_search(best_seq, deadline - 1, max_rounds=2000)
                try_update(a_p, f"final_peak_{rnd}")

            if time.monotonic() < deadline - 1:
                a_l, _ = _local_search(best_seq, deadline - 1, max_rounds=3000)
                try_update(a_l, f"final_ls_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
