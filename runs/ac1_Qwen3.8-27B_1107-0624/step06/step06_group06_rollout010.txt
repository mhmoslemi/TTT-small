"""
Time-domain coordinate descent with exact 1D line search, combined with
L-BFGS log-sum-exp smoothing and active-set subgradient polishing,
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations: exact per-coordinate 1D optimization (no smoothing needed),
randomized coordinate ordering, 2-coordinate block updates, and adaptive
method switching based on convergence behavior.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(z, a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(z, a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.85
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _coord_descent_1d(a, deadline, n_passes=200, rng=None):
    """
    Cyclic (or randomized) coordinate descent with exact 1D line search.
    For each coordinate j, minimize g(t) = max_k b_k(t) / (S+t)^2 over t>=0,
    where b_k(t) is the convolution with a_j replaced by t.

            # b_k = c_k + t^2 for k = 2j

            # if k != 2j: 2 * t0 * a[k-j] (if 0<=k-j<n)

            # For k = 2j: only one term (i=j=k-j), total = a_j^2

                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = obj_of_t(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = obj_of_t(d_gs)

            t_new = (a_gs + b_gs) / 2.0
            t_new = max(0.0, t_new)

            new_score_t = obj_of_t(t_new)

                    # Update b

                        kj = int(k) - j
                        if 0 <= kj < n:
                            if k == 2 * j:

                            else:
                                b[k] += 2.0 * delta * a[kj]

                    S_total += delta

                    new_s = float(a.sum())
                    if new_s < 1e-15:
                        a[j] = t0
                        S_total -= delta
                        # revert b
                        delta = -delta
                        for k in ks:

                            if 0 <= kj < n:

            if p > 20:

    """
    2-coordinate block descent: pick two random coordinates, do a 2D grid search.
    Helps escape 1D local minima.
    """
    n = len(a)
    if rng is None:
        rng = np.random.default_rng()
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        j = rng.integers(0, n)
        m = rng.integers(0, n)
        if j == m:
            m = (j + 1) % n

        t0j = a[j]
        t0m = a[m]
        S = float(a.sum()) - t0j - t0m

        if S < 1e-15:
            continue

        # 2D grid search
        # Search over (tj, tm) in a small neighborhood
        scale = max(t0j, t0m, 0.01 / n)
        deltas_j = [-0.5, -0.2, -0.1, 0.0, 0.1, 0.2, 0.5]
        deltas_m = [-0.5, -0.2, -0.1, 0.0, 0.1, 0.2, 0.5]

        best_tj, best_tm = t0j, t0m
        best_val = np.inf

        for dj in deltas_j:
            tj = t0j + dj * scale
            if tj < 0:
                continue
            for dm in deltas_m:
                tm = t0m + dm * scale
                if tm < 0:
                    continue
                a[j] = tj
                a[m] = tm
                s = float(a.sum())
                if s < 1e-10:
                    continue
                c = _full_convolve(a, a)
                sc = float(2.0 * n * c.max() / (s * s))
                if sc < best_val:
                    best_val = sc
                    best_tj, best_tm = tj, tm

        a[j] = best_tj
        a[m] = best_tm
        sc = _score(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _run_lbfgs_refined(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation (improved version)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)] if len(c_sorted) > 5 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 10))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _multiplicative_update(a_init, deadline, n_iters=200, eta_init=0.1):
    """
    Multiplicative weight update for the log-sum-exp smoothed objective.
    Keeps a >= 0 automatically.
    """
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_a = a.copy()
    best_sc = _score(a)
    eta = eta_init

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        beta = c_max * 0.01
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s_e = e.sum()
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        # Gradient of smoothed max w.r.t. a_j
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Normalize gradient for stable updates
        g_max = np.abs(grad).max()
        if g_max < 1e-15:
            break
        grad_normed = grad / g_max

        # Multiplicative update
        a = a * np.exp(-eta * grad_normed)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
            break
        a = a / s

        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            eta = min(eta * 1.02, 0.5)
        else:
            eta *= 0.9
            if eta < 0.001:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer: coordinate descent (exact 1D) + L-BFGS + 
    multiplicative updates + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
                update_best(parent, _score(parent), "parent")
    except Exception:
        pass

    if parent is None:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === PHASE 1: Coordinate descent (main workhorse) ===
    print(f"\n[Phase 1] Coordinate descent 1D (t={time.monotonic() - t_start:.1f}s)")
    cd_deadline = min(deadline - 300, t_start + budget_s * 0.45)
    a_cd, sc_cd = _coord_descent_1d(parent, cd_deadline, n_passes=100, rng=rng)
    update_best(a_cd, sc_cd, "coord_1d")

    # === PHASE 2: L-BFGS refinement on coordinate descent result ===
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = min(deadline - 200, time.monotonic() + budget_s * 0.25)
    a_lbfgs = _run_lbfgs_refined(best_seq, lbfgs_deadline, min_time_remaining=100.0)
    update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_p1")

    # === PHASE 3: More coordinate descent from improved point ===
    print(f"\n[Phase 3] Coordinate descent round 2 (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        cd2_deadline = min(deadline - 100, time.monotonic() + budget_s * 0.2)
        a_cd2, sc_cd2 = _coord_descent_1d(best_seq, cd2_deadline, n_passes=80, rng=rng)
        update_best(a_cd2, sc_cd2, "coord_1d_r2")

    # === PHASE 4: Multiplicative update ===
    print(f"\n[Phase 4] Multiplicative update (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        mu_deadline = min(deadline - 80, time.monotonic() + 120)
        a_mu, sc_mu = _multiplicative_update(best_seq, mu_deadline, n_iters=300, eta_init=0.05)
        update_best(a_mu, sc_mu, "mult_update")

    # === PHASE 5: 2-coordinate block descent ===
    print(f"\n[Phase 5] 2D block descent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        b2_deadline = min(deadline - 60, time.monotonic() + 100)
        a_2d, sc_2d = _coord_descent_2d(best_seq, b2_deadline, n_iters=100, rng=rng)
        update_best(a_2d, sc_2d, "block_2d")

    # === PHASE 6: Subgradient polish ===
    print(f"\n[Phase 6] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        sg_deadline = min(deadline - 40, time.monotonic() + 80)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=3000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # === PHASE 7: Multi-n exploration ===
    print(f"\n[Phase 7] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 5120]
    if n_parent not in n_candidates and n_parent not in [1024, 4096]:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 50:
            break
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        else:
            a_init = parent.copy()

        # Quick coord descent + L-BFGS
        if time.monotonic() < deadline - 30:
            cd_dl = min(deadline - 20, time.monotonic() + 60)
            a_opt, sc_opt = _coord_descent_1d(a_init, cd_dl, n_passes=30, rng=rng)
            update_best(a_opt, sc_opt, f"n_{n_try}_cd")

            if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 15:
                a_lb = _run_lbfgs_refined(a_opt, min(deadline - 10, time.monotonic() + 30),
                                          min_time_remaining=10.0)
                update_best(a_lb, _score(a_lb), f"n_{n_try}_lbfgs")

    # === PHASE 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for i in range(3):
            if time.monotonic() > deadline - 10:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)
            else:
                p_start = _perturb(best_seq, 0.08, rng)

            sc_start = _score(p_start)
            # Quick coord descent
            a_r, sc_r = _coord_descent_1d(p_start, min(deadline - 5, time.monotonic() + 40),
                                          n_passes=20, rng=rng)
            update_best(a_r, sc_r, f"restart_{i}")

    # === PHASE 9: Final L-BFGS fine polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin = _run_lbfgs_refined(best_seq, deadline - 1, min_time_remaining=2.0)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Final coord descent
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _coord_descent_1d(best_seq, deadline - 0.5, n_passes=10, rng=rng)
            update_best(a_fin2, sc_fin2, "final_cd")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
