def _coord_descent(a_init, deadline, max_rounds=100):

    #   change = 2 * delta * a_{k-j}

    # c_k_rest = c_k - (terms involving a_j)

    from scipy.optimize import minimize
    n = len(a_init)

        return _soft_value(a, beta)

        a = _softmax(z)
        g = _soft_grad(a, beta)
        return a * (g - float(a @ g))
    
    max_time = deadline - time.monotonic() - 2.0

        return a_init
    
    try:
        result = minimize(
            obj, z0, jac=grad, method='L-BFGS-B',

    return _softmax(best_z)
