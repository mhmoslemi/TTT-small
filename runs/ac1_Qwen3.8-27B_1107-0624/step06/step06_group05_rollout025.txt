        # Auto temperature: gap between max and (top_k+1)-th largest
        sorted_c = np.sort(c)[::-1]
        if len(sorted_c) > top_k:
            temperature = max(1e-12, (c_max - sorted_c[top_k]) * 0.5)
        else:
            temperature = c_max * 0.01
    w = np.exp((c - c_max) / temperature)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    grad = _compute_grad(a, w)
    return grad, c, c_max


def _fista_opt(a_init, beta, deadline, max_iter):
    """FISTA on the smoothed objective: 2n * logsumexp(conv/beta - max/beta)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 * n
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _subgrad_line_search(a_init, deadline, max_iter=50000, top_k=8, momentum=0.9):
    """
    Subgradient method with backtracking line search and momentum.
    Uses soft active set with decreasing temperature.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c_curr = _full_convolve(a, a)
    c_max_0 = float(c_curr.max())
    
    # Temperature schedule: start moderate, decrease over iterations
    temp_start = 0.05 * c_max_0
    temp_end = 0.001 * c_max_0
    
    v = np.zeros(n)  # momentum
    base_step = 1.0 / (4.0 * n * c_max_0)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Decay temperature
        frac = min(1.0, it / (max_iter * 0.8))
        temperature = temp_start + frac * (temp_end - temp_start)
        
        # Compute subgradient
        c = _full_convolve(a, a)
        c_max = float(c.max())
        w = np.exp((c - c_max) / temperature)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(a, w)
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm < 1e-15:
            break
        
        # Momentum update
        v = momentum * v - grad
        v_norm = float(np.linalg.norm(v))
        if v_norm < 1e-15:
            break
        
        # Backtracking line search
        step = base_step
        current_score = _score_norm(a)
        found = False
        for ls_iter in range(20):
            a_trial = _proj_simplex(a + step * v)
            sc_trial = _score_norm(a_trial)
            if sc_trial < current_score - 1e-12:
                found = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        
        if found:
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a.copy()
        else:
            # Try a smaller effective step with the subgradient directly
            a_trial = _proj_simplex(a - step * grad / grad_norm)
            sc_trial = _score_norm(a_trial)
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a.copy()
                a = a_trial
            else:
                # Decay momentum to avoid stalling
                v = 0.5 * v
    
    return best_a, best_score


def _peak_shave(a_init, deadline, max_swaps=200000):
    """
    Peak shaving via coordinate swaps: identify the peak convolution position,
    then swap mass from high-contributing coordinates to low-contributing ones.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    swaps_made = 0
    stall_count = 0
    max_stall = 5000
    
    for it in range(max_swaps):
        if time.monotonic() > deadline:
            break
        if swap_shave_one(a):
            swaps_made += 1
            stall_count = 0
            # Check score periodically
            if swaps_made % 100 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                elif sc > best_score + 1e-6:
                    # Revert to best
                    a = best_a.copy()
        else:
            stall_count += 1
            if stall_count > max_stall:
                break
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def swap_shave_one(a):
    """Attempt one peak-shaving swap. Returns True if a swap was made."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max_pos = int(np.argmax(c))
    c_max = float(c[c_max_pos])
    
    # For peak position k*, the contribution to conv from coordinate i is a[i]*a[k*-i]
    # We want to reduce this. Find the coordinate i with the largest product a[i]*a[k*-i]
    # and swap some mass to a coordinate j with small a[j]*a[k*-j].
    
    # Compute contributions: for each i, contribution_i = a[i] * a[k*-i] (if valid)
    k = c_max_pos
    # Valid range of i: need 0 <= i < n and 0 <= k-i < n
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    
    if i_max < i_min:
        return False
    
    # Contribution of each valid i to the peak
    contributions = np.zeros(n)
    valid_indices = np.arange(i_min, i_max + 1)
    contributions[valid_indices] = a[valid_indices] * a[k - valid_indices]
    
    # Find the coordinate with the largest contribution (and non-negligible a[i])
    # We want to decrease a[i] where contribution is high
    # And increase a[j] where a[j]*a[k-j] is low
    total_contrib = float(contributions.sum())
    if total_contrib < 1e-15:
        return False
    
    # Score for "source" (to decrease): high contribution, high a[i]
    source_score = contributions.copy()
    # Score for "target" (to increase): low contribution at peak, but we also care
    # about not creating new peaks elsewhere
    target_score = -np.abs(a)  # prefer moving to coordinates with small a
    
    # Pick source: top contributor with significant mass
    source_candidates = np.where(contributions > 0.05 * total_contrib)[0]
    if len(source_candidates) == 0:
        return False
    
    # Weight by contribution and pick
    src_weights = contributions[source_candidates]
    src_total = src_weights.sum()
    if src_total < 1e-15:
        return False
    src_probs = src_weights / src_total
    i = source_candidates[np.random.choice(len(source_candidates), p=src_probs)]
    
    # Pick target: avoid the same position, prefer low contribution at peak
    # and also check that increasing a[j] won't create a bigger peak elsewhere
    # Simple heuristic: pick j that minimizes a[j]*a[k-j] (contribution to peak)
    # and also minimizes the "worst case" increase at other positions
    
    # For efficiency, just pick from a random subset of low-contribution positions
    low_contrib = np.where(contributions < 0.01 * total_contrib)[0]
    low_contrib = low_contrib[low_contrib != i]
    if len(low_contrib) == 0:
        # Fallback: any position other than i
        low_contrib = np.array([j for j in range(n) if j != i])
    
    # Pick a few random candidates and evaluate
    n_try = min(5, len(low_contrib))
    candidates = np.random.choice(low_contrib, size=n_try, replace=False)
    
    # Determine swap amount: small fraction of a[i]
    delta = 0.01 * a[i]
    if delta < 1e-10:
        return False
    
    best_j = -1
    best_new_score = np.inf
    current_score = float(2.0 * n * c_max)
    
    for j in candidates:
        a[i] -= delta
        a[j] += delta
        # Quick check: recompute conv at the peak position
        # conv_k = sum_m a[m]*a[k-m]
        # Change: -delta*a[k-i] + delta*a[k-j] (approximately, ignoring the cross term)
        # Actually the change is: (a[i]-delta)*a[k-i] + a[j]+delta)*a[k-j] - a[i]*a[k-i] - a[j]*a[k-j]
        # = -delta*a[k-i] + delta*a[k-j]
        # But this only changes conv at position k. Other positions also change.
        # For a quick check, just evaluate the full score (expensive but safe)
        
        # Actually, let's just do a full evaluation for the top candidate
        a_new_conv = _full_convolve(a, a)
        new_max = float(a_new_conv.max())
        new_score = 2.0 * n * new_max
        if new_score < best_new_score:
            best_new_score = new_score
            best_j = j
        # Revert
        a[i] += delta
        a[j] -= delta
    
    if best_j >= 0 and best_new_score < current_score * (1 - 1e-6):
        a[i] -= delta
        a[best_j] += delta
        return True
    
    # Try smaller delta
    delta = 0.003 * a[i]
    if delta > 1e-10:
        for j in candidates[:2]:
            a[i] -= delta
            a[j] += delta
            a_new_conv = _full_convolve(a, a)
            new_score = 2.0 * n * float(a_new_conv.max())
            if new_score < current_score * (1 - 1e-6):
                return True
            a[i] += delta
            a[j] -= delta
    
    return False


def _adam_logspace(a_init, deadline, beta, lr_init=0.005, max_iter=100000):
    """Adam optimization in log-space with softened max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    prev_score = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_safe(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - v_buf ** 0 + 1e-30)  # fixed: use b2**t
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()
        
        if t % 20 == 0:
            a_chk = _softmax_safe(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                prev_score = sc
        if t % 5000 == 0:
            current_lr = max(0.0001, current_lr * 0.9)

    a_final = _softmax_safe(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_polish(a_init, deadline, beta):
    """L-BFGS refinement on smoothed objective in log-space."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_safe(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_safe(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 40)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_safe(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_safe(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shaped":
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (2 * t - 1) ** 2
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_safe(z)

            b_r = 0.0005 * float(c_p.max())
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.5, 30000)
            p, _ = _subgrad_line_search(p, time.monotonic() + per_r * 0.4, max_iter=20000, momentum=0.7)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final evaluation ===
    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
