def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):

    n = len(v)
    u = np.sort(v)[::-1]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: FISTA with adaptive log-sum-exp smoothing, top-k subgradient focusing,
Adam fine-tuning in log-space, L-BFGS polish, and multi-length search (n=4096, 8192).
Key improvements: aggressive beta continuation, top-k gradient focusing, and trying larger n.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_vec(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_flip = a[::-1]
    wc = _full_convolve(w, a_flip)
    return 2.0 * wc[n - 1:2 * n - 1]


def _make_beta_schedule(max_c, n_phases=30, decay=0.45):
    betas = []
    beta = 0.05 * max_c
    for _ in range(n_phases):
        betas.append(beta)
        beta *= decay
    return betas


def _fista_opt(a_init, beta_schedule, deadline, topk=100):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-30:
        a = a / s
    a_prev = a.copy()
    t_val = 1.0
    best_score = _score_norm(a)
    best_a = a.copy()

    for phase, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1.0:
            break
        c_curr = _full_convolve(a, a)
        c_max = float(c_curr.max())
        if c_max < 1e-300:
            break
        L = 4.0 * n / max(beta, 1e-300)
        step = 1.0 / L
        max_iterations = int(remaining * 3000)
        max_iterations = max(2000, min(max_iterations, 500000))

        for it in range(max_iterations):
            if time.monotonic() > deadline:
                break
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            c_max_y = float(c_y.max())
            # Top-k softmax weights
            k_eff = min(topk, 2 * n - 1)
            if k_eff >= 2 * n - 1:
                w = np.exp((c_y - c_max_y) / beta)
            else:
                part_idx = np.argpartition(-c_y, k_eff - 1)[:k_eff]
                w = np.zeros(2 * n - 1)
                w_vals = np.exp((c_y[part_idx] - c_max_y) / beta)
                w_sum = w_vals.sum()
                if w_sum < 1e-300:
                    w[part_idx] = 1.0 / k_eff
                else:
                    w[part_idx] = w_vals / w_sum

            grad = _compute_grad(y, w)
            a_next = _proj_simplex(y - step * grad)
            a_prev = a.copy()
            a = a_next
            t_val = t_next

            if (it + 1) % 500 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if phase % 5 == 0:

    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init=0.005, topk=100):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(2000000):
        if time.monotonic() > deadline:
            break
        t_step = it + 1
        a = _softmax_vec(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_eff = min(topk, 2 * n - 1)
        if k_eff >= 2 * n - 1:
            e = np.exp((c - cm) / beta)
        else:
            part_idx = np.argpartition(-c, k_eff - 1)[:k_eff]
            e = np.zeros(2 * n - 1)
            e_vals = np.exp((c[part_idx] - cm) / beta)
            e[part_idx] = e_vals
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros(2 * n - 1)
            w[int(c.argmax())] = 1.0
        else:
            w = e / s_e

        grad = _compute_grad(a, w)
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t_step % 50 == 0:
            a_chk = _softmax_vec(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if t_step % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_vec(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_and_grad(z):
        a = _softmax_vec(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            val = 2.0 * n * cm
            w = np.zeros(2 * n - 1)
            w[int(c.argmax())] = 1.0
        else:
            val = 2.0 * n * (cm + beta * np.log(s))
            w = e / s
        grad_a = _compute_grad(a, w)
        correction = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - correction)
        return val, grad_z

    max_time = deadline - time.monotonic() - 5.0
    if max_time <= 2.0:
        a_best = _softmax_vec(best_z)
        return a_best, best_score

    try:
        result = minimize(
            lambda z: obj_and_grad(z)[0],
            z0,
            jac=lambda z: obj_and_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 25)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_vec(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as ex:
        print(f"  L-BFGS error: {ex}")

    a_best = _softmax_vec(best_z)
    return a_best, best_score


def _resample_sequence(a, new_n):
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    resampled = np.interp(new_x, old_x, a)
    resampled = np.maximum(resampled, 0.0)
    s = resampled.sum()
    if s > 1e-30:
        resampled = resampled / s
    else:
        resampled = np.ones(new_n) / new_n
    return resampled

            beta_sched_r = _make_beta_schedule(max_cp, n_phases=15, decay=0.5)
            p, _ = _fista_opt(p, beta_sched_r, time.monotonic() + per_r * 0.5, topk=100)
            p, _ = _adam_opt(p, time.monotonic() + per_r * 0.5, 0.0001 * max_cp, 0.003, topk=50)
            p = p / p.sum()
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_ref = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 6: Final ultra-polish ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_fin = best_seq.copy() / best_seq.sum()
        c_f = _full_convolve(a_fin, a_fin)
        max_cf = float(c_f.max())
        per_t = deadline - time.monotonic() - 8.0
        if per_t > 10:
            # Very fine Adam
            a_fin, _ = _adam_opt(a_fin, time.monotonic() + per_t * 0.5,
                                 0.00001 * max_cf, 0.001, topk=50)
            a_fin = a_fin / a_fin.sum()
            sc_f = _score(a_fin)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_fin.copy()
                print(f"  Final Adam: score={sc_f:.10f}")
            # L-BFGS on remainder
            if time.monotonic() < deadline - 10:
                c_ff = _full_convolve(a_fin, a_fin)
                b_f = 0.000005 * float(c_ff.max())
                a_fin, _ = _lbfgs_opt(a_fin, deadline - 5.0, b_f)
                a_fin = a_fin / a_fin.sum()
                sc_ff = _score(a_fin)
                if sc_ff < best_score:
                    best_score = sc_ff
                    best_seq = a_fin.copy()
                    print(f"  Final L-BFGS: score={sc_ff:.10f}")

    print(f"Done: best_score={best_score:.10f} t={time.monotonic()-t_start:.0f}s")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.0f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
