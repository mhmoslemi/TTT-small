"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA + Adam + L-BFGS with aggressive beta continuation,
LP-guided direction finding, and diverse restarts.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_from_w(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Estimate Lipschitz constant
    c_init = _full_convolve(a_curr, a_curr)
    max_c0 = float(c_init.max())
    L = max(4.0 * n / max(beta, 1e-30), 1.0)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        diff = (c_y - c_max) / beta
        diff = np.clip(diff, -50, 50)
        w = np.exp(diff)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad_from_w(y, w)
        # Adaptive step size (backtracking)
        while step > 1e-15:
            a_next = _proj_simplex(y - step * grad)
            c_next = _full_convolve(a_next, a_next)
            if float(c_next.max()) < c_max + step * float(np.dot(grad, a_next - y)) + 1e-12:
                break
            step *= 0.5
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        diff = (c - cm) / beta
        diff = np.clip(diff, -50, 50)
        e = np.exp(diff)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        diff = (c - cm) / beta
        diff = np.clip(diff, -50, 50)
        e = np.exp(diff)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        diff = (c - cm) / beta
        diff = np.clip(diff, -50, 50)
        e = np.exp(diff)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bumped":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def _lp_polish(a, deadline):
    """Try to improve using LP-based direction finding."""
    from scipy.optimize import linprog
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for round_idx in range(50):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top active lags
        top_k = min(10, len(c))
        active = np.argsort(c)[::-1][:top_k]
        # Only keep lags within 1% of max
        active = active[c[active] >= c_max * 0.995]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # LP: find d such that conv(a,d)[k] < 0 for k in active, sum(d)=0, -a <= d
        # Maximize sum(d) to make the strongest step
        epsilon = 1e-10 * c_max
        
        A_rows = []
        b_vals = []
        
        for k in active:
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = a[idx]
            A_rows.append(row)
            b_vals.append(-epsilon)
        
        # sum(d) = 0
        A_rows.append(np.ones(n))
        b_vals.append(1e-10)
        A_rows.append(-np.ones(n))
        b_vals.append(1e-10)
        
        A = np.array(A_rows)
        b = np.array(b_vals)
        
        # Maximize -||d||_1 (or sum of negative parts) to get a descent direction
        # Actually, let's minimize the max increase: min t s.t. conv(a,d)[k] <= t for k in active, sum(d)=0, |d| <= 1
        # This is a LP with n+1 variables and |active|+2+n constraints
        
        # Simpler: minimize sum(d^2) approximately -> just find any feasible d
        # Let's use: minimize c_obj @ d where c_obj encourages spread
        c_obj = np.zeros(n)
        # Try to decrease the active conv values as much as possible
        # c_obj = grad of max conv (negative = good direction)
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = _compute_grad_from_w(a, w)
        c_obj = grad  # We want to minimize grad @ d (descent direction)
        
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(-a[i], a[i]*2) for i in range(n)], 
                      method='highs', options={'presolve': True})
        
        if res.success and res.fun < -1e-15:
            d = res.x
            # Line search
            best_t = 0
            best_sc = _score_norm(a)
            for t_try in [0.5, 0.3, 0.1, 0.05]:
                a_try = np.maximum(a + t_try * d, 0)
                s_try = a_try.sum()
                if s_try > 0.01:
                    a_try = a_try / s_try
                    sc_try = _score_norm(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_t = t_try
            if best_t > 0:
                a = np.maximum(a + best_t * d, 0)
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"    LP polish round {round_idx}: score={sc:.10f}")
    
    return best_a, best_score

"""
Multi-phase optimizer: FISTA + dual-tracking Adam + L-BFGS + LP polish
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements: aggressive beta continuation, dual variable tracking,
LP-guided direction finding, diverse restarts with spectral perturbations.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_from_w(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter, step_scale=1.0):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c0 = float(c_init.max())
    L = max(4.0 * n / max(beta, 1e-30), 1.0) * step_scale
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    backtracks = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        diff = (c_y - c_max) / beta
        diff = np.clip(diff, -50.0, 50.0)
        w = np.exp(diff)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad_from_w(y, w)
        # Backtracking (limited)
        for _bt in range(5):
            a_next = _proj_simplex(y - step * grad)
            c_next = _full_convolve(a_next, a_next)
            cn = float(c_next.max())
            diff_a = float(np.dot(grad, a_next - y))
            if cn <= c_max + step * diff_a + 1e-14 * c_max:
                break
            step *= 0.5
            backtracks += 1
        else:
            a_next = _proj_simplex(y - step * grad)
        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
        # Periodically increase step if no backtracks
        if it % 200 == 199 and backtracks < 10:
            step *= 1.1
            backtracks = 0

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _dual_adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam with dual variable tracking for smoother convergence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    # Dual variable (soft weights on convolution lags)
    c_init = _full_convolve(a_init, a_init)
    w_dual = np.exp((c_init - c_init.max()) / max(beta, 1e-30))
    w_dual = w_dual / w_dual.sum()
    dual_lr = 2.0 / max(beta, 1e-30)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Update dual variable
        w_target = np.exp((c - cm) / beta)
        w_target = np.exp(np.clip((c - cm) / beta, -50, 50))
        ws = float(w_target.sum())
        if ws > 1e-300 and np.isfinite(ws):
            w_target = w_target / ws
        else:
            w_target = np.zeros_like(c)
            w_target[int(np.argmax(c))] = 1.0
        # Exponential tracking of dual
        w_dual = 0.95 * w_dual + 0.05 * w_target
        w_dual = w_dual / w_dual.sum()
        # Adam gradient step
        a_rev = a[::-1]
        wc = _full_convolve(w_dual, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w_dual, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.25:
            update = update * (0.25 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 40 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 4000 == 0:
            current_lr = max(0.0002, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        diff = (c - cm) / beta
        diff = np.clip(diff, -50.0, 50.0)
        e = np.exp(diff)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        diff = (c - cm) / beta
        diff = np.clip(diff, -50.0, 50.0)
        e = np.exp(diff)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 40)), 'gtol': 1e-14, 'maxls': 25}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _lp_direction(a, active_lags, epsilon):
    """Find a descent direction using LP. Returns (d, obj_value) or (None, 0)."""
    from scipy.optimize import linprog
    n = len(a)

    A_rows = []
    b_vals = []
    for k in active_lags:
        row = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row[j] = a[k - j]
        A_rows.append(row)
        b_vals.append(-epsilon)

    A_rows.append(np.ones(n))
    b_vals.append(1e-10)
    A_rows.append(-np.ones(n))
    b_vals.append(1e-10)

    A = np.array(A_rows)
    b = np.array(b_vals)

    w = np.zeros(len(a) + len(a) - 1)
    w[active_lags] = 1.0 / len(active_lags)
    grad = _compute_grad_from_w(a, w)
    c_obj = grad

    lower = -a * 0.5
    upper = a * 1.0

    res = linprog(c_obj, A_ub=A, b_ub=b,
                  bounds=list(zip(lower, upper)),
                  method='highs', options={'presolve': True})

    if res.success and res.fun < -1e-12:
        return res.x, res.fun
    return None, 0.0


def _lp_polish(a, deadline, max_rounds=30):
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        top_k = min(20, len(c))
        sorted_idx = np.argsort(c)[::-1][:top_k]
        active = sorted_idx[c[sorted_idx] >= c_max * 0.99]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        epsilon = 1e-9 * c_max
        d, obj = _lp_direction(a, active, epsilon)

        if d is not None:
            best_sc = _score_norm(a)
            best_t = 0.0
            for t_try in [0.4, 0.2, 0.1, 0.05, 0.02]:
                a_try = np.maximum(a + t_try * d, 0.0)
                s_try = a_try.sum()
                if s_try > 0.01:
                    a_try = a_try / s_try
                    sc_try = _score_norm(a_try)
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_t = t_try
            if best_t > 0:
                a = np.maximum(a + best_t * d, 0.0)
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if round_idx % 5 == 0:
                        print(f"    LP r{round_idx}: score={sc:.10f}")
        else:
            break

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bumped":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _spectral_perturb(a, sigma, rng):
    """Perturb in frequency domain to explore different shapes."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    # Add random phase perturbation (preserves magnitude)
    freqs = np.arange(fa.shape[0])
    noise_amp = sigma * rng.standard_normal(fa.shape[0])
    noise_phase = sigma * rng.standard_normal(fa.shape[0])
    fa_new = fa * (1.0 + noise_amp) * np.exp(1j * noise_phase)
    fa_new[0] = abs(fa_new[0])  # Keep DC real and positive
    p = np.fft.irfft(fa_new, N)[:n]
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s > 0.01:
        p = p / s
    else:
        p = a.copy()
    return p

    # === Phase 4: L-BFGS refinement ===
    print("Phase 4: L-BFGS")
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        lbf_betas = [3e-6*max_cc, 1e-6*max_cc, 4e-7*max_cc, 1.5e-7*max_cc]
        per_b = (phase4_end - time.monotonic() - 5) / len(lbf_betas)
        for beta in lbf_betas:
            if time.monotonic() > phase4_end - 5:
                break
            a_work, _ = _lbfgs_opt(a_work, time.monotonic() + per_b, beta)
            a_work = a_work / a_work.sum()
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  L-BFGS beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: LP polish ===
    print("Phase 5: LP polish")
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end - 10:
        a_work, _ = _lp_polish(a_work, phase5_end, max_rounds=20)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"  LP polish: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print("Phase 6: Diverse restarts")
    phase6_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < phase6_end - 20:
        n_restarts = 6
        per_r = (phase6_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break

            if i == 0:
                p = _perturb(a_work, 0.008, rng)
            elif i == 1:
                p = _spectral_perturb(a_work, 0.015, rng)
            elif i == 2:
                shift = rng.integers(1, max(2, n_main // 128))
                p = np.roll(a_work, shift).copy()
                p = p / p.sum()
            elif i == 3:
                q = _make_profile(n_main, "raised_cosine")
                p = 0.9 * a_work + 0.1 * q
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 4:
                p = _spectral_perturb(a_work, 0.03, rng)
            else:
                # Coarse-to-fine
                factor = 4
                small_n = n_main // factor
                p_small = a_work[:small_n * factor].reshape(small_n, factor).mean(axis=1)
                p_small = p_small / p_small.sum()
                p_small = _perturb(p_small, 0.04, rng)
                p = np.repeat(p_small, factor)
                if len(p) < n_main:
                    p = np.concatenate([p, a_work[len(p):]])
                p = p[:n_main]
                p = np.maximum(p, 1e-12)
                p = p / p.sum()

            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 2e-5 * max_cp
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.4, 20000)
            p = p / p.sum()
            p, _ = _dual_adam_opt(p, time.monotonic() + per_r * 0.35, b_r * 0.3, 0.002, 40000)
            p = p / p.sum()
            if time.monotonic() < phase6_end - 12:
                p, _ = _lbfgs_opt(p, time.monotonic() + per_r * 0.2, b_r * 0.1)
                p = p / p.sum()
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-polish ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 4
        if per_t > 4:
            # Ultra-fine Adam
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.4,
                                 5e-8 * max_cp, 0.0008, 80000)
            a_pol = a_pol / a_pol.sum()
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # L-BFGS
            if time.monotonic() < deadline - 5:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 2e-8 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 2.0, b_f)
                a_pol = a_pol / a_pol.sum()
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
