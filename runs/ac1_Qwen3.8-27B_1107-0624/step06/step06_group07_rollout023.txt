Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses polynomial parameterized global search, L-BFGS with log-sum-exp smoothing,
active-set subgradient with backtracking, and coordinate descent polishing.
Key novelty: low-dimensional polynomial parameterization for global exploration,
combined with multi-scale refinement and aggressive active-set tracking.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _poly_sequence(coeffs, n):
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for j in range(len(coeffs)):
        a += coeffs[j] * t**j
    a = np.maximum(a, 1e-15)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s


def _poly_score(coeffs, n):
    a = _poly_sequence(coeffs, n)
    return _score(a)


def _poly_gradient(coeffs, n, num_step=1e-6):
    a = _poly_sequence(coeffs, n)
    c = _full_convolve(a, a)
    c_max = c.max()
    s = a.sum()
    base = 2.0 * n * c_max / (s * s)
    grad = np.zeros(len(coeffs))
    for j in range(len(coeffs)):
        cp = coeffs.copy()
        cp[j] += num_step
        sp = coeffs.copy()
        sp[j] -= num_step
        fp = _poly_score(cp, n)
        fm = _poly_score(sp, n)
        grad[j] = (fp - fm) / (2.0 * num_step)
    return base, grad

            coeffs0[0] = 1.0
            coeffs0[-1] = 0.5

                best_sc = sc
                best_coeffs = res.x.copy()
        except Exception:
            pass

    if best_coeffs is None:
        return None, np.inf
    a = _poly_sequence(best_coeffs, n)
    return a, best_sc


def _spectral_init(n, n_harmonics=5, rng=None):
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    if rng is None:
        rng = np.random.default_rng(42)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.02, 0.15) / k
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * k * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _ramp_init(n, alpha=1.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + alpha * (t - 0.3)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _two_bump_init(n):
    t = np.linspace(0, 1, n)
    a = np.exp(-20 * (t - 0.3)**2) + np.exp(-20 * (t - 0.7)**2) + 0.1
    return a / a.sum()


def _zigzag_init(n, periods=3):
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * np.abs(np.sin(np.pi * periods * t))
    return a / a.sum()


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-15, 'maxcor': 80,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002, 0.00001,
                              0.000005, 0.000002, 0.000001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_iter_time = 0.002 * (n / 2048.0)**1.3
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 20))))

        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _active_set_subgrad(a_init, deadline, tol_frac=1e-5, max_steps=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    lr = 0.02
    stall = 0
    prev_sc = _score(a)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + 0.001 * i)
        z_new = z - step * grad
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        a_new = _softmax(z_new)
        sc_new = _score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            z = z_new
            stall = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stall += 1
            if stall % 50 == 0:
                lr *= 0.8
            if lr < 1e-8:
                break
            if stall > 500:
                break

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=2000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng_c = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = int(rng_c.integers(0, n))
        c = _full_convolve(a, a)
        c_max = c.max()

        best_delta = 0.0
        best_local_sc = best_sc
        base_val = a[idx]
        for frac in [0.5, 0.7, 0.85, 0.95, 1.1, 1.3, 1.5, 2.0]:
            new_val = base_val * frac
            if new_val < 0:
                continue
            a_try = a.copy()
            a_try[idx] = new_val
            s = a_try.sum()
            if s < 1e-30:
                continue
            sc_try = _score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_delta = new_val - base_val
                best_a_try = a_try.copy()

        if best_delta != 0.0:
            a = best_a_try
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = a.copy()

        if rnd % 100 == 0 and rnd > 0:
            c_check = _full_convolve(a, a)
            if c_check.max() < c_max * 0.999:
                pass

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.003 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: polynomial global search -> L-BFGS refinement ->
    active-set subgradient -> coordinate descent polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                try_update(parent, "parent")
                print(f"Parent: n={len(parent)}, score={best_score_val:.10f}")
    except Exception:
        pass

    if parent is None:
        a0 = np.ones(2048) / 2048
        try_update(a0, "uniform_init")
        print(f"Default init: score={best_score_val:.10f}")

    # === Phase 1: Polynomial parameterized global search ===
    print(f"\n=== Phase 1: Polynomial search (t={time.monotonic()-t_start:.1f}s) ===")
    poly_deadline = t_start + 250.0

    for n_try in [2048, 4096]:
        if time.monotonic() > poly_deadline:
            break
        for degree in [3, 5, 7, 10, 15]:
            if time.monotonic() > poly_deadline:
                break
            max_iter = max(500, int(30000 / (degree + 1)))
            a_poly, sc_poly = _optimize_poly(n_try, degree, max_iter,
                                             deadline=poly_deadline)
            if a_poly is not None:
                try_update(a_poly, f"poly_d{degree}_n{n_try}")

    # Try different initial profiles
    print(f"\n=== Phase 1b: Profile inits (t={time.monotonic()-t_start:.1f}s) ===")
    profile_deadline = t_start + 350.0
    n_base = 4096

    profiles = [
        ("ramp", _ramp_init(n_base, 0.5)),
        ("ramp2", _ramp_init(n_base, 1.0)),
        ("two_bump", _two_bump_init(n_base)),
        ("zigzag", _zigzag_init(n_base, 3)),
        ("zigzag5", _zigzag_init(n_base, 5)),
        ("spectral", _spectral_init(n_base, 5, rng)),
        ("spectral2", _spectral_init(n_base, 8, rng)),
    ]

    for name, a_prof in profiles:
        if time.monotonic() > profile_deadline:
            break
        try_update(a_prof, f"init_{name}")
        a_refined = _lbfgs_schedule(a_prof, deadline, min_time_remaining=600.0)
        try_update(a_refined, f"init_{name}_lbfgs")

    # === Phase 2: Main L-BFGS with fine schedule ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        a_p2 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=400.0)
        try_update(a_p2, "phase2_lbfgs")

    # === Phase 3: Active-set subgradient ===
    print(f"\n=== Phase 3: Active-set subgrad (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sg, sc_sg = _active_set_subgrad(best_seq, deadline - 200,
                                          tol_frac=1e-5, max_steps=30000)
        try_update(a_sg, "phase3_subgrad")

    # === Phase 4: Fine L-BFGS (tiny beta) ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(8000, max(500, int(remaining / 0.003)))
            a_new = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate descent (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 80, max_rounds=3000)
        try_update(a_cd, "coord_descent")

    # === Phase 6: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 6: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            ("g002", "gauss", 0.02),
            ("g005", "gauss", 0.05),
            ("g010", "gauss", 0.10),
            ("four3", "fourier", 0),
            ("four7", "fourier", 0),
        ]

        for name, ptype, sigma in restart_configs:
            if time.monotonic() > deadline - 40:
                break
            if ptype == "gauss":
                p_start = _perturb(best_seq, sigma, rng)
            else:
                n_modes = int(name[4:]) if name.startswith("four") else 3
                p_start = _fourier_perturb(best_seq, rng, n_modes)

            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=30.0)
            try_update(a_r, f"restart_{name}")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 30:
                break
            if target_n == len(best_seq):
                continue
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=20.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 8: Final aggressive polish ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())

            # Ultra-fine L-BFGS
            beta = mb * 5e-7
            a_f = _lbfgs_run(best_seq, beta, n, 5000)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _active_set_subgrad(best_seq, deadline - 5,
                                             tol_frac=5e-6, max_steps=10000)
                try_update(a_s, f"final_sg_{rnd}")

            # Quick coordinate descent
            if time.monotonic() < deadline - 5:
                a_c, _ = _coordinate_descent(best_seq, deadline - 5, max_rounds=1000)
                try_update(a_c, f"final_cd_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
