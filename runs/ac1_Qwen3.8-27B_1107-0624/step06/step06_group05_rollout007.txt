        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        # Adaptive step size (backtracking)
        a_next = _proj_simplex(y - step * grad)
        # Check if step is too large (score increased a lot)
        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _mirror_descent_opt(a_init, deadline, max_iter):
    """Mirror descent with softmax-weighted active set."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive tolerance: start wide, narrow over time
        progress = min(1.0, it / (0.5 * max_iter))
        tol = (1.0 - progress) * 0.1 * c_max + progress * 1e-8 * c_max
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        grad = _compute_grad(a, w)
        lr = 2.0 / (n * (it + 2.0))
        z = z - lr * grad
        a = _softmax_compute(z)
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam in log-space on smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad = _compute_grad(a, w)
        # Correct for softmax parameterization
        correction = float(np.dot(a, grad))
        grad_z = a * (grad - correction)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)

        vh = v_buf / (1.0 - b2 ** t)
