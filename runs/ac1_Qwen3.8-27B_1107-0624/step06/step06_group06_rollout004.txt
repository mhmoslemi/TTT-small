Optimized step-function search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses combination of: spectral initialization, L-BFGS with adaptive log-sum-exp
smoothing, LP-based direction finding (SOTA approach), and multi-n search.

import numpy as np
import time
from scipy.optimize import minimize, linprog
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()

        w[int(np.argmax(c))] = 1.0

    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]

def _gradient_active(a, c, n, tol_frac=0.002):

        active[int(np.argmax(c))] = True

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns the optimal b (normalized to sum=1) or None if infeasible."""
    n = len(f)
    # Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<=n-1
    num_constraints = 2 * n - 1
    C = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        C[k, js] = f[k - js]
    
    # LP: max 1^T b s.t. C b <= M, b >= 0
    # scipy linprog solves: min c^T x s.t. A_ub x <= b_ub, A_lb x <= b_lb
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = C
    b_ub = np.full(num_constraints, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            b = result.x
            s = b.sum()
            if s > 1e-15:
                return b / s
            return None
    except Exception:
        pass
    return None

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: L-BFGS with adaptive log-sum-exp smoothing, LP-based direction finding
(SOTA approach), spectral/Walsh initializations, active-set subgradient,
multi-n exploration, and support reduction post-processing.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time_per_iter = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 8))))
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxcor': 40, 'maxls': 80}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.85
                if patience > 30:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _build_conv_matrix(f):
    n = len(f)
    num_rows = 2 * n - 1
    C = np.zeros((num_rows, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _solve_lp_direction(f, M, deadline):
    n = len(f)
    if time.monotonic() > deadline:
        return None
    try:
        C = _build_conv_matrix(f)
        num_constraints = 2 * n - 1
        c_obj = -np.ones(n)
        b_ub = np.full(num_constraints, M)
        bounds = [(0.0, None)] * n
        t0 = time.monotonic()
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': max(1.0, deadline - time.monotonic() - 5.0)})
        if time.monotonic() - t0 > 120:
            return None
        if result.success:
            b = result.x
            s = float(b.sum())
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None


def _lp_iteration(a, deadline, n_lp_solves=3):
    n = len(a)
    a_work = a.copy()
    a_work = a_work / a_work.sum()
    best_a = a_work.copy()
    best_sc = _score(a_work)
    for _ in range(n_lp_solves):
        if time.monotonic() > deadline - 10:
            break
        c = _full_convolve(a_work, a_work)
        M = float(c.max())
        b_star = _solve_lp_direction(a_work, M, deadline - 10)
        if b_star is None:
            break
        sc_b = _score(b_star)
        if sc_b < best_sc:
            best_a = b_star.copy()
            best_sc = sc_b
            a_work = b_star.copy()
        else:
            for t in [0.3, 0.5, 0.7]:
                a_try = (1.0 - t) * a_work + t * b_star
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_a = a_try.copy()
                    best_sc = sc_try
                    a_work = a_try.copy()
                    break
            else:
                break
    return best_a, best_sc


def _make_walsh_init(n, alpha=0.5, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    mask = rng.integers(0, 2, size=n) * 2 - 1
    a = 1.0 + alpha * mask
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def _make_spectral_init(n, seed=0):
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)
    phases = rng.uniform(0, 2 * np.pi, size=N // 2 + 1)
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = 1.0
    for j in range(1, N // 2 + 1):
        A[j] = np.exp(1j * phases[j]) * (1.0 / np.sqrt(2.0 * n))
    a_padded = np.fft.irfft(A, N)
    a = np.maximum(a_padded[:n], 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _make_tapered_init(n, kind="sin"):
    t = np.linspace(0, 1, n)
    if kind == "sin":
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "cos":
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
    elif kind == "quadratic":
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t * t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_chirp_init(n, alpha=0.3):
    t = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t * (t + 1) / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample_profile(profile, n_target):
    n_src = len(profile)
    if n_src == n_target:
        return profile.copy()
    x_old = np.linspace(0, 1, n_src)
    x_new = np.linspace(0, 1, n_target)
    a = np.interp(x_new, x_old, profile)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    return a / s


def _fine_lbfgs(a_init, deadline, n_betas=20):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    betas = max_c * np.geomspace(0.001, 0.00001, n_betas)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break
        max_iter = min(3000, max(20, int(remaining / 0.02)))
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a
