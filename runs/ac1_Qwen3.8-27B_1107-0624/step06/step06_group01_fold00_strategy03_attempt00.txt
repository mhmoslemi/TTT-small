def _full_convolve(a, b):
    """
    Compute the full linear convolution of two sequences using FFT.
    This is an efficient way to compute the convolution. If the input sequences

    # Determine the next power of two for the convolution length
    n_out = len(a) + len(b) - 1
    next_power = 2**np.ceil(np.log2(n_out))
    
    # Pad sequences if necessary
    if a.shape[0] < next_power:
        a_pad = np.zeros(next_power)
        a_pad[:len(a)] = a
    else:
        a_pad = a
    
    if b.shape[0] < next_power:
        b_pad = np.zeros(next_power)
        b_pad[:len(b)] = b
    else:
        b_pad = b

    fa = np.fft.rfft(a_pad)

    return convolve_result

        # These buffers are being clipped to avoid exploding gradients
        # Notation kept from the previous program but we'll use actual gradients from full_convolve.
        grad = _compute_grad_at_z(a_log)
        
        # Adam update rule: first moment (moving average) and second moment (Adam's moment)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2_listof_vars) * (grad ** 2)
        
        # Bias correction
        m_corr = m / (1 - beta1)
        v_corr = v / (1 - beta2)
        
        # Updates
        update = -learning_rate * m_corr / (np.sqrt(v_corr) + epsilon)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update / update_norm
            
        a_log = a_log + update
        
        # Reset if confidence too low
        if np.prod(np.exp(a_log)) < 0.01:
            a_log = np.random.randn(n) + 10.0
            a_log = a_log - a_log.mean()  # Center

        a_work = _project_simplex(np.exp(a_log))

        # Instead, we'll use a first-order approximation: compute J at the new candidate.
        candidate_work = a_work

    # Use is objective

      - Then, try a multi-phase optimization: use Adam optimization for finer tuning, interleaved with annealing for exploration.

        a_current = np.ones(n_fixed)
    
    # Normalize a_current to ensure sum>=0.01
