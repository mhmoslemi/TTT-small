"""
Improved multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
1. Nesterov-accelerated subgradient with adaptive active sets
2. Backtracking line search for step size selection

4. Structured perturbations based on Fourier Analisis
5. Solution merging (convex combination of good solutions)

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _nesterov_subgrad(a_init, deadline, tol_frac=1e-4, max_steps=20000,
                       lr_init=0.05, momentum=0.9):
    """Nesterov-accelerated subgradient with adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()  # Nesterov lookahead
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0
    adap_tol = tol_frac
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(y)
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Adaptive active set width
        c_range = c.max() - c.min()
        if c_range > 1e-15:
            adap_tol = max(1e-6, min(tol_frac, 0.5 * c_range / (2*n-1)))
        
        tol = adap_tol * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Nesterov update
        z = y
        y = z - lr * grad
        y = np.clip(y, -80.0, 80.0)
        y = y - y.mean()
        
        # Momentum blend
        if i > 0:
            y = momentum * y + (1 - momentum) * z
        
        # Adaptive step size
        if i % 100 == 0:
            a_check = _softmax(y)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.3)
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.8
                    momentum = min(momentum + 0.01, 0.98)
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.95
    
    a_final = _softmax(y)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
    
    return best_a, best_sc

def _backtrack_subgrad(a_init, deadline, tol_frac=1e-3, max_steps=5000):
    """Subgradient with backtracking line search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.1
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad = grad / (np.linalg.norm(grad) + 1e-15)  # Normalize
        
        # Backtracking line search
        step = lr
        found = False
        for _ in range(20):
            z_try = z - step * grad
            z_try = np.clip(z_try, -80.0, 80.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)

def _merge_solutions(a1, a2, weights=None):
    """Convex combination of two solutions."""
    if weights is None:
        weights = np.random.dirichlet([1, 1])
    n1, n2 = len(a1), len(a2)
    if n1 != n2:
        # Resample to common length
        target_n = max(n1, n2)
        if n1 != target_n:
            a1 = np.interp(np.linspace(0, 1, target_n), np.linspace(0, 1, n1), a1)
        if n2 != target_n:
            a2 = np.interp(np.linspace(0, 1, target_n), np.linspace(0, 1, n2), a2)
    a = weights[0] * a1 + weights[1] * a2
    a = a / a.sum()
    return a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-10 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    """For pair (i,j), find optimal delta to minimize max autocorrelation.
    Returns the optimal delta."""
    n = len(a)
    x, y = a[i], a[j]
    if x + y < 1e-15:
        return 0.0
    
    # d_k = a[k-i] + a[i-k] - a[k-j] - a[j-k] for k = 0..2n-2
    # where a[m] = 0 if m < 0 or m >= n
    d = np.zeros(2*n-1)
    # Contribution from a[i]: for lag k, a[i] appears in Q_k if k-i is in [0,n-1] or i-k is in [0,n-1]
    # Q_k = sum_l a[l]*a[k-l]
    # dQ_k/da[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)

    for k in range(2*n-1):
        contrib = 0.0
        if 0 <= k - i < n:
            contrib += a[k - i]
        if 0 <= i - k < n:
            contrib += a[i - k]
        d[k] += contrib
    
    # For a[j] (negative):
    for k in range(2*n-1):
        contrib = 0.0
        if 0 <= k - j < n:
            contrib += a[k - j]
        if 0 <= j - k < n:
            contrib += a[j - k]
        d[k] -= contrib

    # We want to minimize max_k Q_k(δ) subject to:
    # -x <= δ <= y (to keep a[i]>=0 and a[j]>=0)
    
    # The function f(δ) = max_k (c[k] + δ*d[k]) is convex in δ
    # Use ternary search

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Barzilai-Borwein adaptive subgradient, pair-coordinate descent,
aggressive L-BFGS beta continuation, solution sewing, and diverse restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _pair_descent(a_init, deadline, n_sweeps=50, pairs_per_sweep=200):
    """Pair-coordinate descent: redistribute mass between pairs to reduce peak."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)

    sweep = 0
    improvement_count = 0

    while sweep < n_sweeps and time.monotonic() < deadline:
        improved_this_sweep = 0
        # Random subset of pairs
        indices = np.random.permutation(n)
        for p in range(pairs_per_sweep):
            if time.monotonic() > deadline:
                break
            i = int(indices[p])
            j = int(indices[(p + 1) % n])
            if i == j:
                continue
            x, y = a[i], a[j]
            if x + y < 1e-12:
                continue

            # Compute d[k] = grad_Q_k wrt a[i] - grad_Q_k wrt a[j]
            # d[k] = (a[k-i] + a[i-k]) - (a[k-j] + a[j-k]) with bounds
            d = np.zeros(2 * n - 1)
            for k in range(max(0, i - n + 1), min(2 * n - 1, i + n)):
                if 0 <= k - i < n:
                    d[k] += a[k - i]
                if 0 <= i - k < n:
                    d[k] += a[i - k]
            for k in range(max(0, j - n + 1), min(2 * n - 1, j + n)):
                if 0 <= k - j < n:
                    d[k] -= a[k - j]
                if 0 <= j - k < n:
                    d[k] -= a[j - k]

            # f(delta) = max_k (c[k] + delta * d[k])
            # Convex in delta, minimize over [-x, y]
            lo, hi = -x, y
            # Quick evaluation
            f_lo = np.max(c + lo * d)
            f_hi = np.max(c + hi * d)
            f_0 = float(c.max())

            if min(f_lo, f_hi) >= f_0 - 1e-15:
                continue

            # Ternary search (30 iterations for good precision)
            for _ in range(30):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                v1 = np.max(c + m1 * d)
                v2 = np.max(c + m2 * d)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1

            delta = (lo + hi) / 2.0
            if abs(delta) < 1e-12:
                continue

            a_new = a.copy()
            a_new[i] = x + delta
            a_new[j] = y - delta
            if a_new[i] < 0 or a_new[j] < 0:
                continue
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)

            if sc_new < best_sc - 1e-14:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
                c = _full_convolve(a, a)
                improved_this_sweep += 1

        sweep += 1
        if improved_this_sweep == 0 and sweep > 5:
            break
        improvement_count += improved_this_sweep

    return best_a, best_sc


def _sewing(a1, a2, n_segments=4):
    """Take best segments from two solutions of the same length."""
    n1, n2 = len(a1), len(a2)
    if n1 != n2:
        return None
    n = n1
    seg_size = n // n_segments
    best = np.zeros(n)
    for s in range(n_segments):
        start = s * seg_size
        end = start + seg_size if s < n_segments - 1 else n
        # Pick the segment with lower local autocorrelation contribution
        # Approximate: pick the one with lower max local energy
        e1 = np.sum(a1[start:end] ** 2)
        e2 = np.sum(a2[start:end] ** 2)
        if e1 <= e2:
            best[start:end] = a1[start:end]
        else:
            best[start:end] = a2[start:end]
    best = np.maximum(best, 0)
    s = best.sum()
    if s < 1e-15:
        return None
    return best / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
