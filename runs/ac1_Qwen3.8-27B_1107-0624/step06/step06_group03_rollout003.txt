```python
"""
P-norm annealing optimizer with exact coordinate descent for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA with p-norm smoothing of the max convolution, followed by
exact 1D coordinate descent polishing and Fourier-domain diversity restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1: 2n * max(c)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - 1.0) / idx > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_pnorm(a, c, p):
    """Compute gradient of (sum c_k^p)^(1/p) with respect to a.
    
    The gradient is proportional to 2 * cross_corr(a, c^(p-1)).
    We compute the cross-correlation using FFT.
    """
    n = len(a)
    N = _next_pow2(2 * n)
    
    # Clamp c to be non-negative (should already be, but for safety)
    c_clipped = np.maximum(c, 0.0)
    
    if p == 2:
        w = c_clipped.copy()
    else:
        # c_k^(p-1), with numerical stability
        c_max = float(c_clipped.max())
        if c_max < 1e-300:
            w = np.ones_like(c_clipped)
        else:
            # Scale to avoid overflow
            c_scaled = c_clipped / c_max
            w = c_scaled ** (p - 1)
            # The actual weight is c^(p-1) = c_max^(p-1) * c_scaled^(p-1)
            # But since we only need the gradient up to a scale factor for FISTA,
            # we can use c_scaled^(p-1) and adjust the step size.
    
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _fista_pnorm(a_init, deadline, p_schedule, top_k_extra=0):
    """FISTA on simplex with p-norm objective and p-annealing.
    
    a_init: initial point on simplex (sum=1)
    p_schedule: list of p values to anneal through
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        
        # Reset momentum at phase boundary
        a_prev = a_curr.copy()
        t_val = 1.0
        
        # Compute Lipschitz constant estimate
        # ||grad||_L <= 2 * ||w||_1 * max(||a||_infty) ... rough bound
        # Use adaptive step size instead
        c_curr = _full_convolve(a_curr, a_curr)
        c_max = float(c_curr.max())
        
        if p == 2:
            w = np.maximum(c_curr, 0)
        else:
            c_scaled = np.maximum(c_curr, 0) / max(c_max, 1e-300)
            w = c_scaled ** (p - 1)
        
        # Rough Lipschitz: L ≈ 2 * sum(w) * max(a) or similar
        # Use a conservative step and adapt
        L = 2.0 * float(np.sum(w)) * float(np.sum(a_curr)) + 1e-10
        # Actually, the Lipschitz constant of the gradient of the p-norm
        # is bounded by p * (p-1) * ||w||_inf * ... it's complicated.
        # Let's use backtracking instead.
        
        max_iter = 50000  # Will be limited by time
        step = 1.0 / L
        
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            grad = _grad_pnorm(y, c_y, p)
            
            # Backtracking line search
            while step > 1e-15:
                a_next = _proj_simplex(y - step * grad)
                # Check if descent condition holds ( Lion smoother)
                # For simplicity, just use the step
                break  # Skip backtracking for speed
            
            a_next = _proj_simplex(y - step * grad)
            
            # Update best
            if it % 100 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
            
            # Decrease step size slowly
            if it % 5000 == 0:
                step *= 0.95
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if best_t != a_i_old and best_f < _score(a):
                a[i] = best_t
                S = S - a_i_old + best_t
                improved += 1
        
        # Update score after full pass
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            print(f"    CD pass {p}: score={sc:.10f}, improved={improved}")
    
    return a, best_score
```

Wait, the coordinate descent above is very slow because for each coordinate, I'm computing a full convolution trial. That's O(n) per coordinate per pass, and O(n^2) per pass. For n=4096, that's about 16M operations per pass, which should be OK for a few passes.

But actually, the `c_trial = c.copy()` and the modifications are O(n) each, and I do this for each t_candidate (maybe 10-20 candidates). So it's O(n * num_candidates) per coordinate, O(n^2 * num_candidates) per pass. For n=4096 and num_candidates=20, that's about 200M operations per pass. With numpy vectorization, this should take maybe 0.1-0.5 seconds per pass. So I can do 100+ passes in a few minutes.

Actually wait, I'm doing `c_trial = c.copy()` which is O(n), and then the max is O(n). For each candidate t, that's O(n). With ~20 candidates per coordinate and n coordinates, that's O(n^2 * 20) per pass = O(4096^2 * 20) ≈ 340M operations. In numpy, this might take about 1-2 seconds per pass. So I can do maybe 100-300 passes in 5 minutes.

Let me optimize: instead of copying the full c array for each candidate, I can precompute the modifications and only evaluate the max at the relevant positions.

Actually, let me think more carefully. The key insight is:

For a given coordinate i, changing a_i to t:
- c_k changes only for k in [i, i+n-1] (the "affected" range)
- For k in [i, i+n-1], k ≠ 2i: c_k_new = c_k + (t - t_old) * 2 * a[k-i]
- For k = 2i: c_k_new = c_k + t^2 - t_old^2
- For k outside [i, i+n-1]: c_k_new = c_k (unchanged)

So the new max is:
`max( max_{k outside [i,i+n-1]} c_k,  max_{k in [i,i+n-1], k≠2i} (c_k + (t-t_old)*2*a[k-i]),  c[2i] + t^2 - t_old^2 )`

The first term is a constant (call it `max_unaffected`). The second term is the max of lines in t (piecewise linear). The third term is a quadratic in t.

So I can precompute `max_unaffected` once per pass (or update it as I go). Then for each coordinate, I only need to evaluate the max over the affected range, which has at most n elements.

This is much faster! Let me restructure:

def _exact_coord_descend_fast(a, n_passes, deadline):
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    max_c = float(c.max())
    best_score = 2.0 * n * max_c / (S * S)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        # Precompute prefix and suffix max of c for fast "unaffected max" queries
        # For coordinate i, unaffected = max(c[0:i], c[i+n:2n-1])
        # (note: c has length 2n-1)
        c_len = 2 * n - 1
        prefix_max = np.maximum.accumulate(c)  # prefix_max[k] = max(c[0..k])
        suffix_max = np.maximum.accumulate(c[::-1])[::-1]  # suffix_max[k] = max(c[k..c_len-1])
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            t_old = a[i]
            if t_old < 1e-15:
                continue
            
            # Unaffected max: max of c[0:i] (if i>0) and c[i+n:c_len] (if i+n < c_len)
            # Note: affected range is [i, i+n-1], which has n elements
            # Unaffected: [0, i-1] U [i+n, 2n-2]
            if i > 0:
                max_left = prefix_max[i - 1]
            else:
                max_left = -np.inf
            if i + n <= c_len - 1:
                max_right = suffix_max[i + n]
            else:
                max_right = -np.inf
            max_unaffected = max(max_left, max_right)
            
            # Affected: c[i:i+n] (indices i to i+n-1)
            # For k in [i, i+n-1]:
            #   if k == 2i: c_k_new = c[2i] + t^2 - t_old^2
            #   else: c_k_new = c[k] + (t - t_old) * 2 * a[k-i]
            
            # The "linear" part: for k in [i, i+n-1], k != 2i
            #   c_k_new = c[k] + 2*(t-t_old)*a[k-i]
            #   = (c[k] - 2*t_old*a[k-i]) + 2*t*a[k-i]
            #   = B_k + d_k * t  where B_k = c[k] - 2*t_old*a[k-i], d_k = 2*a[k-i]
            
            # The "quadratic" part: k = 2i (if 2i is in [i, i+n-1], i.e., i <= 2i <= i+n-1, i.e., 0 <= i <= n-1, always true)
            #   c_{2i}(t) = c[2i] - t_old^2 + t^2
            
            # So max_affected(t) = max( max_k (B_k + d_k*t), c[2i] - t_old^2 + t^2 )
            # where k ranges over [i, i+n-1] \ {2i}
            
            # For the linear part, max_k (B_k + d_k*t) is the upper envelope of lines.
            # Since d_k = 2*a[k-i] >= 0, the max is achieved at the k with the largest
            # B_k + d_k*t. For a given t, this is the k that maximizes B_k + d_k*t.
            
            # For the overall max: max_affected(t) = max(linear_envelope(t), quadratic(t))
            # And the overall max of c is: max(max_unaffected, max_affected(t))
            
            # The objective: F(t) = 2n * max(max_unaffected, max_affected(t)) / (S - t_old + t)^2
            
            # To find the minimum, I'll use a grid + refinement approach:
            # 1. Evaluate F at t = 0, t_old, and a few heuristics
            # 2. For the active k's (those giving the max), compute critical points
            # 3. Refine around the best
            
            # Quick evaluation at a few points
            t_candidates = [0.0, t_old, t_old * 0.5, t_old * 1.5]
            
            # Find the k that gives the max at current t_old
            affected_c = c[i:i+n]
            # At t_old, the affected values are just c[i:i+n]
            # The k giving the max among affected:
            k_local_max = int(np.argmax(affected_c))
            k_global_max = i + k_local_max
            
            # Critical point for the active k:
            if k_global_max == 2 * i:
                # Quadratic: c[2i] - t_old^2 + t^2
                # F(t) = 2n * max(max_unaffected, c[2i] - t_old^2 + t^2) / (S - t_old + t)^2
                # If quadratic dominates: (c[2i] - t_old^2 + t^2) / (S - t_old + t)^2
                # Critical: (2t)(S-t_old+t) - 2(c[2i]-t_old^2+t^2) = 0
                # 2tS - 2t*t_old + 2t^2 - 2c[2i] + 2t_old^2 - 2t^2 = 0
                # 2tS - 2t*t_old - 2c[2i] + 2t_old^2 = 0
                # t(S - t_old) = c[2i] - t_old^2
                # t = (c[2i] - t_old^2) / (S - t_old)  [if S != t_old]
                if S - t_old > 1e-15:
                    t_star = (c[2 * i] - t_old * t_old) / (S - t_old)
                    if t_star > 0:
                        t_candidates.extend([t_star, t_star * 0.7, t_star * 1.3])

                    # Critical: d_k*(S-t_old+t) - 2*(B_k+d_k*t) = 0
                    # d_k*S - d_k*t_old + d_k*t - 2*B_k - 2*d_k*t = 0
                    # -d_k*t = 2*B_k - d_k*S + d_k*t_old

                    #   = S - t_old - 2*B_k/d_k

                    if t_star > 0:
