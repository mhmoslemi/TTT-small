import numpy as np

def _next_pow2(n):

def _full_convolve(a, b):

        if remaining < 20:
            break

        est_iter_time = 0.004 * (n / 4096.0) ** 1.3
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 20))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            # Check convergence: if gradient is small relative to objective, reduce beta
            z_check = np.log(np.maximum(current_a, 1e-15))
            z_check = z_check - z_check.mean()
            g = _smograd(z_check, beta, n)
            g_norm = np.linalg.norm(g)
            obj_val = _smoobj(z_check, beta, n)

            if g_norm < 1e-10 * max(1.0, abs(obj_val)):
                beta = max(min_beta, beta * beta_factor)
            elif niter < max_iter * 0.3:
                beta = max(min_beta, beta * 0.7)
        else:
            beta = max(min_beta, beta * 0.5)

        if beta <= min_beta and success:
            break

    return current_a


def _nesterov_subgrad(a_init, deadline, tol_frac=1e-4, lr_init=0.03,
                       momentum_init=0.5):
    """Nesterov-accelerated subgradient in the log domain."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()  # Nesterov lookahead
    z_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    alpha = momentum_init
    lr = lr_init
    stall = 0
    n_steps = 0

    while time.monotonic() < deadline:
        n_steps += 1
        # Nesterov update: y = z + alpha * (z - z_prev)
        y = z + alpha * (z - z_prev)
        z_prev = z.copy()

        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_y * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        # Adaptive step size with cosine annealing component
        decay = max(0.3, 1.0 - 0.7 * n_steps / max(1, int((deadline - time.monotonic()) * 10)))
        step = lr * decay / (1.0 + 0.001 * n_steps)

        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Check score periodically
        if n_steps % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                lr = min(lr * 1.02, 0.1)
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.75
                    tol_frac = max(1e-6, tol_frac * 0.7)
                    if lr < 1e-8:
                        break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _nesterov_subgrad_direct(a_init, deadline, tol_frac=1e-5, max_steps=50000):
    """Nesterov subgradient in direct space with projection onto simplex."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.5 / np.sqrt(n)
    stall = 0
    momentum = 0.3

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        a_y = a + momentum * (a - a_prev)
        a_y = np.maximum(a_y, 0.0)
        s_y = a_y.sum()
        if s_y < 1e-30:
            break
        a_y = a_y / s_y

        c = _full_convolve(a_y, a_y)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        a_prev = a.copy()
        a = a - lr * grad
        # Project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
        else:
            a = a / s

        if i % 200 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                stall = 0
                lr = min(lr * 1.01, 2.0 / np.sqrt(n))
            else:
                stall += 1
                if stall > 4:
                    lr *= 0.7
                    if lr < 1e-12:
                        break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_structured(a, rng, n_modes=5):
    """Perturb using low-frequency Fourier modes - more effective for this problem."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.005 * rng.uniform(0.5, 4.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(8, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.3, 1.5)))
        z[start:end] += rng.uniform(-0.02, 0.02)
    z = z - z.mean()
    return _softmax(z)


def _perturb_antisym(a, rng, strength=0.01):
    """Perturb in the anti-symmetry direction to reduce central peak."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    # Create a perturbation that emphasizes antisymmetry
    t = np.arange(n)
    pert = np.sin(np.pi * t / n)  # Anti-symmetric about center
    pert = pert - pert.mean()
    scale = strength * rng.uniform(0.5, 2.0)
    z = z + scale * pert
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _kkt_polish(a_init, deadline, max_iter=200):
    """KKT-inspired polishing: make the active gradient more uniform."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.01

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Adaptive active set: top fraction by value
        sorted_idx = np.argsort(c)[::-1]
        n_active = max(10, int(0.1 * len(c)))
        active_set = set(sorted_idx[:n_active].tolist())

        # Weighted gradient: weight by how close to max
        threshold = c[sorted_idx[n_active - 1]]
        w = np.zeros_like(c)
        mask = c > threshold
        if mask.sum() == 0:
            mask[sorted_idx[0]] = True
        w[mask] = (c[mask] - threshold) / (c_max - threshold + 1e-15)
        w = w / w.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        # Make gradient more uniform: subtract its mean effect
        # The KKT condition wants h - W to be constant, i.e., h = W + const
        # So we want to reduce variance of h
        h_var = np.var(h)
        if h_var < 1e-20:
            break

        # Step to reduce h variance
        a_try = a - lr * grad
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            break
        a_try = a_try / s

        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break

    return best_a, best_sc


def _simulated_annealing(a_init, deadline, n_iterations=200000, T_init=0.001, T_final=1e-7):
    """Simulated annealing in the log domain with structured perturbations."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    current_sc = best_sc
    rng = np.random.default_rng(12345)

    for i in range(n_iterations):
        if time.monotonic() > deadline:
            break

        # Linear cooling
        progress = i / n_iterations
        T = T_init * (T_final / T_init) ** progress

        # Random perturbation
        pert_type = rng.integers(0, 4)
        if pert_type == 0:
            # Small Gaussian
            z_try = z + 0.01 * rng.standard_normal(n)
        elif pert_type == 1:
            # Block perturbation
            z_try = z.copy()
            start = int(rng.integers(0, n - 32))
            end = min(n, start + int(rng.integers(16, 128)))
            z_try[start:end] += rng.uniform(-0.02, 0.02)
        elif pert_type == 2:
            # Fourier perturbation
            z_try = z.copy()
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = rng.uniform(0.002, 0.01)
            phase = rng.uniform(0, 2 * np.pi)
            t = np.arange(n)
            z_try += amp * np.cos(2 * np.pi * freq * t / n + phase)
        else:
            # Antisymmetric perturbation
            z_try = z.copy()
            t = np.arange(n)
            freq = int(rng.integers(1, max(2, n // 16)))
            amp = rng.uniform(0.002, 0.008)
            z_try += amp * np.sin(2 * np.pi * freq * t / n)

        z_try = z_try - z_try.mean()
        a_try = _softmax(z_try)
        sc_try = _score(a_try)

        # Accept/reject
        delta = sc_try - current_sc
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            z = z_try
            current_sc = sc_try
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with Nesterov acceleration, adaptive log-sum-exp
    smoothing, KKT polishing, simulated annealing, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(4096) / 4096
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    n_main = len(best_seq)
    print(f"Main length: n={n_main}")

    # === Phase 1: Adaptive L-BFGS (400s) ===
    t_phase1_end = t_start + 400.0
    print(f"\n=== Phase 1: Adaptive L-BFGS (budget 400s) ===")
    a_p1 = _adaptive_lbfgs(best_seq, min(deadline, t_phase1_end), min_time_remaining=50.0)
    try_update(a_p1, "phase1_adaptive_lbfgs")

    # === Phase 2: Nesterov subgradient in log domain (300s) ===
    t_phase2_end = t_start + 700.0
    print(f"\n=== Phase 2: Nesterov subgrad log-domain (budget 300s) ===")
    if time.monotonic() < min(deadline, t_phase2_end):
        a_sg, sc_sg = _nesterov_subgrad(best_seq, min(deadline, t_phase2_end),
                                        tol_frac=1e-4, lr_init=0.025)
        try_update(a_sg, "phase2_nesterov_log")

        # Second pass with tighter tolerance
        if time.monotonic() < min(deadline, t_phase2_end) - 30:
            a_sg2, sc_sg2 = _nesterov_subgrad(best_seq, min(deadline, t_phase2_end) - 10,
                                              tol_frac=1e-5, lr_init=0.01)
            try_update(a_sg2, "phase2b_nesterov_tight")

    # === Phase 3: Nesterov subgradient in direct space (200s) ===
    t_phase3_end = t_start + 900.0
    print(f"\n=== Phase 3: Nesterov subgrad direct (budget 200s) ===")
    if time.monotonic() < min(deadline, t_phase3_end):
        a_sg3, sc_sg3 = _nesterov_subgrad_direct(best_seq, min(deadline, t_phase3_end),
                                                 tol_frac=1e-5, max_steps=50000)
        try_update(a_sg3, "phase3_nesterov_direct")

    # === Phase 4: KKT polishing (150s) ===
    t_phase4_end = t_start + 1050.0
    print(f"\n=== Phase 4: KKT polish (budget 150s) ===")
    if time.monotonic() < min(deadline, t_phase4_end):
        a_kkt, sc_kkt = _kkt_polish(best_seq, min(deadline, t_phase4_end), max_iter=500)
        try_update(a_kkt, "phase4_kkt")

    # === Phase 5: Simulated annealing (350s) ===
    t_phase5_end = t_start + 1400.0
    print(f"\n=== Phase 5: Simulated annealing (budget 350s) ===")
    if time.monotonic() < min(deadline, t_phase5_end):
        remaining = min(deadline, t_phase5_end) - time.monotonic()
        # Estimate iterations
        n_iters = int(remaining * 800)  # ~800 iters/sec
        n_iters = min(n_iters, 500000)
        a_sa, sc_sa = _simulated_annealing(best_seq, min(deadline, t_phase5_end),
                                           n_iterations=n_iters,
                                           T_init=0.002, T_final=1e-8)
        try_update(a_sa, "phase5_sa")

    # === Phase 6: Multi-restart with structured perturbations (200s) ===
    t_phase6_end = t_start + 1600.0
    print(f"\n=== Phase 6: Multi-restart (budget 200s) ===")
    if best_seq is not None and time.monotonic() < min(deadline, t_phase6_end):
        restart_configs = [
            ("antisym", "antisym", 0.01, 0),
            ("antisym2", "antisym", 0.02, 0),
            ("fourier_5", "fourier", 0, 5),
            ("fourier_8", "fourier", 0, 8),
            ("block_6", "block", 0, 6),
            ("gauss_003", "gauss", 0.03, 0),
            ("gauss_008", "gauss", 0.08, 0),
        ]

        per_restart = (min(deadline, t_phase6_end) - time.monotonic()) / len(restart_configs)

        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > min(deadline, t_phase6_end) - 20:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_structured(best_seq, rng, modes)
            elif ptype == "block":
                p_start = _perturb_block(best_seq, rng, modes)
            elif ptype == "antisym":
                p_start = _perturb_antisym(best_seq, rng, sigma)
            else:
                p_start = _perturb_gaussian(best_seq, 0.05, rng)

            sc_start = _score(p_start)

            # Quick Nesterov polish
            a_r, sc_r = _nesterov_subgrad(p_start, time.monotonic() + max(5, per_restart * 0.6),
                                          tol_frac=1e-4, lr_init=0.02)
            try_update(a_r, f"restart_{name}")

            # Quick L-BFGS
            if time.monotonic() < min(deadline, t_phase6_end) - 15:
                c_b = _full_convolve(a_r, a_r)
                mb = float(c_b.max())
                beta = mb * 0.01
                a_f, _, _ = _lbfgs_run(a_r, beta, n_main, 2000)
                try_update(a_f, f"restart_{name}_lbfgs")

    # === Phase 7: Try different lengths (150s) ===
    t_phase7_end = t_start + 1750.0
    print(f"\n=== Phase 7: Different lengths (budget 150s) ===")
    if best_seq is not None and time.monotonic() < min(deadline, t_phase7_end):
        for target_n in [2048, 8192]:
            if time.monotonic() > min(deadline, t_phase7_end) - 30:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            if sc_interp < best_score_val:
                print(f"  Interpolated n={target_n}: {sc_interp:.10f}")
                try_update(a_interp, f"interp_{target_n}")

            # Quick optimization at new length
            if time.monotonic() < min(deadline, t_phase7_end) - 20:
                c_b = _full_convolve(a_interp, a_interp)
                mb = float(c_b.max())
                beta = mb * 0.02
                max_iter = min(3000, max(500, int((min(deadline, t_phase7_end) - 20 - time.monotonic()) / 0.005)))
                a_new, _, _ = _lbfgs_run(a_interp, beta, target_n, max_iter)
                try_update(a_new, f"opt_n_{target_n}")

                # Nesterov at new length
                if time.monotonic() < min(deadline, t_phase7_end) - 10:
                    a_ns, _ = _nesterov_subgrad(a_new, min(deadline, t_phase7_end) - 5,
                                                tol_frac=1e-4, lr_init=0.02)
                    try_update(a_ns, f"nesterov_n_{target_n}")

    # === Phase 8: Final aggressive polish (remaining time) ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            n_cur = len(best_seq)
            c_b = _full_convolve(best_seq, c_b if False else best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-6
            max_iter = min(3000, max(500, int((deadline - 5 - time.monotonic()) / 0.005)))
            a_f, _, _ = _lbfgs_run(best_seq, beta, n_cur, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight Nesterov
            if time.monotonic() < deadline - 5:
                a_s, _ = _nesterov_subgrad(best_seq, deadline - 3,
                                           tol_frac=1e-5, lr_init=0.008)
                try_update(a_s, f"final_nesterov_{rnd}")

            # KKT polish
            if time.monotonic() < deadline - 3:
                a_k, _ = _kkt_polish(best_seq, deadline - 2, max_iter=300)
                try_update(a_k, f"final_kkt_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
