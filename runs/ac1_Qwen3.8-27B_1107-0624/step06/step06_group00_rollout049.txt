"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines structured initialization, FFT-based Adam in log-space with adaptive
temperature, coordinate descent with analytical line search, LP refinement,
and multi-scale exploration. Designed to improve upon previous 1.5057 result.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1, i.e., 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, T):
    """Compute gradient of smoothed max of convolution."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5, eps_frac=0.01):
    """Compute gradient using uniform distribution over near-max indices."""
    n = len(a)
    c_max = c.max()
    threshold = c_max - eps_frac * c_max
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    if len(active) > top_k:
        active = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[active] = 1.0 / len(active)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=15):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _adam_optimize(a, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, use_active=False, active_k=5,
                   eval_interval=200, clip_norm=0.8, use_nesterov=True):
    """Adam optimizer in log-space with adaptive temperature and optional Nesterov."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    lr_mult = 1.0
    nesterov_z = z.copy()

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        if use_nesterov:
            z_eval = nesterov_z
        else:
            z_eval = z

        a_cur = _softmax_stable(z_eval)
        c = _fft_convolve(a_cur, a_cur)
        if use_active and T < T_start * 0.003:
            g = _compute_active_gradient(a_cur, c, active_k)
        else:
            g = _compute_gradient(a_cur, c, T)

        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if use_nesterov:
            nesterov_z = z - b1 * update
            nesterov_z = np.clip(nesterov_z, -80.0, 80.0)
            nesterov_z = nesterov_z - nesterov_z.mean()

        if t % eval_interval == 0:
            a_check = _softmax_stable(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                nesterov_z = z.copy()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, max_passes=200, eval_interval=5):
    """
    Coordinate descent with golden section line search.
    For each coordinate, find the optimal value via line search.
    """
    n = len(a)
    a = a.copy()
    c = _fft_convolve(a, a)
    S = float(a.sum())
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(12345)

    for path in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Randomize coordinate order for diversity
        order = rng.permutation(n)
        improved = 0

        for idx in order:
            if time.monotonic() > deadline:
                break

            old_val = a[idx]
            # Quick grid search to find approximate best
            candidates = [0.0]
            if old_val > 1e-15:
                candidates.extend([old_val * f for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]])
            else:
                max_a = a.max()
                candidates.extend([max_a * f for f in [0.01, 0.05, 0.1, 0.2]])

            best_t = old_val
            best_c_max = c.max()
            best_denom = S

            for trial in candidates:
                # Compute new convolution in O(n)
                delta = trial - old_val
                if abs(delta) < 1e-15:
                    continue
                c_trial = c.copy()
                # Update: c_new[k] = c_old[k] + 2*delta*a[k-idx] for k in [idx, idx+n-1]
                k_start = idx
                k_end = min(idx + n, 2 * n - 1)
                a_slice = a[k_start - idx:k_end - idx]  # a[0:n] but shifted
                # Actually: a[k-idx] for k in [idx, idx+n-1] is a[0], a[1], ..., a[n-1]
                # Wait: k ranges from idx to min(idx+n-1, 2n-2)
                # k-idx ranges from 0 to min(n-1, 2n-2-idx)
                # We need a[k-idx] which is a[0: min(n, 2n-1-idx)]
                valid_len = min(n, 2 * n - 1 - idx)
                if valid_len > 0:
                    c_trial[k_start:k_start + valid_len] += 2.0 * delta * a[:valid_len]
                # Add delta^2 at position 2*idx if valid
                if 2 * idx < 2 * n - 1:
                    c_trial[2 * idx] += delta * delta

                new_S = S + delta
                if new_S < 1e-15:
                    continue
                c_max_trial = c_trial.max()
                sc_trial = 2.0 * n * c_max_trial / (new_S * new_S)

                if sc_trial < 2.0 * n * best_c_max / (best_denom * best_denom):
                    best_sc_trial = sc_trial
                    best_t = trial
                    best_c = c_trial.copy()
                    best_c_max = c_max_trial
                    best_denom = new_S

            # Golden section refinement around best_t
            if best_t < old_val * 0.3 or best_t > old_val * 1.5:
                lo = 0.0
                hi = max(best_t * 2, a.max() * 0.5) if best_t > 0 else a.max() * 0.3
            else:
                spread = old_val * 0.2 if old_val > 1e-10 else a.mean() * 0.2
                lo = max(0.0, best_t - spread)
                hi = best_t + spread

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)

            def _eval_coord(t_val):
                delta = t_val - old_val
                if abs(delta) < 1e-15:
                    c_t = c
                else:
                    c_t = c.copy()
                    k_start = idx
                    valid_len = min(n, 2 * n - 1 - idx)
                    if valid_len > 0:
                        c_t[k_start:k_start + valid_len] += 2.0 * delta * a[:valid_len]
                    if 2 * idx < 2 * n - 1:
                        c_t[2 * idx] += delta * delta
                new_S = S + (t_val - old_val)
                if new_S < 1e-15:
                    return np.inf
                return 2.0 * n * c_t.max() / (new_S * new_S)

            f1 = _eval_coord(x1)
            f2 = _eval_coord(x2)
            for _ in range(25):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = hi - gr * (hi - lo)
                    f1 = _eval_coord(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + gr * (hi - lo)
                    f2 = _eval_coord(x2)

            best_x = (lo + hi) / 2.0
            f_best = _eval_coord(best_x)

            # Compare with current
            f_curr = _eval_coord(old_val)
            if f_best < f_curr - 1e-12:
                # Update
                delta = best_x - old_val
                c[k_start:k_start + valid_len] += 2.0 * delta * a[:valid_len]
                if 2 * idx < 2 * n - 1:
                    c[2 * idx] += delta * delta
                c = np.maximum(c, 0.0)  # numerical safety
                a[idx] = best_x
                S += delta
                improved += 1

        # End of pass
        if path % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        if improved < n // 10:
            # Nearly converged
            if path > 5:
                break

    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _window_hanning(n):
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))


def _window_blackman(n):
    n = max(n, 3)
    m = np.arange(n)
    return 0.42 - 0.5 * np.cos(2.0 * np.pi * m / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * m / (n - 1))


def _window_blackman_harris(n):
    m = np.arange(n)
    a = [0.35875, 0.48829, 0.14128, 0.01168]
    w = np.zeros(n)
    for i, ai in enumerate(a):
        w += ai * np.cos(np.pi * (2 * m + i - i) / (n - 1) * (1 - i % 2) + np.pi * i / 2 * (i % 2))
    # Simpler: use the standard formula
    w = 0.35875 - 0.48829 * np.cos(2 * np.pi * m / (n - 1)) + \
        0.14128 * np.cos(4 * np.pi * m / (n - 1)) - 0.01168 * np.cos(6 * np.pi * m / (n - 1))
    return w


def _window_gaussian(n, sigma_frac=0.4):
    m = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * (n - 1) / 2.0
    return np.exp(-0.5 * ((m - center) / sigma) ** 2)


def _window_tukey(n, alpha=0.5):
    m = np.arange(n)
    w = np.ones(n)
    edge = int(alpha * (n - 1) / 2)
    if edge > 0:
        for i in range(edge):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
            w[n - 1 - i] = w[i]
    return w


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fista_optimize(a_init, deadline, T_start, T_end, max_iter=100000):
    """FISTA with simplex projection for the smoothed problem."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    L = 16.0 * n
    step = 1.0 / L

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        c = _fft_convolve(y, y)
        g = _compute_gradient(y, c, T)
        a_new = y - step * g
        # Project to simplex (a >= 0, sum = 1)
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new

        if t % 500 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = int(np.max(ind[cond]))
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

                print(f"  LP[{lp_iter}]: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
        except Exception as ex:
            print(f"  LP error: {ex}")

    # --- Phase 3: Coordinate descent on best (550s - 1000s) ---
    ph3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < ph3_end - 10:
        n_cd = len(best_a)
        a_cd, sc_cd_norm = _coordinate_descent(best_a, ph3_end - 5, max_passes=500)
        full_sc = _score(a_cd)
        print(f"  Phase3 CoordDesc: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # --- Phase 4: Active-set Adam polish (1000s - 1300s) ---
    ph4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < ph4_end - 10:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_active, sc_active = _adam_optimize(
            best_a, ph4_end - 5,
            T_start=0.001 * max_c, T_end=0.000005 * max_c,
            lr_start=0.015, lr_end=0.001,
            max_iter=300000, use_active=True, active_k=5,
            eval_interval=100, clip_norm=0.5,
            use_nesterov=True
        )
        full_sc = _score(a_active)
        print(f"  Phase4 ActiveAdam: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_active.copy()

    # --- Phase 5: Coordinate descent again (1300s - 1550s) ---
    ph5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < ph5_end - 10:
        a_cd2, sc_cd2 = _coordinate_descent(best_a, ph5_end - 5, max_passes=300)
        full_sc = _score(a_cd2)
        print(f"  Phase5 CoordDesc2: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_cd2.copy()

    # --- Phase 6: Multi-scale exploration (1550s - 1700s) ---
    ph6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < ph6_end - 10:
        n_large = 8192
        n_orig = len(best_a)
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())

        a_large_opt, _ = _adam_optimize(
            a_large, min(ph6_end - 10, time.monotonic() + 60),
            T_start=0.002 * max_cl, T_end=0.00001 * max_cl,
            lr_start=0.02, lr_end=0.002,
            max_iter=100000, use_active=True, active_k=5,
            eval_interval=200, use_nesterov=True
        )

        # Interpolate back
        a_back = _interp_to_length(a_large_opt, n_orig)
        # Quick coordinate descent on the result
        a_back_opt, _ = _coordinate_descent(a_back, ph6_end - 5, max_passes=50)
        full_sc = _score(a_back_opt)
        print(f"  Phase6 MultiScale: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_back_opt.copy()

        # Also try the large sequence directly
        full_sc_large = _score(a_large_opt)
        if full_sc_large < best_score:
            best_score = full_sc_large
            best_a = a_large_opt.copy()
            print(f"  Phase6: large seq better! score={full_sc_large:.10f}")

    # --- Phase 7: Final coordinate descent polish (1700s - deadline) ---
    if time.monotonic() < deadline - 10:
        a_final_cd, _ = _coordinate_descent(best_a, deadline - 5, max_passes=200)
        full_sc = _score(a_final_cd)
        print(f"  Phase7 Final CD: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_final_cd.copy()

    # --- Final validation and return ---
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.1
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
