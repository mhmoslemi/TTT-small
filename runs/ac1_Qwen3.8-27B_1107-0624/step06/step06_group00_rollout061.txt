"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
1. Adaptive active set (percentile-based instead of top-k)
2. Coordinate descent polish phase
3. Multi-n exploration (2048, 4096, 8192)
4. Better LP with warm-starting
5. Improved time management and restart diversity

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_soft(a, c, T):
    """Compute gradient using soft (Boltzmann) active set."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_gradient_active(a, c, top_k=5):
    """Compute gradient using hard active set (top-k peaks)."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_gradient_percentile(a, c, frac=0.05):
    """Compute gradient using percentile-based active set."""
    n = len(a)
    threshold = np.percentile(c, (1 - frac) * 100)
    active = c >= threshold
    if not np.any(active):
        active[np.argmax(c)] = True
    p = np.zeros_like(c)
    p[active] = 1.0 / np.sum(active)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_matrix(n):
    """Build sparse matrix for LP constraints: conv(a, g)_k = sum_j A[k,j] * g[j] <= M."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.ones(len(rows))
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_step(a, A, time_limit=15):
    """LP step: find g maximizing sum(g) subject to conv(a,g) <= M, g >= 0."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        # Line search for best blend
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M

    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    best_patience_score = best_score
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if grad_mode == 'active':
            g = _compute_gradient_active(a, c, active_k)
        elif grad_mode == 'percentile':
            g = _compute_gradient_percentile(a, c, active_frac)
        else:
            g = _compute_gradient_soft(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, n_passes=3):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_score = _score_norm(a)
    
    for pass_i in range(n_passes):
        if time.monotonic() > deadline:
            break
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Compute current convolution
            c = _fft_convolve(a, a)
            M = float(c.max())
            
            # The effect of changing a_j by delta:
            # c_k changes by: 2*a_{k-j}*delta for k where 0<=k-j<n and k!=2j
            # c_{2j} changes by: 2*a_j*delta + delta^2
            # We want to find delta that minimizes max_k (c_k + effect_k)
            
            # Simple approach: try a few values of a_j
            best_val = a[j]
            best_M = M
            # Try: 0, 0.5*a_j, a_j, 1.5*a_j, 2*a_j
            candidates = [0.0, 0.25*a[j], 0.5*a[j], 0.75*a[j], a[j], 
                         1.25*a[j], 1.5*a[j], 2.0*a[j]]
            for new_val in candidates:
                a[j] = new_val
                c_new = _fft_convolve(a, a)
                M_new = float(c_new.max())
                if M_new < best_M:
                    best_M = M_new
                    best_val = new_val
            a[j] = best_val
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    return a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
1. LP that directly minimizes the convolution peak (min t s.t. conv(a+d,a+d)_k<=t)
2. Adaptive active set via percentile (not just top-k)
3. Multi-n exploration (2048, 4096, 8192)
4. Better time management and diverse restarts
5. Dual-informed gradient for polishing

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_soft(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_gradient_active(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_gradient_percentile(a, c, frac=0.05):
    n = len(a)
    threshold = np.percentile(c, (1.0 - frac) * 100.0)
    active = c >= threshold
    if not np.any(active):
        active[np.argmax(c)] = True
    p = np.zeros_like(c)
    p[active] = 1.0 / float(np.sum(active))
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_convolution_rows(n):
    """Build row/col indices for the convolution matrix (sparsity pattern)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_minimize_peak(a, rows, cols, eps=0.02, time_limit=20):
    """LP: minimize t s.t. 2*conv(a,d)_k + c_k <= t, sum(d)=0, |d|<=eps."""
    n = len(a)
    c = _fft_convolve(a, a)
    # Build A_ub: for each k, row is [2*a_{k-j} for valid j] concatenated with [-1] for t
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        a_vals = 2.0 * a[k - j_range]
        vals_list.append(a_vals)
    vals = np.concatenate(vals_list)
    # Add -1 column for t variable (column n)
    A_data = np.concatenate([vals, -np.ones(2 * n - 1)])
    A_rows = np.concatenate([rows, np.arange(2 * n - 1)])
    A_cols = np.concatenate([cols, np.full(2 * n - 1, n, dtype=np.int32)])
    A_ub = sp.csc_matrix((A_data, (A_rows, A_cols)), shape=(2 * n - 1, n + 1))
    b_ub = -c
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(-eps, eps)] * n + [(0.0, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            d = res.x[:n]
            a_new = a + d
            a_new = np.maximum(a_new, 0.0)
            if a_new.sum() > 1e-15:
                a_new = a_new / a_new.sum()
            return a_new
        return a
    except Exception:
        return a


def _lp_maximize_sum(a, rows, cols, time_limit=15):
    """LP: maximize sum(g) s.t. conv(a,g)_k <= M, g>=0."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = np.ones(len(rows))
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _adam_optimize(a, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, grad_mode='soft', active_k=5,
                   active_frac=0.03, eval_interval=200, clip_norm=0.8):
    """Adam optimizer in z-space with various gradient modes."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)

        if grad_mode == 'active':
            g = _compute_gradient_active(a_cur, c, active_k)
        elif grad_mode == 'percentile':
            g = _compute_gradient_percentile(a_cur, c, active_frac)
        else:
            g = _compute_gradient_soft(a_cur, c, T)

        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps_adam)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_check = _softmax_stable(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with LP peak minimization, adaptive active sets,
    multi-n exploration, and diverse restarts.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Build LP sparsity pattern
    rows, cols = _build_convolution_rows(n)
    print(f"LP pattern built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # ============ Phase 1: LP peak minimization (0 - 280s) ============
    phase1_end = min(deadline, t_start + 280.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0
    while time.monotonic() < phase1_end:
        a_new = _lp_minimize_peak(a, rows, cols, eps=0.015, time_limit=18)
        sc_new = _score_norm(a_new)
        lp_count += 1
        if sc_new < prev_sc - 1e-10:
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try the maximize-sum LP as an alternative
            if stall % 3 == 0:
                a_alt = _lp_maximize_sum(a, rows, cols, time_limit=10)
                sc_alt = _score_norm(a_alt)
                if sc_alt < prev_sc - 1e-10:
                    a = a_alt
                    sc_new = sc_alt
                    stall = 0
            else:
                break
        prev_sc = _score_norm(a)
        full_sc = _score(a)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()
        if lp_count % 3 == 0:
            print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
        if stall > 12:
            break
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ============ Phase 2: Adam soft gradient (280s - 530s) ============
    phase2_end = min(deadline, t_start + 530.0)
    if time.monotonic() < phase2_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_opt, sc_opt = _adam_optimize(
            best_a, phase2_end - 5,
            T_start=0.005 * max_c, T_end=0.00003 * max_c,
            lr_start=0.04, lr_end=0.003,
            max_iter=300000, grad_mode='soft',
            eval_interval=200)
        full_sc = _score(a_opt)
        print(f"Phase 2 Adam-soft: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # ============ Phase 3: Adam percentile (530s - 780s) ============
    phase3_end = min(deadline, t_start + 780.0)
    if time.monotonic() < phase3_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_opt, sc_opt = _adam_optimize(
            best_a, phase3_end - 5,
            T_start=0.001 * max_c, T_end=0.00001 * max_c,
            lr_start=0.02, lr_end=0.0015,
            max_iter=300000, grad_mode='percentile', active_frac=0.02,
            eval_interval=200)
        full_sc = _score(a_opt)
        print(f"Phase 3 Adam-percentile: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # ============ Phase 4: Adam hard active (780s - 1020s) ============
    phase4_end = min(deadline, t_start + 1020.0)
    if time.monotonic() < phase4_end:
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_opt, sc_opt = _adam_optimize(
            best_a, phase4_end - 5,
            T_start=0.0005 * max_c, T_end=0.000003 * max_c,
            lr_start=0.012, lr_end=0.0008,
            max_iter=300000, grad_mode='active', active_k=3,
            eval_interval=150)
        full_sc = _score(a_opt)
        print(f"Phase 4 Adam-active: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # ============ Phase 5: Dual-gradient polish (1020s - 1200s) ============
    phase5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase5_end:
        a_dgp, sc_dgp = _dual_grad_polish(
            best_a, phase5_end - 5,
            max_iter=100000, lr=0.003, eval_interval=100)
        full_sc = _score(a_dgp)
        print(f"Phase 5 DualGrad: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_dgp.copy()

    # ============ Phase 6: Multi-n exploration (1200s - 1430s) ============
    phase6_end = min(deadline, t_start + 1430.0)
    if time.monotonic() < phase6_end:
        for n_try in [2048, 8192]:
            if time.monotonic() > phase6_end - 70:
                break
            a_try = _interp_to_length(best_a, n_try)
            c_try = _fft_convolve(a_try, a_try)
            max_ct = float(c_try.max())
            phase_sub_end = min(phase6_end - 30, time.monotonic() + 100)
            a_try_opt, _ = _adam_optimize(
                a_try, phase_sub_end,
                T_start=0.002 * max_ct, T_end=0.00002 * max_ct,
                lr_start=0.025, lr_end=0.0015,
                max_iter=100000, grad_mode='percentile', active_frac=0.02,
                eval_interval=200)
            full_sc = _score(a_try_opt)
            print(f"  Multi-n ({n_try}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_try_opt.copy()
        # If we found a better n, try to refine at that n
        if len(best_a) != n:
            a_refined, _ = _adam_optimize(
                best_a, min(phase6_end - 5, time.monotonic() + 30),
                T_start=0.0005 * _fft_convolve(best_a, best_a).max(),
                T_end=0.000005 * _fft_convolve(best_a, best_a).max(),
                lr_start=0.01, lr_end=0.0005,
                max_iter=50000, grad_mode='active', active_k=2,
                eval_interval=100)
            full_sc = _score(a_refined)
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_refined.copy()

    # ============ Phase 7: Perturbation restarts (1430s - 1640s) ============
    phase7_end = min(deadline, t_start + 1640.0)
    n_best = len(best_a)
    if time.monotonic() < phase7_end:
        for attempt in range(5):
            if time.monotonic() > phase7_end - 50:
                break
            sigma = 0.008 + 0.004 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_deadline = min(phase7_end - 10, time.monotonic() + 60)
            a_pert_opt, _ = _adam_optimize(
                a_pert, sub_deadline,
                T_start=0.001 * max_cp, T_end=0.00002 * max_cp,
                lr_start=0.02, lr_end=0.001,
                max_iter=50000, grad_mode='active', active_k=3,
                eval_interval=150)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # ============ Phase 8: Final ultra-fine polish (1640s - deadline) ============
    if time.monotonic() < deadline - 10:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _adam_optimize(
            best_a, deadline - 8,
            T_start=0.0001 * max_cf, T_end=0.000001 * max_cf,
            lr_start=0.003, lr_end=0.0001,
            max_iter=200000, grad_mode='active', active_k=2,
            eval_interval=80)
        full_sc = _score(a_final)
        print(f"Phase 8 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.001
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
