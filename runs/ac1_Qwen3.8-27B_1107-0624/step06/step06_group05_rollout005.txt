"""
Multi-phase optimizer with aggressive L-BFGS, peak equalization, active-set
exploit, and multi-length exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over prior: tighter beta schedules, peak-equalization step,
symmetric-constrained optimization, and adaptive multi-n search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _lbfgs_smooth(z0, n, beta, deadline, max_iter=2000):
    """L-BFGS on softmax parameterization with log-sum-exp smoothing."""
    z0 = z0 - z0.mean()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    remaining = deadline - time.monotonic() - 1.0
    if remaining <= 0.5:
        return z0, _score_norm(_softmax_compute(z0))

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(remaining * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc = _score_norm(a_opt)
        return z_opt, sc
    except Exception:
        return z0, _score_norm(_softmax_compute(z0))


def _peak_equalize(a, deadline, max_rounds=200):
    """Iteratively equalize the top peaks using active-set gradient averaging."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 1e-9 * c_max
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        grad = _compute_grad(a, w)
        # Polyak-like step size
        a_fb = _softmax_compute(z)
        cur_val = _score_norm(a_fb)
        lower_bound = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
        gap = cur_val - lower_bound
        if gap < 1e-12:
            break
        gnorm_sq = float(np.dot(grad, grad))
        if gnorm_sq < 1e-30:
            break
        lr = min(1.0 / np.sqrt(r + 1), max(1e-6, gap / (gnorm_sq + 1e-30)))
        lr = min(lr, 0.5)
        z = z - lr * grad
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        a_chk = _softmax_compute(z)
        sc = _score_norm(a_chk)
        if sc < best_sc:
            best_sc = sc
            best_a = a_chk.copy()

    return best_a, best_sc


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)

        if t % 30 == 0:

def _combing_init(n, num_teeth, rng):
    """Initialize with a comb-like pattern for diversity."""
    a = np.zeros(n)
    tooth_width = max(1, n // (2 * num_teeth))
    for i in range(num_teeth):
        start = int(i * n / num_teeth)
        end = min(start + tooth_width, n)
        a[start:end] = 1.0
    a = np.maximum(a, 1e-8)
    # Add small random perturbation
    a += 0.01 * rng.random(n)
    a = a / a.sum()
    return a

def _interpolate_sequence(a, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 4096

    if parent is not None:
        sc0 = _score(parent)
        print(f"Parent: n={n_parent}, score={sc0:.10f}")
        best_seq = parent.copy()
        best_score = sc0
    else:
        print("No parent available, using default init")
        a_def = _uniform_init(4096)
        best_seq = a_def.copy()
        best_score = _score(a_def)
        print(f"Default uniform n=4096: score={best_score:.10f}")

    # Determine which n values to try
    n_list = [n_parent]
    # Add other sizes (only if time permits)
    for n_try in [2048, 8192]:
        if n_try != n_parent and n_try >= 512 and n_try <= 16384:
            n_list.append(n_try)

    # Budget allocation: give parent's n the most time
    n_sorted = sorted(n_list, key=lambda x: (x != n_parent, abs(x - n_parent)))

    if len(n_sorted) == 1:
        time_per_n = deadline - time.monotonic() - 30
    elif len(n_sorted) == 2:
        time_per_n = [(deadline - time.monotonic() - 20) * 0.6,
                      (deadline - time.monotonic() - 20) * 0.4]
    else:
        time_per_n = [(deadline - time.monotonic() - 15) * 0.55,
                      (deadline - time.monotonic() - 15) * 0.25,
                      (deadline - time.monotonic() - 15) * 0.20]

    for ni, n in enumerate(n_sorted):
        if time.monotonic() > deadline - 40:
            break
        alloc_end = time.monotonic() + time_per_n[ni] - 10
        alloc_end = min(alloc_end, deadline - 30)
        print(f"\n=== Optimizing n={n} (budget until t={alloc_end - t_start:.0f}s) ===")

        # Initialize
        if n == n_parent and parent is not None:
            a0 = parent.copy()
        else:
            # Interpolate from parent or use default
            if parent is not None:
                a0 = _interpolate_sequence(parent, n)
            else:
                a0 = _uniform_init(n)

        # Run optimization
        a_opt, sc_opt = _optimize_for_n(a0, alloc_end, label=f"n={n}")
        sc_eval = _score(a_opt)
        if sc_eval < best_score:
            best_score = sc_eval
            best_seq = a_opt.copy()
            print(f"  NEW BEST: n={n}, score={sc_eval:.10f}")

    # === Final polish phase ===
    print(f"\n=== Final Polish (best so far: {best_score:.10f}) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 8
        print(f"  Final polish: n={n_best}, remaining={remaining:.1f}s")

        # Ultra-fine L-BFGS
        z = np.log(np.maximum(a_pol, 1e-15))
        z = z - z.mean()
        # Try a few very tight betas
        fine_betas = [5e-7 * max_cp, 2e-7 * max_cp, 1e-7 * max_cp, 5e-8 * max_cp]
        per_fb = remaining / (len(fine_betas) + 2)
        for beta in fine_betas:
            if time.monotonic() > deadline - 10:
                break
            z, _ = _lbfgs_smooth(z, n_best, beta, time.monotonic() + per_fb)
            a_pol = _softmax_compute(z)
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Fine L-BFGS beta={beta:.2e}: score={sc:.10f}")

        # Peak equalization final
        if time.monotonic() < deadline - 10:
            a_pe, sc_pe = _peak_equalize(a_pol, deadline - 5, max_rounds=200)
            sc = _score(a_pe)
            if sc < best_score:
                best_score = sc
                best_seq = a_pe.copy()
                print(f"  Final peak eq: score={sc:.10f}")

    # Safety: ensure valid output
    if best_seq is None:
        best_seq = _uniform_init(4096)

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _uniform_init(256)

    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
