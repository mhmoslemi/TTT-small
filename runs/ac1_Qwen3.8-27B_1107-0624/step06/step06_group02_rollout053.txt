LP-based optimizer for minimizing the peak autocorrelation of non-negative step functions.
Uses linear programming to find descent directions, combined with FISTA/Adam gradient polish.

from scipy.sparse.diagonal import diag_indices

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score_norm(a):
    """Score assuming sum(a)=1: returns 2*n*max(a*a)"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _solve_lp(f, M, n):

    Solve LP: maximize sum(g) subject to g >= 0 and (f*g)_k <= M for all k.
    Returns the optimal g or None if infeasible.

    cols = []
    vals = []

        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(num_constraints, n))

"""
LP-augmented optimizer for minimizing peak autocorrelation of non-negative step functions.
Combines linear programming for global direction finding with FISTA (smoothed max) and
Adam (log-space) for gradient-based refinement. Targets sub-1.5027 score.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista_smoothed(a_init, beta, deadline, max_iter=100000):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        if c_max < 1e-300:
            c_max = 1e-300
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -600, 600, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0 or it == max_iter - 1:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.01, max_iter_total=1000000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -600, 600, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _build_lp_matrix(n):
    """Precompute row/col indices for the LP constraint matrix (Toeplitz structure)."""
    num_k = 2 * n - 1
    rows_list = []
    cols_list = []
    for k in range(num_k):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows_list.append(k)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def _solve_lp_direction(f, M, n, row_idx, col_idx, timeout_s=60):
    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k.
    Returns optimal g or None.
    """
    num_k = 2 * n - 1
    vals = f[col_idx]  # A[col_idx_row, col_idx] = f[col_idx] since A[k,i]=f[i]
    A = csr_matrix((vals, (row_idx, col_idx)), shape=(num_k, n))
    c_obj = -np.ones(n)
    b_ub = np.full(num_k, M)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'time_limit': timeout_s, 'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-15:
                return g
        else:
            return None
    except Exception:
        return None
    return None


def _lp_iteration(f, n, row_idx, col_idx, deadline, timeout_s=45):
    """
    One LP-based iteration: find direction g via LP, line search, update f.
    Returns updated f.
    """
    c = _convolve(f, f)
    M = float(c.max())
    if M < 1e-300:
        return f.copy()

    g0 = _solve_lp_direction(f, M, n, row_idx, col_idx, timeout_s)
    if g0 is None:
        return f.copy()

    sum_g0 = float(g0.sum())
    if sum_g0 < 1e-15:
        return f.copy()

    # Check if LP direction is meaningful (not just f itself)
    corr = float(np.dot(f, g0) / (np.linalg.norm(f) * np.linalg.norm(g0) + 1e-300))
    if corr > 0.9999:
        # LP didn't find a new direction
        return f.copy()

    # Rescale g to sum=1
    g = g0 / sum_g0

    # Line search for best t
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() > deadline:
            break
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            continue
        f_new = f_new / s
        sc = _score_norm(f_new)
        if sc < best_sc:
            best_sc = sc
            best_t = t

    if best_t == 0.0:
        return f.copy()

    f_new = (1.0 - best_t) * f + best_t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def _coord_descent(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 2048:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            for factor in [0.95, 0.98, 1.02, 1.05]:
                new_val = old_val * factor
                a[j] = new_val
                s = a.sum()
                if s < 1e-10:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val

        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}, n={n_main}")

    # Precompute LP matrix structure
    row_idx, col_idx = _build_lp_matrix(n_main)
    print(f"LP matrix: {2*n_main-1}x{n_main}, {len(row_idx)} nnz")

    # === Phase 1: FISTA warmup (200s) ===
    print("Phase 1: FISTA warmup")
    phase1_end = min(deadline, t_start + 200.0)
    a_curr = a0.copy()
    betas = [0.05 * max_c0, 0.02 * max_c0, 0.01 * max_c0,
             0.005 * max_c0, 0.002 * max_c0, 0.001 * max_c0]
    per_beta = (phase1_end - t_start) / len(betas)
    for beta in betas:
        if time.monotonic() > phase1_end - 5:
            break
        a_curr, sc = _fista_smoothed(a_curr, beta, time.monotonic() + max(5, per_beta - 2), 80000)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: best_score={best_score:.10f}")

    # === Phase 2: LP + gradient cycles (700s) ===
    print("Phase 2: LP + gradient cycles")
    phase2_end = min(deadline, t_start + 950.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    num_cycles = 0
    while time.monotonic() < phase2_end - 60:
        cycle_start = time.monotonic()
        num_cycles += 1
        remaining = phase2_end - cycle_start

        # LP step (budget: min(60s, 40% of remaining))
        lp_budget = min(60.0, 0.4 * remaining)
        if lp_budget > 15:
            a_lp = _lp_iteration(a_work, n_main, row_idx, col_idx,
                                 cycle_start + lp_budget, timeout_s=min(40, lp_budget - 5))
            if a_lp is not a_work:
                sc_lp = _score(a_lp)
                print(f"  Cycle {num_cycles} LP: score={sc_lp:.10f} t={time.monotonic()-t_start:.1f}s")
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = a_lp.copy()
                a_work = a_lp

        # Gradient polish after LP (budget: remaining - 10)
        if time.monotonic() < phase2_end - 20:
            c_w = _convolve(a_work, a_work)
            max_cw = float(c_w.max())
            grad_budget = phase2_end - time.monotonic() - 20
            if grad_budget > 20:
                # FISTA brief
                beta_f = 0.001 * max_cw
                fista_time = min(60.0, grad_budget * 0.4)
                if fista_time > 10:
                    a_work, _ = _fista_smoothed(a_work, beta_f,
                                                time.monotonic() + fista_time, 50000)
                    a_work = a_work / a_work.sum()

                # Adam brief
                if time.monotonic() < phase2_end - 15:
                    c_w2 = _convolve(a_work, a_work)
                    max_cw2 = float(c_w2.max())
                    betas_a = [0.001 * max_cw2, 0.0005 * max_cw2,
                               0.0002 * max_cw2, 0.0001 * max_cw2]
                    adam_time = phase2_end - time.monotonic() - 15
                    if adam_time > 15:
                        a_work, _ = _adam_logspace(
                            a_work, time.monotonic() + adam_time,
                            betas_a, lr_init=0.008, max_iter_total=500000)
                        a_work = a_work / a_work.sum()

                sc_w = _score(a_work)
                if sc_w < best_score:
                    best_score = sc_w
                    best_seq = a_work.copy()
                    print(f"  Cycle {num_cycles} polished: score={sc_w:.10f} t={time.monotonic()-t_start:.1f}s")

        # If LP gives no improvement twice in a row, reduce LP usage
        if num_cycles >= 3:
            c_last = _convolve(a_work, a_work)
            # Check convergence
            pass

    print(f"End Phase 2: best_score={best_score:.10f}, cycles={num_cycles}")

    # === Phase 3: Adam fine optimization (500s) ===
    print("Phase 3: Adam fine optimization")
    phase3_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase3_end - 30:
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        c_fin = _convolve(a_fin, a_fin)
        max_cf = float(c_fin.max())
        betas_fine = [0.0005 * max_cf, 0.0002 * max_cf, 0.0001 * max_cf,
                      0.00005 * max_cf, 0.00002 * max_cf, 0.00001 * max_cf]
        adam_deadline = min(phase3_end, time.monotonic() + (phase3_end - time.monotonic()))
        a_fin, sc_fin = _adam_logspace(
            a_fin, adam_deadline, betas_fine, lr_init=0.006, max_iter_total=2000000)
        a_fin = a_fin / a_fin.sum()
        sc_fin_full = _score(a_fin)
        print(f"  Adam fine: score={sc_fin_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin.copy()

    # === Phase 4: Multi-restart (200s) ===
    print("Phase 4: Multi-restart")
    phase4_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase4_end - 30:
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_restarts = 4
        per_restart = (phase4_end - time.monotonic()) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            if i == 0:
                p_start = _perturb(a_base, 0.015, rng)
            elif i == 1:
                shift = rng.integers(1, n_main // 1024)
                p_start = np.roll(a_base, shift).copy()
                p_start = p_start / p_start.sum()
            elif i == 2:
                p_start = _perturb(a_base, 0.03, rng)
            else:
                p_start = (a_base + 0.01 * rng.standard_normal(n_main))
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs,
                  0.00005 * max_cs]
            p_best, _ = _adam_logspace(
                p_start, time.monotonic() + max(5, per_restart * 0.8),
                bs, lr_init=0.006, max_iter_total=200000)
            p_best = p_best / p_best.sum()
            sc_r = _score(p_best)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Coordinate descent polish (80s) ===
    print("Phase 5: Coordinate descent")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 10, time.monotonic() + 80)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=300)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Final: Try n=8192 if time permits ===
    print("Phase 6: n=8192 exploration")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        a_big = a_big / a_big.sum()
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < deadline - 60:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            beta_big = 0.005 * max_cb
            a_big, _ = _fista_smoothed(a_big, beta_big,
                                       min(deadline - 30, time.monotonic() + 40), 30000)
            a_big = a_big / a_big.sum()
            sc_bb = _score(a_big)
            if sc_bb < best_score:
                best_score = sc_bb
                best_seq = a_big.copy()
                print(f"  Optimized n=8192: score={sc_bb:.10f}")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:

            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
