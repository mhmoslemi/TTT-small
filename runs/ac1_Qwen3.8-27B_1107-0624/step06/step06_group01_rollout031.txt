and multi-length exploration with adaptive time allocation.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    """Project onto probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _topk_mask(c, k):
    """Get mask of top-k elements in convolution."""
    if k >= len(c):
        return np.ones(len(c), dtype=bool)
    idx = np.argpartition(c, -k)[-k:]
    mask = np.zeros(len(c), dtype=bool)
    mask[idx] = True
    return mask

def _gradient_topk(a, c, k):
    """Compute gradient using top-k approximation of max."""
    n = len(a)
    # Top-k weighted gradient
    c_max = c.max()
    c_min = c.min()
    range_c = c_max - c_min + 1e-30
    # Use top-k: weight = 1 for top-k, 0 for rest (sharp approximation)
    mask = _topk_mask(c, k)
    # For smoother version, use a sharp sigmoid
    w = np.zeros_like(c)
    w[mask] = 1.0
    w = w / w.sum()
    
    # Gradient of sum(w * c) w.r.t. a
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

def _gradient_softmax(a, c, beta):
    """Compute gradient using softmax approximation of max."""
    n = len(a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

def _fista_topk(a_init, k, deadline, max_iter, step_init=None):
    """FISTA with top-k max approximation."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    
    # Estimate Lipschitz constant
    if step_init is not None:
        step = step_init
    else:
        step = 1.0 / (4.0 * n * max_c)
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    no_improve = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        grad = _gradient_topk(y, c_y, k)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 100 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_next.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive step size
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            step = 1.0 / (4.0 * n * max_c_new)
            
            if no_improve > 20:
                step *= 0.5
                no_improve = 0
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, lr_init=0.01, max_iter=100000):
    """Adam optimization in log-space with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    # Adaptive beta: start coarse, get finer
    c_curr = _full_convolve(a_init, a_init)
    max_cc = float(c_curr.max())
    
    while t < max_iter and time.monotonic() < deadline:
        t += 1
        
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta: decrease over time
        progress = t / max_iter
        beta = max_cc * 0.01 * (1.0 - progress) + max_cc * 0.0001
        
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient in z-space
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % 100 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        
        if t % 5000 == 0:
            current_lr = max(0.001, current_lr * 0.9)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _peak_targeting(a, deadline, n_peaks=5, n_steps=500):
    """Target specific peaks in the convolution for reduction."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(best_a, best_a)
        # Find top peaks
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        peak_idx = peak_idx[np.argsort(c[peak_idx])[::-1]]
        
        improved = False
        for pk in peak_idx:
            # c[pk] = sum_i a[i] * a[pk-i] (with valid indices)
            # Find which pair contributes most
            lo = max(0, pk - (n - 1))
            hi = min(n - 1, pk)
            if lo > hi:
                continue
            products = best_a[lo:hi+1] * best_a[pk-lo:pk-hi:(-1) if False else 1]  # hmm this is tricky
            # Actually: c[pk] = sum_{i=max(0,pk-n+1)}^{min(n-1,pk)} a[i] * a[pk-i]
            i_vals = np.arange(lo, hi+1)
            j_vals = pk - i_vals
            products = best_a[i_vals] * best_a[j_vals]
            max_prod_idx = np.argmax(products)
            i_star = i_vals[max_prod_idx]
            j_star = j_vals[max_prod_idx]
            
            # Try reducing the larger of a[i_star], a[j_star]
            if best_a[i_star] >= best_a[j_star]:
                src = i_star
            else:
                src = j_star
            
            # Find a good target position (one that minimizes increase in max)
            # Heuristic: position with smallest contribution to all peaks
            max_target = min(10, n // 50)
            targets = rng.choice(n, max_target, replace=False)
            
            for tgt in targets:
                if tgt == src:
                    continue
                # Try moving a small amount
                for frac in [0.01, 0.005, 0.02]:
                    amount = best_a[src] * frac
                    if amount < 1e-15:
                        continue
                    a_try = best_a.copy()
                    a_try[src] -= amount
                    a_try[tgt] += amount
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                        sc = _score_norm(a_try)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_try.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        
        if not improved and step > 100:
            # Try random perturbations
            a_try = best_a.copy()
            idx = rng.integers(0, n)
            a_try[idx] *= (0.9 + 0.2 * rng.random())
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc = _score_norm(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()
    
    return best_a, best_score

def _spectral_init(n, rng, alpha=1.0):
    """Initialize using spectral method: flat magnitude with random phase."""
    N = _next_pow2(2 * n)
    # Create target flat spectrum
    target_mag = np.ones(N // 2 + 1)
    # Random phase
    phase = rng.random(N // 2 + 1) * 2 * np.pi
    spectrum = target_mag * np.exp(1j * phase)
    # Make it Hermitian for real output
    X = np.zeros(N, dtype=complex)
    X[:N//2+1] = spectrum
    for i in range(1, N//2 + 1):
        X[N-i] = np.conj(X[i])
    
    a = np.fft.irfft(X, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "broad_gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _coordinate_polish(a, deadline, n_steps=5000):
    """Coordinate descent: try small multiplicative changes to individual elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        # Try a range of multiplicative factors
        for factor in [0.95, 0.9, 1.05, 1.1, 0.85, 1.15]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    break
    
    return best_a, best_score

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses top-K max gradient, spectral initialization, peak targeting, adaptive FISTA,
Adam in log-space, and multi-length exploration with time-budget management.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_conv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _gradient_topk(a, c, k):
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    # Sharp top-k with small smoothing among top-k
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros_like(c)
    top_vals = c[idx]
    ts = (top_vals - top_vals.min()) / (top_vals.max() - top_vals.min() + 1e-30)
    w[idx] = ts + 0.1
    w = w / w.sum()
    return _grad_conv(a, w)


def _gradient_softmax(a, c, beta):
    n = len(a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return _grad_conv(a, w)


def _fista_adaptive(a_init, deadline, max_iter=50000, mode="topk", k_val=None, beta_init=None):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())

    if mode == "topk":
        if k_val is None:
            k_val = max(10, n // 100)
        step = 1.0 / (4.0 * n * max_c)
    else:
        if beta_init is None:
            beta_init = 0.01 * max_c
        step = 1.0 / (4.0 * n * max_c)

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    no_improve = 0
    best_no_improve_score = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)

        if mode == "topk":
            grad = _gradient_topk(y, c_y, k_val)
        else:
            # Adaptive beta: decrease over iterations
            beta = beta_init * (0.8 ** (it / 5000.0))
            grad = _gradient_softmax(y, c_y, beta)

        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_next.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 20:
                step *= 0.7
                no_improve = 0
                # Also try a small random perturbation to escape
                rng_escape = np.random.default_rng(it)
                a_next = a_next + 0.001 * rng_escape.standard_normal(n)
                a_next = np.maximum(a_next, 0)
                a_next = a_next / a_next.sum()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, lr_init=0.01, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    c_curr = _full_convolve(a_init, a_init)
    max_cc = float(c_curr.max())

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        progress = min(1.0, t / max_iter)
        beta = max_cc * (0.02 * (1.0 - progress) + 0.0005)

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 200 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if t % 10000 == 0:
            current_lr = max(0.0005, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_targeting(a, deadline, n_peaks=5, n_steps=3000):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        peak_idx = np.argpartition(c, -n_peaks)[-n_peaks:]
        peak_idx = peak_idx[np.argsort(c[peak_idx])[::-1]]

        improved = False
        for pk in peak_idx:
            lo = max(0, pk - (n - 1))
            hi = min(n - 1, pk)
            if lo > hi:
                continue
            i_vals = np.arange(lo, hi + 1)
            j_vals = pk - i_vals
            products = best_a[i_vals] * best_a[j_vals]
            max_pi = np.argmax(products)
            i_star = i_vals[max_pi]
            j_star = j_vals[max_pi]

            if best_a[i_star] >= best_a[j_star]:
                src = i_star
            else:
                src = j_star

            n_targets = min(20, n // 20)
            targets = rng.choice(n, n_targets, replace=False)

            for tgt in targets:
                if tgt == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.003]:
                    amount = best_a[src] * frac
                    if amount < 1e-18:
                        continue
                    a_try = best_a.copy()
                    a_try[src] -= amount
                    a_try[tgt] += amount
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                        sc = _score_norm(a_try)
                        if sc < best_score - 1e-15:
                            best_score = sc
                            best_a = a_try.copy()
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break

        if not improved and step % 50 == 0:
            a_try = best_a.copy()
            idx = rng.integers(0, n)
            a_try[idx] *= (0.9 + 0.2 * rng.random())
            a_try = np.maximum(a_try, 0)
            a_try = a_try / a_try.sum()
            sc = _score_norm(a_try)
            if sc < best_score:
                best_score = sc
                best_a = a_try.copy()

    return best_a, best_score


def _spectral_init(n, rng):

    # Target: flat magnitude spectrum

    phase[0] = 0.0
    X = np.zeros(N, dtype=complex)
    X[:N // 2 + 1] = target_mag * np.exp(1j * phase)
    for i in range(1, N // 2 + 1):
        X[N - i] = np.conj(X[i])
    a = np.fft.irfft(X, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "broad_gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-1.5 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "flat_noise":
        rng = np.random.default_rng(99)
        a = 1.0 + 0.3 * rng.standard_normal(n)
        a = np.maximum(a, 0.1)
        return a / a.sum()
    elif kind == "trapezoid":
        t = np.linspace(0, 1, n)
        alpha = 0.2
        a = np.ones(n)
        a[:int(alpha * n)] = np.linspace(0.5, 1.0, int(alpha * n))
        a[-int(alpha * n):] = np.linspace(1.0, 0.5, int(alpha * n))
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _coordinate_polish(a, deadline, n_steps=30000):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    # Precompute which indices contribute to the top peaks for targeted search
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        for factor in [0.97, 0.94, 1.03, 1.06, 0.9, 1.1]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_try.copy()
                    break

    return best_a, best_score


def _block_optimize(a, deadline, n_blocks=8):
    """Optimize blocks of the sequence alternately."""
    n = len(a)
    block_size = max(1, n // n_blocks)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)

    for it in range(200):
        if time.monotonic() > deadline:
            break
        b = it % n_blocks
        lo = b * block_size
        hi = min(n, (b + 1) * block_size)

        # Try perturbing this block
        a_try = best_a.copy()
        block = a_try[lo:hi].copy()
        block *= (0.95 + 0.1 * rng.random(hi - lo))
        a_try[lo:hi] = block
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 1e-15:
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_score - 1e-15:
                best_score = sc
                best_a = a_try.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        best_seq = _make_profile(4096, "uniform")
        best_score = _score(best_seq)
        print(f"No parent, uniform n=4096, score={best_score:.10f}")

    # Determine lengths to explore
    if parent is not None:
        n_parent = len(parent)
        lengths_to_try = [n_parent]
        # Explore nearby and different lengths
        candidates = [n_parent, n_parent + 512, n_parent - 512,
                      n_parent * 2, n_parent // 2,
                      4096, 8192, 6144, 2048]
        for c in candidates:
            if 1024 <= c <= 16384 and c not in lengths_to_try:
                lengths_to_try.append(c)
    else:
        lengths_to_try = [4096, 8192, 2048, 6144]

    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    # Sort: parent first, then by size (larger = more DOF)
    if parent is not None:
        np_ = len(parent)
        lengths_to_try.sort(key=lambda x: (x != np_, -x))
    else:
        lengths_to_try.sort(key=lambda x: -x)

    # Keep at most 4 lengths to focus effort
    lengths_to_try = lengths_to_try[:4]
    print(f"Lengths to try: {lengths_to_try}")

    # Time allocation
    total_time = deadline - t_start - 80.0
    # 45% primary, rest split
    primary_time = total_time * 0.45
    if len(lengths_to_try) > 1:
        secondary_time_each = (total_time - primary_time) / (len(lengths_to_try) - 1)
    else:
        secondary_time_each = 0

    for li, n_target in enumerate(lengths_to_try):
        if time.monotonic() > deadline - 40:
            break

        if li == 0:
            phase_end = t_start + 40.0 + primary_time
        else:
            phase_end = min(deadline - 40, time.monotonic() + max(30, secondary_time_each))

        if phase_end <= time.monotonic() + 10:
            continue

        time_left = phase_end - time.monotonic()
        print(f"\n=== n={n_target}, time={time_left:.0f}s ===")

        # Build initializations
        inits = []
        if parent is not None:
            if n_target == len(parent):
                inits.append(("parent", parent.copy()))
            else:
                inits.append(("parent_interp", _interp_profile(parent, n_target)))
        inits.append(("raised_cosine", _make_profile(n_target, "raised_cosine")))
        inits.append(("broad_gaussian", _make_profile(n_target, "broad_gaussian")))
        inits.append(("flat_noise", _make_profile(n_target, "flat_noise")))
        if n_target <= 8192:
            inits.append(("spectral", _spectral_init(n_target, rng)))
            inits.append(("trapezoid", _make_profile(n_target, "trapezoid")))

        best_for_n = None
        best_score_for_n = np.inf

        # Allocate time per init
        time_per_init = max(10.0, (phase_end - time.monotonic() - 5) / len(inits))

        for ii, (name, a_init) in enumerate(inits):
            if time.monotonic() > phase_end - 5:
                break

            a_work = a_init.copy()
            a_work = np.maximum(a_work, 0)
            s = a_work.sum()
            if s < 1e-15:
                a_work = np.ones(n_target) / n_target
            else:
                a_work = a_work / s

            sc_init = _score(a_work)
            if sc_init < best_score_for_n:
                best_score_for_n = sc_init
                best_for_n = a_work.copy()

            if ii < 3:
                print(f"  [{name}] init score={sc_init:.10f}")

            # Skip expensive optimization if already worse than global best by a lot
            if sc_init > best_score * 1.05 and ii > 0:
                continue

            # Phase A: FISTA top-k (coarse to fine)
            if time.monotonic() < phase_end - 10:
                fista_deadline = min(phase_end - 10, time.monotonic() + time_per_init * 0.5)
                k_val = max(10, n_target // 80)
                a_work, sc_f = _fista_adaptive(a_work, fista_deadline, max_iter=30000, mode="topk", k_val=k_val)
                a_work = a_work / a_work.sum()
                sc_f_full = _score(a_work)
                if sc_f_full < best_score_for_n:
                    best_score_for_n = sc_f_full
                    best_for_n = a_work.copy()

            # Phase B: FISTA softmax (finer)
            if time.monotonic() < phase_end - 8:
                c_tmp = _full_convolve(a_work, a_work)
                max_tmp = float(c_tmp.max())
                fista2_deadline = min(phase_end - 8, time.monotonic() + time_per_init * 0.25)
                a_work, sc_f2 = _fista_adaptive(a_work, fista2_deadline, max_iter=20000,
                                                 mode="softmax", beta_init=0.005 * max_tmp)
                a_work = a_work / a_work.sum()
                sc_f2_full = _score(a_work)
                if sc_f2_full < best_score_for_n:
                    best_score_for_n = sc_f2_full
                    best_for_n = a_work.copy()

            # Phase C: Adam fine
            if time.monotonic() < phase_end - 5:
                adam_deadline = min(phase_end - 3, time.monotonic() + time_per_init * 0.25)
                a_opt, sc_opt = _adam_logspace(a_work, adam_deadline, lr_init=0.008, max_iter=100000)
                sc_opt_full = _score(a_opt)
                if sc_opt_full < best_score_for_n:
                    best_score_for_n = sc_opt_full
                    best_for_n = a_opt.copy()

            if best_for_n is not None:
                print(f"  [{name}] best so far: {best_score_for_n:.10f}")

        # Update global
        if best_for_n is not None and best_score_for_n < best_score:
            best_score = best_score_for_n
            best_seq = best_for_n.copy()
            print(f"  >> New global best: {best_score:.10f} (n={n_target})")

    # === Refinement phases on global best ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_best = len(best_seq)
        print(f"\n=== Refinement (n={n_best}, score={best_score:.10f}) ===")

        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        # Peak targeting
        if time.monotonic() < deadline - 25:
            pt_deadline = min(deadline - 25, time.monotonic() + 40.0)
            a_work, sc_pt = _peak_targeting(a_work, pt_deadline, n_peaks=8, n_steps=5000)
            sc_pt_full = _score(a_work)
            if sc_pt_full < best_score:
                best_score = sc_pt_full
                best_seq = a_work.copy()
                print(f"  Peak targeting: {sc_pt_full:.10f}")

        # Adam fine polish
        if time.monotonic() < deadline - 20:
            adam_deadline = min(deadline - 20, time.monotonic() + 60.0)
            a_work, sc_adam = _adam_logspace(a_work, adam_deadline, lr_init=0.003, max_iter=150000)
            sc_adam_full = _score(a_work)
            if sc_adam_full < best_score:
                best_score = sc_adam_full
                best_seq = a_work.copy()
                print(f"  Adam polish: {sc_adam_full:.10f}")

        # Block optimization
        if time.monotonic() < deadline - 15:
            blk_deadline = min(deadline - 15, time.monotonic() + 30.0)
            a_work, sc_blk = _block_optimize(a_work, blk_deadline, n_blocks=16)
            sc_blk_full = _score(a_work)
            if sc_blk_full < best_score:
                best_score = sc_blk_full
                best_seq = a_work.copy()
                print(f"  Block opt: {sc_blk_full:.10f}")

        # Coordinate descent
        if time.monotonic() < deadline - 10:
            cd_deadline = min(deadline - 10, time.monotonic() + 30.0)
            a_work, sc_cd = _coordinate_polish(a_work, cd_deadline, n_steps=50000)
            sc_cd_full = _score(a_work)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_work.copy()
                print(f"  Coord polish: {sc_cd_full:.10f}")

    # === Multi-restart from best ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\n=== Multi-restart (score={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = 3
        per_restart = max(5.0, (deadline - time.monotonic() - 15.0) / n_restarts)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            if ri == 0:
                # Small perturbation
                z = np.log(np.maximum(a_base, 1e-15)) + 0.01 * rng.standard_normal(n_best)
                a_start = _softmax_compute(z)
            elif ri == 1:
                # Shift
                sh = rng.integers(1, max(2, n_best // 512))
                a_start = np.roll(a_base, sh).copy()
                a_start = a_start / a_start.sum()
            else:
                # Blend with different profile
                alt = _make_profile(n_best, "raised_cosine")
                a_start = 0.8 * a_base + 0.2 * alt
                a_start = np.maximum(a_start, 1e-10)
                a_start = a_start / a_start.sum()

            sc_r = _score(a_start)
            if sc_r > best_score * 1.02:
                continue

            r_deadline = time.monotonic() + per_restart
            a_r, sc_r2 = _adam_logspace(a_start, r_deadline, lr_init=0.005, max_iter=80000)
            sc_r_full = _score(a_r)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = a_r.copy()
                print(f"  Restart {ri}: {sc_r_full:.10f}")

    # === Final ultra-fine ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        fin_deadline = deadline - 2
        a_fin, sc_fin = _adam_logspace(a_fin, fin_deadline, lr_init=0.001, max_iter=50000)
        sc_fin_full = _score(a_fin)
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin.copy()
            print(f"  Ultra-fine: {sc_fin_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}, "
          f"t={time.monotonic()-t_start:.1f}s ===")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
