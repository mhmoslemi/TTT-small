def _next_pow2(n):
    k = 1
    while k < n:

def _convolve(a, b):

    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _interp_to_length(a, n_new):
    """Interpolate sequence to new length, preserving sum."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a * n0)  # Scale to preserve mass
    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s > 0:
        p_new = p_new / s
    else:
        p_new = np.ones(n_new) / n_new
    return p_new

"""
Multi-scale SPO+ optimizer for minimizing 2n*max(a*a)/sum(a)^2.
Key strategies: aggressive large-n exploration (8192, 16384, 32768),
FISTA with Adam polishing, coordinate descent, and diverse restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _autocorr(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _autocorr(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _autocorr(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_weights(a, w):
    n = len(a)
    a_rev = a[::-1]
    N = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, N)
    fr = np.fft.rfft(a_rev, N)
    wc = np.fft.irfft(fw * fr, N)[:len(w) + n - 1]
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista_spo(a_init, deadline, beta_init, min_beta_ratio, beta_decay, max_iter=200000):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    beta = beta_init
    min_beta = min_beta_ratio * beta_init
    L = 4.0 / max(beta, 1e-15)
    step = 1.0 / L
    best_a = a_curr.copy()
    c_best = _autocorr(best_a)
    best_sc = float(c_best.max())
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        if it > 0 and it % 400 == 0:
            beta *= beta_decay
            if beta < min_beta:
                beta = min_beta
            L = 4.0 / max(beta, 1e-15)
            step = 1.0 / L

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        c_y = _autocorr(y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -300, 300, out=exponents)
        w = np.exp(exponents)
        ws = float(w.sum())
        if ws < 1e-300 or not np.isfinite(ws):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / ws

        grad = _grad_from_weights(y, w)
        a_next = _proj_simplex(y - step * grad)

        it += 1
        if it % 150 == 0:
            sc = float(_autocorr(a_next).max())
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = float(_autocorr(a_curr).max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.005, max_iter_total=10000000, max_norm=0.15):
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-15)
    z = np.log(a_safe)
    z = z - z.mean()
    best_a = a_init.copy()
    c_best = _autocorr(best_a)
    best_score = float(c_best.max())
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        while it_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            zm = z - z.max()
            e = np.exp(zm)
            s = e.sum()
            a = e / s if s > 1e-300 else np.ones(n) / n

            c = _autocorr(a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -300, 300, out=c_shifted)
            ew = np.exp(c_shifted)
            sw = float(ew.sum())
            if sw < 1e-300 or not np.isfinite(sw):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = ew / sw

            grad = _grad_from_weights(a, w)
            W = float(np.dot(w, c))
            grad_z = 4.0 * n * a * (grad / 2.0 - W)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > max_norm:
                update *= max_norm / un
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z -= z.mean()

            if t % 200 == 0:
                zm2 = z - z.max()
                e2 = np.exp(zm2)
                s2 = e2.sum()
                a_chk = e2 / s2 if s2 > 1e-300 else np.ones(n) / n
                sc = float(_autocorr(a_chk).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_phase % 3000 == 0:
                current_lr = max(1e-5, current_lr * 0.95)

    zm = z - z.max()
    e = np.exp(zm)
    s = e.sum()
    a_final = e / s if s > 1e-300 else np.ones(n) / n
    sc_final = float(_autocorr(a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=3000):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    c = _autocorr(a)
    best_sc = float(c.max())
    rng = np.random.default_rng(77777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        size = min(512, n) if n > 4096 else n
        indices = rng.choice(n, size=size, replace=False) if n > 4096 else np.arange(n)
        rng.shuffle(indices)
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-12:
                continue
            for factor in [0.99, 0.995, 1.005, 1.01]:
                a[j] = old * factor
                sc = float(_autocorr(a).max())
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
                a[j] = old
        if not improved and r > 50:
            break
    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a * n0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    return p / s if s > 0 else np.ones(n_new) / n_new
