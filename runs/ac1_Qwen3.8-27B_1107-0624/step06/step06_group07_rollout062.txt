    cols = []
    vals = []

            rows.append(k)
            cols.append(j)
            vals.append(a[i])

    # Then rescale b to have the right normalization.

    # Build A matrix: A[k,j] = a[k-j] if 0 <= k-j < n

            cols.append(j)

    prob.solve(solver=cp.ECOS, max_iters=10000)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-12):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, float(result.fun)
    except Exception:
        return a_init, _score(a_init)


def _lbfgs_schedule(a_start, deadline, min_remaining=200.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.12, 0.06, 0.03, 0.015, 0.008, 0.004,
                              0.002, 0.001, 0.0005, 0.0002, 0.0001,
                              0.00005, 0.00002, 0.00001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_per_iter = 0.0025 * (n / 4096.0) ** 1.3
        max_iter = min(12000, max(200, int(remaining / (est_per_iter * 20))))
        a_new, _ = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
        else:
            if i > 4:
                break

    return current_a

def _subgrad_direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=20000):

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0
    c = _full_convolve(a, a)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        c_max = c.max()
        tol = tol_frac * c_max

        if active.sum() == 0:

            active[int(np.argmax(c))] = True

        W = float(np.dot(w, c))

        if s_try < 1e-30:

            step_size *= 0.88

        step_size *= (1.0 - 5e-6)
        step_size = max(step_size, 1e-12)

            break

def _lp_direction(a, M, deadline):
    """Solve LP: max sum(d) s.t. conv(a,d) <= M, d >= 0.
    Uses cvxpy with sparse matrix for moderate n."""
    try:
        import cvxpy as cp
    except ImportError:
        return None

    n = len(a)
    if n > 3000:
        # Downsample for LP
        n_lp = 2048
        t = np.linspace(0, 1, n)
        t_lp = np.linspace(0, 1, n_lp)
        a_lp = np.interp(t_lp, t, a)
        a_lp = np.maximum(a_lp, 1e-10)
        a_lp = a_lp / a_lp.sum()
    else:
        n_lp = n
        a_lp = a.copy()

    n_lp_actual = len(a_lp)
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n_lp_actual - 1):
        j_min = max(0, k - n_lp_actual + 1)
        j_max = min(n_lp_actual - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_lp[k - j])

    A = coo_matrix((vals, (rows, cols)), shape=(2 * n_lp_actual - 1, n_lp_actual))

    d = cp.Variable(n_lp_actual, nonneg=True)
    constraints = [A @ d <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(d)), constraints)

    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None

    if d.value is None:
        return None

    d_val = np.maximum(d.value, 0.0)
    s = d_val.sum()
    if s < 1e-15:
        return None
    d_val = d_val / s

    # Interpolate back to original length
    if len(d_val) != len(a):
        t = np.linspace(0, 1, len(d_val))
        t_full = np.linspace(0, 1, len(a))
        d_full = np.interp(t_full, t, d_val)
        d_full = np.maximum(d_full, 0.0)
        d_full = d_full / d_full.sum()
        return d_full
    return d_val


def _lp_phase(a_start, deadline, min_remaining=300.0):
    """Iteratively apply LP direction finding."""
    n = len(a_start)
    current_a = a_start.copy()

        a_new = (1.0 - t_blend) * current_a + t_blend * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:

            if t_blend < 0.01:
                break

    return current_a


def _perturb_gaussian(a, sigma, rng):

def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.008 * rng.uniform(0.3, 4.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=6):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(8, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.3, 1.8)))
        z[start:end] += rng.uniform(-0.02, 0.02)
    z = z - z.mean()
    return _softmax(z)


def _perturb_antipodal(a, rng):

def _local_search(a, deadline, max_rounds=500):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.008
    c = _full_convolve(best_a, best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        idx = int((rnd * 7919 + 13) % n)
        jdx = int((rnd * 104729 + 7) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        for delta_frac in [0.2, 0.4, 0.6, 0.8]:
            s_pair = best_a[idx] + best_a[jdx]
            new_val_i = s_pair * (1.0 - delta_frac + step)
            new_val_j = s_pair * (delta_frac - step)
            if new_val_i < 0 or new_val_j < 0:
                continue
            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            s = a_try.sum()
            if s < 1e-30:
                continue
            a_try = a_try / s
            c_try = _full_convolve(a_try, a_try)
            sc_try = float(2.0 * n * c_try.max())

            if sc_try < best_sc - 1e-12:
                best_a = a_try.copy()
                best_sc = sc_try
                c = c_try
                improved = True

        if not improved and rnd > 30:
            step *= 0.75
            if step < 1e-7:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with log-sum-exp smoothing,
    LP-based direction finding, subgradient polishing, local coordinate
    search, diversified restarts, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"delta={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Main L-BFGS ===
    print(f"\n=== Phase 1: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, deadline, min_remaining=600.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: LP-based direction finding ===
    print(f"\n=== Phase 2: LP direction (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_lp = _lp_phase(best_seq, deadline, min_remaining=300.0)
        try_update(a_lp, "phase2_lp")
        # Follow up with L-BFGS
        if time.monotonic() < deadline - 250:
            a_lp2 = _lbfgs_schedule(a_lp, deadline, min_remaining=200.0)
            try_update(a_lp2, "phase2_lp_refined")

    # === Phase 3: Z-space subgradient ===
    print(f"\n=== Phase 3: Z-space subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 250,
                                        tol_frac=1e-4, max_steps=12000, lr_init=0.03)
        try_update(a_sg, "phase3_sg_z")

    # === Phase 4: Direct-space subgradient ===
    print(f"\n=== Phase 4: Direct subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 180:
        a_sg2, sc_sg2 = _subgrad_direct_polish(best_seq, deadline - 180,
                                               tol_frac=1e-5, max_steps=20000)
        try_update(a_sg2, "phase4_sg_direct")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - 80 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(8000, max(300, int(remaining / 0.005)))
            a_new, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 6: Local search ===
    print(f"\n=== Phase 6: Local search (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_ls, sc_ls = _local_search(best_seq, deadline - 80, max_rounds=800)
        try_update(a_ls, "local_search")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            ("g_003", "gauss", 0.03),
            ("g_008", "gauss", 0.08),
            ("g_015", "gauss", 0.15),
            ("f_4", "fourier", 4),
            ("f_8", "fourier", 8),
            ("b_6", "block", 6),
            ("b_12", "block", 12),
            ("anti_1", "antipodal", 0),
            ("anti_2", "antipodal", 0),
        ]
        for name, ptype, param in restart_configs:
            if time.monotonic() > deadline - 40:
                break
            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, param, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, param)
            elif ptype == "block":
                p_start = _perturb_block(best_seq, rng, param)
            else:
                p_start = _perturb_antipodal(best_seq, rng)

            sc_start = _score(p_start)
            if sc_start < best_score_val + 0.02:
                a_r = _lbfgs_schedule(p_start, deadline, min_remaining=30.0)
                try_update(a_r, f"restart_{name}")
                if time.monotonic() < deadline - 25 and _score(a_r) < best_score_val + 0.01:
                    a_r2, _ = _subgrad_z_polish(a_r, deadline - 25,
                                                tol_frac=1e-4, max_steps=2000, lr_init=0.02)
                    try_update(a_r2, f"restart_{name}_sg")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        cur_n = len(best_seq)
        for target_n in [2048, 3072, 5120, 6144, 8192]:
            if time.monotonic() > deadline - 30:
                break
            if target_n == cur_n:
                continue
            if target_n > 6000 and time.monotonic() > deadline - 50:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_remaining=20.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 3:
        for rnd in range(5):
            if time.monotonic() > deadline - 3:
                break
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-7
            max_iter = min(5000, 2000 + rnd * 500)
            a_f, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            if time.monotonic() < deadline - 3:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 3,
                                           tol_frac=5e-5, max_steps=3000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
