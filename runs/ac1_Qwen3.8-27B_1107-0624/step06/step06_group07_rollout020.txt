def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _half_to_full(half):
    n = 2 * len(half)
    return np.concatenate([half, half[::-1]])


def _full_to_half(a):
    n = len(a)
    if n % 2 == 0:
        return a[:n // 2].copy()
    else:
        return a[:n // 2 + 1].copy()


def _smooth_max_value(z, beta):
    half = np.exp(z - z.max())
    half = half / half.sum()
    a = _half_to_full(half)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * len(a) * c_max
    return 2.0 * len(a) * (c_max + beta * np.log(s))


def _smooth_max_grad(z, beta):
    half = np.exp(z - z.max())
    half = half / half.sum()
    a = _half_to_full(half)
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_full = 2.0 * n * a * (h - W)
    grad_half = grad_full[:n // 2] + grad_full[n // 2:][::-1]
    return grad_half


def _fista_optimize(z0, beta, max_iter=5000, deadline=None, verbose=False):
    n_half = len(z0)
    n = 2 * n_half
    x = z0.copy()
    y = z0.copy()
    t_val = 1.0
    best_x = z0.copy()
    best_f = _smooth_max_value(z0, beta)
    f_prev = best_f

    for i in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        f_val = _smooth_max_value(y, beta)
        grad = _smooth_max_grad(y, beta)
        grad_norm = np.linalg.norm(grad)

        if grad_norm < 1e-18:
            break

        L = 4.0 * n * (n + 1) / 2
        step = 1.0 / L

        x_new = y - step * grad

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - x)

        f_new = _smooth_max_value(x_new, beta)

        if f_new > f_val + step * (f_val - f_prev):
            x_new = y.copy()
            t_new = 1.0
            f_new = f_val
        else:
            x_new = x_new.copy()

        x = x_new
        y = y_new
        t_val = t_new
        f_prev = f_val
        f_val = f_new

        if f_val < best_f:
            best_f = f_val
            best_x = y.copy()

        if verbose and i % 1000 == 0:
            print(f"    FISTA iter {i}: f={best_f:.10f}")

    half_best = np.exp(best_x - best_x.max())
    half_best = half_best / half_best.sum()
    a_best = _half_to_full(half_best)
    return a_best, _score(a_best), best_x


def _lbfgs_symmetric(z0, beta, n, max_iter=3000, deadline=None):
    def fun(z):
        return _smooth_max_value(z, beta)

    def jac(z):
        return _smooth_max_grad(z, beta)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 100,
                                   'maxls': 50, 'disp': False})
        a_out = _half_to_full(np.exp(result.x - result.x.max()))
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        a_out = _half_to_full(np.exp(z0 - z0.max()))
        return a_out / a_out.sum()


def _beta_schedule(a_start, deadline, min_remaining=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    z0 = _full_to_half(a_start / a_start.sum())
    z0 = np.log(np.maximum(z0, 1e-15))

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 20:
            break

        z_cur = _full_to_half(current_a / current_a.sum())
        z_cur = np.log(np.maximum(z_cur, 1e-15))

        max_iter = min(5000, max(200, int(remaining / 0.002)))
        a_new, sc_new, z_new = _fista_optimize(z_cur, beta, max_iter,
                                               deadline=deadline - min_remaining)
        if sc_new < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc_new
            if verbose_flag:
                print(f"    Beta schedule step {i}: score={sc_new:.10f}")

    return current_a


def _fista_schedule(a_start, deadline, min_remaining=100.0, verbose=False):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    z0 = _full_to_half(a_start / a_start.sum())
    z0 = np.log(np.maximum(z0, 1e-15))

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 30:
            break

        z_cur = _full_to_half(current_a / current_a.sum())
        z_cur = np.log(np.maximum(z_cur, 1e-15))

        max_iter = min(8000, max(500, int(remaining / 0.003)))
        a_new, sc_new, z_new = _fista_optimize(z_cur, beta, max_iter,
                                               deadline=deadline - min_remaining,
                                               verbose=verbose)
        if sc_new < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc_new
        else:
            a_l = _lbfgs_symmetric(z_new, beta, n, max_iter=2000,
                                   deadline=deadline - min_remaining)
            sc_l = _score(a_l)
            if sc_l < current_sc - 1e-12:
                current_a = a_l
                current_sc = sc_l

    return current_a


def _lp_direction(a_current, M, deadline, n_lp=None):
    n = len(a_current)
    if n_lp is None:
        n_lp = min(n, 512)

    a_red = a_current[:n_lp] if n_lp < n else a_current
    a_red = a_red / a_red.sum()

    c = _full_convolve(a_red, a_red)
    c_max = float(c.max())
    M_eff = M * np.sum(a_current[:n_lp]) ** 2 / (2 * n) if n_lp < n else M

    if c_max < 1e-30:
        return None

    A_ub = np.zeros((2 * n_lp - 1, n_lp))
    b_ub = np.full(2 * n_lp - 1, M_eff)

    for k in range(2 * n_lp - 1):
        for i in range(max(0, k - n_lp + 1), min(n_lp, k + 1)):
            A_ub[k, i] = a_red[k - i]

    c_obj = -np.ones(n_lp)
    bounds = [(0, None)] * n_lp

    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if result.success:
        g = np.maximum(result.x, 0)
        if g.sum() > 1e-10:
            g = g / g.sum()
            g_full = np.zeros(n)
            g_full[:n_lp] = g
            if n_lp < n:
                g_full[n_lp:] = g[::-1][:n - n_lp]
            return g_full

    return None


def _lp_mix_iteration(a_current, deadline, n_iterations=50):
    n = len(a_current)
    c = _full_convolve(a_current, a_current)
    M = float(c.max())
    sum_a = float(a_current.sum())
    current_sc = 2.0 * n * M / (sum_a * sum_a)

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break

        g = _lp_direction(a_current, M, deadline)
        if g is None:
            break

        g_sum = g.sum()
        t = 0.1 + 0.1 * it / max(1, n_iterations)
        a_new = (1 - t) * a_current / sum_a + t * g / g_sum
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()

        c_new = _full_convolve(a_new, a_new)
        M_new = float(c_new.max())
        sc_new = 2.0 * n * M_new / 1.0

        if M_new < M:
            a_current = a_new
            M = M_new
        else:
            t *= 0.5
            a_new = (1 - t) * a_current / sum_a + t * g / g_sum
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-10:
                break
            a_new = a_new / a_new.sum()
            c_new = _full_convolve(a_new, a_new)
            M_new = float(c_new.max())
            if M_new < M:
                a_current = a_new
                M = M_new

    return a_current


def _make_profile_sym(n, kind="uniform"):
    half_n = n // 2
    if kind == "uniform":
        half = np.ones(half_n)
    elif kind == "tent":
        t = np.linspace(0, 1, half_n)
        half = 1.0 - t
    elif kind == "parabola":
        t = np.linspace(0, 1, half_n)
        half = 1.0 - t * t
    elif kind == "sine":
        t = np.linspace(0, 1, half_n)
        half = np.sin(np.pi * t / 2)
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, half_n)
        half = 1.0 + 0.1 * np.cos(4 * np.pi * t)
    elif kind == "power":
        t = np.linspace(0, 1, half_n)
        half = (1.0 - t) ** 0.5
    elif kind == "bump":
        t = np.linspace(0, 1, half_n)
        half = np.exp(-4 * (t - 0.3) ** 2) + 0.5 * np.exp(-4 * (t - 0.7) ** 2)
    else:
        half = np.ones(half_n)
    half = np.maximum(half, 1e-10)
    full = _half_to_full(half)
    return full / full.sum()


def _subgradient_polish_sym(a_init, deadline, max_steps=20000, lr_init=0.02, tol_frac=1e-5):
    n = len(a_init)
    z = _full_to_half(a_init / a_init.sum())
    z = np.log(np.maximum(z, 1e-15))
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        half = np.exp(z - z.max())
        half = half / half.sum()
        a = _half_to_full(half)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_full = 2.0 * n * a * (h - W)
        grad_half = grad_full[:n // 2] + grad_full[n // 2:][::-1]

        gn = np.linalg.norm(grad_half)
        if gn < 1e-18:
            break

        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad_half
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if i % 500 == 0:
            half_ch = np.exp(z - z.max())
            half_ch = half_ch / half_ch.sum()
            a_ch = _half_to_full(half_ch)
            sc_ch = _score(a_ch)
            if sc_ch < best_sc - 1e-12:
                best_sc = sc_ch
                best_a = a_ch.copy()
                lr *= 1.02
                stall = 0
            else:
                stall += 1
                if stall > 4:
                    lr *= 0.7
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.9

    half_f = np.exp(z - z.max())
    half_f = half_f / half_f.sum()
    a_f = _half_to_full(half_f)
    sc_f = _score(a_f)
    if sc_f < best_sc:
        best_sc = sc_f
        best_a = a_f.copy()

    return best_a, best_sc


def _perturb_sym(a, sigma, rng, mode="gauss"):
    n = len(a)
    half = _full_to_half(a)
    z = np.log(np.maximum(half, 1e-15))

    if mode == "gauss":
        z = z + sigma * rng.standard_normal(len(z))
    elif mode == "fourier":
        t = np.arange(len(z))
        for _ in range(3):
            freq = int(rng.integers(1, max(2, len(z) // 8)))
            amp = 0.003 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2 * np.pi * freq * t / len(z) + phase)
    elif mode == "block":
        for _ in range(4):
            start = int(rng.integers(0, max(1, len(z) - 10)))
            length = int(rng.integers(5, len(z) // 4))
            end = min(len(z), start + length)
            z[start:end] += rng.uniform(-0.01, 0.01)

    z = z - z.mean()
    half_new = np.exp(z)
    half_new = half_new / half_new.sum()
    return _half_to_full(half_new)


def _interpolate_sym(a, target_n):
    n_cur = len(a)
    half = _full_to_half(a)
    t_old = np.linspace(0, 1, len(half))
    new_half_n = target_n // 2
    t_new = np.linspace(0, 1, new_half_n)
    half_new = np.interp(t_new, t_old, half)
    half_new = np.maximum(half_new, 1e-10)
    full = _half_to_full(half_new)
    return full / full.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Symmetric multi-scale optimizer combining FISTA, L-BFGS, LP direction finding,
    subgradient polishing, and multi-restart diversification.
    """
    global verbose_flag
    verbose_flag = True

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n_init = 2048
        a0 = _make_profile_sym(n_init, "uniform")
        print(f"No parent. Uniform n={n_init}, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    if best_seq is None:
        a0 = _make_profile_sym(1024, "uniform")
        try_update(a0, "fallback_init")

    # === Phase 1: FISTA with beta continuation on existing best ===
    print(f"\n=== Phase 1: FISTA beta continuation (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        a_p1 = _fista_schedule(best_seq, deadline, min_remaining=400.0, verbose=True)
        try_update(a_p1, "phase1_fista")

    # === Phase 2: Subgradient polish ===
    print(f"\n=== Phase 2: Subgradient polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _subgradient_polish_sym(best_seq, deadline - 300,
                                              max_steps=15000, lr_init=0.02, tol_frac=1e-5)
        try_update(a_sg, "phase2_sg")

    # === Phase 3: LP direction mixing ===
    print(f"\n=== Phase 3: LP direction mixing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_lp = _lp_mix_iteration(best_seq, deadline - 150, n_iterations=30)
        try_update(a_lp, "phase3_lp")
        if best_seq is not None:
            a_lp2 = _fista_schedule(best_seq, deadline, min_remaining=120.0)
            try_update(a_lp2, "phase3_lp_refine")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        z0 = _full_to_half(best_seq / best_seq.sum())
        z0 = np.log(np.maximum(z0, 1e-15))

        fine_betas = mb * np.array([2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - 80 - time.monotonic()
            if remaining < 15:
                break
            max_iter = min(3000, max(200, int(remaining / 0.003)))
            a_l = _lbfgs_symmetric(z0, beta, n, max_iter,
                                   deadline=deadline - 80)
            try_update(a_l, f"fine_{i}")
            if a_l is not None and _score(a_l) < best_score_val:
                z0 = _full_to_half(a_l / a_l.sum())
                z0 = np.log(np.maximum(z0, 1e-15))

    # === Phase 5: Multi-restart with diverse shapes ===
    print(f"\n=== Phase 5: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            ("gauss_003", "gauss", 0.03),
            ("gauss_008", "gauss", 0.08),
            ("gauss_015", "gauss", 0.15),
            ("fourier", "fourier", 0),
            ("block", "block", 0),
            ("tent", None, 0),
            ("parabola", None, 0),
            ("sine", None, 0),
            ("power", None, 0),
            ("bump", None, 0),
        ]

        for name, ptype, sigma in restart_configs:
            if time.monotonic() > deadline - 60:
                break

            if ptype is None:
                p_start = _make_profile_sym(len(best_seq), name)
            else:
                p_start = _perturb_sym(best_seq, sigma, rng, mode=ptype)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            if sc_start < best_score_val + 0.02:
                a_r = _fista_schedule(p_start, deadline, min_remaining=50.0)
                try_update(a_r, f"restart_{name}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [1024, 2048, 3072, 5120, 6144]:
            if time.monotonic() > deadline - 40:
                break
            if abs(target_n - len(best_seq)) < 100:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_sym(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")
            if sc_interp < best_score_val + 0.05:
                a_try = _fista_schedule(a_interp, deadline, min_remaining=30.0)
                try_update(a_try, f"n_{target_n}")

    # === Phase 7: Final aggressive polish ===
    print(f"\n=== Phase 7: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 3:
                break

            a_s, _ = _subgradient_polish_sym(best_seq, deadline - 3,
                                             max_steps=5000, lr_init=0.01, tol_frac=1e-6)
            try_update(a_s, f"final_sg_{rnd}")

            if time.monotonic() > deadline - 3:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            z0 = _full_to_half(best_seq / best_seq.sum())
            z0 = np.log(np.maximum(z0, 1e-15))
            beta = mb * 5e-7
            a_f, sc_f, _ = _fista_optimize(z0, beta, max_iter=3000,
                                           deadline=deadline - 3)
            try_update(a_f, f"final_fista_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile_sym(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile_sym(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
