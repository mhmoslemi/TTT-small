"""
Newton-based optimizer with LP-weighted active set for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient + Newton direction with FFT-based Hessian matvec,
periodic LP-based weight optimization, and projected line search.
Key improvements: direct Newton on active set (not smoothed), LP for optimal
subgradient weights, adaptive support reduction, multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _corr_1d(w_full, a_vec, n):
    """Compute cross-correlation: result[j] = sum_k w_full[k] * a_vec[k-j] for j=0..n-1."""
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w_full, N)
    A = np.fft.rfft(a_vec, N)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]


def _hess_matvec(w_full, v, n):
    """Compute H*v = 2 * corr(w_full, v) where H is the active-set Hessian."""
    return 2.0 * _corr_1d(w_full, v, n)


def _project_simplex(v):
    """Project v onto {a: a>=0, sum(a)=1}."""
    n = len(v)
    if n == 0:
        return v
    # Binary search for threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _compute_weights(c, c_max, beta):
    """Compute exponential weights on the convolution entries."""
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -500, 0))
    s = e.sum()
    if s < 1e-300:
        idx = np.argmax(c)
        w = np.zeros_like(c)
        w[idx] = 1.0
        return w
    return e / s


def _solve_lp_weights(a, active_indices, n, deadline):
    """
    Solve LP to find optimal weights lambda on active set that make
    sum_k lambda_k * a[k-i] most constant over the support.
    Returns (lambda_vector, gap).
    """
    s = len(active_indices)
    m = n  # full support for now
    
    # Build B matrix: B[i, k_idx] = a[k - i] where k = active_indices[k_idx]
    # a is defined on [0, n-1], so a[k-i] is nonzero when 0 <= k-i <= n-1
    B = np.zeros((m, s))
    for idx, k in enumerate(active_indices):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            B[lo:hi+1, idx] = a[k - np.arange(lo, hi + 1)]
    
    # LP: min t
    # s.t. B@lambda - M <= t/2  (m constraints)
    #      M - B@lambda <= t/2  (m constraints)
    #      sum(lambda) = 1
    #      lambda >= 0
    # Variables: [lambda (s), M (1), t (1)] total s+2
    # 
    # c = [0, 0, 1] (minimize t)
    c_obj = np.zeros(s + 2)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # B@lambda - M <= t/2  =>  B@lambda - M - t/2 <= 0
    # M - B@lambda <= t/2  =>  -B@lambda + M - t/2 <= 0
    A_ineq = np.zeros((2 * m, s + 2))
    b_ineq = np.zeros(2 * m)
    
    # First m rows: B@lambda - M - t/2 <= 0
    A_ineq[:m, :s] = B
    A_ineq[:m, s] = -1.0
    A_ineq[:m, s+1] = -0.5
    
    # Next m rows: -B@lambda + M - t/2 <= 0
    A_ineq[m:, :s] = -B
    A_ineq[m:, s] = 1.0
    A_ineq[m:, s+1] = -0.5
    
    # Equality: sum(lambda) = 1
    A_eq = np.zeros((1, s + 2))
    A_eq[0, :s] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: lambda >= 0 (indices 0..s-1), M free, t free
    bounds = [(0, None)] * s + [(None, None)] * 2
    
    try:
        if time.monotonic() > deadline:
            return None, None
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status == 0:
            lam = res.x[:s]
            lam = np.maximum(lam, 0)
            lam_sum = lam.sum()
            if lam_sum > 1e-15:
                lam = lam / lam_sum
            # Compute the gap
            val = B @ lam
            gap = float(val.max() - val.min())
            return lam, gap
        else:
            return None, None
    except Exception:
        return None, None


def _newton_optimize(a_init, deadline, max_iter=2000, tol_gap=1e-10):
    """
    Newton optimization on the simplex with active set identification.
    Returns (best_a, best_score, best_gap).
    """
    n = len(a_init)
    a = a_init.copy()
    a = _project_simplex(a)
    best_a = a.copy()
    best_sc = _score(a)
    best_gap = np.inf
    
    # Beta schedule for weight smoothing
    c0 = _full_convolve(a, a)
    c0_max = float(c0.max())
    beta_start = max(c0_max * 0.005, 1e-8)
    
    lr = 0.5
    patience = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-30:
            break
        
        # Adaptive beta: decrease over iterations
        beta = beta_start * (0.95 ** it)
        beta = max(beta, c_max * 0.0001)
        
        # Compute weights
        w_conv = _compute_weights(c, c_max, beta)
        
        # Build w_full on [0, 2n-2]
        w_full = w_conv  # already on [0, 2n-2]
        
        # Compute subgradient: g[j] = 2 * sum_k w[k] * a[k-j]
        g = 2.0 * _corr_1d(w_full, a, n)
        
        # Project gradient onto tangent space of simplex
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        
        if gnorm < 1e-14:
            break
        
        # Newton direction via CG
        # Solve (H + mu*I) * d = -g_t
        mu = 0.01  # regularization
        d = _cg_solve(w_full, -g_t, n, mu, max_cg_iter=100, tol=1e-8)
        
        # Project d onto tangent space
        d = d - np.mean(d)
        
        # If CG failed, use gradient direction
        dnorm = np.linalg.norm(d)
        if dnorm < 1e-15 or not np.isfinite(dnorm):
            d = -g_t
            dnorm = gnorm
        
        # Line search
        alpha = lr
        a_new = a.copy()
        f_old = c_max
        found = False
        for ls in range(40):
            a_try = _project_simplex(a + alpha * d)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a_new = a_try
                found = True
                break
            alpha *= 0.5
        
        if not found:
            # Try gradient direction
            alpha = lr * 0.1
            for ls in range(20):
                a_try = _project_simplex(a - alpha * g_t / gnorm * gnorm)
                c_try = _full_convolve(a_try, a_try)
                f_try = float(c_try.max())
                if f_try < f_old - 1e-13:
                    a_new = a_try
                    found = True
                    break
                alpha *= 0.5
            
            if not found:
                patience += 1
                lr *= 0.5
                if patience > 30:
                    a = best_a.copy()
                    patience = 0
                    lr = 0.5
                continue
        
        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            patience = 0
            lr = min(lr * 1.1, 1.0)
        else:
            patience += 1
            lr *= 0.7
        
        # Compute KKT gap (how constant is the weighted correlation)
        if it % 10 == 0:
            val = _corr_1d(w_full, a, n)
            # Only consider the support (where a > threshold)
            support = a > max(1e-10, np.max(a) * 1e-8)
            if support.sum() > 1:
                gap = float(val[support].max() - val[support].min())
                if gap < best_gap:
                    best_gap = gap
        
        # Periodic LP-based weight refinement
        if it % 40 == 20 and time.monotonic() < deadline - 60:
            active_tol = 0.01 * c_max
            active_mask = c >= (c_max - active_tol)
            active_indices = np.where(active_mask)[0]
            if len(active_indices) > 2 and len(active_indices) < 4 * n:
                lam_opt, gap_lp = _solve_lp_weights(a, active_indices, n, min(deadline - 30, time.monotonic() + 60))
                if lam_opt is not None:
                    # Update weights with LP solution
                    w_new = np.zeros(2 * n - 1)
                    w_new[active_indices] = lam_opt
                    g_lp = 2.0 * _corr_1d(w_new, a, n)
                    g_lp_t = g_lp - np.mean(g_lp)
                    if np.linalg.norm(g_lp_t) > 1e-12:
                        # Take a step in the LP-optimal direction
                        d_lp = _cg_solve(w_new, -g_lp_t, n, 0.01, max_cg_iter=50, tol=1e-8)
                        d_lp = d_lp - np.mean(d_lp)
                        alpha_lp = 0.3
                        for ls in range(15):
                            a_try = _project_simplex(a + alpha_lp * d_lp)
                            c_try = _full_convolve(a_try, a_try)
                            if float(c_try.max()) < c_max:
                                a = a_try
                                break
                            alpha_lp *= 0.5
                    print(f"    LP iter {it}: gap={gap_lp:.2e}, active_set_size={len(active_indices)}")
        
        # Check convergence
        if best_gap < tol_gap:
            print(f"    Converged at iter {it}: gap={best_gap:.2e}")
            break
    
    a_final = _project_simplex(a)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    # Final verification
    c_fin = _full_convolve(best_a, best_a)
    w_fin = _compute_weights(c_fin, float(c_fin.max()), float(c_fin.max()) * 0.0001)
    val_fin = _corr_1d(w_fin, best_a, n)
    support = best_a > max(1e-10, np.max(best_a) * 1e-8)
    final_gap = float(val_fin[support].max() - val_fin[support].min()) if support.sum() > 1 else np.inf
    
    return best_a, best_sc, final_gap


def _cg_solve(w_full, rhs, n, mu, max_cg_iter=100, tol=1e-8):
    """Solve (H + mu*I) * x = rhs using conjugate gradient.
    H*v = 2*corr(w_full, v)."""
    x = np.zeros(n)
    r = rhs - _hess_matvec(w_full, x, n) - mu * x
    p = r.copy()
    rs = float(r @ r)
    rhs_norm = np.linalg.norm(rhs)
    if rhs_norm < 1e-30:
        return x
    
    for i in range(max_cg_iter):
        if np.sqrt(rs) < tol * rhs_norm:
            break
        Hp = _hess_matvec(w_full, p, n) + mu * p
        pHp = float(p @ Hp)
        if abs(pHp) < 1e-30:
            break
        alpha = rs / pHp
        x += alpha * p
        r_new = r - alpha * Hp
        rs_new = float(r_new @ r_new)
        if not np.isfinite(rs_new):
            break
        beta = rs_new / rs
        p = r_new + beta * p
        r = r_new
        rs = rs_new
    
    return x


def _subgradient_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Simple subgradient method with decreasing step size."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set gradient (uniform weights on near-max entries)
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[np.argmax(c)] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        g = 2.0 * _corr_1d(w, a, n)
        g_t = g - np.mean(g)
        gnorm = np.linalg.norm(g_t)
        if gnorm < 1e-14:
            break
        
        step = lr / (1.0 + 0.001 * i)
        a_new = _project_simplex(a - step * g_t)
        sc = _score(a_new)
        
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.03, lr_init)
        else:
            lr *= 0.9
            if (i + 1) % 100 == 0:
                a = best_a + 0.01 * np.random.randn(n)
                a = _project_simplex(a)
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-10 * (t - 0.5) ** 2)
        a = a + 0.05
        return a / a.sum()
    elif kind == "flat_top":
        t = np.linspace(0, 1, n)
        a = np.ones(n) * 0.5 + np.cos(2 * np.pi * t) ** 2 * 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _run_smoothed_lbfgs(a_init, deadline, min_time=30.0):

    n = len(a_init)
    c0 = _full_convolve(a_init, a_init)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    betas = c0_max * np.array([0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    
    for beta in betas:
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break
        
        def fun(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)
            c_max = float(c_local.max())

            e = np.exp(np.clip(shifted, -500, 0))
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def jac(z_local):
            a_local = _softmax(z_local)
            c_local = _full_convolve(a_local, a_local)

            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c_local))
            return 4.0 * n * a_local * (h - W)
        
        try:
            max_iter = min(3000, max(50, int(remaining / 0.05)))
            from scipy.optimize import minimize
            result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                             options={'maxiter': max_iter, 'ftol': 1e-15,

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

    else:
        a_init = parent.copy()
    
    a_newton, sc_newton, gap = _newton_optimize(
        a_init, min(deadline - 300, t_start + 600), max_iter=3000)
    print(f"  Newton done: score={sc_newton:.10f}, gap={gap:.2e}, n={n_newton}")
    update_best(a_newton, sc_newton, "newton_main")
    
    # --- Phase 2: Smoothed L-BFGS from Newton result ---
    print(f"\n[Phase 2] Smoothed L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_lbfgs = _run_smoothed_lbfgs(best_seq, min(deadline - 200, t_start + 900), min_time=50)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_smoothed")
    
    # --- Phase 3: Subgradient polish ---
    print(f"\n[Phase 3] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgradient_polish(best_seq, min(deadline - 150, t_start + 1100),
                                          n_max_steps=3000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")
    
    # --- Phase 4: Second Newton pass (from best) ---
    print(f"\n[Phase 4] Second Newton pass (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 120:
        a_newton2, sc_newton2, gap2 = _newton_optimize(
            best_seq, min(deadline - 120, t_start + 1300), max_iter=2000, tol_gap=1e-12)
        print(f"  Newton2 done: score={sc_newton2:.10f}, gap={gap2:.2e}")
        update_best(a_newton2, sc_newton2, "newton_2")
    
    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 5120, 6144]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        # Interpolate parent to new length
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, n_try)
        a_init_n = np.interp(x_new, x_old, parent)
        a_init_n = np.maximum(a_init_n, 0)
        a_init_n = a_init_n / a_init_n.sum()
        
        a_opt_n, sc_opt_n, _ = _newton_optimize(
            a_init_n, min(deadline - 60, time.monotonic() + 120), max_iter=500)
        update_best(a_opt_n, sc_opt_n, f"n_{n_try}")
    
    # --- Phase 6: Diverse restarts ---
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [
            ("noise", lambda: _perturb(best_seq, 0.05, rng)),
            ("fourier", lambda: _fourier_perturb(best_seq, rng, n_modes=5)),
            ("small_noise", lambda: _perturb(best_seq, 0.02, rng)),
        ]
        for name, gen in restarts:
            if time.monotonic() > deadline - 20:
                break
            a_r = gen()
            a_r2, sc_r2, _ = _newton_optimize(
                a_r, min(deadline - 15, time.monotonic() + 60), max_iter=300)
            update_best(a_r2, sc_r2, f"restart_{name}")
    
    # --- Phase 7: Final fine Newton ---
    print(f"\n[Phase 7] Final fine Newton (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin, gap_fin = _newton_optimize(
            best_seq, deadline - 3, max_iter=500, tol_gap=1e-12)
        update_best(a_fin, sc_fin, "final_newton")
        
        # Quick subgradient
        if time.monotonic() < deadline - 2:
            a_sg2, sc_sg2 = _subgradient_polish(best_seq, deadline - 2,
                                                n_max_steps=200, lr_init=0.005)
            update_best(a_sg2, sc_sg2, "final_sg")
    
    # --- Final validation and return ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
