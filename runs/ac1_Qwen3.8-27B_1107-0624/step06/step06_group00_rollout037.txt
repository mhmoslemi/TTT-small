def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    c = _fft_convolve(a, a)

def _softmax_stable(z):

def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, top_k_dynamic=False, eval_interval=200,
                  clip_norm=0.8):
    """Adam optimizer in z-space with optional dynamic top-k."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        
        a = _softmax_stable(z)
        
        # Dynamic top-k: more aggressive early, focused late
        if top_k_dynamic:
            if frac < 0.3:
                tk = None
            elif frac < 0.6:
                tk = max(10, n // 20)
            elif frac < 0.85:
                tk = max(5, n // 100)
            else:
                tk = 3
        else:
            tk = None
        
        c = _fft_convolve(a, a)
        g = _compute_gradient_z(a, c, T, tk)
        
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * g
        v_buf = b2 * v_buf + (1.0 - b2) * (g * g)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.02, 0.05, 0.08, 0.15]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _multi_start_adam(a_init, deadline, n_starts=3, seed=42):
    """Try multiple perturbations and optimize each."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for i in range(n_starts):
        if time.monotonic() > deadline:
            break
        sigma = 0.02 + 0.02 * i
        z = np.log(np.maximum(best_a, 1e-15))
        z = z + sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax_stable(z)
        
        sub_deadline = deadline - (deadline - time.monotonic()) * (n_starts - 1 - i) / n_starts
        c_p = _fft_convolve(a_pert, a_pert)
        max_cp = float(c_p.max())
        
        a_opt, sc_opt = _adam_z_space(a_pert, sub_deadline,
                                       T_start=0.002 * max_cp, T_end=0.00005 * max_cp,
                                       lr_start=0.03, lr_end=0.003,
                                       max_iter=100000,
                                       top_k_dynamic=True,
                                       eval_interval=150)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS, active-set Adam, LP moves,
    multi-n exploration, and informed restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 2048
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Build LP indices
    rows, cols = _build_lp_indices(n)
    print(f"LP indices built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP iterations (0 - 80s)
    phase1_end = min(deadline, t_start + 80.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0
    while time.monotonic() < phase1_end:
        a, M = _lp_step(a, rows, cols, time_limit=10)
        sc = _score_norm(a)
        lp_count += 1
        if sc < prev_sc - 1e-10:
            stall = 0
        else:
            stall += 1
        full_sc = _score(a)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()
        if lp_count % 5 == 0:
            print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
        if stall > 10:
            break
        prev_sc = sc
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: L-BFGS with annealing (80s - 400s)
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_lbfgs, sc_lbfgs = _lbfgs_phase(best_a, phase2_end - 3,
                                          T_start=0.01 * max_c, T_end=0.0002 * max_c,
                                          max_iter=200, top_k=None, m=30)
        full_sc = _score(a_lbfgs)
        print(f"Phase 2 L-BFGS: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_lbfgs.copy()

    # Phase 3: L-BFGS with top-k (400s - 650s)
    phase3_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase3_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        # Use decreasing top-k
        for tk in [max(10, n // 50), max(5, n // 100), 3]:
            if time.monotonic() > phase3_end:
                break
            a_tk, sc_tk = _lbfgs_phase(best_a, min(phase3_end - 2, time.monotonic() + 70),
                                        T_start=0.003 * max_c, T_end=0.00005 * max_c,
                                        max_iter=150, top_k=tk, m=30)
            full_sc = _score(a_tk)
            print(f"  L-BFGS top-k={tk}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_tk.copy()

    # Phase 4: Active-set Adam with dynamic top-k (650s - 1000s)
    phase4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase4_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_adam, sc_adam = _adam_z_space(best_a, phase4_end - 3,
                                         T_start=0.003 * max_c, T_end=0.00002 * max_c,
                                         lr_start=0.035, lr_end=0.003,
                                         max_iter=500000,
                                         top_k_dynamic=True,
                                         eval_interval=150)
        full_sc = _score(a_adam)
        print(f"Phase 4 Adam-dynamic: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # Phase 5: Multi-n exploration (1000s - 1350s)
    phase5_end = min(deadline, t_start + 1350.0)
    n_best = len(best_a)
    if time.monotonic() < phase5_end:
        # Try larger n
        for n_try in [4096, 8192]:
            if time.monotonic() > phase5_end - 100:
                break
            if n_try == n_best:
                continue
            a_try = _interp_to_length(best_a, n_try)
            c_try = _fft_convolve(a_try, a_try)
            max_ct = float(c_try.max())
            sub_dl = min(phase5_end - 20, time.monotonic() + 120)
            a_opt, sc_opt = _adam_z_space(a_try, sub_dl,
                                           T_start=0.002 * max_ct, T_end=0.00003 * max_ct,
                                           lr_start=0.03, lr_end=0.002,
                                           max_iter=200000,
                                           top_k_dynamic=True,
                                           eval_interval=200)
            full_sc = _score(a_opt)
            print(f"  Multi-n n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()
                n_best = n_try
                rows, cols = _build_lp_indices(n_best)
        # Try smaller n
        if n_best > 2048:
            if time.monotonic() < phase5_end - 50:
                a_small = _interp_to_length(best_a, 2048)
                c_sm = _fft_convolve(a_small, a_small)
                max_csm = float(c_sm.max())
                sub_dl = min(phase5_end - 5, time.monotonic() + 60)
                a_opt_s, sc_opt_s = _adam_z_space(a_small, sub_dl,
                                                   T_start=0.002 * max_csm,
                                                   T_end=0.00003 * max_csm,
                                                   lr_start=0.03, lr_end=0.002,
                                                   max_iter=100000,
                                                   top_k_dynamic=True,
                                                   eval_interval=150)
                full_sc_s = _score(a_opt_s)
                print(f"  Multi-n n=2048: score={full_sc_s:.10f}, t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc_s) and full_sc_s < best_score:
                    best_score = full_sc_s
                    best_a = a_opt_s.copy()
                    n_best = 2048
                    rows, cols = _build_lp_indices(n_best)

    # Phase 6: Informed restarts (1350s - 1620s)
    phase6_end = min(deadline, t_start + 1620.0)
    n_current = len(best_a)
    if time.monotonic() < phase6_end:
        a_restarts, sc_restarts = _multi_start_adam(best_a, phase6_end - 5,
                                                     n_starts=4, seed=int(seed) + 100)
        full_sc_r = _score(a_restarts)
        print(f"Phase 6 restarts: score={full_sc_r:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_r) and full_sc_r < best_score:
            best_score = full_sc_r
            best_a = a_restarts.copy()

    # Phase 7: LP + Adam hybrid polish (1620s - 1750s)
    phase7_end = min(deadline, t_start + 1750.0)
    n_curr = len(best_a)
    if time.monotonic() < phase7_end:
        # A few more LP steps
        rows_cur, cols_cur = _build_lp_indices(n_curr)
        for _ in range(3):
            if time.monotonic() > phase7_end - 30:
                break
            best_a, M = _lp_step(best_a, rows_cur, cols_cur, time_limit=8)
            sc = _score(best_a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                print(f"  LP polish: score={sc:.10f}")
        
        # Final Adam
        if time.monotonic() < phase7_end - 10:
            c_fin = _fft_convolve(best_a, best_a)
            max_cf = float(c_fin.max())
            a_final, sc_final = _adam_z_space(best_a, phase7_end - 3,
                                               T_start=0.0005 * max_cf, T_end=0.00001 * max_cf,
                                               lr_start=0.015, lr_end=0.001,
                                               max_iter=200000,
                                               top_k_dynamic=True,
                                               eval_interval=100)
            full_sc_f = _score(a_final)
            print(f"  Final Adam: score={full_sc_f:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_f) and full_sc_f < best_score:
                best_score = full_sc_f
                best_a = a_final.copy()

    # Phase 8: Ultra-fine final polish (1750s - deadline)
    if time.monotonic() < deadline - 5:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_ultra, sc_ultra = _adam_z_space(best_a, deadline - 3,
                                           T_start=0.0001 * max_cf, T_end=0.000001 * max_cf,
                                           lr_start=0.005, lr_end=0.0002,
                                           max_iter=300000,
                                           top_k_dynamic=True,
                                           eval_interval=50,
                                           clip_norm=0.3)
        full_sc_u = _score(a_ultra)
        print(f"Phase 8 ultra polish: score={full_sc_u:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_u) and full_sc_u < best_score:
            best_score = full_sc_u
            best_a = a_ultra.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
