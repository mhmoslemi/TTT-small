```python
import numpy as np
import time

def _next_pow2(n):
    return 1 << (n - 1).bit_length() if n > 1 else 2

def _full_conv(a):
    n = len(a)
    out_len = 2 * n - 1
    N = _next_pow2(out_len)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:out_len]
    return c

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_conv(a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    # Assumes sum(a) = 1
    n = len(a)
    c = _full_conv(a)
    return 2.0 * n * c.max()

def _proj_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target)
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses diverse asymmetric initializations, Nesterov-accelerated smoothed subgradient (FISTA),
Adam in log-space with annealed softmax, and block coordinate descent polish.

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b):
    n_a = len(a)
    n_b = len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _full_conv(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_conv(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_weights(c, beta):
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_conv(a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    stall_count = 0
    stall_threshold = 5000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:

        c_y = _full_conv(y)
        w = _softmax_weights(c_y, beta)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0 or it < 5:

                    stall_count = 0

        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)

def _make_profile(n, kind, rng=None):

    elif kind == "sinusoidal_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t + np.pi / 3) + 0.2 * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "multi_freq":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t) + 0.15 * np.cos(6.0 * np.pi * t + 0.5) + 0.3 * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.5 * t + 0.3 * t * t
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(99999)
        a_base = np.linspace(0, 1, n)
        a = a_base + 0.3 * np.sin(3.7 * np.pi * a_base + rng.random() * 2 * np.pi)
        a = a + 0.2 * rng.random(n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return e / s


def _block_coordinate_descent(a, deadline, block_size=16, n_rounds=50):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        for _ in range(n // block_size + 1):
            if time.monotonic() > deadline:
                break
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(start + block_size, n)
            block = best_a[start:end].copy()
            s_rest = best_a.sum() - block.sum()
            b_sum = block.sum()
            if b_sum < 1e-15:
                continue

            best_block = block.copy()
            best_local = _score_norm(best_a)

            for attempt in range(20):
                if time.monotonic() > deadline + 5:
                    break
                if attempt == 0:
                    trial_block = block * (1.0 + 0.05 * rng.standard_normal(len(block)))
                elif attempt == 1:
                    trial_block = block * (1.0 + 0.15 * rng.standard_normal(len(block)))
                elif attempt == 2:
                    idx = int(rng.integers(0, len(block)))
                    trial_block = block.copy()
                    trial_block[idx] *= 0.7
                elif attempt == 3:
                    idx = int(rng.integers(0, len(block)))
                    trial_block = block.copy()
                    trial_block[idx] *= 1.3
                elif attempt == 4:
                    trial_block = block * np.exp(0.1 * rng.standard_normal(len(block)))
                else:
                    f = 0.8 + 0.4 * rng.random(len(block))
                    trial_block = block * f
                    trial_block = np.maximum(trial_block, 0.0)

                t_sum = trial_block.sum()
                if t_sum < 1e-15:
                    continue
                total_sum = s_rest + t_sum
                trial_a = best_a.copy()
                trial_a[start:end] = trial_block * (1.0 / total_sum) * n
                # Renormalize to sum=1
                trial_a = trial_a / trial_a.sum()
                sc = _score_norm(trial_a)
                if sc < best_local - 1e-12:
                    best_local = sc
                    best_block = trial_block.copy()

            if np.any(best_block != block):
                total_sum = s_rest + best_block.sum()
                best_a[start:end] = best_block / total_sum
                best_a = best_a / best_a.sum()
                sc_check = _score_norm(best_a)
                if sc_check < best_score:
                    best_score = sc_check

    return best_a, best_score


def _spectral_init(n, rng):
    """Create initialization with flat-ish power spectrum."""
    N = _next_pow2(2 * n)
    num_freqs = N // 2 + 1
    # Random phases, flat magnitude
    phase = rng.random(num_freqs) * 2 * np.pi
    spectrum = np.exp(1j * phase)
    # Make it conjugate symmetric for real output
    spec_full = np.zeros(N, dtype=complex)
    spec_full[:num_freqs] = spectrum
    for k in range(1, num_freqs):
        if N - k < num_freqs:
            spec_full[N - k] = np.conj(spectrum[k])
    a_full = np.fft.irfft(spec_full, N)
    # Take first n elements, rectify
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # Determine lengths to try
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
        lengths_to_try = [n_parent]
        if n_parent <= 4096:
            lengths_to_try.append(8192)
        if n_parent > 2048:
            lengths_to_try.append(n_parent // 2)
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
        lengths_to_try = [4096, 8192]

    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    time_remaining = deadline - t_start
    time_per_length = max(30.0, time_remaining / len(lengths_to_try) - 20.0)

    # Profile types to try (prioritize asymmetric ones)
    profile_kinds = [
        "ramp", "exp_decay", "ramp_steep", "multi_freq",
        "sinusoidal_asym", "quadratic_asym", "exp_decay_fast",
        "uniform", "tapered", "random_smooth"
    ]

    for n_target in lengths_to_try:
        if time.monotonic() > deadline - 40:
            break

        print(f"\n=== n={n_target} (time_left={deadline - time.monotonic():.0f}s) ===")

        # Build initialization list
        inits = []
        labels = []
        if parent is not None:
            if n_target == len(parent):
                inits.append(parent.copy())
                labels.append("parent")
            else:
                inits.append(_interp_profile(parent, n_target))
                labels.append("parent_interp")

        for pk in profile_kinds[:6]:
            inits.append(_make_profile(n_target, pk, rng))
            labels.append(pk)

        # Also try spectral init
        inits.append(_spectral_init(n_target, rng))
        labels.append("spectral")

        best_for_n = None
        best_score_for_n = np.inf

        # Allocate time per init
        n_inits = len(inits)
        time_per_init = max(15.0, (min(deadline - 30, time.monotonic() + time_per_length) - time.monotonic()) / n_inits)

        for init_idx, (a_init, label) in enumerate(zip(inits, labels)):
            if time.monotonic() > deadline - 30:
                break

            a_work = a_init.copy()
            a_work = np.maximum(a_work, 1e-15)
            a_work = a_work / a_work.sum()

            sc_init = _score(a_work)
            print(f"  [{label}] init score={sc_init:.8f} t={time.monotonic()-t_start:.0f}s")

            # Phase A: FISTA with coarse-to-fine beta
            c_curr = _full_conv(a_work)
            max_cc = float(c_curr.max())

            betas_coarse = []
            beta = 0.05 * max_cc
            while beta > 0.002 * max_cc:
                betas_coarse.append(beta)
                beta *= 0.5
            if len(betas_coarse) > 8:
                betas_coarse = betas_coarse[:8]

            for i_beta, beta in enumerate(betas_coarse):
                phase_time = time_per_init * (0.6 if i_beta < len(betas_coarse) // 2 else 0.4)
                phase_dl = min(deadline - 30, time.monotonic() + max(5.0, phase_time))
                max_it = max(3000, int((phase_dl - time.monotonic()) * 5000))
                a_work, sc = _fista_smoothed(a_work, beta, phase_dl, max_it)
                a_work = a_work / a_work.sum()
                sc_full = _score(a_work)
                if sc_full < best_score_for_n:
                    best_score_for_n = sc_full
                    best_for_n = a_work.copy()

            # Phase B: Adam log-space fine optimization
            if time.monotonic() < deadline - 35:
                c_curr = _full_conv(a_work)
                max_cc = float(c_curr.max())
                betas_fine = []
                beta = 0.003 * max_cc
                while beta > 1e-5 * max_cc:
                    betas_fine.append(beta)
                    beta *= 0.5
                if len(betas_fine) > 8:
                    betas_fine = betas_fine[:8]

                adam_dl = min(deadline - 25, time.monotonic() + time_per_init * 0.5)
                a_opt, sc_opt = _adam_logspace(
                    a_work, adam_dl,
                    betas_fine, lr_init=0.02, max_iter_total=200000
                )
                sc_opt_full = _score(a_opt)
                if sc_opt_full < best_score_for_n:
                    best_score_for_n = sc_opt_full
                    best_for_n = a_opt.copy()
                    print(f"    Adam improved: {sc_opt_full:.10f}")

            if best_for_n is not None:
                print(f"  [{label}] best after opt: {best_score_for_n:.10f} t={time.monotonic()-t_start:.0f}s")

        # Update global best
        if best_for_n is not None and best_score_for_n < best_score:
            best_score = best_score_for_n
            best_seq = best_for_n.copy()
            print(f"  *** New global best: {best_score:.10f}, n={n_target} ***")

    # Multi-restart on the best found
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Multi-restart (score={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        strategies = [
            ("perturb_small", lambda: _perturb(a_work, 0.01, rng)),
            ("perturb_med", lambda: _perturb(a_work, 0.04, rng)),
            ("roll", lambda: np.roll(a_work, max(1, n_best // 2000)) / (np.roll(a_work, max(1, n_best // 2000)) / (np.roll(a_work, max(1, n_best // 2000))).sum()).sum()),
            ("blend_taper", lambda: (0.8 * a_work + 0.2 * _make_profile(n_best, "tapered")) / (0.8 * a_work + 0.2 * _make_profile(n_best, "tapered")).sum()),
            ("blend_ramp", lambda: (0.8 * a_work + 0.2 * _make_profile(n_best, "ramp")) / (0.8 * a_work + 0.2 * _make_profile(n_best, "ramp")).sum()),
        ]

        per_restart = max(5.0, (deadline - time.monotonic() - 30.0) / len(strategies))

        for i, (name, gen_fn) in enumerate(strategies):
            if time.monotonic() > deadline - 25:
                break
            try:
                p_start = gen_fn()
            except Exception:
                p_start = _perturb(a_work, 0.02, rng)
            p_start = np.maximum(p_start, 1e-15)
            p_start = p_start / p_start.sum()

            c_s = _full_conv(p_start)
            max_cs = float(c_s.max())
            bs = [0.0005 * max_cs, 0.0002 * max_cs, 0.00005 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + per_restart,
                bs, lr_init=0.005, max_iter_total=100000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart [{name}]: {sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # Final ultra-fine Adam polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Final ultra-fine polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_conv(a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00002 * max_cp, 0.000005 * max_cp, 0.000001 * max_cp]
        per_t = max(5.0, deadline - time.monotonic() - 10)
        a_pol, sc_pol = _adam_logspace(
            a_pol, time.monotonic() + per_t,
            betas_ult, lr_init=0.001, max_iter_total=50000
        )
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: {sc_pol_full:.10f}")

    # Block coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 8:
        print(f"\n=== Block coordinate descent ===")
        a_cd, sc_cd = _block_coordinate_descent(
            best_seq, min(deadline - 5, time.monotonic() + 12),
            block_size=max(8, len(best_seq) // 256), n_rounds=30
        )
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Block CD: {sc_cd_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"Final score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
```
