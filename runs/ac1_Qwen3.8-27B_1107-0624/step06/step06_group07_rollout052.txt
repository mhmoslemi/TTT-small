from scipy import sparse
from scipy.optimize import linprog

def solve_lp(f, M):
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: for each k=0,...,2n-2, sum_j f_j * g_{k-j} <= M
    # Build sparse matrix
    rows = []
    cols = []
    data = []
    
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            i = k - j  # index into g
            rows.append(k)
            cols.append(i)
            data.append(f[j])
    
    A = sparse.csc_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)

    bounds = [(0, None)] * n

    else:

"""
LP-guided optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming with active-set reduction for fast iterations,
combined with log-sum-exp gradient polishing and multi-restart strategies.
"""
import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog, minimize


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(f):
    """Build sparse matrix A such that A @ g = f * g (full convolution)."""
    n = len(f)
    rows_parts = []
    cols_parts = []
    data_parts = []
    for i in range(n):
        k_end = min(i + n - 1, 2 * n - 2)
        length = k_end - i + 1
        rows_parts.append(np.arange(i, k_end + 1))
        cols_parts.append(np.full(length, i, dtype=np.int32))
        data_parts.append(f[:length])
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    data = np.concatenate(data_parts)
    return sparse.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_reduced(f, M, active_rows, max_time=30.0):
    """Solve reduced LP: max sum(g) s.t. A[active] @ g <= M, g >= 0."""
    n = len(f)
    # Build submatrix for active rows
    A_sub = _submatrix_by_rows(f, active_rows)
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_rows), M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': max_time, 'presolve': True})
    
    if res.success and res.x is not None:
        return res.x
    return None


def _submatrix_by_rows(f, row_indices):
    """Build submatrix of convolution matrix for given rows."""
    n = len(f)
    rows_parts = []
    cols_parts = []
    data_parts = []
    for k in row_indices:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        length = j_max - j_min + 1
        if length <= 0:
            continue
        # A[k, i] = f[k-i] for i in range
        i_vals = np.arange(k - j_max, k - j_min + 1)
        rows_parts.append(np.full(length, k))
        cols_parts.append(i_vals.astype(np.int32))
        data_parts.append(f[j_min:j_max + 1])
    
    if not rows_parts:
        return sparse.csc_matrix((len(row_indices), n))
    
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    data = np.concatenate(data_parts)
    return sparse.csc_matrix((data, (rows, cols)), shape=(len(row_indices), n))

"""
LP-guided optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming with active-set constraint generation for fast
direction-finding, combined with log-sum-exp gradient polishing (L-BFGS),
multi-restart diversification, and adaptive sequence length exploration.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    s = float(a.sum())
    if s < 1e-15:
        return a
    return a / s


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_submatrix(f, row_indices):
    n = len(f)
    rows_parts = []
    cols_parts = []
    data_parts = []
    for k in row_indices:
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        length = j_max - j_min + 1
        if length <= 0:
            continue
        i_vals = np.arange(k - j_max, k - j_min + 1)
        rows_parts.append(np.full(length, k, dtype=np.int32))
        cols_parts.append(i_vals.astype(np.int32))
        data_parts.append(f[j_min:j_max + 1])
    if not rows_parts:
        return sparse.csc_matrix((len(row_indices), n))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    data = np.concatenate(data_parts)
    return sparse.csc_matrix((data, (rows, cols)), shape=(len(row_indices), n))


def _lp_direction(f, M, active_rows, time_limit=10.0):
    n = len(f)
    A_sub = _build_conv_submatrix(f, active_rows)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_rows), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = float(g.sum())
            if gs < 1e-15:
                return None
            return g / gs
    except Exception:
        pass
    return None


def _lp_optimize(f, deadline, min_time=60.0, max_iters=200):

    f = _normalize(f.copy())
    best_f = f.copy()
    best_sc = _score(f)
    c = _full_convolve(f, f)
    M = float(c.max())

    for it in range(max_iters):

        c = _full_convolve(f, f)

            s = f_try.sum()

            f = _normalize(f)
            sc = _score(f)
            if sc < best_sc - 1e-12:

                best_f = f.copy()
        else:

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_polish(a_start, deadline, min_time=30.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001,
                              0.00005, 0.00002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break
        est_time_per_iter = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(5000, max(50, int(remaining / (est_time_per_iter * 20))))

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-18,
                                       'gtol': 1e-14, 'maxcor': 50,
                                       'maxls': 40, 'disp': False})
            a_new = _softmax(result.x)
            sc = _score(a_new)
            if sc < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc
        except Exception:
            pass

    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.12 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_hump":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * t) * np.exp(-((t - 0.5) ** 2) / 0.05)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
