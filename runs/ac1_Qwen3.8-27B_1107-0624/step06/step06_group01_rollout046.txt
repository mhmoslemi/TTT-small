"""
Optimized step-function sequence minimizer using adaptive smoothed-max gradient descent
with top-k focusing, Adam in log-space, active-set refinement, and multi-length exploration.
Improves on previous 1.505769 by: (1) top-k weighted gradients focusing on active lags,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Gradient of sum_k w_k * c_k(a) where c_k = sum_j a_j * a_{j+k}."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Gradient of the smoothed objective w.r.t. z in a=softmax(z) parameterization."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter, top_k_frac=1.0):
    """FISTA with softmax-smoothed max, optionally focusing on top-k fraction of lags."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / max(beta, 1e-30)
    step = 1.0 / L
    step_min = 1e-10 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    best_c = c_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        if top_k_frac < 1.0:
            k_top = max(1, int(top_k_frac * len(c_y)))
            top_idx = np.argpartition(c_y, -k_top)[-k_top:]
            w_masked = np.zeros_like(w)
            w_masked[top_idx] = w[top_idx]
            w_masked_sum = w_masked.sum()
            if w_masked_sum > 1e-30:
                w = w_masked / w_masked_sum

        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / max(beta, 1e-30)
            step = 1.0 / L
            step = max(step, step_min)

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace_v2(a_init, deadline, beta_schedule, lr_init, max_iter_total, top_k_frac=1.0):
    """Adam in log-space with top-k focusing and adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init
    patience = 0
    patience_limit = 2000

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            if top_k_frac < 1.0:
                k_top = max(1, int(top_k_frac * len(c)))
                top_idx = np.argpartition(c, -k_top)[-k_top:]
                w_masked = np.zeros_like(w)
                w_masked[top_idx] = w[top_idx]
                wms = w_masked.sum()
                if wms > 1e-30:
                    w = w_masked / wms

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    patience = 0
                else:
                    patience += 1

            if patience >= patience_limit:
                current_lr *= 0.7
                patience = 0
                if current_lr < 1e-5:
                    break

            if it_in_phase % 10000 == 0:
                current_lr = max(1e-5, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "chirp_abs":
        t = np.linspace(0, 1, n)
        a = np.abs(np.cos(2.0 * np.pi * 3.0 * t * (t - 1))) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

                best_score_for_n = sc_ref_full
                best_for_n = a_ref.copy()
            print(f"  Active-set: score={sc_ref_full:.10f} t={time.monotonic()-t_start:.0f}s")

        # Update global best
        if best_score_for_n < best_score:
            best_score = best_score_for_n
            best_seq = best_for_n.copy()
            print(f"  ** New global best: score={best_score:.10f}, n={n_target}")

    # Diversification: try a few different initializations
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Diversification (current best={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_work = best_seq.copy() / best_seq.sum()

        strategies = [
            ("perturb_small", lambda: _perturb(a_work, 0.015, rng)),
            ("perturb_med", lambda: _perturb(a_work, 0.04, rng)),
            ("shift", lambda: np.roll(a_work, rng.integers(2, max(3, n_best // 2000))).copy()),
            ("blend_tapered", lambda: (0.8 * a_work + 0.2 * _make_profile(n_best, "tapered")) / (0.8 * a_work + 0.2 * _make_profile(n_best, "tapered")).sum()),
        ]

        per_strat = (deadline - time.monotonic() - 30.0) / len(strategies)

        for name, gen_fn in strategies:
            if time.monotonic() > deadline - 25:
                break
            try:
                p_start = gen_fn()
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()
            except Exception:
                continue

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0004 * max_cs, 0.00015 * max_cs, 0.00005 * max_cs]

            strat_deadline = time.monotonic() + max(5.0, per_strat)
            p_best, sc_best = _adam_logspace_v2(
                p_start, strat_deadline,
                bs, lr_init=0.006, max_iter_total=200000, top_k_frac=0.3
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  {name}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")
            else:
                print(f"  {name}: score={sc_full:.10f} (no improvement)")

    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Final polish (score={best_score:.10f}) ===")
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [5e-5 * max_cp, 2e-5 * max_cp, 1e-5 * max_cp, 5e-6 * max_cp]
        per_t = deadline - time.monotonic() - 8
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace_v2(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.002, max_iter_total=200000, top_k_frac=0.2
            )
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Ultra-fine: score={sc_pol_full:.10f}")

    # Coordinate refinement
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_cd, sc_cd = _coordinate_refine(best_seq, deadline - 3)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Coord refine: score={sc_cd_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
