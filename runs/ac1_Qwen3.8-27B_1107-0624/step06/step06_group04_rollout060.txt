Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1: returns 2n*max(a*a)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_logspace(a, c, w, n):
    """Gradient of 2n*dot(w,c) w.r.t. log-space z."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)

def _fista_beta(a_init, beta, deadline, max_iter=10000):

    n = len(a_init)
    a_curr = a_init.copy()

    a_prev = a_curr.copy()

    max_c_init = float(c_init.max())
    L = 4.0 * n / beta
    step = 1.0 / L

    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:

        c_y = _full_convolve(y, y)

            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc = _score_norm(a_curr)
    if sc < best_score:

        best_a = a_curr.copy()
    
    return best_a, best_score

    a_norm = a_init.copy()

        inner_t = 0
        while t < max_iter:

            inner_t += 1

def _lbfgs_logspace(a_init, beta, deadline, max_iter=5000):
    """L-BFGS-B optimization in log-space with log-sum-exp smoothing."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    
    z0 = np.log(np.maximum(a_norm, 1e-15))
    z0 = z0 - z0.mean()
    
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    
    def objective(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s_w = float(e.sum())
        if s_w < 1e-300 or not np.isfinite(s_w):
            return 2.0 * n * cm  # approximate
        val = 2.0 * n * float(np.sum((c / s_w * e)))  # this is 2n * dot(c, softmax(c/beta))
        # Actually, let's just use 2n * max for the objective to track
        # But for gradient, we need the smooth version
        return float(2.0 * n * (cm + beta * np.log(s_w)))  # log-sum-exp * 2n
    
    def gradient(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s_w = float(e.sum())
        if s_w < 1e-300 or not np.isfinite(s_w):
            return np.zeros(n)
        w = e / s_w
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        return 2.0 * n * a * (h - W)
    
    # Actually, let me be more careful with the objective.
    # f(z) = 2n * logsumexp(c(z) / beta) where c(z) = conv(softmax(z), softmax(z))
    # The gradient of logsumexp(c/beta) w.r.t. z is:
    # (1/beta) * sum_j (dc[j]/dz) * exp(c[j]/beta) / sum_j exp(c[j]/beta)
    # = (1/beta) * dot(dc/dz, w) where w = softmax(c/beta)
    
    # Let me define the objective and gradient properly.
    
    def obj_and_grad(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s_w = float(e.sum())
        w = e / s_w
        # Objective: 2n * (cm + beta * log(s_w)) = 2n * logsumexp(c/beta)
        obj_val = 2.0 * n * (cm + beta * np.log(s_w))
        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        # d logsumexp / dz_i = (1/beta) * dot(dc/dz_i, w)
        # dc[j]/dz_i = 2 * a[i] * (w-related term)
        # Actually: dc[j]/da[i] = a[j-i] (for appropriate j-i)
        # da[i]/dz[k] = a[i](delta_ik - a[k])
        # So dc[j]/dz[k] = sum_i a[i](delta_ik - a[k]) * dc[j]/da[i]
        # = a[k] * dc[j]/da[k] - a[k] * sum_i a[i] * dc[j]/da[i]
        # = a[k] * (h[k] - W)  [this is what we had before, without the 2n factor]
        
        # So d obj / dz_k = 2n * (1/beta) * sum_j w[j] * dc[j]/dz_k
        # = 2n * (1/beta) * a[k] * (sum_j w[j] * dc[j]/da[k] - a[k] * sum_j w[j] * sum_i a[i] dc[j]/da[i])
        # Hmm, this is getting complicated. Let me just use the formula:
        # grad_z[k] = 2n/beta * a[k] * (H[k] - a[k] * sum(H))
        # where H = conv(w, a_rev)[n-1:2n-1] represents sum_j w[j] * a[j-k] (or similar)
        
        # Actually, from the previous code:
        # h[k] = sum_j w[j] * a_rev[j - (k - (n-1))] ... this is the cross-correlation
        # W = dot(w, c) = sum_j w[j] * c[j]
        # grad[k] = 2n * a[k] * (h[k] - W) / beta
        # Wait, the 2n/beta factor... let me be careful.
        
        # f(z) = 2n * logsumexp(c/beta)
        # df/dz_k = 2n * (1/beta) * sum_j (dc[j]/dz_k) * w[j]
        # where w[j] = exp(c[j]/beta) / sum exp(c[j']/beta)
        
        # dc[j]/dz_k = a[k] * (d c[j]/da[k]) - a[k] * sum_i a[i] * (dc[j]/da[i])
        # d c[j]/da[k] = a[j-k+n-1] (the appropriate element of reversed a)
        #   Actually c = a * a (full convolution), so c[j] = sum_i a[i] a[i+j-(n-1)]
        #   dc[j]/da[k] = a[k+j-(n-1)] if 0 <= k+j-(n-1) < n, else 0
        #   = reversed_a[k - (j-(n-1))] = reversed_a[k - j + n - 1]
        
        # So sum_j w[j] * dc[j]/da[k] = sum_j w[j] * a_rev[k - j + n - 1]
        # This is (w * a_rev)[k + n - 1] where * is full convolution
        # = wc[k + n - 1] where wc = conv(w, a_rev)
        
        # And sum_j w[j] * sum_i a[i] * dc[j]/da[i] = sum_j w[j] * c[j] = W
        
        # So df/dz_k = 2n/beta * a[k] * (wc[k+n-1] - W)
        # = 2n/beta * a[k] * (h[k] - W) where h = wc[n-1:2n-1]
        
        grad = (2.0 * n / beta) * a * (h - W)
        return obj_val, grad
    
    # Use scipy L-BFGS-B with the objective and gradient
    # We need to track the best exact score separately
    
    n_evals = [0]
    
    def fun(z):
        n_evals[0] += 1
        val, _ = obj_and_grad(z)
        return val
    
    def jac(z):
        # To avoid recomputation, we'll just recompute (scipy L-BFGS-B should cache)
        _, g = obj_and_grad(z)
        return g
    
    try:
        result = scipy_minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                   'gtol': 1e-12, 'factr': 1e12}

        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
    except Exception as ex:
        print(f"  L-BFGS-B error: {ex}")
    
    return best_a, best_score

def _spectral_soften(a, alpha, n_iter, deadline=None):
    """Partially flatten the spectrum."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    N_fft = 2 * n - 1
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Soft flattening: A_new = A * (avg / |A|)^alpha
        ratio = avg_mag / (mag + 1e-30)
        A_new = A * (ratio ** alpha)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

Wait, I need to double-check the L-BFGS-B gradient computation. Let me be more careful.

We have:
- `z` is the log-space parameter (length n)
- `a = softmax(z) = exp(z - max(z)) / sum(exp(z - max(z)))` (but for gradient purposes, we can use `a = exp(z)/S` where `S = sum(exp(z))`)
- `c = a * a` (full convolution, length 2n-1)
- `f(z) = 2n * logsumexp(c / beta)`

The gradient:
`df/dz_k = 2n * (1/beta) * sum_j w[j] * dc[j]/dz_k`

where `w[j] = exp(c[j]/beta) / sum_j' exp(c[j']/beta)`.

Now, `c[j] = sum_i a[i] * a[i + j - (n-1)]` (with the convention that a[k] = 0 for k outside [0, n-1]).
