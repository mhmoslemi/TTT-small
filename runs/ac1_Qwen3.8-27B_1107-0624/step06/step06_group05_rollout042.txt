"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA + Adam + L-BFGS with multi-n exploration, spectral initialization, symmetry exploitation,
and local search fine-tuning.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a) / (s * s)


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_conv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _symmetrize(a):
    return 0.5 * (a + a[::-1])


def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / max(beta, 1e-15)
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        scaled = (c_y - c_max) / beta
        scaled = np.clip(scaled, -50.0, 0.0)
        w = np.exp(scaled)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 64 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter, symmetrize=False):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        if symmetrize:
            a = _symmetrize(a)
            a = _normalize(a)
        c = _full_convolve(a, a)
        cm = float(c.max())
        scaled = (c - cm) / beta
        scaled = np.clip(scaled, -50.0, 0.0)
        e = np.exp(scaled)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad_a = 4.0 * n * (h - W_val * a)
        if symmetrize:
            grad_a = _symmetrize(grad_a)
        if not np.all(np.isfinite(grad_a)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_a
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_a * grad_a)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 20 == 0:
            a_chk = _softmax_compute(z)
            if symmetrize:
                a_chk = _symmetrize(a_chk)
                a_chk = _normalize(a_chk)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    if symmetrize:
        a_final = _symmetrize(a_final)
        a_final = _normalize(a_final)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta, symmetrize=False):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-20))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def make_obj(z):
        a = _softmax_compute(z)
        if symmetrize:
            a = _symmetrize(a)
            a = _normalize(a)
        c = _full_convolve(a, a)
        cm = float(c.max())
        scaled = (c - cm) / beta
        scaled = np.clip(scaled, -50.0, 0.0)
        e = np.exp(scaled)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def make_grad(z):
        a = _softmax_compute(z)
        if symmetrize:
            a = _symmetrize(a)
            a = _normalize(a)
        c = _full_convolve(a, a)
        cm = float(c.max())
        scaled = (c - cm) / beta
        scaled = np.clip(scaled, -50.0, 0.0)
        e = np.exp(scaled)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad_a = 4.0 * n * (h - W_val * a)
        if symmetrize:
            grad_a = _symmetrize(grad_a)
        correction = float(np.dot(a, grad_a))
        g = a * (grad_a - correction)
        if symmetrize:
            g = _symmetrize(g)
        return g

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            make_obj, z0, jac=make_grad, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 40)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        if symmetrize:
            a_opt = _symmetrize(a_opt)
            a_opt = _normalize(a_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    if symmetrize:
        a_best = _symmetrize(a_best)
        a_best = _normalize(a_best)
    return a_best, best_score


def _local_search(a_init, deadline, rng, max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.01

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        idx = rng.integers(0, n)
        delta = rng.standard_normal() * step_size
        a_new = a.copy()
        a_new[idx] = max(0.0, a_new[idx] + delta)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            step_size = max(0.001, step_size * 1.05)
            a = a_new
        else:
            step_size = max(0.0005, step_size * 0.95)

    return best_a, best_score


def _block_local_search(a_init, deadline, rng, block_size=8, max_iter=10000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        idx = rng.integers(0, n - block_size)
        block = a[idx:idx+block_size].copy()
        perturber = rng.standard_normal(block_size) * 0.005
        new_block = block + perturber
        new_block = np.maximum(new_block, 0.0)
        if new_block.sum() < 1e-15:
            continue
        scale = block.sum() / new_block.sum()
        new_block = new_block * scale
        a_new = a.copy()
        a_new[idx:idx+block_size] = new_block
        sc = _score_norm(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def _spectral_flatten(a, n_iters=20):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_padded = np.zeros(N)
    a_padded[:n] = a
    for _ in range(n_iters):
        A = np.fft.rfft(a_padded)
        mag = np.abs(A)
        target = np.full_like(mag, mag[0] / np.sqrt(N))
        target[0] = mag[0]
        phase = np.angle(A)
        new_mag = 0.5 * mag + 0.5 * target
        A_new = new_mag * np.exp(1j * phase)
        a_new_full = np.fft.irfft(A_new, N)
        a_padded = np.maximum(a_new_full[:n], 0.0)
        s = a_padded.sum()
        if s < 1e-15:
            break
        a_padded = a_padded / s
    return a_padded


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.3
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

                a_parent_resized = np.interp(
                    np.linspace(0, 1, n_try),
                    np.linspace(0, 1, len(parent)),
                    parent
                )
                a_try = 0.5 * a_try + 0.5 * a_parent_resized
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
        per_time = min(phase1_end - time.monotonic() - 10, 120.0)
        if per_time < 10:
            continue
        print(f"  Trying n={n_try}, time_budget={per_time:.0f}s")
        a_try, sc_try = _run_pipeline(a_try, time.monotonic() + per_time, use_sym=False)
        sc_try = _score(a_try)
        print(f"  n={n_try}: score={sc_try:.10f}")
        if sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()

    # Phase 2: Detailed optimization on best n (600s)
    print(f"Phase 2: Detailed optimization, best_n={len(best_seq)}")
    n_best = len(best_seq)
    phase2_end = min(deadline, t_start + 850.0)
    a_work = best_seq.copy()
    a_work = _normalize(a_work)

    # Spectral flattening as initialization boost
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    beta_pf = 0.03 * max_cw

    sub_end = min(phase2_end, time.monotonic() + (phase2_end - time.monotonic()) * 0.4)
    n_subphases = 20
    for phase in range(n_subphases):
        if time.monotonic() > sub_end - 2:
            break
        per_time = min(sub_end - time.monotonic() - 1, 25.0)
        if per_time < 2:
            break
        max_it = max(3000, int(per_time * 6000))
        a_work, _ = _fista_opt(a_work, beta_pf, time.monotonic() + per_time, max_it)
        a_work = _normalize(a_work)
        sc = _score_norm(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 5 == 0:
                print(f"  FISTA p{phase}: score={sc / (a_work.sum()**2):.10f} t={time.monotonic()-t_start:.1f}s")
        c_work = _full_convolve(a_work, a_work)
        beta_pf = 0.35 * float(c_work.max())
        if beta_pf < 1e-13:
            beta_pf = 1e-13

    # Adam fine with multiple betas
    c_work = _full_convolve(a_work, a_work)
    max_cc = float(c_work.max())
    betas_fine = [0.0005 * max_cc, 0.00015 * max_cc, 0.00005 * max_cc, 0.000015 * max_cc]
    remaining = phase2_end - time.monotonic() - 5
    if remaining > 20:
        per_beta = remaining / len(betas_fine)
        for beta in betas_fine:
            if time.monotonic() > phase2_end - 5:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.004, max_iter=400000, symmetrize=False)
            a_work = _normalize(a_work)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # Symmetrized Adam attempt
    if time.monotonic() < phase2_end - 30:
        a_sym = _symmetrize(a_work)
        a_sym = _normalize(a_sym)
        sc_sym = _score(a_sym)
        sym_end = min(phase2_end, time.monotonic() + 60.0)
        a_sym, _ = _adam_opt(a_sym, sym_end, 0.0001 * max_cc, lr_init=0.003, max_iter=100000, symmetrize=True)
        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  Sym Adam: score={sc_sym:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 3: L-BFGS refinement (300s)
    print("Phase 3: L-BFGS refinement")
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end - 20:
        a_work = best_seq.copy()
        a_work = _normalize(a_work)
        c_work = _full_convolve(a_work, a_work)
        beta_lb = 0.00002 * float(c_work.max())
        a_lb, _ = _lbfgs_opt(a_work, phase3_end, beta_lb, symmetrize=False)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Local search fine-tuning (350s)
    print("Phase 4: Local search fine-tuning")
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 15:
        a_ls = best_seq.copy()
        a_ls = _normalize(a_ls)

        # Coordinate-wise local search
        ls_end1 = min(phase4_end, time.monotonic() + (phase4_end - time.monotonic()) * 0.6)
        a_ls, _ = _local_search(a_ls, ls_end1, rng, max_iter=200000)
        sc_ls = _score(a_ls)
        if sc_ls < best_score:
            best_score = sc_ls
            best_seq = a_ls.copy()
            print(f"  Local search: score={sc_ls:.10f} t={time.monotonic()-t_start:.1f}s")

        # Block local search
        if time.monotonic() < phase4_end - 10:
            bs_end = phase4_end
            a_ls, _ = _block_local_search(a_ls, bs_end, rng, block_size=4, max_iter=50000)
            sc_bs = _score(a_ls)
            if sc_bs < best_score:
                best_score = sc_bs
                best_seq = a_ls.copy()
                print(f"  Block search: score={sc_bs:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Final ultra-polish with Adam + L-BFGS
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = _normalize(a_pol)
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 8
        if per_t > 15:
            half_t = per_t * 0.5
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + half_t,
                                 0.000003 * max_cp, lr_init=0.001, max_iter=200000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            if time.monotonic() < deadline - 10:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000001 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 5.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    # Phase 6: Symmetrization check
    if best_seq is not None:
        a_sym_final = _symmetrize(best_seq)
        a_sym_final = _normalize(a_sym_final)
        sc_sym = _score(a_sym_final)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym_final.copy()
            print(f"  Final symmetrization improved: score={sc_sym:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
