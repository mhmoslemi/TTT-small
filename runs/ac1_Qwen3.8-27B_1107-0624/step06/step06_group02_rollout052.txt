The score is `2n * max_k c[k] / (sum a)^2`. With sum a = 1, it's `2n * max_k c[k]`.

For the convolution c = a * a (linear):
- c has 2n-1 entries
- c[k] = sum_j a[j] * a[k-j] (with valid indices)

The sum of all c[k] is (sum a)^2 = 1.
The average c[k] is 1/(2n-1).

We want to minimize the max. By the pigeonhole principle, max >= average = 1/(2n-1), giving a lower bound of score >= 2n/(2n-1) ≈ 1.

But this lower bound is not achievable. A better bound comes from the fact that the convolution of a non-negative sequence has specific structural constraints.

For the "ramp" part (k < n-1 and k > n-1), the number of terms in the sum is limited:
- c[k] = sum_{j=0}^{k} a[j]*a[k-j] for k < n
- c[k] = sum_{j=k-n+1}^{n-1} a[j]*a[k-j] for k >= n

So c[0] = a[0]^2, c[1] = 2a[0]a[1], etc.

If all a[j] are equal (= 1/n), then:
- c[k] = (min(k+1, 2n-1-k)) / n^2 for k = 0, ..., 2n-2
- max c[k] = (n) / n^2 = 1/n at k = n-1
- score = 2n * (1/n) = 2

To do better than 2, we need to reduce the center peak c[n-1] = sum a[j]*a[n-1-j].

If we make a[j] smaller at the center and larger at the edges, we can reduce c[n-1] but might increase c[0] = a[0]^2.

def _next_pow2(n):
    k = 1
    while k < n:

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            active.add(int(j))

        active_list = list(active)
        rng.shuffle(active_list)
        active_list = active_list[:300]

        improved = False
        for j in active_list:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting to a small positive value
                for trial in [1e-6, 1e-5, 1e-4]:
                    a[j] = trial / n
                    a_norm = a / a.sum()
                    sc = _score_norm(a_norm)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = old_val
                        break
            else:
                for factor in [0.94, 0.97, 1.03, 1.06]:
                    new_val = old_val * factor
                    a[j] = new_val
                    a_norm = a / a.sum()
                    sc = _score_norm(a_norm)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        old_val = new_val
                    else:
                        a[j] = old_val
                        break

        a = a / a.sum()
        if not improved and r > 80:
            break

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _adam_short(a_init, deadline, beta, lr=0.005, max_iter=50000):
    """Short Adam run for restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        np.clip(c_shifted, -500.0, 500.0, out=c_shifted)
        e = np.exp(c_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        a_rev = a[::-1]
        wc = _convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W_dot = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W_dot)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if it % 100 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Main search function. Returns list of non-negative floats minimizing the score."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # Determine working size
    n_main = 16384

    if parent is not None:
        a0 = _interp_profile(parent, n_main)
        sc0 = _score(a0)
        print(f"Interpolated to n={n_main}: score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        # Try a few structured initializations
        candidates = []
        # Uniform
        a_unif = np.ones(n_main) / n_main
        candidates.append(a_unif)
        # Raised cosine
        j = np.arange(n_main)
        a_cos = np.cos(np.pi * j / (n_main - 1)) ** 2
        a_cos = np.maximum(a_cos, 1e-10)
        a_cos = a_cos / a_cos.sum()
        candidates.append(a_cos)
        # Slightly peaked (wider than delta)
        center = (n_main - 1) / 2.0
        sigma_j = n_main / 6.0
        a_gauss = np.exp(-0.5 * ((j - center) / sigma_j) ** 2)
        a_gauss = a_gauss / a_gauss.sum()
        candidates.append(a_gauss)

        best_init = min(candidates, key=_score)
        sc0 = _score(best_init)
        print(f"Best initialization at n={n_main}: score={sc0:.10f}")
        a0 = best_init.copy()
        best_seq = a0.copy()
        best_score = sc0

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")

    # === Phase 1: FISTA with adaptive beta ===
    print("--- Phase 1: FISTA adaptive ---")
    phase1_deadline = min(deadline, t_start + 500.0)
    beta_init = 0.1 * max_c0
    a_curr, _ = _fista_adaptive(a0, phase1_deadline, beta_init, beta_decay=0.9995, beta_min=1e-14)
    a_curr = a_curr / a_curr.sum()
    sc1 = _score(a_curr)
    print(f"  FISTA done: score={sc1:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a_curr.copy()

    # === Phase 2: Adam log-space ===
    print("--- Phase 2: Adam log-space ---")
    phase2_deadline = min(deadline, t_start + 1100.0)
    c_curr = _convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas_fine = []
    beta = 0.005 * max_cc
    while beta > 1e-15 * max_c0:
        betas_fine.append(beta)
        beta *= 0.4
    if len(betas_fine) > 12:
        betas_fine = [betas_fine[i] for i in range(0, len(betas_fine), len(betas_fine) // 12)]
    betas_fine.append(1e-15 * max_c0)

    a_opt, _ = _adam_logspace(a_curr, phase2_deadline, betas_fine, lr_init=0.006)
    a_opt = a_opt / a_opt.sum()
    sc2 = _score(a_opt)
    print(f"  Adam done: score={sc2:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc2 < best_score:
        best_score = sc2
        best_seq = a_opt.copy()

    # === Phase 3: Diverse restarts ===
    print("--- Phase 3: Diverse restarts ---")
    phase3_deadline = min(deadline, t_start + 1500.0)
    a_work = best_seq.copy()
    a_work = np.clip(a_work, 0, 1000)
    a_work = a_work / a_work.sum()

    n_restarts = 6
    per_restart = max(20.0, (phase3_deadline - time.monotonic() - 30.0) / n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > phase3_deadline - 15:
            break

        if i == 0:
            p_start = _perturb(a_work, 0.015, rng)
        elif i == 1:
            p_start = _perturb(a_work, 0.04, rng)
        elif i == 2:
            # Block shift
            block = n_main // 8
            shift = rng.integers(1, block)
            p_start = a_work.copy()
            p_start[:block] = np.roll(p_start[:block], int(shift))
            p_start = p_start / p_start.sum()
        elif i == 3:
            # Symmetric mirror perturbation
            p_start = a_work.copy()
            half = n_main // 2
            noise = 0.01 * rng.standard_normal(half)
            p_start[:half] = np.maximum(p_start[:half] + noise, 0)
            p_start[half:] = p_start[:half][::-1]
            p_start = p_start / p_start.sum()
        elif i == 4:
            # Add a sinusoidal perturbation
            freq = rng.uniform(1, 5)
            j = np.arange(n_main)
            p_start = a_work * (1 + 0.05 * np.sin(2 * np.pi * freq * j / n_main))
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            p_start = _perturb(a_work, 0.08, rng)

        c_s = _convolve(p_start, p_start)
        max_cs = float(c_s.max())
        beta_r = 0.001 * max_cs

        # Short Adam
        p_best, _ = _adam_short(p_start, time.monotonic() + per_restart * 0.6, beta_r, lr=0.004)
        p_best = p_best / p_best.sum()
        sc_r = _score(p_best)

        if sc_r < best_score:
            best_score = sc_r
            best_seq = p_best.copy()
            a_work = p_best.copy()
            print(f"  Restart {i}: score={sc_r:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Coordinate descent polish ===
    print("--- Phase 4: Coordinate descent ---")
    if best_seq is not None and time.monotonic() < deadline - 150:
        cd_deadline = min(deadline - 120, time.monotonic() + 200)
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        a_cd, _ = _coord_descent_focused(a_cd, cd_deadline)
        a_cd = a_cd / a_cd.sum()
        sc_cd = _score(a_cd)
        print(f"  Coord descent: score={sc_cd:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()

    # === Phase 5: Final ultra-fine Adam ===
    print("--- Phase 5: Final polish ---")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00005 * max_cp, 0.00002 * max_cp, 0.00001 * max_cp,
                     0.000005 * max_cp, 0.000002 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, _ = _adam_logspace(a_pol, time.monotonic() + per_t, betas_ult, lr_init=0.002)
            a_pol = a_pol / a_pol.sum()
            sc_pol = _score(a_pol)
            print(f"  Final polish: score={sc_pol:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 256.0
        # Scale so sum is reasonable (keep as is since score is scale-invariant)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
