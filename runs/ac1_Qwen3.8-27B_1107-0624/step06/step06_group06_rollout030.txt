Multi-stage optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with active-set gradient descent using momentum,
adaptive step sizes, and multi-resolution polishing.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    return e / s

def _active_gradient_simple(a, c, n, tol_frac=0.001):
    """Simpler but correct active gradient via direct computation for small active sets."""

        return np.zeros(n)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros(len(c), dtype=bool)
        active[int(np.argmax(c))] = True
    
    n_active = int(active.sum())
    active_indices = np.where(active)[0]
    
    # If active set is small, compute directly
    if n_active <= 20:
        grad = np.zeros(n)
        for k in active_indices:
            # dc_k/da_i = a[k-i] if 0<=k-i<n, plus a[i-k] if 0<=i-k<n
            for i in range(n):
                if 0 <= k - i < n:
                    grad[i] += a[k - i]
                if 0 <= i - k < n:
                    grad[i] += a[i - k]
        grad /= n_active
        return grad
    else:
        return _active_gradient(a, c, n, tol_frac)

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 50,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, max_passes=100):
    """
    Coordinate descent: for each coordinate i, find the optimal value 
    of a_i that minimizes the max convolution, holding other coordinates fixed.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try adjusting a[i]
            # The contribution of a[i] to c_k is: a[i]*a[k-i] + a[i]*a[i-k] (for valid indices)
            # We want to find a[i] that minimizes max_k c_k
            # Current c_k = c_k_without_i + a[i] * (a[k-i] + a[i-k])
            # where c_k_without_i is the convolution with a[i] set to 0
            
            # This is complex to do exactly. Let's just try a few values.
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            # Try reducing a[i] by various amounts
            for frac in [0.5, 0.7, 0.9, 1.1, 1.5]:
                a_try = a.copy()
                a_try[i] = old_val * (1.0 + (frac - 1.0) * 0.3)
                if a_try[i] < 0:
                    continue
                s = a_try.sum()
                a_try = a_try / s
                c_try_max = float(_full_convolve(a_try, a_try).max())
                if c_try_max < c_max - 1e-15:
                    a = a_try
                    c = _full_convolve(a, a)
                    c_max = c_try_max
                    improved = True
                    break
        
        if not improved:
            break
    
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with extended beta-continuation, active-set gradient descent with 
momentum and backtracking, multi-n exploration, and targeted peak reduction.
Key improvements: tighter active set tolerances, momentum-based refinement,
adaptive step sizes, and expanded search over sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _active_grad_fft(a, active, n):
    """
    Compute gradient of (1/|S|) * sum_{k in S} c_k w.r.t. a,
    where c = conv(a,a) and S is the active set.
    Uses FFT for efficiency.
    """
    n_active = int(active.sum())
    if n_active == 0:
        return np.zeros(n)
    
    N = _next_pow2(4 * n)
    A = np.fft.rfft(a, N)
    Ac = np.conj(A)
    
    act_pad = np.zeros(N, dtype=np.float64)
    act_pad[:2*n-1] = active.astype(np.float64)
    Act = np.fft.rfft(act_pad, N)
    
    # grad_i = (1/|S|) * [sum_j a[j]*active[i+j] + sum_j a[j]*active[i-j]]
    f1 = np.fft.irfft(Ac * Act, N)[:n]
    f2 = np.fft.irfft(A * Act, N)[:n]
    
    return (f1 + f2) / n_active


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_mult=1.0):
    """L-BFGS with adaptive beta continuation - extended schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.003) * beta_mult

    # Extended beta schedule for finer optimization
    betas = base_beta * np.array([
        1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125,
        0.015625, 0.0078125, 0.00390625, 0.001953125,
        0.0009765625, 0.00048828125, 0.000244140625,
        0.0001220703125, 0.00006103515625, 0.000030517578125
    ])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        scale = max(1.0, n / 2048.0)
        max_iter = min(4000, max(20, int(remaining / (0.004 * scale * 6))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_descent(a_init, deadline, lr_init=0.01, momentum_init=0.9, max_steps=50000):
    """
    Active-set gradient descent with momentum and adaptive learning rate.
    Directly targets the peak of the autocorrelation.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    momentum = momentum_init
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < 1e-30:
            break
        
        # Adaptive tolerance: start loose, tighten over time
        progress = min(1.0, i / max_steps)
        tol_frac = 0.005 * (1 - progress) + 0.0005 * progress
        
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        
        # Compute gradient via FFT
        grad = _active_grad_fft(a, active, n)
        
        # Project onto tangent space (sum=1 constraint in log-space means mean-preserving)
        grad = grad - np.mean(grad)
        
        # Adaptive learning rate based on gradient norm
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        
        # Momentum update
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                # Slightly increase LR if making good progress
                lr = min(lr * 1.02, lr_init)
            else:
                patience += 1
                lr *= 0.92
                momentum = max(0.5, momentum * 0.995)
                if patience > 50:
                    # Reset from best
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    v = np.zeros(n)
                    patience = 0
                    lr = lr_init * 0.4
                    momentum = 0.85

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _peak_targeting(a_init, deadline, n_finish=200):
    """
    Targeted peak reduction: identify the top peaks and use local 
    coordinate adjustments to reduce them.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    c = _full_convolve(a, a)
    
    for it in range(n_finish):
        if time.monotonic() > deadline:
            break
        
        c_max = float(c.max())
        peak_k = int(np.argmax(c))
        
        # Find which a_i values most contribute to c[peak_k]
        # c[peak_k] = sum_i a_i * a[peak_k - i] (valid indices)
        contribs = np.zeros(n)
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        for i in range(lo, hi + 1):
            contribs[i] = a[i] * a[peak_k - i]
        
        # Also account for the symmetric contribution
        # The gradient of c[peak_k] w.r.t. a[i] is a[peak_k-i] + a[i-peak_k] (bounded)
        grad_k = np.zeros(n)
        for i in range(n):
            if 0 <= peak_k - i < n:
                grad_k[i] += a[peak_k - i]
            if 0 <= i - peak_k < n:
                grad_k[i] += a[i - peak_k]
        
        # Try reducing the largest contributors
        # Sort by contribution descending
        sorted_idx = np.argsort(contribs)[::-1]
        
        improved = False
        for idx in sorted_idx[:10]:
            if time.monotonic() > deadline:
                break
            old_val = a[idx]
            if old_val < 1e-12:
                continue
            
            # Try reducing this element
            for frac in [0.7, 0.85, 0.95, 0.6, 0.5]:
                a_try = a.copy()
                a_try[idx] = old_val * frac
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                c_try = _full_convolve(a_try, a_try)
                c_try_max = float(c_try.max())
                if c_try_max < c_max * (1 - 1e-10):
                    a = a_try
                    c = c_try
                    c_max = c_try_max
                    improved = True
                    break
            
            if not improved:
                break
        
        if not improved:
            # Try increasing small elements (to spread mass)
            small_idx = np.argsort(a)[:5]
            for idx in small_idx:
                a_try = a.copy()
                a_try[idx] += a.max() * 0.01
                a_try = a_try / a_try.sum()
                c_try = _full_convolve(a_try, a_try)
                c_try_max = float(c_try.max())
                if c_try_max < c_max * (1 - 1e-10):
                    a = a_try
                    c = c_try
                    c_max = c_try_max
                    break
        
        if not improved:
            break
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t) + 0.05 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3, amp_scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, new_n):
    """Resample sequence to new length using linear interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s

        a_pt, sc_pt = _peak_targeting(best_seq, p4_deadline, n_finish=500)
        update_best(a_pt, sc_pt, "peak_targeting")

    # === Phase 5: Multi-n exploration (7% budget) ===
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        # Try a few nearby n values
        n_tries = []
        for delta in [-512, -256, 256, 512, 768]:
            nt = n_parent + delta
            if 1024 <= nt <= 10000:
                n_tries.append(nt)
        
        for n_try in n_tries[:3]:
            if time.monotonic() > deadline - 30:
                break
            a_init = _resample(best_seq, n_try) if best_seq is not None else _make_profile(n_try, "uniform")
            a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=15.0, beta_mult=0.5)
            update_best(a_opt, _score(a_opt), f"n_{n_try}")

    # === Phase 6: Perturbation restarts + final polish (remaining time) ===
    print(f"\n[Phase 6] Perturbation restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Try a few perturbations
        for trial in range(3):
            if time.monotonic() > deadline - 8:
                break
            if trial == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif trial == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3, amp_scale=0.015)
            else:
                p_start = _perturb(best_seq, 0.08, rng)
            
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=5.0, beta_mult=0.3)
            update_best(a_r, _score(a_r), f"perturb_{trial}")
            
            # Quick active descent on the result
            if time.monotonic() < deadline - 5:
                a_r2, sc_r2 = _active_descent(a_r, min(deadline - 3, time.monotonic() + 30), lr_init=0.005, momentum_init=0.9, max_steps=5000)
                update_best(a_r2, sc_r2, f"perturb_{trial}_sg")

    # === Final: one more active descent pass ===
    print(f"\n[Final] Last active descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _active_descent(best_seq, deadline - 2, lr_init=0.003, momentum_init=0.95, max_steps=20000)
        update_best(a_fin, sc_fin, "final_active")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Verify with evaluate_sequence
        eval_sc = evaluate_sequence([float(x) for x in final])
        print(f"Verified with evaluate_sequence: {eval_sc:.10f}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
