```python
"""
Multi-scale optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: adaptive beta schedule, multi-n exploration (4096/8192),
structured inits (sine/hann), active-set-directed refinement, and upsampled parent init.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _smoothed_max_obj(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if beta < 1e-15:
        return 2.0 * n * cm
    c_sh = (c - cm) / beta
    e = np.exp(c_sh)
    s = float(e.sum())
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _smoothed_max_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h * n


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    s_0 = float(a_curr.sum())
    L = 4.0 * n / max(beta, 1e-15)
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        if beta < 1e-15:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            e = np.exp((c_y - c_max) / beta)
            w_sum = float(e.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c_y)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = e / w_sum
        grad = _compute_grad(y, w)
        step_sz = min(step, 1.0 / max(float(np.max(np.abs(grad))), 1e-12))
        a_next = _proj_simplex(y - step_sz * grad)
        if it % 60 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if beta < 1e-15:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            e = np.exp((c - cm) / beta)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        return _smoothed_max_obj(a, beta)

    def grad_fn(z):
        a = _softmax_compute(z)
        g_a = _smoothed_max_grad(a, beta)
        correction = float(np.dot(a, g_a))
        return a * (g_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        a_final = _softmax_compute(best_z)
        return a_final, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 40)), 'gtol': 1e-13, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _active_set_polish(a_init, deadline, max_iter=50000):
    """Polish using exact subgradient (uniform on active set) with diminishing step."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    lr0 = 0.5 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 1e-10 * c_max
        active = np.where(c >= c_max - tol)[0]
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        grad = _compute_grad(a, w)
        lr = lr0 / np.sqrt(it + 1)
        z = z - lr * grad
        a = _softmax_compute(z)
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _make_init(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "raised_cos2":
        t = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))) ** 2
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "blackman":
        t = np.linspace(0, 1, n)
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "sqrt_hann":
        t = np.linspace(0, 1, n)
        a = np.sqrt(0.5 + 0.5 * np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _upsample_parent(parent, n_target):
    """Upsample parent to n_target using sinc-like interpolation."""
    n_src = len(parent)
    if n_src >= n_target:
        return parent[:n_target].copy() / parent[:n_target].sum()
    t_src = np.linspace(0, 1, n_src, endpoint=False)
    t_target = np.linspace(0, 1, n_target, endpoint=False)
    # Use np.interp for simplicity (linear is fine as init)
    a_new = np.interp(t_target, t_src, parent)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _adaptive_fista(a_init, deadline, max_iter, n_beta_steps=20):
    """FISTA with geometric beta decay, adaptive to active set size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    c = _full_convolve(a, a)
    max_c0 = float(c.max())
    best_score = 2.0 * n * max_c0
    best_a = a.copy()

    start_beta = 0.1 * max_c0
    end_beta = 1e-12 * max_c0
    time_per = (deadline - time.monotonic()) / n_beta_steps
    it_per = max(1000, int(max_iter / n_beta_steps))

    for phase in range(n_beta_steps):
        if time.monotonic() > deadline - 2:
            break
        t_frac = phase / max(1, n_beta_steps - 1)
        beta = start_beta * (end_beta / start_beta) ** t_frac
        phase_deadline = min(deadline - 1, time.monotonic() + time_per * 1.2)
        a, sc = _fista_opt(a, beta, phase_deadline, it_per)
        a = a / a.sum()
        sc_full = _score_norm(a)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a.copy()
            if phase % 5 == 0:
                print(f"    AF phase {phase}/{n_beta_steps} beta={beta:.2e}: sn={sc_full:.6f}")

    return best_a, best_score
```
