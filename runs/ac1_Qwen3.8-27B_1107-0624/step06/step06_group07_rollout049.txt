"""
Multi-scale optimizer combining L-BFGS with adaptive log-sum-exp continuation,
Prospero-style frequency domain spectral equalization, subgradient polishing,
structured multi-restart diversification, and targeted coordinate descent
to minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_continuation(a_start, deadline, min_time_remaining=30.0,
                        beta_start_frac=0.10, beta_end_frac=1e-5,
                        n_betas=20):
    n = len(a_start)
    c_cur = _fft_conv(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Geometric beta schedule from beta_start to beta_end
    b_start = max_c * beta_start_frac
    b_end = max_c * beta_end_frac
    if b_end >= b_start:
        b_end = b_start * 1e-3
    ratios = (b_end / b_start) ** (1.0 / (n_betas - 1))
    betas = [b_start * (ratios ** i) for i in range(n_betas)]

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate per-iteration time and set max_iter
        est_time = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(10000, max(200, int(remaining / (est_time * 20))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _prospero_polish(a_init, deadline, max_iters=200, alpha=0.4):
    """Prospero-style frequency domain spectral equalization."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flatten the magnitude spectrum
        # Scale so that the mean of |A|^2 is preserved
        target_mag = np.full_like(mag, np.mean(mag))
        # Preserve DC bin (index 0) to maintain the sum
        # Blend: new_mag = (1-alpha)*mag + alpha*target_mag
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        # Always keep DC component
        new_mag[0] = mag[0]

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Project to non-negative
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        a = a_new

    return best_a


def _prospero_iterative(a_init, deadline, max_iters=500):
    """Iterative Prospero with adaptive alpha and phase updates."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    alpha = 0.3
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Adaptive target: partial flattening
        target_mag = np.full_like(mag, np.mean(mag))
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        new_mag[0] = mag[0]  # preserve DC

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            stall = 0
            a = a_new
            alpha = min(alpha * 1.05, 0.8)
        else:
            stall += 1
            alpha *= 0.95
            if stall > 50:
                break

    return best_a

def _coordinate_descent(a_init, deadline, n_rounds=200):
    """Cyclic coordinate descent: for each coordinate, find optimal value."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Process a random subset of coordinates per round
        n_subset = min(n, max(64, n // 4))
        idxs = rng.choice(n, size=n_subset, replace=False)

        for idx in idxs:
            if time.monotonic() > deadline:
                break

            # Binary search for optimal a[idx] given all others
            lo = 0.0
            hi = a.sum() * 2.0 / n  # upper bound
            cur_val = a[idx]

            # Try a few candidate values around current
            candidates = [cur_val * 0.5, cur_val * 0.7, cur_val * 0.85,
                          cur_val * 1.0, cur_val * 1.15, cur_val * 1.3,
                          cur_val * 1.5, lo, hi]
            candidates = [max(0.0, c) for c in candidates]

            best_trial = cur_val
            best_trial_sc = best_sc

            for cv in candidates:
                a[idx] = cv
                s_now = a.sum()
                if s_now < 1e-15:
                    a[idx] = cur_val
                    continue
                a_norm = a / s_now
                sc_trial = _score(a_norm)
                if sc_trial < best_trial_sc:
                    best_trial_sc = sc_trial
                    best_trial = cv

            if best_trial != cur_val:
                a[idx] = best_trial
                s_now = a.sum()
                if s_now > 1e-15:
                    a = a / s_now
                    sc_now = _score(a)
                    if sc_now < best_sc:
                        best_sc = sc_now
                        best_a = a.copy()
            else:
                a[idx] = cur_val

    return best_a


def _interpolate(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _make_structured_init(n, kind, rng):
    """Create structured initial sequences."""
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "cosine_mod":
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * t) + 0.15 * np.cos(2 * np.pi * 7 * t)
    elif kind == "multi_sine":
        a = 1.0
        for k in range(1, 6):
            a += 0.1 / k * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
    elif kind == "tapered_cos":
        a = np.sin(np.pi * t) ** 0.7 + 0.1
    elif kind == "bimodal":
        a = 1.0 + 0.25 * np.cos(4 * np.pi * t)
    elif kind == "quartic":
        a = 1.0 - 0.3 * (2.0 * t - 1.0) ** 4
    elif kind == "random_smooth":
        # Random smooth via Fourier
        freqs = np.arange(n // 2 + 1)
        amps = rng.exponential(0.1, size=len(freqs)) * np.exp(-freqs / (n / 10.0))
        phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
        a = np.fft.irfft(amps * np.exp(1j * phases), n)
        a = np.abs(a) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _perturb(a, rng, sigma=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):

    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 120:
        a_fine = _lbfgs_continuation(best_seq, deadline - 120,
                                     min_time_remaining=20.0,
                                     beta_start_frac=2e-5, beta_end_frac=1e-7,
                                     n_betas=15)
        try_update(a_fine, "phase5_fine_lbfgs")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        a_cd = _coordinate_descent(best_seq, deadline - 80, n_rounds=100)
        try_update(a_cd, "phase6_coord")

    # === Phase 7: Multi-restart with diverse initializations ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    restart_configs = [
        ("gauss_002", "gauss", 0.02),
        ("gauss_005", "gauss", 0.05),
        ("gauss_010", "gauss", 0.10),
        ("fourier_3", "fourier", 3),
        ("fourier_6", "fourier", 6),
        ("struct_cos_mod", "struct", "cosine_mod"),
        ("struct_multi_sine", "struct", "multi_sine"),
        ("struct_rand_smooth", "struct", "random_smooth"),
    ]

    for name, rtype, param in restart_configs:
        if time.monotonic() > deadline - 40:
            break

        if rtype == "gauss":
            p_start = _perturb(best_seq, rng, sigma=float(param))
        elif rtype == "fourier":
            p_start = _perturb_fourier(best_seq, rng, n_modes=int(param))
        elif rtype == "struct":
            # Blend structured with current best
            p_struct = _make_structured_init(n_target, str(param), rng)
            p_start = 0.5 * best_seq + 0.5 * p_struct
            p_start = p_start / p_start.sum()
        else:
            continue

        sc_start = _score(p_start)
        print(f"  Restart [{name}]: start={sc_start:.10f}")

        # Quick L-BFGS
        if time.monotonic() < deadline - 30:
            a_r = _lbfgs_continuation(p_start, deadline - 30,
                                      min_time_remaining=20.0,
                                      beta_start_frac=0.05, beta_end_frac=1e-4,
                                      n_betas=12)
            try_update(a_r, f"restart_{name}")

            # Quick subgradient if promising
            if time.monotonic() < deadline - 25 and _score(a_r) < best_score_val + 0.003:
                a_r2, _ = _subgrad_z_polish(a_r, deadline - 25,
                                            tol_frac=1e-4, max_steps=2000, lr_init=0.03)
                try_update(a_r2, f"restart_{name}_sg")

    # === Phase 8: Multi-scale interpolation exploration ===
    print(f"\n=== Phase 8: Multi-scale (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 50:
        for target_n in [1024, 2048, 3072, 5120, 6144]:
            if time.monotonic() > deadline - 40:
                break
            if target_n <= 0:
                continue
            print(f"  Interpolating to n={target_n}...")
            a_interp = _interpolate(best_seq, target_n)
            a_try = _lbfgs_continuation(a_interp, deadline - 20,
                                        min_time_remaining=15.0,
                                        beta_start_frac=0.05, beta_end_frac=5e-4,
                                        n_betas=10)
            try_update(a_try, f"scale_{target_n}")

    # === Phase 9: Final aggressive polish cycle ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 3:
                break

            # Tiny-beta L-BFGS
            a_f = _lbfgs_continuation(best_seq, deadline - 3,
                                      min_time_remaining=2.0,
                                      beta_start_frac=1e-6, beta_end_frac=1e-8,
                                      n_betas=8)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Subgradient
            if time.monotonic() < deadline - 3:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 3,
                                           tol_frac=1e-5, max_steps=3000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

            # Coordinate descent
            if time.monotonic() < deadline - 3:
                a_c = _coordinate_descent(best_seq, deadline - 3, n_rounds=50)
                try_update(a_c, f"final_cd_{rnd}")

            # Prospero
            if time.monotonic() < deadline - 3:
                a_p = _prospero_iterative(best_seq, deadline - 3, max_iters=100)
                try_update(a_p, f"final_prospero_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
