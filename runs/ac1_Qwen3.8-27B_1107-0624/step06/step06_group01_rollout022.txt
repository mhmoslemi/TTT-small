    corr = np.fft.irfft(np.conj(W) * A, N)[:n]
    # But wait, I need both (w ⋆ a)_i and (w ⋆ a)_{-i}
    # (w ⋆ a)_i = corr[i]
    # (w ⋆ a)_{-i} = corr[? ]
    # Actually, let me reconsider. 
    # (w ⋆ a)_i = sum_k w[k+i]*a[k] for k=0..n-1
    # This is exactly the cross-correlation, computable as ifft(conj(FFT(w)) * FFT(a)) with padding
    # But w has length 2n-1 and a has length n, so I need padding of at least 3n-1
    
    # Hmm, let me just use a simpler approach: compute the gradient via two convolutions
    # grad_i = sum_k w_k * a[k-i] + sum_k w_k * a[i-k]
    # First part: for each i, sum over k of w[k]*a[k-i]
    # This is: (w * a)[i] where * is cross-correlation = conv(w, rev(a)) shifted
    
    # You know what, let me just use the matrix-free approach with a single convolution:
    # Note that sum_k w_k * a[k-i] = (w ⋆ a)_i
    # And sum_k w_k * a[i-k] = (w ⋆ a)_{n-1-i}... no
    
    # I'll just compute it directly using two rfft-based cross-correlations.
    # For efficiency, I'll compute corr = ifft(conj(rfft(w, N)) * rfft(a, N))
    # This gives (w ⋆ a)_i for i = 0, 1, ..., N-1 (circular)
    # With N >= (2n-1) + n - 1 = 3n-2, it's the linear cross-correlation.
    # Then grad_i = corr[i] + corr_rev[i] where corr_rev[i] = corr related to the other term.
    
    # Actually, I just realized:
    # sum_k w_k * a[i-k] = sum_k w_k * a[i-k]
    # Let's reverse: = sum_k w[k] * a[i-k] = conv(w, a)[i] (standard convolution!)
    # And sum_k w_k * a[k-i] = sum_k w[k] * a[k-i]
    # = sum_k w[k+i] * a[k] = cross_correlation(w, a)[i]
    
    # So grad_i = conv(w, a)[i] + cross_correlation(w, a)[i]
    
    # For the cross-correlation, I can compute it as:
    # cross_correlation(w, a)[i] = conv(w, rev(a))[i] ... let me verify
    # conv(w, rev(a))[i] = sum_j w[j] * rev(a)[i-j] = sum_j w[j] * a[n-1-i+j]
    # Hmm, that's not quite right either because of the indexing.
    
    # OK I'll just use the FFT cross-correlation formula:
    # (f ⋆ g)[n] = sum_k f[k+n] * conj(g[k])
    # FFT: ifft(conj(FFT(f)) * FFT(g))[n] (with zero padding for linear)
    
    N = _next_pow2(3 * n)  # w has length 2n-1, a has length n, need >= (2n-1)+n-1 = 3n-2
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(Wf) * Af, N)
    # corr[i] = sum_k w[k+i] * a[k] (linear cross-correlation, valid for i=0..n-1)
    
    # Now conv(w, a)[i] for i=0..n-1:
    conv_wa = np.fft.irfft(Wf * Af, N)[:n]
    
    grad = corr[:n] + conv_wa
    return 2.0 * grad  # factor of 2 from the two terms in the derivative


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA for minimizing 2*n*max(conv(a,a)) subject to a>=0, sum(a)=1
    using softmax smoothing of the max with parameter beta."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_conv_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            # Adaptively update L
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        phase_max = min(remain_iter, max(1000, remain_iter // len(beta_schedule)))
        
        while it_in_phase < phase_max:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_conv_grad(a, w)
            # Transform gradient to z-space: d/dz = d/da * da/dz
            # a = softmax(z), da_i/dz_j = a_i*(delta_ij - a_j)
            # grad_z_j = sum_i grad_i * a_i * (delta_ij - a_j) = a_j * (grad_j - sum_i grad_i * a_i)
            weighted = np.dot(grad, a)
            grad_z = a * (grad - weighted)
            
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _block_coordinate_descent(a, deadline, block_size=32, n_rounds=50):
    """Block coordinate descent: optimize small blocks at a time."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(7)

    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Pick random block
        start = rng.integers(0, n - block_size)
        end = start + block_size

        # Get current convolution
        c = _full_convolve(best_a, best_a)
        
        # For the block, compute the "reduced" problem
        # We want to minimize max(c_new) where c_new is the updated convolution
        # Only entries of c that involve the block change
        a_block = best_a[start:end].copy()
        
        # Compute the contribution of the block to the convolution
        # c[k] = sum_i a[i]*a[k-i]
        # The block contributes to c[k] when start <= i < end or start <= k-i < end
        # i.e., k in [start, end+n-2] (from first condition) or k in [start, end+n-2] (same)
        
        # For a quick improvement, just try to minimize the max convolution value
        # by adjusting the block to reduce the peak
        peak_idx = int(np.argmax(c))
        
        # Try to reduce a[start:end] at positions that contribute to the peak
        # Simple heuristic: reduce elements in the block that contribute most to the peak
        contributions = np.zeros(block_size)
        for j in range(block_size):
            i = start + j
            # Contribution of a[i] to c[peak_idx]:
            # c[peak_idx] includes a[i]*a[peak_idx-i] and a[peak_idx-i]*a[i]
            k1 = peak_idx - i
            k2 = i - (peak_idx - i)  # not right
            # Actually c[peak_idx] = sum_m a[m]*a[peak_idx-m]
            # The terms involving a[i] are: a[i]*a[peak_idx-i] (from m=i) 
            # and if peak_idx-i is in the block, a[peak_idx-i]*a[i] (from m=peak_idx-i)
            if 0 <= k1 < n:
                contributions[j] = a_block[j] * best_a[k1] + best_a[i] * best_a[k1]
        
        # Try reducing the most contributing elements
        for factor in [0.7, 0.8, 0.85, 0.9]:
            a_try = best_a.copy()
            # Reduce the top contributors
            top_idx = np.argsort(contributions)[::-1][:block_size // 4]
            for j in top_idx:
                a_try[start + j] *= factor
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    break

    return best_a, best_score


def _spectral_perturb(a, rng, strength=0.1):
    """Perturb the sequence in the spectral domain for diversity."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    
    # Perturb the phases
    magnitudes = np.abs(A)
    phases = np.angle(A)
    phases += strength * rng.standard_normal(len(phases))
    
    # Reconstruct
    A_new = magnitudes * np.exp(1j * phases)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(int(kind) if isinstance(kind, int) else 12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses accurate gradient computation via FFT, FISTA with adaptive Lipschitz estimation,
Adam in log-space with cosine-annealed smoothing, block coordinate descent,
and spectral perturbation for diversity. Explores lengths 4096-16384.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_conv_grad(a, w):
    """
    Compute gradient of sum_k w_k * conv(a,a)[k] with respect to a.
    w has length 2n-1 (the full convolution), a has length n.
    Result: grad_i = sum_k w_k * (a[k-i] + a[i-k]) for i=0..n-1
    Computed as: conv(w, a)[:n] + cross_corr(w, a)[:n]
    """
    n = len(a)
    N = _next_pow2(3 * n + 1)
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a, N)
    conv_part = np.fft.irfft(Wf * Af, N)[:n]
    corr_part = np.fft.irfft(np.conj(Wf) * Af, N)[:n]
    return conv_part + corr_part


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_conv_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    n_phases = len(beta_schedule)
    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        phase_max = min(remain_iter, max(500, remain_iter // n_phases))

        while it_in_phase < phase_max:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_conv_grad(a, w)
            weighted = np.dot(grad, a)
            grad_z = a * (grad - weighted)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buff + (1.0 - b1) * grad_z if False else b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 4000 == 0:
                current_lr = max(0.0002, current_lr * 0.88)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _block_cd(a, deadline, block_size=64, n_rounds=100):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(7)

    for round_idx in range(n_rounds):
        if time.monotonic() > deadline:
            break

        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(start + block_size, n)

        c = _full_convolve(best_a, best_a)
        peak_idx = int(np.argmax(c))

        a_block = best_a[start:end].copy()
        contributions = np.zeros(end - start)
        for j in range(end - start):
            i = start + j
            k1 = peak_idx - i
            if 0 <= k1 < n:
                contributions[j] = abs(a_block[j] * best_a[k1])
            k2 = i - (peak_idx - i)
            if 0 <= (peak_idx - i) < n and k1 != i:
                contributions[j] += abs(best_a[i] * best_a[peak_idx - i])

        improved = False
        for factor in [0.7, 0.8, 0.85, 0.9, 1.1, 1.2]:
            a_try = best_a.copy()
            top_frac = max(1, (end - start) // 4)
            top_idx = np.argsort(contributions)[::-1][:top_frac]
            for j in top_idx:
                a_try[start + j] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    improved = True
                    break

    return best_a, best_score


def _spectral_perturb(a, rng, strength=0.05):
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    magnitudes = np.abs(A)
    phases = np.angle(A)
    phases += strength * rng.standard_normal(len(phases))
    A_new = magnitudes * np.exp(1j * phases)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            adam_end = min(phase_deadline - 15, time.monotonic() + t_alloc * 0.35)
            a_opt, sc_opt = _adam_logspace(
                a_curr, adam_end, betas_fine, lr_init=0.018, max_iter_total=400000
            )
            sc_n = _score_norm(a_opt)
            if sc_n < best_score_n:
                best_score_n = sc_n
                best_for_n = a_opt.copy()

            if best_score_n < best_score:
                best_score = best_score_n
                best_seq = best_for_n.copy()
                print(f"  After Adam: {best_score:.12f}")

        # Phase 3: Block coordinate descent (remaining)
        if time.monotonic() < phase_deadline - 5:
            cd_end = min(phase_deadline - 3, time.monotonic() + t_alloc * 0.1)
            a_cd, sc_cd = _block_cd(best_for_n, cd_end, block_size=48)
            sc_n = _score_norm(a_cd)
            if sc_n < best_score_n:
                best_score_n = sc_n
                best_for_n = a_cd.copy()
            if best_score_n < best_score:
                best_score = best_score_n
                best_seq = best_for_n.copy()
                print(f"  After block CD: {best_score:.12f}")

    # Spectral perturbation restarts
    if best_seq is not None and time.monotonic() < deadline - 25:
        print(f"\n=== Spectral restarts (best={best_score:.12f}) ===")
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_best = len(a_base)

        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 15.0) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > deadline - 12:
                break

            if i == 0:
                a_try = _spectral_perturb(a_base, rng, strength=0.04)
            elif i == 1:
                z = np.log(np.maximum(a_base, 1e-15))
                z += 0.025 * rng.standard_normal(n_best)
                z = z - z.mean()
                a_try = _softmax_compute(z)
            else:
                alt = _make_profile(n_best, "cosine")
                a_try = 0.85 * a_base + 0.15 * alt
                a_try = np.maximum(a_try, 1e-10)
                a_try = a_try / a_try.sum()

            c_try = _full_convolve(a_try, a_try)
            max_ct = float(c_try.max())
            bs = [0.0004 * max_ct, 0.00015 * max_ct, 0.00006 * max_ct, 0.00002 * max_ct]

            a_opt, sc_opt = _adam_logspace(
                a_try, time.monotonic() + max(5.0, per_restart * 0.75),
                bs, lr_init=0.008, max_iter_total=150000
            )
            sc_n = _score_norm(a_opt)
            if sc_n < best_score:
                best_score = sc_n
                best_seq = a_opt.copy()
                print(f"  Restart {i}: {best_score:.12f}")

    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 8:
        print(f"\n=== Final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000004 * max_cp, 0.000001 * max_cp]

        a_pol, sc_pol = _adam_logspace(
            a_pol, deadline - 4, betas_ult, lr_init=0.002, max_iter_total=80000
        )
        sc_n = _score_norm(a_pol)
        if sc_n < best_score:
            best_score = sc_n
            best_seq = a_pol.copy()
            print(f"  Final: {best_score:.12f}")

    # Output
    print(f"\n=== FINAL: score={best_score:.12f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"Final _score: {sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.12f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.12f}")
    return [float(x) for x in fallback]
