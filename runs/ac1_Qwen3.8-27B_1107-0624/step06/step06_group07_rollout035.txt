"""
IRLS (Iteratively Reweighted Least Squares) optimizer with spectral shaping
for minimizing 2*n*max(convolve(a,a))/sum(a)^2.

Key innovations over the previous log-sum-exp + L-BFGS approach:
1. Explicit reweighting focuses directly on autocorrelation peaks (no smoothing parameter tuning)
2. Projected gradient on weighted QP with O(n log n) FFT-based gradient
3. Fienup-style spectral shaping for diverse starting points
4. Active-set subgradient polishing at the end
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _cross_corr_fft(a, w):
    """
    Compute c[i] = sum_j a[j] * w[i+j] for i = 0, ..., n-1.
    w has length 2n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    # Need N >= n + (2n-1) - 1 = 3n-2 for linear correlation
    N = _next_pow2(3 * n - 2)
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w, N)
    # ifft(fft(a) * conj(fft(w)))[k] = sum_j a[j] * w[j-k] (linear, due to zero-padding)
    result = np.fft.ifft(fa * np.conj(fw)).real
    # I want c[i] = sum_j a[j] * w[i+j] = result[-i mod N]
    c = np.empty(n)
    c[0] = result[0]
    if n > 1:
        c[1:] = result[N - np.arange(1, n)][::-1]
    return c


def _weighted_grad(a, w):
    """
    Compute gradient of Q(a) = sum_m w[m] * b[m] where b = convolve(a, a).
    grad[i] = c[i] + w[2*i] * a[i]
    """
    n = len(a)
    c = _cross_corr_fft(a, w)
    grad = c + w[2 * np.arange(n)] * a
    return grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Simple approach: clip and renormalize (approximate but fast for small steps)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return v / s


def _project_simplex_exact(v):
    """Exact projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu) - 1.0
    ind = np.arange(1, n + 1)
    cond = mu - cv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.max(ind[cond])
    theta = cv[i - 1] / i
    return np.maximum(v - theta, 0.0)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = 1.0 - 2.0 * np.abs(t - 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def _irls_optimize(a_init, deadline, min_time=30.0):
    """
    IRLS with p-continuation: minimize weighted autocorrelation,
    gradually increasing emphasis on peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = _make_profile(n, "uniform")
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)
    eps = 1e-12

    # p-continuation: start smooth, get sharper
    p_schedule = [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0, 20.0, 40.0, 80.0]

    for p in p_schedule:
        if time.monotonic() + min_time > deadline:
            break

        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        # Number of gradient steps for this p value
        n_steps = min(500, max(50, int(remaining / (0.005 * (n / 1024.0)))))

        # Initial learning rate for this p
        b = _full_convolve(a, a)
        b_mean = b.mean()
        # Estimate Lipschitz constant
        L_est = 4.0 * n * (b_max / b_mean if b_mean > 0 else 1.0)
        lr = 0.5 / (L_est + 1.0)
        lr_min = lr * 1e-6

        for step_i in range(n_steps):
            if time.monotonic() + min_time > deadline:
                break

            # Compute weights
            b = _full_convolve(a, a)
            b_max = b.max()
            b_mean = max(b.mean(), eps)
            # Weights: emphasize bins close to max
            w = (b / (b_mean + eps)) ** p
            w = w / w.sum() * (2 * n - 1)  # normalize so sum(w) = 2n-1

            # Compute gradient
            grad = _weighted_grad(a, w)

            # Projected gradient step
            a_new = a - lr * grad
            a_new = _project_simplex(a_new)

            # Check if score improved (do every few steps to save time)
            if step_i % 10 == 0 or step_i == n_steps - 1:
                sc = _score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                if sc < _score(a) + 1e-10:
                    a = a_new
                    lr = min(lr * 1.01, 2.0 / (L_est + 1.0))
                else:
                    lr *= 0.8
                    if lr < lr_min:
                        break

        # After p round, do a final check
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        elapsed = time.monotonic()
        if elapsed > 0:
            print(f"  IRLS p={p}: score={best_score:.10f}")

    return best_a, best_score


def _irls_aggressive(a_init, deadline, min_time=20.0):
    """
    More aggressive IRLS: use only active set for weights (hard peaks).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, _score(a)
    a = a / s

    best_a = a.copy()
    best_score = _score(a)
    eps = 1e-12
    lr = 0.02

    for it in range(5000):
        if time.monotonic() + min_time > deadline:
            break

        b = _full_convolve(a, a)
        b_max = b.max()
        tol = 0.01 * b_max
        active = b >= (b_max - tol)
        if active.sum() == 0:
            active = np.zeros(2 * n - 1, dtype=bool)
            active[int(np.argmax(b))] = True

        # Weight on active set only
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / active.sum() * (2 * n - 1)

        grad = _weighted_grad(a, w)

        a_new = a - lr * grad
        a_new = _project_simplex(a_new)

        sc_new = _score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        if sc_new < _score(a) - 1e-14:
            a = a_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.9
            if lr < 1e-8:
                # Try widening active set
                tol *= 2
                if tol > 0.5:
                    break

    return best_a, best_score


def _fienup_iterate(a_init, deadline, min_time=30.0, n_iter_max=2000):
    """
    Fienup/Gerchberg-Saxton style: alternate between time domain (a >= 0)
    and frequency domain (flat spectrum target).
    
    Idea: In frequency domain, |A(w)|^2 relates to autocorrelation.
    We want autocorrelation to be flat, so |A(w)|^2 should be concentrated
    at DC. We iterate:
    1. FFT(a) -> A
    2. Modify |A| to be more DC-concentrated (flatten non-DC parts)
    3. IFFT -> new a
    4. Project to a >= 0
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)
    N = _next_pow2(2 * n)

    # Target spectrum: mostly DC with some tolerance
    target_alpha = 0.9  # fraction of energy to keep at DC

    for it in range(n_iter_max):
        if time.monotonic() + min_time > deadline:
            break

        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compute current spectrum energy distribution
        dc_energy = mag[0] ** 2
        total_energy = np.sum(mag ** 2)
        dc_frac = dc_energy / total_energy if total_energy > 0 else 0

        # Modify magnitude: boost DC, suppress non-DC
        # New target: keep DC, reduce non-DC by factor
        new_mag = mag.copy()
        non_dc = np.arange(N) != 0
        suppress = 0.98  # reduce non-DC components slightly
        new_mag[non_dc] *= suppress

        # Reconstruct
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new).real[:n]

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        if it % 200 == 0 and it > 0:
            print(f"  Fienup it={it}: score={best_score:.10f}, dc_frac={dc_frac:.4f}")

    return best_a, best_score


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=8000, lr_init=0.03):

def _perturb_sequence(a, rng, method="gaussian", strength=0.05):
    """Perturb a sequence for diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    if method == "gaussian":
        z += strength * rng.standard_normal(n)
    elif method == "fourier":
        t = np.arange(n)
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 6)))
            amp = strength * 0.5 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif method == "block":
        n_blocks = 6
        block_size = n // n_blocks
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += strength * 0.3 * rng.uniform(-1, 1)
    else:
        z += strength * rng.standard_normal(n)

    z = z - z.mean()
    # Softmax back
    z = z - z.max()
    e = np.exp(z)
    a_out = e / e.sum()
    return a_out


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()

                a_p2, _ = _irls_optimize(a_p, deadline - 5, min_time=3)
                try_update(a_p2, f"final_pert_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
