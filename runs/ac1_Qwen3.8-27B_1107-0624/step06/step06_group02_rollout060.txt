def _fft_conv(a, b):

def _grad_smooth_max(a, beta):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    conv = _fft_conv(a, a)
    cm = conv.max()
    shifted = (conv - cm) / beta
    np.clip(shifted, -500, 500, out=shifted)
    e = np.exp(shifted)
    w = e / e.sum()
    
    # Compute P = correlate(w, a, 'full')[:n] and Q = convolve(a, w, 'full')[:n]
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    
    # P = ifft(fw * conj(fa))[:n]
    P = np.fft.irfft(fw * np.conj(fa), N)[:n]
    # Q = ifft(fa * fw)[:n]
    Q = np.fft.irfft(fa * fw, N)[:n]
    
    return (P + Q) / beta

def _adam_optimize(a_init, deadline, beta_schedule, lr_init=0.01, max_iter=100000):
    """Adam optimization in log-space with smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        phase_iter = 0
        while phase_iter < max_iter:
            if time.monotonic() > deadline:
                break
            phase_iter += 1
            t += 1
            
            a = _softmax(z)
            grad_a = _grad_smooth_max(a, beta)
            # Transform to z-space
            grad_z = a * (grad_a - np.dot(a, grad_a))
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m = b1 * m + (1 - b1) * grad_z
            v = b2 * v + (1 - b2) * grad_z**2
            mh = m / (1 - b1**t)
            vh = v / (1 - b2**t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            # Clip update
            un = np.linalg.norm(update)
            if un > 1.0:
                update *= 1.0 / un
            
            z -= update
            z = np.clip(z, -100, 100)
            z = z - z.mean()

                a_chk = _softmax(z)
                sc = _score_norm(a_chk)

    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score

Optimized step function sequence generator using Adam in log-space with
smoothed-max gradient, multi-scale exploration (n=4096, 8192), beta annealing,
and multi-restart. Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)

def _score_norm(a):
    n = len(a)
    conv = _fft_conv(a, a)
    return float(2.0 * n * conv.max())


def _score(a):
    n = len(a)
    s = float(a.sum())

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smooth_max(a, beta):
    """Gradient of smooth max of conv(a,a) w.r.t. a, using softmax weighting."""
    n = len(a)
    conv = _fft_conv(a, a)
    cm = conv.max()
    shifted = (conv - cm) / beta
    np.clip(shifted, -500, 500, out=shifted)
    e = np.exp(shifted)
    w = e / e.sum()

    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)

    # P[j] = sum_k w[k]*a[k-j] (cross-correlation)
    P = np.fft.irfft(fw * np.conj(fa), N)[:n]
    # Q[j] = sum_k a[k]*w[k-j] (convolution)
    Q = np.fft.irfft(fa * fw, N)[:n]

    return (P + Q) / beta


def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.01, max_iter=200000):
    """Adam optimizer in log-space with smoothed max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    if not np.isfinite(best_score):
        best_score = 2.0

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)

            if not np.all(np.isfinite(grad_a)):

            grad_z = a * (grad_a - np.dot(a, grad_a))

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)

def _fista_smoothed(a_init, beta, deadline, max_iter=50000):
    """FISTA on simplex with smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 / beta
    step = 1.0 / L

    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        grad_a = _grad_smooth_max(y, beta)
        a_next = y - step * grad_a

        # Project onto simplex (sum=1, a>=0)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc = _score_norm(a_curr)
    if sc < best_sc:
        best_sc = sc
        best_a = a_curr.copy()
    return best_a, best_sc


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _coord_descent(a_init, deadline, max_rounds=300):

    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.95, 0.98, 1.02, 1.05]:
                a[j] = old_val * factor
                sc = _score_norm(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    old_val = a[j]
                else:
                    a[j] = old_val
        a = a / a.sum()
        if not improved and r > 30:
            break

    return best_a, best_sc


def _interp_profile(p, n_new):
    """Interpolate a profile to a new length, preserving shape."""

    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

        a_big = _interp_profile(best_a, n_big)
        sc_big = _score_norm(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_a = a_big.copy()

        c_big = _fft_conv(a_big, a_big)
        max_cb = float(c_big.max())

        # FISTA on big
        per_t = min(60.0, phase2_end - time.monotonic() - 200)
        if per_t > 20:
            a_big, _ = _fista_smoothed(a_big, 0.01 * max_cb,
                                       time.monotonic() + per_t, max_iter=20000)
            a_big = a_big / a_big.sum()
            sc = _score_norm(a_big)
            if sc < best_score:
                best_score = sc
                best_a = a_big.copy()
                print(f"  FISTA n=8192: score={sc:.10f}, t={time.monotonic()-t_start:.1f}s")

        # Adam on big
        betas_big = [max_cb * f for f in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
        per_t = phase2_end - time.monotonic() - 100
        if per_t > 40:
            a_big, sc_big_opt = _adam_logspace(a_big, time.monotonic() + per_t,
                                               betas_big, lr_init=0.006, max_iter=300000)
            sc_big_full = _score_norm(a_big)
            if sc_big_full < best_score:
                best_score = sc_big_full
                best_a = a_big.copy()
                print(f"  Adam n=8192: score={sc_big_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart with diverse strategies ===
    print(f"Phase 3: Multi-restart exploration")
    phase3_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase3_end - 80:
        n_r = len(best_a)
        n_restarts = 6
        per_restart = (phase3_end - time.monotonic() - 40) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 40:
                break

            # Different perturbation strategies
            if i % 3 == 0:
                # Gaussian perturbation in log-space
                z = np.log(np.maximum(best_a, 1e-15))
                sigma = 0.01 + 0.005 * i
                z += sigma * rng.standard_normal(n_r)
                z -= z.mean()
                a_r = _softmax(z)
            elif i % 3 == 1:
                # Circular shift
                shift = rng.integers(1, max(2, n_r // 8))
                a_r = np.roll(best_a, int(shift)).copy()
                a_r = a_r / a_r.sum()
            else:
                # Block shuffle
                block = max(1, n_r // 16)
                a_r = best_a.copy().reshape(-1, block)
                order = rng.permutation(a_r.shape[0])
                a_r = a_r[order].ravel()
                a_r = a_r / a_r.sum()

            c_r = _fft_conv(a_r, a_r)
            max_cr = float(c_r.max())
            betas_r = [max_cr * f for f in [0.003, 0.001, 0.0005, 0.0002, 0.0001]]

            a_r_opt, sc_r = _adam_logspace(a_r, time.monotonic() + per_restart,
                                           betas_r, lr_init=0.005, max_iter=80000)
            sc_r_full = _score_norm(a_r_opt)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_a = a_r_opt.copy()
                print(f"  Restart {i}: score={sc_r_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Ultra-fine Adam polish ===
    print(f"Phase 4: Ultra-fine polish")
    phase4_end = min(deadline, t_start + 1700.0)
    if best_a is not None and time.monotonic() < phase4_end - 30:
        a_pol = best_a.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _fft_conv(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [max_cp * f for f in [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7]]
        per_t = phase4_end - time.monotonic() - 20
        if per_t > 20:
            a_pol, sc_pol = _adam_logspace(a_pol, time.monotonic() + per_t,
                                           betas_ult, lr_init=0.002, max_iter=200000)
            sc_pol_full = _score_norm(a_pol)
            print(f"  Ultra-fine: score={sc_pol_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_a = a_pol.copy()

    # === Phase 5: Coordinate descent ===
    if best_a is not None and time.monotonic() < deadline - 30:
        a_cd = best_a.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 20, time.monotonic() + 80)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=200)
        sc_cd_full = _score_norm(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_a = a_cd.copy()

    # === Final validation ===
    print(f"\nBest: score={best_score:.10f}, n={len(best_a) if best_a is not None else 0}")

    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0).astype(np.float64)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(4096, dtype=np.float64)
        # Verify
        try:
            eval_sc = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator verification: {eval_sc:.10f}")
            if not np.isfinite(eval_sc) or eval_sc > best_score + 0.01:
                print("WARNING: evaluator disagrees, using uniform fallback")
                final = np.ones(4096, dtype=np.float64)
        except Exception as ex:
            print(f"Eval check error: {ex}")

        print(f"FINAL: n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(4096, dtype=np.float64)
    print(f"FALLBACK: uniform n=4096")
    return [float(x) for x in fallback]
