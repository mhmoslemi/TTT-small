def _lp_active_set(a, active_constraints, delta=0.0):
    """LP step using only active constraints."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    target = M * (1.0 + delta)
    
    # Active constraints: indices k where c[k] is close to M
    # Plus some additional constraints for robustness
    
    # Build sparse A for only the active constraints
    rows = active_constraints
    cols_list = []
    vals_list = []
    
    for k in active_constraints:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            cols_list.append(j_range)
            vals_list.append(a[k - j_range])
    
    if not cols_list:
        return a
    
    all_cols = np.concatenate(cols_list)
    all_vals = np.concatenate(vals_list)
    all_rows = np.repeat(active_constraints, [len(cl) for cl in cols_list])
    
    A_sp = sp.csc_matrix((all_vals, (all_rows, all_cols)), shape=(len(active_constraints), n))
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_constraints), target)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_sp, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': 10})
    if res.status != 0:
        return a
    
    g = np.maximum(res.x, 0.0)
    s_g = g.sum()
    if s_g < 1e-15:
        return a
    g = g / s_g
    
    # Check if g violates any non-active constraint
    c_new = _fft_convolve(a, g)
    violated = np.where(c_new > target + 1e-10)[0]
    
    return g, violated

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: LP with symmetry & active-set, IRLS/QP, coordinate descent,
and multi-length exploration. Builds on parent solution.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_max(a):
    """Just the max of convolution (assumes sum(a)=1)."""
    c = _fft_convolve(a, a)
    return float(c.max())


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _build_lp_structure(n):
    """Precompute sparse LP structure (rows, cols) that stays fixed."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _get_lp_values(a, rows, cols):
    """Get the values for the LP constraint matrix given current a."""
    return a[rows - cols].astype(np.float64)


def _lp_solve(a, rows, cols, n, delta=0.0, top_k=None, time_limit=15):
    """Solve the LP: max sum(g) s.t. g>=0, a*g[k]<=M(1+delta) for active k."""
    c = _fft_convolve(a, a)
    M = float(c.max())
    target = M * (1.0 + delta)

    if top_k is not None:
        # Only use the top-k tightest constraints
        tightness = c.copy()
        k_order = np.argsort(tightness)[::-1][:top_k]
        mask = np.zeros(2 * n - 1, dtype=bool)
        mask[k_order] = True
        # Filter rows/cols
        row_idx = np.where(mask[rows])[0]
        rows_f = rows[row_idx]
        cols_f = cols[row_idx]
    else:
        rows_f = rows
        cols_f = cols

    vals = a[rows_f - cols_f].astype(np.float64)
    n_constraints = len(np.unique(rows_f))
    A_sp = sp.csc_matrix((vals, (rows_f, cols_f)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, target)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_sp, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status != 0:
            return None, M
        g = np.maximum(res.x, 0.0)
        s_g = float(g.sum())
        if s_g < 1e-15:
            return None, M
        g = g / s_g
        return g, M
    except Exception:
        return None, M


def _golden_section_search(a, g, n_iter=40):
    """Golden section search for optimal t in [0,1] minimizing max(conv)."""
    def obj(t):
        a_t = (1.0 - t) * a + t * g
        c = _fft_convolve(a_t, a_t)
        return float(c.max())

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1, f2 = obj(c1), obj(c2)

    for _ in range(n_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)

    return (lo + hi) / 2.0

def _irls_phase(a, deadline, p_start=2.0, p_end=8.0, max_iter=500):
    """IRLS phase: iterate QP solves with sharpening weights."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_max(a)
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1
        c = _fft_convolve(a, a)
        M = float(c.max())
        if M < 1e-15:
            break

        frac = min(1.0, it / max_iter)
        p = p_start * (p_end / p_start) ** frac

        w = np.maximum(c / M, 1e-10) ** p
        w = w / w.sum()

        # Build dense Toeplitz matrix A[i,j] = w[i+j]
        I, J = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
        A_dense = w[I + J]

        # Augmented system: [A  1; 1^T  0] [a; lambda] = [0; 1]
        A_aug = np.zeros((n + 1, n + 1))
        A_aug[:n, :n] = A_dense
        A_aug[:n, n] = 1.0
        A_aug[n, :n] = 1.0
        b_aug = np.zeros(n + 1)
        b_aug[n] = 1.0

        try:
            sol = np.linalg.solve(A_aug, b_aug)
            a_new = sol[:n]
        except np.linalg.LinAlgError:
            # Fallback: use conjugate gradient or just keep current
            a_new = a.copy()
            break

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_new = _symmetrize(a_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_max(a_new)
        a = a_new

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        if it % 10 == 0:
            full_sc = _score(a)
            print(f"  IRLS[{it}] p={p:.1f} max_c={sc:.10f} full={full_sc:.10f} t={time.monotonic()-a_start_time:.0f}s")

    return best_a, best_sc


def _coord_descent(a, deadline, n_sweeps=100, sym=True):

    n = len(a)

    c = _fft_convolve(a, a)
    M = c.max()
    S = a.sum()

        improved = False
        for i in range(n):

            old_val = a[i]
            # Try several candidate values
            best_local = old_val
            best_local_sc = 2.0 * n * c.max() / (S * S)

            # Candidate values: 0, old*0.5, old, old*1.5, and gradient-based
            candidates = [0.0, old_val * 0.5, old_val, old_val * 1.5]
            # Gradient-based: the gradient of max_c w.r.t. a_i
            # d(max_c)/da_i ~ 2 * sum_k w_k * a[k-i] where w_k is subdual weight
            k_max = np.argmax(c)
            grad_approx = 2.0 * a[(k_max - i) % (2 * n - 1) if (k_max - i) % (2 * n - 1) >= n else k_max - i] if 0 <= k_max - i < n else 0.0
            # Simple: just try perturbing in both directions
            step = max(old_val * 0.1, S * 0.001)
            candidates.append(old_val + step)
            candidates.append(max(0.0, old_val - step))
            candidates.append(max(0.0, old_val + step * 2))
            candidates.append(max(0.0, old_val - step * 2))

            for cand in candidates:
                if cand < 0 or cand > 1000:
                    continue
                a[i] = cand
                S_new = S + (cand - old_val)
                if S_new < 1e-15:
                    a[i] = old_val
                    continue
                # Update convolution incrementally
                # c[k] changes by (cand - old_val) * (a[k-i] + a[i-k]) + (cand-old_val)^2 * [k=2i] - old_val*(cand-old_val)*[k=i or k=2i-or...]
                # Too complex; just recompute

                    best_local_sc = sc_new
                    best_local = cand
                    c = c_new
                    M = c_new.max()
                    S = S_new

                c = _fft_convolve(a, a)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0 + 1)[:-1]
    x1 = np.linspace(0, 1, n_new + 1)[:-1]
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    p_new = _symmetrize(p_new)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

                                   top_k=200, sym=True, line_search=True)
        full_sc = _score(a_fin)
        print(f"Phase 6 final LP: score={full_sc:.10f}")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fin.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
