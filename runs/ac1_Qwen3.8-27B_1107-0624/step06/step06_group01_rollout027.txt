"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses L-BFGS on log-softmax parameterization, subgradient methods with true max,
adaptive multi-constraint averaging, and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is normalized (sum=1). Returns 2*n*max_c."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _conv_and_grad(a, beta):
    """Compute convolution, softmax weights, and gradient.
    
    Returns: c (convolution), w (softmax weights), grad (gradient of smoothed objective)
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    w = np.exp(shifted)
    w_sum = float(w.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    
    # Gradient of sum_k w_k * c_k with respect to a
    # dc_k/da_i = 2 * a_{k-i} (with a_m = 0 for m outside [0,n-1])
    # So grad_i = sum_k w_k * 2 * a_{k-i}
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad = 2.0 * h
    
    obj = 2.0 * n * float(np.dot(w, c))  # smoothed objective (times n for scale)
    return c, w, grad, obj


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lgbfgs_smoothed(a_init, deadline, beta_sequence, max_iter_per_phase=50000):
    """Run L-BFGS on the log-softmax parameterization with decreasing beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for beta in beta_sequence:
        if time.monotonic() > deadline:
            break
        
        def obj_grad(z_vec):
            a = _softmax(z_vec)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            obj = 2.0 * n * float(np.dot(w, c))

            # = da_grad_j * a_j - a_j * sum_i da_grad_i * a_i
            
            dot = float(np.dot(da_grad, a))

def _coordinate_descent(a_init, deadline, n_cycles=5):
    """Coordinate descent: try to optimize each coordinate."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    rng = np.random.default_rng(123)
    
    for cycle in range(n_cycles):
        if time.monotonic() > deadline:
            break
        
        # Shuffle coordinates
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            # Try different values for a_i
            a_i_orig = a_curr[i]
            if a_i_orig < 1e-10:
                continue
            
            # Try multiplicative factors
            for factor in [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]:

                a_try[i] = a_i_orig * factor

                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try.copy()
                    a_curr = a_try.copy()

    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fista_coarse(a_init, deadline, max_iter=200000):
    """Coarse FISTA optimization with adaptive smoothing."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    
    beta = 0.05 * max_c
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            
            # Update step size
            c_new = _full_convolve(a_next, a_next)
            max_c = float(c_new.max())
            L = 4.0 * n * max_c / beta
            step = 1.0 / L
            
            # Decrease beta every 2000 iterations
            if it > 0 and it % 2000 == 0:
                beta *= 0.5
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _multi_active_subgrad(a_init, deadline, n_iter=50000):
    """Subgradient method with multiple active constraints and adaptive step."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    c_curr = _full_convolve(a_curr, a_curr)
    c_max0 = float(c_curr.max())
    
    # Initial step size based on problem scale
    base_step = 1.0 / (2.0 * n * c_max0)
    patience = 0
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c_curr = _full_convolve(a_curr, a_curr)
        c_max = float(c_curr.max())
        
        # Adaptive threshold for active constraints
        threshold = c_max - 0.005 * c_max0
        active = np.where(c_curr >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c_curr))])
        # Limit to top 20 active constraints
        if len(active) > 20:
            top_vals = np.sort(c_curr[active])[::-1][:20]
            top_thresh = top_vals[-1]
            active = active[c_curr[active] >= top_thresh]
        
        # Compute subgradient
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        
        a_rev = a_curr[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Step size: decrease if no improvement
        step = base_step / np.sqrt(1.0 + it * 0.001)
        
        a_new = a_curr - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        
        sc = _score_norm(a_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a_new.copy()
            a_curr = a_new
            patience = 0
        else:
            # Don't accept, try smaller step
            a_new2 = a_curr - step * 0.1 * grad
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-15:
                a_new2 = a_new2 / s2
            sc2 = _score_norm(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a_curr = a_new2
                patience = 0
            else:
                patience += 1
                if patience > 200:
                    base_step *= 0.5
                    patience = 0
                    if base_step < 1e-10:
                        break
    
    return best_a, best_score
