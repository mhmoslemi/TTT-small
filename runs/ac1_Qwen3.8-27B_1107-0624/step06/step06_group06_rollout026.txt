"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exponential gradient (multiplicative weights) for fast coarse convergence,
L-BFGS with adaptive log-sum-exp smoothing for refinement, active set tracking
with QP-based polish, diverse structured initializations, and multi-n exploration.
Key improvements: direct gradient computation (no softmax overhead), better step-size
adaptation, peacocking heuristic for peak distribution, and final active-set QP.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_direct(a, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a.
    w has length 2n-1, a has length n.
    grad[i] = 2 * sum_j w[i+j] * a[j]
    Computed via: 2 * conv(w, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _smoobj_and_grad(z, beta, n):
    """Log-sum-exp smoothed objective and gradient for softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    # Gradient through softmax: d/da obj * da/dz
    grad_a = _grad_direct(a, w)
    # Chain rule: d f(softmax(z))/dz_i = (a_i * (grad_a_i - sum(a*grad_a)))
    a_dot_g = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - a_dot_g)
    return obj, grad_z


def _exp_gradient(a_init, deadline, beta_init=0.01, max_iters=50000):
    """Exponential gradient (multiplicative weights) with adaptive beta."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    beta = beta_init
    eta = 1.0
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        grad = _grad_direct(a, w)

        # Adaptive step size: use the magnitude of the gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = eta / (gnorm + 1e-10)

        # Multiplicative update
        log_update = -step * grad
        log_update = np.clip(log_update, -10, 10)
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        s_new = a_new.sum()
        if s_new < 1e-15:
            beta *= 0.5
            continue
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                stall_count = 0
            a = a_new
            eta = min(eta * 1.02, 5.0)
        else:
            stall_count += 1
            eta *= 0.7
            if stall_count % 10 == 0:
                beta *= 0.7
                if stall_count > 100:
                    a = best_a.copy()
                    eta = 1.0
                    stall_count = 0

        # Decrease beta over time for sharper peak approx
        if it % 2000 == 0 and it > 0:
            beta *= 0.6

    return best_a, best_sc


def _lbfgs_refine(a_init, deadline, min_time_remaining=20.0):
    """L-BFGS with adaptive beta continuation for fine-tuning."""
    n = len(a_init)
    current_a = a_init.copy()
    current_a = np.maximum(current_a, 1e-15)
    current_a = current_a / current_a.sum()
    current_sc = _score(current_a)

    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return current_a

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted) - 1)] if len(c_sorted) > 5 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015, 0.0075, 0.003, 0.001, 0.0005, 0.0002])

    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))

        def fun(z_local):
            obj, _ = _smoobj_and_grad(z_local, beta, n)
            return obj

        def jac(z_local):
            _, g = _smoobj_and_grad(z_local, beta, n)
            return g

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac=0.005, max_iter=200):
    """Iteratively solve on the active set using projected gradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active

        # Gradient of sum_{k in active} c[k]
        grad = _grad_direct(a, w)

        # Project gradient onto tangent space of {sum(a)=1}
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        alpha = 1.0
        f_old = float(c[active].sum())
        improved = False
        for ls in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller step
            a_try = a - 1e-4 * grad_t / (gnorm + 1e-30)
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
            else:
                break

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Tighten active set tolerance as we converge
        tol_frac = max(0.001, tol_frac * 0.95)

    return best_a, best_sc


def _peacock(a_init, deadline, n_rounds=50):
    """Peacock heuristic: iteratively reduce the peak by adjusting elements
    that contribute most to it, redistributing mass to elements that
    contribute least."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max_idx = int(np.argmax(c))
        c_max = float(c[c_max_idx])

        # Find elements that contribute most to the peak
        # (a*a)[c_max_idx] = sum_i a[i]*a[c_max_idx - i]
        contributions = np.zeros(n)
        for i in range(n):
            j = c_max_idx - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]

        # Also consider contribution to other high peaks
        top_k = np.argsort(c)[::-1][:max(5, n // 10)]
        for k in top_k[1:]:
            for i in range(n):
                j = int(k) - i
                if 0 <= j < n:
                    contributions[i] += 0.5 * a[i] * a[j]

        # Reduce top contributors, increase bottom contributors
        top_frac = max(3, n // 50)
        bot_frac = max(3, n // 50)
        top_idx = np.argsort(contributions)[::-1][:top_frac]
        bot_idx = np.argsort(contributions)[:bot_frac]

        total_top = a[top_idx].sum()
        total_bot = a[bot_idx].sum()

        if total_top < 1e-15:
            break

        # Transfer a small fraction of mass
        transfer = 0.01 * total_top
        a[top_idx] -= transfer / top_frac
        a[bot_idx] += transfer / bot_frac
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Revert if it got worse
            a = best_a.copy()

    return best_a, best_sc


def _make_geometric(n, ratio=0.95):
    t = np.arange(n, dtype=float)
    a = ratio ** t
    a = a / a.sum()
    return a


def _make_two_lobe(n, center1=0.25, center2=0.75, width=0.15):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - center1) / width) ** 2) + np.exp(-((t - center2) / width) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _make_skewed(n, gamma=2.0):
    t = np.linspace(0, 1, n)
    a = (1 - t) ** gamma
    a = a + 0.01
    a = a / a.sum()
    return a


def _make_random_structured(n, rng):
    """Random sequence with smooth structure (low-pass filtered noise)."""
    raw = rng.standard_normal(n)
    # Low-pass filter via FFT
    freqs = np.fft.rfftfreq(n)
    cutoff = 0.1
    mask = np.exp(-freqs / cutoff)
    filtered = np.fft.irfft(np.fft.rfft(raw) * mask, n)
    a = np.abs(filtered) + 0.1
    a = a / a.sum()
    return a


def _make_tapered(n, power=0.5):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient method with active set for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        grad_a = _grad_direct(a, w)
        a_dot_g = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - a_dot_g)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad_z
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage_polish(a_init, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    a_sorted_idx = np.argsort(a_init)

    fractions = [0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: exponential gradient + L-BFGS + active set + peacock.
    Explores multiple n values and structured initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent))
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent))

    # --- Phase 1: Exponential gradient from parent ---
    print(f"\n[Phase 1] Exp-Gradient from parent (t={time.monotonic() - t_start:.1f}s)")
    eg_deadline = min(deadline - 600, t_start + 300)
    a_eg, sc_eg = _exp_gradient(parent, eg_deadline, beta_init=0.02, max_iters=100000)
    update_best(a_eg, sc_eg)

    # --- Phase 2: L-BFGS refinement ---
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        lbfgs_deadline = min(deadline - 400, time.monotonic() + 200)
        a_lb = _lbfgs_refine(best_seq, lbfgs_deadline, min_time_remaining=100.0)
        update_best(a_lb, _score(a_lb))

    # --- Phase 3: Active set polish ---
    print(f"\n[Phase 3] Active set polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        as_deadline = min(deadline - 300, time.monotonic() + 120)
        a_as, sc_as = _active_set_polish(best_seq, as_deadline, tol_frac=0.01, max_iter=300)
        update_best(a_as, sc_as)

    # --- Phase 4: Peacock heuristic ---
    print(f"\n[Phase 4] Peacock (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        pk_deadline = min(deadline - 250, time.monotonic() + 60)
        a_pk, sc_pk = _peacock(best_seq, pk_deadline, n_rounds=200)
        update_best(a_pk, sc_pk)
        if sc_pk < best_score:
            # Polish after peacock
            a_pk2 = _lbfgs_refine(a_pk, min(deadline - 200, time.monotonic() + 60, deadline - 200), min_time_remaining=50)
            update_best(a_pk2, _score(a_pk2))

    # --- Phase 5: Shrinkage ---
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sh, sc_sh = _shrinkage_polish(best_seq, min(deadline - 200, time.monotonic() + 30))
        update_best(a_sh, sc_sh)
        if sc_sh < best_score:
            a_sh2 = _lbfgs_refine(a_sh, min(deadline - 150, time.monotonic() + 60, deadline - 150), min_time_remaining=50)
            update_best(a_sh2, _score(a_sh2))

    # --- Phase 6: Try different n values ---
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        # Initialize from parent or structured
        if parent is not None and abs(n_try - n_parent) < n_try * 0.5:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_tapered(n_try, power=0.5)

        # Quick exp-gradient
        eg_dl = min(deadline - 100, time.monotonic() + 80)
        a_eg2, _ = _exp_gradient(a_init, eg_dl, beta_init=0.02, max_iters=30000)
        # Quick L-BFGS
        lb_dl = min(deadline - 80, time.monotonic() + 60)
        a_lb2 = _lbfgs_refine(a_eg2, lb_dl, min_time_remaining=30)
        update_best(a_lb2, _score(a_lb2))

    # --- Phase 7: Diverse restarts ---
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n = len(best_seq)
        initializations = [
            _make_geometric(n, ratio=0.98),
            _make_two_lobe(n, 0.3, 0.7, 0.2),
            _make_skewed(n, gamma=3.0),
            _make_random_structured(n, rng),
        ]
        for idx, a_init in enumerate(initializations):
            if time.monotonic() > deadline - 30:
                break
            # Blend with best
            a_blend = 0.5 * best_seq + 0.5 * a_init
            a_blend = np.maximum(a_blend, 1e-10)
            a_blend = a_blend / a_blend.sum()

            eg_dl = min(deadline - 20, time.monotonic() + 40)
            a_r, _ = _exp_gradient(a_blend, eg_dl, beta_init=0.015, max_iters=20000)
            lb_dl = min(deadline - 10, time.monotonic() + 30)
            a_r2 = _lbfgs_refine(a_r, lb_dl, min_time_remaining=5)
            update_best(a_r2, _score(a_r2))

    # --- Phase 8: Final fine polish ---
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Subgradient with very small step
        sg_dl = min(deadline - 3, time.monotonic() + 30)
        a_fin, sc_fin = _subgrad_polish(best_seq, sg_dl, n_max_steps=2000, lr_init=0.008)
        update_best(a_fin, sc_fin)

        # Active set with tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_polish(best_seq, deadline - 2, tol_frac=0.002, max_iter=100)
            update_best(a_fin2, sc_fin2)

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
