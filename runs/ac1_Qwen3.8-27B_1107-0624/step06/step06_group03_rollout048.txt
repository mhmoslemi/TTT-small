"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam in log-space + exact coordinate descent (golden section) + multi-length exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _topk_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
        m_buf *= 0.3
        v_buf *= 0.3
    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _coord_descend_golden(a, n_passes, deadline, n_iter_golden=60):
    """Coordinate descent with golden section search for exact 1D minimization."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = 0
        for i in range(n):
            if time.monotonic() > deadline:
                break
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue
            
            # Compute base convolution (with a_i set to 0)
            # (a*a)_k = base_k + 2*t*a_{k-i} + (t^2 if k==2i)
            # where base_k is conv with a_i = 0
            
            # Efficient computation:
            # base_k = current_conv_k - 2*a_i*a_{k-i} - (a_i^2 if k==2i)
            # Actually: current_conv_k = base_k + 2*a_i*a_{k-i} + (a_i^2 if k==2i)
            # So: base_k = current_conv_k - 2*a_i*a_{k-i} - (a_i^2 if k==2i)
            
            # But we need to be careful with bounds.
            # a_{k-i} is valid only when 0 <= k-i < n
            # a_i^2 contributes to k = 2i (if 2i < 2n-1)
            
            c = _full_convolve(a, a)
            
            # d_k = a_{k-i} if 0 <= k-i < n, else 0
            d = np.zeros(2 * n - 1)
            valid_k = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_k] = a[valid_k - i]
            
            # base_k = c_k - 2*a_i*d_k - (a_i^2 if k==2i else 0)
            base = c - 2.0 * a_i_old * d
            if 2 * i < 2 * n - 1:
                base[2 * i] -= a_i_old * a_i_old
            
            S = 1.0 - a_i_old  # sum of all other elements (a is normalized to sum=1)
            
            # Define F(t) = 2n * max_k(base_k + 2*t*d_k + (t^2 if k==2i)) / (S+t)^2
            # Minimize over t in [0, some upper bound]
            
            def _f_of_t(t):
                c_t = base + 2.0 * t * d
                if 2 * i < 2 * n - 1:
                    c_t = c_t.copy()
                    c_t[2 * i] += t * t
                max_c = float(c_t.max())
                return 2.0 * n * max_c / ((S + t) ** 2)
            
            # Golden section search on [0, upper]
            # Upper bound: a_i can't be more than 1 (since sum=1 and all non-negative)
            # But let's be a bit more generous
            upper = min(1.0, max(a_i_old * 3.0, 0.1))
            
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
            a_lb = 0.0
            a_ub = upper
            c_pt = a_ub - gr * (a_ub - a_lb)
            d_pt = a_lb + gr * (a_ub - a_lb)
            c_val = _f_of_t(c_pt)
            d_val = _f_of_t(d_pt)
            
            for _ in range(n_iter_golden):
                if c_val < d_val:
                    a_ub = d_pt
                    d_pt = c_pt
                    d_val = c_val
                    c_pt = a_ub - gr * (a_ub - a_lb)
                    c_val = _f_of_t(c_pt)
                else:
                    a_lb = c_pt
                    c_pt = d_pt
                    c_val = d_val
                    d_pt = a_lb + gr * (a_ub - a_lb)
                    d_val = _f_of_t(d_pt)
            
            t_opt = (a_lb + a_ub) / 2.0
            t_opt = max(0.0, min(t_opt, 1.0))
            
            if abs(t_opt - a_i_old) > 1e-8:
                a[i] = t_opt
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1
            
            if improved % 500 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
        
        sc_pass = _score_norm(a)
        if sc_pass < best_score:
            best_score = sc_pass
        if sc_pass > best_score * (1 + 1e-6):
            # Didn't improve, might be converged
            break
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score

def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _structured_init(n, kind="cos_sum", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4]:
            a += 0.2 / freq * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        # Smooth with a window
        w = np.ones(32) / 32
        a = np.convolve(a, w, mode='same')
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        return np.ones(n) / n
