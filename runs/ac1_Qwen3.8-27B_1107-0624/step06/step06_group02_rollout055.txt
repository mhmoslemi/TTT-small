This is a quadratically constrained program. For a small active set (say 10-20 constraints), this can be solved efficiently.

    z_full[1:] = z_sub

    z_max = z_full.max()
    exp_z = np.exp(z_full - z_max)
    a = exp_z / exp_z.sum()

    c = _convolve(a, a)
    c_max = float(c.max())

    tc = tau * (c - c_max)
    np.clip(tc, -500.0, 500.0, out=tc)
    exp_tc = np.exp(tc)
    s_exp = float(exp_tc.sum())
    if s_exp < 1e-300:
        m_tau = c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        m_tau = c_max + np.log(s_exp) / tau
        w = exp_tc / s_exp

    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]

    dot = float(np.dot(grad_a, a))
    grad_z_full = a * (grad_a - dot)
    grad_z = grad_z_full[1:]

    obj = 2.0 * n * m_tau
    return obj, grad_z


def _lbfgs_optimize(a_init, tau_list, deadline, n):
    """Run L-BFGS optimization across multiple tau levels."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z[0] = 0.0

    best_a = a_init.copy()
    best_sc = _score(a_init)

    for i, tau in enumerate(tau_list):
        if time.monotonic() > deadline:
            break

        tau_c = _convolve(best_a, best_a).max()
        effective_tau = tau * tau_c  # Make tau relative to scale

        time_budget = min(120.0, max(10.0, (deadline - time.monotonic()) / max(1, len(tau_list) - i)))
        start_t = time.monotonic()
        max_iter = int(time_budget * 800)  # ~800 iterations per second estimate

        try:
            result = minimize(
                _obj_grad_lbfgs,
                z[1:].copy(),
                jac=True,
                method='L-BFGS-B',
                args=(effective_tau, n),
                options={
                    'maxiter': min(max_iter, 2000),
                    'maxfun': min(max_iter * 10, 50000),
                    'maxls': 100,
                    'ftol': 1e-18,
                    'gtol': 1e-14,
                    'gtol_rel': 1e-14,
                    'iprint': -1,
                    'max_correction': 10,
                }
            )
            z[1:] = result.x
        except Exception:
            pass

        a = _softmax_full(z)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        elapsed = time.monotonic() - start_t
        print(f"    L-BFGS tau_idx={i}, tau={effective_tau:.2e}: score={sc:.8f}, "
              f"iters={result.nit if 'result' in dir() else '?'}, time={elapsed:.1f}s")

    return best_a, best_sc


def _coord_descent(a_init, deadline, max_rounds=300, verbose=True):
    """Coordinate descent with 1D line search."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break

        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break

            old_val = a[j]
            # Bounds for a_j: [0, 2*max(a)]
            lo = 0.0
            hi = max(2.0 * old_val, 2.0 * a.max())

            def _1d_obj(x):
                a[j] = x
                s2 = a.sum()
                if s2 < 1e-30:
                    return 1e10
                return _score_norm(a / s2)

            try:
                res = minimize_scalar(_1d_obj, bounds=(lo, hi), method='bounded',
                                      options={'maxiter': 30, 'xatol': 1e-12})
                if res.success and res.fun < best_sc - 1e-14:
                    a[j] = res.x
                    s2 = a.sum()
                    a = a / s2
                    best_sc = res.fun
                    best_a = a.copy()
                    improved = True
                    a[j] = a[j]  # already updated
                else:
                    a[j] = old_val
            except Exception:
                a[j] = old_val

        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if not improved and r > 10:
            break
        if verbose and r % 50 == 0:
            print(f"    CD round {r}: score={best_sc:.10f}")

    return best_a, best_sc


def _adam_logspace_polish(a_init, deadline, n):
    """Quick Adam polish in log-space as backup."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.005

    c_init = _convolve(a_init, a_init)
    beta = 0.001 * c_init.max()

    while time.monotonic() < deadline:
        a = _softmax_full(z)
        c = _convolve(a, a)
        cm = float(c.max())

        cs = (c - cm) / beta
        np.clip(cs, -500, 500, out=cs)
        e = np.exp(cs)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * h  # gradient of 2n*max_c in log-space (simplified)
        # Actually need proper chain rule
        dot = float(np.dot(2.0 * h, a))
        grad_z = a * (2.0 * h - dot)

        if not np.all(np.isfinite(grad_z)):
            break

        t += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        upd_norm = np.linalg.norm(update)
        if upd_norm > 0.2:
            update *= 0.2 / upd_norm

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 20 == 0:
            a_chk = _softmax_full(z)
            sc = _score_norm(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if t % 5000 == 0:
            beta *= 0.5
            lr = max(0.0005, lr * 0.9)

    a_final = _softmax_full(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        parent = np.ones(4096) / 4096
        n_parent = 4096
        sc_parent = _score(parent)
        best_seq = parent.copy()
        best_score = sc_parent
        print(f"No parent, uniform n=4096, score={sc_parent:.10f}")

    # === Phase 1: Improve parent at n=4096 with L-BFGS ===
    print("Phase 1: L-BFGS at n=4096")
    n1 = 4096
    a1 = parent.copy()
    c1 = _convolve(a1, a1)
    max_c1 = float(c1.max())

    tau_list_1 = [0.02 * max_c1, 0.005 * max_c1, 0.001 * max_c1,
                  0.0003 * max_c1, 0.0001 * max_c1, 0.00003 * max_c1,
                  0.00001 * max_c1]
    phase1_end = min(deadline, t_start + 300.0)
    a1, sc1 = _lbfgs_optimize(a1, tau_list_1, phase1_end, n1)
    sc1_full = _score(a1)
    print(f"  Phase 1 done: score={sc1_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc1_full < best_score:
        best_score = sc1_full
        best_seq = a1.copy()

    # Coordinate descent polish at n=4096
    if time.monotonic() < deadline - 200:
        a1_cd, sc1_cd = _coord_descent(a1, min(deadline - 200, time.monotonic() + 60),
                                       max_rounds=100)
        sc1_cd_full = _score(a1_cd)
        print(f"  CD polish n=4096: score={sc1_cd_full:.10f}")
        if sc1_cd_full < best_score:
            best_score = sc1_cd_full
            best_seq = a1_cd.copy()

    # === Phase 2: Optimize at n=8192 ===
    print("Phase 2: L-BFGS at n=8192")
    n2 = 8192
    # Generate diverse initializations for n=8192
    inits_2 = []

    # From best n=4096
    a_interp = _interp_profile(best_seq, n2)
    inits_2.append(('interp', a_interp))

    # Flat spectrum random
    a_flat = _flat_spectrum_init(n2, rng)
    inits_2.append(('flat_spec', a_flat))

    # Chirp
    a_chirp = _chirp_init(n2)
    inits_2.append(('chirp', a_chirp))

    # Raised cosine
    a_rc = _raised_cosine_init(n2)
    inits_2.append(('raised_cos', a_rc))

    # Best init
    best_init2 = None
    best_init2_sc = np.inf
    for name, init in inits_2:
        sc = _score(init)
        print(f"  Init '{name}': score={sc:.8f}")
        if sc < best_init2_sc:
            best_init2_sc = sc
            best_init2 = init

    c2 = _convolve(best_init2, best_init2)
    max_c2 = float(c2.max())
    tau_list_2 = [0.02 * max_c2, 0.005 * max_c2, 0.001 * max_c2,
                  0.0003 * max_c2, 0.0001 * max_c2, 0.00003 * max_c2,
                  0.00001 * max_c2, 0.000003 * max_c2]

    phase2_end = min(deadline, t_start + 900.0)
    a2, sc2 = _lbfgs_optimize(best_init2, tau_list_2, phase2_end, n2)
    sc2_full = _score(a2)
    print(f"  Phase 2 done: score={sc2_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc2_full < best_score:
        best_score = sc2_full
        best_seq = a2.copy()

    # Try other initializations if time permits
    if time.monotonic() < phase2_end - 120:
        for name, init in inits_2:
            if name == 'interp':
                continue
            if time.monotonic() > phase2_end - 60:
                break
            sc_init = _score(init)
            if sc_init > best_score + 0.5:
                continue  # Skip if too far from best
            c_i = _convolve(init, init)
            max_ci = float(c_i.max())
            tau_i = [0.01 * max_ci, 0.001 * max_ci, 0.0001 * max_ci, 0.00001 * max_ci]
            a_i, sc_i = _lbfgs_optimize(init, tau_i, min(phase2_end - 30, time.monotonic() + 120), n2)
            sc_i_full = _score(a_i)
            print(f"  Alt init '{name}': score={sc_i_full:.10f}")
            if sc_i_full < best_score:
                best_score = sc_i_full
                best_seq = a_i.copy()

    # Adam polish on best n=8192
    if time.monotonic() < deadline - 200:
        a2_adam, sc2_adam = _adam_logspace_polish(best_seq, min(deadline - 200, time.monotonic() + 120), n2)
        sc2_adam_full = _score(a2_adam)
        print(f"  Adam polish n=8192: score={sc2_adam_full:.10f}")
        if sc2_adam_full < best_score:
            best_score = sc2_adam_full
            best_seq = a2_adam.copy()

    # Coordinate descent on best n=8192
    if time.monotonic() < deadline - 100:
        a2_cd, sc2_cd = _coord_descent(best_seq, min(deadline - 100, time.monotonic() + 90),
                                       max_rounds=80)
        sc2_cd_full = _score(a2_cd)
        print(f"  CD polish n=8192: score={sc2_cd_full:.10f}")
        if sc2_cd_full < best_score:
            best_score = sc2_cd_full
            best_seq = a2_cd.copy()

    # === Phase 3: Explore n=16384 ===
    print("Phase 3: n=16384 exploration")
    n3 = 16384
    phase3_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase3_end - 120:
        a3 = _interp_profile(best_seq, n3)
        sc3_init = _score(a3)
        print(f"  Interpolated n=16384: score={sc3_init:.8f}")
        if sc3_init < best_score:
            best_score = sc3_init
            best_seq = a3.copy()

        c3 = _convolve(a3, a3)
        max_c3 = float(c3.max())
        tau_list_3 = [0.01 * max_c3, 0.003 * max_c3, 0.001 * max_c3,
                      0.0003 * max_c3, 0.0001 * max_c3]
        a3, sc3 = _lbfgs_optimize(a3, tau_list_3, min(phase3_end - 30, time.monotonic() + 240), n3)
        sc3_full = _score(a3)
        print(f"  L-BFGS n=16384: score={sc3_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc3_full < best_score:
            best_score = sc3_full
            best_seq = a3.copy()

        # Quick CD on n=16384
        if time.monotonic() < deadline - 80:
            a3_cd, sc3_cd = _coord_descent(a3, min(deadline - 80, time.monotonic() + 60),
                                           max_rounds=40, verbose=False)
            sc3_cd_full = _score(a3_cd)
            print(f"  CD n=16384: score={sc3_cd_full:.10f}")
            if sc3_cd_full < best_score:
                best_score = sc3_cd_full
                best_seq = a3_cd.copy()

    # === Phase 4: Final ultra-fine polish on best sequence ===
    print("Phase 4: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        tau_final = [0.00001 * max_cp, 0.000003 * max_cp, 0.000001 * max_cp,
                     0.0000003 * max_cp]
        a_pol, sc_pol = _lbfgs_optimize(a_pol, tau_final, min(deadline - 20, time.monotonic() + 120), n_best)
        sc_pol_full = _score(a_pol)
        print(f"  Final polish: score={sc_pol_full:.10f}")
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 256.0
        # Scale to use full range (helps with evaluator's min(1000) cap)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
