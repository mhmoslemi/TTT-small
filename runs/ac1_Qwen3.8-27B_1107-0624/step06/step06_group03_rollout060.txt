"""
Alternating LP + FISTA polishing for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Haffner-type alternating linear programming as the main optimizer, which
provides exact best-response steps and naturally discovers the non-symmetric
structure needed to beat the symmetric lower bound of 2. Combined with
smoothed-max FISTA and coordinate descent for fine polishing.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_rows_cols(n):
    """Pre-build row/col index arrays for the response LP constraint matrix."""
    total_nz = n * n + (2 * n - 1)
    rows = np.empty(total_nz, dtype=np.int32)
    cols = np.empty(total_nz, dtype=np.int32)
    idx = 0

    for d in range(n):

        cols[idx:idx + n] = jj
        idx += n
    rows[idx:idx + (2 * n - 1)] = np.arange(2 * n - 1, dtype=np.int32)
    cols[idx:idx + (2 * n - 1)] = n
    return rows, cols, total_nz


def _solve_response_lp(b, n, rows, cols, total_nz):
    """Solve: min t s.t. conv(a,b)[k]<=t, sum(a)=1, a>=0. Returns (a_opt, t_opt)."""
    vals = np.empty(total_nz, dtype=np.float64)

    b_eq = np.array([1.0])

def _alternating_lp(a_init, n, max_time_s, damping=1.0, verbose=True):
    """Run alternating LP until time budget or convergence."""
    t_start = time.monotonic()
    b = a_init.copy()
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-15:

    else:
        b = b / s

    rows, cols, total_nz = _build_lp_rows_cols(n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    best_score = _score_norm(b)

            break

        a_new = np.maximum(result.x[:n], 0.0)

        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Damped update
        if damping < 1.0:

            a_new = np.maximum(a_new, 0.0)

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_softmax_weights(c, beta, k):
    """Compute softmax weights over top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([np.argmax(c)])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a: grad_i = 2 * sum_k w[i+k] * a_k."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    """Gradient in log-space for normalized problem (sum(a)=1)."""
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_polish(a_init, deadline, beta, top_k, max_iter):
    """FISTA on simplex with top-K softened max of conv(a,a)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _fft_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _fft_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter_per_phase):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for it in range(max_iter_per_phase):
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_normalize(z)
            c = _fft_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                t = 0
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descend(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent: for each i, optimize a_i with 1D search."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break

            c = _fft_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue

            # d[j] = a[j-i] for valid j (where 0 <= j-i < n)
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            # Find active positions
            idx_top = np.argsort(c)[-top_k_w:]

            # Candidate t values
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2, a_i_old * 2.0]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 5.0 * max(a_i_old, 0.01):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.8))
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            # Deduplicate
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = _score_norm(a)
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max_c / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="asym_cos"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "asym_cos":
        # Asymmetric cosine: break symmetry to reduce central peak
        t = np.arange(n) / n
        a = 1.0 + 0.5 * np.cos(2 * np.pi * t) + 0.3 * np.cos(4 * np.pi * t)
        # Add asymmetry: linear ramp
        a += 0.4 * (t - 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered_asym":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a *= (1.0 + 0.3 * t)  # asymmetry
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "step_asym":
        # Step function: high on left, low on right
        a = np.ones(n)
        a[:n // 2] = 1.5
        a[n // 2:] = 0.7
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "asym_cos")
        sc_parent = _score(parent)
        print(f"No parent. Init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Determine which lengths to try
    if n_parent == 4096:
        lengths = [4096, 8192]
    elif n_parent == 8192:
        lengths = [8192, 4096]
    else:
        lengths = [4096, 8192]

    for n_try in lengths:
        if time.monotonic() > deadline - 90:
            break

        remaining = deadline - time.monotonic()
        print(f"\n=== n={n_try}, remaining={remaining:.0f}s ===")

        # Initialize
        if n_try == n_parent:
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()

        sc0 = _score(a0)
        print(f"  Init score: {sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Phase 1: Alternating LP
        # Budget: 60% of remaining for this length
        lp_budget = min(remaining * 0.55, 700.0)
        print(f"  Phase 1: AltLP (budget {lp_budget:.0f}s)")

        a_lp, sc_lp = _alternating_lp(a0, n_try, lp_budget, damping=1.0, verbose=True)

        sc_lp_full = _score(a_lp)
        if sc_lp_full < best_score:
            best_score = sc_lp_full
            best_seq = a_lp.copy()
            print(f"  After AltLP: score={sc_lp_full:.10f}")

        a_work = a_lp

        # Try with damping if not converged
        if time.monotonic() < deadline - 60:
            remaining2 = deadline - time.monotonic()
            lp_budget2 = min(remaining2 * 0.2, 150.0)
            if lp_budget2 > 20:
                a_lp2, sc_lp2 = _alternating_lp(a0, n_try, lp_budget2, damping=0.8, verbose=False)
                sc_lp2_full = _score(a_lp2)
                if sc_lp2_full < best_score:
                    best_score = sc_lp2_full
                    best_seq = a_lp2.copy()
                    a_work = a_lp2
                    print(f"  After damped AltLP: score={sc_lp2_full:.10f}")

        # Phase 2: FISTA polishing
        if time.monotonic() < deadline - 40:
            remaining3 = deadline - time.monotonic()
            fista_budget = min(remaining3 * 0.2, 120.0)
            print(f"  Phase 2: FISTA (budget {fista_budget:.0f}s)")

            c_w = _fft_convolve(a_work, a_work)
            max_cw = float(c_w.max())
            beta0 = 0.01 * max_cw
            fista_end = time.monotonic() + fista_budget

            # Coarse to fine beta
            betas = [beta0, beta0 * 0.3, beta0 * 0.1, beta0 * 0.03]
            for bi, beta in enumerate(betas):
                if time.monotonic() > fista_end - 5:
                    break
                top_k = max(50, int(100 * (0.7 ** bi)))
                a_work, sc_w = _fista_polish(a_work, time.monotonic() + max(5, fista_end - time.monotonic()),
                                              beta, top_k, max_iter=200000)
                a_work = a_work / a_work.sum()
                sc_full = _score(a_work)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = a_work.copy()
                    print(f"  FISTA beta={beta:.6f}: score={sc_full:.10f}")

        # Phase 3: Adam log-space
        if time.monotonic() < deadline - 30:
            remaining4 = deadline - time.monotonic()
            adam_budget = min(remaining4 * 0.25, 200.0)
            print(f"  Phase 3: Adam (budget {adam_budget:.0f}s)")

            c_w = _fft_convolve(a_work, a_work)
            max_cw = float(c_w.max())
            betas_adam = [0.002 * max_cw, 0.001 * max_cw, 0.0005 * max_cw,
                          0.0002 * max_cw, 0.0001 * max_cw]
            adam_end = time.monotonic() + adam_budget
            a_work, sc_adam = _adam_logspace(a_work, adam_end, betas_adam,
                                             lr_init=0.008, top_k=80,
                                             max_iter_per_phase=30000)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
                print(f"  After Adam: score={sc_full:.10f}")

        # Phase 4: Coordinate descent
        if time.monotonic() < deadline - 20:
            remaining5 = deadline - time.monotonic()
            cd_budget = min(remaining5 * 0.3, 100.0)
            if cd_budget > 10:
                print(f"  Phase 4: CoordDescend (budget {cd_budget:.0f}s)")
                cd_end = time.monotonic() + cd_budget
                a_work, sc_cd = _coord_descend(a_work, n_passes=5, top_k_w=50, deadline=cd_end)
                sc_full = _score(a_work)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = a_work.copy()
                    print(f"  After CoordDescend: score={sc_full:.10f}")

    # Phase 5: Diversity restarts from best
    if time.monotonic() < deadline - 30 and best_seq is not None:
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        remaining6 = deadline - time.monotonic() - 20
        n_restarts = min(3, int(remaining6 // 40))

        if n_restarts > 0:
            print(f"\n=== Diversity restarts: {n_restarts} (remaining {remaining6:.0f}s) ===")
            per_restart = remaining6 / n_restarts

            for ri in range(n_restarts):
                if time.monotonic() > deadline - 20:
                    break

                if ri == 0:
                    p = a_base + 0.03 * rng.standard_normal(n_best)
                elif ri == 1:
                    # Frequency perturbation
                    freq = rng.uniform(1.0, 4.0)
                    p = a_base + 0.04 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                else:
                    # Shift
                    shift = rng.integers(1, max(2, n_best // 512))
                    p = np.roll(a_base, shift).copy()

                p = np.maximum(p, 1e-10)
                p = p / p.sum()

                # Quick AltLP from this start
                p_lp, sc_lp = _alternating_lp(p, n_best, max(5, per_restart * 0.6),
                                              damping=1.0, verbose=False)
                sc_full = _score(p_lp)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = p_lp.copy()
                    print(f"  Restart {ri}: score={sc_full:.10f}")

                # Quick Adam polish
                if time.monotonic() < deadline - 25:
                    c_p = _fft_convolve(p_lp, p_lp)
                    max_cp = float(c_p.max())
                    betas_r = [0.001 * max_cp, 0.0005 * max_cp, 0.0002 * max_cp]
                    p_adam, _ = _adam_logspace(p_lp, time.monotonic() + max(5, per_restart * 0.4),
                                               betas_r, lr_init=0.005, top_k=60,
                                               max_iter_per_phase=20000)
                    sc_ad = _score(p_adam)
                    if sc_ad < best_score:
                        best_score = sc_ad
                        best_seq = p_adam.copy()
                        print(f"  Restart {ri} Adam: score={sc_ad:.10f}")

    # Final validation
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
