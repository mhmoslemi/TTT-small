def _next_pow2(n):

    while k < n:
        k <<= 1

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]

"""
Multi-phase optimizer using L-BFGS with adaptive smoothing and exact coordinate descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _smooth_max_and_grad(c, beta):
    """Compute smooth max of c and its gradient weights (softmax)."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / beta)
    w = exp_c / exp_c.sum()
    phi = beta * np.log(exp_c.sum()) + c_max
    return phi, w

def _conv_grad(a, w):
    """Compute d/d a_i [sum_k w_k * (a*a)_k] = 2 * cross_corr(a, w)[:n]"""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # cross_corr(a, w)[i] = sum_j a[j] * w[j+i]
    # = ifft(conj(FFT(a)) * FFT(w))
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _smoothed_objective(z, beta, n):
    """Objective in log-space z. a = softmax(z), minimize 2n * smooth_max(conv(a,a)) / sum(a)^2.
    Since a is softmax, sum(a) = 1, so we minimize 2n * smooth_max(conv(a,a))."""
    a = _softmax(z)
    c = _full_convolve(a)
    phi, w = _smooth_max_and_grad(c, beta)
    return 2.0 * n * phi  # sum(a)=1 so this is the score (up to normalization)

def _smoothed_grad(z, beta, n):
    """Gradient of smoothed objective w.r.t. z in log-space.
    dF/dz_i = dF/da_i * da_i/dz_i = dF/da_i * a_i * (1 - a_i) ... 
    Actually, for softmax: da_i/dz_j = a_i*(delta_ij - a_j)
    So dF/dz_i = sum_j dF/da_j * da_j/dz_i = sum_j dF/da_j * a_j * (delta_ji - a_i)
    = a_i * dF/da_i - a_i * sum_j dF/da_j * a_j
    = a_i * (dF/da_i - <dF/da, a>)
    """
    a = _softmax(z)
    c = _full_convolve(a)
    phi, w = _smooth_max_and_grad(c, beta)
    # dF/da_i where F = 2n*phi (since sum=1)
    grad_a = _conv_grad(a, w)
    F = 2.0 * n * phi
    dF_da = 2.0 * n * grad_a  # since sum(a)=1, dF/da = 2n*dphi/da
    mean_dF = float(np.dot(dF_da, a))
    return a * (dF_da - mean_dF)

    e = np.exp(np.clip(z, -50, 50))

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_max_and_weights(c, beta):
    c_max = float(c.max())
    exp_c = np.exp((c - c_max) / beta)
    w = exp_c / exp_c.sum()
    phi = beta * np.log(exp_c.sum()) + c_max
    return phi, w


def _conv_grad(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _lbfgs_smoothed(a_init, beta_schedule, deadline, max_iter_per_phase=500):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break

        def func(zv):
            a = _softmax(zv)
            c = _full_convolve(a)
            phi, _ = _smooth_max_and_grad(c, beta)
            return 2.0 * n * phi

        def flat(zv):
            a = _softmax(zv)
            c = _full_convolve(a)
            phi, w = _smooth_max_and_grad(c, beta)
            grad_a = _conv_grad(a, w)
            dF_da = 2.0 * n * grad_a
            mean_dF = float(np.dot(dF_da, a))
            return a * (dF_da - mean_dF)

        result = minimize(func, z, jac=flat, method='L-BFGS-B',
                          options={'maxiter': max_iter_per_phase,
                                   'gtol': 1e-8, 'ftol': 1e-12})
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            print(f"    L-BFGS beta={beta:.2e}: score={sc_new:.10f}, iters={result.nit}")

        z = result.x

    return best_a, best_sc


def _coord_descent(a, n_passes=10, top_k=200, deadline=None):
    n = len(a)
    a = a.copy()
    best_sc = _score(a)
    s = float(a.sum())
    c = _full_convolve(a)

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = 0
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            vi = a[i]
            if vi < 1e-15:
                continue

            d = np.zeros(2 * n - 1)
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            d[lo:hi] = a[hi - i - 1 : -1 : -1] if False else a[i:i]  # placeholder
            # Actually: d[k] = a[k-i] if 0 <= k-i < n
            idx_valid = np.arange(max(0, i), min(2 * n - 1, i + n))
            d[idx_valid] = a[idx_valid - i]

            # Find top-K positions
            if top_k < 2 * n - 1:
                top_idx = np.argpartition(c, -top_k)[-top_k:]
            else:
                top_idx = np.arange(2 * n - 1)

            candidates = [0.0, vi]

            for k in top_idx:
                ck = c[k]
                dk = d[k]
                delta_i2 = 1.0 if k == 2 * i else 0.0
                # u* = (ck - dk*s) / (delta_i2*s - dk)
                denom = delta_i2 * s - dk
                if abs(denom) > 1e-15:
                    u_star = (ck - dk * s) / denom
                    t_star = vi + u_star
                    if t_star > 0:
                        candidates.append(t_star)
                        candidates.append(max(0.0, t_star * 0.5))
                        candidates.append(t_star * 1.5)

            # Also try some heuristic candidates
            candidates.append(vi * 0.5)
            candidates.append(vi * 2.0)

            best_t = vi
            best_f = np.inf
            s_new_base = s - vi

            for tc in candidates:
                if tc < 0:
                    continue
                u = tc - vi
                # Compute max of c' = c + 2*d*u + u^2 * delta_{2i}
                c_trial_max = 0.0
                # For speed, only check top positions
                max_val = float(np.max(c[top_idx] + 2 * d[top_idx] * u + u * u * (np.array([1.0 if k == 2*i else 0.0 for k in top_idx]))))
                # Also check if the overall max shifted
                # For efficiency, just use top_idx
                s_trial = s_new_base + tc
                if s_trial < 1e-15:
                    continue
                f_val = 2.0 * n * max_val / (s_trial * s_trial)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != vi:
                a[i] = best_t
                improved += 1

        # Update s and c after full pass
        s = float(a.sum())
        c = _full_convolve(a)
        sc = 2.0 * n * float(c.max()) / (s * s)
        if sc < best_sc:
            best_sc = sc
        if improved == 0:
            break

    a = a / a.sum()
    final_sc = _score(a)
    if final_sc < best_sc:
        best_sc = final_sc
    return a, best_sc


def _coord_descent_fast(a, n_passes, top_k, deadline):
    """Faster coordinate descent using vectorized operations."""
    n = len(a)
    a = a.copy()
    best_sc = _score(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        s = float(a.sum())
        c = _full_convolve(a)
        improved = 0

        # Find top positions (update less frequently)
        if top_k < 2 * n - 1:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
        else:
            top_idx = np.arange(2 * n - 1)

        for i in range(n):
            vi = a[i]
            if vi < 1e-14:
                continue

            # d[k] = a[k-i] if 0 <= k-i < n
            # Only compute for top_idx
            diffs = top_idx - i
            valid = (diffs >= 0) & (diffs < n)
            d_top = np.where(valid, a[diffs[valid]], 0.0)
            d_top = np.where(valid, a[np.clip(diffs, 0, n-1)], 0.0)
            d_top = np.where(valid, d_top, 0.0)

            delta_arr = (top_idx == 2 * i).astype(np.float64)
            c_top = c[top_idx]

            # Candidate t values
            denom = delta_arr * s - d_top
            with np.errstate(divide='ignore', invalid='ignore'):
                u_stars = np.where(np.abs(denom) > 1e-14,
                                   (c_top - d_top * s) / denom, 0.0)
                t_stars = vi + u_stars

            t_candidates = np.concatenate([[0.0, vi, vi * 0.5, vi * 2.0],
                                           t_stars[t_stars > 0][:10]])
            t_candidates = np.clip(t_candidates, 0, 10.0 * max(vi, 0.01))

            s_new_base = s - vi
            best_t = vi
            best_f = np.inf

            for tc in t_candidates:
                u = tc - vi
                c_trial = c_top + 2.0 * d_top * u + u * u * delta_arr
                max_val = float(c_trial.max())
                s_trial = s_new_base + tc
                if s_trial < 1e-15:
                    continue
                f_val = 2.0 * n * max_val / (s_trial * s_trial)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != vi:
                a[i] = best_t
                s = s_new_base + best_t
                improved += 1

        sc = 2.0 * n * float(_full_convolve(a).max()) / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
        if improved == 0:
            break

    a = a / a.sum()
    final_sc = _score(a)
    if final_sc < best_sc:
        best_sc = final_sc
    return a, best_sc


def _structured_init(n, kind="cos_sum", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    if kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        return a / a.sum()
    elif kind == "alternating":
        a = np.ones(n)
        a[::2] *= 0.3
        return a / a.sum()
    elif kind == "random":
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        betas.append(beta)
        beta *= 0.5

    if sc_lbfgs < best_sc:
        best_sc = sc_lbfgs
        best_a = a_lbfgs.copy()
        print(f"  After L-BFGS: score={best_sc:.10f}")

    # === Phase 2: Coordinate descent polishing ===
    print(f"\n=== Phase 2: Coordinate Descent ===")
    phase2_deadline = min(deadline, t_start + (deadline - t_start) * 0.5)

        c_curr = _full_convolve(best_a)
