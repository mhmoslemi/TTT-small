Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses structured initialization (ramp, geometric, FFT-based), 
FISTA with adaptive smoothing, Adam in log-space with backtracking,
peak-shaving heuristics, and multi-scale refinement.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_w(a, w, n):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a, using FFT."""
    a_rev = a[::-1]
    # We want: grad[i] = sum_k w[k] * (a[k-i] + a[i+k]) with bounds
    # First term: conv(w, a_rev) gives us part of it
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _compute_grad_logspace(a, w, n):
    """Gradient in log-space parameterization z where a = softmax(z)."""
    # da/dz = a * (1 - a) (diagonal) for the softmax
    # Actually for a = softmax(z), da_i/dz_j = a_i*(delta_ij - a_j)
    # The gradient of f(a) w.r.t. z is: J^T * grad_a where J is Jacobian of a w.r.t. z
    # J_ij = a_i*(delta_ij - a_j)
    # So (J^T * g)_j = sum_i a_i*(delta_ij - a_j)*g_i = a_j*sum_i g_i - a_j*sum_i a_i*g_i
    # = a_j * (sum(g) - sum(a*g))
    # = a_j * sum(g - a*sum(g))... 
    # Actually: (J^T g)_j = a_j * sum_i g_i - a_j * sum_i a_i * g_i = a_j * (1^T g - a^T g)
    # = a_j * (sum(g) - dot(a,g))
    g = _compute_grad_w(a, w, n)
    scalar = float(np.sum(g) - np.dot(a, g))
    return a * scalar

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA for minimizing smoothed max convolution on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / max(beta, 1e-10)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_w(y, w, n)
        a_next = _proj_simplex(y - step * grad)

        # Adaptive L update
        if it % 100 == 0:
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / max(beta, 1e-10)
            step = 1.0 / L
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space for fine-tuning."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_logspace(a, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _peak_shave(a, deadline, n_iters=500):
    """Iteratively reduce the peak convolution by scaling down elements that contribute most to it."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(best_a, best_a)
        k_max = int(np.argmax(c))
        max_val = c[k_max]
        
        # Find which elements contribute most to conv[k_max]
        # conv[k_max] = sum_i a[i]*a[k_max-i]
        i_start = max(0, k_max - n + 1)
        i_end = min(n - 1, k_max)
        contributors = []
        for i in range(i_start, i_end + 1):
            j = k_max - i
            contributors.append((i, j, best_a[i] * best_a[j]))
        
        if not contributors:
            break
        
        # Scale down the top contributor
        contributors.sort(key=lambda x: -x[2])
        i_top, j_top, val_top = contributors[0]
        
        # Try scaling down
        for factor in [0.99, 0.98, 0.95, 0.9]:
            a_try = best_a.copy()
            a_try[i_top] *= factor
            a_try = a_try / a_try.sum()
            c_try = _full_convolve(a_try, a_try)
            if c_try.max() < max_val:
                best_a = a_try
                best_score = c_try.max()
                break
    
    return best_a, best_score

def _make_init_ramp(n, direction=1):
    """Create a ramp initialization."""
    if direction == 1:
        a = np.arange(1, n + 1, dtype=np.float64)
    else:
        a = np.arange(n, 0, dtype=np.float64)
    return a / a.sum()

def _make_init_geometric(n, r=0.98):
    """Create a geometric progression initialization."""
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()

def _make_init_sinusoidal(n, freq=1, amp=0.5):
    """Create a sinusoidal modulation."""
    t = np.arange(n, dtype=np.float64) / n
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_init_trapezoid(n, taper=0.3):
    """Trapezoidal profile."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.ones(n)
    mask_left = t < taper
    mask_right = t > (1 - taper)
    a[mask_left] = t[mask_left] / taper
    a[mask_right] = (1 - t[mask_right]) / taper
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

Actually, let me think about a completely different approach: **SOCP formulation**.

The problem: minimize max_k conv(a,a)_k subject to sum(a) = 1, a >= 0.

This can be written as: minimize t subject to conv(a,a)_k <= t for all k, sum(a) = 1, a >= 0.

The constraint conv(a,a)_k <= t is: sum_i a_i * a_{k-i} <= t, which is a quadratic inequality. This is NOT a standard SOCP constraint.

However, we can use the following trick: for non-negative a, conv(a,a)_k = sum_i a_i * a_{k-i} <= sqrt(sum a_i^2 * sum a_{k-i}^2) by Cauchy-Schwarz. But this gives an upper bound, not an equality.

Another approach: **Semidefinite relaxation**. Let X = a * a^T (outer product, rank 1). Then conv(a,a)_k = sum_{i: valid} X[i, k-i]. The problem becomes:
minimize t
subject to sum_{i: valid} X[i, k-i] <= t for all k
         trace(X) + 2*sum_{i<j} X[i,j] = 1 (from (sum a)^2 = 1)
         X >= 0, rank(X) = 1
         sum_i X[i,i] + 2*sum_{i<j} X[i,j] = 1

Dropping the rank constraint gives an SDP relaxation. But this is a very large SDP (4096x4096), which might be too slow.

What about a **second-order cone relaxation**? We can write:
sum_i a_i * a_{k-i} <= t

For non-negative a, this is equivalent to: there exists r >= 0 such that a_i <= r and a_{k-i} <= t/r for all i... no, that's not right.

Actually, for non-negative a, the product a_i * a_{k-i} can be bounded using the AM-GM inequality: a_i * a_{k-i} <= (a_i^2 + a_{k-i}^2) / 2. But this gives a loose bound.

Let me abandon the SOCP/SDP approach and focus on the heuristic optimization.

**New idea: Use the structure of the active set.**

At the optimum, there should be a set of lags S (the "active set") where conv(a,a)_k = M (the maximum), and for k not in S, conv(a,a)_k < M.

The KKT conditions give us:
- For each i with a_i > 0: sum_{k in S} w_k * (a[k-i] + a[i+k]) = lambda (the Lagrange multiplier for the sum constraint)
- For each i with a_i = 0: sum_{k in S} w_k * (a[k-i] + a[i+k]) <= lambda

Where w_k are the weights for the active lags (which should be non-negative and sum to 1).

This gives us a system of equations that we can solve iteratively:
1. Guess the active set S
2. Solve the linear system for a and lambda
3. Check if a >= 0 and if the inactive lags are indeed less than M
4. Update S and repeat

For a given active set S with |S| = m, the equations are:
sum_{k in S} w_k * (a[k-i] + a[i+k]) = lambda for all i with a_i > 0

This is a linear system in the variables {a_i : a_i > 0} and lambda.

But we also need sum_i a_i = 1 (normalization).

So we have (number of positive a_i) + 1 equations and (number of positive a_i) + 1 unknowns (the a_i's and lambda). We can solve this linear system.

Then we check:
- Are all a_i >= 0?
- For k not in S, is conv(a,a)_k < M where M = conv(a,a)_k for k in S?

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses structured initialization, FISTA with adaptive smoothing, Adam in log-space with
aggressive fine-tuning, active-set equalization heuristics, and multi-scale refinement.
Key improvements: finer beta schedules, active-set KKT-inspired equalization,
and more aggressive final polishing with very small smoothing parameters.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_w(a, w, n):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a via FFT."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, w, n):
    """Gradient in log-space where a = softmax(z)."""
    g = _compute_grad_w(a, w, n)
    scalar = float(np.sum(g) - np.dot(a, g))
    return a * scalar


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA for smoothed max-convolution on the simplex with adaptive step."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / max(beta, 1e-12)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    stuck_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_w(y, w, n)
        a_next = _proj_simplex(y - step * grad)

            c_new = _full_convolve(a_next, a_next)

            L = 4.0 * n * max_c_new / max(beta, 1e-12)
            step = 1.0 / L
            sc = _score_norm(a_next)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_next.copy()
                stuck_count = 0
            else:
                stuck_count += 1
            if stuck_count > 20:
                break

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with adaptive learning rate decay."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        phase_best = best_score
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_logspace(a, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                if no_improve > 300:
                    break

        if time.monotonic() > deadline:
            break

    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_equalize(a, deadline, n_top=8, n_outer=20):
    """Try to equalize the top convolution entries using KKT-inspired updates."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for outer in range(n_outer):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        idx = np.argsort(c)[::-1][:n_top]
        top_vals = c[idx]
        spread = top_vals.max() - top_vals.min()

        if spread < 1e-10 * M:
            break

        # Compute "excess" at each top lag
        # Try to reduce high lags and allow low lags to increase
        w = np.zeros(len(c))
        for i in idx:
            w[i] = 1.0 / n_top

        grad = _compute_grad_w(best_a, w, n)
        grad_centered = grad - grad.mean()

        # Line search
        best_alpha = None
        best_sc = best_score
        for alpha in [0.001, 0.005, 0.01, 0.05, 0.1, 0.5]:
            a_try = best_a - alpha * grad_centered
            a_try = _proj_simplex(a_try)
            sc = _score_norm(a_try)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_alpha = a_try.copy()

        if best_alpha is not None:
            best_a = best_alpha
            if best_sc < best_score:
                best_score = best_sc
        else:
            break

    return best_a, best_score

    print(f"\n=== Phase 2: Alternative initializations (n={n_main}) ===")
    alternates = [
        ("ramp", _make_init_ramp(n_main, 1)),
        ("ramp_rev", _make_init_ramp(n_main, -1)),
        ("geom_099", _make_init_geometric(n_main, 0.99)),
        ("geom_097", _make_init_geometric(n_main, 0.97)),
        ("asym_03", _make_init_asym(n_main, 0.3)),
        ("asym_05", _make_init_asym(n_main, 0.5)),
        ("sin1", _make_init_sinusoidal(n_main, 1, 0.4)),
        ("sin2", _make_init_sinusoidal(n_main, 2, 0.2)),
    ]

    time_per_alt = max(20.0, (deadline - time.monotonic() - 180) / len(alternates))

    for name, a_init in alternates:
        if time.monotonic() > deadline - 120:
            break
        alt_deadline = time.monotonic() + time_per_alt
        a_try = _optimize_sequence(a_init, alt_deadline, aggressive=False)
        sc_try = _score(a_try)
        print(f"  {name}: score={sc_try:.10f}")
        if sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            print(f"    -> New best!")

    # Phase 3: Try length 8192 by interpolating best
    print(f"\n=== Phase 3: Try n=8192 ===")
    if time.monotonic() < deadline - 200:
        n_8k = 8192
        a_interp = _interp_profile(best_seq, n_8k)
        phase3_deadline = min(deadline - 100, time.monotonic() + 250)
        a_8k = _optimize_sequence(a_interp, phase3_deadline, aggressive=False)
        sc_8k = _score(a_8k)
        print(f"  n=8192: score={sc_8k:.10f}")
        if sc_8k < best_score:
            best_score = sc_8k
            best_seq = a_8k.copy()
            print(f"    -> New best (n=8192)!")
            n_main = 8192

    # Phase 4: Multi-restart from best
    print(f"\n=== Phase 4: Multi-restart fine-tuning ===")
    n_best = len(best_seq)
    a_base = best_seq.copy() / best_seq.sum()

    n_restarts = 6
    per_restart = max(15.0, (deadline - time.monotonic() - 60.0) / n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > deadline - 40:
            break

        if i == 0:
            p_start = _perturb(a_base, 0.005, rng)
        elif i == 1:
            p_start = _perturb(a_base, 0.02, rng)
        elif i == 2:
            p_start = _perturb(a_base, 0.05, rng)
        elif i == 3:
            ramp = _make_init_ramp(n_best, 1)
            p_start = 0.9 * a_base + 0.1 * ramp
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        elif i == 4:
            geom = _make_init_geometric(n_best, 0.995)
            p_start = 0.9 * a_base + 0.1 * geom
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            # Block perturbation
            p_start = a_base.copy()
            n_blocks = 32
            block_size = n_best // n_blocks
            b = rng.integers(0, n_blocks)
            start = b * block_size
            end = min(n_best, start + block_size)
            p_start[start:end] *= 1.0 + 0.1 * rng.standard_normal(end - start)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

        c_s = _full_convolve(p_start, p_start)
        max_cs = float(c_s.max())
        restart_betas = [0.0003 * max_cs, 0.0001 * max_cs, 0.00003 * max_cs,
                         0.00001 * max_cs, 0.000003 * max_cs]
        p_best, _ = _adam_logspace(
            p_start, time.monotonic() + per_restart,
            restart_betas, lr_init=0.004, max_iter_total=200000
        )
        sc_full = _score(p_best)
        print(f"  Restart {i}: score={sc_full:.10f}")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_best.copy()
            print(f"    -> New best!")

    # Phase 5: Ultra-fine final polish
    print(f"\n=== Phase 5: Ultra-fine final polish ===")
    a_pol = best_seq.copy()
    a_pol = a_pol / a_pol.sum()

    c_pol = _full_convolve(a_pol, a_pol)
    max_cp = float(c_pol.max())

    polish_betas = [0.00002 * max_cp, 0.000005 * max_cp, 0.000001 * max_cp,
                    0.0000003 * max_cp, 0.0000001 * max_cp]
    polish_deadline = deadline - 5
    if time.monotonic() < polish_deadline:
        a_pol, _ = _adam_logspace(
            a_pol, polish_deadline,
            polish_betas, lr_init=0.001, max_iter_total=300000
        )
        sc_pol = _score(a_pol)
        print(f"  Final polish: score={sc_pol:.10f}")
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()

    # Phase 6: One more active-set equalization pass
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_eq, _ = _active_set_equalize(best_seq, deadline - 1, n_top=10, n_outer=50)
        sc_eq = _score(a_eq)
        if sc_eq < best_score:
            best_score = sc_eq
            best_seq = a_eq.copy()
            print(f"  Active-set final: score={sc_eq:.10f}")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\n=== FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s ===")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
