            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _exp_map(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                    best_iter = t
                else:
                    no_improve += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.001, current_lr * lr_decay_factor)

            if no_improve > 30 and t > 500:
                current_lr = max(0.001, current_lr * 0.95)

            # Periodic restart to escape local minima
            if t > restart_every and no_improve > 50:
                z += restart_sigma * np.random.standard_normal(n)
                z = z - z.mean()
                no_improve = 0
                current_lr = lr_init * 0.5

    a_final = _exp_map(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten_interleaved(a_init, n_iter=200, deadline=None, flatten_strength=0.5):
    """Interleave spectral flattening with time-domain peak reduction."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        if it % 3 == 0:
            # Spectral flatten step
            A = np.fft.rfft(a, N_fft)
            mag = np.abs(A)
            # Reduce peaks in spectrum (keep DC)
            if len(mag) > 1:
                non_dc_mag = mag[1:]
                avg_non_dc = non_dc_mag.mean()
                target = np.ones_like(mag)
                target[0] = mag[0]
                target[1:] = avg_non_dc
                # Blend
                mag_new = (1 - flatten_strength) * mag + flatten_strength * target
                phase = A / (mag + 1e-30)
                A_new = mag_new * phase
                a_new = np.fft.irfft(A_new, N_fft)[:n]
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
        else:
            # Time-domain peak reduction
            c = _full_convolve(a, a)
            cm = float(c.max())
            # Top-3 peaks
            top_k = min(3, len(c))
            top_idx = np.argsort(c)[::-1][:top_k]
            w = np.zeros(len(c))
            w[top_idx] = 1.0 / top_k
            grad = _compute_grad_simplex(a, w)
            step = 0.3 / (1.0 + 0.005 * it)
            a_new = _proj_simplex(a - step * grad)
            a = a_new

        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _coord_polish(a_init, n_steps=5000, deadline=None):
    """Coordinate descent polishing: randomly pick elements and adjust them."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    # Precompute convolution
    c = _full_convolve(a, a)

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Pick 1-3 random indices to modify
        n_mod = rng.integers(1, 4)
        idxs = rng.integers(0, n, size=n_mod)

        best_local = a.copy()
        best_local_sc = best_score

        for idx in idxs:
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                test = best_local.copy()
                test[idx] = a[idx] * mult
                test = np.maximum(test, 0)
                s_t = test.sum()
                if s_t < 1e-15:
                    continue
                test = test / s_t
                sc = _score_norm(test)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = test.copy()

        if best_local_sc < best_score:
            best_score = best_local_sc
            best_a = best_local.copy()
            a = best_local.copy()
            # Recompute convolution
            c = _full_convolve(a, a)

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _exp_map(z)
