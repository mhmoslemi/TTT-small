def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:

    N = max(2, N)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
LP-alternating optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based alternating direction method with gradient polishing,
subgradient descent, and multi-restart diversification.
"""
import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _make_A_vals(f, n):
    """Compute values for the convolution matrix given pre-computed structure."""
    total = n * n
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(n):
        count = k + 1
        vals[idx:idx+count] = f[k::-1]
        idx += count
    for k in range(n, 2 * n - 1):

        vals[idx:idx+count] = f[n-1:n-1-count:-1]

def _lp_alternate_step(f, M, A_rows, A_cols, n):
    """Solve the LP: max sum(g) s.t. A@g <= M, g >= 0. Return g."""
    vals = _make_A_vals(f, n)
    A = csr_matrix((vals, (A_rows, A_cols)), shape=(2 * n - 1, n))
    
    c = -np.ones(n)
    b_ub = np.ones(2 * n - 1) * M
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs')
        if res.success:
            return res.x
        else:
            # Fallback: return f itself
            return f.copy()
    except Exception:
        return f.copy()


def _lp_alternating(f_init, deadline, max_iters=500, t_mix=0.15):
    """LP-based alternating optimization."""
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    
    A_rows, A_cols = _precompute_A_structure(n)
    
    best_f = f.copy()
    best_score = _score(f)
    stall_count = 0
    
    print(f"  LP-Alt: n={n}, initial score={best_score:.10f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(f, f)
        M = float(c.max())
        
        g = _lp_alternate_step(f, M, A_rows, A_cols, n)
        
        # Rescale g to have same sum as f
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g = g * (f.sum() / g_sum)
        
        # Try different mixing parameters
        best_t = 0.0
        best_new_score = best_score
        
        for t in [t_mix, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = (1 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-15:
                continue
            f_try = f_try / s
            sc = _score(f_try)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            
            sc = _score(f)
            if sc < best_score - 1e-12:
                best_score = sc
                best_f = f.copy()
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        
        if it % 50 == 0:

        if stall_count > 100:

            f = np.maximum(f, 0.0)
            f = f / f.sum()
            stall_count = 0
    
    return best_f, best_score


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(8000, max(100, int(remaining / (est_time * 15))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.0003)
        z = z - step * grad
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 3:
                    lr *= 0.7
                    if lr < 1e-6:
                        break
                else:
                    lr *= 0.93

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _local_search(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.005

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        idx = int((rnd * 997) % n)
        jdx = int((rnd * 997 + 7) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        for delta_frac in [0.3, 0.5, 0.7]:
            s_pair = best_a[idx] + best_a[jdx]
            new_val_i = s_pair * (1.0 - delta_frac + step)
            new_val_j = s_pair * (delta_frac - step)
            if new_val_i < 0 or new_val_j < 0:
                continue

            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True

        if not improved and rnd > 20:
            step *= 0.7
            if step < 1e-6:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()
