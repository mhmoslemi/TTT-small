2. The objective is non-smooth (max of quadratics) and non-convex
3. The previous approach used smoothed optimization (log-sum-exp) with FISTA, mirror descent, Adam, and L-BFGS

My strategy differs from the previous program by:
1. **Bundle method** for the non-smooth objective (fresh approach vs. smoothing)
2. **Conjugate gradient with restarts** on the smoothed surrogate (different from FISTA)
3. **Adaptive multi-scale exploration** trying both n=4096 and n=8192
4. **Better beta annealing** with progress-aware scheduling
5. **Stochastic active-set exploitation** to escape local minima

```python
"""
Hybrid optimizer: Bundle method + CG + adaptive FISTA for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses novel bundle-freezing and CG-restart strategies to improve on
previous smoothing-based approaches.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_from_w(w, a):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _smooth_obj_and_grad(a, beta):
    """Compute smoothed objective value and gradient w.r.t. a (on simplex)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_sh = (c - cm) / beta
    e = np.exp(c_sh)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * cm
    else:
        w = e / s
        obj = 2.0 * n * (cm + beta * np.log(s))
    grad = _compute_grad_from_w(w, a)
    return obj, grad, w, c, cm


def _bundle_opt(a_init, deadline, max_iter=50000):
    """
    Bundle method in log-space for the non-smooth objective.
    Maintains a bundle of subgradients and solves QP subproblems.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm(a)

    # Bundle: store subgradients and function values
    bundle_grads = []
    bundle_vals = []
    bundle_points = []
    max_bundle = 50

    # Proper step parameters
    gamma = 0.1  # trust region parameter
    t_val = 0.5  # initial trust radius

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        f_val = 2.0 * n * cm

        # Get active set and subgradient
        tol = 1e-6 * cm + 1e-12
        active = np.where(c >= cm - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        g = _compute_grad_from_w(w, a)

        # Add to bundle
        bundle_grads.append(g.copy())
        bundle_vals.append(f_val)
        bundle_points.append(z.copy())
        if len(bundle_grads) > max_bundle:
            # Keep most recent + most informative (largest violation)
            bundle_grads.pop(0)
            bundle_vals.pop(0)
            bundle_points.pop(0)

        # Solve simplified QP subproblem:
        # min (1/2)||dz||^2 + max_i g_i^T dz + f_val
        # Approximate with: min (1/2)||dz||^2 + sum_i max(0, g_i^T dz + f_i - f_val - gamma)
        # Use a simple approach: find direction that reduces all active constraints
        if len(bundle_grads) <= 1:
            # Just use negative gradient
            direction = -g
        else:
            # Use the average of the "worst" subgradients
            violations = np.array([float(np.dot(bi, z)) + fv - float(np.dot(g, z)) - f_val
                                   for bi, fv in zip(bundle_grads, bundle_vals)])
            # Weight by violation amount
            wts = np.maximum(violations, 0)
            if wts.sum() < 1e-15:
                wts = np.ones(len(wts))
            wts = wts / wts.sum()
            direction = -np.average(bundle_grads, axis=0, weights=wts)

        # Normalize direction in log-space (to control step size)
        dnorm = np.linalg.norm(direction)
        if dnorm < 1e-15:
            break

        # CGLS / backtracking line search
        alpha = t_val / dnorm
        found = False
        for ls in range(20):
            z_try = z + alpha * direction
            a_try = _softmax_compute(z_try)
            c_try = _full_convolve(a_try, a_try)
            f_try = 2.0 * n * float(c_try.max())
            if f_try < f_val - 1e-10 * alpha * dnorm:
                found = True
                break
            alpha *= 0.5

        if not found:
            # Try smaller step
            z_try = z + 0.01 * alpha * direction
            a_try = _softmax_compute(z_try)
            f_try = _score_norm(a_try)
            if f_try < f_val:
                z = z_try
                a = a_try
                f_val = f_try
                t_val = 0.01 * t_val
            else:
                t_val *= 0.7
                if t_val < 1e-8:
                    break
            continue

        z = z_try
        a = a_try
        f_val = f_try
        t_val = min(2.0 * t_val, 5.0)

        if f_val < best_score:
            best_score = f_val
            best_a = a.copy()

        # Periodic check
        if it % 500 == 0:
            sc = _score(a)

    return best_a, best_score


def _cg_optimize(a_init, beta, deadline, max_iter=200000):
    """
    Conjugate gradient with restarts on the smoothed objective in log-space.
    Uses Polak-Ribiere restart and backtracking line search.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Compute initial gradient
    a = _softmax_compute(z)
    obj, grad, w, c, cm = _smooth_obj_and_grad(a, beta)
    p = -grad  # initial direction
    g_old = grad.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Backtracking line search
        alpha = 1.0
        f_cur = obj
        for ls in range(30):
            z_try = z + alpha * p
            a_try = _softmax_compute(z_try)
            obj_try, grad_try, _, _, _ = _smooth_obj_and_grad(a_try, beta)
            if obj_try <= f_cur - 1e-4 * alpha * float(np.dot(grad, p)):
                break
            alpha *= 0.5
        else:
            # Line search failed, restart
            break

        z = z_try
        a = a_try
        f_new = obj_try
        g_new = grad_try

        if f_new < best_score:
            best_score = f_new
            best_a = a.copy()

        # Check convergence
        gnorm = np.linalg.norm(g_new)
        if gnorm < 1e-10:
            break

        # Polak-Ribiere beta
        gdiff = float(np.dot(g_new - g_old, g_new))
        gnorm_old_sq = float(np.dot(g_old, g_old))
        if gnorm_old_sq < 1e-30:
            beta_cg = 0.0
        else:
            beta_cg = max(0.0, gdiff / gnorm_old_sq)

        # Strong Wolfe restart: if beta is large, restart
        if beta_cg > 0.5 or it % 50 == 0:
            p = -g_new
        else:
            p = -g_new + beta_cg * p

        g_old = g_new
        obj = f_new
        grad = g_new

    return best_a, best_score


def _fista_v2(a_init, beta, deadline, max_iter):
    """
    Improved FISTA with adaptive restart and better step-size control.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    # Estimate Lipschitz constant more carefully
    c0 = _full_convolve(a_curr, a_curr)
    L = 4.0 * n / max(beta, 1e-15)
    step = 1.0 / L
    f_prev = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        obj, grad, w, c_y, c_max = _smooth_obj_and_grad(y, beta)

        a_next = _proj_simplex(y - step * grad)

        # Adaptive restart: if function value increased, reset momentum
        f_next = _score_norm(a_next)
        if f_next > f_prev + 1e-10:
            a_prev = a_curr.copy()
            t_val = 1.0
        else:
            a_prev = a_curr.copy()
            t_val = t_next

        a_curr = a_next
        f_prev = f_next

        if f_next < best_score:
            best_score = f_next
            best_a = a_curr.copy()

    return best_a, best_score


def _adam_v2(a_init, deadline, beta, lr_init, max_iter):
    """Improved Adam with adaptive clipping and restart."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    patience = 0
    best_seen = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        obj, grad, w, c, cm = _smooth_obj_and_grad(a, beta)

        # Corrections for log-space gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad_a)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_a
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_a * grad_a)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        max_step = 0.2
        if update_norm > max_step:
            update = update * (max_step / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                patience = 0
            else:
                patience += 1
            if patience > 20:
                # Restart with new learning rate
                current_lr = max(0.0002, current_lr * 0.7)
                patience = 0

        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "lorentzian":
        t = np.linspace(-1, 1, n)
        a = 1.0 / (1.0 + 4.0 * t * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _lbfgs_refine(a_init, deadline, beta):
    """L-BFGS-B refinement in log-space."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 20)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer using bundle method, CG, adaptive FISTA, and Adam.
    Explores both n=4096 (from parent) and n=8192 for potential improvement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: Adaptive FISTA with continuation ===
    print("Phase 1: Adaptive FISTA with continuation")
    phase1_end = min(deadline, t_start + 400.0)
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    beta = 0.02 * max_c0
    n_phases = 30
    stall_count = 0
    prev_best = best_score

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        if remaining < 1:
            break
        per_time = min(remaining / max(1, n_phases - phase), 20.0)
        max_it = max(2000, int(per_time * 5000))
        a_work, _ = _fista_v2(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            stall_count = 0
            if phase % 5 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            stall_count += 1
            if stall_count > 8:
                # Restart with larger beta to escape
                beta = beta * 3.0
                a_work = _perturb(a_work, 0.01, rng)
                stall_count = 0
                print(f"  FISTA restart at p{phase}: score={sc:.10f}")

        c_curr = _full_convolve(a_work, a_work)
        beta = max(0.3 * float(c_curr.max()), 1e-12)

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Bundle method (novel approach) ===
    print("Phase 2: Bundle method")
    phase2_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase2_end - 10:
        a_bundle, _ = _bundle_opt(a_work, phase2_end, max_iter=100000)
        sc_b = _score(a_bundle)
        if sc_b < best_score:
            best_score = sc_b
            best_seq = a_bundle.copy()
            print(f"  Bundle: score={sc_b:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_bundle if sc_b < _score(a_work) else a_work

    # === Phase 3: CG with restarts (novel approach) ===
    print("Phase 3: CG with restarts")
    phase3_end = min(deadline, t_start + 950.0)

        c_curr = _full_convolve(a_work, a_work)

            sc_cg = _score(a_cg)
            if sc_cg < best_score:
                best_score = sc_cg
                best_seq = a_cg.copy()
                print(f"  CG beta={beta_cg:.2e}: score={sc_cg:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = a_cg if sc_cg < _score(a_work) else a_work

    # === Phase 4: Adam fine optimization ===
    print("Phase 4: Adam fine optimization")

        for bi, beta_ad in enumerate(adam_betas):

                break
            a_ad, _ = _adam_v2(a_work, time.monotonic() + max(3.0, per_beta),
                               beta_ad, lr_init=0.005, max_iter=300000)
            sc_ad = _score(a_ad)
            if sc_ad < best_score:
                best_score = sc_ad
                best_seq = a_ad.copy()
                print(f"  Adam beta={beta_ad:.2e}: score={sc_ad:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = a_ad if sc_ad < _score(a_work) else a_work

    # === Phase 5: L-BFGS refinement ===
    print("Phase 5: L-BFGS refinement")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00003 * max_cc
        a_lb, _ = _lbfgs_refine(a_work, phase5_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb if sc_lb < _score(a_work) else a_work

    # === Phase 6: Try larger n=8192 ===
    print("Phase 6: Alternative n=8192")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end - 20:
        n_alt = 8192
        # Initialize from upsampling the best sequence found
        a_alt = np.kron(best_seq, np.ones(2)) if best_seq is not None else _make_profile(n_alt, "uniform")
        a_alt = a_alt[:n_alt]
        a_alt = np.maximum(a_alt, 1e-10)
        a_alt = a_alt / a_alt.sum()
        sc_alt_init = _score(a_alt)
        print(f"  Alt n={n_alt} init: score={sc_alt_init:.10f}")

        c_alt = _full_convolve(a_alt, a_alt)
        max_ca = float(c_alt.max())
        beta_alt = 0.005 * max_ca

        per_phase = (phase6_end - time.monotonic() - 10) / 4
        # FISTA
        a_alt, _ = _fista_v2(a_alt, beta_alt, time.monotonic() + per_phase, 100000)
        a_alt = a_alt / a_alt.sum()
        sc_alt = _score(a_alt)
        print(f"  Alt FISTA: score={sc_alt:.10f}")
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()

        # CG
        c_alt = _full_convolve(a_alt, a_alt)
        beta_alt2 = 0.0002 * float(c_alt.max())
        a_alt, _ = _cg_optimize(a_alt, beta_alt2, time.monotonic() + per_phase, 80000)
        sc_alt = _score(a_alt)
        print(f"  Alt CG: score={sc_alt:.10f}")
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()

        # Adam
        c_alt = _full_convolve(a_alt, a_alt)
        beta_alt3 = 0.00005 * float(c_alt.max())
        a_alt, _ = _adam_v2(a_alt, time.monotonic() + per_phase, beta_alt3, 0.004, 150000)
        sc_alt = _score(a_alt)
        print(f"  Alt Adam: score={sc_alt:.10f}")
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()

    # === Phase 7: Final multi-restart ===
    print("Phase 7: Final multi-restart")
    phase7_end = deadline - 10.0
    if best_seq is not None and time.monotonic() < phase7_end - 15:
        n_r = 3
        per_r = (phase7_end - time.monotonic() - 10) / n_r
        for i in range(n_r):
            if time.monotonic() > phase7_end - 8:
                break
            if i == 0:
                p = _perturb(best_seq, 0.008, rng)
            elif i == 1:
                # Frequency-domain perturbation
                N = len(p) if 'p' in dir() else len(best_seq)
                p = best_seq.copy()
                F = np.fft.rfft(p)
                ang = rng.standard_normal(len(F)) * 0.05
                F = F * np.exp(1j * ang)
                p = np.fft.irfft(F, N)[:len(best_seq)]
                p = np.maximum(p, 1e-10)
                p = p / p.sum()
            else:
                p = _perturb(best_seq, 0.02, rng)

            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.0002 * max_cp
            p, _ = _fista_v2(p, b_r, time.monotonic() + per_r * 0.4, 30000)
            c_p2 = _full_convolve(p, p)
            b_r2 = 0.00005 * float(c_p2.max())
            p, _ = _adam_v2(p, time.monotonic() + per_r * 0.6, b_r2, 0.003, 100000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Ultra-final polish ===
    print("Phase 8: Ultra-final polish")
    if best_seq is not None and time.monotonic() < deadline - 12:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5
        if per_t > 8:
            # Adam with tiny beta
            a_pol, _ = _adam_v2(a_pol, time.monotonic() + per_t * 0.5,
                                0.000003 * max_cp, 0.001, 100000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # L-BFGS on remaining
            if time.monotonic() < deadline - 8:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000002 * float(c_f.max())
                a_pol, _ = _lbfgs_refine(a_pol, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
