"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive Adam in log-space with power-weighted peak targeting, fast coordinate descent
with incremental convolution updates, and multi-phase parameter scheduling.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_power_weights(c, p, k):
    """Compute power-based weights over top-k elements. w_j = (c_j/c_max)^p normalized."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    if c_max < 1e-30:
        w = np.zeros(n_c)
        w[topk_indices[0]] = 1.0
        return w
    ratios = c_topk / c_max
    exp_vals = ratios ** p
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _topk_softmax_weights(c, beta, k):
    """Compute softmax weights over top-k elements."""
    n_c = len(c)

        k = n_c

    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a. = 2*cross_correlation(a, w)."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    """Gradient in log-space for normalized problem. a is on simplex (sum=1)."""
    grad_a = _grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)

            best_f = np.inf

            for delta in candidates:
                new_ai = a_i + delta
                if new_ai < -1e-10:
                    continue
                new_ai = max(0.0, new_ai)
                new_S = S + delta
                if new_S < 1e-10:
                    continue

                # Compute new max of conv (only need to check affected positions + global max)
                # Affected positions: j in [j_lo, j_hi]
                # conv[j] + change
                changes = np.zeros(j_hi - j_lo + 1)
                for idx, j in enumerate(range(j_lo, j_hi + 1)):
                    jmi = j - i
                    if 0 <= jmi < n:
                        if j == 2 * i:
                            changes[idx] = 2.0 * a_i * delta + delta * delta
                        else:
                            changes[idx] = 2.0 * delta * a[jmi]

                new_conv_seg = conv[j_lo:j_hi + 1] + changes
                seg_max = float(new_conv_seg.max())

                # Unaffected max: we need max of conv outside [j_lo, j_hi]
                # For efficiency, precompute: the global max minus the affected part
                # Quick approximation: use current global max (might overestimate, but safe)
                unaff_max = float(conv.max())
                # This is an upper bound; the true new max is max(seg_max, true_unaff_max)
                # For candidate selection, using the upper bound is conservative (won't miss improvements)
                # But it might reject valid improvements. Let's compute it properly if needed.
                # Actually, the unaffected part doesn't change, so:
                if j_lo > 0:
                    unaff_left = float(conv[:j_lo].max())
                else:
                    unaff_left = 0.0
                if j_hi < 2 * n - 2:
                    unaff_right = float(conv[j_hi + 1:].max())
                else:
                    unaff_right = 0.0
                true_unaff = max(unaff_left, unaff_right)

                new_max = max(seg_max, true_unaff)
                f_val = 2.0 * n * new_max / (new_S * new_S)

                if f_val < best_f:
                    best_f = f_val
                    best_delta = delta

            if best_delta != 0.0:
                # Apply the change
                delta = best_delta
                new_ai = a_i + delta
                a[i] = max(0.0, new_ai)
                S_new = S + delta
                S = S_new

                # Update conv incrementally
                for j in range(j_lo, j_hi + 1):
                    jmi = j - i
                    if 0 <= jmi < n:
                        if j == 2 * i:
                            conv[j] += 2.0 * a_i * delta + delta * delta
                        else:
                            conv[j] += 2.0 * delta * a[jmi]

                improved += 1

                if improved % 500 == 0:
                    sc = _score_norm(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()

        # End of pass
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _coord_descend_fast2(a, deadline, top_k_w, max_passes=5000):
    """Even faster coordinate descent - vectorized inner loop."""
    n = len(a)
    a = a.copy()
    conv = _full_convolve(a, a)
    S = float(a.sum())
    best_score = _score_norm(a)
    best_a = a.copy()
    rng = np.random.default_rng(42)

    passes = 0
    while passes < max_passes:
        if time.monotonic() > deadline:
            break
        passes += 1
        order = rng.permutation(n)
        improved = 0

        for i in order:
            a_i = a[i]
            if a_i < 1e-14:
                continue

            j_lo = max(0, i)
            j_hi = min(2 * n - 2, i + n - 1)
            seg_len = j_hi - j_lo + 1

            # Get segment of conv
            conv_seg = conv[j_lo:j_hi + 1]

            # Find top-k in segment
            if seg_len > top_k_w:
                top_local = np.argpartition(conv_seg, -top_k_w)[-top_k_w:]
            else:
                top_local = np.arange(seg_len)

            # Generate candidates
            best_f = 2.0 * n * float(conv.max()) / (S * S)
            best_delta = 0.0

            for idx in top_local:
                j = j_lo + idx
                jmi = j - i
                if 0 <= jmi < n:
                    ajmi = a[jmi]
                    if j == 2 * i:
                        d_j = 2.0 * a_i
                    else:
                        d_j = 2.0 * ajmi
                    if d_j > 1e-14:
                        delta_star = S - 2.0 * conv[j] / d_j
                        delta_star = max(-a_i + 1e-8, min(delta_star, S))
                        # Evaluate
                        new_S = S + delta_star
                        if new_S < 1e-10:
                            continue
                        # Update only this one position in conv for quick eval
                        if j == 2 * i:
                            new_val = conv[j] + 2.0 * a_i * delta_star + delta_star * delta_star
                        else:
                            new_val = conv[j] + 2.0 * delta_star * ajmi
                        # New max is at least new_val, at most max(new_val, old_max)
                        old_max = float(conv.max())
                        new_max_est = max(new_val, old_max)
                        f = 2.0 * n * new_max_est / (new_S * new_S)
                        if f < best_f:
                            # Verify with full segment update
                            changes = np.zeros(seg_len)
                            for k2 in range(seg_len):
                                j2 = j_lo + k2
                                jmi2 = j2 - i
                                if 0 <= jmi2 < n:
                                    if j2 == 2 * i:
                                        changes[k2] = 2.0 * a_i * delta_star + delta_star * delta_star
                                    else:
                                        changes[k2] = 2.0 * delta_star * a[jmi2]
                            new_seg = conv_seg + changes
                            seg_max = float(new_seg.max())
                            # Unaffected max
                            if j_lo > 0:
                                ul = float(conv[:j_lo].max())
                            else:
                                ul = 0.0
                            if j_hi < 2 * n - 2:
                                ur = float(conv[j_hi + 1:].max())
                            else:
                                ur = 0.0
                            true_new_max = max(seg_max, ul, ur)
                            f_true = 2.0 * n * true_new_max / (new_S * new_S)
                            if f_true < best_f:
                                best_f = f_true
                                best_delta = delta_star

            if best_delta != 0.0:
                delta = best_delta
                a[i] = max(0.0, a_i + delta)
                S += delta
                # Update conv
                for j in range(j_lo, j_hi + 1):
                    jmi = j - i
                    if 0 <= jmi < n:
                        if j == 2 * i:
                            conv[j] += 2.0 * a_i * delta + delta * delta
                        else:
                            conv[j] += 2.0 * delta * a[jmi]
                improved += 1

        if improved == 0:
            break

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4, 5]:
            a += 0.15 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.001)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.005, 0.995, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(0)
        a = rng.random(n)
        # Smooth
        kernel = np.ones(16) / 16
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fista_simplex(a_init, deadline, beta, top_k, max_iter, step_scale=1.0):
    """FISTA on simplex with top-K softmax."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = step_scale / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _grad_conv(y, w)

        # Projected gradient step onto simplex
        a_next = y - step * grad
        # Simplex projection
        n_ = len(a_next)
        u = np.sort(a_next)[::-1]
        cssv = np.cumsum(u)
        idx_ = np.arange(1, n_ + 1, dtype=np.float64)
        cond = u * idx_ > (cssv - 1.0)
        if np.any(cond):
            rho = np.where(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_next = np.maximum(a_next - theta, 0.0)
        else:
            a_next = np.ones(n_) / n_
        a_next = a_next / a_next.sum()

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    n_work = 4096

    # Main optimization loop with multiple strategies
    strategies = [
        ("adam_coarse_fine", 0.30),
        ("coord_descend_heavy", 0.35),
        ("adam_ultra_fine", 0.15),
        ("coord_descend_polish", 0.10),
        ("restarts", 0.08),
        ("reserve", 0.02),
    ]

    total_budget = deadline - time.monotonic()
    phase_times = {}
    for name, frac in strategies:
        phase_times[name] = total_budget * frac

    a_work = parent.copy() if len(parent) == n_work else _interp_profile(parent, n_work)
    a_work = np.maximum(a_work, 1e-12)
    a_work = a_work / a_work.sum()

    # === Phase 1: Adam coarse to fine ===
    print(f"\n=== Phase 1: Adam coarse-to-fine (budget={phase_times['adam_coarse_fine']:.0f}s) ===")
    phase1_end = time.monotonic() + phase_times['adam_coarse_fine']

    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    # Beta schedule: start coarse, end fine
    betas = []
    beta = 0.05 * max_cc
    while beta > 0.0005 * max_cc:
        betas.append(beta)
        beta *= 0.5
    if len(betas) > 10:
        betas = betas[:10]

    a_curr = a_work.copy()
    for pi, beta in enumerate(betas):
        if time.monotonic() > phase1_end - 2:
            break
        per_beta_time = min(phase1_end - time.monotonic() - 1, 15.0)
        top_k = max(20, min(150, int(100 * (1.0 - 0.08 * pi))))
        lr = 0.005 * (1.0 - 0.05 * pi)
        a_curr, sc = _adam_logspace(a_curr, time.monotonic() + per_beta_time,
                                    [beta], lr, top_k, max_iter=100000,
                                    use_power=False, power_p=30)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  Adam beta={beta:.6f} k={top_k}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = a_curr.copy()

    # === Phase 2: Heavy coordinate descent ===
    print(f"\n=== Phase 2: Coordinate descent heavy (budget={phase_times['coord_descend_heavy']:.0f}s) ===")
    phase2_end = time.monotonic() + phase_times['coord_descend_heavy']
    a_cd, sc_cd = _coord_descend_fast(a_work, phase2_end, n_passes=20000, top_k_w=30)
    a_cd = np.maximum(a_cd, 0.0)
    a_cd = a_cd / a_cd.sum()
    sc_cd_full = _score(a_cd)
    if sc_cd_full < best_score:
        best_score = sc_cd_full
        best_seq = a_cd.copy()
    print(f"  CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
    a_work = a_cd.copy()

    # === Phase 3: Ultra-fine Adam ===
    print(f"\n=== Phase 3: Ultra-fine Adam (budget={phase_times['adam_ultra_fine']:.0f}s) ===")
    phase3_end = time.monotonic() + phase_times['adam_ultra_fine']
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    # Very small betas
    betas_ult = [0.002 * max_cw, 0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw]
    a_ult, sc_ult = _adam_logspace(a_work, phase3_end, betas_ult,
                                   lr_init=0.003, top_k=30,
                                   max_iter=200000, use_power=False, power_p=50)
    a_ult = np.maximum(a_ult, 0.0)
    a_ult = a_ult / a_ult.sum()
    sc_ult_full = _score(a_ult)
    if sc_ult_full < best_score:
        best_score = sc_ult_full
        best_seq = a_ult.copy()
    print(f"  Ultra-fine Adam: score={sc_ult_full:.10f} t={time.monotonic()-t_start:.1f}s")
    a_work = a_ult.copy()

    # === Phase 4: More coordinate descent ===
    print(f"\n=== Phase 4: Coord descent polish (budget={phase_times['coord_descend_polish']:.0f}s) ===")
    phase4_end = time.monotonic() + phase_times['coord_descend_polish']
    a_cd2, sc_cd2 = _coord_descend_fast(a_work, phase4_end, n_passes=10000, top_k_w=20)
    a_cd2 = np.maximum(a_cd2, 0.0)
    a_cd2 = a_cd2 / a_cd2.sum()
    sc_cd2_full = _score(a_cd2)
    if sc_cd2_full < best_score:
        best_score = sc_cd2_full
        best_seq = a_cd2.copy()
    print(f"  CoordDescend2: score={sc_cd2_full:.10f} t={time.monotonic()-t_start:.1f}s")
    a_work = a_cd2.copy()

    # === Phase 5: Diversity restarts ===
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 5: Restarts (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 10) / n_restarts
        a_base = a_work.copy()

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            if ri == 0:
                pert = 0.01
                p_start = a_base + pert * rng.standard_normal(n_work)
            elif ri == 1:
                # Block swap
                p_start = a_base.copy()
                blk = n_work // 4
                s1 = rng.integers(0, blk * 2)
                s2 = rng.integers(blk * 2, n_work - blk)
                p_start[s1:s1+blk], p_start[s2:s2+blk] = p_start[s2:s2+blk].copy(), p_start[s1:s1+blk].copy()
            else:
                # Frequency perturbation
                freq = rng.uniform(2.0, 8.0)
                p_start = a_base + 0.02 * np.sin(2 * np.pi * freq * np.arange(n_work) / n_work)

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.003 * max_cs, 0.001 * max_cs, 0.0005 * max_cs]

            p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart * 0.6),
                                             betas_r, lr_init=0.005, top_k=40,
                                             max_iter=50000)
            # Quick CD polish
            cd_end = time.monotonic() + max(3, per_restart * 0.35)
            p_best, sc_best2 = _coord_descend_fast(p_best, cd_end, n_passes=2000, top_k_w=20)

            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
            print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = p_best.copy()

    # Final coordinate descent punch with remaining time
    if time.monotonic() < deadline - 5:
        print(f"\n=== Final CD punch (remaining: {deadline - time.monotonic():.1f}s) ===")
        a_final, sc_final_cd = _coord_descend_fast(a_work, deadline - 3, n_passes=50000, top_k_w=20)
        a_final = np.maximum(a_final, 0.0)
        a_final = a_final / a_final.sum()
        sc_final_full = _score(a_final)
        if sc_final_full < best_score:
            best_score = sc_final_full
            best_seq = a_final.copy()
        print(f"  Final CD: score={sc_final_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Ensure valid output
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
