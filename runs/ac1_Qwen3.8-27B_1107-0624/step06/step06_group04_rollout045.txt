"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: FISTA with aggressive beta-continuation, exact subgradient with Polyak step,
Adam log-space fine-tuning, local element-perturbation search, and multi-length exploration
(n=4096, 8192, 16384). Starts from parent if available.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _fista_backtrack(a_init, beta, deadline, max_iter):
    """FISTA with backtracking line search for the smoothed objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta * 1.5
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        step = 1.0 / L
        a_next = _proj_simplex(y - step * grad)

        if it % 150 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=10000, lr_decay_factor=0.97, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 80 and t > 2000:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_polyak(a_init, n_steps, deadline, lower_bound=1.01, top_k=50):
    """Exact subgradient with Polyak-type step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    current_score = best_score

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Use top-k peaks for the subgradient
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        # Weight by how close to max
        weights = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = weights ** 3
        ws = w_vec.sum()
        if ws > 0:
            w_vec = w_vec / ws
        else:
            w_vec[top_idx[0]] = 1.0

        grad = _compute_grad_simplex(a, w_vec)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break

        # Polyak step size
        ub = current_score
        step_size = (ub - lower_bound) / (gnorm * gnorm)
        step_size = min(step_size, 1.0)  # Cap step size
        step_size = max(step_size, 1e-8)

        a_new = _proj_simplex(a - step_size * grad)
        sc = _score_norm(a_new)

        if sc < current_score:
            a = a_new
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            # Halve step and retry
            step_size *= 0.5
            a_new = _proj_simplex(a - step_size * grad)
            if _score_norm(a_new) < current_score:
                a = a_new
                current_score = _score_norm(a)
                if current_score < best_score:
                    best_score = current_score
                    best_a = a.copy()

        # Reduce top_k over time for convergence
        if it % 500 == 0 and it > 0:
            top_k = max(5, top_k // 2)

    return best_a, best_score


def _local_search(a_init, n_kicks, deadline, kick_size=10):
    """Local search: randomly perturb a few elements, accept if improvement."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for it in range(n_kicks):
        if deadline is not None and time.monotonic() > deadline:
            break

        # Pick random indices to perturb
        idx = rng.integers(0, n, size=kick_size)
        a_trial = a.copy()
        # Multiplier between 0.5 and 2.0
        mults = rng.uniform(0.5, 2.0, size=kick_size)
        a_trial[idx] *= mults
        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial

    return best_a, best_score


def _interpolate(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score

                p_struct = np.maximum(p_struct, 1e-10)
                mix = 0.2 + 0.1 * i
                p_start = mix * p_struct + (1 - mix) * a_work
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_fast = 0.001 * max_cs
            p_opt, _ = _fista_smoothed(p_start, beta_fast,
                                        time.monotonic() + max(3.0, per_restart * 0.25), 4000)
            p_opt = p_opt / p_opt.sum()

            # Adam fine
            if time.monotonic() < phase5_end - 15:
                betas_r = [0.0003 * max_cs, 0.0001 * max_cs, 0.00005 * max_cs]
                per_t_r = phase5_end - time.monotonic() - 10
                p_opt, sc_r = _adam_logspace(p_opt, time.monotonic() + per_t_r,
                                              betas_r, lr_init=0.006, max_iter_total=60000)
                sc_r_full = _score(p_opt)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_seq = p_opt.copy()
                    n_best = len(best_seq)
                    print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print("=== Phase 6: Local search ===")

        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
