"""
Multi-phase optimizer using L-BFGS-B, Adam with adaptive active-set,
basin jumping, and coordinate descent to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1, so just 2*n*max(conv)"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=5,
                  eval_interval=200, clip_norm=0.8,
                  switch_to_active_at=0.3):
    """Adam in z-space with soft gradient, switching to active-set at a fraction of iterations."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if frac > switch_to_active_at:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_soft_gradient(a, c, T)
        
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.1:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_smooth(a_init, deadline, T, max_iter=2000):

def _lbfgs_smooth_z(a_init, deadline, T_start, T_end, max_iter=3000):
    """L-BFGS-B in z-space with smooth objective."""

    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    t_count = [0]

            return 0.0, np.zeros(n)
        t_count[0] += 1
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Interpolate T
        frac = min(1.0, t_count[0] / max_iter)
        T = T_start * (T_end / T_start) ** frac
        
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s_e
        
        f_val = 2.0 * n * T * np.log(max(s_e, 1e-300))
        
        # Gradient w.r.t. z: dz/dz = a*(p_conv - W) where p_conv is the conv-based gradient
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g_a = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]

    except Exception:

def _coordinate_descent(a, deadline, n_rounds=100, subset_size=32):
    """Optimize small subsets of coordinates while keeping others fixed."""
    n = len(a)
    rng = np.random.default_rng(12345)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        # Pick a random subset of coordinates
        idx = rng.choice(n, size=min(subset_size, n), replace=False)
        idx_sorted = np.sort(idx)
        
        # Get current values at these coordinates
        a_subset = best_a[idx_sorted].copy()
        a_rest = np.delete(best_a, idx_sorted)
        sum_rest = a_rest.sum()
        
        # Optimize the subset: minimize max_k conv(a_full, a_full) subject to a_subset >= 0
        # conv(a_full, a_full) depends on the subset through cross terms
        # This is a smaller QP problem
        
        # Simple approach: linesearch on a scaling factor
        # Or: do a few Adam steps on just the subset
        
        # Let's do a simple approach: try to reduce the max conv by adjusting the subset
        c = _fft_convolve(best_a, best_a)
        max_idx = np.argmax(c)
        max_val = c[max_idx]
        
        # Gradient at the maximizing lag
        # For lag max_idx, the gradient w.r.t. a[i] is:
        # a[max_idx - i] + a[i] (simplified, with boundary handling)
        grad_subset = np.zeros(subset_size)
        for j, i in enumerate(idx_sorted):
            k1 = max_idx - i
            k2 = i  # the "other" index
            g = 0.0
            if 0 <= k1 < n:
                g += best_a[k1]
            if k2 != k1 and 0 <= k2 < n:
                g += best_a[k2]
            # Handle the case where k1 == k2 (self-term)
            if k1 == i:  # This means max_idx = 2*i
                g = 2.0 * best_a[i]
            grad_subset[j] = g
        
        # Simple projected gradient step
        lr = 0.01 / (np.linalg.norm(grad_subset) + 1e-10)
        a_subset_new = a_subset - lr * grad_subset
        a_subset_new = np.maximum(a_subset_new, 0.0)
        
        # Renormalize to keep sum = 1
        total = a_subset_new.sum() + sum_rest
        if total > 0:
            a_subset_new = a_subset_new / total
            # Scale rest proportionally too... actually this changes rest
            # Better: just try the update and see if score improves
            a_trial = best_a.copy()
            a_trial[idx_sorted] = a_subset_new
            a_trial = a_trial / a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial
    
    return best_a, best_score


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=20):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _fourier_perturb(a, rng, n_harmonics=10):
    """Perturb a in Fourier space for diverse basin jumping."""
    n = len(a)
    A = np.fft.rfft(a)
    # Perturb the low-frequency components
    for k in range(1, min(n_harmonics, len(A))):
        A[k] *= (1.0 + 0.1 * rng.standard_normal())
        A[k] += 0.05 * rng.standard_normal()
    a_new = np.fft.irfft(A, n)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 1e-10:
        a_new = a_new / a_new.sum()
    else:
        a_new = a.copy()
    return a_new
