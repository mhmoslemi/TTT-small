"""
Optimized step-function sequence finder minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-space Adam with adaptive top-K softmax smoothing, multi-phase cooling,
coordinate descent polishing with golden-section 1D search, and diversity restarts.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):

    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_softmax_weights(c, beta, k):
    """Compute softmax weights over top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a via cross-correlation."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logspace_grad(a, w, n):
    """Gradient in log-space (z-space) for score = 2n*max(conv)/sum(a)^2 with sum=1.
    On simplex: dF/dz_i = 2n * a_i * (grad_a_i - <grad_a, a>)
    """
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, lr_end, top_k_schedule, max_iter):
    """Adam optimizer in log-space with top-K smoothed max and cosine LR annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    total_iters = 0
    for phase_idx, (beta, top_k) in enumerate(zip(beta_schedule, top_k_schedule)):
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            total_iters += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine annealing learning rate
            progress = min(1.0, total_iters / (max_iter * len(beta_schedule)))
            lr = lr_end + 0.5 * (lr_init - lr_end) * (1.0 + np.cos(np.pi * progress))

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 0.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if total_iters % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        # Dampen momentum at phase boundaries
        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_polish(a, deadline, n_passes=3, top_k_w=80):
    """Cyclic coordinate descent with golden-section 1D search."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    # Golden section ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            if a_i_old < 1e-14:
                continue

            # Compute d_j = a[j-i] for valid j (contribution of a[i] to conv at position j)
            d = np.zeros(2 * n - 1)
            valid_start = i
            valid_end = min(i + n, 2 * n - 1)
            d[valid_start:valid_end] = a[0:n - (valid_end - valid_start)]

            # c_base[j] = conv(a,a)[j] - a_i * d[j]  (convolution without a[i]'s contribution to term i)
            # Actually: conv(a,a)[j] = sum_k a[k]*a[j-k]
            # Terms involving a[i]: a[i]*a[j-i] + a[j-i]*a[i] (if j-i != i)
            #                     = a[i]*a[j-i] (if j-i == i, i.e., j=2i)
            # More carefully: the terms where one factor is a[i]:
            #   a[i]*a[j-i] (when 0<=j-i<n) + a[j-i]*a[i] (when 0<=i-(j-i)<n, i.e., 0<=2i-j<n)
            # Let's just compute: new_conv[j] = old_conv[j] - a[i]*d[j] where d[j] accounts for both terms
            # d[j] = a[j-i] (if valid) + a[2i-j] (if valid and 2i-j != j-i, i.e., i != j-i)
            # Simpler: just recompute the change
            # old contribution of a[i] to conv[j]: a[i]*a[j-i] (if 0<=j-i<n) + a[j-i]*a[i] (if 0<=j-i<n and j-i != i)
            # = a[i]*a[j-i] + (a[j-i]*a[i] if j-i != i else 0)
            # = a[i]*(a[j-i] + (a[j-i] if j!=2i else 0))
            # This is getting complicated. Let's use a simpler approach:
            # new_conv[j] = old_conv[j] - a_old*a[j-i] (if valid) - a[j-i]*a_old (if valid and j-i != i)
            # + t*a[j-i] (if valid) + a[j-i]*t (if valid and j-i != i)
            # where t is the new value of a[i]
            # = old_conv[j] - a_old*(a[j-i] + (a[j-i] if j-i!=i else 0)) + t*(a[j-i] + (a[j-i] if j-i!=i else 0))
            # = old_conv[j] + (t - a_old) * (a[j-i] + (a[j-i] if j!=2*i else 0))  [if 0<=j-i<n]

            # Let me just use: slope[j] = d coefficient for changing a[i] to t
            # conv[j] as function of a[i]=t:
            #   = (other terms) + t * a[j-i] (if 0<=j-i<n) + a[j-i] * t (if 0<=j-i<n and j-i != i)
            # The "other terms" don't involve a[i].
            # So slope[j] = a[j-i] (if 0<=j-i<n) + a[j-i] (if 0<=j-i<n and j-i != i)
            #             = a[j-i] * (1 + (1 if j != 2*i else 0))  [if 0<=j-i<n]
            #               + 0 otherwise

            slope = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            slope[valid_j] = a[valid_j - i]
            # Add the symmetric term (when j-i != i, i.e., j != 2i)
            sym_j = 2 * i - np.arange(0, min(n, 2 * i + 1))
            sym_j = sym_j[(sym_j >= 0) & (sym_j < 2 * n - 1) & (sym_j != 2 * i)]
            if len(sym_j) > 0:
                slope[sym_j] += a[i - (sym_j - i)]  # a[2i - j - i] = a[i - j + i]... 

            # Actually let me just do it more carefully with a direct computation
            # For changing a[i] from old to t, the change in conv[j] is:
            # delta[j] = (t - old) * (a[j-i] if 0<=j-i<n else 0) + (t - old) * (a[j-i] if 0<=j-i<n and j-i!=i else 0)
            # Wait, conv[j] = sum_{k: 0<=k<n, 0<=j-k<n} a[k]*a[j-k]
            # Terms with k=i: a[i]*a[j-i] (if 0<=j-i<n)
            # Terms with j-k=i, i.e., k=j-i: a[j-i]*a[i] (if 0<=j-i<n and j-i != i)
            # So total coefficient of a[i] in conv[j] (for j != 2i): a[j-i] + a[j-i] = 2*a[j-i] (if 0<=j-i<n)
            # For j = 2i: a[i]*a[i], coefficient of a[i] is a[i] itself (it's a[i]^2)
            # So: slope[j] = 2*a[j-i] if (0<=j-i<n and j!=2*i), or a[j-i] = a[i] if j=2*i, else 0

            slope = np.zeros(2 * n - 1)
            # j ranges from i to i+n-1 for the first term
            j_vals = np.arange(i, min(i + n, 2 * n - 1))
            slope[j_vals] = 2.0 * a[j_vals - i]
            # Fix j=2i case
            if 2 * i < 2 * n - 1:
                slope[2 * i] = a[i]  # only one factor of a[i]

            # conv_new[j] = conv_old[j] + (t - a_i_old) * slope[j]
            # S_new = S_old + (t - a_i_old) = 1 + (t - a_i_old) (since sum=1)
            # But we're working with sum=1, so we rescale after.
            # Actually in log-space we don't rescale; the softmax handles it.
            # For coordinate descent, let's work with the raw values.
            S_old = float(a.sum())
            S_base = S_old - a_i_old  # sum of all other elements

            c_old = _full_convolve(a, a)
            c_base = c_old - a_i_old * slope

            # F(t) = 2n * max_j (c_base[j] + slope[j]*t) / (S_base + t)^2
            # We want to minimize F(t) over t >= 0

            # Find top-K active constraints
            idx_top = np.argsort(c_old)[-top_k_w:]

            # Golden section search on t in [0, upper_bound]
            t_lo = 0.0
            t_hi = max(4.0 * a_i_old, 0.1)

            def F_of_t(t):
                c_trial = c_base + slope * t
                return 2.0 * n * float(c_trial.max()) / ((S_base + t) ** 2 + 1e-30)

            # Golden section
            c1 = t_lo + gr * (t_hi - t_lo)
            c2 = t_hi - gr * (t_hi - t_lo)
            f1 = F_of_t(c1)
            f2 = F_of_t(c2)
            for _ in range(30):
                if f1 < f2:
                    t_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = t_lo + gr * (t_hi - t_lo)
                    f1 = F_of_t(c1)
                else:
                    t_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = t_hi - gr * (t_hi - t_lo)
                    f2 = F_of_t(c2)

            t_opt = (t_lo + t_hi) / 2.0
            # Also check t=0 and t=a_i_old
            candidates = [0.0, a_i_old, t_opt]
            best_t = a_i_old
            best_f = F_of_t(a_i_old)
            for tc in candidates:
                fc = F_of_t(tc)
                if fc < best_f:
                    best_f = fc
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        # Check score after each pass
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    return a, best_score


def _fista_smoothed(a_init, deadline, beta, top_k, max_iter=5000):
    """FISTA on simplex with top-K smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 16.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4, 5]:
            a += 0.2 / freq * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.005, 0.995, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n) / n
        a = np.abs(np.sin(2.0 * np.pi * (16.0 * t ** 2))) ** 0.3 + 0.1
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(123)
        a = rng.random(n)
        # Smooth it
        for _ in range(3):
            a = np.convolve(a, np.ones(5) / 5, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.25) ** 2) + np.exp(-30 * (t - 0.75) ** 2) + 0.05
        return a / a.sum()
    elif kind == "gabor":
        t = np.arange(n) / n
        a = np.exp(-10 * (t - 0.5) ** 2) * (1 + 0.5 * np.cos(2 * np.pi * 20 * t))
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """Main search function. Returns best sequence found within time budget."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                parent_norm = parent.copy()
    except Exception:
        parent = None
        parent_norm = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        parent_norm = parent.copy()
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Primary optimization at n=4096
    n_primary = 4096
    if parent is not None and n_parent == n_primary:
        a0 = parent_norm.copy()
    elif parent is not None:
        a0 = _interp_profile(parent_norm, n_primary)
    else:
        a0 = parent_norm.copy()

    sc0 = _score(a0)
    print(f"  n={n_primary} initial: score={sc0:.10f}")
    if sc0 < best_score:
        best_score = sc0
        best_seq = a0.copy()

    # Phase 1: FISTA coarse
    remaining = deadline - time.monotonic()
    if remaining > 30:
        c0 = _full_convolve(a0, a0)
        max_c0 = float(c0.max())
        betas_fista = [0.02 * max_c0, 0.008 * max_c0, 0.003 * max_c0]
        top_ks_fista = [200, 150, 100]
        phase1_end = time.monotonic() + min(remaining * 0.25, 200.0)

        a_curr = a0.copy()
        for bi in range(len(betas_fista)):
            if time.monotonic() > phase1_end:
                break
            per_budget = phase1_end - time.monotonic() - 2
            if per_budget < 5:
                break
            max_it = max(2000, int(per_budget * 2000))
            a_curr, _ = _fista_smoothed(a_curr, time.monotonic() + min(60, per_budget),
                                        betas_fista[bi], top_ks_fista[bi], max_it)
            a_curr = a_curr / a_curr.sum()
            sc = _score(a_curr)
            if sc < best_score:
                best_score = sc
                best_seq = a_curr.copy()
            print(f"  FISTA phase {bi}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Adam log-space (main workhorse)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())

        # Multi-phase beta schedule
        n_phases = 10
        betas_adam = [max_cc * 0.01 * (0.5 ** i) for i in range(n_phases)]
        top_ks_adam = [max(50, int(200 * (0.9 ** i))) for i in range(n_phases)]

        phase2_end = time.monotonic() + min(remaining * 0.45, 450.0)
        a_opt, sc_opt = _adam_logspace(a_curr, phase2_end, betas_adam,
                                       lr_init=0.015, lr_end=0.002,
                                       top_k_schedule=top_ks_adam,
                                       max_iter=200000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam main: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

        a_work = a_opt

    # Phase 3: Coordinate descent polishing
    remaining = deadline - time.monotonic()
    if remaining > 20:
        phase3_end = time.monotonic() + min(remaining * 0.20, 150.0)
        a_cd, sc_cd = _coord_descend_polish(a_work, deadline=phase3_end,
                                            n_passes=5, top_k_w=100)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = a_cd

    # Phase 4: Ultra-fine Adam
    remaining = deadline - time.monotonic()
    if remaining > 20:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_ult = [max_cw * 0.0005, max_cw * 0.0002, max_cw * 0.00008]
        top_ks_ult = [50, 40, 30]
        phase4_end = time.monotonic() + min(remaining * 0.15, 100.0)
        a_ult, sc_ult = _adam_logspace(a_work, phase4_end, betas_ult,
                                       lr_init=0.005, lr_end=0.0005,
                                       top_k_schedule=top_ks_ult,
                                       max_iter=100000)
        sc_ult_full = _score(a_ult)
        if sc_ult_full < best_score:
            best_score = sc_ult_full
            best_seq = a_ult.copy()
            print(f"  Ultra-fine: score={sc_ult_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = a_ult

    # Phase 5: Coordinate descent on ultra-fine result
    remaining = deadline - time.monotonic()
    if remaining > 15 and a_work is not None:
        phase5_end = time.monotonic() + min(remaining * 0.15, 80.0)
        a_cd2, sc_cd2 = _coord_descend_polish(a_work, deadline=phase5_end,
                                              n_passes=4, top_k_w=60)
        sc_cd2_full = _score(a_cd2)
        if sc_cd2_full < best_score:
            best_score = sc_cd2_full
            best_seq = a_cd2.copy()
            print(f"  CoordDescend2: score={sc_cd2_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Diversity restarts
    remaining = deadline - time.monotonic()
    if remaining > 30 and best_seq is not None:
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = 4
        per_restart = (remaining - 10) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break

            if ri == 0:
                # Gaussian perturbation
                p_start = a_base + 0.03 * rng.standard_normal(n_best)
            elif ri == 1:
                # Circular shift perturbation
                shift = rng.integers(1, max(2, n_best // 256))
                p_start = np.roll(a_base, shift).copy()
                # Slight blend with original to avoid pure shift
                p_start = 0.7 * p_start + 0.3 * a_base
            elif ri == 2:
                # Frequency modulation perturbation
                freq = rng.uniform(1.0, 8.0)
                pert = 0.04 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            elif ri == 3:
                # Block scaling perturbation
                p_start = a_base.copy()
                n_blocks = 16
                block_size = n_best // n_blocks
                for b in range(n_blocks):
                    scale = rng.uniform(0.8, 1.2)
                    p_start[b * block_size:(b + 1) * block_size] *= scale
            else:
                p_start = a_base.copy()

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [max_cs * 0.003, max_cs * 0.001, max_cs * 0.0003, max_cs * 0.0001]
            top_ks_r = [100, 80, 60, 40]

            p_best, _ = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                       betas_r, lr_init=0.01, lr_end=0.003,
                                       top_k_schedule=top_ks_r,
                                       max_iter=50000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
            print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 7: Try n=2048 as alternative
    remaining = deadline - time.monotonic()
    if remaining > 60:
        n_alt = 2048
        if best_seq is not None:
            a_alt = _interp_profile(best_seq, n_alt)
        else:
            a_alt = _structured_init(n_alt, "cos_sum")

        sc_alt0 = _score(a_alt)
        if sc_alt0 < best_score:
            best_score = sc_alt0
            best_seq = a_alt.copy()

        c_alt = _full_convolve(a_alt, a_alt)
        max_ca = float(c_alt.max())
        betas_alt = [max_ca * 0.005, max_ca * 0.002, max_ca * 0.0005]
        top_ks_alt = [100, 70, 50]

        phase_alt_end = time.monotonic() + min(remaining * 0.3, 120.0)
        a_alt_opt, _ = _adam_logspace(a_alt, phase_alt_end, betas_alt,
                                      lr_init=0.01, lr_end=0.002,
                                      top_k_schedule=top_ks_alt,
                                      max_iter=100000)
        sc_alt_full = _score(a_alt_opt)
        # Also do coord descend
        if sc_alt_full < best_score and time.monotonic() < deadline - 10:
            a_alt_cd, _ = _coord_descend_polish(a_alt_opt, deadline=min(deadline - 5, time.monotonic() + 30),
                                                n_passes=3, top_k_w=60)
            sc_alt_cd = _score(a_alt_cd)
            if sc_alt_cd < sc_alt_full:
                sc_alt_full = sc_alt_cd
                a_alt_opt = a_alt_cd
        
        if sc_alt_full < best_score:
            best_score = sc_alt_full
            best_seq = a_alt_opt.copy()
        print(f"  n=2048 attempt: best={sc_alt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        # Verify with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"Verified score via evaluate_sequence: {ev_score:.10f}")
            if ev_score < best_score:
                best_score = ev_score
        except Exception as e:
            print(f"Verification failed: {e}")
        print(f"FINAL: score={best_score:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
