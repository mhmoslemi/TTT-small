Key improvements over previous version:
1. Coordinate descent polishing (selective coordinates near peaks)
2. More aggressive FISTA with finer beta schedule

import numpy as np
import time
import math

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

    best_a = a_norm.copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w
            
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:

                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            
            if t % lr_decay_every == 0:

    a_final = _softmax_compute(z)

    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

            #
            # Oh wait, I see the issue. When we set a[j] = 0, the sum changes, and
            # the score is 2n*max(conv)/(sum)^2. If we scale back to maintain sum=1,
            # the other elements get larger, which might increase their contribution to
            # the convolution.
            #
            # So the coordinate descent needs to account for the normalization.
            # This makes it much more complex.
            #
            # Alternative: do "paired" updates - decrease a[j] and increase some a[j']
            # to maintain the sum. This is more like a 2D optimization.
            #
            # For now, let's use a simpler approach: just try scaling a[j] by factors
            # and check if the score improves.
            
            # Try several scaling factors
            candidates = [0.0, 0.1*old_x, 0.3*old_x, 0.5*old_x, 0.7*old_x,
                        0.9*old_x, 1.1*old_x, 1.3*old_x, 1.5*old_x]
            best_x_try = old_x
            best_score_try = float(c.max())
            
            # Also try to find the optimal by minimizing the max of affected values
            # without the normalization constraint (as a heuristic)
            # The idea: reduce x until the max affected value equals the second-max
            # (like water-filling)
            
            # For now, just try the candidates
            for x_try in candidates:
                a[j] = x_try
                # Quick check: compute only the affected convolution values
                # (for speed, just check a few of the largest)
                local_max = c.max()  # This is O(n) which is fast enough
                # Actually we need to recompute the full convolution for accuracy
                # Let's just do it for the top candidates
                if x_try != old_x and x_try > 0:
                    c_check = _full_convolve(a, a)
                    if c_check.max() < best_score_try:
                        best_score_try = float(c_check.max())
                        best_x_try = x_try
            
            a[j] = best_x_try
            a = a / a.sum()
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def _peak_directed_search(a_init, deadline, max_iters=500):
    """
    Directly target the peak: at each step, identify the peak location and
    find a direction that reduces it most effectively.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    step_size = 0.05
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        peak_idx = np.argmax(c)
        
        # The peak is at position peak_idx.
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
        # To reduce this, we want to reduce a[i]*a[peak_idx-i] for large products.
        
        # Strategy: find the pair (i, peak_idx-i) with the largest product and
        # transfer weight from one to a random other position.
        
        products = np.zeros(n)
        for i in range(n):
            t = peak_idx - i
            if 0 <= t < n:
                products[i] = a[i] * a[t]
        
        # Find the top contributor
        top_i = np.argmax(products)
        top_t = peak_idx - top_i
        
        if top_t < 0 or top_t >= n:
            top_t = 0
        
        # Transfer some weight from top_i to a random position
        # that doesn't contribute much to the peak
        candidates = np.where(products < 0.1 * products[top_i])[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        
        target_j = candidates[np.random.randint(len(candidates))]
        if target_j == top_i:
            continue
        
        # Transfer weight
        transfer = step_size * a[top_i]
        if transfer > a[top_i]:
            transfer = 0.5 * a[top_i]
        
        a_new = a.copy()
        a_new[top_i] -= transfer
        a_new[target_j] += transfer
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reject, reduce step size
            step_size *= 0.95
            if step_size < 0.001:
                step_size = 0.01
    
    return best_a, best_score

def _freq_domain_shape(a_init, deadline, n_iters=2000):

    """

    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()

    N_fft = _next_pow2(2 * n - 1)
    
    # Get the DFT
    A = np.fft.rfft(a, N_fft)

    mag = np.abs(A)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        ps = mag ** 2
        # Inverse FFT of power spectrum gives autocorrelation (up to scaling)
        ac = np.fft.irfft(ps, N_fft)
        ac = np.concatenate([ac[n-1:], ac[:n-1]])  # Circular autocorrelation
        # Actually, let's just use the direct convolution for accuracy

        cm = float(c.max())
