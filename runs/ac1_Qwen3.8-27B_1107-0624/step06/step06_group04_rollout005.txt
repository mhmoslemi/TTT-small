    n = len(c)
    k = min(k, n)
    top_idx = np.argpartition(c, -k)[-k:]

    # Softmax over top-k only
    shifted = top_vals - top_vals.max()
    w_top = np.exp(shifted / beta)

    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

def _grad_logspace(a, c, w, n):
    """Gradient for log-space optimization."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                                       max_iter_total=300000, lr_decay_every=5000,
                                       lr_decay_factor=0.96, grad_clip=0.3)
                sc8 = _score(a8)
            if sc8 < best_score:
                best_score = sc8
                best_seq = a8.copy()
                print(f"  8192 result: {sc8:.10f} t={time.monotonic()-t_start:.0f}s")

    a_work = best_seq.copy() / best_seq.sum()
    print(f"End Phase 3: {best_score:.10f}")

    # Phase 4: Ultra-fine Adam on best
    print("Phase 4: Ultra-fine Adam")
    p4_end = min(deadline, t_start + 1500.0)
    c_f = _full_convolve(a_work, a_work)
    mcf = float(c_f.max())
    beta_uf = [0.00003 * mcf, 0.00001 * mcf, 0.000005 * mcf, 0.000002 * mcf, 0.000001 * mcf]
    k_uf = [15, 12, 10, 8, 6]
    a_uf, _ = _adam_logspace(a_work, p4_end, beta_uf, k_uf,
                             lr_init=0.004, max_iter_total=400000,
                             lr_decay_every=8000, lr_decay_factor=0.95, grad_clip=0.3)
    sc_uf = _score(a_uf)
    if sc_uf < best_score:
        best_score = sc_uf
        best_seq = a_uf.copy()
        print(f"  Ultra-fine: {sc_uf:.10f} t={time.monotonic()-t_start:.0f}s")

    a_work = best_seq.copy() / best_seq.sum()
    print(f"End Phase 4: {best_score:.10f}")

    # Phase 5: Subgradient polish
    print("Phase 5: Subgradient polish")
    p5_end = min(deadline, t_start + 1650.0)
    a_sg, _ = _subgradient_polish(a_work, 8000, p5_end)
    sc_sg = _score(a_sg)
    if sc_sg < best_score:
        best_score = sc_sg
        best_seq = a_sg.copy()
        print(f"  Subgrad: {sc_sg:.10f} t={time.monotonic()-t_start:.0f}s")
    a_work = best_seq.copy() / best_seq.sum()

    # Phase 6: Final Adam micro-polish + spectral
    print("Phase 6: Final polish")
    c_fin = _full_convolve(a_work, a_work)
    mfin = float(c_fin.max())
    if time.monotonic() < deadline - 20:
        beta_fin = [0.000005 * mfin, 0.000002 * mfin, 0.000001 * mfin]
        k_fin = [8, 6, 4]
        per_t = deadline - time.monotonic() - 15
        a_fin, _ = _adam_logspace(a_work, time.monotonic() + per_t * 0.6, beta_fin, k_fin,
                                  lr_init=0.002, max_iter_total=200000,
                                  lr_decay_every=10000, lr_decay_factor=0.94, grad_clip=0.2)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  Final Adam: {sc_fin:.10f} t={time.monotonic()-t_start:.0f}s")

    if time.monotonic() < deadline - 10:
        a_sf, _ = _spectral_flatten(best_seq, 500, deadline - 5)
        sc_sf = _score(a_sf)
        if sc_sf < best_score:
            best_score = sc_sf
            best_seq = a_sf.copy()
            print(f"  Spectral: {sc_sf:.10f} t={time.monotonic()-t_start:.0f}s")

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.0f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
