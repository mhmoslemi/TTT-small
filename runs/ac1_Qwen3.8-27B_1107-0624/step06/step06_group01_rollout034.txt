    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_smooth_max(c, beta):
    """Compute softmax weights for the convolution entries with temperature beta."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, c_max


def _compute_grad_conv(a, c, w, n):
    """
    Compute the gradient of the smoothed max objective.
    The smoothed objective is sum_k w[k] * c[k] where c = conv(a,a).
    Grad = sum_k w[k] * d(c[k])/da = 2 * conv(w, a_reversed)[n-1:2n-1]
    """
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h


def _adam_optimize(a_init, deadline, betas, lr_init=0.01, momentum_b1=0.9, momentum_b2=0.999, eps=1e-8):
    """
    Adam optimization in log-space with softmax parameterization.
    a_init should have sum approximately 1.
    betas is a list of temperatures for the softmax smoothing.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        phase_iter_limit = 50000
        it_count = 0
        
        while it_count < phase_iter_limit:
            if time.monotonic() > deadline:
                break
            it_count += 1
            t += 1
            
            # Compute a from z via softmax
            z_shifted = z - z.max()
            e = np.exp(z_shifted)
            s = e.sum()
            a = e / s
            
            # Compute convolution and softmax weights
            c = _full_convolve(a, a)
            w, c_max = _softmax_smooth_max(c, beta)
            
            # Compute gradient in z-space
            # d/dz_i [sum_k w[k]*c[k]] = sum_k w[k] * dc[k]/da_i * da_i/dz_i
            # da_i/dz_i = a_i * (delta_{ij} - a_j) for the softmax
            # For simplicity, we compute the gradient in a-space and chain rule
            grad_a = _compute_grad_conv(a, c, w, n)
            # Chain rule for softmax: grad_z[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])
            mean_grad = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - mean_grad)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            # Adam update
            m_buf = momentum_b1 * m_buf + (1 - momentum_b1) * grad_z
            v_buf = momentum_b2 * v_buf + (1 - momentum_b2) * grad_z * grad_z
            mh = m_buf / (1 - momentum_b1 ** t)
            vh = v_buf / (1 - momentum_b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            
            # Periodic evaluation
            if t % 100 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            
            # Decay learning rate within phase
            if it_count % 5000 == 0:
                current_lr *= 0.9
        
        # Contribution of the current phase
        a_final = _softmax_from_z(z)
        sc = _score_norm(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final
    
    return best_a, best_score


def _softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-10.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "block_random":
        if rng is None:
            rng = np.random.default_rng(42)
        n_blocks = 4
        block_size = n // n_blocks
        a = np.zeros(n)
        for i in range(n_blocks):
            val = rng.random() + 0.1
            a[i*block_size:(i+1)*block_size] = val
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_from_z(z)


def _block_permute(a, rng, n_blocks=4):
    """Randomly permute blocks of the sequence."""
    n = len(a)
    block_size = n // n_blocks
    blocks = [a[i*block_size:(i+1)*block_size] for i in range(n_blocks)]
    remaining = n - n_blocks * block_size
    if remaining > 0:
        blocks.append(a[n_blocks*block_size:])
    rng.shuffle(blocks)
    result = np.concatenate(blocks)
    return result / result.sum()


def _peak_shave_cd(a, deadline, n_iterations=5000, rng=None):
    """
    Coordinate descent that specifically targets reducing the peak of the convolution.
    At each step, identify the argmax of conv, and reduce the contribution of the
    pairs that form the peak.
    """
    if rng is None:
        rng = np.random.default_rng(123)
    
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    a = a.copy()
    
    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        # Find the top-k peak positions
        k_peak = int(np.argmax(c))
        
        # The peak is contributed by pairs (j, k_peak - j)
        # Try to reduce a[j] for j that contributes most
        j_options = []
        for j in range(max(0, k_peak - n + 1), min(n, k_peak + 1)):
            other = k_peak - j
            if 0 <= other < n:
                contrib = a[j] * a[other]
                j_options.append((j, other, contrib))
        
        if not j_options:
            continue
        
        j_options.sort(key=lambda x: -x[2])
        
        # Try reducing the top contributor
        improved = False
        for (j, other, contrib) in j_options[:5]:
            for factor in [0.95, 0.9, 0.85, 0.8, 1.05, 1.1]:
                a_try = a.copy()
                old_val = a_try[j]
                a_try[j] *= factor
                # Compensate by increasing a random other position
                r = rng.integers(0, n)
                if r != j and r != other:
                    a_try[r] += (old_val - a_try[j])
                    a_try[r] = max(a_try[r], 0)
                
                if a_try.sum() > 1e-15:
                    a_try = a_try / a_try.sum()
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
            if improved:
                break
        
        if not improved and it % 1000 == 999:
            # If stuck, perturb
            a = _perturb(a, 0.01, rng)
    
    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter=10000):
    """
    FISTA (accelerated proximal gradient) on the simplex for the smoothed max objective.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = np.maximum(a_curr, 0)
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    if max_c < 1e-15:
        max_c = 1e-15
    
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w, c_max = _softmax_smooth_max(c_y, beta)
        
        # Gradient of smoothed max
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h
        
        a_next = y - step * grad
        # Project onto simplex
        a_next = _proj_simplex(a_next)
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            # Update L adaptively
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            if max_c_new > 1e-15:
                L = 4.0 * n * max_c_new / beta
                step = 1.0 / L
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-stage optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses FISTA with softmax-smoothed max, Adam in log-space, peak-shaving coordinate descent,
and diverse multi-restart strategies. Built for n up to 16384.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_smooth_max(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _grad_a_space(a, c, w, n):
    """Gradient of sum_k w[k]*C[k] with respect to a, in a-space. Returns 2*h."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)

def _grad_z_space(a, h, n):

    W = float(np.dot(a, h))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fista_smoothed(a_init, beta, deadline, max_iter=20000):
    n = len(a_init)
    a_curr = np.maximum(a_init, 0.0)
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    if max_c < 1e-15:
        max_c = 1e-15

    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _softmax_smooth_max(c_y, beta)
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = y - step * grad
        a_next = _proj_simplex(a_next)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            if max_c_new > 1e-15:
                L = 4.0 * n * max_c_new / beta
                step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_optimize(a_init, deadline, betas, lr_init=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in betas:
        if time.monotonic() > deadline:
            break

        lr = lr_init
        phase_limit = 80000
        it_count = 0

        while it_count < phase_limit:
            if time.monotonic() > deadline:
                break
            it_count += 1
            t += 1

            z_shifted = z - z.max()
            e = np.exp(z_shifted)
            s = e.sum()
            a = e / s

            c = _full_convolve(a, a)
            w = _softmax_smooth_max(c, beta)
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]

            grad_z = _grad_z_space(a, h, n)
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.5:
                update = update * (1.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if it_count % 10000 == 0:
                lr *= 0.85

        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        a_final = e / e.sum()
        sc = _score_norm(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final

    return best_a, best_score


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-25.0 * (t - 0.3) ** 2) + np.exp(-25.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "trimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = (np.exp(-40.0 * (t - 0.2) ** 2) + np.exp(-40.0 * (t - 0.5) ** 2)
             + np.exp(-40.0 * (t - 0.8) ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(42)
        # Generate random frequencies and amplitudes
        n_freqs = 8
        freqs = rng.integers(1, n // 4, size=n_freqs).astype(float)
        amps = rng.random(n_freqs)
        phases = rng.random(n_freqs) * 2 * np.pi
        t = np.arange(n, dtype=float)
        a = np.ones(n)
        for f, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2.0 * np.pi * f * t / n + ph)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def _block_permute(a, rng, n_blocks=4):
    n = len(a)
    block_size = n // n_blocks
    blocks = [a[i * block_size:(i + 1) * block_size].copy() for i in range(n_blocks)]
    remaining = n - n_blocks * block_size
    if remaining > 0:
        blocks.append(a[n_blocks * block_size:].copy())
    rng.shuffle(blocks)
    result = np.concatenate(blocks)
    return result / result.sum()


def _mirror_permute(a, rng):
    """Reverse blocks of the sequence."""
    n = len(a)
    half = n // 2
    b = half
    result = np.concatenate([a[b:], a[:b]])
    return result / result.sum()


def _peak_shave_cd(a, deadline, n_iter=20000, rng=None):
    if rng is None:
        rng = np.random.default_rng(777)

    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    a_work = a.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_work, a_work)
        k_peak = int(np.argmax(c))

        # Find contributing pairs
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)
        if j_min > j_max:
            # Try a random coordinate update instead
            i = rng.integers(0, n)
            for factor in [0.9, 0.85, 0.95, 1.05, 1.1]:
                a_try = a_work.copy()
                old = a_try[i]
                a_try[i] *= factor
                r = rng.integers(0, n)
                if r != i:
                    a_try[r] += (old - a_try[i])
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() > 1e-15:
                    a_try = a_try / a_try.sum()
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_try.copy()
                        a_work = a_try.copy()
                        break
            continue

        # Get contributions sorted
        contribs = []
        for j in range(j_min, j_max + 1):
            other = k_peak - j
            if 0 <= other < n:
                contribs.append((j, other, a_work[j] * a_work[other]))

        contribs.sort(key=lambda x: -x[2])

        improved = False
        for (j, other, _) in contribs[:8]:
            for factor in [0.95, 0.9, 0.85, 0.8, 1.05, 1.1]:
                a_try = a_work.copy()
                old_val = a_try[j]
                a_try[j] *= factor
                # Redistribute to a random low-value position
                r = rng.integers(0, n)
                if r == j or r == other:
                    r = (r + 1) % n
                if r == j or r == other:
                    continue
                a_try[r] += (old_val - a_try[j])
                a_try = np.maximum(a_try, 0.0)
                if a_try.sum() > 1e-15:
                    a_try = a_try / a_try.sum()
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_try.copy()
                        a_work = a_try.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved and it % 5000 == 4999:
            a_work = _perturb(a_work, 0.005, rng)

    return best_a, best_score


def _pair_swap_cd(a, deadline, n_iter=10000, rng=None):
    """Try swapping portions of two coordinates to reduce the peak."""
    if rng is None:
        rng = np.random.default_rng(999)

    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    a_work = a.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_work, a_work)
        k_peak = int(np.argmax(c))

        # Pick two random coordinates, try to transfer mass
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        for frac in [0.05, 0.1, 0.2, 0.3]:
            a_try = a_work.copy()
            transfer = a_try[i] * frac
            if a_try[i] - transfer < 0:
                continue
            a_try[i] -= transfer
            a_try[j] += transfer
            if a_try.sum() > 1e-15:
                a_try = a_try / a_try.sum()
                sc = _score_norm(a_try)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_try.copy()
                    a_work = a_try.copy()
                    break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        a0 = np.ones(n_parent) / n_parent
        best_seq = a0.copy()
        best_score = _score(a0)
        print(f"No parent. Uniform n={n_parent}, score={best_score:.6f}")

    # ============================================================
    # PHASE 1: FISTA optimization on multiple initializations
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 1: FISTA (deadline in {deadline - time.monotonic():.0f}s)")
    print(f"{'='*60}")

    n_main = len(best_seq)

    # Define initializations to try for the main length
    init_kinds = [None, "uniform", "sinusoidal", "tapered", "bimodal", "trimodal",
                  "random_smooth", "gaussian"]

    for idx, kind in enumerate(init_kinds):
        if time.monotonic() > deadline - 180:
            break

        if kind is None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_main, kind, rng)

        a_init = np.maximum(a_init, 0.0)
        a_init = a_init / a_init.sum()
        sc_init = _score(a_init)

        # FISTA with coarse-to-fine beta
        c_init = _full_convolve(a_init, a_init)
        max_ci = float(c_init.max())

        betas = []
        beta = 0.04 * max_ci
        while beta > 0.0005 * max_ci:
            betas.append(beta)
            beta *= 0.5
        if len(betas) > 10:
            betas = betas[:10]

        a_work = a_init.copy()
        for bi, b in enumerate(betas):
            if time.monotonic() > deadline - 180:
                break
            remaining = deadline - 180 - time.monotonic()
            phase_time = min(20.0, remaining / (len(betas) - bi))
            phase_dl = time.monotonic() + max(5.0, phase_time)
            a_work, sc = _fista_smoothed(a_work, b, phase_dl, max_iter=15000)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
                print(f"  FISTA init={kind or 'parent'} beta_idx={bi}: {sc_full:.10f}")

        print(f"  After FISTA init={kind or 'parent'}: best={best_score:.10f} "
              f"(t={time.monotonic() - t_start:.0f}s)")

    # ============================================================
    # PHASE 2: Adam fine optimization
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 2: Adam fine (deadline in {deadline - time.monotonic():.0f}s)")
    print(f"{'='*60}")

    a_ref = best_seq.copy()
    a_ref = a_ref / a_ref.sum()
    c_ref = _full_convolve(a_ref, a_ref)
    max_cr = float(c_ref.max())

    # Fine betas for Adam
    betas_fine = []
    beta = 0.003 * max_cr
    while beta > 0.000005 * max_cr:
        betas_fine.append(beta)
        beta *= 0.5
    if len(betas_fine) > 15:
        betas_fine = betas_fine[:15]

    adam_deadline = min(deadline - 120, time.monotonic() + 240.0)
    a_opt, sc_opt = _adam_optimize(a_ref, adam_deadline, betas_fine, lr_init=0.015)
    a_opt = a_opt / a_opt.sum()
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam fine: {sc_opt_full:.10f}")
    print(f"  After Adam: best={best_score:.10f} (t={time.monotonic() - t_start:.0f}s)")

    # ============================================================
    # PHASE 3: Multi-restart with diverse perturbations
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 3: Multi-restart (deadline in {deadline - time.monotonic():.0f}s)")
    print(f"{'='*60}")

    a_ref = best_seq.copy()
    a_ref = a_ref / a_ref.sum()
    n_ref = len(a_ref)
    c_ref = _full_convolve(a_ref, a_ref)
    max_cr = float(c_ref.max())

    restart_betas = [0.001 * max_cr, 0.0005 * max_cr, 0.0002 * max_cr,
                     0.0001 * max_cr, 0.00005 * max_cr]

    n_restarts = 8
    per_restart_time = (deadline - time.monotonic() - 60.0) / n_restarts

    for ri in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break

        if ri == 0:
            p_start = _perturb(a_ref, 0.008, rng)
        elif ri == 1:
            p_start = _perturb(a_ref, 0.02, rng)
        elif ri == 2:
            p_start = _perturb(a_ref, 0.04, rng)
        elif ri == 3:
            p_start = _block_permute(a_ref, rng, n_blocks=4)
        elif ri == 4:
            p_start = _block_permute(a_ref, rng, n_blocks=8)
        elif ri == 5:
            p_start = _mirror_permute(a_ref, rng)
        elif ri == 6:
            alt = _make_profile(n_ref, "sinusoidal", rng)
            p_start = 0.85 * a_ref + 0.15 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            alt = _make_profile(n_ref, "tapered", rng)
            p_start = 0.7 * a_ref + 0.3 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

        rd = time.monotonic() + max(8.0, per_restart_time)
        p_best, sc_best = _adam_optimize(p_start, rd, restart_betas, lr_init=0.008)
        p_best = p_best / p_best.sum()
        sc_full = _score(p_best)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_best.copy()
            a_ref = p_best.copy()
            print(f"  Restart {ri}: {sc_full:.10f} (t={time.monotonic() - t_start:.0f}s)")
        else:
            print(f"  Restart {ri}: no improvement ({sc_full:.10f})")

    # ============================================================
    # PHASE 4: Peak-shaving coordinate descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 4: Peak shaving (deadline in {deadline - time.monotonic():.0f}s)")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 30:
        cd_deadline = min(deadline - 20, time.monotonic() + 90.0)
        a_cd, sc_cd = _peak_shave_cd(best_seq, cd_deadline, n_iter=30000, rng=rng)
        a_cd = a_cd / a_cd.sum()
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Peak shaving: {sc_cd_full:.10f}")
        else:
            print(f"  Peak shaving: no improvement ({sc_cd_full:.10f})")

    # ============================================================
    # PHASE 5: Pair swap coordinate descent
    # ============================================================
    if time.monotonic() < deadline - 20:
        print(f"\nPHASE 5: Pair swap CD")
        ps_deadline = min(deadline - 10, time.monotonic() + 40.0)
        a_ps, sc_ps = _pair_swap_cd(best_seq, ps_deadline, n_iter=15000, rng=rng)
        a_ps = a_ps / a_ps.sum()
        sc_ps_full = _score(a_ps)
        if sc_ps_full < best_score:
            best_score = sc_ps_full
            best_seq = a_ps.copy()
            print(f"  Pair swap: {sc_ps_full:.10f}")

    # ============================================================
    # PHASE 6: Final ultra-fine Adam polish
    # ============================================================
    if best_seq is not None and time.monotonic() < deadline - 8:
        print(f"\nPHASE 6: Final polish")
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        c_fin = _full_convolve(a_fin, a_fin)
        max_cf = float(c_fin.max())

        betas_ult = [0.00003 * max_cf, 0.00001 * max_cf, 0.000003 * max_cf,
                     0.000001 * max_cf]

        fin_dl = time.monotonic() + min(40.0, deadline - time.monotonic() - 5)
        a_final, sc_final = _adam_optimize(a_fin, fin_dl, betas_ult, lr_init=0.003)
        a_final = a_final / a_final.sum()
        sc_final_full = _score(a_final)
        if sc_final_full < best_score:
            best_score = sc_final_full
            best_seq = a_final.copy()
            print(f"  Final Adam: {sc_final_full:.10f}")

    # ============================================================
    # PHASE 7: One more round of peak shaving on the final result
    # ============================================================
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_ps2, sc_ps2 = _peak_shave_cd(best_seq, time.monotonic() + min(20.0,
            deadline - time.monotonic() - 5), n_iter=10000, rng=rng)
        a_ps2 = a_ps2 / a_ps2.sum()
        sc_ps2_full = _score(a_ps2)
        if sc_ps2_full < best_score:
            best_score = sc_ps2_full
            best_seq = a_ps2.copy()
            print(f"  Final peak shave: {sc_ps2_full:.10f}")

    # ============================================================
    # Final validation and output
    # ============================================================
    print(f"\n{'='*60}")
    print(f"FINAL RESULT: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    print(f"{'='*60}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        # Normalize to sum=1 for cleanliness (score is scale-invariant)

        sc_check = _score(final)
        print(f"Verification score: {sc_check:.10f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluator error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(1024) / 1024
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
