Hybrid optimizer combining Gerchberg-Saxton spectral shaping,

and local search to minimize 2*n*max(convolve(a,a))/sum(a)^2.

def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_grad(a, c=None):
    """Return (score, subgradient of score w.r.t. a, normalized so sum(a)=1)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, np.zeros_like(a)
    if c is None:
        c = _fft_conv(a, a)
    c_max = c.max()
    # Active set
    tol = 1e-6 * c_max
    active = c >= (c_max - tol)
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    # Gradient of c_w = sum_k w_k c_k w.r.t. a
    # d(c_k)/d(a_i) = 2 * a_{k-i} (with valid index)

    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n-1:2*n-1]
    # Score = 2*n*c_max / s^2 (using weighted max as surrogate)
    # d(score)/d(a_i) = 2n * (d(c_w))/d(a_i) / s^2 - 2 * 2n * c_w / s^2
    # = (2n/s^2) * (h_i - c_w)
    # But we need to account for the normalization
    c_w = float(np.dot(w, c))
    grad = (2.0 * n / (s * s)) * (h - c_w)
    sc = 2.0 * n * c_max / (s * s)
    return sc, grad

"""
Hybrid optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines Gerchberg-Saxton spectral shaping, aggressive L-BFGS beta-continuation,
adaptive subgradient with backtracking line search, and multi-scale local search.
Key innovations: (1) spectral preprocessing via alternating projections,
(2) much finer beta schedule reaching 1e-10, (3) true-score backtracking,
(4) adaptive active set with multiple tolerance levels, (5) block-based local search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_and_conv(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, np.zeros(2 * n - 1)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s)), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_conv(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_active(a, c, n, tol_frac):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _gs_project(a_init, n_iters=200, relax=1.0):
    """Gerchberg-Saxton: alternate spatial and Fourier domain projections."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a_init.copy()
    a = a / s
    N = _next_pow2(2 * n - 1)
    for _ in range(n_iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        # Target flat magnitude: |A| = 1/sqrt(N) (so that |A|^2 * N = 1, giving c ~ 1/N)
        # Actually A[0] = sum(a) = 1, so target |A_m| = 1 for all m
        # But that's too aggressive. Use: target |A_m| = 1/sqrt(N) * N = sqrt(N)? No.
        # Let's use: we want c_k ~ constant. c_k = IDFT(|A|^2)_k.
        # If |A_m|^2 = 1 for all m, then c_0 = 1/N * sum |A_m|^2 = 1, c_k = 0 for k>0.
        # If |A_m|^2 = N for all m, then c_0 = N, c_k = 0.
        # For flat c: need |A|^2 concentrated at m=0.
        # Alternative: just flatten |A| towards a constant (reduces peak of c)
        # Use target: |A_m| = (sum |A|) / N (mean magnitude)
        target_mag = mag.mean()
        new_mag = mag * relax + target_mag * (1.0 - relax)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a = a_new / s2
    return a


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_aggressive(a_start, deadline, min_time_remaining=100.0):
    """Aggressive beta continuation with very small final betas."""
    n = len(a_start)
    c_cur = _fft_conv(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    # Much finer schedule going down to 1e-10
    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.002,
                              0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                              0.00002, 0.00001, 0.000005, 0.000002,
                              0.000001, 0.0000005, 0.0000002, 0.0000001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        est_iter_time = 0.002 * (n / 2048.0) ** 1.3
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 20))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.2e} -> score={sc:.10f} (iter={niter})")

    return current_a


def _subgrad_backtrack(a_init, deadline, max_steps=50000, tol_fracs=None):
    """Subgradient method with backtracking line search on true score."""
    if tol_fracs is None:
        tol_fracs = [1e-3, 1e-4, 1e-5, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break

        # Adaptive tolerance: start wide, narrow down
        tol_idx = min(i // 3000, len(tol_fracs) - 1)
        tol = tol_fracs[tol_idx] * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _fft_conv(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        # Center the gradient (since we're on the simplex)
        grad = grad - np.dot(grad, a)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break

        # Backtracking line search
        step = lr
        improved = False
        for bt in range(15):
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                step *= 0.4
                continue
            a_try = a_try / s
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
            if sc_try < _score(a) - 1e-16:
                a = a_try
                improved = True
                break
            step *= 0.5

        if not improved:
            stall += 1
            lr *= 0.85
            # Try a smaller direct step
            a_try = np.maximum(a - 0.01 * lr * grad / (gnorm + 1e-30), 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                if sc_try < _score(a):
                    a = a_try
                    stall = 0
        else:
            stall = 0
            lr = min(lr * 1.01, 0.1)

        lr *= (1.0 - 5e-6)
        if stall > 300:
            break

    return best_a, best_sc


def _local_block_search(a, deadline, max_rounds=2000):
    """Block-based local search: modify blocks of consecutive coordinates."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Choose a block
        block_sizes = [2, 3, 4, 5, 8, 12]
        bs = block_sizes[int(rng.integers(0, len(block_sizes)))]
        start = int(rng.integers(0, max(1, n - bs)))
        end = start + bs

        # Try redistribution within the block
        block_sum = best_a[start:end].sum()
        if block_sum < 1e-15:
            continue

        for trial in range(4):
            # Random redistribution
            new_block = rng.dirichlet(np.ones(bs) * 2.0) * block_sum
            a_try = best_a.copy()
            a_try[start:end] = new_block
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                break

            # Tapered redistribution
            t = np.linspace(0, 1, bs)
            new_block = np.sin(np.pi * t)**1.5 * block_sum
            new_block = np.maximum(new_block, 0)
            ns = new_block.sum()
            if ns > 0:
                new_block = new_block / ns * block_sum
            a_try = best_a.copy()
            a_try[start:end] = new_block
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                break

            # Partial zero-out
            mask = rng.random(bs) > 0.3
            if mask.sum() > 0:
                new_block = best_a[start:end].copy()
                new_block[~mask] = 0.0
                ns = new_block.sum()
                if ns > 1e-15:
                    new_block = new_block / ns * block_sum
                    a_try = best_a.copy()
                    a_try[start:end] = new_block
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-13:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        break

    return best_a, best_sc


def _coordinate_search(a, deadline, max_rounds=5000):
    """Pairwise coordinate search with adaptive step."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(999)
    step = 0.01

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n

        s_pair = best_a[i] + best_a[j]
        if s_pair < 1e-15:
            continue

        # Current allocation
        frac = best_a[i] / s_pair

        # Try several new allocations
        for delta in [0.01, 0.02, 0.05, 0.1, 0.2, -0.01, -0.02, -0.05, -0.1, -0.2]:
            new_frac = np.clip(frac + delta, 0.0, 1.0)
            a_try = best_a.copy()
            a_try[i] = new_frac * s_pair
            a_try[j] = (1.0 - new_frac) * s_pair
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _make_varied_start(n, kind, rng):
    """Create diverse starting sequences."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t * rng.uniform(0.5, 3.0))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind as "gaussian":
        t = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.4)
        a = np.exp(-((t - center) / width) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        c1, c2 = 0.25, 0.75
        w1, w2 = 0.1, 0.15
        a = np.exp(-((t - c1) / w1) ** 2) + np.exp(-((t - c2) / w2) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "m_seq Inspired":
        # Inspired by m-sequences: create a pseudo-random binary-like pattern
        a = (rng.random(n) > 0.5).astype(float)
        # Smooth it
        kernel = np.ones(4) / 4.0
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chebyshev":
        t = np.linspace(0, 1, n)
        a = 1.0 - (2 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

                try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final intensive polish ===
    print(f"\n=== Phase 9: Final Polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            n_cur = len(best_seq)
            c_b = _fft_conv(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-8
            a_f, _, _ = _lbfgs_run(best_seq, beta, n_cur, 5000)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_backtrack(best_seq, deadline - 5,
                                            max_steps=20000,
                                            tol_fracs=[1e-6, 5e-7, 2e-7, 1e-7, 5e-8])
                try_update(a_s, f"final_sg_{rnd}")

            # Quick block search
            if time.monotonic() < deadline - 5:
                a_b, _ = _local_block_search(best_seq, deadline - 5, max_rounds=2000)
                try_update(a_b, f"final_block_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
