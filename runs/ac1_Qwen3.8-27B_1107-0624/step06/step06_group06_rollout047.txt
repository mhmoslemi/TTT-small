```python
    a_p = np.zeros(N)
    a_p[:n] = a
    A = np.fft.rfft(a_p)  # real FFT, size N//2+1
    
    # Target magnitude: flat
    # DC: A[0] = S (keep)
    # Others: target = S / sqrt(N-1) approximately
    # But we should use the actual energy constraint
    # sum |A[k]|^2 = N * sum(a^2) (Parseval, with the N factor for numpy's FFT convention)
    # Actually numpy's FFT: if A = fft(x), then sum(|A|^2) = N * sum(|x|^2)
    current_energy = np.sum(np.abs(A)**2)  # = N * sum(a^2)
    target_energy_nonDC = current_energy - S**2  # energy in non-DC components
    n_nonDC = N // 2  # number of independent non-DC components (for rfft)
    # But we need to be careful: for rfft, A has N//2+1 elements
    # A[0] is DC, A[N//2] is the Nyquist (if N is even)
    # The "power" in each non-DC bin is counted once for rfft (except DC and Nyquist)
    
    # Simpler: just scale all non-DC components to have the same magnitude
    mag = np.abs(A)
    nonDC_mask = np.arange(len(mag)) > 0
    current_nonDC_power = np.sum(mag[nonDC_mask]**2)

        scale = np.sqrt(target_energy_nonDC / current_nonDC_power)
    else:
        scale = 1.0
    
    new_mag = mag.copy()
    new_mag[nonDC_mask] = mag[nonDC_mask] * scale  # equalize all non-DC
    
    # Actually, to truly flatten, set all non-DC to the same value:
    target_nonDC_mag = np.sqrt(current_nonDC_power / np.sum(nonDC_mask))
    new_mag[nonDC_mask] = target_nonDC_mag
    
    # Blend with current (for gradual convergence)
    new_mag = (1-alpha) * mag + alpha * new_mag
    new_mag[0] = S  # keep DC

    A_new = new_mag * np.exp(1j * phase)

    a_new = np.maximum(a_p_new[:n].real, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a
```

```python
"""
Multi-phase optimizer combining Gerchberg-Saxton spectral flattening,
variance-minimizing gradient descent, L-BFGS with adaptive log-sum-exp smoothing,
and active-set refinement. Key novel ideas: (1) GS iterations to flatten the
power spectrum of the zero-padded sequence, directly targeting convolution flatness;
(2) variance-based gradient descent as a smooth alternative to max-based objectives;
(3) basin-hopping with GS perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gs_step(a, N, alpha):
    """Single Gerchberg-Saxton step: flatten spectrum, project to non-negative."""
    n = len(a)
    S = float(a.sum())
    a_p = np.zeros(N)
    a_p[:n] = a
    A = np.fft.rfft(a_p)
    mag = np.abs(A)
    phase = np.angle(A)
    # Target: equalize all non-DC magnitudes
    if len(mag) > 1:
        nonDC_power = np.sum(mag[1:] ** 2)
        if nonDC_power > 1e-30:
            target_mag = np.sqrt(nonDC_power / (len(mag) - 1))
        else:
            target_mag = 0.0
        new_mag = (1.0 - alpha) * mag.copy()
        new_mag[1:] = (1.0 - alpha) * mag[1:] + alpha * target_mag
        new_mag[0] = S  # preserve DC
    else:
        new_mag = mag.copy()
    A_new = new_mag * np.exp(1j * phase)
    a_p_new = np.fft.irfft(A_new, N)
    a_new = np.maximum(a_p_new[:n].real, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy()
    a_new = a_new * S / s_new
    return a_new


def _gs_optimize(a_init, n_steps, alpha_init, alpha_end):
    """Run GS iterations with decaying alpha."""
    n = len(a_init)
    N = 2 * n
    a = a_init.copy()
    S = float(a.sum())
    best_a = a.copy()
    best_sc = _score(a)
    for i in range(n_steps):
        alpha = alpha_init * (alpha_end / alpha_init) ** (i / max(1, n_steps - 1))
        a = _gs_step(a, N, alpha)
        if i % 10 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _variance_grad(a):
    """Compute gradient of sum(b_k^2) where b = conv(a,a). O(n log n) via FFT."""
    n = len(a)
    b = _full_convolve(a, a)
    # grad_j = 4 * sum_i a[i] * b[i+j] for i=0..n-1, valid i+j
    # This is cross-correlation of a with b
    N = _next_pow2(2 * n + n)  # need at least n + (2n-1) - 1 = 3n-2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    # corr[j] = sum_i a[i] * b[i+j] (with circular wrap, but N is large enough)
    grad = 4.0 * corr[:n]
    return grad


def _variance_grad_descend(a_init, deadline, max_steps=5000, lr_init=0.05):
    """Gradient descent on sum(b_k^2) with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        b = _full_convolve(a, a)
        obj = float(np.sum(b ** 2))
        grad = _variance_grad(a)
        # Project gradient to tangent of simplex (sum=1)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.0002 * i)
        z_new = z - step * grad_t
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        if obj < best_sc * 0.5:  # rough check
            pass
        if i % 20 == 0:
            sc = _score(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.9
                z_new = z  # revert
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4
        z = z_new
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk(z, a, c, n, k=5, tol_frac=0.003):
    """Gradient focusing on top-k peaks of convolution."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    # Find top-k peaks
    sorted_idx = np.argsort(c)[::-1]
    topk = sorted_idx[:min(k, len(c))]
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() < k:
        active[topk] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxcor': 50, 'maxls': 60}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _subgrad_polish(a_init, deadline, max_steps=3000, lr_init=0.02, topk=5):
    """Subgradient polish focusing on top-k peaks."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_topk(z, a, c, n, k=topk)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 35:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.05 * np.cos(2.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_mode":
        t = np.arange(n)
        a = 1.0 + 0.03 * np.cos(2.0 * np.pi * t / n) + 0.02 * np.cos(4.0 * np.pi * t / n + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_gs(a, rng, n_gs=5, alpha=0.3):
    """Perturb using a few GS steps with random phase addition."""
    n = len(a)
    N = 2 * n
    S = float(a.sum())
    a_p = np.zeros(N)
    a_p[:n] = a
    A = np.fft.rfft(a_p)
    # Add random phase perturbation
    phase_noise = rng.standard_normal(len(A)) * 0.1
    A = A * np.exp(1j * phase_noise)
    a_p_new = np.fft.irfft(A, N)
    a_new = np.maximum(a_p_new[:n].real, 0.0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy()
    a_new = a_new * S / s_new
    # Run a few GS steps
    for _ in range(n_gs):
        a_new = _gs_step(a_new, N, alpha)
    return a_new


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: GS init -> L-BFGS -> variance GD -> topk subgrad -> basin hopping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: GS initialization from parent ---
    print(f"\n[Phase 1] GS optimization from parent (t={time.monotonic() - t_start:.1f}s)")
    a_gs, sc_gs = _gs_optimize(parent, n_steps=200, alpha_init=0.8, alpha_end=0.1)
    update_best(a_gs, sc_gs, "gs_init")

    # Also try GS from a few different profiles
    for kind in ["cos_mod", "two_mode", "tapered"]:
        if time.monotonic() > deadline - 600:
            break
        a_prof = _make_profile(n_parent, kind)
        a_gs2, sc_gs2 = _gs_optimize(a_prof, n_steps=100, alpha_init=0.5, alpha_end=0.05)
        update_best(a_gs2, sc_gs2, f"gs_{kind}")

    # --- Phase 2: Main L-BFGS from best GS result ---
    print(f"\n[Phase 2] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_start = best_seq.copy() if best_seq is not None else parent
    a_main = _run_lbfgs(a_start, deadline, min_time_remaining=400.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # --- Phase 3: Variance gradient descent ---
    print(f"\n[Phase 3] Variance gradient descent (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_vg, sc_vg = _variance_grad_descend(best_seq, min(deadline - 100, time.monotonic() + 180),
                                             max_steps=3000, lr_init=0.03)
        update_best(a_vg, sc_vg, "variance_gd")

    # --- Phase 4: Re-run L-BFGS after variance GD (if improved) ---
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_main2 = _run_lbfgs(best_seq, deadline, min_time_remaining=100.0)
        update_best(a_main2, _score(a_main2), "lbfgs_post_var")

    # --- Phase 5: Subgradient polish with topk ---
    print(f"\n[Phase 5] Topk subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        for k in [3, 5, 10]:
            if time.monotonic() > deadline - 60:
                break
            a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 50, time.monotonic() + 80),
                                          max_steps=2000, lr_init=0.015, topk=k)
            update_best(a_sg, sc_sg, f"subgrad_k{k}")

    # --- Phase 6: Basin hopping with GS perturbations ---
    print(f"\n[Phase 6] Basin hopping (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_hops = 5
        for i in range(n_hops):
            if time.monotonic() > deadline - 30:
                break
            # Different perturbation strategies
            if i % 3 == 0:
                a_pert = _perturb_gs(best_seq, rng, n_gs=3, alpha=0.5)
            elif i % 3 == 1:
                a_pert = _perturb_fourier(best_seq, rng, n_modes=5)
            else:
                # Random block swap
                z = np.log(np.maximum(best_seq, 1e-30))
                n = len(z)
                bs = max(1, n // 16)
                idx = rng.integers(0, n - 2 * bs)
                idx2 = rng.integers(0, n - 2 * bs)
                if abs(idx - idx2) > bs:
                    tmp = z[idx:idx + bs].copy()
                    z[idx:idx + bs] = z[idx2:idx2 + bs]
                    z[idx2:idx2 + bs] = tmp
                a_pert = _softmax(z)
            sc_pert = _score(a_pert)
            # Quick L-BFGS from perturbed point
            remaining = deadline - time.monotonic() - 20
            if remaining > 20:
                a_hop = _run_lbfgs(a_pert, deadline, min_time_remaining=20.0)
                update_best(a_hop, _score(a_hop), f"hop_{i}")
            else:
                update_best(a_pert, sc_pert, f"hop_{i}_raw")

    # --- Phase 7: Multi-n exploration ---
    print(f"\n[Phase 7] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [3072, 4096, 5120, 6144]
    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            a_init_n = parent.copy()
        else:
            a_init_n = _resample(parent, n_try)
            # Quick GS
            a_init_n, _ = _gs_optimize(a_init_n, n_steps=50, alpha_init=0.5, alpha_end=0.1)
        a_opt_n = _run_lbfgs(a_init_n, deadline, min_time_remaining=60.0)
        update_best(a_opt_n, _score(a_opt_n), f"n_{n_try}")

    # --- Phase 8: Final intensive polish ---
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very fine L-BFGS with tiny beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            max_iter = min(2000, max(50, int(remaining / 0.04)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(
                    fun_final, z, jac=jac_final, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 80}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient with very small steps
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 5, max_steps=500, lr_init=0.005, topk=3)
            update_best(a_fin2, sc_fin2, "final_sg")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
