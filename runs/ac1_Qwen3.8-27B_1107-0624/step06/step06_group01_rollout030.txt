    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Initial gradient estimation
    c_curr = _full_convolve(a_curr, a_curr)
    cm = float(c_curr.max())
    L = 4.0 * n * cm / beta  # Lipschitz estimate
    step = 1.0 / L
    
    best_score = float(2.0 * n * cm)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # FISTA momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute smoothed gradient
        c_y = _full_convolve(y, y)
        cm_y = float(c_y.max())
        shifted = (c_y - cm_y) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_next = _proj_simplex(y - step * grad)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA with top-m smoothed max, penalty-based refinement with binary search on M,
multi-length exploration (n=4096, 8192), and coordinate-descent polishing.
Key improvements over previous: top-m focused gradients, penalty method for final convergence,
and adaptive step sizes with backtracking.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    w has length 2n-1 (the full convolution length)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista_topm(a_init, beta, topm_frac, deadline, max_iter):
    """FISTA with top-m fraction smoothing. topm_frac in (0,1] - fraction of conv values to smooth over."""
    n = len(a_init)
    conv_len = 2 * n - 1
    m = max(1, int(conv_len * topm_frac))
    
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / max(beta, 1e-15)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)

        # Top-m selection for focused gradient
        if m < conv_len:
            top_idx = np.argpartition(c_y, -m)[-m:]
            c_sel = c_y[top_idx]
            c_local_max = float(c_sel.max())
            shifted = (c_sel - c_local_max) / beta
            w_sel = np.exp(shifted)
            w_sel_sum = float(w_sel.sum())
            if w_sel_sum < 1e-300 or not np.isfinite(w_sel_sum):
                w = np.zeros(conv_len)
                w[top_idx[np.argmax(c_sel)]] = 1.0
            else:
                w = np.zeros(conv_len)
                w[top_idx] = w_sel / w_sel_sum
        else:
            c_local_max = float(c_y.max())
            shifted = (c_y - c_local_max) / beta
            w = np.exp(shifted)
            w_sum = float(w.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros(conv_len)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 80 == 0 or it < 5:
            c_chk = _full_convolve(a_next, a_next)
            sc = float(2.0 * n * c_chk.max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            # Adaptive L
            max_c_new = float(c_chk.max())
            L = 4.0 * n * max_c_new / max(beta, 1e-15)
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _penalty_fista(a_init, M, deadline, max_iter):

    best_max_c = max_c

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

        c_y = _full_convolve(y, y)
        excess_y = np.maximum(c_y - M, 0.0)

            grad = _compute_grad(y, w)
            # Gradient clipping for stability

    excess_final = np.maximum(c_final - M, 0.0)

    max_c_final = float(c_final.max())
    if obj_final < best_obj:
        best_obj = obj_final
        best_a = a_curr.copy()

def _binary_search_penalty(a_init, deadline, n_iterations_per_M=2000):
    """Binary search on M using penalty optimization."""
    n = len(a_init)
    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())
    min_possible = float(c_init.min())

    # Binary search bounds
    M_hi = max_c_init
    M_lo = min_possible * 0.999
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    n_bisection = 6  # number of bisection steps

    for bisection in range(n_bisection):
        if time.monotonic() > deadline:
            break

        M_mid = (M_lo + M_hi) / 2.0
        phase_deadline = min(deadline, time.monotonic() + (deadline - time.monotonic()) / (n_bisection - bisection))

        a_pen, obj, max_c_pen = _penalty_fista(a_init, M_mid, phase_deadline, n_iterations_per_M)

        # Check if M_mid is feasible (or nearly so)
        c_pen = _full_convolve(a_pen, a_pen)
        actual_max = float(c_pen.max())
        excess = actual_max - M_mid

        if excess < 0.001 * abs(M_mid) + 1e-10:
            # M_mid is (approximately) feasible
            M_hi = M_mid
            sc = _score_norm(a_pen)
            if sc < best_score:
                best_score = sc
                best_a = a_pen.copy()
            a_init = a_pen.copy()
        else:
            # Not feasible, need larger M
            M_lo = M_mid

        print(f"    Penalty bisection {bisection+1}/{n_bisection}: M_mid={M_mid:.6e}, actual_max={actual_max:.6e}, excess={excess:.2e}")

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z)
    return e / e.sum()


def _coordinate_descent_polish(a, deadline, n_rounds=50):
    """Try individual coordinate modifications to reduce the max convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)
    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 0.5, 1.5]

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # Try random coordinates
        n_tries = min(n, 200)
        indices = rng.integers(0, n, size=n_tries)
        for i in indices:
            for factor in factors:
                a_try = best_a.copy()
                a_try[i] *= factor
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    c_try = _full_convolve(a_try, a_try)
                    sc = float(2.0 * n * c_try.max())
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_try.copy()
                        break

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        t_phase_start = t
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            # Softmax to get a
            z_shift = z - z.max()
            e = np.exp(z_shift)
            s = e.sum()
            a = e / s

            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e2 = np.exp(c_shifted)
            s2 = float(e2.sum())
            if s2 < 1e-300 or not np.isfinite(s2):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e2 / s2

            grad = _compute_grad(a, w)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Convert gradient from a-space to z-space
            # a_i = exp(z_i)/sum(exp(z)), so da_i/dz_j = a_i*(delta_ij - a_j)
            # grad_z_j = sum_i grad_a_i * da_i/dz_j = grad_j * a_j - a_j * sum_i grad_i * a_i
            grad_z = grad * a - a * np.dot(grad, a)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if (t - t_phase_start) % 5000 == 0 and (t - t_phase_start) > 0:
                current_lr = max(0.0003, current_lr * 0.92)
                break  # Move to next beta phase

    # Final evaluation
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a_final = e / e.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

        best_seq = a0.copy()
        best_score = sc0

    # ============ PHASE 1: FISTA with top-m on parent length ============
    print(f"\n=== Phase 1: FISTA top-m on n={n_parent} ===")
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    # Coarse-to-fine beta schedule with top-m
    phase1_deadline = min(deadline, t_start + 500.0)
    beta = 0.02 * max_cc
    topm_fracs = [1.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    for i, tmf in enumerate(topm_fracs):
        if time.monotonic() > phase1_deadline:
            break
        sub_deadline = min(phase1_deadline, time.monotonic() + 60.0)
        a_work, sc = _fista_topm(a_work, beta, tmf, sub_deadline, max_iter=50000)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  Phase 1, topm={tmf:.3f}, beta={beta:.2e}: score={sc_full:.10f}")
        beta *= 0.5

    # ============ PHASE 2: Penalty-based refinement ============
    print(f"\n=== Phase 2: Penalty refinement on n={n_parent} ===")
    phase2_deadline = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_deadline - 60:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        a_pen, sc_pen = _binary_search_penalty(a_work, phase2_deadline, n_iterations_per_M=3000)
        sc_pen_full = _score(a_pen)
        if sc_pen_full < best_score:
            best_score = sc_pen_full
            best_seq = a_pen.copy()
            print(f"  Penalty: score={sc_pen_full:.10f}")

    # ============ PHASE 3: Adam log-space fine optimization ============
    print(f"\n=== Phase 3: Adam log-space ===")
    phase3_deadline = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_deadline - 30:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas_fine = [0.001 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc]
        a_opt, sc_opt = _adam_logspace(a_work, phase3_deadline, betas_fine, lr_init=0.01, max_iter_total=200000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f}")

    # ============ PHASE 4: Try n=8192 ============
    print(f"\n=== Phase 4: Try n=8192 ===")
    n_large = 8192
    phase4_deadline = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_deadline - 60:
        if parent is not None:
            a_large = _interp_profile(parent, n_large)
        else:
            a_large = _interp_profile(best_seq, n_large)
        a_large = a_large / a_large.sum()
        sc_large_init = _score(a_large)
        print(f"  Large init score: {sc_large_init:.10f}")

        # FISTA on large
        c_l = _full_convolve(a_large, a_large)
        max_cl = float(c_l.max())
        beta_l = 0.02 * max_cl

        topm_large = [1.0, 0.5, 0.2, 0.05, 0.01]
        for tmf in topm_large:
            if time.monotonic() > phase4_deadline:
                break
            sub_dl = min(phase4_deadline, time.monotonic() + 60.0)
            a_large, sc_l = _fista_topm(a_large, beta_l, tmf, sub_dl, max_iter=30000)
            a_large = a_large / a_large.sum()
            sc_l_full = _score(a_large)
            if sc_l_full < best_score:
                best_score = sc_l_full
                best_seq = a_large.copy()
                print(f"  Large n={n_large}, topm={tmf}: score={sc_l_full:.10f}")
            beta_l *= 0.5

    # ============ PHASE 5: Multi-restart diversification ============
    print(f"\n=== Phase 5: Multi-restart ===")
    phase5_deadline = min(deadline, t_start + 1780.0)
    if time.monotonic() < phase5_deadline - 30 and best_seq is not None:
        n_best = len(best_seq)
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        restarts = 3
        per_restart = max(5.0, (phase5_deadline - time.monotonic() - 10.0) / restarts)

        for i in range(restarts):
            if time.monotonic() > phase5_deadline:
                break

            if i == 0:
                p_start = _perturb(a_work, 0.01, rng)
            elif i == 1:
                sz = max(1, n_best // 2048)
                shift = int(rng.integers(1, max(2, sz)))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                alt = _make_profile(n_best, "tapered")
                p_start = 0.8 * a_work + 0.2 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0005 * max_cs, 0.0002 * max_cs, 0.00005 * max_cs]

            p_best, sc_b = _adam_logspace(
                p_start, time.monotonic() + per_restart,
                bs, lr_init=0.005, max_iter_total=100000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f}")

    # ============ PHASE 6: Final coordinate descent polish ============
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_cd, sc_cd = _coordinate_descent_polish(
            best_seq, time.monotonic() + min(15.0, deadline - time.monotonic() - 3),
            n_rounds=30
        )
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc_cd_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
