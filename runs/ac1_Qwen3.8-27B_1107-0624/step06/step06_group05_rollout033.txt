def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    best_a = a.copy()
    
    total_iter = max_iter
    for it in range(total_iter):
        if time.monotonic() > deadline:
            break
        
        # Annealing schedule (geometric)
        frac = it / max(1, total_iter)
        beta = beta_start * (beta_end / beta_start) ** frac
        eta = eta_start * (eta_end / eta_start) ** frac
