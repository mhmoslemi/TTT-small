```python
def _lp_update(f, M):
    """Solve LP to find best g given f and M."""
    n = len(f)
    # Constraint matrix A: (2n-1) x n
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # i.e., A[k, j] = f[j] if k-n+1 <= j <= min(k, n-1)
    
    # Build A as a sparse matrix
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[j]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_vec = np.full(2*n-1, M)
    
    # Bounds: g_j >= 0
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A.tocsc(), b_ub=b_vec, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    return None
```

```python
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.signal import fftconvolve
from scipy.sparse import csr_matrix

def _convolve(a, b):
    return fftconvolve(a, b)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def _subgradient(a, tol_frac=1e-4):
    """Compute subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n), c_max
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h, c_max

def _soft_subgradient(a, temperature=1e-3):
    """Compute soft subgradient using exp weighting."""
    n = len(a)
    c = _convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / temperature
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h, c_max

def _active_sg_optimize(a0, deadline, tol_frac=1e-4, max_iters=50000):
    """Active-set subgradient with backtracking line search."""
    n = len(a0)
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_cmax = _convolve(a, a).max()
    lr = 0.01
    stall = 0
    patience = 2000
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        sg = 2.0 * h
        sg = sg - sg.mean()  # Center for simplex
        
        # Backtracking line search
        best_alpha = lr
        best_trial_cmax = c_max
        for alpha_try in [lr, lr*0.5, lr*0.25, lr*0.125, lr*0.0625]:
            a_try = a - alpha_try * sg
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-30:

            if cmax_try < best_trial_cmax:

                best_alpha = alpha_try

        a = np.maximum(a, 0)
        s = a.sum()

        a = a / s

            stall = 0

def _fista_optimize(a0, deadline, temp_init=0.01, max_iters=20000):

    n = len(a0)
    z = _normalize(a0.copy())
    y = z.copy()
    t_k = 1.0
    best_a = z.copy()
    best_cmax = _convolve(z, z).max()

        sg = sg - sg.mean()

        z_new = np.maximum(z_new, 0)
        z_new = z_new / z_new.sum()

def _lp_direction(f, M, n_max=None):
    """Solve LP to find direction g."""
    n = len(f)
    if n_max is not None and n > n_max:
        # Interpolate to smaller size
        t_new = np.linspace(0, 1, n_max)
        t_old = np.linspace(0, 1, n)
        f_small = np.interp(t_new, t_old, f)
        f_small = f_small / f_small.sum()
        n = n_max
    else:
        f_small = f
    
    n = len(f_small)
    M = max(M, 0)
    
    # Build constraint matrix
    # A[k, j] = f_small[j] if 0 <= k-j < n, else 0
    # Number of constraints: 2n-1
    # Number of variables: n
    
    # For memory efficiency, build as CSR
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f_small[j])
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_vec = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            # Interpolate back if we downsized
            if n_max is not None and n < len(f):
                t_new = np.linspace(0, 1, len(f))
                t_old = np.linspace(0, 1, n)
                g_full = np.interp(t_new, t_old, g)
                g_full = np.maximum(g_full, 0)
                if g_full.sum() > 0:
                    g_full = g_full / g_full.sum()
                return g_full
            else:
                g = np.maximum(g, 0)
                if g.sum() > 0:
                    g = g / g.sum()
                return g
    except Exception as e:
        print(f"    LP failed: {e}")
    
    return None

def _lp_iteration(a0, deadline, n_max=2048, max_iters=20):
    """LP-based iteration from state of the art."""
    a = _normalize(a0.copy())
    best_a = a.copy()

    for i in range(max_iters):

                best_cmax = cmax_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return _normalize(best_a)
```

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding, active-set subgradient with adaptive
tolerance and momentum, FISTA acceleration, multi-start asymmetric
profiles, variable length search, and intensive final polishing.
Key insight: optimal sequence is asymmetric; peak of convolution is off-center.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.signal import fftconvolve
from scipy.sparse import csr_matrix


def _convolve(a, b):
    return fftconvolve(a, b)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def _subgrad_vec(a, w):
    """Compute 2*conv(w, a_rev)[n-1:2n-1] efficiently."""
    n = len(a)
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _make_profile(n, kind="uniform"):

    elif kind == "ramp_up":

    elif kind == "ramp_down":

        return a / a.sum()
    elif kind == "exp_ramp":
        t = np.linspace(0, 1, n)
        a = np.exp(0.8 * t)

    elif kind == "quad_ramp":

def _active_sg_optimize(a0, deadline, tol_frac=1e-4, max_iters=50000, lr_init=0.01):

    momentum = 0.0

            active[int(np.argmax(c))] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()

        sg = sg - sg.mean()

        if sg_norm < 1e-15:

        best_trial = c_max + 1e-10

            alpha_try = lr * mult
            v_new = momentum * v - alpha_try * sg
            a_try = np.maximum(a + v_new, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-30:

            a_try = a_try / s_try

            stall += 1
            if stall > patience:

            v = momentum * v

            v = np.zeros(n)

def _soft_sg_optimize(a0, deadline, temp_init=0.005, max_iters=20000, lr_init=0.005):

        w = e / s_e

            lr = min(lr * 1.02, 0.03)
            temp *= 0.98
            stall = 0
        else:

            if stall > 5000:

def _lp_direction(f, M, size_limit=2048):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0."""
    n = len(f)
    if n > size_limit:
        f_small = _interpolate_to_n(f, size_limit)
    else:
        f_small = f.copy()
    n_eff = len(f_small)

    M = float(M)
    if M < 1e-30:
        return None

    # Build constraint matrix (sparse)
    rows = []
    cols = []
    data = []
    for k in range(2 * n_eff - 1):
        j_start = max(0, k - n_eff + 1)
        j_end = min(n_eff - 1, k)
        js = range(j_start, j_end + 1)
        for j in js:
            rows.append(k)
            cols.append(j)
            data.append(f_small[j])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n_eff - 1, n_eff))
    c_obj = -np.ones(n_eff)
    b_vec = np.full(2 * n_eff - 1, M)
    bounds = [(0, None)] * n_eff

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                         options={'time_limit': 30})
        if result.success:
            g = np.maximum(result.x, 0.0)
            sg = g.sum()
            if sg > 1e-30:
                g = g / sg
                if n > size_limit:
                    g = _interpolate_to_n(g, n)
                return g
    except Exception:
        pass
    return None

            break

        c = _convolve(a, a)
        M = float(c.max())

        t_start_lp = time.monotonic()
        g = _lp_direction(a, M, size_limit=size_limit)
        lp_time = time.monotonic() - t_start_lp
        print(f"    LP round {r}: time={lp_time:.1f}s")

        if g is None:
            break

        # Try different mixing fractions
        improved = False
        for t in [0.15, 0.25, 0.35, 0.5, 0.7]:
            a_try = (1.0 - t) * a + t * g
            a_try = _normalize(a_try)
            c_try = _convolve(a_try, a_try)
            cmax_try = float(c_try.max())
            if cmax_try < best_cmax - 1e-15:
                best_cmax = cmax_try
                best_a = a_try.copy()
                a = a_try.copy()
                improved = True
                break

        if not improved:
            break

    return _normalize(best_a)


def _coordinate_push(a0, deadline, max_rounds=50000):
    """Targeted coordinate descent: push down the peak of the convolution."""
    n = len(a0)
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_cmax = float(_convolve(a, a).max())

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        # Find the peak position
        peak_k = int(np.argmax(c))
        c_max = c[peak_k]

        # The peak c[peak_k] = sum_j a[j]*a[peak_k-j]
        # To reduce it, reduce some a[j] that contributes most
        # and add it to a position that doesn't contribute to the peak
        j_vals = np.arange(max(0, peak_k - n + 1), min(n - 1, peak_k) + 1)
        contributions = a[j_vals] * a[peak_k - j_vals]
        total_contrib = contributions.sum()
        if total_contrib < 1e-30:
            break

        # Find the j with largest contribution
        j_best = j_vals[np.argmax(contributions)]
        # Find a target index that doesn't contribute to the peak
        target_candidates = [j for j in range(n) if j not in j_vals or (peak_k - j) not in j_vals]
        if not target_candidates:
            break
        # Pick the target with smallest current value (to minimize disruption)
        target_j = min(target_candidates, key=lambda j: a[j])

        # Move a small amount
        frac = min(0.001, 0.01 * contributions[np.argmax(contributions)] / total_contrib)
        amount = frac * a[j_best]
        a_try = a.copy()
        a_try[j_best] -= amount
        a_try[target_j] += amount
        a_try = _normalize(a_try)

        c_try = _convolve(a_try, a_try)
        cmax_try = float(c_try.max())

        if cmax_try < best_cmax - 1e-16:
            best_cmax = cmax_try
            best_a = a_try.copy()
            a = a_try.copy()

    return _normalize(best_a)


def _perturb(a, rng, method="gauss", scale=0.02):
    """Perturb a sequence."""
    n = len(a)
    if method == "gauss":
        z = np.log(np.maximum(a, 1e-15))
        z = z + scale * rng.standard_normal(n)
        z = z - z.mean()
        return np.exp(z) / np.exp(z).sum()
    elif method == "fourier":
        z = np.log(np.maximum(a, 1e-15))
        t = np.arange(n)
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 10)))
            amp = scale * 0.5 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        z = z - z.mean()
        return np.exp(z) / np.exp(z).sum()
    else:
        # Block perturbation
        z = np.log(np.maximum(a, 1e-15))
        n_blocks = 4
        block_size = max(10, n // n_blocks)
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-scale, scale)
        z = z - z.mean()
        return np.exp(z) / np.exp(z).sum()

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
