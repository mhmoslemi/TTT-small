"""
L-BFGS with softmax-continuation + exact coordinate descent + multi-n exploration.
Key innovation: uses L-BFGS (curvature-aware) instead of first-order methods,
combined with exact coordinate descent for final polish.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:n + m - 1]


def _conv_self(a):
    return _fft_conv(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assume sum(a)=1, return 2n*max_conv"""
    n = len(a)
    c = _conv_self(a)
    return float(2.0 * n * c.max())


def _softmax_obj_grad(z, beta):
    """Compute smoothed objective and gradient in log-space z.
    a = softmax(z), objective = 2n * soft_max(conv(a,a), beta)
    """
    n = len(z)
    a = np.exp(z - z.max())
    s = a.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = a / s

    c = _conv_self(a)
    cm = c.max()
    c_shift = (c - cm) / beta
    np.clip(c_shift, -50, 50, out=c_shift)
    w = np.exp(c_shift)
    w_sum = float(w.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        w_sum = 1.0
    else:
        w = w / w_sum

    obj = 2.0 * n * (cm + beta * np.log(w_sum))

    # Gradient computation:
    # h_i = sum_k w_k * dc_k/da_i = cross_corr(w, a)[i] + a_i * sum(w)
    # cross_corr(w, a)[i] = sum_k w_k * a_{k-i} = (w * a_rev)[n-1+i]
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    h = wc[n - 1:2 * n - 1] + a  # length n

    # dF/dz_j = a_j * (h_j - sum_i h_i * a_i) * 2n
    W = float(np.dot(h, a))
    grad = a * (h - W) * 2.0 * n

    return obj, grad


def _topk_obj_grad(z, beta, top_k):
    """Smoothed objective using only top-k peaks for the gradient."""
    n = len(z)
    a = np.exp(z - z.max())
    s = a.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = a / s

    c = _conv_self(a)
    cm = c.max()

    if top_k < len(c) and top_k > 0:
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        base = c_top.max()
        c_shift = (c_top - base) / beta
        np.clip(c_shift, -50, 50, out=c_shift)
        w_top = np.exp(c_shift)
        w_sum_top = float(w_top.sum())
        if w_sum_top < 1e-300:
            top_k_eff = 1
            idx = idx[np.argmax(c_top):np.argmax(c_top) + 1]
            w_top = np.array([1.0])
            w_sum_top = 1.0
        else:
            w_top = w_top / w_sum_top
            top_k_eff = len(idx)

        obj = 2.0 * n * (cm + beta * np.log(w_sum_top))

        # Gradient: only top-k peaks contribute
        a_rev = a[::-1]
        # For each peak position k with weight w_k:
        # contribution to h_i = w_k * (a_{k-i} + a_i)
        # Sum over k in idx: h = sum_k w_k * cross_corr_component
        # cross_corr(w, a)[i] where w is nonzero only at positions in idx
        w_full = np.zeros_like(c)
        for kk, ww in zip(idx, w_top):
            w_full[kk] = ww

        wc = _fft_conv(w_full, a_rev)
        h = wc[n - 1:2 * n - 1] + a * w_full.sum()

        W = float(np.dot(h, a))
        grad = a * (h - W) * 2.0 * n
    else:
        c_shift = (c - cm) / beta
        np.clip(c_shift, -50, 50, out=c_shift)
        w = np.exp(c_shift)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            w_sum = 1.0
        else:
            w = w / w_sum
        obj = 2.0 * n * (cm + beta * np.log(w_sum))
        a_rev = a[::-1]
        wc = _fft_conv(w, a_rev)
        h = wc[n - 1:2 * n - 1] + a
        W = float(np.dot(h, a))
        grad = a * (h - W) * 2.0 * n

    return obj, grad


def _lbfgs_phase(z0, beta_schedule, top_k_schedule, deadline, max_iter_per_phase=20000):
    """Run L-BFGS with continuation over beta and top_k."""
    n = len(z0)
    z = z0.copy()
    best_z = z.copy()
    best_score = _score_norm(_safe_softmax(z))

    for phase in range(len(beta_schedule)):
        if time.monotonic() > deadline:
            break
        beta = beta_schedule[phase]
        top_k = top_k_schedule[phase] if phase < len(top_k_schedule) else 0

        def fun_and_grad(zv):
            return _topk_obj_grad(zv, beta, top_k)

        per_deadline = min(deadline, time.monotonic() + max(5.0, (deadline - time.monotonic()) / max(1, len(beta_schedule) - phase)))

        result = minimize(
            fun_and_grad, z,
            jac=True,
            method='L-BFGS-B',
            options={'maxiter': max_iter_per_phase, 'gtol': 1e-10,
                     'maxls': 50, 'ftol': 1e-12}
        )

        if result.success or result.nit > 0:
            z_try = result.x.copy()
            z_try = z_try - z_try.mean()  # Remove gauge
            a_try = _safe_softmax(z_try)
            sc = _score_norm(a_try)
            if sc < best_score:
                best_score = sc
                best_z = z_try.copy()
                z = z_try.copy()
        # Always update z to the result (even if not better, for continuation)
        z = result.x.copy()
        z = z - z.mean()

        print(f"    L-BFGS phase {phase}: beta={beta:.2e}, top_k={top_k}, "
              f"nit={result.nit}, score={best_score:.10f}")

    return best_z, best_score


def _safe_softmax(z):
    z = z - z.max()
    a = np.exp(z)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return a / s


def _coord_descent(a, b, max_sweeps=500, time_budget=300.0):
    """Exact coordinate descent with analytical 1D updates."""
    n = len(a)
    S = float(a.sum())
    best_score = float(2.0 * n * b.max())
    best_a = a.copy()
    t_start = time.monotonic()

    fracs = np.array([-0.4, -0.2, -0.1, -0.05, -0.02, 0.02, 0.05, 0.1, 0.2, 0.4])

    for sweep in range(max_sweeps):
        if time.monotonic() > t_start + time_budget:
            break

        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-14:
                continue

            # b_affected = b[j : j+n] (positions where b changes)
            k_hi = min(j + n, 2 * n - 1)
            b_aff = b[j:k_hi].copy()
            # influence: a[k-j] for k in [j, k_hi-1] → a[0 : k_hi-j]
            infl = a[0:k_hi - j]
            # self-term position: k=2j, index in b_aff = 2j - j = j
            self_pos = 2 * j - j  # = j, but only if j < len(b_aff)

            # Unaffected max (upper bound)
            unaff_max = b.max()  # conservative

            best_t = 0.0
            best_sc = best_score

            for frac in fracs:
                t = frac * old_val
                new_val = old_val + t
                if new_val < 0:
                    continue
                new_S = S + t
                if new_S < 0.01:
                    continue

                new_aff = b_aff + 2.0 * t * infl
                if self_pos < len(new_aff):
                    new_aff[self_pos] += t * t

                # New max: max of affected and unaffected
                aff_max = float(new_aff.max())
                # For the unaffected part, it's at most b.max() (but we can be tighter)
                # The unaffected part is b[0:j] and b[k_hi:]
                unaff = 0.0
                if j > 0:
                    unaff = max(unaff, float(b[:j].max()))
                if k_hi < 2 * n - 1:
                    unaff = max(unaff, float(b[k_hi:].max()))

                new_max = max(aff_max, unaff)
                sc = 2.0 * n * new_max / (new_S * new_S)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t

            if best_t != 0.0:
                a[j] += best_t
                S += best_t
                # Update b incrementally
                # b[k] += 2*best_t*a_old[k-j] for k in [j, k_hi-1]
                a_old_j = a[j] - best_t
                # For k in [j, j+n): b[k] += 2*best_t*a[k-j]
                # But a has changed at position j, so use old value for the influence
                # Actually the correct update is:
                # b_new[k] = b_old[k] + 2*best_t*a_old[k-j] + best_t^2 (if k=2j)
                # where a_old[k-j] uses the state BEFORE this update
                # Since only a[j] changed, and a[k-j] for k-j≠j is unchanged,
                # we need a_old[k-j] which equals a[k-j] for k-j≠j, and a_old_j for k-j=j
                b[j:k_hi] += 2.0 * best_t * infl  # infl was a[0:k_hi-j], which is correct (old values)
                # Wait, infl = a[0:k_hi-j] was computed BEFORE the update, so it's correct
                if 2 * j < 2 * n - 1:
                    b[2 * j] += best_t * best_t
                improved = True
                best_score = best_sc

        # Periodic recomputation to remove drift
        if sweep % 50 == 49:
            b = _conv_self(a)
            S = float(a.sum())
            best_score = float(2.0 * n * b.max())
            if best_score < _score_norm(best_a):
                best_a = a.copy()

        if not improved and sweep > 10:
            break

    b_final = _conv_self(a)
    final_score = float(2.0 * n * b_final.max())
    if final_score < best_score:
        best_a = a.copy()
        best_score = final_score

    return best_a, b_final, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

        final = np.ones(256)
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
