"""
Flat-square-root initialization + adaptive L-BFGS for minimizing
2*n*max(conv(a,a))/sum(a)^2.
Key insight: The coefficients of (1-z)^(-1/2) (= C(2k,k)/4^k) make the linear
autocorrelation approximately flat, yielding scores ~1.05-1.2 for large n,
far below the uniform initialization (score=2) which is a local minimum.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import gammaln


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_z(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _flat_sqrt_coeffs(n):
    """Coefficients of (1-z)^(-1/2): C(2k,k)/4^k for k=0..n-1."""
    k = np.arange(n, dtype=np.float64)
    log_coeffs = gammaln(2.0 * k + 1.0) - 2.0 * gammaln(k + 1.0) - k * np.log(4.0)
    a = np.exp(log_coeffs)
    return a


def _flat_sqrt_init(n):
    """Normalized flat-sqrt initialization."""
    a = _flat_sqrt_coeffs(n)
    s = a.sum()
    return a / s


def _smoobj(z, beta, n):
    """Smoothed objective: 2n/beta * log(sum exp(C(k)/beta))."""
    a = _softmax_z(z)
    c = _full_convolve(a, a)
    c_scaled = c / beta
    c_max = c_scaled.max()
    e = np.exp(c_scaled - c_max)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c.max()
    return 2.0 * n * (c_max + np.log(s))


def _smograd_z(z, beta, n):
    """Gradient of smoothed objective w.r.t. z (log-space parameters)."""
    a = _softmax_z(z)
    c = _full_convolve(a, a)
    c_scaled = c / beta
    c_max = c_scaled.max()
    e = np.exp(c_scaled - c_max)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # h_j = sum_k w[k] * a[k-j], computed as convolve(w, a_rev)[n-1+j]
    a_rev = a[::-1]
    h_full = _full_convolve(w, a_rev)
    h = h_full[n - 1:2 * n - 1]

    # Diagonal correction: w[2j] * a[j] where 2j is valid
    two_j = 2 * np.arange(n)
    diag = np.zeros(n)
    valid = two_j < len(w)
    diag[valid] = w[two_j[valid]] * a[np.arange(n)[valid]]

    # g_j = 2n/beta * (2*h_j - diag_j)
    g = (2.0 * n / beta) * (2.0 * h - diag)

    # Convert to z-gradient: dz_j = a_j * (g_j - sum(g*a))
    ga = float(np.dot(g, a))
    dz = a * (g - ga)
    return dz


def _gradient_active_z(z, n, tol_frac=0.005):
    """Gradient using active set (top lags) instead of log-sum-exp."""
    a = _softmax_z(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()

    a_rev = a[::-1]
    h_full = _full_convolve(w, a_rev)
    h = h_full[n - 1:2 * n - 1]

    two_j = 2 * np.arange(n)
    diag = np.zeros(n)
    valid = two_j < len(w)
    diag[valid] = w[two_j[valid]] * a[np.arange(n)[valid]]

    g = 4.0 * n * (2.0 * h - diag)
    ga = float(np.dot(g, a))
    dz = a * (g - ga)
    return dz

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation in log-space."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta schedule
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[min(5, len(c_sorted) - 1)]
    base_beta = max((c_sorted[0] - gap) * 0.5, max_c * 0.005, max_c * 1e-6)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0004, 0.00015])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        time_per_iter = 0.003 * max(1.0, n / 4096.0)
        max_iter = min(3000, max(50, int(remaining / (time_per_iter * 15))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd_z(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax_z(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                print(f"    beta={beta:.2e} sc={current_sc:.10f} iter={result.niter}")
        except Exception:
            pass

def _dual_iteration(a, n, n_dual_steps=50, tol=1e-8):
    """
    Dual (KKT) iteration: alternate between fixing w (dual) and solving for a (primal).
    At optimum: 2*sum_k w_k * a_{k-j} - w_{2j} * a_j = lambda for all j with a_j > 0.
    """
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    n_lags = 2 * n - 1

    for it in range(n_dual_steps):
        # Compute convolution
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Update dual: uniform over active lags
        tol_active = 0.01 * c_max
        active = c >= (c_max - tol_active)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros(n_lags)
        w[active] = 1.0 / active.sum()

        # Solve (2*T_w - D_w) a = lambda * 1 (approximately)
        # Using iterative method: a_new = lambda^{-1} * (D_w a + ... )
        # Actually, let's use a simpler approach: gradient step on the dual objective
        # Minimize max_k C(k) = minimize t subject to C(k) <= t for all k
        # Dual: maximize sum_k w_k * t_k subject to ...
        # Let's just do a projected gradient step

        # The KKT condition says: grad_a (sum_k w_k C(k)) = lambda * 1 (where a > 0)
        # grad_a_j (sum_k w_k C(k)) = 2 * sum_k w_k * a_{k-j} - w_{2j} * a_j
        # We want this to be proportional to 1 (for the support of a).

        # Compute the gradient
        a_rev = a[::-1]
        h_full = _full_convolve(w, a_rev)
        h = h_full[n - 1:2 * n - 1]
        two_j = 2 * np.arange(n)
        diag = np.zeros(n)
        valid = two_j < n_lags
        diag[valid] = w[two_j[valid]] * a[np.arange(n)[valid]]
        grad = 2.0 * h - diag

        # Project: we want grad_j = lambda for all j in support
        # lambda = mean of grad over support
        support = a > 1e-10
        if support.sum() == 0:
            break
        lam = float(grad[support].mean())

        # Direction: a_new_j proportional to (something that makes grad = lambda)
        # a_new = a + alpha * (something)
        # For now, use a multiplicative update: a_new_j = a_j * exp(alpha * (lambda - grad_j))
        alpha = 0.5
        a_new = a * np.exp(alpha * (lam - grad))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        if _score(a) - best_sc > 0.01:
            # Diverged, revert
            a = best_a.copy()
            break

    return best_a, best_sc

        if time.monotonic() > deadline - 80:
            break

        # Run L-BFGS
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"lbfgs_n{n_try}")

        # Quick subgradient polish
        if time.monotonic() < deadline - 60:
            a_sg, sc_sg = _subgrad_polish(a_opt, min(deadline - 50, time.monotonic() + 90),
                                          n_max_steps=2000, lr_init=0.015)
            update_best(a_sg, sc_sg, f"sg_n{n_try}")

        # Dual iteration
        if time.monotonic() < deadline - 40:
            a_dual, sc_dual = _dual_iteration(a_sg if a_sg is not None else a_opt,
                                              n_try, n_dual_steps=30)
            update_best(a_dual, sc_dual, f"dual_n{n_try}")

    # --- Phase 2: If parent exists, also try from parent ---
    if parent is not None and time.monotonic() < deadline - 100:
        print(f"\n[Phase 2] From parent (t={time.monotonic()-t_start:.1f}s)")
        n_p = len(parent)
        a_from_parent = _run_lbfgs(parent, deadline, min_time_remaining=60.0)
        update_best(a_from_parent, _score(a_from_parent), f"parent_n{n_p}")

        # Blend flat-sqrt with parent
        if best_seq is not None and len(best_seq) == n_p:
            for alpha in [0.3, 0.5, 0.7]:
                if time.monotonic() > deadline - 40:
                    break
                mixed = alpha * _flat_sqrt_init(n_p) + (1 - alpha) * parent
                mixed = mixed / mixed.sum()
                sc_mix = _score(mixed)
                update_best(mixed, sc_mix, f"mixed_{alpha}")
                if sc_mix < best_score + 0.01:
                    a_m = _run_lbfgs(mixed, deadline, min_time_remaining=30.0)
                    update_best(a_m, _score(a_m), f"mixed_lbfgs_{alpha}")

    # --- Phase 3: Diverse restarts from best ---
    print(f"\n[Phase 3] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_restarts = 4
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=6)
            elif i == 2:
                # Try a different n
                n_alt = len(best_seq) + 512
                a_alt = _flat_sqrt_init(n_alt)
                p_start = a_alt
            else:
                p_start = _perturb(best_seq, 0.08, rng)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 10:
                a_r2, sc_r2 = _subgrad_polish(a_r, min(deadline - 8, time.monotonic() + 40),
                                              n_max_steps=1000, lr_init=0.01)
                update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # --- Phase 4: Final intensive polish ---
    print(f"\n[Phase 4] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_final = len(best_seq)

        # Very fine L-BFGS
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_l: _smoobj(z_l, beta_final, n_final),
                    z,
                    jac=lambda z_l: _smograd_z(z_l, beta_final, n_final),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax_z(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 3, n_max_steps=2000, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Final dual
        if time.monotonic() < deadline - 2:
            a_d, sc_d = _dual_iteration(best_seq, len(best_seq), n_dual_steps=50)
            update_best(a_d, sc_d, "final_dual")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _flat_sqrt_init(256)
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            print(f"FINAL: internal={sc_final:.10f}, evaluated={sc_eval:.10f}, "
                  f"n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        except Exception:
            print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
                  f"time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _flat_sqrt_init(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
