"""
Optimized multi-stage sequence optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses refined Adam in log-space with adaptive top-K smoothing, FISTA warm-up,
coordinate descent polishing, and peak-shaving post-processing.
Key improvements over previous: more aggressive fine-beta schedule, adaptive
learning rate, momentum resets, and a dedicated peak-shaving phase.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a. grad_i = 2 * cross_corr(a, w)[i]."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_softmax_weights(c, beta, k):
    """Softmax weights over top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -(k))[-(k)]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, n, grad_a):
    """Gradient in log-space: dF/dz_i = 2n * a_i * (grad_i - <grad, a>)."""
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k_init, top_k_min,
                   max_iter_per_phase, momentum_decay):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        current_lr = lr_init * (0.7 ** phase_idx)
        top_k = max(top_k_min, int(top_k_init * (0.85 ** phase_idx)))

        c_curr = _full_convolve(_softmax_normalize(z), _softmax_normalize(z))
        max_c = float(c_curr.max())
        if max_c < 1e-15:
            beta_eff = max(beta, 1e-10)
        else:
            beta_eff = beta * max_c

        it_in_phase = 0
        phase_best = best_score
        phase_best_a = best_a.copy()

        while it_in_phase < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta_eff, top_k)
            grad_a = _compute_grad_conv(a, w)
            grad = _logspace_grad(a, n, grad_a)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 0.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < phase_best:
                    phase_best = sc
                    phase_best_a = a_chk.copy()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_chk.copy()

        a_final = _softmax_normalize(z)
        sc_final = _score_norm(a_final)
        if np.isfinite(sc_final):
            if sc_final < phase_best:
                phase_best = sc_final
                phase_best_a = a_final.copy()
            if sc_final < best_score:
                best_score = sc_final
                best_a = a_final.copy()

        z = np.log(np.maximum(phase_best_a, 1e-15))
        z = z - z.mean()
        m_buf *= momentum_decay
        v_buf *= momentum_decay

    return best_a, best_score


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    """FISTA on simplex with top-K softened max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    if max_c < 1e-10:
        max_c = 1e-10
    beta_eff = beta * max_c
    L = 8.0 * n / max_c
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta_eff, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent polishing with 1D optimization per coordinate."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(98765)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            if a_i_old < 1e-10:
                continue

            c = _full_convolve(a, a)
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            idx_top = np.argsort(c)[-top_k_w:]

            t_candidates = np.array([a_i_old * f for f in
                                     [0.0, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]])

            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 5.0 * max(a_i_old, 0.05):
                        t_candidates = np.append(t_candidates,
                                                 [t_star, t_star * 0.7, t_star * 1.3])

            t_candidates = np.clip(t_candidates, 0.0, 10.0)
            t_candidates = np.unique(np.round(t_candidates, 12))

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(np.max(c_trial[idx_top]))
                global_max = float(c_trial.max())
                f_val = 2.0 * n * global_max / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if abs(best_t - a_i_old) > 1e-10:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        sc_after = _score_norm(a)
        if sc_after < best_score:
            best_score = sc_after

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _peak_shave(a, n_iter, deadline):
    """Direct peak shaving: reduce coordinates contributing to the max convolution peak."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(54321)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])

        if M < 1e-10:
            break

        # Find contributors to c[k_star]
        # c[k_star] = sum_i a[i] * a[k_star - i] for valid i
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Try reducing the top contributors
        top_contrib = np.argsort(contrib)[-20:]

        best_delta = 0
        best_score_try = best_score
        best_idx = -1

        for i in top_contrib:
            if contrib[i] < 1e-12:
                continue
            # Try reducing a[i] by a small fraction
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = a[i] * frac
                if delta < 1e-12:
                    continue
                a[i] -= delta
                s = a.sum()
                if s < 0.01:
                    a[i] += delta
                    break
                a_norm = a / s
                c_check = _full_convolve(a_norm, a_norm)
                sc = float(2.0 * n * c_check.max())
                if sc < best_score_try - 1e-8:
                    best_score_try = sc
                    best_delta = delta
                    best_idx = i
                a[i] += delta

        if best_idx >= 0:
            a[best_idx] -= best_delta
            s = a.sum()
            if s > 0.01:
                a = a / s
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
        else:
            # No single coordinate helped; try random perturbation
            idx = rng.integers(0, n)
            if a[idx] > 1e-8:
                delta = a[idx] * 0.01
                a[idx] -= delta
                s = a.sum()
                if s > 0.01:
                    a_try = a / s
                    sc_try = _score_norm(a_try)
                    if sc_try < best_score:
                        a = a_try
                        best_score = sc_try
                    else:
                        a[idx] += delta

    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 7]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + 0.3 * freq)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_positive":
        rng = np.random.default_rng(42)
        a = rng.random(n) + 0.1
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "sawtooth":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * (2.0 * (t - np.floor(t + 0.5)))
        a = np.maximum(a, 0.05)
        return a / a.sum()
    else:
        return np.ones(n) / n

        remaining_b = deadline - time.monotonic()
        phase_b_budget = remaining_b * 0.45
        phase_b_end = time.monotonic() + phase_b_budget
        print(f"  Phase B (Adam main): budget={phase_b_budget:.1f}s")

        c_main = _full_convolve(a_work, a_work)
        max_c_main = float(c_main.max())

        # Build beta schedule: geometric from coarse to fine
        n_betas = 18
        beta_start = 0.02
        beta_end = 0.00003
        ratio = (beta_end / beta_start) ** (1.0 / (n_betas - 1))
        beta_schedule = [beta_start * (ratio ** i) for i in range(n_betas)]

        # Allocate more iterations to later (finer) phases
        weights = np.array([1.0] * 4 + [1.5] * 7 + [2.5] * 7)
        weights = weights / weights.sum()
        iters_per_phase = [max(2000, int(phase_b_budget * w * 2000)) for w in weights]
        iters_per_phase = iters_per_phase[:n_betas]

        a_work, sc_work = _adam_logspace(a_work, phase_b_end, beta_schedule,
                                         lr_init=0.008, top_k_init=250, top_k_min=15,
                                         max_iter_per_phase=max(iters_per_phase),
                                         momentum_decay=0.5)
        a_work = a_work / a_work.sum()
        sc_work_full = _score(a_work)
        if sc_work_full < best_score:
            best_score = sc_work_full
            best_seq = a_work.copy()
            print(f"    Adam main: score={sc_work_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # === Phase C: Fine Adam (25% of budget) ===
        remaining_c = deadline - time.monotonic()
        phase_c_budget = remaining_c * 0.30
        phase_c_end = time.monotonic() + phase_c_budget
        print(f"  Phase C (Adam fine): budget={phase_c_budget:.1f}s")

        c_fine = _full_convolve(a_work, a_work)
        max_c_fine = float(c_fine.max())

        fine_betas = [0.0002, 0.0001, 0.00005, 0.00003, 0.00002, 0.00001, 0.000005, 0.000003]
        n_fine = len(fine_betas)
        fine_weights = np.array([1.0, 1.0, 1.5, 1.5, 2.0, 2.0, 2.5, 3.0])
        fine_weights = fine_weights / fine_weights.sum()
        fine_iters = [max(3000, int(phase_c_budget * w * 2500)) for w in fine_weights]

        a_work, sc_fine = _adam_logspace(a_work, phase_c_end, fine_betas,
                                         lr_init=0.003, top_k_init=50, top_k_min=5,
                                         max_iter_per_phase=max(fine_iters),
                                         momentum_decay=0.3)
        a_work = a_work / a_work.sum()
        sc_fine_full = _score(a_work)
        if sc_fine_full < best_score:
            best_score = sc_fine_full
            best_seq = a_work.copy()
            print(f"    Adam fine: score={sc_fine_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # === Phase D: Coordinate descent polishing (10% of budget) ===
        remaining_d = deadline - time.monotonic()
        phase_d_budget = remaining_d * 0.25
        phase_d_end = time.monotonic() + phase_d_budget
        print(f"  Phase D (CoordDescend): budget={phase_d_budget:.1f}s")

        if phase_d_budget > 5:
            n_passes = max(1, int(phase_d_budget / 15))
            a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=n_passes,
                                                top_k_w=40, deadline=phase_d_end)
            a_cd = a_cd / a_cd.sum()
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"    CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_work = a_cd

        # === Phase E: Peak shaving (5% of budget) ===
        remaining_e = deadline - time.monotonic()
        phase_e_budget = remaining_e * 0.4
        phase_e_end = time.monotonic() + phase_e_budget
        print(f"  Phase E (PeakShave): budget={phase_e_budget:.1f}s")

        if phase_e_budget > 3:
            n_shave = max(50, int(phase_e_budget * 50))
            a_shave, sc_shave = _peak_shave(a_work, n_iter=n_shave, deadline=phase_e_end)
            a_shave = a_shave / a_shave.sum()
            sc_shave_full = _score(a_shave)
            if sc_shave_full < best_score:
                best_score = sc_shave_full
                best_seq = a_shave.copy()
                print(f"    PeakShave: score={sc_shave_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_work = a_shave

        # === Phase F: Ultra-fine Adam on best (remaining) ===
        remaining_f = deadline - time.monotonic()
        if remaining_f > 20 and best_seq is not None:
            phase_f_end = min(deadline - 5, time.monotonic() + remaining_f * 0.8)
            print(f"  Phase F (Ultra-fine): budget={phase_f_end - time.monotonic():.1f}s")

            a_uf = best_seq.copy()
            a_uf = a_uf / a_uf.sum()
            c_uf = _full_convolve(a_uf, a_uf)
            max_c_uf = float(c_uf.max())

            uf_betas = [0.00001, 0.000005, 0.000003, 0.000002]
            a_uf, sc_uf = _adam_logspace(a_uf, phase_f_end, uf_betas,
                                         lr_init=0.001, top_k_init=20, top_k_min=3,
                                         max_iter_per_phase=200000,
                                         momentum_decay=0.2)
            a_uf = a_uf / a_uf.sum()
            sc_uf_full = _score(a_uf)
            if sc_uf_full < best_score:
                best_score = sc_uf_full
                best_seq = a_uf.copy()
                print(f"    Ultra-fine: score={sc_uf_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # === Phase G: Diversity restarts (if time remains) ===
        remaining_g = deadline - time.monotonic()
        if remaining_g > 60 and best_seq is not None:
            print(f"  Phase G (Restarts): budget={remaining_g:.1f}s")
            a_base = best_seq.copy()
            a_base = a_base / a_base.sum()
            n_base = len(a_base)

            n_restarts = min(3, int(remaining_g / 30))
            per_restart = (remaining_g - 10) / n_restarts

            for ri in range(n_restarts):
                if time.monotonic() > deadline - 10:
                    break

                if ri == 0:
                    p_start = a_base + 0.015 * rng.standard_normal(n_base)
                elif ri == 1:
                    shift = rng.integers(1, max(2, n_base // 2048))
                    p_start = np.roll(a_base, shift).copy()
                else:
                    freq = rng.uniform(0.3, 2.0)
                    pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_base) / n_base)
                    p_start = a_base + pert

                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

                c_s = _full_convolve(p_start, p_start)
                max_cs = float(c_s.max())
                restart_betas = [0.001 * max_cs / max_c, 0.0003 * max_cs / max_c,
                                 0.0001 * max_cs / max_c] if max_c > 0 else [0.001, 0.0003, 0.0001]
                # Use absolute beta values scaled to current max
                max_c_ref = float(_full_convolve(p_start, p_start).max())
                restart_betas = [0.001 * max_c_ref, 0.0003 * max_c_ref, 0.0001 * max_c_ref]

                p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                                 restart_betas, lr_init=0.005, top_k_init=50,
                                                 top_k_min=5, max_iter_per_phase=100000,
                                                 momentum_decay=0.5)
                p_best = p_best / p_best.sum()
                sc_full = _score(p_best)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = p_best.copy()
                    print(f"    Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation and output
    print(f"\n{'='*60}")
    print(f"FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Total time: {time.monotonic()-t_start:.1f}s")
    print(f"{'='*60}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        else:
            # Verify with evaluate_sequence
            try:
                score_check = evaluate_sequence([float(x) for x in final])
                print(f"  Verified score: {score_check:.10f}")
                if not np.isfinite(score_check):
                    final = np.clip(best_seq, 0.0, 1000.0)
                    s = float(final.sum())
                    if s > 0.01:
                        final = final / s * 1.0
            except Exception as e:
                print(f"  Verification error: {e}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(1024) * 1.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
