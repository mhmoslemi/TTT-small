Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses hot-lag elimination with multi-subgradient, spectral domain optimization,
targeted mass redistribution, and Adam fine-tuning. Explores n=4096 and n=8192.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-14:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_lag(a, k):
    """Gradient of (a*a)[k] w.r.t. a[j] = 2*a[k-j] (0 if out of range)."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        idx = k - j
        if 0 <= idx < n:
            grad[j] = 2.0 * a[idx]
    return grad


def _gradient_lag_fast(a, k):
    """Fast gradient using array operations."""
    n = len(a)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        idx = k - np.arange(j_start, j_end + 1)
        grad[j_start:j_end + 1] = 2.0 * a[idx]
    return grad


def _multi_lag_gradient(a, c, top_k=10, temperature=0.01):
    """Compute weighted gradient from top-k lags."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    w = c[top_idx].copy()
    w = w - w.min() + 1e-10
    if temperature > 0:
        w = np.exp(w / (temperature * c.max()))
    w = w / w.sum()
    grad = np.zeros(n)
    for i in range(top_k):
        grad += w[i] * _gradient_lag_fast(a, int(top_idx[i]))
    return grad


def _hot_lag_optimize(a_init, deadline, n_iter=50000, top_k=10, eta_init=0.1):
    """Aggressive hot-lag elimination with multi-subgradient."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    eta = eta_init
    patience = 0

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        grad = _multi_lag_gradient(a, c, top_k=top_k)
        a_new = _proj_simplex(a - eta * grad)
        sc = _score_norm(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            a = a_new
            patience += 1
            if patience > 50:
                eta *= 0.95
                patience = 0

        if it % 500 == 0:
            eta = max(0.001, eta * (1.0 - 0.001 * it / n_iter))

    return best_a, best_score


def _spectral_optimize(a_init, deadline, n_iter=2000, alpha=0.1, power=2.0):
    """Spectral domain optimization: flatten the amplitude spectrum."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        phase = np.angle(A)

        target_mag = mag.mean()
        # Blend toward flat spectrum
        new_mag = mag + alpha * (target_mag - mag)
        new_mag = np.maximum(new_mag, 1e-15)

        # Apply power compression
        new_mag = new_mag**power
        avg = new_mag.mean()
        new_mag = new_mag / (new_mag**power) * mag  # keep same average
        new_mag = mag + alpha * (mag.mean() - mag)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def _spectral_flatten_v2(a_init, deadline, n_iter=1000, gamma=0.5):
    """Advanced spectral flattening with adaptive gamma."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)

        # Adaptive gamma based on flatness
        mag_std = mag.std()
        mag_mean = mag.mean()
        flatness = mag_mean / (mag_std + 1e-15)
        g = min(1.0, 0.1 * flatness)

        # Compress dynamic range
        target = mag**gamma
        target = target / target.mean() * mag.mean()

        new_mag = mag + g * (target - mag)
        new_mag = np.maximum(new_mag, 1e-15)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new

    return best_a, best_score


def _targeted_redistribution(a_init, deadline, n_iter=5000):
    """Targeted mass redistribution: move mass from hot-lag contributors to cold positions."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        # Find positions that most contribute to c[k_star]
        # Contribution of a[j] to c[k_star] is a[j] * a[k_star - j]
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        idx = k_star - j_range
        contrib[j_range] = a[j_range] * a[idx]

        # Find "cold" positions (low contribution to all lags)
        # Approximate: positions with low a value
        cold_threshold = np.percentile(a, 20)
        cold_mask = a < cold_threshold

        if not np.any(cold_mask):
            cold_mask = a < np.percentile(a, 50)

        # Move mass from hot contributors to cold positions
        hot_amount = 0.01 * contrib[contrib > 0].sum()
        hot_amount = min(hot_amount, 0.05 * a[j_range].sum())

        if hot_amount < 1e-15:
            continue

        # Reduce hot positions proportionally
        hot_reduce = contrib / (contrib.sum() + 1e-30)
        a_jr = a[j_range].copy()
        a[j_range] -= hot_amount * hot_reduce * a[j_range] / (np.abs(a[j_range]).sum() * a[j_range].max())

        # Direct: reduce the top contributors
        top_hot = j_range[np.argsort(contrib)[::-1][:5]]
        reduce_per = hot_amount / len(top_hot)
        for j in top_hot:
            a[j] = max(0, a[j] - reduce_per)

        # Add to cold positions
        cold_idx = np.where(cold_mask)[0]
        if len(cold_idx) > 0:
            add_per = a[0] * 0 + reduce_per  # same amount we removed
            # Actually add what we removed
            removed = reduce_per * len(top_hot)
            # Distribute to cold
            per_cold = removed / len(cold_idx[:20])
            for j in cold_idx[:20]:
                a[j] += per_cold

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a / s

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init=0.01, max_iter=200000):
    """Adam optimizer in log-space with beta continuation."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            # Gradient in log-space
            a_rev = a[::-1].copy()
            a_rev = np.concatenate([np.zeros(n - 1), a_rev])  # shift
            wc = _full_convolve(w, a)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 10000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter=200000):
    """FISTA with smoothed max via softmax."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        # Gradient
        wc = _full_convolve(w, y)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3)**2) + np.exp(-20 * (t - 0.7)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp":
        t = np.linspace(0, 1, n)
        phase = 20 * t * t
        a = 0.5 + 0.5 * np.cos(phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _combined_iteration(a, deadline, n_iter=10000, mode="alternate"):
    """Alternate between spectral and time-domain optimization."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        if mode == "alternate":
            if it % 3 == 0:
                # Spectral step
                N_fft = 2 * n - 1
                A = np.fft.rfft(a, N_fft)
                mag = np.abs(A)
                phase = np.angle(A)
                # Gentle flattening
                alpha = 0.05
                new_mag = mag + alpha * (mag.mean() - mag)
                new_mag = np.maximum(new_mag, 1e-15)
                A_new = new_mag * np.exp(1j * phase)
                a_new = np.fft.irfft(A_new, N_fft)[:n]
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                a = a_new
            else:
                # Gradient step
                c = _full_convolve(a, a)
                grad = _multi_lag_gradient(a, c, top_k=10)
                eta = 0.05 / (1.0 + 0.01 * it)
                a_new = _proj_simplex(a - eta * grad)
                sc = _score_norm(a_new)
                if sc < _score_norm(a):
                    a = a_new
        else:
            # Pure gradient
            c = _full_convolve(a, a)
            grad = _multi_lag_gradient(a, c, top_k=10)
            eta = 0.05 / (1.0 + 0.01 * it)
            a = _proj_simplex(a - eta * grad)

        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _frank_wolfe_lp_approx(a_init, deadline, n_iter=200):
    """
    Frank-Wolfe style optimization using approximate LP solution.
    At each step, approximately solve:
      max sum(b) s.t. (a*b)_k <= M for all k, b >= 0
    using a few iterations of the dual subgradient method.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    N_conv = 2 * n - 1

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())
        if M < 1e-15:
            break

        # Approximate LP: solve min M*sum(y) s.t. (a_correlate_y)_j >= 1, y >= 0
        # Using subgradient on dual
        y = np.ones(N_conv) / N_conv
        lr = 1.0 / (1.0 + it * 0.1)

        for sub_it in range(5):
            # Compute constraint violations: (correlation of a with y)
            # (a * y)_k for k=0..N_conv-1, but we need the correlation at each position j
            # cross(a, y)[j] = sum_i a[i] * y[i + j - (n-1)] approximately
            ay = _full_convolve(a, y[:n]) if len(y) >= n else _full_convolve(a, np.pad(y, (0, n - len(y))))
            # Actually, let's just use the constraint: for the LP primal,
            # we want to find b that maximizes sum(b) s.t. T_a * b <= M * 1
            # Dual: min M * sum(y) s.t. T_a^T * y >= 1, y >= 0
            # T_a^T * y at position j = sum_k a[k-j] * y[k] (correlation-like)

            # Simpler: just compute the "effective gradient" for b
            # The most violated constraint in the primal tells us where to add b
            # Let's use a different approach: coordinate ascent on b

            # Actually, let's use the primal directly with a simple heuristic:
            # b[j] should be large where a is small (to keep a*b small)
            # and we want to maximize sum(b)
            break  # Skip inner loop, use simpler approach

        # Simple heuristic for LP solution:
        # b[j] ∝ 1 / (sum_k |a[k-j]|)  (inverse of "influence")
        # This makes (a*b)_k approximately uniform
        influence = np.zeros(n)
        for j in range(n):
            k_start = max(0, j)
            k_end = min(n - 1, j + n - 1)
            influence[j] = np.sum(np.abs(a[np.arange(k_start, k_end + 1) - j]))
        influence = np.maximum(influence, 1e-10)

        b = 1.0 / influence
        b = b / b.sum()

        # Check if this direction helps
        c_new = _full_convolve(a + 0.1 * b, a + 0.1 * b)
        sc_test = float(c_new.max())
        sc_curr = float(c.max())

        if sc_test < sc_curr:
            # Good direction, take a step
            t = 0.1
            a_new = (1 - t) * a + t * b
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        else:
            # Try smaller step or ignore
            t = 0.01
            a_new = (1 - t) * a + t * b
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

    return best_a, best_score


def _low_rank_optimize(a_init, deadline, rank=20):
    """Optimize a low-rank approximation of the sequence."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s

    # Fit a low-rank model: a ≈ sum_j c_j * phi_j
    # Use cosine basis
    t = np.linspace(0, 1, n)
    basis = np.ones((rank, n))
    for k in range(rank):
        basis[k] = np.cos((k + 1) * np.pi * t)

    # Solve least squares: a ≈ B @ coeffs
    coeffs, _, _, _ = np.linalg.lstsq(basis, a, rcond=None)

    # Optimize coefficients
    best_coeffs = coeffs.copy()
    best_score = _score(np.maximum(basis @ coeffs, 0))

    for it in range(500):
        if time.monotonic() > deadline:
            break
        a_curr = basis @ coeffs
        a_curr = np.maximum(a_curr, 0)
        sc = _score(a_curr)
        if sc < best_score:
            best_score = sc
            best_coeffs = coeffs.copy()

        # Gradient of score w.r.t. coeffs
        c = _full_convolve(a_curr, a_curr)
        k_star = int(np.argmax(c))
        grad_a = _gradient_lag_fast(a_curr, k_star)
        grad_c = basis.T @ grad_a

        lr = 0.01 / (1 + it * 0.01)
        coeffs = coeffs - lr * grad_c
        coeffs = np.maximum(coeffs, -1)  # allow negative coeffs but bound

    a_final = basis @ best_coeffs
    a_final = np.maximum(a_final, 0)
    s = a_final.sum()
    if s > 0:
        a_final = a_final / s
    return a_final, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining hot-lag elimination, spectral optimization,
    Frank-Wolfe heuristics, and Adam fine-tuning.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: Combined spectral + gradient optimization ===
    print(f"Phase 1: Combined optimization on n={n_main}")
    phase1_end = min(deadline, t_start + 200.0)
    a_work = a0.copy()
    a_work, sc = _combined_iteration(a_work, phase1_end, n_iter=20000, mode="alternate")
    a_work = np.maximum(a_work, 0)
    a_work = a_work / a_work.sum()
    sc_full = _score(a_work)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()
        print(f"  Combined: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Hot-lag elimination ===
    print("Phase 2: Hot-lag elimination")
    phase2_end = min(deadline, t_start + 400.0)
    a_work = best_seq.copy()
    a_work, sc = _hot_lag_optimize(a_work, phase2_end, n_iter=50000, top_k=10, eta_init=0.05)
    a_work = np.maximum(a_work, 0)
    a_work = a_work / a_work.sum()
    sc_full = _score(a_work)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()
        print(f"  Hot-lag: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Spectral flattening (v2) ===
    print("Phase 3: Spectral flattening v2")
    phase3_end = min(deadline, t_start + 550.0)
    a_work = best_seq.copy()
    a_work, sc = _spectral_flatten_v2(a_work, phase3_end, n_iter=3000, gamma=0.5)
    a_work = np.maximum(a_work, 0)
    a_work = a_work / a_work.sum()
    sc_full = _score(a_work)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()
        print(f"  Spectral v2: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: FISTA with beta continuation ===
    print("Phase 4: FISTA beta-continuation")
    phase4_end = min(deadline, t_start + 750.0)
    a_work = best_seq.copy()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    betas = [0.01 * max_cw, 0.005 * max_cw, 0.002 * max_cw, 0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw]
    for b in betas:
        if time.monotonic() > phase4_end - 10:
            break
        per_t = phase4_end - time.monotonic() - 5
        a_work, sc = _fista_smoothed(a_work, b, time.monotonic() + min(25.0, per_t), 30000)
        a_work = np.maximum(a_work, 0)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={b:.8f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try n=8192 ===
    print("Phase 5: n=8192 interpolation and optimization")
    phase5_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase5_end - 30:
        n_8192 = 8192
        a_8192 = _interp_profile(best_seq, n_8192)
        sc_8192 = _score(a_8192)
        print(f"  Interpolated n=8192: score={sc_8192:.10f}")

        # Combined iteration on 8192
        a_8192, sc = _combined_iteration(a_8192, min(deadline, t_start + 850.0), n_iter=15000, mode="alternate")
        a_8192 = np.maximum(a_8192, 0)
        a_8192 = a_8192 / a_8192.sum()
        sc_full = _score(a_8192)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_8192.copy()
            print(f"  8192 combined: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # FISTA on 8192
        if time.monotonic() < phase5_end - 20:
            c_8 = _full_convolve(a_8192, a_8192)
            max_c8 = float(c_8.max())
            for b in [0.005 * max_c8, 0.001 * max_c8, 0.0005 * max_c8]:
                if time.monotonic() > phase5_end - 15:
                    break
                a_8192, sc = _fista_smoothed(a_8192, b, time.monotonic() + min(20.0, phase5_end - time.monotonic() - 10), 20000)
                a_8192 = np.maximum(a_8192, 0)
                a_8192 = a_8192 / a_8192.sum()
                sc_full = _score(a_8192)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = a_8192.copy()
                    print(f"  8192 FISTA: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Adam fine-tuning ===
    print("Phase 6: Adam fine-tuning")
    phase6_end = min(deadline, t_start + 1500.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas_fine = [0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw, 0.00005 * max_cw]
    per_t = phase6_end - time.monotonic() - 10
    a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + max(10, per_t), betas_fine, lr_init=0.008, max_iter=300000)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Multi-restart with diverse strategies ===
    print("Phase 7: Multi-restart")
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end - 20:
        a_work = best_seq.copy()
        n_best = len(a_work)
        n_restarts = 5
        per_restart = (phase7_end - time.monotonic() - 15) / n_restarts

        strategies = ["perturb_small", "perturb_med", "spectral_perturb", "structure_mix", "hot_lag_perturb"]

        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            per_t_r = max(5.0, per_restart)
            strategy = strategies[i % len(strategies)]

            if strategy == "perturb_small":
                p_start = _perturb(a_work, 0.01 + 0.005 * i, rng)
            elif strategy == "perturb_med":
                p_start = _perturb(a_work, 0.03, rng)
            elif strategy == "spectral_perturb":
                p_tmp = _perturb(a_work, 0.02, rng)
                p_start, _ = _spectral_flatten_v2(p_tmp, time.monotonic() + min(5, per_t_r * 0.3), n_iter=200, gamma=0.3)
            elif strategy == "structure_mix":
                p_struct = _make_profile(n_best, ["cosine", "bimodal", "chirp"][i % 3])
                mix = 0.2 + 0.1 * i
                p_start = mix * p_struct + (1 - mix) * a_work
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:  # hot_lag_perturb
                p_start = a_work.copy()
                c_tmp = _full_convolve(p_start, p_start)
                k_star = int(np.argmax(c_tmp))
                # Perturb the positions that contribute most to the peak
                j_start = max(0, k_star - n_best + 1)
                j_end = min(k_star, n_best - 1)
                j_range = np.arange(j_start, j_end + 1)
                contrib = p_start[j_range] * p_start[k_star - j_range]
                top_contrib = j_range[np.argsort(contrib)[::-1][:20]]
                p_start[top_contrib] *= (1 - 0.05 * rng.random(20))
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()

            # Optimize the perturbed start
            p_opt, _ = _combined_iteration(p_start, time.monotonic() + min(per_t_r * 0.5, 30.0), n_iter=5000)
            p_opt = np.maximum(p_opt, 0)
            p_opt = p_opt / p_opt.sum()

            if time.monotonic() < phase7_end - 10:
                c_r = _full_convolve(p_opt, p_opt)
                max_cr = float(c_r.max())
                betas_r = [0.001 * max_cr, 0.0005 * max_cr]
                p_opt, _ = _adam_logspace(p_opt, time.monotonic() + max(5, phase7_end - time.monotonic() - 8), betas_r, lr_init=0.005, max_iter=50000)
                sc_r_full = _score(p_opt)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_seq = p_opt.copy()
                    print(f"  Restart {i} ({strategy}): score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Ultra-fine polish ===
    print("Phase 8: Ultra-fine polish")
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    if time.monotonic() < deadline - 20:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = [0.00002 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 15
        a_pol, _ = _adam_logspace(a_work, time.monotonic() + max(5, per_t), betas_ult, lr_init=0.002, max_iter=200000)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_final = float(final.sum())
        if s_final < 0.01:
            final = _make_profile(256, "uniform")

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
