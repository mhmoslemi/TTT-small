"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses diverse initializations (U-shaped, cosine, parent), multi-scale exploration
(n=4096, 8192), Adam with soft/active gradients, and coordinate-descent local search.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=8):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, active_k=5, use_active_from_frac=0.3,
                   eval_interval=200, clip_norm=0.5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if frac > use_active_from_frac:
            k_eff = max(3, active_k - int(3 * frac))
            g = _compute_active_gradient(a, c, k_eff)
        else:
            g = _compute_soft_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-18))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _local_search(a_init, deadline, n_passes_max=50):
    n = len(a_init)
    a = a_init.copy()
    best_score = _score_norm(a)
    best_a = a.copy()
    c = _fft_convolve(a, a)
    max_c = float(c.max())
    target = 0.85 * max_c

    for _pass in range(n_passes_max):
        if time.monotonic() > deadline:
            break
        improved = False
        step = 0.01
        for iteration in range(200):
            if time.monotonic() > deadline:
                break
            c = _fft_convolve(a, a)
            max_c = float(c.mean())
            peak_indices = np.where(c > target)[0]
            if len(peak_indices) == 0:
                break
            peak_idx = peak_indices[np.argmax(c[peak_indices])]
            # Find which positions contribute most to this peak
            j_start = max(0, peak_idx - n + 1)
            j_end = min(n - 1, peak_idx)
            if j_start > j_end:
                break
            contrib = a[j_start:j_end + 1] * a[peak_idx - j_start:peak_idx - j_end][::-1]
            # Find position with max contribution
            max_local_idx = np.argmax(contrib)
            src = j_start + max_local_idx
            if a[src] < 1e-8:
                break
            # Find position with min contribution to all peak lags
            min_contrib = np.inf
            dst = src
            for pos in range(0, n, max(1, n // 200)):
                if pos == src:
                    continue
                total_contrib = 0.0
                for pk in peak_indices[:5]:
                    j_s = max(0, pk - n + 1)
                    j_e = min(n - 1, pk)
                    if j_s <= pos <= j_e:
                        partner = pk - pos
                        if 0 <= partner < n:
                            total_contrib += a[pos] * a[partner]
                    else:
                        continue
                if total_contrib < min_contrib:
                    min_contrib = total_contrib
                    dst = pos
            if dst == src:
                step *= 0.5
                continue
            # Try moving mass
            delta = step * a[src]
            if delta > a[src] * 0.5:
                delta = a[src] * 0.3
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            sc_new = _score_norm(a_new)
            if sc_new < best_score - 1e-12:
                a = a_new
                best_score = sc_new
                best_a = a.copy()
                improved = True
            else:
                step *= 0.7
        if not improved:
            break
    return best_a, best_score


def _make_profile(n, profile_type, rng):
    if profile_type == 'uniform':
        return np.ones(n) / n
    elif profile_type == 'u_shape':
        x = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + np.abs(x - 0.5) * 2) ** 2
        return a / a.sum()
    elif profile_type == 'v_shape':
        x = np.linspace(0, 1, n)
        a = np.abs(x - 0.5) ** 0.5 + 0.05
        return a / a.sum()
    elif profile_type == 'cosine_bell':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif profile_type == 'gaussian':
        x = np.linspace(0, 1, n)
        sigma = 0.2 + 0.1 * rng.random()
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        return a / a.sum()
    elif profile_type == 'multi_bump':
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        nb = int(2 + 3 * rng.random())
        for _ in range(nb):
            center = rng.random()
            width = 0.05 + 0.15 * rng.random()
            amp = 0.5 + rng.random()
            a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif profile_type == 'sin_power':
        x = np.linspace(0, 1, n)
        p = 1.0 + 2.0 * rng.random()
        a = np.sin(np.pi * x) ** p
        a = np.maximum(a, 1e-8)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer with diverse initialization, multi-scale exploration,
    Adam with adaptive active-set gradients, and local search polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
        else:
            parent = None
    except Exception:
        parent = None

    best_a = None
    best_score = np.inf

    if parent is not None:
        best_a = parent.copy()
        best_score = _score(parent)
        print(f"Parent loaded: n={len(parent)}, score={best_score:.10f}")

    # Phase 1: Quick screening of diverse (n, profile) combinations (0-400s)
    phase1_end = min(deadline, t_start + 400.0)
    candidates = []

    n_values = [4096, 8192]
    profiles = ['u_shape', 'v_shape', 'cosine_bell', 'gaussian', 'multi_bump', 'sin_power', 'uniform']

    if parent is not None:
        # Interpolate parent to different lengths
        for n in n_values:
            a = _interp_to_length(parent, n)
            candidates.append((n, a, 'parent_interp'))

    for n in n_values:
        for prof in profiles:
            a = _make_profile(n, prof, rng)
            candidates.append((n, a, prof))
            # Also try a perturbed version
            a_pert = a * (1.0 + 0.1 * rng.standard_normal(n))
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            candidates.append((n, a_pert, prof + '_pert'))

    # Quick screen: run brief Adam on each candidate
    screen_per = max(5, (phase1_end - t_start - 10) / len(candidates))
    print(f"Phase 1: screening {len(candidates)} candidates, ~{screen_per:.1f}s each")

    for idx, (n, a_init, name) in enumerate(candidates):
        if time.monotonic() > phase1_end - 20:
            break
        t_end = min(phase1_end - 5, time.monotonic() + screen_per)
        c = _fft_convolve(a_init, a_init)
        max_c = float(c.max())
        a_opt, sc_opt = _adam_optimize(
            a_init, t_end,
            T_start=0.01 * max_c, T_end=0.001 * max_c,
            lr_start=0.05, lr_end=0.01,
            max_iter=50000, active_k=5,
            use_active_from_frac=0.5, eval_interval=100, clip_norm=0.5
        )
        full_sc = _score(a_opt)
        if np.isfinite(full_sc):
            print(f"  [{idx:2d}] n={n:5d} {name:20s} -> score={full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()
            candidates[idx] = (n, a_opt, name)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Thorough optimization of top candidate (400s - 1000s)
    phase2_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_end and best_a is not None:
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        print(f"Phase 2: optimizing n={n_best} for {phase2_end - time.monotonic():.0f}s")

        # Adam soft phase
        t_mid = (time.monotonic() + phase2_end) / 2
        a_opt1, sc1 = _adam_optimize(
            best_a, t_mid - 5,
            T_start=0.005 * max_c, T_end=0.0002 * max_c,
            lr_start=0.04, lr_end=0.005,
            max_iter=300000, active_k=8,
            use_active_from_frac=0.7, eval_interval=150, clip_norm=0.4
        )
        full_sc1 = _score(a_opt1)
        print(f"  Adam-soft: score={full_sc1:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc1) and full_sc1 < best_score:
            best_score = full_sc1
            best_a = a_opt1.copy()

        # Adam active phase
        c_best2 = _fft_convolve(best_a, best_a)
        max_c2 = float(c_best2.max())
        a_opt2, sc2 = _adam_optimize(
            best_a, phase2_end - 5,
            T_start=0.001 * max_c2, T_end=0.00003 * max_c2,
            lr_start=0.02, lr_end=0.002,
            max_iter=300000, active_k=5,
            use_active_from_frac=0.0, eval_interval=150, clip_norm=0.3
        )
        full_sc2 = _score(a_opt2)
        print(f"  Adam-active: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # Phase 3: Multi-scale exploration (1000s - 1300s)
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end and best_a is not None:
        n_cur = len(best_a)
        # Try larger n
        for n_try in [max(n_cur * 2, 8192), 16384]:
            if time.monotonic() > phase3_end - 60:
                break
            a_large = _interp_to_length(best_a, n_try)
            c_l = _fft_convolve(a_large, a_large)
            max_cl = float(c_l.max())
            t_end_l = min(phase3_end - 20, time.monotonic() + 120)
            a_lo, sc_lo = _adam_optimize(
                a_large, t_end_l,
                T_start=0.003 * max_cl, T_end=0.0001 * max_cl,
                lr_start=0.03, lr_end=0.003,
                max_iter=100000, active_k=5,
                use_active_from_frac=0.4, eval_interval=200, clip_norm=0.4
            )
            full_sc_l = _score(a_lo)
            print(f"  n={n_try} explore: score={full_sc_l:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_l) and full_sc_l < best_score:
                best_score = full_sc_l
                best_a = a_lo.copy()

        # Try smaller n (sometimes better due to fewer constraints)
        if n_cur > 2048:
            a_small = _interp_to_length(best_a, 2048)
            c_s = _fft_convolve(a_small, a_small)
            max_cs = float(c_s.max())
            t_end_s = min(phase3_end - 10, time.monotonic() + 80)
            a_so, sc_so = _adam_optimize(
                a_small, t_end_s,
                T_start=0.003 * max_cs, T_end=0.0001 * max_cs,
                lr_start=0.03, lr_end=0.003,
                max_iter=80000, active_k=5,
                use_active_from_frac=0.4, eval_interval=150, clip_norm=0.4
            )
            full_sc_s = _score(a_so)
            print(f"  n=2048 explore: score={full_sc_s:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_s) and full_sc_s < best_score:
                best_score = full_sc_s
                best_a = a_so.copy()

    # Phase 4: Local search / coordinate descent (1300s - 1600s)
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end and best_a is not None:
        n_ls = len(best_a)
        print(f"Phase 4: local search on n={n_ls}")

        # Coordinate descent: iteratively adjust individual elements
        a_cd = best_a.copy()
        c_cd = _fft_convolve(a_cd, a_cd)
        max_cd = float(c_cd.max())
        target_cd = 0.92 * max_cd
        lr_cd = 0.02
        n_cd_steps = 3000
        steps_done = 0

        for step_i in range(n_cd_steps):
            if time.monotonic() > phase4_end - 10:
                break
            if step_i % 100 == 0:
                c_cd = _fft_convolve(a_cd, a_cd)
                max_cd = float(c_cd.max())
                target_cd = 0.92 * max_cd
                sc_cd = _score(a_cd)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_a = a_cd.copy()

            # Find peak position
            peak_idx = np.argmax(c_cd)
            # Get contributing positions
            j_start = max(0, peak_idx - n_ls + 1)
            j_end = min(n_ls - 1, peak_idx)
            if j_start > j_end:
                break

            # Compute gradient at peak
            g_peak = np.zeros(n_ls)
            contrib_positions = range(j_start, j_end + 1)
            for j in contrib_positions:
                partner = peak_idx - j
                if 0 <= partner < n_ls:
                    if j == partner:
                        g_peak[j] += a_cd[j]
                    else:
                        g_peak[j] += 2.0 * a_cd[partner]

            # Find position where reducing helps most (high contribution to peak, low elsewhere)
            # Sample a subset for efficiency
            sample_positions = list(range(0, n_ls, max(1, n_ls // 2000)))
            best_j = None
            best_gain = 0

            # Also check adding to low-contribution positions
            for j in sample_positions:
                if j < j_start or j > j_end:
                    continue
                # Try reducing a[j]
                if a_cd[j] < 1e-8:
                    continue
                # Try increasing (add mass to low-peak positions)
                # and decreasing (remove from high-peak positions)
                pass

            # Simpler: use gradient to do a step
            # Find the position with highest g_peak / a ratio (most "expensive")
            valid = g_peak > 0
            if np.any(valid):
                ratios = np.where(valid, g_peak / (a_cd + 1e-15), 0)
                # Move mass from high-ratio to low-ratio positions
                src = np.argmax(ratios)
                # Find destination: position with lowest total gradient
                g_total = np.zeros(n_ls)
                # Approximate: use a few lags
                top_lags = np.argsort(c_cd)[::-1][:5]
                for kl in top_lags:
                    js = max(0, kl - n_ls + 1)
                    je = min(n_ls - 1, kl)
                    if js <= je:
                        for j in range(js, je + 1):
                            p = kl - j
                            if 0 <= p < n_ls:
                                g_total[j] += a_cd[p]
                dst = np.argmin(g_total)
                if src != dst and a_cd[src] > 1e-8:
                    delta = lr_cd * a_cd[src]
                    a_cd[src] -= delta
                    a_cd[dst] += delta
                else:
                    # Try a different source
                    src2 = np.argmax(ratios * (np.arange(n_ls) != src))
                    if src2 != dst and a_cd[src2] > 1e-8:
                        delta = lr_cd * a_cd[src2]
                        a_cd[src2] -= delta
                        a_cd[dst] += delta

            steps_done += 1
            # Adaptive learning rate
            if step_i % 500 == 499:
                c_check = _fft_convolve(a_cd, a_cd)
                if c_check.max() > max_cd * 1.01:
                    lr_cd *= 0.5
                    a_cd = best_a.copy()
                else:
                    lr_cd = min(lr_cd * 1.1, 0.05)

        # Final check
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
        print(f"  Local search done: {steps_done} steps, score={_score(best_a):.10f}")

    # Phase 5: Perturbation restarts (1600s - 1750s)
    phase5_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase5_end and best_a is not None:
        n_p = len(best_a)
        for attempt in range(4):
            if time.monotonic() > phase5_end - 30:
                break
            sigma = 0.01 + 0.01 * attempt
            z = np.log(np.maximum(best_a, 1e-18))
            z = z + sigma * rng.standard_normal(n_p)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            t_end_p = min(phase5_end - 10, time.monotonic() + 80)
            a_po, sc_po = _adam_optimize(
                a_pert, t_end_p,
                T_start=0.001 * max_cp, T_end=0.00005 * max_cp,
                lr_start=0.02, lr_end=0.002,
                max_iter=60000, active_k=4,
                use_active_from_frac=0.3, eval_interval=100, clip_norm=0.3
            )
            full_sc_p = _score(a_po)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc_p:.10f}")
            if np.isfinite(full_sc_p) and full_sc_p < best_score:
                best_score = full_sc_p
                best_a = a_po.copy()

    # Phase 6: Final ultra-fine polish (1750s - deadline)
    if time.monotonic() < deadline - 10 and best_a is not None:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_optimize(
            best_a, deadline - 5,
            T_start=0.0001 * max_cf, T_end=0.000005 * max_cf,
            lr_start=0.003, lr_end=0.0002,
            max_iter=200000, active_k=3,
            use_active_from_frac=0.0, eval_interval=100, clip_norm=0.2
        )
        full_sc_final = _score(a_final)
        print(f"  Final polish: score={full_sc_final:.10f}")
        if np.isfinite(full_sc_final) and full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()

    # Ensure valid output
    final = np.maximum(best_a, 0.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final_check = _score(final)
    print(f"\nFINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
