"""
Improved multi-phase optimizer combining FISTA with adaptive smoothing,
active-set deflation targeting top convolution peaks, efficient coordinate
descent with gradient-weighted steps, and multi-resolution exploration.
Targets sub-1.5027 score.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(c)."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_weights(w, f):
    """Compute gradient of weighted sum of convolution values."""
    n = len(f)
    f_rev = f[::-1]
    wf = _convolve(w, f_rev)
    return 2.0 * wf[n - 1:2 * n - 1]


def _softmax_weights(c, beta):
    """Compute softmax weights over convolution values."""
    cm = float(c.max())
    z = (c - cm) / beta
    np.clip(z, -600, 600, out=z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _fista_optimize(a_init, beta_schedule, deadline, max_iter_per_beta=50000):
    """FISTA with momentum on the smoothed max of auto-convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    stall_count = 0
    stall_threshold = best_sc * 0.999

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        L = 4.0 / max(beta, 1e-15)
        step = 1.0 / L
        t_val = 1.0
        a_prev = a_curr.copy()
        
        per_deadline = min(deadline, time.monotonic() + max_iter_per_beta * 0.01)
        for it in range(max_iter_per_beta):
            if time.monotonic() > per_deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _convolve(y, y)
            w = _softmax_weights(c_y, beta)
            grad = _grad_from_weights(w, y)
            
            a_next = _proj_simplex(y - step * grad)

            a_prev = a_curr
            a_curr = a_next
            t_val = t_next

            if it % 200 == 0:
                sc = _score_norm(a_curr)
                if sc < best_sc:
                    if sc < stall_threshold:
                        stall_threshold = sc * 0.999
                        stall_count = 0
                    else:
                        stall_count += 1
                    best_sc = sc
                    best_a = a_curr.copy()
                else:
                    stall_count += 1

                if stall_count > 30:
                    t_val = 1.0
                    a_prev = a_curr.copy()
                    stall_count = 0

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _deflation_step(f, n_peaks=10, n_iter=50, deadline=None):
    """Actively reduce top convolution peaks by adjusting responsible coefficients."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_sc = _score_norm(f)
    rng = np.random.default_rng(999)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _convolve(f, f)
        cm = float(c.max())
        
        # Find top peak positions
        top_idx = np.argsort(c)[::-1][:n_peaks]
        
        # For each peak, find the most responsible coefficient
        reduced = False
        for k in top_idx:
            if reduced:
                break
            # c[k] = sum_j f[j] * f[k-j]
            # Find j that maximizes f[j]*f[k-j]
            valid_j = np.arange(max(0, k - n + 1), min(k + 1, n))
            if len(valid_j) == 0:
                continue
            partners = k - valid_j
            partners = partners[(partners >= 0) & (partners < n)]
            valid_j = valid_j[(partners >= 0) & (partners < n)]
            if len(valid_j) == 0:
                continue
            
            products = f[valid_j] * f[partners]
            max_j = valid_j[np.argmax(products)]
            max_partner = k - max_j
            
            # Reduce the larger of f[max_j] and f[max_partner]
            if f[max_j] >= f[max_partner]:
                target = max_j
                other = max_partner
            else:
                target = max_partner
                other = max_j
            
            old_val = f[target]
            # Reduce by a fraction
            new_val = old_val * 0.95
            f[target] = new_val
            s = f.sum()
            if s < 1e-30:
                f[target] = old_val
                continue
            f = f / s
            
            # Check improvement
            c_new = _convolve(f, f)
            sc = float(c_new.max())
            if sc < cm - 1e-15:
                reduced = True
                if _score_norm(f) < best_sc:
                    best_sc = _score_norm(f)
                    best_f = f.copy()
            else:
                f = best_f.copy()
                # Try a smaller reduction
                f[target] = best_f[target] * 0.99
                f = f / f.sum()
                c_check = _convolve(f, f)
                if float(c_check.max()) < cm - 1e-15:
                    if _score_norm(f) < best_sc:
                        best_sc = _score_norm(f)
                        best_f = f.copy()
                else:
                    f = best_f.copy()

    return best_f, best_sc


def _coord_descent_improved(f, deadline, max_rounds=200):
    """Coordinate descent with gradient-weighted multiplicative updates."""
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_sc = _score_norm(f)
    rng = np.random.default_rng(777)
    
    # Precompute reversal for gradient
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(f, f)
        cm = float(c.max())
        w = _softmax_weights(c, cm * 0.001)
        grad = _grad_from_weights(w, f)
        
        # Compute "ideal" direction: reduce where grad is high, increase where low
        grad_max = float(grad.max())
        grad_min = float(grad.min())
        if grad_max - grad_min < 1e-15:
            break
        
        # Multiplicative update
        lr = 0.1
        f_new = f * np.exp(-lr * (grad - grad.mean()) / (grad_max + 1e-30))
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        
        sc_new = _score_norm(f_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller step
            f_try = f * np.exp(-lr * 0.3 * (grad - grad.mean()) / (grad_max + 1e-30))
            f_try = np.maximum(f_try, 1e-30)
            f_try = f_try / f_try.sum()
            sc_try = _score_norm(f_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_f = f_try.copy()
                f = f_try
            else:
                # Coordinate-wise: try individual updates
                improved = False
                if n <= 8192:
                    indices = np.arange(n)
                else:
                    indices = rng.choice(n, size=2048, replace=False)
                rng.shuffle(indices)
                
                for j in indices[:min(512, len(indices))]:
                    if time.monotonic() > deadline:
                        break
                    old = f[j]
                    for factor in [0.9, 0.95, 1.05, 1.1]:
                        f[j] = old * factor
                        s = f.sum()
                        if s < 1e-30:
                            f[j] = old
                            continue
                        f[j] = f[j]  # already set
                        sc = _score_norm(f / s)
                        if sc < best_sc - 1e-14:
                            best_sc = sc
                            best_f = (f / s).copy()
                            f = best_f.copy()
                            improved = True
                            break
                        f[j] = old
                
                if not improved and r > 50:
                    break
    
    return best_f, best_sc


def _adam_logspace_fine(a_init, deadline, beta_schedule, lr_init=0.005, max_iter=500000):
    """Adam optimizer in log-space with softmax smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a, a)
            cm = float(c.max())
            
            # Compute gradient
            z_shift = (c - cm) / beta
            np.clip(z_shift, -600, 600, out=z_shift)
            e = np.exp(z_shift)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.2:
                update = update * (0.2 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb_structure(a, rng, mode=0):
    """Apply structural perturbation (not just noise)."""
    n = len(a)
    if mode == 0:
        # Add low-frequency perturbation
        freqs = np.arange(1, 5)
        phase = rng.random(len(freqs)) * 2 * np.pi
        pert = np.zeros(n)
        for fi, ph in zip(freqs, phase):
            pert += np.sin(2 * np.pi * fi * np.arange(n) / n + ph)
        z = np.log(np.maximum(a, 1e-20)) + 0.02 * pert
        z = z - z.mean()
        return np.exp(z) / np.exp(z).sum()
    elif mode == 1:
        # Block shuffle
        nb = n // 16
        blocks = [a[i*nb:(i+1)*nb] for i in range(16)]
        rng.shuffle(blocks)
        new_a = np.concatenate(blocks)
        return new_a / new_a.sum()
    else:
        # Tapered perturbation
        taper = np.linspace(0.5, 1.5, n) ** 2
        z = np.log(np.maximum(a, 1e-20))
        z = z + 0.03 * rng.standard_normal(n) * taper
        z = z - z.mean()
        return np.exp(z) / np.exp(z).sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: FISTA with geometric beta schedule (coarse to fine)
    Phase 2: Active-set deflation (target top peaks directly)
    Phase 3: Fine FISTA (ultra-small beta)
    Phase 4: Adam log-space polish
    Phase 5: Coordinate descent
    Phase 6: Multi-restart with structural perturbations
    Phase 7: n=8192 exploration
    Phase 8: Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")

    # === Phase 1: FISTA coarse-to-fine (400s) ===
    print("Phase 1: FISTA coarse-to-fine")
    phase1_end = min(deadline, t_start + 400.0)
    a_curr = a0.copy()

    # Geometric beta schedule from large to very small
    beta_start = 0.1 * max_c0
    beta_end = 1e-14 * max_c0
    n_betas = 20
    betas = beta_start * (beta_end / beta_start) ** (np.arange(n_betas) / (n_betas - 1))
    
    # Allocate time: more time for later (smaller) betas
    for i, beta in enumerate(betas):
        if time.monotonic() > phase1_end - 5:
            break
        # Allocate proportionally more time to later betas
        time_alloc = (phase1_end - time.monotonic()) / (len(betas) - i) * 0.8
        time_alloc = max(3.0, min(60.0, time_alloc))
        max_it = max(5000, int(time_alloc * 2000))
        
        a_curr, sc = _fista_optimize(a_curr, [beta], time.monotonic() + time_alloc, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: score={best_score:.10f}")

    # === Phase 2: Active-set deflation (250s) ===
    print("Phase 2: Deflation")
    phase2_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase2_end - 10:
        a_def = best_seq.copy()
        a_def, sc_def = _deflation_step(a_def, n_peaks=8, n_iter=200, 
                                         deadline=phase2_end - 10)
        sc_def_full = _score(a_def)
        print(f"  Deflation: score={sc_def_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_def_full < best_score:
            best_score = sc_def_full
            best_seq = a_def.copy()

    # === Phase 3: Fine FISTA (300s) ===
    print("Phase 3: Fine FISTA")
    phase3_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase3_end - 10:
        a_curr = best_seq.copy()
        c_curr = _convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())
        
        betas_fine = max_cc * np.array([5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6])
        betas_fine = betas_fine[betas_fine > 1e-16 * max_cc]
        
        for beta in betas_fine:
            if time.monotonic() > phase3_end - 5:
                break
            time_alloc = min(40.0, phase3_end - time.monotonic() - 3)
            if time_alloc < 3:
                break
            max_it = max(3000, int(time_alloc * 1500))
            a_curr, sc = _fista_optimize(a_curr, [beta], time.monotonic() + time_alloc, max_it)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  Fine FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 3: score={best_score:.10f}")

    # === Phase 4: Adam log-space (300s) ===
    print("Phase 4: Adam log-space")
    phase4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase4_end - 20:
        a_curr = best_seq.copy()
        c_curr = _convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())
        betas_adam = max_cc * np.array([1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5])
        per_t = phase4_end - time.monotonic() - 10
        if per_t > 30:
            a_opt, sc_opt = _adam_logspace_fine(a_curr, time.monotonic() + per_t,
                                                  betas_adam, lr_init=0.008, max_iter=300000)
            sc_opt_full = _score(a_opt)
            print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_opt_full < best_score:
                best_score = sc_opt_full
                best_seq = a_opt.copy()

    # === Phase 5: Coordinate descent (200s) ===
    print("Phase 5: Coordinate descent")
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end - 10:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(phase5_end - 5, time.monotonic() + 190)
        a_cd, sc_cd = _coord_descent_improved(a_cd, cd_deadline, max_rounds=500)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 6: Multi-restart (200s) ===
    print("Phase 6: Multi-restart")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end - 30:
        n_restarts = 5
        per_restart = (phase6_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            p_start = _perturb_structure(best_seq, rng, mode=i % 3)
            
            # Quick FISTA
            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            b1 = 0.0005 * max_cs
            b2 = 0.0001 * max_cs
            p_opt, _ = _fista_optimize(p_start, [b1, b2], 
                                         time.monotonic() + per_restart * 0.6, 30000)
            p_opt = p_opt / p_opt.sum()
            sc = _score(p_opt)
            if sc < best_score:
                best_score = sc
                best_seq = p_opt.copy()
                print(f"  Restart {i}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: n=8192 exploration (150s) ===
    print("Phase 7: n=8192 exploration")
    phase7_end = min(deadline, t_start + 1800.0 - 30)
    if time.monotonic() < phase7_end - 30 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase7_end - 20:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            betas_big = [0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb]
            per_t = min(50.0, phase7_end - time.monotonic() - 10)
            if per_t > 15:
                a_big_opt, _ = _fista_optimize(a_big, betas_big,
                                                time.monotonic() + per_t, 30000)
                a_big_opt = a_big_opt / a_big_opt.sum()
                sc_big_opt = _score(a_big_opt)
                if sc_big_opt < best_score:
                    best_score = sc_big_opt
                    best_seq = a_big_opt.copy()
                    print(f"  Optimized n=8192: score={sc_big_opt:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-fine polish (remaining time) ===
    print("Phase 8: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        
        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        
        # Ultra-fine FISTA
        betas_ult = [max_cp * 1e-6, max_cp * 5e-7, max_cp * 2e-7]
        betas_ult = [b for b in betas_ult if b > 1e-18]
        per_t = min(60.0, deadline - time.monotonic() - 5)
        if per_t > 10:
            a_pol, _ = _fista_optimize(a_pol, betas_ult,
                                        time.monotonic() + per_t, 20000)
            a_pol = a_pol / a_pol.sum()
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Final FISTA: score={sc_pol:.10f}")
        
        # Quick deflation
        if time.monotonic() < deadline - 5:
            a_pol, sc_d = _deflation_step(a_pol, n_peaks=5, n_iter=20,
                                           deadline=deadline - 2)
            sc_d_full = _score(a_pol)
            if sc_d_full < best_score:
                best_score = sc_d_full
                best_seq = a_pol.copy()
                print(f"  Final deflation: score={sc_d_full:.10f}")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    return [float(x) for x in fallback]
