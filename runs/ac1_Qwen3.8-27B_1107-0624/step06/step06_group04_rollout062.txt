        if time.monotonic() > deadline:
            break

        y = a_curr + momentum * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _softmax_weights(c_y, beta)
        grad = _compute_grad(y, w)

        best_a = a_curr.copy()

def _adam_logspace(a_init, deadline, betas, lr_init, max_iter, lr_decay=0.95, decay_every=5000):
    """Adam optimizer in log-space with beta continuation."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 1e-30:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-20))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_sc = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()

            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)
            
            # Gradient in log-space: d(score)/d(z_j) = a_j * (d(score)/d(a_j))
            grad_a = _compute_grad(a, w)
            grad_z = a * grad_a
            # Center the gradient (since scaling z doesn't change a)
            grad_z = grad_z - a * np.dot(grad_z, np.ones(n))  # project out the null direction
            
            # Actually, for softmax parameterization, the gradient of f(a(z)) w.r.t. z is:
            # a * (grad_f - mean(a * grad_f)) where grad_f = d f / d a
            # This is the score function gradient.
            # But since we're using log-param (z = log(a)), not softmax, the gradient is:
            # d f / d z_j = a_j * d f / d a_j
            # and we don't need to center it (the constraint sum(a)=1 is handled by re-normalization)
            
            grad_z = a * grad_a

            m_buf = b1 * m_buf + (1 - b1) * grad_z
            v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            # Gradient clipping
            unorm = np.linalg.norm(update)
            if unorm > 1.0:
                update = update * (1.0 / unorm)

            z = z - update
            z = np.clip(z, -100, 100)
            z = z - z.mean()  # Renormalization in log-space

            if t % 50 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_chk.copy()

            if t % decay_every == 0:
                lr = max(0.001, lr * lr_decay)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc

    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-20))
    z = z + sigma * rng.standard_normal(a.shape[0])
    z = z - z.mean()
    out = np.exp(z - z.max())
    return out / out.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent loaded: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 1: Nesterov + softmax with very gradual beta continuation ===
    print("Phase 1: Nesterov softmax continuation")
    phase1_end = min(deadline, t_start + 350.0)
    c_work = _full_convolve(a_work, a_work)
    max_c = float(c_work.max())

    # Very gradual: from 0.02*max_c down to 0.00001*max_c with factor 0.7
    betas = []
    b = 0.02 * max_c
    while b > 0.00001 * max_c:
        betas.append(b)
        b *= 0.7
    print(f"  Beta levels: {len(betas)}, from {betas[0]:.8f} to {betas[-1]:.10f}")

    a_curr = a_work.copy()
    for bi, beta in enumerate(betas):
        if time.monotonic() > phase1_end - 5:
            break
        remaining = phase1_end - time.monotonic() - 3
        if remaining < 5:
            break
        max_it = max(2000, int(remaining * 5000))
        a_curr, sc = _nesterov_smoothed(a_curr, beta, time.monotonic() + min(remaining, 25.0), max_it, step_scale=0.8)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  Beta {bi+1}/{len(betas)} ({beta:.8f}): score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"  Phase 1 done: best={best_score:.10f}")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 2: Adam log-space with fine beta ===
    print("Phase 2: Adam log-space fine-tuning")
    phase2_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase2_end - 30:
        c_work = _full_convolve(a_work, a_work)
        max_c2 = float(c_work.max())
        betas_fine = []
        b = 0.0001 * max_c2
        while b > 0.0000005 * max_c2:
            betas_fine.append(b)
            b *= 0.65
        if len(betas_fine) > 15:
            betas_fine = betas_fine[:15]
        print(f"  Adam betas: {len(betas_fine)} levels")

        per_t = phase2_end - time.monotonic() - 10
        a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + per_t, betas_fine, lr_init=0.008, max_iter=500000)
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 3: Nesterov with top-k subgradient ===
    print("Phase 3: Top-k Nesterov")
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end - 20:
        for k in [20, 10, 5, 3, 2, 1]:
            if time.monotonic() > phase3_end - 15:
                break
            remaining = phase3_end - time.monotonic() - 10
            if remaining < 10:
                break
            a_opt, sc_opt = _nesterov_topk(a_work, k, time.monotonic() + remaining, max_iter=50000, step_init=0.003)
            sc_full = _score(a_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_opt.copy()
                a_work = a_opt.copy() / a_opt.sum()
                print(f"  Top-{k}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 4: Subgradient polish with adaptive step ===
    print("Phase 4: Subgradient polish")
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end - 10:
        a_opt, sc_opt = _subgrad_polish(a_work, k_list=[3, 2, 1], deadline=phase4_end, steps_per_k=3000)
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Subgrad polish: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 5: Multi-restart with perturbation ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase5_end - 30:
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            sigma = 0.005 + 0.003 * i
            a_pert = _perturb_log(a_work, sigma, rng)
            # Quick Nesterov with moderate beta
            c_p = _full_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            beta_r = 0.0001 * max_cp
            remaining = min(per_restart, phase5_end - time.monotonic() - 10)
            if remaining < 5:
                break
            a_r, _ = _nesterov_smoothed(a_pert, beta_r, time.monotonic() + remaining * 0.4, 10000, step_scale=0.7)
            a_r = a_r / a_r.sum()
            # Quick Adam
            if time.monotonic() < phase5_end - 15:
                betas_r = [0.00005 * max_cp, 0.00002 * max_cp]
                a_r, _ = _adam_logspace(a_r, time.monotonic() + (phase5_end - time.monotonic() - 10) * 0.6, betas_r, lr_init=0.004, max_iter=50000)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                a_work = a_r.copy() / a_r.sum()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 6: Final ultra-fine polish ===
    print("Phase 6: Final polish")
    if time.monotonic() < deadline - 20:
        c_f = _full_convolve(a_work, a_work)
        max_cf = float(c_f.max())
        betas_final = [0.000005 * max_cf, 0.000002 * max_cf]
        remaining = deadline - time.monotonic() - 15
        if remaining > 10:
            a_f, _ = _adam_logspace(a_work, time.monotonic() + remaining * 0.7, betas_final, lr_init=0.002, max_iter=200000)
            sc_f = _score(a_f)
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_f.copy()
                print(f"  Final polish: score={sc_f:.10f}")

            if time.monotonic() < deadline - 15:
                a_f2, _ = _nesterov_topk(a_f, 1, time.monotonic() + (deadline - time.monotonic()) * 0.5, 20000, step_init=0.001)
                sc_f2 = _score(a_f2)
                if sc_f2 < best_score:
                    best_score = sc_f2
                    best_seq = a_f2.copy()
                    print(f"  Final top-1: score={sc_f2:.10f}")

    # Final validation
    print(f"\nAll done: best_score={best_score:.10f}, n={len(best_seq)}")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
        s = final.sum()
        final = final / s

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
