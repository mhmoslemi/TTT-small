import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Compute the full score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _active_gradient(a, c, k=1):
    """Compute subgradient of max(conv) using top-k active positions."""
    n = len(a)

    # This is conv(a, p_reversed) evaluated at appropriate positions

"""
Optimized subgradient/mirror-descent solver for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based cross-correlation for O(n log n) gradient computation, backtracking
line search, adaptive temperature annealing, multi-restart strategy, and a final
coordinate-wise polishing phase. Tries n=4096 and n=8192 for diversity.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, p):
    """Compute d[j] = 2 * sum_m a[m] * p[m+j] efficiently via FFT cross-correlation."""
    n = len(a)
    a_rev = a[::-1]
    conv = _fft_conv(a_rev, p)
    d = conv[n - 1:2 * n - 1] * 2.0
    return d


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _optimize_mirror(a_init, deadline, T_rel_start=0.01, T_rel_end=0.00005,
                     lr_init=1.0, lr_decay=0.95, noise_interval=2000,
                     noise_scale=0.3, eval_interval=50):
    """
    Mirror descent in log-space with adaptive temperature and backtracking line search.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    current_score = best_score

    t = 0
    max_iter = 5000000
    rng = np.random.default_rng(12345 + n)
    stagnation = 0
    lr = lr_init

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _fft_conv(a, a)
        c_max = float(c.max())

        if c_max < 1e-15:
            break

        # Adaptive temperature (relative to c_max)
        frac = min(1.0, t / 200000)
        T_rel = T_rel_start * (T_rel_end / T_rel_start) ** frac
        T = T_rel * c_max

        # Compute gradient probability
        if T < 1e-8 * c_max:
            # Hard active set: top 1
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        else:
            c_shift = (c - c_max) / T
            p = np.exp(c_shift)
            ps = p.sum()
            if ps < 1e-300 or not np.isfinite(ps):
                p = np.zeros(len(c))
                p[np.argmax(c)] = 1.0
            else:
                p /= ps

        d = _compute_grad(a, p)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Backtracking line search
        accepted = False
        for attempt in range(8):
            z_new = z - lr * d
            z_new = np.clip(z_new, -60.0, 60.0)
            z_new -= z_new.mean()
            a_new = _softmax(z_new)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < current_score * (1.0 - 1e-10):
                z = z_new
                current_score = sc
                accepted = True
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    stagnation = 0
                break
            lr *= 0.5

        if not accepted:
            # No improvement even with tiny step; add noise to escape
            stagnation += 1
            if stagnation > 5:
                noise = noise_scale * rng.standard_normal(n)
                z = z + noise
                z -= z.mean()
                lr = lr_init * (lr_decay ** (t // 10000))
                stagnation = 0
            else:
                lr = lr_init * (lr_decay ** (t // 10000))

        # Periodic noise injection to explore
        if t % noise_interval == 0 and t > 0:
            sc_noise = noise_scale * 0.5
            z_pert = z + sc_noise * rng.standard_normal(n)
            z_pert -= z_pert.mean()
            a_pert = _softmax(z_pert)
            sc_pert = _score_norm(a_pert)
            if sc_pert < current_score:
                z = z_pert
                current_score = sc_pert
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_a = a_pert.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_a = a_final
        best_score = sc_final

    return best_a, best_score


def _active_set_optimize(a_init, deadline, top_k=3, lr_init=0.5, eval_interval=30):
    """Focused active-set optimization for final polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    current_score = best_score
    lr = lr_init
    t = 0
    max_iter = 2000000
    stagnation = 0
    rng = np.random.default_rng(99999)

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _fft_conv(a, a)
        c_max = float(c.max())

        if c_max < 1e-15:
            break

        # Top-k active set
        if top_k <= 1:
            top_idx = np.argmax(c)
            p = np.zeros(len(c))
            p[top_idx] = 1.0
        else:
            top_idx = np.argsort(c)[::-1][:top_k]
            p = np.zeros(len(c))
            p[top_idx] = 1.0 / top_k

        d = _compute_grad(a, p)
        d_norm = np.linalg.norm(d)
        if d_norm < 1e-15:
            break
        d = d / d_norm

        # Backtracking
        accepted = False
        for attempt in range(10):
            z_new = z - lr * d
            z_new = np.clip(z_new, -60.0, 60.0)
            z_new -= z_new.mean()
            a_new = _softmax(z_new)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < current_score * (1.0 - 1e-12):
                z = z_new
                current_score = sc
                accepted = True
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    stagnation = 0
                break
            lr *= 0.5

        if not accepted:
            stagnation += 1
            if stagnation > 20:
                # Random restart in neighborhood
                noise = 0.05 * rng.standard_normal(n)
                z = z + noise
                z -= z.mean()
                lr = lr_init
                stagnation = 0

        if t % eval_interval == 0:
            lr = lr_init * max(0.1, 1.0 - 0.01 * (t // 10000))

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_a = a_final
        best_score = sc_final

    return best_a, best_score


def _coordinate_polish(a_init, deadline, passes=3):
    """
    Coordinate-wise local search: for each coordinate, try moving a small
    fraction of its mass to neighbors. Accept if score improves.
    """
    n = len(a_init)
    a = a_init.copy()
    current_score = _score_norm(a)
    best_a = a.copy()
    best_score = current_score
    rng = np.random.default_rng(777)

    for p in range(passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            if a[i] < 1e-10:
                continue
            # Try moving fraction of a[i] to neighbors
            fractions = [0.01, 0.05, 0.1, 0.2]
            neighbors = []
            if i > 0:
                neighbors.append(i - 1)
            if i < n - 1:
                neighbors.append(i + 1)
            if len(neighbors) == 0:
                continue

            for f in fractions:
                for j in neighbors:
                    a_try = a.copy()
                    amount = a_try[i] * f
                    a_try[i] -= amount
                    a_try[j] += amount
                    sc = _score_norm(a_try)
                    if sc < current_score - 1e-12:
                        a = a_try
                        current_score = sc
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                        break
                else:
                    continue
                break

    return best_a, best_score


def _structured_init(n, rng):
    """Generate structured initializations."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular (peak in middle)
    x = np.linspace(0, 1, n)
    a = np.abs(1.0 - 2.0 * x)
    a = 1.0 - a
    a = np.maximum(a, 1e-8)
    inits.append(a / a.sum())

    # Bimodal (peaks at 1/4 and 3/4)
    a = np.exp(-4.0 * (x - 0.25) ** 2 / 0.05) + np.exp(-4.0 * (x - 0.75) ** 2 / 0.05)
    a = np.maximum(a, 1e-8)
    inits.append(a / a.sum())

    # U-shaped (peaks at edges)
    a = x ** 2 + (1.0 - x) ** 2
    a = np.maximum(a, 1e-8)
    inits.append(a / a.sum())

    # Random
    a = rng.random(n) + 1e-8
    inits.append(a / a.sum())

    return inits
