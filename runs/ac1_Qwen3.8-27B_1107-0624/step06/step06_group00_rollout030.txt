"""
Central-binspired multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses central binomial coefficient sequences (theoretically optimal for infinite
sequences) as starting points, combined with Adam in z-space, temperature annealing,
active-set gradients, multi-n exploration, and perturbation restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _central_binomial(n):
    """Generate normalized central binomial coefficient sequence."""
    k = np.arange(n, dtype=np.float64)
    # log(C(2k,k)/4^k) = log(C(2k,k)) - k*log(4)
    # Use the approximation: C(2k,k)/4^k = prod_{j=1}^{k} (1 - 1/(4j))
    # Or compute directly for numerical stability
    # For large k, use the log-gamma: log(C(2k,k)) = lgamma(2k+1) - 2*lgamma(k+1)
    from math import lgamma
    log_vals = np.zeros(n)
    for i in range(n):
        if i == 0:
            log_vals[i] = 0.0
        else:
            log_vals[i] = lgamma(2*i + 1) - 2*lgamma(i + 1) - i*np.log(4.0)
    vals = np.exp(log_vals - np.max(log_vals))
    return vals / vals.sum()


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_idx = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_idx] = 1.0 / len(k_idx)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, use_active=False, active_k=5,
                   eval_interval=200, patience_limit=8, clip_norm=0.8):
    """Adam optimizer in z-space with annealing and patience-based restart."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        # Exponential annealing
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_active and T < T_start * 0.01:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_soft_gradient(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_limit and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                # Reset to best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_optimize_multi_restarts(a_init, deadline, n_restarts=3,
                                  T_scale=0.003, lr_base=0.03):
    """Run Adam with multiple restarts at different temperature scales."""
    n = len(a_init)
    c = _fft_convolve(a_init, a_init)
    max_c = float(c.max())
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        T_s = T_scale * max_c * (0.5 ** r)
        T_e = T_s * 0.02
        lr_s = lr_base * (0.7 ** r)
        lr_e = lr_s * 0.15
        a_opt, sc = _adam_optimize(best_a, deadline, T_s, T_e, lr_s, lr_e,
                                   max_iter=200000, use_active=(r > 0),
                                   active_k=3 + r, eval_interval=150)
        full_sc = _score(a_opt)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    Restart {r}: score={full_sc:.10f}")
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_and_optimize(a_best, deadline, rng, sigma, n_iter=100000):

Central-binspired multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)

    return np.fft.irfft(fa * fb, N)[:out_n]
