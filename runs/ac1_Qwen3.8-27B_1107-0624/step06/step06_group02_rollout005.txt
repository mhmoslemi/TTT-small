import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_weights(y, w):
    """Gradient of weighted sum of convolution values."""
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _topk_weights(c, top_k, temperature):
    """Compute weights using top-K selection with soft weighting."""
    c_max = c.max()
    if top_k >= len(c):
        # Full softmax
        exponents = (c - c_max) / temperature
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws
    else:
        # Top-K with Gaussian decay
        sorted_idx = np.argpartition(-c, top_k)[:top_k]
        w = np.zeros_like(c)
        vals = c[sorted_idx]
        v_max = vals.max()
        exponents = (vals - v_max) / temperature
        np.clip(exponents, -500, 500, out=exponents)
        wv = np.exp(exponents)
        ws = wv.sum()
        if ws < 1e-300:
            w[int(np.argmax(c))] = 1.0
        else:
            w[sorted_idx] = wv / ws
    return w

    """FISTA with top-K active set and adaptive temperature."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    beta = beta_start
    iter_per_beta = max(200, max_iter // 10)

    for bi in range(30):
        if time.monotonic() > deadline:
            break
        if beta < beta_min:
            break

        L = 4.0 / beta
        step = 1.0 / L
        local_iter = 0

        for it in range(iter_per_beta):
            if time.monotonic() > deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:

            w = _topk_weights(c_y, top_k, beta)

def _make_window(n, window_type, rng=None):
    """Create various window shapes as initial candidates."""
    x = np.arange(n, dtype=np.float64)
    xm = x - (n - 1) / 2.0
    xn = (n - 1) / 2.0

    if window_type == 0:  # Uniform
        a = np.ones(n)
    elif window_type == 1:  # Hann
        a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    elif window_type == 2:  # Hamming
        a = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    elif window_type == 3:  # Blackman
        a = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    elif window_type == 4:  # Gaussian
        sigma = n / 4.0
        a = np.exp(-0.5 * (xm / sigma) ** 2)
    elif window_type == 5:  # Power law
        a = 1.0 / (np.abs(xm / xn) + 0.1) ** 0.5
    elif window_type == 6:  # Sine
        a = np.sin(np.pi * (x + 0.5) / n)
    elif window_type == 7:  # Bi-Hann
        a = (0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))) ** 2
    elif window_type == 8:  # Exponential decay from center
        a = np.exp(-np.abs(xm / xn))
    elif window_type == 9:  # Random positive
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n) + 0.1
    elif window_type == 10:  # Triangular
        a = 1.0 - np.abs(xm / xn)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-18))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return np.exp(z - z.max()) / np.exp(z - z.max()).sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

        candidates.append(("parent", parent.copy()))
        candidates.append(("parent_perturbed", _perturb(parent, 0.03, rng)))
        candidates.append(("parent_perturbed2", _perturb(parent, 0.08, rng)))

    for wt in range(11):
        a = _make_window(n_main, wt, rng)
        candidates.append((f"window_{wt}_{n_main}", a))

    # Also try a smaller n for potentially better ratio
    n_small = 2048
    for wt in [0, 1, 4, 9]:
        a = _make_window(n_small, wt, rng)
        candidates.append((f"window_{wt}_{n_small}", a))

    for name, a in candidates:
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        sc = _score(a)
        scores.append((sc, name, a))
        print(f"  {name}: score={sc:.10f}")

    scores.sort(key=lambda x: x[0])
    if scores:
        best_score = scores[0][0]
        best_seq = scores[0][2].copy()
        print(f"  Best initial: {scores[0][1]} with score={best_score:.10f}")

    # Phase 1: FISTA with top-K on top candidates
    print(f"\nPhase 1: FISTA top-K optimization")
    phase1_deadline = min(deadline, t_start + 600.0)
    n_top = min(6, len(scores))

    population = []
    for i in range(n_top):
        sc_i, name_i, a_i = scores[i]
        if time.monotonic() > phase1_deadline - 30:
            break

        a_work = a_i.copy()
        n_w = len(a_work)
        c_w = _convolve(a_work, a_work)
        max_cw = float(c_w.max())

        # Adaptive top-K: start with large K, reduce over time
        top_k = max(50, n_w // 20)
        beta_start = 0.05 * max_cw
        beta_min = 1e-10 * max_cw

        time_per = min(60.0, (phase1_deadline - time.monotonic() - 10) / max(1, n_top - i))
        if time_per < 10:
            break

        a_opt, sc_opt = _fista_topk(
            a_work, time.monotonic() + time_per,
            max_iter=200000, top_k=top_k,
            beta_start=beta_start, beta_min=beta_min, beta_decay=0.5
        )
        a_opt = a_opt / a_opt.sum()
        sc_full = _score(a_opt)
        print(f"  FISTA {name_i} (n={n_w}): score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()

        population.append((sc_full, a_opt.copy()))

    # Phase 2: Adam log-space on best candidates
    print(f"\nPhase 2: Adam log-space polishing")
    phase2_deadline = min(deadline, t_start + 1000.0)

    # Sort population by score
    population.sort(key=lambda x: x[0])
    n_adam = min(4, len(population))

    for i in range(n_adam):
        if time.monotonic() > phase2_deadline - 30:
            break
        sc_i, a_i = population[i]
        n_i = len(a_i)
        c_i = _convolve(a_i, a_i)
        max_ci = float(c_i.max())

        betas = []
        beta = 0.001 * max_ci
        while beta > 1e-12 * max_ci:
            betas.append(beta)
            beta *= 0.4
        betas.append(1e-12 * max_ci)

        time_per = min(80.0, (phase2_deadline - time.monotonic() - 10) / max(1, n_adam - i))
        if time_per < 20:
            break

        a_opt, sc_opt = _adam_logspace(
            a_i, time.monotonic() + time_per,
            betas, lr_init=0.01, max_iter_total=500000, top_k=200
        )
        a_opt = a_opt / a_opt.sum()
        sc_full = _score(a_opt)
        print(f"  Adam on n={n_i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            population[0] = (sc_full, a_opt.copy())

    # Phase 3: Population-based search with diverse mutations
    print(f"\nPhase 3: Population-based exploration")
    phase3_deadline = min(deadline, t_start + 1400.0)

    if best_seq is not None and time.monotonic() < phase3_deadline - 60:
        n_base = len(best_seq)
        n_restarts = 10
        per_restart = (phase3_deadline - time.monotonic() - 30) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase3_deadline - 15:
                break

            # Diverse perturbation strategies
            if i % 5 == 0:
                p_start = _perturb(best_seq, 0.02 + 0.01 * i, rng)
            elif i % 5 == 1:
                # Cross with a window
                wt = (i // 5) % 11
                a_w = _make_window(n_base, wt, rng)
                p_start = 0.5 * best_seq + 0.5 * a_w
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            elif i % 5 == 2:
                # Circular shift
                shift = rng.integers(1, max(2, n_base // 16))
                p_start = np.roll(best_seq, shift).copy()
                p_start = p_start / p_start.sum()
            elif i % 5 == 3:
                # Mix with scaled version
                p_start = best_seq * np.exp(-0.1 * rng.standard_normal(n_base) ** 2)
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            else:
                # Large perturbation
                p_start = _perturb(best_seq, 0.1, rng)

            # Quick FISTA
            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            t_per = max(5.0, per_restart * 0.5)
            p_opt, sc_opt = _fista_topk(
                p_start, time.monotonic() + t_per,
                max_iter=50000, top_k=100,
                beta_start=0.01 * max_cs, beta_min=1e-10 * max_cs, beta_decay=0.5
            )
            p_opt = p_opt / p_opt.sum()

            # Quick Adam
            if time.monotonic() < phase3_deadline - 20 and per_restart > 10:
                c_o = _convolve(p_opt, p_opt)
                max_co = float(c_o.max())
                betas_quick = [0.001 * max_co, 0.0003 * max_co, 0.0001 * max_co]
                p_opt, sc_o = _adam_logspace(
                    p_opt, time.monotonic() + max(5.0, per_restart * 0.4),
                    betas_quick, lr_init=0.008, max_iter_total=100000, top_k=100
                )
                p_opt = p_opt / p_opt.sum()

            sc_full = _score(p_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_opt.copy()
                print(f"  Restart {i}: NEW BEST score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Try larger n
    print(f"\nPhase 4: Larger n exploration")
    phase4_deadline = min(deadline, t_start + 1650.0)

    if best_seq is not None and time.monotonic() < phase4_deadline - 60:
        # Try n=8192
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n={n_big}: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        # Optimize the big version
        if time.monotonic() < phase4_deadline - 30:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            t_per = min(60.0, phase4_deadline - time.monotonic() - 10)
            if t_per > 15:
                a_po, sc_po = _fista_topk(
                    a_big, time.monotonic() + t_per * 0.6,
                    max_iter=100000, top_k=200,
                    beta_start=0.02 * max_cb, beta_min=1e-10 * max_cb, beta_decay=0.5
                )
                a_po = a_po / a_po.sum()
                sc_po_full = _score(a_po)
                print(f"  FISTA n={n_big}: score={sc_po_full:.10f}")
                if sc_po_full < best_score:
                    best_score = sc_po_full
                    best_seq = a_po.copy()

        # Also try n=16384 briefly
        if time.monotonic() < phase4_deadline - 30:
            n_xl = 16384
            a_xl = _interp_profile(best_seq, n_xl)
            sc_xl = _score(a_xl)
            print(f"  Interpolated n={n_xl}: score={sc_xl:.10f}")
            if sc_xl < best_score:
                best_score = sc_xl
                best_seq = a_xl.copy()

    # Phase 5: Final ultra-fine polish
    print(f"\nPhase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000003 * max_cp, 0.000001 * max_cp,
                     0.0000003 * max_cp, 0.0000001 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 15:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.003, max_iter_total=300000, top_k=50
            )
            a_pol = a_pol / a_pol.sum()
            sc_pol_full = _score(a_pol)
            print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

        # One more FISTA pass
        if time.monotonic() < deadline - 10:
            c_pf = _convolve(a_pol, a_pol)
            max_cpf = float(c_pf.max())
            a_pf, sc_pf = _fista_topk(
                a_pol, time.monotonic() + min(20.0, deadline - time.monotonic() - 2),
                max_iter=50000, top_k=50,
                beta_start=0.00001 * max_cpf, beta_min=1e-12 * max_cpf, beta_decay=0.5
            )
            a_pf = a_pf / a_pf.sum()
            sc_pf_full = _score(a_pf)
            if sc_pf_full < best_score:
                best_score = sc_pf_full
                best_seq = a_pf.copy()
                print(f"  Final FISTA: score={sc_pf_full:.10f}")

    # Final validation
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 256.0
        # Scale to avoid issues
        final = final / s * s  # keep as is, scale-invariant
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
