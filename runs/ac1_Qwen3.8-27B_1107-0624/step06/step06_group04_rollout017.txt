"""
Improved step function optimizer using:
1. Exact subgradient with peak-averaged gradient and backtracking line search
2. Log-space Adam with adaptive near-exact beta
3. Focused coordinate descent on peak-contributing elements
4. Multi-n exploration around the best length

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _peak_gradient(a, peak_lags):
    """
    Compute the (approximate) gradient of sum_{k in peak_lags} conv(a,a)[k] w.r.t. a.
    For each peak lag k: d(conv(a,a)[k])/da[j] = 2*a[k-j] (if 0<=k-j<n, else 0)
    Returns gradient vector of length n.
    """
    n = len(a)
    grad = np.zeros(n)
    for k in peak_lags:
        # grad[j] += 2*a[k-j] for j where 0<=k-j<n
        # j ranges from max(0, k-n+1) to min(n-1, k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        # a[k-j] for j in [j_min, j_max]
        # When j = j_min: a[k-j_min]
        # When j = j_max: a[k-j_max]
        # So we need a[k-j_max : k-j_min+1] and assign to grad[j_min : j_max+1]
        a_start = k - j_max
        a_end = k - j_min
        grad[j_min:j_max+1] += 2.0 * a[a_start:a_end+1]
    return grad


def _phase1_subgradient(a_init, deadline, max_iter=5000):
    """
    Exact subgradient method with peak-averaged gradient and backtracking line search.
    Adapts K (number of top peaks to average over).
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    K = 1  # Start with just the max
    step_size_init = 0.1
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1]
        top_K = sorted_idx[:K]
        
        # Compute peak-averaged gradient
        grad = _peak_gradient(a, top_K)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        step = step_size_init / (1.0 + 0.05 * no_improve)
        improved = False
        for backtrack in range(25):
            a_try = _proj_simplex(a - step * grad)
            sc_try = _score_norm(a_try)
            if sc_try < best_score * (1.0 - 1e-8):
                # Accept
                a = a_try
                if sc_try < best_score:
                    best_score = sc_try
                    best_a = a.copy()
                    no_improve = 0
                improved = True
                break
            step *= 0.5
        
        if not improved:
            no_improve += 1
            # Adapt K: if stuck with small K, increase it
            if no_improve > 50 and K < 50:
                K = min(50, K * 2)
                no_improve = 0
            elif no_improve > 200:
                # Try a different strategy: random perturbation
                a = a + 0.01 * np.random.randn(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        
    return best_a, best_score


def _phase2_adam(a_init, deadline, beta_list=None, lr_init=0.008, max_iter=300000):
    """
    Adam optimizer in log-space with adaptive beta (near-exact objective).
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    if beta_list is None:
        c = _full_convolve(a, a)
        max_c = float(c.max())
        beta_list = [max_c * b for b in [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]]
    
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_list:
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1
            
            a_s = _softmax(z)
            c = _full_convolve(a_s, a_s)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w
            
            # Gradient in log-space
            a_rev = a_s[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a_s * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 20 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if t % 5000 == 0:

def _phase3_coord_descent(a_init, deadline, max_rounds=500):
    """
    Coordinate descent focusing on peak-contributing elements.
    For each round, identify the peak lag and the top contributing pairs,
    then try small transfers between elements.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    delta_base = 1e-4  # Base transfer amount
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        peak_lag = int(np.argmax(c))
        
        # Find top contributing pairs to the peak
        # c[peak_lag] = sum_i a[i]*a[peak_lag-i]
        # Find i's where a[i]*a[peak_lag-i] is large
        i_range = range(max(0, peak_lag - n + 1), min(n, peak_lag + 1))
        products = np.array([a[i] * a[peak_lag - i] for i in i_range if 0 <= peak_lag - i < n])
        i_vals = np.array([i for i in i_range if 0 <= peak_lag - i < n])
        
        if len(products) == 0:
            continue
        
        # Sort by contribution
        top_idx = np.argsort(products)[::-1][:min(10, len(products))]
        
        # For each top contributor, try transferring mass to a non-contributing position
        accepted = 0
        for ti in top_idx:
            if time.monotonic() > deadline:
                break
            
            i_src = int(i_vals[ti])
            # The "partner" index in the peak
            i_partner = peak_lag - i_src
            
            # Choose a target: random position that doesn't contribute much to the peak
            # or a position that's "opposite" to the peak
            candidates = []
            for attempt in range(5):
                j_dst = int(rng.integers(n))
                # Check if j_dst contributes to the peak
                if 0 <= peak_lag - j_dst < n:
                    contrib = a[j_dst] * a[peak_lag - j_dst]
                else:
                    contrib = 0
                if contrib < products[ti] * 0.1:  # Much less than the source's contribution
                    candidates.append(j_dst)
                    break
            
            if not candidates:
                j_dst = int(rng.integers(n))
            else:
                j_dst = candidates[0]
            
            if j_dst == i_src:
                continue
            
            # Try transferring delta from i_src to j_dst
            delta = delta_base * (1.0 + 0.5 * rng.random())
            if a[i_src] < delta:
                continue
            
            a[i_src] -= delta
            a[j_dst] += delta
            
            sc_new = _score_norm(a)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
                accepted += 1
            else:
                # Revert
                a[i_src] += delta
                a[j_dst] -= delta
        
        # Also try random pairs
        for _ in range(5):
            if time.monotonic() > deadline:
                break
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            delta = delta_base * 0.5 * rng.random()
            if a[i1] < delta:
                continue
            a[i1] -= delta
            a[i2] += delta
            sc_new = _score_norm(a)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
            else:
                a[i1] += delta
                a[i2] -= delta
        
        # Adaptive delta
        if accepted > 10:
            delta_base = max(1e-6, delta_base * 0.7)
        elif accepted == 0 and rnd > 10:
            delta_base = min(1e-2, delta_base * 1.5)
    
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
