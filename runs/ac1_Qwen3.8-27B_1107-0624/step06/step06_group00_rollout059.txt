- `a[k-j] * a[j]` (when `i = k-j`, which is the same as the previous when `k-j = j` i.e. `k = 2j`)

If `k = 2j`: the contribution is `a[j]^2`

When we change `a[j]` to `new`:
- For `k ≠ 2j`: new contribution is `2 * new * a[k-j]`
- For `k = 2j`: new contribution is `new^2`

So `Δconv[k] = 2*(new - a[j])*a[k-j]` for `k ≠ 2j`, and `new^2 - a[j]^2` for `k = 2j`.

The new max convolution is `max_k (conv[k] + Δconv[k])`. We want to minimize this over `new >= 0`, subject to the simplex constraint (which means we'd also need to adjust another coordinate, or work in a "free" parameterization).

"""
Subgradient mirror descent with spectral flattening for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses adaptive step sizes, alternating
time/frequency domain optimization, and multi-phase annealing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _subgradient(a, c, top_k=10, soft_T=0.0):
    """Compute subgradient of max(conv(a,a)) at point a.
    If soft_T > 0, use softmax-weighted combination (smooth approximation).
    """
    n = len(a)
    if soft_T > 0:
        # Soft approximation
        c_shifted = (c - c.max()) / soft_T
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
    else:
        # Hard: top-k
        idx = np.argsort(c)[::-1][:top_k]
        p = np.zeros_like(c)
        p[idx] = 1.0 / top_k
    
    # sg[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    # = 2 * (p * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_pa = _fft_convolve(p, a_rev)  # length = (2n-1) + n - 1 = 3n-2
    sg = 2.0 * conv_pa[n-1:2*n-1]
    return sg


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_flatten(a, n_iter=10, alpha=0.5):
    """Flatten the power spectrum of a while maintaining non-negativity."""
    n = len(a)
    N = _next_pow2(2 * n)
    for _ in range(n_iter):
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        # Target: geometric mean magnitude (preserves total power)
        log_mag = np.log(mag + 1e-30)
        target_mag = np.exp(np.mean(log_mag))
        # Blend current and target
        new_mag = (1 - alpha) * mag + alpha * target_mag
        # Reconstruct with original phase
        phase = np.angle(F)
        F_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, N)[:n]
        # Project to non-negative and renormalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        a = a_new
    return a


def _mirror_descent(a, deadline, T_init, T_final, n_iter_max=500000,
                    lr_init=0.1, lr_final=0.001, top_k=5,
                    use_spectral=False, spec_interval=100,
                    eval_interval=200, patience_limit=10):
    """Mirror descent with adaptive temperature and step size."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    t = 0
    
    while t < n_iter_max and time.monotonic() < deadline:
        # Check time budget for this iteration
        est_time = (time.monotonic() - (deadline - (deadline - time.monotonic())))  # just time.monotonic()
        if time.monotonic() > deadline:
            break
        
        t += 1
        frac = min(1.0, t / n_iter_max)
        T = T_init * (T_final / T_init) ** frac
        lr = lr_init * (lr_final / lr_init) ** frac
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Compute gradient (soft subgradient)
        if T > T_init * 0.01:
            # Use soft gradient
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s_e = e.sum()
            if s_e < 1e-300 or not np.isfinite(s_e):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s_e
        else:
            # Use hard top-k
            idx = np.argsort(c)[::-1][:top_k]
            p = np.zeros_like(c)
            p[idx] = 1.0 / top_k
        
        # Subgradient
        a_rev = a[::-1]
        conv_pa = _fft_convolve(p, a_rev)
        sg = 2.0 * conv_pa[n-1:2*n-1]
        
        # Mirror descent update
        W = float(np.dot(a, sg))
        z_grad = a * (sg - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        z = z - lr * z_grad
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        # Periodic spectral flattening
        if use_spectral and t % spec_interval == 0:
            a_cur = _softmax_stable(z)
            a_spec = _spectral_flatten(a_cur, n_iter=3, alpha=0.3)
            # Blend
            blend = 0.3
            a_blended = (1 - blend) * a_cur + blend * a_spec
            a_blended = a_blended / a_blended.sum()
            z = np.log(np.maximum(a_blended, 1e-15))
            z = z - z.mean()
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > patience_limit:
                # Restart from best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                patience = 0
                # Reduce learning rate
                lr_init *= 0.5
                lr_final *= 0.5
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _adam_subgradient(a, deadline, T_init, T_final, n_iter_max=500000,
                      lr_init=0.05, lr_final=0.001, top_k=5,
                      use_spectral=False, spec_interval=200,
                      eval_interval=200, patience_limit=15,
                      use_momentum=True):
    """Adam optimizer with subgradient in z-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr_mult = 1.0
    t_start = time.monotonic()
    
    while t < n_iter_max and time.monotonic() < deadline:
        t += 1
        iter_time = (time.monotonic() - t_start) / max(1, t)
        # Adaptive: reduce iterations if too slow
        if iter_time > 0.01:  # more than 10ms per iter
            break
        
        frac = min(1.0, t / n_iter_max)
        T = T_init * (T_final / T_init) ** frac
        lr = lr_init * (lr_final / lr_init) ** frac * current_lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Gradient
        if T > T_init * 0.005:
            c_shifted = (c - c.max()) / T
            e = np.exp(c_shifted)
            s_e = e.sum()
            if s_e < 1e-300 or not np.isfinite(s_e):
                p = np.zeros_like(c)
                p[np.argmax(c)] = 1.0
            else:
                p = e / s_e
        else:
            idx = np.argsort(c)[::-1][:top_k]
            p = np.zeros_like(c)
            p[idx] = 1.0 / top_k
        
        a_rev = a[::-1]
        conv_pa = _fft_convolve(p, a_rev)
        sg = 2.0 * conv_pa[n-1:2*n-1]
        
        # Natural gradient (Fisher metric on simplex)
        W = float(np.dot(a, sg))
        z_grad = a * (sg - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * z_grad
        v_buf = b2 * v_buf + (1 - b2) * (z_grad * z_grad)
        mh = m_buf / (1 - b1**t)
        vh = v_buf / (1 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Spectral flattening
        if use_spectral and t % spec_interval == 0:
            a_cur = _softmax_stable(z)
            a_spec = _spectral_flatten(a_cur, n_iter=5, alpha=0.2)
            blend = 0.2
            a_blended = (1 - blend) * a_cur + blend * a_spec
            a_blended = np.maximum(a_blended, 1e-15)
            a_blended = a_blended / a_blended.sum()
            z = np.log(a_blended)
            z = z - z.mean()
        
        # Evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > patience_limit and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _coordinate_polish(a, n_passes=100):
    """Coordinate-wise optimization: for each coordinate, find the best value."""
    n = len(a)
    a = a.copy()
    for _ in range(n_passes):
        improved = False
        for j in range(n):
            old_val = a[j]
            # Try a few candidate values
            best_val = old_val
            best_max = np.inf
            for scale in [0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.5, 2.0]:
                test_val = old_val * scale
                # Renormalize to keep sum the same
                new_sum = a.sum() - old_val + test_val
                if new_sum <= 0:
                    continue
                a[j] = test_val
                c = _fft_convolve(a, a)
                cur_max = c.max() / (a.sum() ** 2)
                if cur_max < best_max:
                    best_max = cur_max
                    best_val = test_val
            a[j] = best_val
            if best_val != old_val:
                improved = True
        if not improved:
            break
    return a


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase subgradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines Adam in log-space with adaptive temperature, spectral diversification,
multi-scale exploration, and perturbation-based restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, T, top_k):
    """Compute gradient (subgradient approx) of max(conv(a,a))."""
    n = len(a)
    c_max = c.max()
    if T > 1e-12:
        ratio = (c - c_max) / T
        e = np.exp(ratio)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
    else:
        idx = np.argsort(c)[::-1][:top_k]
        p = np.zeros_like(c)
        p[idx] = 1.0 / top_k
    a_rev = a[::-1]
    conv_pa = _fft_convolve(p, a_rev)
    sg = 2.0 * conv_pa[n - 1:2 * n - 1]
    return sg


def _spectral_flatten(a, n_iter=5, alpha=0.3):
    """Flatten power spectrum while maintaining non-negativity."""
    n = len(a)
    N = _next_pow2(2 * n)
    for _ in range(n_iter):
        F = np.fft.rfft(a, N)
        mag = np.abs(F)
        phase = np.angle(F)
        log_mag = np.log(mag + 1e-30)
        target = np.exp(np.mean(log_mag))
        new_mag = (1 - alpha) * mag + alpha * target
        F_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(F_new, N)[:n]
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        a = a_new
    return a


def _adam_optimize(a, deadline, T_init, T_final, lr_init, lr_final,
                   top_k_init, top_k_final, n_iter_max=300000,
                   use_spectral=False, spec_interval=150,
                   eval_interval=200, patience_limit=12,
                   clip_val=0.8):
    """Adam in log-space with adaptive temperature and top-k."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr_mult = 1.0
    t0 = time.monotonic()

    while t < n_iter_max:
        if time.monotonic() > deadline:
            break
        t += 1
        # Time check: if iterating too slowly, stop
        if t > 100:
            avg_time = (time.monotonic() - t0) / t
            if avg_time > 0.005 and t > 500:
                break

        frac = min(1.0, t / n_iter_max)
        T = T_init * (T_final / T_init) ** frac
        lr = lr_init * (lr_final / lr_init) ** frac * lr_mult
        top_k = max(1, int(round(top_k_init + (top_k_final - top_k_init) * frac)))

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        sg = _compute_gradient(a, c, T, top_k)

        # Natural gradient (Fisher metric)
        W = float(np.dot(a, sg))
        z_grad = a * (sg - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > clip_val:
            update = update * (clip_val / update_norm)

        z = z - update
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        # Spectral diversification
        if use_spectral and t % spec_interval == 0:
            a_cur = _softmax_stable(z)
            a_spec = _spectral_flatten(a_cur, n_iter=3, alpha=0.25)
            blend = 0.15
            a_blended = (1 - blend) * a_cur + blend * a_spec
            a_blended = np.maximum(a_blended, 1e-15)
            a_blended = a_blended / a_blended.sum()
            z = np.log(a_blended)
            z = z - z.mean()

        # Periodic evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_limit and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _coordinate_polish_fast(a, n_passes=3):
    """Fast coordinate polish using incremental convolution updates."""
    n = len(a)
    a = a.copy()
    c = _fft_convolve(a, a)
    total = a.sum()
    c_max = c.max()

    for _ in range(n_passes):
        for j in range(n):
            old = a[j]
            best_val = old
            best_cmax = c_max
            # Try geometric scales
            for scale in [0.6, 0.8, 1.2, 1.5, 2.0, 0.5, 0.7, 1.3]:
                new_val = old * scale
                if new_val < 0:
                    continue
                delta = new_val - old
                new_total = total + delta
                if new_total < 1e-10:
                    continue
                # Update convolution: conv[k] += 2*delta*a[k-j] (for k != 2j)
                # conv[2j] += 2*old*delta + delta^2
                c_new = c.copy()
                j_min = max(0, j)
                j_max = min(2 * n - 2, j + n - 1)
                k_range = np.arange(j_min, j_max + 1)
                a_idx = k_range - j  # a[k-j]
                valid = (a_idx >= 0) & (a_idx < n)
                if valid.any():
                    c_new[k_range[valid]] += 2.0 * delta * a[a_idx[valid]]
                if 2 * j < 2 * n - 1:
                    c_new[2 * j] += delta * delta  # extra quadratic term
                # Score change (proportional)
                new_cmax = c_new.max()
                # We compare new_cmax/new_total^2 vs c_max/total^2
                # i.e., new_cmax * total^2 vs c_max * new_total^2
                if new_cmax * total * total < best_cmax * new_total * new_total:
                    best_cmax = new_cmax * (total / new_total) ** 2
                    best_val = new_val
                    c = c_new
                    total = new_total
                    c_max = best_cmax
            if best_val != old:
                a[j] = best_val

    # Renormalize
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: Adam subgradient with adaptive temperature,
    spectral diversification, multi-scale exploration, and coordinate polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Get parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Phase 1: Moderate-temperature Adam (exploration) [0 - 200s]
    phase1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < phase1_end:
        c0 = _fft_convolve(a, a)
        max_c = float(c0.max())
        a1, sc1 = _adam_optimize(
            a, phase1_end,
            T_init=0.005 * max_c, T_final=0.0002 * max_c,
            lr_init=0.04, lr_final=0.004,
            top_k_init=10, top_k_final=3,
            n_iter_max=150000,
            use_spectral=False,
            eval_interval=200, patience_limit=10
        )
        full_sc = _score(a1)
        print(f"Phase 1 moderate-T: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a1.copy()

    # Phase 2: Low-temperature Adam with spectral diversification [200 - 500s]
    phase2_end = min(deadline, t_start + 500.0)
    if time.monotonic() < phase2_end:
        cb = _fft_convolve(best_a, best_a)
        max_cb = float(cb.max())
        a2, sc2 = _adam_optimize(
            best_a, phase2_end,
            T_init=0.001 * max_cb, T_final=0.00001 * max_cb,
            lr_init=0.025, lr_final=0.002,
            top_k_init=5, top_k_final=2,
            n_iter_max=200000,
            use_spectral=True, spec_interval=120,
            eval_interval=200, patience_limit=12
        )
        full_sc = _score(a2)
        print(f"Phase 2 low-T+spec: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a2.copy()

    # Phase 3: Very low temperature, fine tuning [500 - 850s]
    phase3_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase3_end:
        cb = _fft_convolve(best_a, best_a)
        max_cb = float(cb.max())
        a3, sc3 = _adam_optimize(
            best_a, phase3_end,
            T_init=0.0003 * max_cb, T_final=0.000003 * max_cb,
            lr_init=0.012, lr_final=0.0008,
            top_k_init=3, top_k_final=1,
            n_iter_max=250000,
            use_spectral=False,
            eval_interval=150, patience_limit=15
        )
        full_sc = _score(a3)
        print(f"Phase 3 fine: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a3.copy()

    # Phase 4: Multi-scale exploration [850 - 1200s]
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end:
        # Try n=8192
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        sub_deadline = min(phase4_end, time.monotonic() + 150.0)
        a_large_opt, sc_large = _adam_optimize(
            a_large, sub_deadline,
            T_init=0.002 * max_cl, T_final=0.00002 * max_cl,
            lr_init=0.03, lr_final=0.003,
            top_k_init=8, top_k_final=3,
            n_iter_max=80000,
            use_spectral=True, spec_interval=100,
            eval_interval=200, patience_limit=8
        )
        full_sc = _score(a_large_opt)
        print(f"Phase 4 n={n_large}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_large_opt.copy()

        # Interpolate back and re-optimize
        if time.monotonic() < phase4_end - 60:
            a_back = _interp_to_length(a_large_opt, n)
            cb = _fft_convolve(a_back, a_back)
            max_cback = float(cb.max())
            a_back_opt, sc_back = _adam_optimize(
                a_back, min(phase4_end, time.monotonic() + 100.0),
                T_init=0.001 * max_cback, T_final=0.00001 * max_cback,
                lr_init=0.02, lr_final=0.002,
                top_k_init=5, top_k_final=2,
                n_iter_max=60000,
                use_spectral=False,
                eval_interval=100, patience_limit=8
            )
            full_sc_b = _score(a_back_opt)
            print(f"Phase 4 interp back n={n}: score={full_sc_b:.10f}")
            if np.isfinite(full_sc_b) and full_sc_b < best_score:
                best_score = full_sc_b
                best_a = a_back_opt.copy()

    # Phase 5: Perturbation restarts [1200 - 1550s]
    phase5_end = min(deadline, t_start + 1550.0)
    n_best = len(best_a)
    if time.monotonic() < phase5_end:
        n_restarts = 6
        time_per_restart = max(30.0, (phase5_end - time.monotonic()) / n_restarts - 10)
        for attempt in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            # Generate perturbed start
            if attempt % 3 == 0:
                # Gaussian in log-space
                sigma = 0.012 + 0.004 * (attempt % 3)
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + sigma * rng.standard_normal(n_best)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            elif attempt % 3 == 1:
                # Spectral perturbation
                N = _next_pow2(2 * n_best)
                F = np.fft.rfft(best_a, N)
                mag = np.abs(F)
                # Add noise to magnitude
                noise = 1.0 + 0.04 * rng.standard_normal(len(mag))
                noise = np.maximum(noise, 0.3)
                F_new = mag * noise * np.exp(1j * np.angle(F))
                a_pert = np.fft.irfft(F_new, N)[:n_best]
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()
            else:
                # Block perturbation (modify contiguous blocks)
                a_pert = best_a.copy()
                n_blocks = 4
                block_size = n_best // (2 * n_blocks)
                for bi in range(n_blocks):
                    start = bi * 2 * block_size + rng.integers(0, block_size)
                    end = start + block_size
                    scale = 1.0 + 0.15 * rng.standard_normal()
                    a_pert[start:end] *= max(0.1, scale)
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            rd = time.monotonic() + time_per_restart
            a_pert_opt, sc_pert = _adam_optimize(
                a_pert, rd,
                T_init=0.0008 * max_cp, T_final=0.000008 * max_cp,
                lr_init=0.02, lr_final=0.0015,
                top_k_init=4, top_k_final=2,
                n_iter_max=50000,
                use_spectral=(attempt % 2 == 0),
                spec_interval=80,
                eval_interval=100, patience_limit=8
            )
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}]: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 6: Ultra-fine final polish [1550s - deadline]
    if time.monotonic() < deadline - 15:
        cb = _fft_convolve(best_a, best_a)
        max_cb = float(cb.max())
        a_fin, sc_fin = _adam_optimize(
            best_a, deadline - 12,
            T_init=0.0001 * max_cb, T_final=0.000001 * max_cb,
            lr_init=0.004, lr_final=0.0002,
            top_k_init=2, top_k_final=1,
            n_iter_max=200000,
            use_spectral=False,
            eval_interval=100, patience_limit=25
        )
        full_sc = _score(a_fin)
        print(f"Phase 6 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fin.copy()

    # Phase 7: Coordinate polish (if time remains)
    if time.monotonic() < deadline - 8:
        remaining = deadline - time.monotonic() - 8
        if remaining > 30 and len(best_a) <= 4096:
            n_passes = max(1, int(remaining // 40))
            a_cpoly = _coordinate_polish_fast(best_a, n_passes=min(n_passes, 4))
            sc_cpoly = _score(a_cpoly)
            print(f"Coordinate polish: score={sc_cpoly:.10f}")
            if np.isfinite(sc_cpoly) and sc_cpoly < best_score:
                best_score = sc_cpoly
                best_a = a_cpoly.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
