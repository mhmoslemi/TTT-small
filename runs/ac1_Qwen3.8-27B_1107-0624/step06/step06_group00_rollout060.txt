"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
- 1/sqrt(1-x) initialization (near-flat infinite convolution property)
- Projected Adam optimizer with soft-max and active-set gradients
- Variance penalty encouraging flat autocorrelation

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _init_sqrt_inv(n):
    """Generate a[i] = C(2i,i)/4^i = 1/sqrt(1-x) coefficients. O(n)."""
    a = np.zeros(n)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    s = a.sum()
    if s > 0:
        a /= s
    return a

def _soft_gradient(a, c, T):
    """Gradient of soft-max approximation of max(c)."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)

    else:

    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _active_gradient(a, c, top_k=10):
    """Exact subgradient of max using top-k active set."""
    n = len(a)
    k_idx = np.argsort(c)[::-1][:top_k]

    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * conv_full[n - 1:2 * n - 1][::-1]

def _variance_gradient(a, c):
    """Gradient of variance of c (encourages flatness)."""
    n = len(a)
    target = c.mean()

    res_rev = residual[::-1]
    conv_full = _fft_convolve(a, res_rev)
    g = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_a = a_init.copy()
    best_sc = _score_unnorm(a_init / a_init.sum())
    patience = 0
    best_patience = best_sc
    lr_mult = 1.0
    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.5)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _project_simplex(np.exp(z))
        c = _fft_convolve(a, a)

        # Soft or active gradient
        if T < T_start * 0.01:
            g = _active_gradient(a, c, active_k)
        else:
            g = _soft_gradient(a, c, T)

        # Variance penalty
        if use_var_penalty:
            g = g + var_weight * _variance_gradient(a, c)

        # Center gradient (constraint sum(a)=1)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        # Gradient clipping
        un = np.linalg.norm(update)
        if un > 1.0:
            update = update * (1.0 / un)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

            sc = _score(a_cur)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()
                patience = 0
                best_patience = best_sc
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _project_simplex(np.exp(z))
    sc = _score(a_final)
    if sc < best_sc:
        best_sc = sc
        best_a = a_final.copy()
    return best_a, best_sc


def _optimize_projected(a_init, deadline, lr_start, lr_end, max_iter=200000,
                        eval_interval=50, use_var=False, var_w=0.05):
    """Projected subgradient method with momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    v = np.zeros(n)
    b2, eps = 0.999, 1e-10
    best_a = a.copy()
    best_sc = _score(a)
    patience = 0
    lr_mult = 1.0
    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        c = _fft_convolve(a, a)
        k_idx = np.argsort(c)[::-1][:5]
        p = np.zeros_like(c)
        p[k_idx] = 1.0 / len(k_idx)
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g = 2.0 * conv_full[n - 1:2 * n - 1][::-1]

        if use_var:
            g = g + var_w * _variance_gradient(a, c)

        W = float(np.dot(a, g))
        a_new = a - lr * (g - W)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        # Momentum
        v = b2 * v + (1.0 - b2) * (a_new - a)
        a = a_new
        if t % eval_interval == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 6 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                a = best_a.copy()
                v = np.zeros(n)

    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _refine_top_k(a, k=5, steps=200, deadline=None):
    """Directly reduce the top-k convolution values via projected updates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    for step in range(steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        max_c = c.max()
        # Find indices where conv is close to max
        threshold = max_c * 0.95
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(c)[::-1][:k]
        # Gradient pointing to reduce active conv values
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)
        p_rev = p[::-1]
        conv_a = _fft_convolve(a, p_rev)
        g = 2.0 * conv_a[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, g))
        lr = 0.01 / math.sqrt(step + 1)
        a_new = a - lr * (g - W)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_unnorm(a_new)
        if sc < _score_unnorm(a):
            a = a_new
            sc_full = _score(a)
            if sc_full < best_sc:
                best_sc = sc_full
                best_a = a.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining 1/sqrt(1-x) initialization,
    Adam in log-space, projected subgradient, and active-set refinement.
    Explores multiple sequence lengths and restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Try to use parent as one candidate
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={_score(parent):.8f}")
    except Exception:
        pass

    best_a = None
    best_score = np.inf

    def _update_candidate(a, label):
        nonlocal best_a, best_score
        if a is None:
            return
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  NEW BEST [{label}]: score={sc:.8f}, n={len(a)}, t={time.monotonic()-t_start:.0f}s")

    # Phase 1: 1/sqrt(1-x) initialization with Adam (primary strategy)
    # Try multiple lengths
    lengths_to_try = [4096, 8192]
    if parent is not None and len(parent) not in lengths_to_try:
        lengths_to_try.insert(0, len(parent))

    # Deduplicate and sort by length (shorter first for quick wins)
    lengths_to_try = sorted(set(lengths_to_try))

    # Time budget per phase
    n_phases = 4
    phase_time = (deadline - t_start) / n_phases

    for n in lengths_to_try:
        if time.monotonic() > deadline - 10:
            break

        # Strategy A: 1/sqrt(1-x) init + Adam
        if time.monotonic() < deadline - 30:
            a_init = _init_sqrt_inv(n)
            print(f"\n[n={n}] 1/sqrt(1-x) init: score={_score(a_init):.8f}")
            _update_candidate(a_init, f"sqrt_inv_init_n{n}")

            if time.monotonic() < deadline - 30:
                d1 = min(deadline - 60, time.monotonic() + phase_time * 0.6)
                c_init = _fft_convolve(a_init, a_init)
                max_c = float(c_init.max())
                a_opt, sc = _optimize_adam(
                    a_init, d1,
                    T_start=0.01 * max_c, T_end=0.00005 * max_c,
                    lr_start=0.05, lr_end=0.003,
                    use_var_penalty=True, var_weight=0.1,
                    active_k=10, eval_interval=80)
                _update_candidate(a_opt, f"adam_sqrt_n{n}")

                # Refinement
                if time.monotonic() < deadline - 20:
                    a_ref, sc_ref = _refine_top_k(a_opt, k=5, steps=500,
                                                   deadline=min(deadline - 15, time.monotonic() + 120))
                    _update_candidate(a_ref, f"refine_sqrt_n{n}")

        # Strategy B: Uniform init + Adam with variance penalty
        if time.monotonic() < deadline - 60:
            a_uni = _init_uniform(n)
            d2 = min(deadline - 45, time.monotonic() + phase_time * 0.4)
            c_u = _fft_convolve(a_uni, a_uni)
            max_cu = float(c_u.max())
            a_u, sc_u = _optimize_adam(
                a_uni, d2,
                T_start=0.005 * max_cu, T_end=0.00002 * max_cu,
                lr_start=0.04, lr_end=0.002,
                use_var_penalty=True, var_weight=0.15,
                active_k=8, eval_interval=100)
            _update_candidate(a_u, f"adam_uniform_n{n}")

    # Phase 2: Random restarts from best
    if best_a is not None and time.monotonic() < deadline - 60:
        n_best = len(best_a)
        print(f"\nPhase 2: Random restarts from best (n={n_best})")
        n_restarts = min(8, int((deadline - time.monotonic() - 30) / 60))
        for r in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            sigma = 0.05 + 0.03 * r
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _project_simplex(np.exp(z))
            d = min(deadline - 20 - 30 * (n_restarts - 1 - r), time.monotonic() + 50)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            a_r, sc_r = _optimize_adam(
                a_pert, d,
                T_start=0.002 * max_cp, T_end=0.00005 * max_cp,
                lr_start=0.03, lr_end=0.002,
                use_var_penalty=True, var_weight=0.08,
                active_k=5, eval_interval=50)
            _update_candidate(a_r, f"restart_{r}_n{n_best}")

    # Phase 3: Projected subgradient on best
    if best_a is not None and time.monotonic() < deadline - 40:
        n_best = len(best_a)
        print(f"\nPhase 3: Projected subgradient (n={n_best})")
        d3 = min(deadline - 20, time.monotonic() + 120)
        a_pg, sc_pg = _optimize_projected(
            best_a, d3,
            lr_start=0.02, lr_end=0.001,
            max_iter=100000, eval_interval=30,
            use_var=True, var_w=0.1)
        _update_candidate(a_pg, "projected_subgrad")

    # Phase 4: Final ultra-fine polish
    if best_a is not None and time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        print(f"\nPhase 4: Final polish (n={n_fin})")
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        d4 = deadline - 5
        a_fin, sc_fin = _optimize_adam(
            best_a, d4,
            T_start=0.0005 * max_cf, T_end=0.000005 * max_cf,
            lr_start=0.01, lr_end=0.0003,
            use_var_penalty=False, var_weight=0.0,
            active_k=3, eval_interval=30)
        _update_candidate(a_fin, "final_polish")

        # Also try direct refinement
        if time.monotonic() < deadline - 5:
            a_ref2, sc_ref2 = _refine_top_k(a_fin, k=3, steps=300,
                                            deadline=deadline - 3)
            _update_candidate(a_ref2, "final_refine")

    # Final output
    if best_a is None:
        best_a = _init_sqrt_inv(2048)

    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01

    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
