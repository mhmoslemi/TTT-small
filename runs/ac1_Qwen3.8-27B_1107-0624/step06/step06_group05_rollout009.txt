"""
Asymmetric sequence optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses asymmetric initialization, FISTA with continuation, alternating minimization,
Adam refinement, L-BFGS polish, and multi-restart exploration.
Key insight: optimal sequences must be asymmetric (symmetric ones can't beat uniform).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_weights(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):

    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    s_init = a_curr.sum()
    if s_init > 1e-30:
        a_curr = a_curr / s_init
        a_prev = a_curr.copy()
    L = 4.0 * n / max(beta, 1e-15)
    step = 1.0 / L
    if step > 0.1:
        step = 0.1
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

            w = w / w_sum

        a_next = _proj_simplex(y - step * grad)
        a_prev = a_curr.copy()

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    a0 = a_init.copy()
    s0 = a0.sum()
    if s0 > 1e-30:
        a0 = a0 / s0
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a0)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a0, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 25)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score

    profiles.append(a)

        if time.monotonic() > phase2_end - 2:
            break
        remaining = phase2_end - time.monotonic() - 2
        if remaining < 2:
            break
        per_time = min(remaining / max(1, n_phases - phase), 20.0)
        max_it = max(2000, int(per_time * 3000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 5 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.4 * float(c_curr.max())
        if beta < 1e-12:
            beta = 1e-12

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Alternating minimization (next 15%) ===
    print("Phase 3: Alternating minimization")
    phase3_end = t_start + 0.60 * (deadline - t_start)
    if time.monotonic() < phase3_end - 5:
        a_alt, _ = _alt_minimize(a_work, phase3_end, max_iter=50000)
        sc_alt = _score(a_alt)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt.copy()
            print(f"  Alt-min: score={sc_alt:.10f} t={time.monotonic()-t_start:.1f}s")
        # Combine: try averaging alt-min result with current best
        a_combo = 0.5 * a_alt + 0.5 * best_seq
        a_combo = np.maximum(a_combo, 0)
        a_combo = a_combo / a_combo.sum()
        sc_combo = _score(a_combo)
        if sc_combo < best_score:
            best_score = sc_combo
            best_seq = a_combo.copy()
            print(f"  Combo: score={sc_combo:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = best_seq.copy()

    # === Phase 4: Adam multi-beta fine (next 20%) ===
    print("Phase 4: Adam fine optimization")
    phase4_end = t_start + 0.80 * (deadline - t_start)
    if time.monotonic() < phase4_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.001 * max_cc, 0.0002 * max_cc, 0.00005 * max_cc, 0.00001 * max_cc]
        per_beta = (phase4_end - time.monotonic() - 5.0) / len(betas)
        per_beta = max(5.0, per_beta)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase4_end - 5:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.005, max_iter=200000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: L-BFGS refinement (next 10%) ===
    print("Phase 5: L-BFGS refinement")
    phase5_end = deadline - 30.0
    if time.monotonic() < phase5_end - 10:
        c_curr = _full_convolve(best_seq, a_work)
        max_cc = float(_full_convolve(a_work, a_work).max())
        beta_lb = 0.00002 * max_cc
        a_lb, _ = _lbfgs_opt(a_work, phase5_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 6: Multi-restart exploration ===
    print("Phase 6: Multi-restart")
    phase6_end = deadline - 15.0
    if time.monotonic() < phase6_end - 10:
        n_restarts = 5
        per_r = (phase6_end - time.monotonic() - 5.0) / n_restarts
        per_r = max(5.0, min(per_r, 60.0))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 5:
                break
            if i == 0:
                p = _perturb(a_work, 0.01, rng)
            elif i == 1:
                p = _perturb(a_work, 0.02, rng)
            elif i == 2:
                # Shift
                shift = rng.integers(1, max(2, n_main // 256))
                p = np.roll(a_work, shift).copy()
                p = p / p.sum()
            elif i == 3:
                # Blend with a new profile
                new_prof = _make_asymmetric_profiles(n_main, rng)[i % 4]
                p = 0.7 * a_work + 0.3 * new_prof
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            else:
                # Mirror + perturb
                p = 0.5 * a_work + 0.5 * a_work[::-1]
                p = _perturb(p, 0.01, rng)
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.001 * max_cp
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.4, 20000)
            p, _ = _adam_opt(p, time.monotonic() + per_r * 0.6, b_r * 0.1, 0.003, 50000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: score={sc_r:.10f} (not better)")

    # === Phase 7: Final ultra-polish ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 3
        if per_t > 5:
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.5,
                                 0.000003 * max_cp, 0.001, 100000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            if time.monotonic() < deadline - 3:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000002 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 2.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(n_main) / n_main
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
