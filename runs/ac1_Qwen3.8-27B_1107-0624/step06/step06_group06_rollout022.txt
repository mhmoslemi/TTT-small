- Update: `z_{new} = z_old - eta * grad` where grad is the subgradient of the objective

- Use a decreasing step size: `eta_t = C / sqrt(t)` (optimal for convex problems)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

        return np.inf

def _topk_gradient(z, a, c, n, k=5):
    """Compute subgradient of sum of top-k peaks."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)

        top_k = sorted_idx[:1]  # Just use the max

    """Upsample sequence from len(a) to new_n using cubic interpolation."""

            break
        lr = lr_init / (1.0 + 0.005 * np.sqrt(t_val))
        # Nesterov: look-ahead
        z_look = z + momentum * v
        # Compute gradient at look-ahead point
        a_look = _softmax(z_look)
        c_look = _full_convolve(a_look, a_look)
        grad_look = _topk_gradient(a_look, c_look, n, k=k_peaks)
        v = momentum * v - lr * grad_look
        z = z + v
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_optimize(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation (very fine schedule)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    beta_start = max(gap, max_c * 0.01)

    ratios = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.015, 0.007,
              0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    betas = beta_start * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(2000, max(20, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-11, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _upsample(a, new_n):
    """Upsample sequence using cubic interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _make_structured(n, kind, rng):
    """Create structured initializations."""
    t = np.arange(n) / n
    if kind == "cos_mod":
        m = int(rng.integers(1, max(2, n // 100)))
        eps = float(rng.uniform(0.2, 0.6))
        a = 1.0 + eps * np.cos(2.0 * np.pi * m * t)
        a = np.maximum(a, 1e-10)
    elif kind == "multi_cos":
        n_terms = int(rng.integers(2, 6))
        a = np.ones(n)
        for _ in range(n_terms):
            m = int(rng.integers(1, max(2, n // 50)))
            eps = float(rng.uniform(0.05, 0.3))
            phase = float(rng.uniform(0, 2 * np.pi))
            a = a + eps * np.cos(2.0 * np.pi * m * t + phase)
        a = np.maximum(a, 1e-10)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
    elif kind == "superflat":
        k = t * (n - 1)
        tri = np.minimum(k, (n - 1) - k).astype(float) + 1.0
        a = 1.0 / np.sqrt(tri)
    elif kind == "bipolar_smooth":
        m1 = int(rng.integers(2, max(3, n // 200)))
        m2 = int(rng.integers(m1 + 1, max(m1 + 2, n // 100)))
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * m1 * t) + 0.2 * np.cos(2.0 * np.pi * m2 * t)
        a = np.maximum(a, 1e-10)
    else:
        a = np.ones(n)
    return a / a.sum()


def _subgrad_line_search(a_init, deadline, n_steps=3000, k_peaks=5, lr_init=0.02):
    """Subgradient method with backtracking line search on top-k peaks."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _topk_gradient(a, c, n, k=k_peaks)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        # Line search
        alpha = lr
        f_old = float(c.max())
        z_new = z.copy()
        success = False
        for ls in range(15):
            z_try = z - alpha * grad
            z_try = np.clip(z_try, -40.0, 40.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * alpha * grad_norm:
                z_new = z_try
                success = True
                break
            alpha *= 0.5

        z = z_new
        a_check = _softmax(z)
        sc = _score(a_check)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_check.copy()
            patience = 0
            lr = min(lr * 1.03, lr_init)
        else:
            patience += 1
            lr *= 0.9
            if patience > 50:
                z = np.log(np.maximum(best_a, 1e-30))
                z = z - z.mean()
                patience = 0
                lr = lr_init * 0.5
                if lr < lr_init * 0.05:
                    break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimizer:
    Phase 1: Nesterov mirror descent from parent
    Phase 2: L-BFGS with fine beta continuation
    Phase 3: Multi-scale expansion (n=8192, 16384)
    Phase 4: Stochastic restarts
    Phase 5: Structured initializations
    Phase 6: Final fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        par_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={par_sc:.10f}")
        if par_sc < best_score:
            best_seq = parent.copy()
            best_score = par_sc
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, using uniform n={n_parent}, score={best_score:.10f}")

    # --- Phase 1: Nesterov descent (200s) ---
    print(f"\n[Phase 1] Nesterov descent (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = min(t_start + 200, deadline - 600)
    a_nd, sc_nd = _nesterov_descent(parent, p1_deadline, n_steps=8000, k_peaks=8, lr_init=0.03)
    if sc_nd < best_score:
        best_seq = a_nd.copy()
        best_score = sc_nd
        print(f"  Nesterov improved to {best_score:.10f}")
    else:
        print(f"  Nesterov: {sc_nd:.10f} (no improvement from {best_score:.10f})")

    # --- Phase 1b: Subgradient line search (100s) ---
    print(f"\n[Phase 1b] Subgradient line search (t={time.monotonic()-t_start:.1f}s)")
    p1b_deadline = min(t_start + 300, deadline - 500)
    a_sg, sc_sg = _subgrad_line_search(best_seq, p1b_deadline, n_steps=2000, k_peaks=5, lr_init=0.015)
    if sc_sg < best_score:
        best_seq = a_sg.copy()
        best_score = sc_sg
        print(f"  Subgrad improved to {best_score:.10f}")

    # --- Phase 2: L-BFGS main (300s) ---
    print(f"\n[Phase 2] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(t_start + 600, deadline - 500)
    a_lb = _lbfgs_optimize(best_seq, p2_deadline, min_time_remaining=100)
    sc_lb = _score(a_lb)
    if sc_lb < best_score:
        best_seq = a_lb.copy()
        best_score = sc_lb
        print(f"  L-BFGS improved to {best_score:.10f}")
    else:
        print(f"  L-BFGS: {sc_lb:.10f} (no improvement)")

    # --- Phase 3: Multi-scale (500s) ---
    print(f"\n[Phase 3] Multi-scale expansion (t={time.monotonic()-t_start:.1f}s)")
    n_targets = [8192, 16384]
    for n_try in n_targets:
        if time.monotonic() > deadline - 300:
            break
        print(f"  Trying n={n_try}")
        a_init = _upsample(best_seq, n_try)
        # Quick Nesterov
        a_nd2 = _nesterov_descent(a_init, min(time.monotonic() + 80, deadline - 400),
                                  n_steps=2000, k_peaks=8, lr_init=0.02)
        # L-BFGS
        a_opt = _lbfgs_optimize(a_nd2[0], min(time.monotonic() + 200, deadline - 300),
                                min_time_remaining=100)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  n={n_try} improved to {best_score:.10f}")
        else:
            print(f"  n={n_try}: {sc_opt:.10f} (no improvement)")

    # --- Phase 4: Stochastic restarts (200s) ---
    print(f"\n[Phase 4] Stochastic restarts (t={time.monotonic()-t_start:.1f}s)")
    n = len(best_seq)
    for i in range(4):
        if time.monotonic() > deadline - 100:
            break
        z = np.log(np.maximum(best_seq, 1e-30))
        sigma = 0.03 * (1.0 + i * 0.5)
        z = z + sigma * rng.standard_normal(n)
        a_try = _softmax(z)
        # Quick L-BFGS
        a_opt = _lbfgs_optimize(a_try, min(time.monotonic() + 80, deadline - 60),
                                min_time_remaining=30)
        sc = _score(a_opt)
        if sc < best_score:
            best_seq = a_opt.copy()
            best_score = sc
            print(f"  Restart {i} improved to {best_score:.10f}")
        else:
            print(f"  Restart {i}: {sc:.10f}")

    # --- Phase 5: Structured inits (150s) ---
    print(f"\n[Phase 5] Structured inits (t={time.monotonic()-t_start:.1f}s)")
    for kind in ["multi_cos", "bipolar_smooth", "superflat"]:
        if time.monotonic() > deadline - 50:
            break
        a_s = _make_structured(n, kind, rng)
        a_si = _lbfgs_optimize(a_s, min(time.monotonic() + 50, deadline - 30),
                               min_time_remaining=15)
        sc_si = _score(a_si)
        if sc_si < best_score:
            best_seq = a_si.copy()
            best_score = sc_si
            print(f"  Struct {kind} improved to {best_score:.10f}")
        else:
            print(f"  Struct {kind}: {sc_si:.10f}")

    # --- Phase 6: Final fine polish (remaining) ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        n_fin = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 3e-5
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(30, int(remaining / 0.02)))
            try:
                result = minimize(
                    _smoobj, z, args=(beta_final, n_fin),
                    jac=_smograd, method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16,
                             'gtol': 1e-13, 'maxcor': 50, 'maxls': 80}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_seq = a_fin.copy()
                    best_score = sc_fin
                    print(f"  Final L-BFGS improved to {best_score:.10f}")
            except Exception:
                pass

        # Final Nesterov on very small scale
        if time.monotonic() < deadline - 3:
            a_fn, sc_fn = _nesterov_descent(best_seq, deadline - 2,
                                             n_steps=1000, k_peaks=10, lr_init=0.003)
            if sc_fn < best_score:
                best_seq = a_fn.copy()
                best_score = sc_fn
                print(f"  Final Nesterov improved to {best_score:.10f}")

    # --- Return ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
