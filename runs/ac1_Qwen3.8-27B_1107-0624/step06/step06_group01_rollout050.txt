import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_at_k(a, k):
    """Compute gradient of (a*a)_k w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        idx1 = k - j
        idx2 = j - k
        if 0 <= idx1 < n:
            val += a[idx1]
        if 0 <= idx2 < n:
            val += a[idx2]
        g[j] = val
    return g


def _grad_at_k_fast(a, k):
    """Compute gradient of (a*a)_k w.r.t. a using convolution trick."""
    n = len(a)
    # grad_j = a_{k-j} + a_{j-k} (with bounds)
    # First term: a_{k-j} for j where 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(k, n-1)
    g = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(k, n - 1)
    if lo1 <= hi1:
        g[lo1:hi1+1] += a[k-hi1:k-lo1+1][::-1]
    # Second term: a_{j-k} for j where 0 <= j-k < n, i.e., k <= j <= k+n-1, clipped to [0,n-1]
    lo2 = max(k, 0)
    hi2 = min(k + n - 1, n - 1)
    if lo2 <= hi2:
        g[lo2:hi2+1] += a[lo2-k:hi2-k+1]
    return g

def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_smoothed_grad(y, c_y, beta, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _peak_shaving(a_init, deadline, n_steps=5000):
    """Direct peak-shaving using exact subgradient of the max."""
    n = len(a_init)
    a_curr = a_init.copy() / a_init.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    eta = 0.01  # initial step size
    patience = 0
    max_patience = 200
    
    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        
        # Find all positions near the max (within 1% of max)
        threshold = c_max * 0.99
        active = np.where(c >= threshold)[0]
        
        if len(active) == 1:
            # Unique max: use exact subgradient
            k_star = active[0]
            grad = _grad_at_k_fast(a_curr, k_star)
        else:
            # Multiple near-max: use weighted average of their gradients
            weights = c[active] - threshold + 1e-10
            weights = weights / weights.sum()
            grad = np.zeros(n)
            for i, k in enumerate(active):
                grad += weights[i] * _grad_at_k_fast(a_curr, k)
        
        # Projected gradient step
        a_new = _proj_simplex(a_curr - eta * grad)
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        c_new_max = float(c_new.max())
        
        if c_new_max < best_score - 1e-12:
            best_score = c_new_max
            best_a = a_new.copy()
            patience = 0
            a_curr = a_new
            eta = min(eta * 1.05, 0.1)  # increase step slightly
        else:
            patience += 1
            if patience > max_patience:
                eta *= 0.9
                patience = 0
                if eta < 1e-6:
                    break
    
    return best_a, best_score

            # grad_z_i = 2n * sum_k w_k * [(a_{k-i}+a_{i-k}) a_i - a_i * sum_j (a_{k-j}+a_{j-k}) a_j]
            # = 2n * a_i * [sum_k w_k (a_{k-i}+a_{i-k}) - sum_k w_k sum_j (a_{k-j}+a_{j-k}) a_j]
            # The second term is: sum_k w_k * 2*sum_j a_j a_{k-j} = 2 * sum_k w_k c_k = 2 * sum(c*w)
            # So grad_z_i = 2n * a_i * [h_i - 2*sum(c*w)] where h_i = sum_k w_k (a_{k-i}+a_{i-k})
            
            h = _compute_smoothed_grad(a, c, beta, n)
            W = float(np.dot(w, c))
            grad_z = 2.0 * n * a * (h - 2.0 * W)
            
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "antisymmetric":
        # Make a_i large where a_{n-1-i} is small to reduce center correlation
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _multi_scale_optimize(a_init, deadline, n_phases=4):
    """
    Multi-scale optimization: start with coarse smoothing (large beta) and
    gradually refine. Uses a mix of FISTA and peak-shaving.
    """
    n = len(a_init)
    a_curr = a_init.copy() / a_init.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())
    
    # Beta schedule: from coarse to fine
    betas = [0.05 * max_cc, 0.02 * max_cc, 0.005 * max_cc, 0.001 * max_cc,
             0.0003 * max_cc, 0.0001 * max_cc]
    
    time_per_beta = (deadline - time.monotonic()) / len(betas)
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        phase_deadline = min(deadline - 5, time.monotonic() + time_per_beta * 1.5)
        
        # FISTA phase
        a_curr, sc = _fista_smoothed(a_curr, beta, phase_deadline, max_iter=5000)
        a_curr = a_curr / a_curr.sum()
        sc_full = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
        if sc_full < best_score:
            best_score = sc_full
            best_a = a_curr.copy()
        
        # Peak shaving phase (use exact subgradient)
        if time.monotonic() < deadline - 5:
            ps_deadline = min(deadline - 5, time.monotonic() + time_per_beta * 0.5)
            a_ps, sc_ps = _peak_shaving(a_curr, ps_deadline, n_steps=2000)
            sc_ps = float(2.0 * n * _full_convolve(a_ps, a_ps).max())
            if sc_ps < best_score:
                best_score = sc_ps
                best_a = a_ps.copy()
                a_curr = a_ps.copy()
        
        print(f"    Multi-scale phase {i}: beta={beta:.6f}, score={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")
    
    return best_a, best_score


def _hybrid_optimize(a_init, deadline):
    """
    Hybrid optimizer combining FISTA, peak-shaving, and Adam.
    Cycles between methods to avoid getting stuck.
    """
    n = len(a_init)
    a_curr = a_init.copy() / a_init.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())
    
    # Cycle: FISTA -> peak_shave -> FISTA -> Adam -> peak_shave -> ...
    cycles = [
        ("fista", 0.02 * max_cc),
        ("peak", None),
        ("fista", 0.005 * max_cc),
        ("peak", None),
        ("adam", [0.001 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc]),
        ("peak", None),
        ("fista", 0.001 * max_cc),
        ("peak", None),
        ("adam", [0.0001 * max_cc, 0.00003 * max_cc]),
        ("peak", None),
    ]
    
    time_per_cycle = (deadline - time.monotonic()) / len(cycles)
    
    for i, (method, param) in enumerate(cycles):
        if time.monotonic() > deadline - 5:
            break
        phase_dl = min(deadline - 5, time.monotonic() + time_per_cycle)
        
        if method == "fista":
            a_curr, sc = _fista_smoothed(a_curr, param, phase_dl, max_iter=3000)
            a_curr = a_curr / a_curr.sum()
            sc = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
        elif method == "peak":
            a_curr, sc = _peak_shaving(a_curr, phase_dl, n_steps=3000)
            sc = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
        elif method == "adam":
            a_curr, sc = _adam_logspace(a_curr, phase_dl, param, lr_init=0.01, max_iter_total=50000)
            sc = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
        
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
        
        print(f"    Hybrid cycle {i} ({method}): score={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800):
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")

        a0 = _make_profile(n_main, "uniform")

        best_seq = a0.copy()
        best_score = sc0
        parent = a0

    # === Phase 1: Optimize at parent length ===
    n_main = len(parent)
    print(f"\n=== Phase 1: Hybrid optimization at n={n_main} ===")
    a_work = parent.copy() / parent.sum()
    
    phase1_deadline = min(deadline - 300.0, time.monotonic() + 400.0)

    print(f"\n=== Phase 2: Multi-scale refinement at n={n_main} ===")
    phase2_deadline = min(deadline - 200.0, time.monotonic() + 250.0)
    a_opt2, sc_opt2 = _multi_scale_optimize(a_work, phase2_deadline)
    sc_full2 = _score(a_opt2)
    if sc_full2 < best_score:
        best_score = sc_full2

        print(f"  Phase 2 result: score={sc_full2:.10f}")

    # === Phase 3: Try n=8192 ===
    if n_main <= 4096 and time.monotonic() < deadline - 200.0:

        n_8k = 8192
        a_8k_init = _interp_profile(a_work, n_8k)

        if sc_8k_full < best_score:

        else:

            a_work = a_opt2 / a_opt2.sum()
    
    # === Phase 4: Multi-restart diversification ===
    if best_seq is not None and time.monotonic() < deadline - 50.0:
