def _proj_simplex(v):

def _lp_refine(a, M, tol, time_budget):
    """LP-based active set refinement.
    a: current sequence (sum=1)
    M: current max convolution value
    tol: tolerance for active set
    time_budget: seconds allowed
    Returns improved sequence."""
    n = len(a)
    c = _full_convolve(a, a)
    best_score = _score_norm(a)
    best_a = a.copy()
    t_start = time.monotonic()
    
    iter_count = 0
    while time.monotonic() - t_start < time_budget:
        t_iter_start = time.monotonic()
        # Update active set
        c = _full_convolve(a, a)
        M_curr = float(c.max())
        active = np.where(c >= M_curr - tol * M_curr)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Build LP
        # max e^T b s.t. (a*b)[k] <= M_curr for k in active, b >= 0
        # Constraint: sum_j A[k,j] * b[j] <= M_curr
        # A[k,j] = a[j] if 0 <= k-j < n, else 0
        #         = (a * delta_k)[j] where delta_k is 1 at position k
        
        # More efficiently: A[k, j] = a[j] * I(0 <= k-j < n)
        # For fixed k: j ranges from max(0, k-n+1) to min(n-1, k)
        # A[k, j] = a[j] for j in that range
        
        # Build constraint matrix
        n_active = len(active)
        # A is n_active x n
        A = np.zeros((n_active, n))
        for idx, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            A[idx, j_start:j_end+1] = a[j_start:j_end+1]
        
        b_var = cp.Variable(n, non_negative=True)
        objective = cp.Maximize(cp.sum(b_var))
        constraints = [A @ b_var <= M_curr]
        
        prob = cp.Problem(objective, constraints)
        
        # Solve with time limit
        solver = cp.ECOS  # or 'GLOP', 'CBC'
        try:
            prob.solve(solver=solver, time_limit=max(1, int(time_budget - (time.monotonic() - t_start))))
        except Exception:
            break
        
        if b_var.value is None:
            break
        
        b_opt = np.asarray(b_var.value).flatten()
        b_opt = np.maximum(b_opt, 0)
        b_sum = float(b_opt.sum())
        
        if b_sum < 1e-10:
            break
        
        b_new = b_opt / b_sum
        
        # Line search
        best_t = 0
        best_sc = best_score
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            a_trial = (1 - t) * a + t * b_new
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0:
            # Try smaller steps
            for t in [0.005, 0.001, 0.0001]:
                a_trial = (1 - t) * a + t * b_new
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        
        if best_t > 0:
            a = best_a_checkpoint = (1 - best_t) * a + best_t * b_new
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_sc < best_score:
                best_score = best_sc
                best_a = a.copy()
        else:
            # No improvement, tighten tolerance or break
            tol *= 0.5
            if tol < 1e-12:
                break
        
        iter_count += 1
        if iter_count % 50 == 0:
            print(f"  LP iter {iter_count}: score={best_score:.10f}, |A|={len(active)}, t={time.monotonic()-t_start:.1f}s")
    
    return best_a, best_score, iter_count

Hybrid optimizer combining FISTA, exact-subgradient mirror descent,
LP-based active-set refinement, and coordinate descent for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Improves on previous 1.505780 by more aggressive continuation and LP-based refinement.
"""
import numpy as np
import time
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n_len if False else out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / max(beta, 1e-15)
    step = 1.0 / L
    if step > 1.0:
        step = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / max(beta, 1e-15))
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _mirror_descent_exact(a_init, deadline, max_iter):
    """Mirror descent using exact subgradient (uniform over active set)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    lr_decay = 2.0 / np.sqrt(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Tight active set: within 0.1% of max
        tol = 0.001 * c_max
        active = np.where(c >= c_max - tol)[0]
        w = np.zeros(len(c))
        w[active] = 1.0 / len(active)
        grad = _compute_grad(a, w)
        lr = lr_decay / np.sqrt(it + 1)
        z = z - lr * grad
        a = _softmax_compute(z)
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _lp_refine(a_init, deadline, max_iter=5000):
    """LP-based active set refinement.
    At each step: solve max sum(b) s.t. (a*b)[k] <= M for active k, b>=0.
    Then line search along (a, b/sum(b)).
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    tol = 0.005  # initial tolerance for active set (fraction of max)
    t_start = time.monotonic()
    no_improve_count = 0
    step_shrink = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M_curr = float(c.max())
        if M_curr < 1e-15:
            break

        # Active set
        active = np.where(c >= M_curr - tol * M_curr)[0]
        # Limit active set size for speed
        if len(active) > 60:
            # Keep only the top 60
            idx = np.argsort(c[active])[::-1][:60]
            active = active[idx]

        # Build LP constraint matrix
        n_act = len(active)
        A = np.zeros((n_act, n))
        for idx, k in enumerate(active):
            j_start = max(0, k - n + 1)
            j_end = min(n, k + 1)
            A[idx, j_start:j_end] = a[j_start:j_end]

        # Solve LP: max sum(b) s.t. A @ b <= M_curr, b >= 0
        b_var = cp.Variable(n, non_negative=True)
        obj = cp.Maximize(cp.sum(b_var))
        cons = [A @ b_var <= M_curr]
        prob = cp.Problem(obj, cons)

        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS)
                except Exception:
                    break

        if b_var.value is None:
            break

        b_opt = np.asarray(b_var.value, dtype=np.float64).ravel()
        b_opt = np.maximum(b_opt, 0)
        b_sum = float(b_opt.sum())

        if b_sum < 1e-10:
            break

        b_dir = b_opt / b_sum

        # Line search
        improved = False
        for t_mult in [step_shrink, step_shrink * 0.5, step_shrink * 0.25,
                       step_shrink * 0.1, step_shrink * 0.05]:
            t_val = min(1.0, max(0.01, t_mult))
            a_trial = (1.0 - t_val) * a + t_val * b_dir
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial = a_trial / s_t
            sc = _score_norm(a_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                break
            elif sc < _score_norm(a):
                # Accept even if not new best
                a = a_trial
                improved = True
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                break

        if improved:
            no_improve_count = 0
            step_shrink = min(1.0, step_shrink * 1.1)
        else:
            no_improve_count += 1
            step_shrink *= 0.7
            tol *= 0.6
            if no_improve_count > 30 or tol < 1e-10:
                # Try wider active set
                tol = 0.01
                no_improve_count = 0
                if it > 200:
                    break

        if it % 100 == 0:
            print(f"  LP iter {it}: score={best_score:.10f}, |A|={len(active)}, "
                  f"tol={tol:.4f}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_score


def _coord_descent(a_init, deadline, max_iter=200000):
    """Coordinate descent: optimize one coordinate at a time on the simplex.
    For coordinate j, the optimal a[j] (given others) that minimizes max conv
    while maintaining sum=1 is found by 1D search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    order = np.arange(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Random coordinate
        j = order[it % n] if it < n * 2 else rng.integers(0, n)

        # Current value
        aj = a[j]
        rest_sum = 1.0 - aj

        if rest_sum <= 0:
            continue

        # Compute the convolution terms that depend on a[j]
        # c[k] changes by: 2*(x - aj)*a[k-j] for k where both indices valid
        # and (x^2 - aj^2) for k = 2j
        # We want to find x in [0, 1] that minimizes max_k c_new[k]
        # subject to a_new[j] = x and renormalization

        # Actually, let's do a simpler approach: try a few candidate values
        # and pick the best one.
        c = _full_convolve(a, a)
        M_curr = float(c.max())

        # Try: slightly increase, slightly decrease, set to 0
        candidates = [0.0, aj * 0.5, aj * 0.1, aj * 2.0, aj * 5.0,
                      aj + rest_sum * 0.01, aj - aj * 0.01]
        candidates = [max(0.0, min(1.0, c_val)) for c_val in candidates]
        # Add a random perturbation
        candidates.append(max(0.0, min(1.0, aj + rng.normal(0, aj * 0.1 + 0.001))))

        best_x = aj
        best_sc = M_curr

        for x in candidates:
            if abs(x - aj) < 1e-12:
                continue
            a_test = a.copy()
            a_test[j] = x
            s_t = a_test.sum()
            if s_t < 1e-10:
                continue
            a_test = a_test / s_t
            c_test = _full_convolve(a_test, a_test)
            sc_test = float(c_test.max())
            if sc_test < best_sc:
                best_sc = sc_test
                best_x = x

        if best_x != aj:
            a[j] = best_x
            s = a.sum()
            a = a / s
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _adam_fine(a_init, deadline, beta, lr_init, max_iter):
    """Adam in log-space with small beta for fine optimization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / max(beta, 1e-15))
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)
        z = z - update
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))

            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_cd

    # === Phase 6: Final LP polish (remaining time) ===
    print("Phase 6: Final LP polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        remaining = deadline - time.monotonic() - 2
        if remaining > 30:
            a_pol, _ = _lp_refine(a_pol, time.monotonic() + remaining, max_iter=5000)
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Final LP: score={sc_pol:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
