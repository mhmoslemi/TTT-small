"""
Multi-phase optimizer targeting sub-1.5027 score for non-negative step functions.
Uses FISTA, Adam log-space, active-set refinement, and golden-section coordinate descent.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, clip_norm=0.2):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        phase_max_iter = min(max_iter_total, max(5000, int((deadline - time.monotonic()) * 2000)))
        while it_in_phase < phase_max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -600, 600, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > clip_norm:
                update = update * (clip_norm / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 80 == 0:

                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc

            if it_in_phase % 2000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_optimize(a_init, K, deadline, max_iter=50000, lr=0.01):
    """Optimize using only top-K peaks in the gradient (smooth quadratic objective)."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    lr_curr = lr

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        # Get top-K peak indices
        if K <= len(c):
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(len(c))
        w = np.zeros_like(c)
        w[top_idx] = 1.0 / len(top_idx)

        a_rev = a[::-1]
        wc = _convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * (a * (h - W))

        a_next = a - lr_curr * grad
        a_next = np.maximum(a_next, 0.0)
        s_next = a_next.sum()
        if s_next < 1e-30:
            a_next = a
        else:
            a_next = a_next / s_next

        sc = _score_norm(a_next)
        if sc < best_sc:
            best_sc = sc
            best_a = a_next.copy()
            a = a_next
        else:
            lr_curr *= 0.8
            if lr_curr < 1e-10:
                break

        if lr_curr < 0.001:
            lr_curr = min(lr, lr_curr * 1.05)

    return best_a, best_sc


def _golden_coord_descent(a_init, deadline, max_rounds=300, n_trials=8):
    """Coordinate descent with golden-section 1D optimization per coordinate."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)
    golden = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            a_save = a[j]
            # Search range for delta
            lo = -0.5 * a_save
            hi = 1.0 * a_save
            # Golden section search
            c_pt = hi - golden * (hi - lo)
            d_pt = lo + golden * (hi - lo)

            def _eval_delta(d):
                a[j] = max(0.0, a_save + d)
                s = a.sum()
                if s < 1e-30:
                    return np.inf
                return _score_norm(a / s)

            fc = _eval_delta(c_pt - a_save)
            fd = _eval_delta(d_pt - a_save)
            a[j] = a_save

            for _ in range(n_trials):
                if time.monotonic() > deadline:
                    break
                if fc < fd:
                    hi = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = hi - golden * (hi - lo)
                    fc = _eval_delta(c_pt - a_save)
                    a[j] = a_save
                else:
                    lo = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = lo + golden * (hi - lo)
                    fd = _eval_delta(d_pt - a_save)
                    a[j] = a_save

            # Take the best of the two final points
            a[j] = lo + 0.5 * (hi - lo)
            s = a.sum()
            if s > 1e-30:
                a = a / s
                sc = _score_norm(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = a_save / (a.sum())  # revert
            else:
                a[j] = a_save

        if not improved and r > 30:
            break

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 5: Multi-restart with diverse perturbations ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end - 60:
        n_restarts = 6
        per_restart = (phase5_end - time.monotonic() - 20) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 15:
                break
            if i % 3 == 0:
                p_start = _perturb(best_seq, 0.015 + 0.003 * i, rng)
            elif i % 3 == 1:
                sz = max(2, n_main // 2048)
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(best_seq, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                # Block permutation
                block = n_main // 8
                p_start = best_seq.copy()
                order = list(range(n_main // block))
                rng.shuffle(order)
                new_p = np.zeros(n_main)
                for old_idx, new_idx in enumerate(order):
                    new_p[new_idx*block:(new_idx+1)*block] = p_start[old_idx*block:(old_idx+1)*block]
                p_start = np.maximum(new_p, 0.0)
                p_start = p_start / p_start.sum()

            per_t = max(5.0, per_restart * 0.6)
            bs = [0.0003 * max_cc, 0.0001 * max_cc, 3e-05 * max_cc,
                  1e-05 * max_cc, 5e-06 * max_cc]
            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + per_t,
                bs, lr_init=0.005, max_iter_total=100000, clip_norm=0.1
            )
            # Quick active-set polish
            per_t2 = max(3.0, per_restart * 0.3)
            p_best, sc_pb = _active_set_optimize(p_best, 10, time.monotonic() + per_t2, max_iter=20000, lr=0.003)
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Try n=8192 ===
    print("Phase 6: n=8192 exploration")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end - 60 and best_seq is not None and len(best_seq) < 8192:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase6_end - 30:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            per_t = min(60.0, phase6_end - time.monotonic() - 10)
            if per_t > 20:
                a_big_opt, sc_bo = _fista_smoothed(a_big, 0.005 * max_cb, time.monotonic() + per_t * 0.4, 30000)
                a_big_opt = a_big_opt / a_big_opt.sum()
                sc_bo_full = _score(a_big_opt)
                if sc_bo_full < best_score:
                    best_score = sc_bo_full
                    best_seq = a_big_opt.copy()
                    print(f"  FISTA n=8192: score={sc_bo_full:.10f}")

                if time.monotonic() < phase6_end - 15:
                    c_bo = _convolve(a_big_opt, a_big_opt)
                    max_cbo = float(c_bo.max())
                    bs_big = [0.0005 * max_cbo, 0.0002 * max_cbo, 0.0001 * max_cbo,
                              5e-05 * max_cbo, 2e-05 * max_cbo]
                    per_t2 = phase6_end - time.monotonic() - 10
                    if per_t2 > 15:
                        a_big_opt, sc = _adam_logspace(
                            a_big_opt, time.monotonic() + per_t2,
                            bs_big, lr_init=0.004, max_iter_total=50000, clip_norm=0.1
                        )
                        sc_full = _score(a_big_opt)
                        if sc_full < best_score:
                            best_score = sc_full
                            best_seq = a_big_opt.copy()
                            print(f"  Adam n=8192: score={sc_full:.10f}")

    # === Phase 7: Final ultra-fine polish ===
    print("Phase 7: Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        n_pol = len(a_pol)

        # Ultra-fine Adam
        per_t = min(60.0, deadline - time.monotonic() - 10)
        if per_t > 15:
            betas_ult = [1e-05 * max_cp, 5e-06 * max_cp, 2e-06 * max_cp,
                         1e-06 * max_cp, 5e-07 * max_cp]
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t * 0.6,
                betas_ult, lr_init=0.001, max_iter_total=200000, clip_norm=0.05
            )
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Ultra Adam: score={sc_pol_full:.10f}")

        # Active-set final
        if time.monotonic() < deadline - 10:
            a_pol2 = best_seq.copy()
            a_pol2 = a_pol2 / a_pol2.sum()
            per_t2 = min(30.0, deadline - time.monotonic() - 5)
            if per_t2 > 10:
                a_pol2, sc_as2 = _active_set_optimize(a_pol2, 5, time.monotonic() + per_t2, max_iter=20000, lr=0.002)
                sc_as2_full = _score(a_pol2)
                if sc_as2_full < best_score:
                    best_score = sc_as2_full
                    best_seq = a_pol2.copy()
                    print(f"  Ultra Active-set: score={sc_as2_full:.10f}")

        # Golden-section final
        if time.monotonic() < deadline - 10 and len(best_seq) <= 8192:
            a_cd2 = best_seq.copy()
            a_cd2 = a_cd2 / a_cd2.sum()
            per_t3 = min(30.0, deadline - time.monotonic() - 3)
            if per_t3 > 10:
                a_cd2, sc_cd2 = _golden_coord_descent(a_cd2, time.monotonic() + per_t3, max_rounds=50, n_trials=8)
                sc_cd2_full = _score(a_cd2)
                if sc_cd2_full < best_score:
                    best_score = sc_cd2_full
                    best_seq = a_cd2.copy()
                    print(f"  Ultra Coord: score={sc_cd2_full:.10f}")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
