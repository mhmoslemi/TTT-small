"""
Active-set cutting plane + L-BFGS hybrid optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses LP-based active set refinement,
adaptive tolerance, multi-n exploration, and coordinate descent polish.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_lp(a0, active_lags, deadline):
    """Solve the active-set LP to find improved a."""
    n = len(a0)
    c_conv = _full_convolve(a0, a0)

    obj = np.zeros(n + 1)
    obj[-1] = 1.0

    n_active = len(active_lags)
    A_ub = np.zeros((n_active, n + 1))
    b_ub = np.zeros(n_active)

    for idx, k in enumerate(active_lags):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[idx, i] = 2.0 * a0[k - i]
        A_ub[idx, -1] = -1.0
        b_ub[idx] = c_conv[k]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0.0, None)] * n + [(None, None)]

    try:
        result = linprog(obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': max(1.0, deadline - time.monotonic())})
        if result.success:
            return result.x[:n]
    except Exception:
        pass
    return a0.copy()


def _active_set_optimize(a_init, deadline, tol_frac=0.005, max_iter=200):
    """Active-set cutting plane optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    no_improve_count = 0

        t0 = time.monotonic()

        c_max = float(c.max())

def _lbfgs_optimize(a_init, deadline, min_time_remaining=10.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 20))))

        try:

                lambda z_local, b=beta: _smoobj(z_local, b, n),
                z,
                jac=lambda z_local, b=beta: _smograd(z_local, b, n),
                method='L-BFGS-B',

                         'maxcor': 50, 'maxls': 100}

            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_descent(a_init, deadline, n_passes=100):
    """Coordinate descent: try adjusting pairs of elements."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        # Random coordinate pair update
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Try moving mass from i to j (and vice versa)
        for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01]:
            a_try = a.copy()
            a_try[i] += delta
            a_try[j] -= delta
            if a_try[i] < 0 or a_try[j] < 0:
                continue
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-12:
                a = a_try
                best_a = a_try.copy()
                best_sc = sc_try
                break

    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient polishing with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)

                lr *= 0.88
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage(a, deadline):
    """Try zeroing out smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.005, 0.01, 0.02, 0.05, 0.10]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)**0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = a**1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid active-set + L-BFGS optimizer.
    Uses LP-based active set refinement for large jumps,
    L-BFGS for smooth convergence, and coordinate descent for polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        par_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
        update_best(parent, par_sc, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Active-set cutting plane from parent ---
    print(f"\n[Phase 1] Active-set optimization (t={time.monotonic() - t_start:.1f}s)")
    as_deadline = min(deadline - 600.0, t_start + 600.0)
    a_as, sc_as = _active_set_optimize(parent, as_deadline, tol_frac=0.005, max_iter=300)
    update_best(a_as, sc_as, "active_set_1")

    # Second pass with tighter tolerance
    if time.monotonic() < deadline - 500:
        a_as2, sc_as2 = _active_set_optimize(a_as, min(deadline - 500, time.monotonic() + 200),
                                             tol_frac=0.002, max_iter=200)
        update_best(a_as2, sc_as2, "active_set_2")

    # --- Phase 2: L-BFGS refinement ---
    print(f"\n[Phase 2] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_lb = _lbfgs_optimize(best_seq, min(deadline - 300, time.monotonic() + 300),
                               min_time_remaining=200.0)
        update_best(a_lb, _score(a_lb), "lbfgs_1")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 200, time.monotonic() + 120),
                                      n_max_steps=2000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # --- Phase 3: Shrinkage + re-optimize ---
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 200)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score + 0.001:
            a_shr2 = _active_set_optimize(a_shr, min(deadline - 150, time.monotonic() + 100),
                                          tol_frac=0.003, max_iter=100)
            update_best(a_shr2, a_shr2[1], "shrinkage_as")
            a_shr3 = _lbfgs_optimize(a_shr2[0], min(deadline - 100, time.monotonic() + 60),
                                     min_time_remaining=50.0)
            update_best(a_shr3, _score(a_shr3), "shrinkage_lbfgs")

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "hann")
        else:
            a_init = parent.copy() if n_parent == n_try else _make_profile(n_try, "hann")

        sc_init = _score(a_init)
        # Quick active set
        a_opt, sc_opt = _active_set_optimize(a_init, min(deadline - 80, time.monotonic() + 60),
                                             tol_frac=0.005, max_iter=80)
        update_best(a_opt, sc_opt, f"n_{n_try}_as")

        # L-BFGS polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2 = _lbfgs_optimize(a_opt, min(deadline - 30, time.monotonic() + 40),
                                     min_time_remaining=20.0)
            update_best(a_opt2, _score(a_opt2), f"n_{n_try}_lbfgs")

    # --- Phase 5: Diverse restarts ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_restarts = 3
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            if i == 0:
                z = z + 0.03 * rng.standard_normal(n)
            elif i == 1:
                # Fourier perturbation
                t = np.arange(n)
                for _ in range(3):
                    freq = rng.integers(1, max(2, n // 4))
                    amp = 0.02 * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            else:
                # Block swap
                block_size = n // 8
                b1 = rng.integers(0, n - 2 * block_size)
                b2 = rng.integers(0, n - 2 * block_size)
                if abs(b1 - b2) > block_size:
                    tmp = z[b1:b1 + block_size].copy()
                    z[b1:b1 + block_size] = z[b2:b2 + block_size]
                    z[b2:b2 + block_size] = tmp
            z = z - z.mean()
            a_pert = _softmax(z)
            sc_pert = _score(a_pert)

            a_r = _active_set_optimize(a_pert, min(deadline - 20, time.monotonic() + 40),
                                       tol_frac=0.004, max_iter=60)
            update_best(a_r, a_r[1], f"restart_{i}_as")

            if time.monotonic() < deadline - 15:
                a_r2 = _lbfgs_optimize(a_r[0], min(deadline - 10, time.monotonic() + 20),
                                       min_time_remaining=5.0)
                update_best(a_r2, _score(a_r2), f"restart_{i}_lbfgs")

    # --- Phase 6: Final intense polish on best ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight active set
        a_fin1, sc_fin1 = _active_set_optimize(best_seq, min(deadline - 3, time.monotonic() + 60),
                                               tol_frac=0.001, max_iter=100)
        update_best(a_fin1, sc_fin1, "final_as_tight")

        # Final L-BFGS with very small beta
        if time.monotonic() < deadline - 3:
            n = len(a_fin1[0])
            c_best = _full_convolve(a_fin1[0], a_fin1[0])
            mb = float(c_best.max())
            z = np.log(np.maximum(a_fin1[0], 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0003
            remaining = deadline - 3 - time.monotonic()
            if remaining > 5:
                max_iter = min(2000, max(50, int(remaining / 0.03)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                 'maxcor': 50, 'maxls': 100}
                    )
                    a_fin2 = _softmax(result.x)
                    update_best(a_fin2, _score(a_fin2), "final_lbfgs_tiny_beta")
                except Exception:
                    pass

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 2,
                                              n_max_steps=500, lr_init=0.005)
            update_best(a_fin3, sc_fin3, "final_sg")

        # Coordinate descent
        if time.monotonic() < deadline - 2:
            a_cd, sc_cd = _coordinate_descent(best_seq, deadline - 2, n_passes=200)
            update_best(a_cd, sc_cd, "coord_desc")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
