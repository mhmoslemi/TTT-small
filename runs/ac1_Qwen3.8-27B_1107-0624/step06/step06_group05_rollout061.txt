"""
Improved optimizer using adaptive top-K smoothing FISTA, multi-n exploration,
and coordinate-level refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad(a, w):
    """Compute gradient of weighted sum of convolution values.
    grad_i = 2 * sum_j w_{i+j} * a_j
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _topk_smooth_weights(c, beta, K):
    """Compute softmax weights only over the top K convolution values."""
    n_c = len(c)
    if K >= n_c:
        # Use all
        c_max = float(c.max())
        w = np.exp((c - c_max) / beta)
        s = float(w.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / s
        return w
    
    # Find top K
    if K < n_c:
        part = np.argpartition(c, -K)[-K:]
        top_k_vals = c[part]
        c_max = float(top_k_vals.max())
        # Softmax over top K
        w_top = np.exp((top_k_vals - c_max) / beta)
        s = float(w_top.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            return w
        w_top = w_top / s
        # Place back
        w = np.zeros_like(c)
        w[part] = w_top
    return w

    n = len(a_init)

            s = float(w.sum())

                return 2.0 * n * cm

        else:

            s = float(w_top.sum())

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        w = _topk_smooth_weights(c, beta, K)
        grad_a = _compute_grad(a, w)
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        a_best = _softmax_compute(best_z)
        return a_best, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 20)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:

            best_z = z_opt.copy()
    except Exception as e:

    a_best = _softmax_compute(best_z)

def _upsample(a, n_new):
    """Upsample a sequence to a new length using linear interpolation."""
    n_old = len(a)
    if n_new == n_old:
        return a.copy()
    if n_new < n_old:
        # Downsample
        idx = np.linspace(0, n_old - 1, n_new)
        return np.interp(idx, np.arange(n_old), a)
    # Upsample
    idx = np.linspace(0, n_old - 1, n_new)
    return np.interp(idx, np.arange(n_old), a)

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):

    return w


def _fista_v2(a_init, deadline, K, beta_init_frac, use_aggressive_step):
    """FISTA with corrected step size and top-K adaptive smoothing."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = np.maximum(a_curr, 0)
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c0 = float(c_init.max())
    beta = beta_init_frac * max_c0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    stall = 0

    for it in range(3000000):
        if time.monotonic() > deadline:
            break
        if stall > 3000:
            break

        # Adaptive beta decay
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.985)
            if stall > 1000:
                beta = min(max_c0 * 0.05, beta * 2.0)
                stall = 0
                t_val = 1.0
                a_prev = a_curr.copy()

        # Step size: use larger step for corrected analysis
        if use_aggressive_step:
            step = 1.0 / max(4.0, 8.0 / max(beta, 1e-12))
        else:
            step = 1.0 / (4.0 * n)

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, beta, K)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_next.copy()
                stall = 0
            else:
                stall += 1

        # Momentum reset if oscillating
        if it % 200 == 0:
            diff = a_curr - a_prev
            if np.dot(diff, a_next - a_curr) < 0:
                t_val = 1.0
                a_prev = a_curr.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_f = _score_norm(a_curr)
    if sc_f < best_score:
        best_score = sc_f
        best_a = a_curr.copy()
    return best_a, best_score


def _fista_original(a_init, deadline, K, beta_init_frac):
    """Original FISTA with conservative step (as in previous SOTA)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = np.maximum(a_curr, 0)
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c0 = float(c_init.max())
    beta = beta_init_frac * max_c0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(3000000):
        if time.monotonic() > deadline:
            break
        if it % 100 == 0:
            beta = max(1e-12, beta * 0.97)
            L = 4.0 * n / beta
            step = 1.0 / L

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, beta, K)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_f = _score_norm(a_curr)
    if sc_f < best_score:
        best_score = sc_f
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_v2(a_init, deadline, K, beta_frac, lr_init=0.01):
    """Adam in log-space with top-K smoothing and adaptive lr."""
    n = len(a_init)
    a_init = np.maximum(a_init, 0)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    c_init = _full_convolve(a_init, a_init)
    max_c0 = float(c_init.max())
    beta = beta_frac * max_c0
    best_score = _score_norm(a_init)
    best_a = a_init.copy()

    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        w = _topk_weights(c, beta, K)
        grad = _compute_grad(a, w)
        grad = grad - grad.mean()

        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 0.3:
            update = update * (0.3 / un)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 200 == 0:
            beta = max(1e-14, beta * 0.96)
            lr = max(0.001, lr * 0.99)

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_f = _softmax_compute(z)
    sc_f = _score_norm(a_f)
    if np.isfinite(sc_f) and sc_f < best_score:
        best_score = sc_f
        best_a = a_f.copy()
    return best_a, best_score


def _lbfgs_v2(a_init, deadline, K, beta_frac):
    """L-BFGS-B with top-K smoothing."""
    from scipy.optimize import minimize
    n = len(a_init)
    a_init = np.maximum(a_init, 0)
    a_init = a_init / a_init.sum()
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.mean()
    c_init = _full_convolve(a_init, a_init)
    beta = beta_frac * float(c_init.max())
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        w = _topk_weights(c, beta, K)
        return 2.0 * n * float(np.dot(w, c))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        w = _topk_weights(c, beta, K)
        grad_a = _compute_grad(a, w)
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 25)), 'gtol': 1e-14, 'maxls': 25}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    return _softmax_compute(best_z), best_score


def _upsample(a, n_new):
    n_old = len(a)
    if n_new == n_old:
        return a.copy()
    idx = np.linspace(0, n_old - 1, n_new)
    return np.interp(idx, np.arange(n_old), a)


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _optimize_one(a_init, t_start, deadline, label):
    """Run full optimization pipeline on one configuration."""
    n = len(a_init)
    a_init = np.maximum(a_init, 0)
    a_init = a_init / a_init.sum()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    print(f"  [{label}] start: n={n}, score={best_sc:.10f}, t={time.monotonic()-t_start:.1f}s")

    total_time = deadline - t_start
    # Phase A: FISTA aggressive (40%)
    pa_end = t_start + total_time * 0.35
    if time.monotonic() < pa_end - 3:
        a_a, _ = _fista_original(a_init, pa_end, K=16, beta_init_frac=0.015)
        sc_a = _score(a_a)
        if sc_a < best_sc:
            best_sc = sc_a
            best_a = a_a.copy()
            print(f"  [{label}] FISTA-orig: score={sc_a:.10f}, t={time.monotonic()-t_start:.1f}s")
        a_work = a_a

        # Phase A2: FISTA aggressive step (10%)
        pa2_end = min(deadline, t_start + total_time * 0.45)
        if time.monotonic() < pa2_end - 3:
            a_a2, _ = _fista_v2(a_work, pa2_end, K=10, beta_init_frac=0.005, use_aggressive_step=True)
            sc_a2 = _score(a_a2)
            if sc_a2 < best_sc:
                best_sc = sc_a2
                best_a = a_a2.copy()
                print(f"  [{label}] FISTA-aggr: score={sc_a2:.10f}, t={time.monotonic()-t_start:.1f}s")
            a_work = a_a2
    else:
        a_work = a_init.copy()

    # Phase B: Adam (25%)
    pb_end = min(deadline, t_start + total_time * 0.70)
    if time.monotonic() < pb_end - 3:
        a_b, _ = _adam_v2(a_work, pb_end, K=8, beta_frac=0.0005, lr_init=0.008)
        sc_b = _score(a_b)
        if sc_b < best_sc:
            best_sc = sc_b
            best_a = a_b.copy()
            print(f"  [{label}] Adam: score={sc_b:.10f}, t={time.monotonic()-t_start:.1f}s")
        a_work = a_b
    else:
        a_work = best_a.copy()

    # Phase C: L-BFGS (20%)
    pc_end = min(deadline, t_start + total_time * 0.92)
    if time.monotonic() < pc_end - 3:
        a_c, _ = _lbfgs_v2(a_work, pc_end, K=5, beta_frac=0.00005)
        sc_c = _score(a_c)
        if sc_c < best_sc:
            best_sc = sc_c
            best_a = a_c.copy()
            print(f"  [{label}] L-BFGS: score={sc_c:.10f}, t={time.monotonic()-t_start:.1f}s")
        a_work = a_c

    # Phase D: Final tiny Adam polish (8%)
    if time.monotonic() < deadline - 3:
        a_d, _ = _adam_v2(a_work, deadline, K=3, beta_frac=0.00001, lr_init=0.002)
        sc_d = _score(a_d)
        if sc_d < best_sc:
            best_sc = sc_d
            best_a = a_d.copy()
            print(f"  [{label}] Final polish: score={sc_d:.10f}, t={time.monotonic()-t_start:.1f}s")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 18.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    best_seq = None
    best_score = np.inf

    if parent is not None:
        n_main = len(parent)
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = parent.copy()
        best_score = sc0
    else:
        n_main = 4096
        best_seq = np.ones(n_main) / n_main
        best_score = _score(best_seq)
        print(f"No parent, uniform n={n_main}, score={best_score:.10f}")

    # Build list of configurations to try
    configs = []

    # Config 1: parent at native length (primary)
    if parent is not None:
        configs.append(("native", parent.copy(), n_main, 0.40))

    # Config 2: upsample to 8192
    n_up = 8192
    if n_main < n_up:
        a_up = _upsample(parent if parent is not None else best_seq, n_up)
        a_up = np.maximum(a_up, 0)
        a_up = a_up / a_up.sum()
        configs.append(("upsample_8k", a_up, n_up, 0.30))

    # Config 3: perturbed native
    if parent is not None:
        a_pert = _perturb(parent, 0.01, rng)
        configs.append(("perturbed", a_pert, n_main, 0.15))

    # Config 4: downsample to 2048 (in case smaller n works better with optimization)
    if n_main > 2048:
        a_dn = _upsample(parent if parent is not None else best_seq, 2048)
        a_dn = np.maximum(a_dn, 0)
        a_dn = a_dn / a_dn.sum()
        configs.append(("down_2k", a_dn, 2048, 0.10))
    elif n_main < 2048:
        a_up2 = _upsample(parent if parent is not None else best_seq, 2048)
        a_up2 = np.maximum(a_up2, 0)
        a_up2 = a_up2 / a_up2.sum()
        configs.append(("up_2k", a_up2, 2048, 0.10))

    # Allocate time proportionally
    total_frac = sum(c[3] for c in configs)
    available_time = deadline - t_start
    print(f"\nConfigs: {[(c[0], c[2]) for c in configs]}, total time available: {available_time:.1f}s")

    for ci, (name, a_init, n_c, frac) in enumerate(configs):
        if time.monotonic() > deadline - 10:
            break
        cfg_time = available_time * frac / total_frac
        cfg_deadline = min(deadline - 5, t_start + cfg_time)
        print(f"\n--- Config: {name} (n={n_c}, budget={cfg_time:.1f}s) ---")
        a_best, sc_best = _optimize_one(a_init, t_start, cfg_deadline, name)
        sc_full = _score(a_best)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_best.copy()
            print(f"  NEW BEST: {name} score={sc_full:.10f}")

    # Final: if time remains, do one more polish on the best
    if best_seq is not None and time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            print(f"\nFinal polish (budget={remaining:.1f}s)")
            a_final, _ = _adam_v2(best_seq, t_start + remaining, K=3, beta_frac=0.000005, lr_init=0.001)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  Final polish improved: {sc_final:.10f}")

    print(f"\n=== DONE: best_score={best_score:.10f}, n={len(best_seq)} ===")

    # Verify with evaluator
    try:
        ev = float(evaluate_sequence([float(x) for x in best_seq]))
        print(f"Evaluator confirms: {ev:.10f}")
    except Exception as e:
        print(f"Eval error: {e}")

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(4096) / 4096
    return [float(x) for x in final]
