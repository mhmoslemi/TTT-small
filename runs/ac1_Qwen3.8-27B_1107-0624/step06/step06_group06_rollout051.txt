import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so score = 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):

    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)

        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    best_a = a.copy()
    best_sc = _score_normalized(a)

        c_max = float(c.max())
        if c_max < 1e-30:

        active = c >= (c_max - tol)

        if n_active < 2:
            idx = np.argsort(c)[-3:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 3

        # Compute pressure: p_i = sum_{k in active} a_{k-i} (via cross-correlation)
        # (Hv)_i = sum_{k in S} a_{k-i} = conv(a, active_reversed)[2n-2-i] reversed
        # More directly: p = conv(active, a_reversed)[n-1:2n-1]
        a_rev = a[::-1]
        p = _full_convolve(active.astype(float), a_rev)
        p = p[n - 1:2 * n - 1]

        # The KKT pressure includes the diagonal correction
        # p_i = sum_{k in S} 2*a_{k-i} - a_i * 1_{2i in S}
        # For simplicity, use the approximate pressure
        p = np.maximum(p, 1e-30)

        # Target: make p_i proportional to a_i^{-gamma} (water-filling in log space)
        # Update: a_i <- a_i * (mean(p) / p_i)^gamma
        p_mean = np.mean(p)
        ratio = p_mean / p
        a = a * np.power(ratio, gamma)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if it % 20 == 0:
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc_final = _score_normalized(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient descent in log-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    best_patience_sc = best_sc

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 40 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                improved = best_sc - sc
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.02, lr_init)
                best_patience_sc = best_sc
            else:
                patience += 1
                lr *= 0.9
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation on log-space parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.02)

    # Aggressive beta schedule: more steps at small beta
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _coordinate_polish(a, deadline, n_rounds=100):
    """
    Coordinate descent: for each coordinate, try adjusting it to reduce the max.
    Uses the analytical insight that optimal a_i (given others fixed) satisfies
    the KKT condition.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_normalized(a)
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Focus on coordinates that contribute most to the peak
        # Contribution of a_i to (a*a)_k is a_i * a_{k-i}
        # Total "peak contribution" of a_i: sum_{k near max} a_i * a_{k-i}
        tol = 0.05 * c_max
        active = c >= (c_max - tol)
        a_rev = a[::-1]
        contrib = _full_convolve(active.astype(float), a_rev)
        contrib = contrib[n - 1:2 * n - 1] * a

        # Pick top coordinates to modify
        n_try = min(n, max(10, n // 50))
        top_idx = np.argsort(contrib)[-n_try:]
        np.random.shuffle(top_idx)
        top_idx = np.array(rng.permutation(top_idx[:min(len(top_idx), n_try//5)]))

        improved = False
        for i in top_idx:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            c_old = _full_convolve(a, a).max()

            # Try: reduce a_i by various factors
            for factor in [0.5, 0.7, 0.9, 1.1, 1.3]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                c_try = _full_convolve(a, a).max()
                if c_try < c_old - 1e-15:
                    improved = True
                else:
                    a[i] = old_val
                    break

        if improved:
            s = a.sum()
            if s > 1e-15:
                a = a / s
            sc = _score_normalized(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_real":
        # Real part of a chirp, shifted to be positive
        t = np.linspace(0, 1, n)
        a = np.cos(2 * np.pi * 3 * t * t) + 1.5
        return a / a.sum()
    elif kind == "double_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 5 * t) + 0.1 * np.cos(2 * np.pi * 13 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exponential_taper":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2 * 10) + 0.1
        return a / a.sum()
    elif kind == "wavelet_like":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(4 * np.pi * t) * np.exp(-5 * (t - 0.3) ** 2)
        a = a + 0.1 * np.sin(8 * np.pi * t) * np.exp(-5 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _fourier_perturb(a, rng, n_modes=5, amplitude=0.03):
    """Perturb in Fourier domain for structured diversity."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z_hat = np.fft.rfft(z)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        if freq < len(z_hat):
            delta = amplitude * rng.uniform(0.5, 2.0) * rng.standard_normal()
            phase = rng.uniform(0, 2 * np.pi)
            z_hat[freq] += delta * np.exp(1j * phase)
            if freq < len(z_hat) - 1:
                z_hat[-freq] = np.conj(z_hat[freq])
    z_new = np.fft.irfft(z_hat, n)
    z_new = z_new - z_new.mean()
    return _softmax(z_new)


def _shrink_and_reoptimize(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.02, 0.05, 0.10, 0.15]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _multi_stage_optimize(a_init, deadline, min_remaining=30.0, label=""):
    """Run the full optimization pipeline on a single sequence."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    # Stage 1: L-BFGS
    if time.monotonic() < deadline - min_remaining - 30:
        a1 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 30)
        sc1 = _score(a1)
        if sc1 < current_sc:
            current_a = a1
            current_sc = sc1

    # Stage 2: Pressure equalization
    if time.monotonic() < deadline - min_remaining - 10:
        a2, sc2 = _pressure_equalize(current_a, min(deadline - min_remaining - 5, time.monotonic() + 60),
                                     n_iters=150, gamma=0.35)
        if sc2 / (2 * n) < current_sc / (2 * n):  # compare normalized scores
            sc2_full = _score(a2)
            if sc2_full < current_sc:
                current_a = a2
                current_sc = sc2_full

    # Stage 3: L-BFGS again (finer)
    if time.monotonic() < deadline - min_remaining - 10:
        a3 = _run_lbfgs(current_a, deadline, min_time_remaining=min_remaining + 10)
        sc3 = _score(a3)
        if sc3 < current_sc:
            current_a = a3
            current_sc = sc3

    # Stage 4: Subgradient polish
    if time.monotonic() < deadline - min_remaining:
        a4, sc4 = _subgrad_polish(current_a, min(deadline - min_remaining, time.monotonic() + 40),
                                  n_max_steps=2000, lr_init=0.02)
        if sc4 < current_sc:
            current_a = a4
            current_sc = sc4

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search combining L-BFGS, pressure equalization,
    coordinate descent, and diverse restarts across multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        _update(parent)
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        _update(parent)

    # --- Phase 1: Main optimization on parent ---
    print(f"\n[Phase 1] Main optimization on parent n={n_parent} (t={time.monotonic()-t_start:.1f}s)")
    a_main, sc_main = _multi_stage_optimize(parent, deadline, min_remaining=600, label="main")
    _update(a_main, sc_main)

    # --- Phase 2: Pressure equalization on best ---
    print(f"\n[Phase 2] Extended pressure equalization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_pe, sc_pe_norm = _pressure_equalize(best_seq, min(deadline - 400, time.monotonic() + 120),
                                              n_iters=300, gamma=0.3)
        sc_pe_full = _score(a_pe)
        _update(a_pe, sc_pe_full)
        if sc_pe_full < best_score:
            # Follow up with L-BFGS
            a_follow = _run_lbfgs(a_pe, deadline, min_time_remaining=400)
            _update(a_follow, _score(a_follow))

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192]
    # Remove parent n to avoid redundant work
    n_candidates = [x for x in n_candidates if x != n_parent]
    # Sort by expected quality (larger n first for exploration)
    n_candidates.sort(reverse=True)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        remaining_for_this = deadline - 200 - time.monotonic()
        if remaining_for_this < 30:
            break

        # Initialize: resample parent or use profile
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        elif parent is not None:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        min_rem = max(30, remaining_for_this * 0.3)
        a_opt, sc_opt = _multi_stage_optimize(a_init, deadline, min_remaining=min_rem, label=f"n_{n_try}")
        _update(a_opt, sc_opt)

    # --- Phase 4: Diverse restarts from best ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restart_profiles = ["chirp_real", "double_cos", "wavelet_like", "exponential_taper"]

        for i, prof in enumerate(restart_profiles):
            if time.monotonic() > deadline - 30:
                break

            # Blend parent with profile
            a_blend = 0.5 * best_seq + 0.5 * _make_profile(n_best, prof)
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()

            # Add small Fourier perturbation
            a_pert = _fourier_perturb(a_blend, rng, n_modes=4, amplitude=0.02)

            a_r, sc_r = _multi_stage_optimize(a_pert, deadline, min_remaining=30, label=f"restart_{i}")
            _update(a_r, sc_r)

    # --- Phase 5: Coordinate descent polish on best ---
    print(f"\n[Phase 5] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd, sc_cd_norm = _coordinate_polish(best_seq, min(deadline - 10, time.monotonic() + 120), n_rounds=50)
        sc_cd = _score(a_cd)
        _update(a_cd, sc_cd)

    # --- Phase 6: Shrinkage ---
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_sh, sc_sh = _shrink_and_reoptimize(best_seq, min(deadline - 10, time.monotonic() + 60))
        _update(a_sh, sc_sh)
        if sc_sh < best_score + 0.001:
            a_sh2 = _run_lbfgs(a_sh, deadline, min_time_remaining=5)
            _update(a_sh2, _score(a_sh2))

    # --- Phase 7: Final fine L-BFGS + subgradient ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        a_fin = _run_lbfgs(best_seq, deadline, min_time_remaining=3)
        _update(a_fin, _score(a_fin))

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=200, lr_init=0.005)
            _update(a_fin2, sc_fin2)

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
