def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

def _next_pow2(n):

    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
