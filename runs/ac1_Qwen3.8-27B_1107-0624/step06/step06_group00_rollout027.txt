def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _gradient_from_weights(a, w, n):
    """Compute 4*n * sum_k w_k * a_{k-j} using FFT."""
    w_rev = w[::-1]
    conv_full = _fft_convolve(a, w_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adaptive_gradient(a, c, T, threshold_frac=0.90):
    """Adaptive active-set gradient: all peaks above threshold get softmax weights."""
    n = len(a)
    c_max = c.max()
    if c_max <= 0:
        return np.zeros(n)
    threshold = threshold_frac * c_max
    active = c > threshold
    if not active.any():
        active[np.argmax(c)] = True
    c_active = c[active]
    T_safe = max(T, 1e-10)
    shifted = (c_active - c_active.max()) / T_safe
    w_active = np.exp(shifted)
    s = w_active.sum()
    if s < 1e-300:
        w_active = np.ones_like(w_active) / len(w_active)
    else:
        w_active = w_active / s
    w = np.zeros_like(c)
    w[active] = w_active
    return _gradient_from_weights(a, w, n)


def _topk_gradient(a, c, k=5):
    """Gradient using top-k peaks uniformly weighted."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[k_indices] = 1.0 / len(k_indices)
    return _gradient_from_weights(a, w, n)


def _adam_refine(a_init, deadline, T_start, T_end, lr_start, lr_end,
                 max_iter=500000, threshold_frac=0.90,
                 use_topk=False, topk=5, eval_interval=200,
                 clip_norm=1.0, patience_limit=8, verbose=True, label=""):
    """Adam optimizer in z-space with adaptive active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_topk:
            g = _topk_gradient(a, c, topk)
        else:
            g = _adaptive_gradient(a, c, T, threshold_frac)

        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1

            if patience >= patience_limit and lr_mult > 0.15:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                if verbose:
                    print(f"    {label}[{t}] lr_mult reset to {lr_mult:.3f}, best={best_score:.10f}")

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_refine_bt(a_init, deadline, T_start, T_end, lr_start, lr_end,
                    max_iter=200000, threshold_frac=0.92,
                    eval_interval=100, patience_limit=6, verbose=True, label=""):
    """Adam with backtracking line search and adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        g = _adaptive_gradient(a, c, T, threshold_frac)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        # Backtracking: try step, shrink if no improvement
        cur_sc = _score_norm(a)
        step_scale = 1.0
        accepted = False
        for trial in range(5):
            z_new = z - step_scale * update
            z_new = np.clip(z_new, -60.0, 60.0)
            z_new = z_new - z_new.mean()
            a_new = _softmax_stable(z_new)
            sc_new = _score_norm(a_new)
            if np.isfinite(sc_new) and sc_new < cur_sc + 1e-12:
                z = z_new
                accepted = True
                break
            step_scale *= 0.5
        if not accepted:
            # Skip this step (gradient was bad)
            m_buf *= 0.5
            v_buf *= 0.5

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience >= patience_limit and lr_mult > 0.1:
                lr_mult *= 0.5
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                if verbose:
                    print(f"    {label}[{t}] restart, lr_mult={lr_mult:.3f}, best={best_score:.10f}")

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _symmetric_adam(a_init, deadline, T_start, T_end, lr_start, lr_end,
                    max_iter=300000, threshold_frac=0.90,
                    eval_interval=200, patience_limit=6, label=""):
    """Adam optimizer enforcing a_i = a_{n-1-i} symmetry."""
    n = len(a_init)
    if n % 2 != 0:
        return a_init, _score_norm(a_init)
    m = n // 2

    def _make_sym(b):
        a = np.zeros(n)
        a[:m] = b
        a[m:] = b[::-1]
        return a

    b = a_init[:m].copy()
    b = b / b.sum() * 0.5
    z = np.log(np.maximum(b, 1e-15))
    z = z - z.mean()
    z_dim = m
    m_buf = np.zeros(z_dim)
    v_buf = np.zeros(z_dim)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_b = b.copy()
    best_a = _make_sym(b)
    best_score = _score_norm(best_a)
    patience = 0
    lr_mult = 1.0
    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        b_cur = _softmax_stable(z) * 0.5
        a = _make_sym(b_cur)
        c = _fft_convolve(a, a)
        g_full = _adaptive_gradient(a, c, T, threshold_frac)
        # Project gradient to symmetric constraint: grad_b = g_full[:m] + g_full[m:][::-1]
        g_b = g_full[:m] + g_full[m:][::-1]
        W = float(np.dot(b_cur, g_b))
        z_grad = b_cur * (g_b - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(z_dim)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            b_cur = _softmax_stable(z) * 0.5
            a_cur = _make_sym(b_cur)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                best_b = b_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience >= patience_limit and lr_mult > 0.15:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_b, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(z_dim)
                v_buf = np.zeros(z_dim)

    b_final = _softmax_stable(z) * 0.5
    a_final = _make_sym(b_final)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_step_fast(a, deadline, time_limit=10):
    """LP-based direction finding (simplified)."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    # Build sparse matrix for constraints (a*b)_k <= M
    # Use a subset of constraints for speed: only the top ones
    top_k_constraints = min(2 * n, 4 * n)
    all_k = np.arange(2 * n - 1)
    # Use all constraints but with sparse representation
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            rows_list.append(np.full(len(j_range), k, dtype=np.int32))
            cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'mip_rel_gap': 0.1})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        # Line search
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            return (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_diverse_init(n, rng, mode="gaussian"):
    """Create diverse structural initializations."""
    if mode == "uniform":
        return np.ones(n) / n
    elif mode == "gaussian":
        x = np.arange(n) / n
        sigma = 0.15 + 0.1 * rng.random()
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        return a / a.sum()
    elif mode == "triangular":
        x = np.abs(np.arange(n) / n - 0.5) * 2
        a = np.maximum(0, 1 - x)
        return a / a.sum()
    elif mode == "sinusoidal":
        k = 1 + int(rng.integers(0, 4))
        x = np.arange(n) / n
        a = np.abs(np.sin(k * np.pi * x)) + 0.1
        return a / a.sum()
    elif mode == "bimodal":
        x = np.arange(n) / n
        a = np.exp(-0.5 * ((x - 0.3) / 0.1) ** 2) + np.exp(-0.5 * ((x - 0.7) / 0.1) ** 2)
        return a / a.sum()
    elif mode == "random_sparse":
        a = np.zeros(n)
        n_peaks = int(n * (0.05 + 0.1 * rng.random()))
        peaks = rng.choice(n, size=n_peaks, replace=False)
        a[peaks] = 1.0
        # Smooth slightly
        kernel = np.ones(3) / 3
        a = np.convolve(a, kernel, mode='same')
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 0: Load parent, quick LP (60s)
    Phase 1: Adam adaptive active-set (300s)
    Phase 2: Symmetric optimization (200s)
    Phase 3: Diverse restarts (400s)
    Phase 4: n=8192 exploration (300s)
    Phase 5: Backtracking refinement (300s)
    Phase 6: Final ultra-fine polish (remaining)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n
        print("No parent found, using uniform init n=4096")

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # === Phase 0: Quick LP warm-up (0-60s) ===
    phase0_end = min(deadline, t_start + 60.0)
    if time.monotonic() < phase0_end:
        a_lp = a.copy()
        lp_iter = 0
        while time.monotonic() < phase0_end - 2:
            a_lp = _lp_step_fast(a_lp, phase0_end, time_limit=8)
            lp_iter += 1
            sc = _score(a_lp)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_lp.copy()
            if lp_iter % 2 == 0:
                print(f"  LP[{lp_iter}]: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
        print(f"Phase 0 done: {lp_iter} LP iters, best={best_score:.10f}")

    # === Phase 1: Adam adaptive active-set (60s-360s) ===
    phase1_end = min(deadline, t_start + 360.0)
    if time.monotonic() < phase1_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a1, sc1 = _adam_refine(best_a, phase1_end - 3,
                                T_start=0.01 * max_c, T_end=0.0001 * max_c,
                                lr_start=0.04, lr_end=0.004,
                                max_iter=300000, threshold_frac=0.88,
                                eval_interval=150, patience_limit=6,
                                verbose=True, label="Adam-adv")
        full_sc = _score(a1)
        print(f"Phase 1 Adam-adv: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a1.copy()

    # === Phase 2: Symmetric optimization (360s-560s) ===
    phase2_end = min(deadline, t_start + 560.0)
    if time.monotonic() < phase2_end and n % 2 == 0:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_sym, sc_sym = _symmetric_adam(best_a, phase2_end - 3,
                                         T_start=0.005 * max_c, T_end=0.0001 * max_c,
                                         lr_start=0.03, lr_end=0.003,
                                         max_iter=200000, threshold_frac=0.90,
                                         eval_interval=150, patience_limit=6,
                                         label="Sym-Adam")
        full_sc = _score(a_sym)
        print(f"Phase 2 Symmetric: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_sym.copy()

    # === Phase 3: Diverse restarts (560s-960s) ===
    phase3_end = min(deadline, t_start + 960.0)
    n_best = len(best_a)
    if time.monotonic() < phase3_end:
        modes = ["gaussian", "triangular", "sinusoidal", "bimodal", "random_sparse", "random_sparse"]
        per_restart = (phase3_end - time.monotonic() - 10) / len(modes)
        for idx, mode in enumerate(modes):
            if time.monotonic() > phase3_end - 20:
                break
            sub_deadline = min(phase3_end, time.monotonic() + max(30, per_restart))
            # Initialize from a mix of current best and random structure
            a_rand = _make_diverse_init(n_best, rng, mode=mode)
            a_init = 0.7 * best_a + 0.3 * a_rand
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
            c_init = _fft_convolve(a_init, a_init)
            max_ci = float(c_init.max())
            a_opt, sc_opt = _adam_refine(a_init, sub_deadline - 2,
                                          T_start=0.005 * max_ci, T_end=0.0002 * max_ci,
                                          lr_start=0.035, lr_end=0.003,
                                          max_iter=100000, threshold_frac=0.88,
                                          eval_interval=100, patience_limit=5,
                                          label=f"Rst[{mode}]")
            full_sc = _score(a_opt)
            print(f"  Restart[{mode}]: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

    # === Phase 4: n=8192 exploration (960s-1260s) ===
    phase4_end = min(deadline, t_start + 1260.0)
    if time.monotonic() < phase4_end:
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        a_large_opt, sc_large = _adam_refine(a_large, phase4_end - 5,
                                              T_start=0.003 * max_cl, T_end=0.00005 * max_cl,
                                              lr_start=0.03, lr_end=0.002,
                                              max_iter=150000, threshold_frac=0.90,
                                              eval_interval=200, patience_limit=5,
                                              label="n8192")
        full_sc_large = _score(a_large_opt)
        print(f"Phase 4 n=8192: score={full_sc_large:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_large) and full_sc_large < best_score:
            best_score = full_sc_large
            best_a = a_large_opt.copy()
        # Interpolate back to original n
        if full_sc_large < best_score + 0.1:
            n_orig = n
            a_back = _interp_to_length(a_large_opt, n_orig)
            c_back = _fft_convolve(a_back, a_back)
            max_cb = float(c_back.max())
            a_back_opt, sc_b = _adam_refine(a_back, phase4_end - 3,
                                             T_start=0.002 * max_cb, T_end=0.00005 * max_cb,
                                             lr_start=0.025, lr_end=0.002,
                                             max_iter=80000, threshold_frac=0.90,
                                             eval_interval=150, patience_limit=4,
                                             label="n8192->back")
            full_sc_back = _score(a_back_opt)
            print(f"  n=8192 -> n={n_orig}: score={full_sc_back:.10f}")
            if np.isfinite(full_sc_back) and full_sc_back < best_score:
                best_score = full_sc_back
                best_a = a_back_opt.copy()

    # === Phase 5: Backtracking refinement (1260s-1560s) ===
    phase5_end = min(deadline, t_start + 1560.0)
    if time.monotonic() < phase5_end:
        n_cur = len(best_a)
        c_cur = _fft_convolve(best_a, best_a)
        max_cc = float(c_cur.max())
        a_bt, sc_bt = _adam_refine_bt(best_a, phase5_end - 5,
                                       T_start=0.001 * max_cc, T_end=0.00002 * max_cc,
                                       lr_start=0.02, lr_end=0.001,
                                       max_iter=100000, threshold_frac=0.92,
                                       eval_interval=80, patience_limit=5,
                                       label="BT-Adam")
        full_sc = _score(a_bt)
        print(f"Phase 5 Backtracking: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_bt.copy()

    # === Phase 6: Final ultra-fine polish (1560s-deadline) ===
    if time.monotonic() < deadline - 8:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_refine(best_a, deadline - 5,
                                          T_start=0.0002 * max_cf, T_end=0.000005 * max_cf,
                                          lr_start=0.005, lr_end=0.0003,
                                          max_iter=200000, threshold_frac=0.95,
                                          eval_interval=80, patience_limit=4,
                                          label="Polish")
        full_sc = _score(a_final)
        print(f"Phase 6 Polish: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # Final LP polish if time remains
        if time.monotonic() < deadline - 15:
            a_lp2 = _lp_step_fast(best_a, deadline, time_limit=5)
            sc_lp2 = _score(a_lp2)
            if np.isfinite(sc_lp2) and sc_lp2 < best_score:
                best_score = sc_lp2
                best_a = a_lp2.copy()
                print(f"  Final LP polish: score={sc_lp2:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
