"""
Multi-method optimizer: FISTA + Coordinate Descent + L-BFGS + Multi-start
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: exact coordinate descent with 1D subproblem solving.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_passes=200):
    """
    Exact coordinate descent on 2n*max(conv(a,a))/sum(a)^2.
    Maintains the full convolution array and updates it incrementally.
    """
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    if S < 1e-15:
        return a, np.inf
    c = _full_convolve(a, a)
    conv_len = 2 * n - 1
    best_score = 2.0 * n * float(c.max()) / (S * S)
    best_a = a.copy()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        pass_improved = 0

        for i in range(n):
            if time.monotonic() > deadline:
                break
            a_old = a[i]
            S_rest = S - a_old

            # Affected k range: k in [i, i+n-1] capped to [0, conv_len-1]
            k_start = i
            k_end = min(i + n, conv_len)
            if k_start >= k_end:
                continue
            length_aff = k_end - k_start

            # offsets = k - i for k in affected range = 0, 1, ..., length_aff-1
            offsets = np.arange(length_aff, dtype=np.int64)
            a_ki = a[offsets]  # a[k-i] for each affected k

            # alpha: linear coefficient of a_new in c[k]
            # For k=2i (offset=i): quadratic only, no linear
            # For k!=2i: alpha = 2*a[k-i]
            alpha = 2.0 * a_ki.copy()
            if i < length_aff:
                alpha[i] = 0.0  # quadratic case

            # beta: quadratic coefficient (1 if k=2i)
            beta_val = 1.0 if i < length_aff else 0.0

            c_aff = c[k_start:k_end]

            # Unaffected max
            if k_start > 0 or k_end < conv_len:
                if k_start == 0:
                    M_unaff = float(c[k_end:].max()) if k_end < conv_len else 0.0
                elif k_end == conv_len:
                    M_unaff = float(c[:k_start].max()) if k_start > 0 else 0.0
                else:
                    M_unaff = max(float(c[:k_start].max()), float(c[k_end:].max()))
            else:
                M_unaff = 0.0

            # 1D optimization: minimize f(a_new) = 2n * max(M_unaff, max(c_aff + (a_new-a_old)*alpha + (a_new^2-a_old^2)*beta)) / (S_rest + a_new)^2
            # Candidate values
            if a_old > 1e-15:
                candidates = np.array([0.0, a_old * 0.3, a_old * 0.6, a_old * 0.9,
                                       a_old, a_old * 1.1, a_old * 1.4, a_old * 2.0])
            else:
                candidates = np.array([0.0, 0.5 / n, 1.0 / n, 2.0 / n])
            candidates = candidates[(candidates >= 0) & np.isfinite(candidates)]

            best_f = np.inf
            best_new = a_old
            for cand in candidates:
                delta_lin = (cand - a_old) * alpha
                delta_quad = (cand * cand - a_old * a_old) * beta_val
                c_new = c_aff + delta_lin + delta_quad
                M_aff_new = float(c_new.max())
                M_total = max(M_unaff, M_aff_new)
                denom = S_rest + cand
                if denom < 1e-12:
                    continue
                f_val = 2.0 * n * M_total / (denom * denom)
                if f_val < best_f:
                    best_f = f_val
                    best_new = cand

            # Golden section refinement
            if best_new > 1e-10:
                lo = max(0.0, best_new * 0.5)
                hi = best_new * 2.0 + 1e-8
            else:
                lo = 0.0
                hi = 4.0 / n

            for _ in range(12):
                if hi - lo < 1e-14:
                    break
                gr = 0.618
                m1 = lo + (hi - lo) * (1.0 - gr)
                m2 = lo + (hi - lo) * gr

                d1 = S_rest + m1
                d2 = S_rest + m2
                if d1 < 1e-12:
                    f1 = np.inf
                else:
                    c_m1 = c_aff + (m1 - a_old) * alpha + (m1 * m1 - a_old * a_old) * beta_val
                    f1 = 2.0 * n * max(M_unaff, float(c_m1.max())) / (d1 * d1)

                if d2 < 1e-12:
                    f2 = np.inf
                else:
                    c_m2 = c_aff + (m2 - a_old) * alpha + (m2 * m2 - a_old * a_old) * beta_val
                    f2 = 2.0 * n * max(M_unaff, float(c_m2.max())) / (d2 * d2)

                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            a_new = (lo + hi) / 2.0
            if a_new < 0:
                a_new = 0.0

            if abs(a_new - a_old) > 1e-14:
                a[i] = a_new
                delta_lin = (a_new - a_old) * alpha
                delta_quad = (a_new * a_new - a_old * a_old) * beta_val
                c[k_start:k_end] = c_aff + delta_lin + delta_quad
                S += (a_new - a_old)
                pass_improved += 1

        # Renormalize (scale invariance)
        if S > 1e-15:
            a = a / S
            c = c / (S * S)
            S = 1.0

        c_max = float(c.max())
        cur_score = 2.0 * n * c_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if pass_improved == 0:
            break

    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 40)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        center = n / 4.0
        width = n / 8.0
        a = np.exp(-((np.arange(n) - center) / width) ** 2) + np.exp(-((np.arange(n) - (n - center)) / width) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        mask = (t > 0.2) & (t < 0.8)
        a[mask] = 1.5
        a[~mask] = 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

def _alternating_fista_cd(a_init, deadline, n_fista_phases=8, fista_time_frac=0.5):

    best_score = _score_norm(a)
    best_a = a.copy()
    total_time = deadline - time.monotonic()

        if time.monotonic() > deadline:

        # FISTA phase
        c_curr = _full_convolve(a, a)

        a = a / a.sum()
        sc = _score_norm(a)

    return best_a, best_score

                                  b, lr_init=0.005, max_iter=300000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={b:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        # CD after Adam
        if time.monotonic() < phase4_end - 3:
            a_work, _ = _coord_descent(a_work, phase4_end - 1, max_passes=100)
            a_work = a_work / a_work.sum()
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  CD after Adam: score={sc:.10f}")

    # === Phase 5: L-BFGS refinement ===
    print("Phase 5: L-BFGS refinement")
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00003 * max_cc
        a_lb, _ = _lbfgs_opt(a_work, phase5_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 6: Multi-restart exploration ===
    print("Phase 6: Multi-restart")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end - 20:
        n_restarts = 5
        per_r = (phase6_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            if i == 0:
                p = _perturb(a_work, 0.01, rng)
            elif i == 1:
                p = _make_profile(n_main, "raised_cosine")
                p = 0.7 * a_work + 0.3 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                shift = rng.integers(1, max(2, n_main // 256))
                p = np.roll(a_work, shift).copy()
                p = p / p.sum()
            elif i == 3:
                p = _perturb(a_work, 0.025, rng)
            else:
                p = _make_profile(n_main, "bimodal")
                p = 0.5 * a_work + 0.5 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()

            # Quick FISTA
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.001 * max_cp
            fista_t = time.monotonic() + per_r * 0.4
            p, _ = _fista_opt(p, b_r, fista_t, 20000)
            p = p / p.sum()
            # CD
            cd_t = time.monotonic() + per_r * 0.5
            p, _ = _coord_descent(p, cd_t, max_passes=50)
            p = p / p.sum()
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final polish ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 5

        # Final CD
        if remaining > 10:
            a_pol, _ = _coord_descent(a_pol, time.monotonic() + remaining * 0.5, max_passes=200)
            a_pol = a_pol / a_pol.sum()
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
