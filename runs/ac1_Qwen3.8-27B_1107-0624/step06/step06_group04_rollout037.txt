    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return min_beta
    w = e / s
    H = -float(np.sum(w * np.log(np.maximum(w, 1e-300))))
    H_max = np.log(len(c))
    # If entropy is high (many active peaks), use larger beta for stability
    # If entropy is low (concentrated on few peaks), use smaller beta for precision
    ratio = H / H_max
    beta = min_beta + (max_beta - min_beta) * ratio
    return float(np.clip(beta, min_beta, max_beta))


def _fista_optimize(a_init, betas, deadline, max_iter_per_beta=20000):
    """FISTA with given beta schedule. Returns best (a, score_norm)."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for beta in betas:
        if time.monotonic() > deadline:
            break
        L = 4.0 * n / beta
        step = 1.0 / L
        a_prev = a_curr.copy()
        t_val = 1.0

        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta)
            grad = _compute_grad_simplex(y, w)
            a_next = _proj_simplex(y - step * grad)

            if it % 200 == 0 or it < 2:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

        sc_final = _score_norm(a_curr)
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace_optimize(a_init, deadline, betas, lr_init, max_iter,
                            lr_schedule="cosine", grad_clip=1.0):
    """Adam in log-space with given beta schedule and LR schedule."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Learning rate schedule
            progress = t / max_iter
            if lr_schedule == "cosine":
                current_lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            elif lr_schedule == "linear":
                current_lr = lr_init * max(0.1, 1.0 - progress)
            else:
                current_lr = lr_init

            current_lr = max(current_lr, lr_init * 0.01)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if no_improve > 80:
                # Reset momentum to escape stagnation
                m_buf *= 0.5
                v_buf *= 0.5
                no_improve = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_shave(a_init, n_steps=500, deadline=None):
    """Directly target and reduce the top peaks of the convolution."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Focus on top-k peaks
        top_k = min(32, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w = np.zeros(len(c))
        w[top_idx] = (c[top_idx] - cm + 1e-8) ** 3
        ws = w.sum()
        if ws < 1e-300:
            w[int(np.argmax(c))] = 1.0
        else:
            w /= ws

        grad = _compute_grad_simplex(a, w)
        step_size = 1.0 / (1.0 + 0.005 * it)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.7
            a_new = _proj_simplex(a - step_size * grad)
            sc2 = _score_norm(a_new)
            if sc2 < _score_norm(a):
                a = a_new

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_noise(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _perturb_structured(a, rng):
    """Add a low-frequency sinusoidal perturbation in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freq = rng.integers(2, 8)
    phase = rng.uniform(0, 2 * np.pi)
    amplitude = rng.uniform(0.05, 0.2)
    t = np.arange(n) / n
    perturbation = amplitude * np.cos(2 * np.pi * freq * t + phase)
    z = z + perturbation
    z = z - z.mean()
    return _softmax_compute(z)


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    """Flatten the spectrum of the sequence to reduce convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        # Target: flatten magnitude spectrum
        target_mag = np.median(mag)
        # Blend between current and target
        blend = 0.1 + 0.05 * (it / n_iter)
        new_mag = (1 - blend) * mag + blend * target_mag
        A_new = A / (mag + 1e-30) * new_mag
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score
