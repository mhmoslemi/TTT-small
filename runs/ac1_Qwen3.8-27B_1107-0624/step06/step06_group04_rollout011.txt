"""
Multi-phase optimizer using L_p norm continuation and adaptive softmax smoothing.
Exploits frequency-domain gradient computation (FFT-based cross-correlation) for
efficient optimization of max(conv(a,a)) over the probability simplex.
Uses log-parameterization (softmax) for natural constraint handling, adaptive
p-continuation, multi-length exploration, and diverse restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _compute_gradient_from_weights(a, w, n):
    """Compute gradient given weight vector w (length 2n-1).
    grad_i = sum_k w[k] * a[k-i] for valid indices.
    This equals conv(a, w)[n-1 : 2n-1].
    """
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    conv_aw = np.fft.irfft(A * W, N)[:3 * n - 2]
    return conv_aw[n - 1:2 * n - 1]


def _softmax_z_to_a(z):
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_in_z_space(a, grad_a):
    """Convert gradient wrt a to gradient wrt z where a = softmax(z)."""
    mean_grad = float(np.dot(a, grad_a))
    return a * (grad_a - mean_grad)


def _adam_optimize_l_p(a_init, deadline, p_schedule, lr_init=0.01,
                       max_iter_total=500000, lr_decay_every=5000,
                       lr_decay_factor=0.95, grad_clip=10.0, check_freq=20):
    """Adam optimizer using L_p norm gradient with log-parameterization."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_z_to_a(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            if cm < 1e-30:
                break

            # Compute weights v_k = c_k^(p-1) normalized
            # Use log-domain for stability
            log_c = np.log(np.maximum(c, 1e-300))
            log_v = (p - 1.0) * log_c
            log_v_max = float(log_v.max())
            v = np.exp(log_v - log_v_max)
            v_sum = float(v.sum())
            if v_sum < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = v / v_sum

            grad_a = _compute_gradient_from_weights(a, w, n)
            if not np.all(np.isfinite(grad_a)):
                current_lr *= 0.5
                continue

            grad_z = _grad_in_z_space(a, grad_a)

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % check_freq == 0:
                a_chk = _softmax_z_to_a(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0001, current_lr * lr_decay_factor)

            if no_improve > 30 and t > 500:
                current_lr = max(0.0001, current_lr * 0.9)
                no_improve = 0

    a_final = _softmax_z_to_a(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

    L = 8.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_gradient_from_weights(y, w, n)

        # Project onto simplex
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    """Flatten the power spectrum to reduce peak autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        # Progressive flattening: blend current magnitude with target
        target_mag = np.full_like(mag, mag.mean())
        blend = 0.05 + 0.05 * it / n_iter
        new_mag = (1 - blend) * mag + blend * target_mag
        phase = A / (mag + 1e-30)
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _freq_domain_random(n, rng):
    """Generate a random non-negative sequence with approximately flat power spectrum."""
    N = _next_pow2(2 * n - 1)
    # Random phases, flat magnitude
    mag = np.full(N // 2 + 1, 1.0)
    phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    A = mag * (np.cos(phase) + 1j * np.sin(phase))
    # Make A Hermitian-symmetric for real output
    A[0] = abs(A[0])
    if N % 2 == 0:
        A[N // 2] = abs(A[N // 2])
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        # Try again with different phases
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        A = mag * (np.cos(phase) + 1j * np.sin(phase))
        A[0] = abs(A[0])
        if N % 2 == 0:
            A[N // 2] = abs(A[N // 2])
        a = np.fft.irfft(A, N)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_z_to_a(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
