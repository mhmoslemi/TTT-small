    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.25:
            update = update * (0.25 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 25 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 4000 == 0:
            current_lr = max(0.0002, current_lr * 0.88)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(80, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 25}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sine_power":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        center = n / 2.0
        sigma = n / 5.0
        a = np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _optimize_single(a_init, n, t_start, budget_frac):
    """Run full optimization pipeline on a single sequence. Returns (best_a, best_score)."""
    deadline = t_start + budget_frac
    best_a = a_init.copy()
    best_score = _score(a_init)
    a_work = a_init.copy()
    a_work = a_work / a_work.sum()

    # Phase A: FISTA with geometrically decreasing beta
    c_curr = _full_convolve(a_work, a_work)
    max_c = float(c_curr.max())
    n_phases = 20
    beta = 0.05 * max_c
    phase_end = min(deadline, t_start + budget_frac * 0.30)
    for phase in range(n_phases):
        if time.monotonic() > phase_end - 2:
            break
        per_t = min(15.0, (phase_end - time.monotonic() - 1) / max(1, n_phases - phase))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_t, max(2000, int(per_t * 5000)))
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()
        c_curr = _full_convolve(a_work, a_work)
        max_c = float(c_curr.max())
        beta = 0.30 * max_c
        if beta < 1e-12:
            beta = 1e-12

    # Phase B: Top-k Nesterov subgradient
    phase_b_end = min(deadline, t_start + budget_frac * 0.50)
    if time.monotonic() < phase_b_end - 5:
        a_work, _ = _topk_nesterov_opt(a_work, phase_b_end)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()

    # Phase C: Adam with decreasing beta
    phase_c_end = min(deadline, t_start + budget_frac * 0.75)
    if time.monotonic() < phase_c_end - 5:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.001 * max_cc, 0.0002 * max_cc, 0.00005 * max_cc, 0.00001 * max_cc]
        per_beta = max(3.0, (phase_c_end - time.monotonic() - 5) / len(betas))
        for beta in betas:
            if time.monotonic() > phase_c_end - 5:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + per_beta, beta,
                                  lr_init=0.005, max_iter=500000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_a = a_work.copy()

    # Phase D: L-BFGS
    phase_d_end = min(deadline, t_start + budget_frac * 0.92)
    if time.monotonic() < phase_d_end - 5:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        a_work, _ = _lbfgs_opt(a_work, phase_d_end, 0.00005 * max_cc)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()

    # Phase E: Final Adam polish
    if time.monotonic() < deadline - 3:
        c_curr = _full_convolve(best_a, a_work)
        max_cc = float(_full_convolve(a_work, a_work).max())
        a_work, _ = _adam_opt(a_work, deadline - 2, 0.000003 * max_cc,
                              lr_init=0.001, max_iter=100000)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")

        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Main optimization on parent length ===
    print(f"\n=== Main optimization (n={n_main}) ===")
    main_deadline = t_start + budget_s * 0.60
    a_main, sc_main = _optimize_single(a0, n_main, t_start, budget_s * 0.60)

    if sc_main_full < best_score:

        best_seq = a_main.copy()

    n8192 = 8192

        a8192_init = a8192_init / a8192_init.sum()

        print(f"n=8192 init score={sc_init:.10f}")

        remaining_budget = min(400.0, deadline - time.monotonic() - 200)
        if remaining_budget > 100:
            a8192, sc8192 = _optimize_single(a8192_init, n8192, t_start, 0)  # dummy
            # Re-do with proper timing
            a8192_deadline = time.monotonic() + remaining_budget
            a8192 = a8192_init.copy()
            # Quick FISTA
            c = _full_convolve(a8192, a8192)
            beta = 0.03 * float(c.max())
            a8192, _ = _fista_opt(a8192, beta, a8192_deadline * 0.5, 30000)
            a8192, _ = _topk_nesterov_opt(a8192, a8192_deadline * 0.7)
            c = _full_convolve(a8192, a8192)
            beta = 0.0001 * float(c.max())
            a8192, _ = _adam_opt(a8192, a8192_deadline * 0.9, beta, 0.003, 200000)
            if time.monotonic() < a8192_deadline - 10:
                c = _full_convolve(a8192, a8192)
                a8192, _ = _lbfgs_opt(a8192, a8192_deadline, 0.00003 * float(c.max()))
            sc8192 = _score(a8192)
            print(f"n=8192 result: score={sc8192:.10f}, time={time.monotonic()-t_start:.1f}s")
            if sc8192 < best_score:

                best_seq = a8192.copy()

    # === Try n=2048 (smaller n, might be easier to optimize) ===

        x_fine = np.linspace(0, 1, n2048)

        remaining = min(200.0, deadline - time.monotonic() - 100)
        if remaining > 60:
            dl = time.monotonic() + remaining
            c = _full_convolve(a2048_init, a2048_init)
            beta = 0.03 * float(c.max())
            a2048, _ = _fista_opt(a2048_init, beta, dl * 0.4, 20000)
            a2048, _ = _topk_nesterov_opt(a2048, dl * 0.6)
            c = _full_convolve(a2048, a2048)
            beta = 0.0001 * float(c.max())
            a2048, _ = _adam_opt(a2048, dl * 0.85, beta, 0.004, 100000)
            if time.monotonic() < dl - 8:

                a2048, _ = _lbfgs_opt(a2048, dl, 0.00003 * float(c.max()))
            sc2048 = _score(a2048)

    if a_best is not None:
        a_best = a_best / a_best.sum()

    if time.monotonic() < restart_end - 30 and a_best is not None:

        per_r = (restart_end - time.monotonic() - 20) / n_restarts

                p = 0.7 * a_best + 0.3 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            else:
                shift = rng.integers(1, max(2, n_best // 256))
                p = np.roll(a_best, int(shift)).copy()
                p = p / p.sum()
            dl_r = time.monotonic() + max(5.0, per_r)
            c = _full_convolve(p, p)
            beta_r = 0.0005 * float(c.max())
            p, _ = _fista_opt(p, beta_r, dl_r * 0.5, 10000)
            p, _ = _adam_opt(p, dl_r * 0.8, beta_r * 0.1, 0.003, 50000)
            if time.monotonic() < dl_r - 5:
                c = _full_convolve(p, p)
                p, _ = _lbfgs_opt(p, dl_r, 0.00002 * float(c.max()))
            sc_r = _score(p)
            print(f"  Restart {i}: score={sc_r:.10f}")
            if sc_r < best_score:

                a_best = p.copy()

    # === Final ultra-polish on best ===

    if best_seq is not None and time.monotonic() < deadline - 5:

        a_pol = a_pol / a_pol.sum()

        max_cp = float(c_pol.max())
        remaining_t = deadline - time.monotonic() - 4

            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
            if time.monotonic() < deadline - 5:
                c_f = _full_convolve(a_pol, a_pol)
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 3, 0.000002 * float(c_f.max()))
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()

    print(f"\nDone: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
