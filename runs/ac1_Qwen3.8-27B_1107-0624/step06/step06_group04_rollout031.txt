```python
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses aggressive Adam in log-space with decaying beta, LP-inspired coordinate moves,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_topk(y, w, top_k):

def _adam_logspace_toppk(a_init, deadline, beta_schedule, lr_init, max_iter_total, top_k=100, lr_decay_every=5000, lr_decay_factor=0.95, grad_clip=1.0):
    """Adam optimizer in log-space with top-K active set gradients."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0
    iter_per_eval = max(1, top_k // 10)  # Evaluate every few iterations

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            # Top-K smoothing
            k = min(top_k, len(c))
            top_idx = np.argpartition(c, -k)[-k:]
            c_top = c[top_idx]
            c_shifted = (c_top - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                ep_idx_cm = int(np.argmax(c))  # fallback to argmax
                ep_idx = ep_idx_cm
                w_full = np.zeros(len(c))
                w_full[ep_idx] = 1.0
            else:
                w_top = e / s_w
                w_full = np.zeros(len(c))
                w_full[top_idx] = w_top

            # Compute gradient
            a_rev = a[::-1]
            wc = _full_convolve(w_full, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w_full, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % iter_per_eval == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 30 and t > 500:
                current_lr = max(0.0003, current_lr * 0.9)
                if current_lr <= 0.0003:
                    break  # Move to next beta

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_ascent_lp(a, M, n_passes=3, deadline=None):

    max sum(b) s.t. b >= 0, conv(a, b) <= M

    b = np.ones(n) / n  # Start with uniform
    conv_abs = _full_convolve(a, b)  # Current conv(a, b)
    
    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            # b[j] appears in conv(a,b)[k] for k = j, j+1, ..., j+n-1 (clipped to 2n-2)
            # conv(a,b)[k] = sum_i a[i] * b[k-i]
            # b[j] appears when k-i = j, i.e., i = k-j
            # So b[j] affects conv(a,b)[j], conv(a,b)[j+1], ..., conv(a,b)[j+n-1]
            # (each by a[k-j] * delta)
            
            # The constraints that limit b[j]:
            # For k in [j, min(j+n-1, 2n-2)]:
            #   conv(a,b)[k] + a[k-j] * delta <= M
            #   delta <= (M - conv(a,b)[k]) / a[k-j] (where a[k-j] > 0)
            
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            
            # Get the relevant a values
            a_slice = a[0:n-j] if j < n else np.array([])  # a[0], a[1], ..., a[n-1-j]
            # Actually a[k-j] for k = j, j+1, ..., j+n-1 is a[0], a[1], ..., a[n-1]
            # Wait: a[k-j] where k goes from j to j+n-1 gives a[0] to a[n-1]
            # But we only need k up to 2n-2, so k-j goes from 0 to min(n-1, 2n-2-j)
            
            relevant_a = a[:min(n, 2*n-1-j)]  # a[0] to a[min(n, 2n-1-j)-1]
            relevant_conv = conv_abs[k_start:k_start + len(relevant_a)]
            
            # delta_max = min of (M - relevant_conv[i]) / relevant_a[i] for relevant_a[i] > 0
            mask = relevant_a > 1e-15
            if not np.any(mask):
                delta_max = np.inf
            else:
                headroom = M - relevant_conv[mask]
                delta_max = np.min(headroom / relevant_a[mask])
            
            # Also, b[j] >= 0, so we can't decrease below 0
            # For coordinate ascent, we only increase: delta in [0, delta_max]
            # To maximize sum(b), we set b[j] as large as possible: delta = max(0, delta_max)
            # But we also need b[j] >= 0, and we're increasing, so delta >= 0
            
            delta = max(0.0, delta_max)
            if delta > 1e-15:
                b[j] += delta
                # Update conv_abs
                conv_abs[k_start:k_start + len(relevant_a)] += relevant_a * delta
                improved = True
        
        if not improved:
            break
    
    # Normalize b
    s = b.sum()
    if s > 1e-15:
        b = b / s
    else:
        b = np.ones(n) / n
    return b
```

```python
def _coord_ascent_lp(a, M, n_passes=5, deadline=None):
    n = len(a)
    b = np.zeros(n)
    
    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Compute full conv(a, b)
        conv_ab = _full_convolve(a, b)
        
        for j in range(n):
            # Find max feasible b[j]
            # b[j] contributes a[k-j]*b[j] to conv_ab[k] for k in [j, j+n-1] ∩ [0, 2n-2]
            k_start = j
            k_end = min(j + n, 2*n - 1)
            if k_start >= k_end:
                continue
            
            # Coefficients: a[0], a[1], ..., a[k_end-1-k_start]
            coeff = a[:k_end - k_start]
            # Current conv values at affected positions (including b[j]'s contribution)
            current_vals = conv_ab[k_start:k_end]
            # Remove b[j]'s current contribution
            other_vals = current_vals - coeff * b[j]
            # Max b[j] such that other_vals + coeff * b[j] <= M
            # => b[j] <= (M - other_vals[i]) / coeff[i] for all i with coeff[i] > 0
            mask = coeff > 1e-15
            if np.any(mask):
                b_max = np.min((M - other_vals[mask]) / coeff[mask])
            else:
                b_max = 1e6
            
            new_bj = max(0.0, b_max)
            if abs(new_bj - b[j]) > 1e-15:
                # Update conv_ab incrementally
                delta = new_bj - b[j]
                conv_ab[k_start:k_end] += coeff * delta
                b[j] = new_bj
    
    # Normalize
    s = b.sum()
    if s > 1e-15:
        b = b / s
    else:
        b = np.ones(n) / n
    return b
```
