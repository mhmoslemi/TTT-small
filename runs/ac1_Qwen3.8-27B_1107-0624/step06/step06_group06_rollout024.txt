```python
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import time
from scipy.optimize import minimize

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

def _project_simplex(v):
    """Project v onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)

        return np.ones(n) / n

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]

        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)

    h = wc[n - 1:2 * n - 1]

    """Nesterov-accelerated subgradient with QP-optimized weights."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    v = np.zeros(n)  # momentum
    beta_mom = 0.9  # momentum parameter
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov look-ahead
        y = z + beta_mom * (z - (z + beta_mom * (z - np.zeros(n))))  # simplified
        # Actually, let's use the standard approach:
        # y_t = x_t + beta_t * (x_t - x_{t-1})
        # But we're working in log-space, so let's just use momentum on z
        
        a_y = _softmax(z)
        b = _full_convolve(a_y, a_y)
        g = _active_gradient(a_y, b, tol_frac)
        
        # Project gradient to tangent space (sum=0)
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        # Step size: eta = c / sqrt(t+1)
        eta = 0.05 / np.sqrt(t + 1)
        
        v = beta_mom * v + (1 - beta_mom) * (g / g_norm)
        z = z - eta * v
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        if t % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

                best_a = a_check.copy()

    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()
```

Now, let me also think about the "equiripple refinement" step. The idea is:

Given the current `a`, linearize the autocorrelation around `a`:
`b_k(a + δ) ≈ b_k(a) + 2 * sum_i a_i * δ_{k-i}`

We want to find `δ` such that the top few `b_k` values are equalized (equi-ripple).

Let the active lags be `k_1, ..., k_m` and let `T` be the target value (e.g., the average of the active lags).

We want: `b_k(a) + 2 * sum_i a_i * δ_{k-i} = T` for each active `k`.

This is: `sum_i a_i * δ_{k-i} = (T - b_k(a)) / 2` for each active `k`.

Let `M` be the `m × n` matrix with `M[k, i] = a_{k-i}` (0 if out of bounds).
Let `r` be the `m` vector with `r[k] = (T - b_k(a)) / 2`.

We want `M @ δ = r`, with the constraint `sum(δ) = 0` and `a + δ >= 0`.

The minimum-norm solution (ignoring the non-negativity constraint) is:
`δ = M^T @ (M @ M^T + εI)^{-1} @ r`

Then we project: `δ_proj = δ - mean(δ)` (to enforce sum=0)
And `a_new = a + α * δ_proj` where `α` is chosen to maintain non-negativity.

```python
    """Linearized equiripple refinement."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        b = _full_convolve(a, a)
        c_max = float(b.max())
        
        # Get top n_top lags
        top_idx = np.argsort(b)[::-1][:n_top]
        T = b[top_idx].mean()  # target: average of top lags
        
        # Build M matrix (n_top x n)
        M = np.zeros((n_top, n))
        for idx, k in enumerate(top_idx):
            k = int(k)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    M[idx, i] = 2.0 * a[j]
        
        # Target residual
        r = (T - b[top_idx]) / 1.0  # We want M @ δ ≈ r, but actually:
        # b_k + (M @ δ)_k = T, so (M @ δ)_k = T - b_k
        r = T - b[top_idx]
        
        # Minimum norm solution: δ = M^T (M M^T)^{-1} r
        MMt = M @ M.T + 1e-10 * np.eye(n_top)
        try:
            delta = M.T @ np.linalg.solve(MMt, r)
        except np.linalg.LinAlgError:
            break

        delta = delta - np.mean(delta)

        alpha_max = 1.0
        for i in range(n):

        a = a + best_alpha * delta
        a = np.maximum(a, 0)
        a = a / a.sum()
```

Actually, let me reconsider the problem from scratch. The score is:

`score = 2 * n * max(b) / S^2`

where `b = conv(a, a)` and `S = sum(a)`.

Key properties:
- The score is homogeneous: `score(c*a) = score(a)` for any `c > 0`.
- So we can fix `S = 1` and minimize `2n * max(b)`.
- `sum(b) = S^2 = 1`, and there are `2n-1` elements of `b`.
- So the mean of `b` is `1/(2n-1)`.
- `max(b) >= 1/(2n-1)` (by the pigeonhole principle).
- Therefore `score >= 2n/(2n-1) ≈ 1` for large `n`.

The question is: how close can we get to this lower bound?

For a random sequence with `S = 1`, `b_k ≈ 1/(2n)` for most `k` (with some fluctuation), and `max(b) ≈ (log(2n))/(2n)` (by extreme value theory for chi-squared variables). So the score would be approximately `2n * log(2n) / (2n) = log(2n)`, which is much worse than 1.5.

For a structured sequence (like the one we're optimizing), we can do much better.

The key insight is that the optimal sequence should have a power spectrum that is as flat as possible (subject to the DC constraint). This is similar to the design of "flat-top" windows in signal processing.

A flat-top window has the property that its sidelobes decay very rapidly, which means the autocorrelation is close to a delta function. But we can't make it exactly a delta (because the sequence must be non-negative and have fixed sum).

The optimal trade-off is:
- Make the power spectrum as flat as possible (to minimize `max(b)`)
- Subject to: `A(0) = 1` (DC constraint) and `a_i >= 0` (non-negativity)

This is essentially a constrained optimization problem in the frequency domain.

**New approach: Optimize in the frequency domain**

Let `A(f) = FFT(a)` (for a periodic extension of length N >= 2n-1).

The circular autocorrelation is `b_circ(k) = IFFT(|A(f)|^2)(k)`.

For N >> n, the circular autocorrelation approximates the linear autocorrelation in the "center" region.

We want to minimize `max_k b_circ(k)` subject to:
- `A(0) = 1` (DC = sum = 1)
- `a = IFFT(A) >= 0` (non-negativity)
- `A` is Hermitian (for real `a`)

The non-negativity constraint `IFFT(A) >= 0` is the hard part. Without it, the problem would be:
- Minimize `max_k IFFT(|A|^2)(k)` subject to `A(0) = 1`, `A` Hermitian.

This is a convex problem! (The objective is convex in `|A|^2`, and the constraints are linear.)

With the non-negativity constraint, it becomes non-convex. But we can use it as a guiding principle.

**Practical approach: Alternating projection**

1. Start with a valid `a` (non-negative, sum=1)
2. Compute `A = FFT(a)`
3. Modify `|A|^2` to be flatter (while keeping `A(0) = 1`)
4. Recover `A` from the modified `|A|^2` (keeping the phase)
5. Project back to the non-negative simplex
6. Repeat

This is similar to the Gerchberg-Saxton algorithm (used in phase retrieval).

```python
"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: QP-optimized active-set weights, equiripple linear refinement,
Nesterov-accelerated subgradient, aggressive L-BFGS beta continuation,
multi-n exploration (4096 and 8192), and Fourier-domain structured restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _solve_qp_simplex(G, max_iter=200):
    m = G.shape[0]
    x = np.ones(m) / m
    G_sym = 0.5 * (G + G.T)
    eta = 1.0 / max(1e-10, 2.0 * np.abs(G_sym).max() * m)
    for _ in range(max_iter):
        grad = 2.0 * (G_sym @ x)
        x_new = _project_simplex(x - eta * grad)
        if np.linalg.norm(x_new - x) < 1e-13:
            break
        x = x_new
    return x


def _compute_active_gradient(a, b, tol_frac):
    n = len(a)
    c_max = float(b.max())
    if c_max < 1e-30:
        return np.zeros(n)
    tol = tol_frac * c_max
    active_idx = np.where(b >= (c_max - tol))[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(b))])
    m = len(active_idx)
    if m == 1:
        k = int(active_idx[0])
        g = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                g[i] = 2.0 * a[j]
        return g
    G = np.zeros((m, m))
    for ii in range(m):
        ki = int(active_idx[ii])
        for jj in range(m):
            kj = int(active_idx[jj])
            k_min = max(0, ki - (n - 1), kj - (n - 1))
            k_max = min(ki, kj, n - 1)
            if k_min <= k_max:
                kr = np.arange(k_min, k_max + 1)
                G[ii, jj] = np.dot(a[ki - kr], a[kj - kr])
    G = 0.5 * (G + G.T)
    lam = _solve_qp_simplex(G)
    g = np.zeros(n)
    for idx in range(m):
        ki = int(active_idx[idx])
        li = lam[idx]
        if li < 1e-15:
            continue
        k_min = max(0, ki - (n - 1))
        k_max = min(ki, n - 1)
        if k_min <= k_max:
            i_range = np.arange(k_min, k_max + 1)
            g[i_range] += 2.0 * li * a[ki - i_range]
    return g

def _equiripple_refine(a_init, deadline, n_top=12, max_outer=30):

    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    for outer in range(max_outer):

        top_idx = np.argsort(b)[::-1][:n_top]

        M = np.zeros((n_top, n))
        for idx, k in enumerate(top_idx):
            k = int(k)
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            if i_start <= i_end:

        d_norm = np.linalg.norm(delta)
        if d_norm < 1e-14:

        if neg_mask.any():

        else:
            alpha_max = 1.0
        alpha_max = min(alpha_max, 2.0 / max(d_norm, 1e-10))
        best_alpha = 0.0
        best_f = _score(a)
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha_try = alpha_max * frac
            if alpha_try < 1e-12:
                break
            a_try = a + alpha_try * delta
            if a_try.min() < -1e-10:
                continue
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            f_try = _score(a_try)
            if f_try < best_f:
                best_f = f_try
                best_alpha = alpha_try
        if best_alpha < 1e-12:
            break
        a = a + best_alpha * delta
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.003)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.06, 0.03, 0.015,
                                   0.008, 0.004, 0.002, 0.001, 0.0005,
                                   0.0002, 0.0001, 0.00005])
    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_tpiter = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(30, int(remaining / max(est_tpiter, 0.001))))
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 60, 'maxls': 200}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            sc_cur = _score(current_a)
            if sc_new < sc_cur - 1e-14:
                current_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    return current_a


def _make_profile(n, kind):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.3, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_gaussian(a, rng, sigma=0.04):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _upsample(a, n_new):
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def update_best(p, sc, label):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            print("  NEW BEST: {:.10f} (n={}) [{}] improve={:.2e} t={:.1f}s".format(
                best_score_val, len(p), label, prev - best_score_val, time.monotonic() - t_start))

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print("Parent: n={}, score={:.10f}".format(n_parent, parent_sc))
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "tapered")
        parent_sc = _score(parent)
        print("No parent, using tapered n={}".format(n_parent))
        update_best(parent, parent_sc, "init")

    # Phase 1: Aggressive L-BFGS on parent (n=4096)
    print("\n[Phase 1] L-BFGS aggressive on parent t={:.1f}s".format(time.monotonic() - t_start))
    a1 = _run_lbfgs_aggressive(parent, deadline, min_time_remaining=600.0)
    sc1 = _score(a1)
    update_best(a1, sc1, "lbfgs_p1")

    # Phase 2: QP-weighted Nesterov subgradient
    print("\n[Phase 2] QP subgradient t={:.1f}s".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 500:
        a2, sc2 = _nesterov_subgrad(best_seq, min(deadline - 500, time.monotonic() + 180), tol_frac=0.004)
        update_best(a2, sc2, "qp_sg_p2")

    # Phase 3: Equiripple refinement
    print("\n[Phase 3] Equiripple t={:.1f}s".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 400:
        a3, sc3 = _equiripple_refine(best_seq, min(deadline - 400, time.monotonic() + 120), n_top=15, max_outer=40)
        update_best(a3, sc3, "equiripple_p3")
        if sc3 < best_score_val:
            # Follow up with L-BFGS
            a3b = _run_lbfgs_aggressive(a3, deadline, min_time_remaining=300.0)
            sc3b = _score(a3b)
            update_best(a3b, sc3b, "lbfgs_post_eq")

    # Phase 4: Subgradient again on refined
    print("\n[Phase 4] Subgradient round 2 t={:.1f}s".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 350:
        a4, sc4 = _nesterov_subgrad(best_seq, min(deadline - 350, time.monotonic() + 100), tol_frac=0.002)
        update_best(a4, sc4, "qp_sg_p4")

    # Phase 5: Try n=8192
    print("\n[Phase 5] n=8192 t={:.1f}s".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 250:
        a5_init = _upsample(best_seq, 8192)
        a5 = _run_lbfgs_aggressive(a5_init, deadline, min_time_remaining=100.0)
        sc5 = _score(a5)
        update_best(a5, sc5, "n8192_lbfgs")
        if sc5 < best_score_val + 0.01 and time.monotonic() < deadline - 80:
            a5b, sc5b = _nesterov_subgrad(a5, min(deadline - 60, time.monotonic() + 60), tol_frac=0.003)
            update_best(a5b, sc5b, "n8192_sg")

    # Phase 6: Fourier-domain restarts from best
    print("\n[Phase 6] Fourier restarts t={:.1f}s".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 60:
        for ri in range(4):
            if time.monotonic() > deadline - 40:
                break
            if ri % 2 == 0:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)
            else:
                p_start = _perturb_gaussian(best_seq, rng, sigma=0.03)
            sc_start = _score(p_start)
            a_r = _run_lbfgs_aggressive(p_start, deadline, min_time_remaining=30.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, "restart_{}".format(ri))
            if sc_r < best_score_val + 0.005 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _nesterov_subgrad(a_r, min(deadline - 10, time.monotonic() + 20), tol_frac=0.002)
                update_best(a_r2, sc_r2, "restart{}_sg".format(ri))

    # Phase 7: Final equiripple + subgradient polish
    print("\n[Phase 7] Final polish t={:.1f}s".format(time.monotonic() - t_start))
    if best_seq is not None and time.monotonic() < deadline - 10:
        a7, sc7 = _equiripple_refine(best_seq, min(deadline - 8, time.monotonic() + 30), n_top=8, max_outer=15)
        update_best(a7, sc7, "final_eq")
        if time.monotonic() < deadline - 5:
            a7b, sc7b = _nesterov_subgrad(best_seq, deadline - 5, tol_frac=0.001)
            update_best(a7b, sc7b, "final_sg")

    # Phase 8: One more L-BFGS with very small beta
    if best_seq is not None and time.monotonic() < deadline - 3:
        n_best = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00005
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(3000, max(50, int(remaining / 0.06)))
            try:
                result = minimize(
                    _smoobj, z, args=(beta_final, n_best),
                    jac=_smograd,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                sc_fin = _score(a_fin)
                update_best(a_fin, sc_fin, "final_lbfgs_tiny")
            except Exception:
                pass

    print("\n" + "=" * 60)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print("FINAL: score={:.10f}, n={}, time={:.1f}s".format(sc_final, len(final), time.monotonic() - t_start))
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print("FALLBACK: score={:.10f}".format(_score(fallback)))
    return [float(x) for x in fallback]
```
