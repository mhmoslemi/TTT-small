        e_z = np.exp(z - z.max())
        a = e_z / e_z.sum()

        grad, c, cm = _compute_grad_smooth(a, beta, N)
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = float(np.linalg.norm(update))
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 50 == 0:
            e_z2 = np.exp(z - z.max())
            a_chk = e_z2 / e_z2.sum()
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()

        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    e_z_final = np.exp(z - z.max())
    a_final = e_z_final / e_z_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_opt(a0, deadline, beta):
    from scipy.optimize import minimize
    n = len(a0)
    N = _next_pow2(2 * n - 1)
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.mean()
    best_sc = _score_norm(a0)
    best_z = z0.copy()

    def obj_fn(z):
        e_z = np.exp(z - z.max())
        a = e_z / e_z.sum()
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:2 * n - 1]
        cm = float(c.max())
        shifted = (c - cm) / beta
        e_c = np.exp(shifted)
        s_c = float(e_c.sum())
        if s_c < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s_c))

    def grad_fn(z):
        e_z = np.exp(z - z.max())
        a = e_z / e_z.sum()
        g, c, cm = _compute_grad_smooth(a, beta, N)
        correction = float(np.dot(a, g))
        return a * (g - correction)

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        return a0, best_sc

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 20)), 'gtol': 1e-12, 'maxls': 10}
        )
        z_opt = result.x
        e_z_o = np.exp(z_opt - z_opt.max())
        a_opt = e_z_o / e_z_o.sum()
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_sc:
            best_sc = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    e_z_b = np.exp(best_z - best_z.max())
    a_best = e_z_b / e_z_b.sum()
    return a_best, best_sc


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0 and float(parent_raw.sum()) > 0.01:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    best_seq = None
    best_score = np.inf

    n_values = [4096, 8192]
    if parent is not None and len(parent) not in n_values:
        n_values = [len(parent)] + n_values

    for n in n_values:
        if time.monotonic() > deadline - 30:
            break

        # Initialize
        if parent is not None and n == len(parent):
            a0 = parent.copy()
        elif parent is not None:
            a0 = _resample(parent, n)
        else:
            a0 = np.ones(n) / n

        sc0 = _score(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        N = _next_pow2(2 * n - 1)

        # Phase 1: Spectral flattening
        phase1_time = min(120.0, (deadline - t_start) * 0.3)
        p1_end = time.monotonic() + phase1_time
        print(f"n={n}: Phase 1 (spectral) time={phase1_time:.0f}s")
        a1, sc1 = _spectral_optimize(a0, p1_end)
        sc1_full = _score(a1)
        if sc1_full < best_score:
            best_score = sc1_full
            best_seq = a1.copy()
            print(f"  spectral: score={sc1_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase 2: Projected gradient
        if time.monotonic() < deadline - 20:
            c_curr = np.fft.irfft(np.fft.rfft(a1, N) * np.fft.rfft(a1, N), N)[:2 * n - 1]
            beta_pg = 0.005 * float(c_curr.max())
            p2_end = min(deadline - 15, time.monotonic() + 60.0)
            a2, sc2 = _proj_grad_opt(a1, p2_end, beta_init=beta_pg)
            sc2_full = _score(a2)
            if sc2_full < best_score:
                best_score = sc2_full
                best_seq = a2.copy()
                print(f"  proj_grad: score={sc2_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a1 = a2

        # Phase 3: Adam refinement
        if time.monotonic() < deadline - 15:
            c_curr = np.fft.irfft(np.fft.rfft(a1, N) * np.fft.rfft(a1, N), N)[:2 * n - 1]
            max_cc = float(c_curr.max())
            betas = [0.001 * max_cc, 0.0002 * max_cc, 0.00005 * max_cc]
            remaining = deadline - time.monotonic() - 15
            per_beta = remaining / len(betas)
            for bi, beta in enumerate(betas):
                if time.monotonic() > deadline - 12:
                    break
                a1, _ = _adam_opt(a1, time.monotonic() + per_beta, beta, lr_init=0.004, max_iter=500000)
                sc_a = _score(a1)
                if sc_a < best_score:
                    best_score = sc_a
                    best_seq = a1.copy()
                    print(f"  Adam b={beta:.2e}: score={sc_a:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase 4: L-BFGS polish
        if time.monotonic() < deadline - 10:
            c_curr = np.fft.irfft(np.fft.rfft(a1, N) * np.fft.rfft(a1, N), N)[:2 * n - 1]
            beta_lb = 0.00003 * float(c_curr.max())
            a1, _ = _lbfgs_opt(a1, deadline - 5.0, beta_lb)
            sc_lb = _score(a1)
            if sc_lb < best_score:
                best_score = sc_lb
                best_seq = a1.copy()
                print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final: try a few random restarts with spectral method if time remains
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_best = len(best_seq)
        N_best = _next_pow2(2 * n_best - 1)
        n_restarts = 3
        per_r = (deadline - time.monotonic() - 10) / n_restarts
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 8:
                break
            if ri == 0:
                a_r = _resample(best_seq, n_best) + 0.02 * rng.standard_normal(n_best)
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            elif ri == 1:
                a_r = _resample(best_seq, n_best)
                shift = rng.integers(1, max(2, n_best // 1024))
                a_r = np.roll(a_r, shift)
                a_r = a_r / a_r.sum()
            else:
                a_r = 0.7 * best_seq + 0.3 * (np.ones(n_best) / n_best)
                a_r = np.maximum(a_r, 0)
                a_r = a_r / a_r.sum()
            a_r, _ = _spectral_optimize(a_r, time.monotonic() + per_r * 0.6)
            c_r = np.fft.irfft(np.fft.rfft(a_r, N_best) * np.fft.rfft(a_r, N_best), N_best)[:2 * n_best - 1]
            b_r = 0.0005 * float(c_r.max())
            a_r, _ = _proj_grad_opt(a_r, time.monotonic() + per_r * 0.4, beta_init=b_r)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {ri}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    if best_seq is None:
        best_seq = np.ones(4096) / 4096

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.1
    print(f"FINAL: score={_score(final):.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
