Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral shaping initialization, active-set gradient methods,
and multi-resolution refinement.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume sum(a) = 1, return 2n * max(c)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    # So |A_target(k)| = sqrt(C_target(k))
    C_target = np.real(C_target)
    C_target = np.maximum(C_target, 0)
    mag_target = np.sqrt(C_target)
    
    # Initialize with random phases
    n_half = N // 2 + 1
    phases = rng.uniform(0, 2*np.pi, n_half)
    A = mag_target * np.exp(1j * phases)
    A[0] = mag_target[0]  # Real positive

Instead of the generic coordinate descent that tries a few candidate values, I'll implement a proper 1D optimization for each coordinate:

    For each active j, g_j(t) = c_base_j + d_j*t
    f(t) = 2n * max_j g_j(t) / (S0+t)^2
    
    The breakpoints are where g_j(t) = g_k(t) for different j,k
    Inside each region where a specific j is the max:
    f(t) = 2n * (c_base_j + d_j*t) / (S0+t)^2
    f'(t) = 2n * [d_j*(S0+t)^2 - (c_base_j+d_j*t)*2*(S0+t)] / (S0+t)^4
         = 2n * [d_j*(S0+t) - 2*(c_base_j+d_j*t)] / (S0+t)^3
         = 2n * [d_j*S0 + d_j*t - 2*c_base_j - 2*d_j*t] / (S0+t)^3
         = 2n * [d_j*S0 - 2*c_base_j - d_j*t] / (S0+t)^3
    
    Setting f'(t) = 0: d_j*S0 - 2*c_base_j - d_j*t = 0
    => t* = S0 - 2*c_base_j/d_j (if d_j > 0)
    """
    n = len(a)
    c = _full_convolve(a, a)
    S0 = a.sum() - a[i]  # sum of all except a[i]
    a_i_old = a[i]
    
    # Compute d_j: coefficient of t in c_j(t)
    d = np.zeros(2*n-1)
    # c_j = sum_k a_k * a_{j-k}
    # When a_i changes to t: c_j_new = c_j - a_i * a_{j-i} + t * a_{j-i} = c_j + (t-a_i)*a_{j-i}
    # So d_j = a_{j-i} for valid j (0 <= j-i < n)
    for j in range(i, min(i+n, 2*n-1)):
        d[j] = a[j-i]
    
    c_base = c - a_i_old * d
    
    # Find top positions by c_base + d * a_i_old (= c)
    # We only need to consider positions where c_j is close to max
    c_max = c.max()
    threshold = c_max * 0.9  # Consider positions within 90% of max
    relevant = np.where(c >= threshold)[0]
    
    # For each relevant position, compute the optimal t
    best_t = a_i_old
    best_f = 2.0 * n * c.max() / (a.sum())**2
    
    for j in relevant:
        if d[j] <= 1e-15:
            continue
        t_star = S0 - 2.0 * c_base[j] / d[j]
        if t_star < 0:
            t_star = 0.0
        # Evaluate f at t_star and nearby
        for t_eval in [t_star, max(0, t_star*0.5), t_star*1.5, 0.0]:
            if t_eval < 0 or t_eval > 10.0:
                continue
            c_trial = c_base + d * t_eval
            # The max might be at a different position now
            f_val = 2.0 * n * c_trial.max() / (S0 + t_eval)**2
            if f_val < best_f:
                best_f = f_val
                best_t = t_eval
    
    return best_t, best_f

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral shaping initialization, Adam in log-space with cosine LR,
exact coordinate descent polishing, and multi-restart diversity.
Key insight: gradient of <w, conv(a,a)> wrt a_i = 2*(a cross-corr w)[i].
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_weights(c, beta, k):
    """Compute softmax weights over top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    thresh = np.partition(c, -k)[-k]
    mask = c >= thresh
    top_idx = np.where(mask)[0]
    c_top = c[top_idx]
    c_max = float(c_top.max())
    exp_vals = np.exp((c_top - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[top_idx[0]] = 1.0
    else:
        w[top_idx] = exp_vals / w_sum
    return w


def _cross_corr(a, w, n):
    """Compute (a cross-corr w)[i] = sum_k a[k]*w[i+k] for i=0..n-1."""
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    return np.fft.irfft(np.conj(fa) * fw, N)[:n]


def _grad_conv_weighted(a, w, n):
    """Gradient of <w, conv(a,a)> wrt a. Returns 2*(a cross-corr w)."""
    return 2.0 * _cross_corr(a, w, n)


def _proj_simplex(v):
    """Project v onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _spectral_refine(a, n_iters=50, relax=0.7):
    """Iterative spectral shaping to flatten autocorrelation.
    Uses Gerchberg-Saxton style iteration in frequency domain."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: flat autocorrelation means |A|^2 should be "spread"
        # Current autocorr energy at each lag
        c = _full_convolve(a, a)
        c_norm = c / c.sum()  # Normalize to sum to 1
        c_target = np.ones(2 * n - 1) / (2 * n - 1)
        # Error in frequency domain
        C_curr = np.abs(A) ** 2
        C_target = np.fft.rfft(np.concatenate([c_target, np.zeros(N - (2 * n - 1))]))
        C_target = np.real(C_target)
        C_target = np.maximum(C_target, 0)

        # Blend magnitudes
        target_mag = np.sqrt(C_target)
        # Normalize target so that DC matches
        if target_mag[0] > 0:
            ratio = A[0].real / target_mag[0]
            # Scale to preserve sum(a)
            new_mag = target_mag * np.abs(A[0].real) / (target_mag[0] + 1e-30)
        else:
            new_mag = mag

        # Relaxation
        new_mag = relax * new_mag + (1 - relax) * mag
        new_mag[0] = mag[0]  # Keep DC (preserves sum)

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        if it > 5 and np.max(np.abs(a_new - a)) < 1e-12:
            a = a_new
            break
        a = a_new

    return a


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter,
                   cosine_warmup=500):
    """Adam optimizer in log-space with top-K softened max and cosine LR."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad_a = _grad_conv_weighted(a, w, n)
            mean_grad = float(np.dot(grad_a, a))
            grad_z = n * a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine LR schedule with restarts
            progress = min(1.0, t / max(1, len(beta_schedule) * max_iter))
            lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            lr = max(lr, lr_init * 0.01)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        # Dampen momentum at phase boundaries
        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -40, 40))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

        w = _topk_weights(c_y, beta, top_k)
        grad = _grad_conv_weighted(y, w, n)

        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_fista = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum", seed=42):
    rng = np.random.default_rng(seed)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "binary_random":
        a = rng.integers(0, 2, n).astype(float)
        a += 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
        n_parent = len(parent)
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple lengths
    if n_parent == 4096:
        lengths_to_try = [4096, 8192]
    elif n_parent == 8192:
        lengths_to_try = [8192, 4096]
    else:
        lengths_to_try = [4096, 8192, 2048]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic()
        print(f"\n=== n={n_try} (remaining: {remaining:.1f}s) ===")

        # Initialize
        if n_try == n_parent:
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)

        # Spectral refinement for better initialization
        a0_spec = _spectral_refine(a0, n_iters=30, relax=0.6)
        sc_spec = _score(a0_spec)
        print(f"  After spectral refine: score={sc_spec:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_spec < best_score:
            best_score = sc_spec
            best_seq = a0_spec.copy()
            a0 = a0_spec.copy()

        # Phase A: FISTA (coarse)
        phase_a_end = time.monotonic() + min(remaining * 0.2, 200.0)
        c_curr = _full_convolve(a0, a0)
        max_cc = float(c_curr.max())
        a_curr = a0.copy()

        betas_fista = []
        beta = 0.03 * max_cc
        while beta > 0.001 * max_cc:
            betas_fista.append(beta)
            beta *= 0.5
        betas_fista = betas_fista[:6]

        for bi, beta in enumerate(betas_fista):
            if time.monotonic() > phase_a_end - 2:
                break
            per_time = phase_a_end - time.monotonic() - 1
            max_it = max(1000, int(per_time * 2000))
            top_k = max(50, min(300, int(80 * (1 + bi * 0.3))))
            a_curr, sc = _fista_simplex(a_curr, time.monotonic() + min(30, per_time),
                                        beta, top_k, max_it)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  FISTA b{bi} beta={beta:.6f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase B: Adam in log-space (refinement)
        phase_b_end = time.monotonic() + min(remaining * 0.4, 400.0)
        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())
        betas_fine = []
        beta = 0.005 * max_cc
        while beta > 0.0001 * max_cc:
            betas_fine.append(beta)
            beta *= 0.5
        betas_fine = betas_fine[:10]

        a_opt, sc_opt = _adam_logspace(a_curr, phase_b_end, betas_fine,
                                       lr_init=0.01, top_k=100,
                                       max_iter=100000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

        a_work = a_opt if sc_opt < _score_norm(a_curr) else a_curr.copy()
        a_work = np.clip(a_work, 0, 1000)
        a_work = a_work / a_work.sum()

        # Phase C: Coordinate descent polishing
        phase_c_end = time.monotonic() + min(remaining * 0.15, 150.0)
        if phase_c_end - time.monotonic() > 10:
            a_cd, sc_cd = _coord_descend_exact(a_work, n_passes=5, deadline=phase_c_end, top_j=40)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_work = a_cd

        # Phase D: Ultra-fine Adam with very small beta
        if time.monotonic() < deadline - 60:
            phase_d_end = min(deadline - 30, time.monotonic() + 80)
            c_work = _full_convolve(a_work, a_work)
            max_cw = float(c_work.max())
            betas_ult = [0.001 * max_cw, 0.0003 * max_cw, 0.0001 * max_cw]
            a_ult, sc_ult = _adam_logspace(a_work, phase_d_end, betas_ult,
                                           lr_init=0.003, top_k=50,
                                           max_iter=50000)
            sc_ult_full = _score(a_ult)
            if sc_ult_full < best_score:
                best_score = sc_ult_full
                best_seq = a_ult.copy()
                print(f"  Ultra-fine: score={sc_ult_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase E: Multi-restart with diverse perturbations
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Multi-restart (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = 6
        per_restart = (deadline - time.monotonic() - 30) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            if ri == 0:
                # Gaussian noise
                p_start = a_base + 0.015 * rng.standard_normal(n_best)
            elif ri == 1:
                # Circular shift
                shift = rng.integers(1, max(2, n_best // 512))
                p_start = np.roll(a_base, shift).copy()
            elif ri == 2:
                # Low-frequency perturbation
                freq = rng.uniform(0.3, 2.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            elif ri == 3:
                # Mid-frequency perturbation
                freq = rng.uniform(3.0, 10.0)
                pert = 0.015 * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + rng.uniform(0, 6.28))
                p_start = a_base + pert
            elif ri == 4:
                # Block scaling
                p_start = a_base.copy()
                block = max(1, n_best // 16)
                si = rng.integers(0, n_best - block)
                p_start[si:si + block] *= rng.uniform(0.7, 1.4)
            else:
                # Spectral perturbation
                A = np.fft.rfft(a_base, _next_pow2(2 * n_best - 1))
                phase_pert = 0.05 * rng.standard_normal(len(A))
                A = np.abs(A) * np.exp(1j * (np.angle(A) + phase_pert))
                p_start = np.fft.irfft(A, _next_pow2(2 * n_best - 1))[:n_best]

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.003 * max_cs, 0.001 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs]

            p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                             betas_r, lr_init=0.008, top_k=80,
                                             max_iter=30000)
            # Quick coord descent polish
            p_best2, sc_best2 = _coord_descend_exact(p_best, n_passes=2,
                                                     deadline=time.monotonic() + min(10, per_restart * 0.3),
                                                     top_j=30)
            if sc_best2 < sc_best:
                p_best = p_best2
                sc_best = sc_best2

            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {ri}: score={sc_full:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # Final verification
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
