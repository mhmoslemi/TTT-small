"""
Sequence optimizer using FISTA + L-BFGS-B + subgradient polishing + coordinate descent.
Key improvements: L-BFGS-B with analytical gradient in log-space for second-order polish,
subgradient steps targeting the actual max position, adaptive perturbation search,
and multi-beta schedule for both FISTA and L-BFGS-B phases.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.ones(n) * (target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient_max(a, j_star):
    """Compute subgradient of conv(a,a)[j_star] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    valid_start = max(0, j_star - (n - 1))
    valid_end = min(n - 1, j_star)
    for m in range(valid_start, valid_end + 1):
        grad[m] = 2.0 * a[j_star - m]
    return grad


def _subgradient_max_fft(a, j_star):
    """Compute subgradient of conv(a,a)[j_star] w.r.t. a using FFT."""
    n = len(a)
    # subgrad[m] = 2*a[j_star - m] if 0 <= j_star - m < n
    # This is 2 * a reversed, shifted
    N = _next_pow2(2 * n)
    # Create a delta at position j_star, cross-correlate with a
    delta = np.zeros(N)
    delta[j_star] = 1.0
    fa = np.fft.rfft(a, N)
    fd = np.fft.rfft(delta, N)
    h = np.fft.irfft(np.conj(fa) * fd, N)[:n]
    return 2.0 * h


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    """FISTA on simplex with top-K softened max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _lbfgs_logspace(a_init, deadline, beta, top_k, max_iter=3000):
    """Run L-BFGS-B in log-space with analytical gradient."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.mean()

    def obj_fn(z):
        a = _softmax_normalize(z)
        c = _full_convolve(a, a)
        w = _topk_softmax_weights(c, beta, top_k)
        smooth_max = float(np.dot(w, c))
        return 2.0 * n * smooth_max

    def grad_fn(z):
        a = _softmax_normalize(z)
        c = _full_convolve(a, a)
        w = _topk_softmax_weights(c, beta, top_k)
        g = _compute_grad_conv(a, w)
        mean_g = float(np.dot(g, a))
        return a * (g - mean_g)

    t_start = time.monotonic()

    best_score = _score_norm(a_init)
    best_a = a_init.copy()

    try:
        result = scipy_minimize(
            obj_fn,
            z0,
            jac=grad_fn,
            method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-16,
                'gtol': 1e-10,
                'maxfun': max_iter * 2,
                'maxls': 30,
                'maxcor': 80
            }
        )
        z_result = result.x
        a_result = _softmax_normalize(z_result)
        sc = _score_norm(a_result)
        if sc < best_score:
            best_score = sc
            best_a = a_result.copy()
        elapsed = time.monotonic() - t_start
        print(f"    L-BFGS-B b={beta:.2e} k={top_k}: score={sc:.10f} nit={result.nit} t={elapsed:.1f}s")
    except Exception as e:
        print(f"    L-BFGS-B error: {e}")

    return best_a, best_score


def _subgradient_polish(a, n_steps, deadline):
    """Polish using subgradient of the actual max."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    best_a = a.copy()

    c = _full_convolve(a, a)
    max_c = float(c.max())

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        j_star = int(np.argmax(c))
        max_c = float(c[j_star])

        # Adaptive step size
        step = 0.01 * max_c / (n * max_c / max(1e-10, a.mean() * n))
        step = min(step, 0.5 * a.mean())

        sg = _subgradient_max_fft(a, j_star)
        a_new = _proj_simplex(a - step * sg)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce step
            step *= 0.5
            a_new = _proj_simplex(a - step * sg)
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Coordinate descent polishing with proper 1D minimization."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for i in order:
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue

            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            idx_top = np.argsort(c)[-top_k_w:]

            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5, a_i_old * 2.0,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2]

            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 10.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.8))
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(10.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if abs(best_t - a_i_old) > 1e-15:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1

            if improved % 200 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _perturbation_search(a, n_attempts, deadline, rng, sigma=0.01):
    """Try random perturbations in log-space and keep improvements."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break

        # Random perturbation
        pert = rng.standard_normal(n) * sigma
        z_pert = z + pert
        a_pert = _softmax_normalize(z_pert)

        sc = _score_norm(a_pert)
        if sc < best_score:
            best_score = sc
            a = a_pert.copy()
            z = z_pert.copy()
            # Decrease sigma on success
            sigma *= 0.95

    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        x = np.arange(n) / n
        for freq in [1, 2, 3, 4, 5]:
            a += 0.2 / freq * np.cos(2.0 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "multi_freq":
        x = np.arange(n) / n
        a = 1.0
        freqs = [1, 3, 5, 7, 11]
        for f in freqs:
            a += 0.15 * np.cos(2 * np.pi * f * x + f * 0.5)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        # Multiple beta values for L-BFGS-B, from coarse to fine
        betas_lbfgs = [0.005 * max_cc, 0.003 * max_cc, 0.002 * max_cc,
                       0.001 * max_cc, 0.0005 * max_cc, 0.0003 * max_cc, 0.0002 * max_cc]

        per_beta_time = phase_b_budget / len(betas_lbfgs)
        for bi, beta in enumerate(betas_lbfgs):
            if time.monotonic() > phase_b_end - 2:
                break
            beta_deadline = time.monotonic() + min(per_beta_time, phase_b_end - time.monotonic() - 1)
            top_k = max(50, min(200, int(200 * (1 - 0.1 * bi))))
            a_work, sc = _lbfgs_logspace(a_work, beta_deadline, beta, top_k,
                                         max_iter=2000)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
                print(f"  After L-BFGS-B b={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # === Phase C: Subgradient polishing ===
        phase_c_budget = min(remaining * 0.10, 80.0)
        phase_c_end = time.monotonic() + phase_c_budget
        if phase_c_budget > 5:
            a_work, sc_sg = _subgradient_polish(a_work, n_steps=500, deadline=phase_c_end)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
                print(f"  Subgradient: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # === Phase D: Coordinate descent ===
        phase_d_budget = min(remaining * 0.15, 150.0)
        phase_d_end = time.monotonic() + phase_d_budget
        if phase_d_budget > 10:
            a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=5,
                                                top_k_w=80, deadline=phase_d_end)
            a_cd = a_cd / a_cd.sum()
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_work = a_cd

        # === Phase E: Final L-BFGS-B polish on best ===
        if time.monotonic() < deadline - 60:
            phase_e_end = min(deadline - 60, time.monotonic() + 60)
            a_best_local = best_seq.copy()
            a_best_local = a_best_local / a_best_local.sum()
            c_b = _full_convolve(a_best_local, a_best_local)
            max_cb = float(c_b.max())
            betas_final = [0.0001 * max_cb, 0.00005 * max_cb, 0.00003 * max_cb]
            for beta in betas_final:
                if time.monotonic() > phase_e_end:
                    break
                a_pol, sc_pol = _lbfgs_logspace(a_best_local, phase_e_end, beta, 30,
                                                max_iter=3000)
                a_pol = a_pol / a_pol.sum()
                sc_pol_full = _score(a_pol)
                if sc_pol_full < best_score:
                    best_score = sc_pol_full
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS b={beta:.2e}: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_best_local = a_pol

    # === Phase F: Perturbation search from best ===
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Perturbation search (remaining: {deadline - time.monotonic():.1f}s) ===")
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_base = len(a_base)

        pert_deadline = min(deadline - 20, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        n_attempts = int((pert_deadline - time.monotonic()) * 500)  # ~500 attempts/sec
        a_pert, sc_pert = _perturbation_search(a_base, n_attempts, pert_deadline, rng, sigma=0.005)
        a_pert = a_pert / a_pert.sum()
        sc_pert_full = _score(a_pert)
        if sc_pert_full < best_score:
            best_score = sc_pert_full
            best_seq = a_pert.copy()
            print(f"  Perturbation: score={sc_pert_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase G: Diversity restarts ===
    if time.monotonic() < deadline - 20 and best_seq is not None:
        print(f"\n=== Diversity restarts (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        per_restart = max(5, (deadline - time.monotonic() - 15) / 3)

        for ri in range(3):
            if time.monotonic() > deadline - 15:
                break

            if ri == 0:
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ri == 1:
                # Frequency shift
                shift = rng.integers(1, max(2, n_best // 2048))
                p_start = np.roll(a_base, shift).copy()
            else:
                # Anti-phase perturbation
                freq = rng.uniform(1.0, 5.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())

            # Quick L-BFGS-B
            p_best, sc = _lbfgs_logspace(p_start, time.monotonic() + per_restart,
                                         0.001 * max_cs, 50, max_iter=2000)
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        else:
            final = final / s * s  # Keep sum the same
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
