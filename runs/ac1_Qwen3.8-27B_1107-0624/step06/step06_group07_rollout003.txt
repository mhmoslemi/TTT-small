"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-guided active set method, Nesterov-accelerated subgradient
in log-space, L-BFGS with adaptive beta continuation, and multi-restart
diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgrad_active(a, c, n, tol_frac=1e-4):
    """Compute subgradient using active set."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)

    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=30.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.10, 0.05, 0.025, 0.012, 0.006, 0.003,
                              0.0015, 0.0008, 0.0004, 0.0002,
                              0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_iter_time = 0.002 * (n / 2048.0) ** 1.3
        max_iter = min(12000, max(200, int(remaining / (est_iter_time * 12))))
        max_iter = min(max_iter, 20000)

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _nesterov_mirrordesc(a_init, deadline, max_steps=50000):
    """Nesterov-accelerated mirror descent in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()

    best_a = a_init.copy()
    best_sc = _score(a_init)
    t_val = 1.0
    momentum = 0.9
    stall = 0
    gamma = 1.0 / np.sqrt(n)

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a_y = _softmax(y)
        c_y = _full_convolve(a_y, a_y)
        c_max = c_y.max()

        # Adaptive active set
        tol_frac = min(0.01, 1e-4 * (1 + i / 500.0))
        tol = tol_frac * c_max
        active = c_y >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c_y, dtype=bool)
            active[int(np.argmax(c_y))] = True
        w = np.zeros_like(c_y)
        w[active] = 1.0 / active.sum()

        # Subgradient in z-space
        a_rev = a_y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c_y))
        grad = 4.0 * n * a_y * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        f_lower = 2.0 * n / (2.0 * n - 1.0)
        f_current = 2.0 * n * c_max
        step = gamma / np.sqrt(t_val + 1.0)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        theta = (t_val - 1.0) / t_new
        z_new = y - step * grad
        y = (1.0 - theta) * z_new + theta * z
        z = z_new
        t_val = t_new

        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        y = np.clip(y, -80.0, 80.0)
        y = y - y.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
            else:
                stall += 1
                gamma *= 0.95
                if stall > 5:
                    break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _lp_active_set(a_init, deadline, max_iters=500):
    """LP-guided active set method for final polishing."""
    if not HAS_CVXPY:
        return a_init, _score(a_init)

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.05
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Adaptive active set: start broad, tighten over iterations
        tol_frac = max(1e-6, 0.01 * (1.0 - it / float(max_iters)))
        tol = tol_frac * c_max
        active_mask = c >= (c_max - tol)
        active_idx = np.where(active_mask)[0]

        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])

        # Limit active set size for LP efficiency
        max_active = min(len(active_idx), 100)
        if len(active_idx) > max_active:
            top_vals = c[active_idx]
            top_order = np.argsort(top_vals)[::-1][:max_active]
            active_idx = active_idx[top_order]

        # Solve LP: min t s.t. (a*d)(k) <= t for k in active, ||d||_inf <= 1, sum(d)=0
        d_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = []
        for k in active_idx:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            d_indices = k - np.arange(j_min, j_max + 1)
            constraints.append(a[j_min:j_max + 1] @ d_var[d_indices] <= t_var)

        constraints.append(d_var <= 1.0)
        constraints.append(d_var >= -1.0)
        constraints.append(cp.sum(d_var) == 0)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                break

        if d_var.value is None or t_var.value is None:
            break

        t_opt = float(t_var.value)
        if t_opt >= -1e-10:
            # No descent direction found
            stall += 1
            if stall > 5:
                break
            continue

        d = np.array(d_var.value, dtype=np.float64).ravel()
        d = np.clip(d, -1.0, 1.0)
        d = d - d.mean()

        # Line search
        found = False
        for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
            trial = a + step_size * shrink * d
            trial = np.maximum(trial, 0.0)
            s = float(trial.sum())
            if s < 1e-15:
                continue
            trial = trial / s
            sc_trial = _score(trial)
            if sc_trial < best_sc - 1e-14:
                best_a = trial.copy()
                best_sc = sc_trial
                a = trial
                found = True
                stall = 0
                break

        if not found:
            step_size *= 0.7
            stall += 1
            if stall > 20:
                break

    return best_a, best_sc


def _subgrad_z_polish(a_init, deadline, tol_frac_init=0.01, max_steps=20000):
    """Subgradient polish in z-space with adaptive parameters."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.03
    stall_count = 0
    momentum_buf = np.zeros(n)
    momentum = 0.85

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Adaptive tolerance
        tol_frac = max(1e-5, tol_frac_init * (0.95 ** (i // 100)))
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Momentum
        momentum_buf = momentum * momentum_buf + (1 - momentum) * grad
        step = lr / (1.0 + i * 0.0002)
        z = z - step * momentum_buf
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.02, 0.1)
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 4:
                    lr *= 0.7
                    momentum *= 0.9
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.93

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.005 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_local_scale(a, rng, n_perturb=50):
    """Perturb a random subset of coordinates multiplicatively."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    indices = rng.choice(n, size=min(n_perturb, n), replace=False)
    z[indices] += rng.uniform(-0.05, 0.05, size=len(indices))
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _local_pair_search(a, deadline, max_rounds=1000):
    """Local search: try pairwise ratio adjustments."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.003
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        idx = int(rng.integers(0, n))
        jdx = int(rng.integers(0, n))
        if idx == jdx:
            jdx = (jdx + 1) % n

        s_pair = best_a[idx] + best_a[jdx]
        if s_pair < 1e-15:
            continue

        for delta_frac in [0.3, 0.5, 0.7]:
            for dir_sign in [-1, 1]:
                new_val_i = s_pair * (1.0 - delta_frac + dir_sign * step)
                new_val_j = s_pair * (delta_frac - dir_sign * step)
                if new_val_i < 0 or new_val_j < 0:
                    continue

                a_try = best_a.copy()
                a_try[idx] = new_val_i
                a_try[jdx] = new_val_j
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)

                if sc_try < best_sc - 1e-13:
                    best_a = a_try.copy()
                    best_sc = sc_try
                    improved = True

        if not improved and rnd > 50:
            step *= 0.85
            if step < 1e-8:
                break
        elif improved:
            step = min(step * 1.01, 0.01)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with adaptive beta,
    Nesterov mirror descent, LP-guided active set, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Main L-BFGS with beta continuation ===
    print(f"\n=== Phase 1: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=600.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Nesterov mirror descent ===
    print(f"\n=== Phase 2: Nesterov MD (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_nm, sc_nm = _nesterov_mirrordesc(best_seq, deadline - 400, max_steps=30000)
        try_update(a_nm, "phase2_nesterov")

    # === Phase 3: Z-space subgradient polish ===
    print(f"\n=== Phase 3: Z-space subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 300,
                                        tol_frac_init=0.005, max_steps=15000)
        try_update(a_sg, "phase3_sg_z")

    # === Phase 4: LP-guided active set ===
    print(f"\n=== Phase 4: LP active set (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200 and HAS_CVXPY:
        a_lp, sc_lp = _lp_active_set(best_seq, deadline - 200, max_iters=300)
        try_update(a_lp, "phase4_lp")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(8000, max(500, int(remaining / 0.003)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 6: Second round of LP + subgradient ===
    print(f"\n=== Phase 6: LP + SG round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        if HAS_CVXPY:
            a_lp2, _ = _lp_active_set(best_seq, deadline - 100, max_iters=200)
            try_update(a_lp2, "phase6_lp2")

        if best_seq is not None and time.monotonic() < deadline - 50:
            a_sg2, _ = _subgrad_z_polish(best_seq, deadline - 50,
                                         tol_frac_init=0.001, max_steps=10000)
            try_update(a_sg2, "phase6_sg2")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            ("gauss_002", "gauss", 0.02, 0),
            ("gauss_005", "gauss", 0.05, 0),
            ("fourier_3", "fourier", 0, 3),
            ("fourier_5", "fourier", 0, 5),
            ("local_scale", "local", 0, 0),
        ]

        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 60:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, modes)
            else:
                p_start = _perturb_local_scale(best_seq, rng, n_perturb=100)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            if sc_start < best_score_val + 0.01:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=30.0)
                try_update(a_r, f"restart_{name}")

    # === Phase 8: Final aggressive polish ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 3:
                break

            # Super-fine L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-7
            a_f, _, _ = _lbfgs_run(best_seq, beta, n, max_iter=5000)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # LP step
            if HAS_CVXPY and time.monotonic() < deadline - 5:
                a_l, _ = _lp_active_set(best_seq, deadline - 3, max_iters=50)
                try_update(a_l, f"final_lp_{rnd}")

            # Local pair search
            if time.monotonic() < deadline - 3:
                a_ls, _ = _local_pair_search(best_seq, deadline - 2, max_rounds=200)
                try_update(a_ls, f"final_ls_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
