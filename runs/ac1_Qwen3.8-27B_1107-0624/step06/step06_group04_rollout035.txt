```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _lp_step(a, M, top_k=20):
    """Solve the LP: max sum(b) s.t. (a*b)[k] <= M for top-k k values, b >= 0"""
    n = len(a)
    conv = _full_convolve(a, a)
    # Get top-k indices
    top_idx = np.argsort(conv)[::-1][:top_k]
    
    # Build constraint matrix: A[i, j] = a[top_idx[i] - j] if 0 <= top_idx[i] - j < n
    K = len(top_idx)
    A = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        # (a*b)[k] = sum_j a[k-j] * b[j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        A[i, j_min:j_max+1] = a[k - j_max:k - j_min + 1][::-1]
    
    # LP: max 1^T b s.t. A b <= M * ones, b >= 0
    # Convert to min: min -1^T b s.t. A b <= M * ones, -b <= 0
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
```

```python
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a, conv=None):
    n = len(a)
    if conv is None:
        conv = _full_convolve(a, a)
    return float(2.0 * n * float(conv.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    u = np.sort(v / s)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v / s - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        a_next = _proj_simplex(y - step * grad)
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=5000, lr_decay_factor=0.97, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > grad_clip:
                update = update * (grad_clip / un)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)
            if no_improve > 40 and t > 500:
                current_lr = max(0.0003, current_lr * 0.9)
                no_improve = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_direction(a, M, top_k=30, time_limit=10.0):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for top-k k, b >= 0.
    Uses constraint generation to ensure feasibility.
    Returns rescaled b (sum=1) or None.
    """
    n = len(a)
    conv_aa = _full_convolve(a, a)
    top_idx = list(np.argsort(conv_aa)[::-1][:top_k].astype(int))
    t_start = time.monotonic()

    for cg_iter in range(30):
        if time.monotonic() - t_start > time_limit:
            break
        K = len(top_idx)
        # Build sparse constraint matrix
        row_list = []
        col_list = []
        val_list = []
        for i, k in enumerate(top_idx):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                cols = np.arange(j_min, j_max + 1)
                vals = a[k - j_max:k - j_min + 1][::-1]
                row_list.extend([i] * (j_max - j_min + 1))
                col_list.extend(cols.tolist())
                val_list.extend(vals.tolist())

        if not val_list:
            return None

        A = coo_matrix(
            (np.array(val_list, dtype=np.float64),
             (np.array(row_list, dtype=np.int32),
              np.array(col_list, dtype=np.int32))),
            shape=(K, n)
        ).tocsr()

        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(1.0, time_limit - (time.monotonic() - t_start))})
        except Exception:
            return None

        if not res.success:
            return None

        b = res.x.copy()
        s_b = b.sum()
        if s_b < 1e-15:
            return b

        # Check constraint violations
        conv_ab = _full_convolve(a, b)
        violations = conv_ab - M
        max_viol = float(violations.max())

        if max_viol < 1e-8 * max(M, 1e-10):
            b = b / s_b
            return b

        # Add most violated constraint
        new_k = int(np.argmax(violations))
        if new_k not in top_idx:
            top_idx.append(new_k)

    # Final solve with current active set
    K = len(top_idx)
    row_list = []
    col_list = []
    val_list = []
    for i, k in enumerate(top_idx):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            cols = np.arange(j_min, j_max + 1)
            vals = a[k - j_max:k - j_min + 1][::-1]
            row_list.extend([i] * (j_max - j_min + 1))
            col_list.extend(cols.tolist())
            val_list.extend(vals.tolist())

    if not val_list:
        return None

    A = coo_matrix(
        (np.array(val_list, dtype=np.float64),
         (np.array(row_list, dtype=np.int32),
          np.array(col_list, dtype=np.int32))),
        shape=(K, n)
    ).tocsr()

    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 15.0})
    except Exception:
        return None

    if res.success:
        b = res.x.copy()
        s_b = b.sum()
        if s_b > 1e-15:
            return b / s_b
    return None


def _lp_phase(a_init, deadline, max_iters=200):
    """LP-based optimization phase."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    consec_no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        conv = _full_convolve(a, a)
        M = float(conv.max())
        s = float(a.sum())

        b_dir = _lp_direction(a, M, top_k=40, time_limit=8.0)
        if b_dir is None:
            break

        # Line search
        best_t = 0.0
        best_sc = _score_norm(a)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            a_trial = (1.0 - t) * a + t * b_dir
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t == 0.0:
            consec_no_improve += 1
            if consec_no_improve > 10:
                break
            continue

        a = (1.0 - best_t) * a + best_t * b_dir
        a = np.maximum(a, 0)
        a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            consec_no_improve = 0
        else:
            consec_no_improve += 1

        if it % 20 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t={best_t:.3f}")

    sc_final = _score_norm(best_a)
    if sc_final < best_score:
        best_score = sc_final
    return best_a, best_score


def _spectral_flatten(a_init, n_iter=200, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _subgradient_polish(a_init, n_steps=3000, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.3

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        conv = _full_convolve(a, a)
        cm = float(conv.max())
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(conv)[::-1][:top_k]
        w_vals = np.maximum(conv[top_idx] - cm, 0) + 1e-10
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_idx] = w_vals ** 3
        w_vec = w_vec / w_vec.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w_vec, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_new = _proj_simplex(a - step_size * grad)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.97
            a_new2 = _proj_simplex(a - step_size * grad)
            if _score_norm(a_new2) < _score_norm(a):
                a = a_new2

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        a = parent.copy()
        n = len(a)
        print(f"Loaded parent: n={n}")
    else:
        n = 4096
        a = np.ones(n) / n
        print("No parent, using uniform n=4096")

    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial score: {best_score:.10f}")

    # === Phase 1: Quick FISTA to get a good starting point ===
    print("Phase 1: FISTA beta-continuation")
    phase1_deadline = min(deadline, t_start + 200.0)
    a_curr = a.copy()
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas = []
    beta = 0.01 * max_cc
    while beta > 5e-6 * max_cc:
        betas.append(beta)
        beta *= 0.5

    for i, b in enumerate(betas):
        if time.monotonic() > phase1_deadline - 5:
            break
        remaining = phase1_deadline - time.monotonic() - 5
        if remaining < 3:
            break
        per_time = min(remaining / (len(betas) - i), 15.0)
        a_curr, sc = _fista_smoothed(a_curr, b, time.monotonic() + per_time, 20000)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a_curr.copy()
            print(f"  FISTA beta={b:.6e}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    a_work = best_a.copy()
    a_work = a_work / a_work.sum()
    print(f"After Phase 1: score={best_score:.10f}")

    # === Phase 2: LP-based optimization (the key new phase) ===
    print("Phase 2: LP-based constraint generation")
    phase2_deadline = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_deadline - 30:
        a_lp, sc_lp = _lp_phase(a_work, phase2_deadline, max_iters=300)
        sc_lp_full = _score(a_lp)
        if sc_lp_full < best_score:
            best_score = sc_lp_full
            best_a = a_lp.copy()
            print(f"  LP phase: score={sc_lp_full:.10f} t={time.monotonic()-t_start:.0f}s")
        a_work = best_a.copy()
        a_work = a_work / a_work.sum()

    # === Phase 3: Adam log-space fine-tuning ===
    print("Phase 3: Adam log-space fine-tuning")
    phase3_deadline = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_deadline - 30:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_fine = []
        b = 0.0003 * max_cw
        while b > 1e-7 * max_cw:
            betas_fine.append(b)
            b *= 0.5
        if len(betas_fine) > 10:
            betas_fine = betas_fine[:10]

        per_t = phase3_deadline - time.monotonic() - 10
        a_adam, sc_adam = _adam_logspace(
            a_work, time.monotonic() + per_t, betas_fine,
            lr_init=0.008, max_iter_total=500000,
            lr_decay_every=8000, lr_decay_factor=0.96)
        sc_adam_full = _score(a_adam)
        if sc_adam_full < best_score:
            best_score = sc_adam_full
            best_a = a_adam.copy()
            print(f"  Adam: score={sc_adam_full:.10f} t={time.monotonic()-t_start:.0f}s")
        a_work = best_a.copy()
        a_work = a_work / a_work.sum()

    # === Phase 4: Second LP pass on the improved solution ===
    print("Phase 4: Second LP pass")
    phase4_deadline = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_deadline - 30:
        a_lp2, sc_lp2 = _lp_phase(a_work, phase4_deadline, max_iters=150)
        sc_lp2_full = _score(a_lp2)
        if sc_lp2_full < best_score:
            best_score = sc_lp2_full
            best_a = a_lp2.copy()
            print(f"  LP2: score={sc_lp2_full:.10f} t={time.monotonic()-t_start:.0f}s")
        a_work = best_a.copy()
        a_work = a_work / a_work.sum()

    # === Phase 5: Multi-restart with perturbations ===
    print("Phase 5: Multi-restart diversification")
    phase5_deadline = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_deadline - 60:
        n_restarts = 4
        per_restart = (phase5_deadline - time.monotonic() - 20) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_deadline - 10:
                break
            sigmas = [0.01, 0.02, 0.03, 0.05]
            p_start = _perturb(a_work, sigmas[i], rng)
            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            p_opt, _ = _fista_smoothed(p_start, 0.001 * max_cs,
                                       time.monotonic() + max(3.0, per_restart * 0.3), 5000)
            p_opt = p_opt / p_opt.sum()
            # Quick Adam
            if time.monotonic() < phase5_deadline - 15:
                betas_r = [0.0003 * max_cs, 0.0001 * max_cs, 5e-5 * max_cs]
                per_t_r = phase5_deadline - time.monotonic() - 10
                p_opt, sc_r = _adam_logspace(
                    p_opt, time.monotonic() + per_t_r, betas_r,
                    lr_init=0.005, max_iter_total=50000)
                sc_r_full = _score(p_opt)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_a = p_opt.copy()
                    print(f"  Restart {i} (sigma={sigmas[i]}): score={sc_r_full:.10f}")

    # === Phase 6: Subgradient + ultra-fine Adam polish ===
    print("Phase 6: Final polish")
    a_work = best_a.copy()
    a_work = a_work / a_work.sum()

    if time.monotonic() < deadline - 40:
        a_sg, _ = _subgradient_polish(a_work, n_steps=2000, deadline=min(deadline - 60, time.monotonic() + 60))
        sc_sg = _score(a_sg)
        if sc_sg < best_score:
            best_score = sc_sg
            best_a = a_sg.copy()
            print(f"  Subgradient: score={sc_sg:.10f}")
        a_work = best_a.copy()
        a_work = a_work / a_work.sum()

    if time.monotonic() < deadline - 20:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = [1e-5 * max_cp, 5e-6 * max_cp, 2e-6 * max_cp]
        per_t = deadline - time.monotonic() - 10
        if per_t > 10:
            a_pol, _ = _adam_logspace(
                a_work, time.monotonic() + per_t, betas_ult,
                lr_init=0.002, max_iter_total=200000,
                lr_decay_every=10000, lr_decay_factor=0.95)
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol.copy()
                print(f"  Ultra-fine Adam: score={sc_pol:.10f}")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
        s = final.sum()
        final = final / s

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
