"""
Improved multi-phase optimizer with frequency-domain fixed-point and coordinate descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume sum(a) = 1, return 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    # Project v onto {x >= 0, sum(x) = target_sum}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] with respect to a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 80 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 25 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 4000 == 0:
            current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 30)), 'gtol': 1e-12, 'maxls': 15}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _freq_fixed_point(a_init, deadline, max_iter=200):
    """Frequency-domain fixed-point iteration for flat autocorrelation."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Softmax weights (smoothed active set)
        beta_fp = 0.01 * cm
        w = np.exp((c - cm) / beta_fp)
        ws = float(w.sum())
        if ws < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws
        
        # Frequency domain: |A_new[k]|^2 should be proportional to 1/W[k]
        # where W is the DFT of w (zero-padded to N)
        W = np.fft.rfft(w, N)
        # Avoid division by zero
        W_abs2 = np.abs(W) ** 2
        W_abs2 = np.maximum(W_abs2, 1e-30)
        # Target power spectrum
        target_power = 1.0 / W_abs2
        # Normalize
        target_power = target_power / target_power.sum()
        # We want |A[k]|^2 = target_power * (sum a)^2 = target_power
        # Phase: use the current phase of A (or try zero phase)
        A_curr = np.fft.rfft(a, N)
        # Option 1: Keep current phase
        A_new = np.sign(A_curr) * np.sqrt(target_power)
        # But rfft has real first element, handle carefully
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        
        # Blend with current to maintain stability
        alpha = 0.5
        a_new = (1 - alpha) * a + alpha * a_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_score

def _coordinate_descent(a_init, deadline, max_passes=200):
    """Coordinate-wise optimization: adjust one element at a time."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Pre-compute the convolution
    c = _full_convolve(a, a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order for diversity
        order = np.arange(n)
        np.random.shuffle(order)
        for i in order:
            if time.monotonic() > deadline:
                break
            # Try different values for a[i]
            # The contribution of a[i] to c[k] is: a[i]*a[k-i] + a[i]*a[i-k] (for valid k)
            # To minimize max(c), we want to set a[i] to reduce the peak c values.
            
            # Binary search for optimal a[i] given others fixed
            # c[k] = a[i]^2 (if k=0) + a[i]*(stuff) + (rest)
            # Actually c[k] is quadratic in a[i]: c[k] = a[i]^2 * delta(k) + a[i] * b[k] + r[k]
            # where b[k] = sum of other terms involving a[i], r[k] = terms not involving a[i]
            
            # Simpler: try a few values
            a_old = a[i]
            # Try: 0, half, 2x, and a few intermediate values
            candidates = [0.0, a_old * 0.25, a_old * 0.5, a_old * 0.75, a_old, a_old * 1.25, a_old * 1.5, 2.0 * a_old]
            # Also try based on gradient direction
            c_max = float(c.max())
            # Gradient of c_max w.r.t. a[i] (approximate)
            # c[k] involves a[i] if k-i >= 0 and k-i < n, or i-k >= 0 and i-k < n
            # grad_c[i] = sum of a[j] for j such that |j-i| < n (in the convolution)
            
            best_val = a_old
            best_local_score = c_max
            
            for val in candidates:
                if val < 0:
                    continue
                a[i] = val
                # Update convolution efficiently
                # c[k] changes by: delta = val^2 - a_old^2 (at k=0... wait, this is full conv)
                # Actually for full convolution c = conv(a,a), changing a[i] by delta:
                # c[k] += 2*delta*a[k-i] + delta^2 (for valid k-i)
                # This is getting complex. Let me just recompute.
                if abs(val - a_old) > 1e-15:
                    c_new = _full_convolve(a, a)
                    sc_new = float(c_new.max())
                    if sc_new < best_local_score:
                        best_local_score = sc_new
                        best_val = val
                        c = c_new  # Keep updated
                a[i] = a_old
            
            if best_val != a_old:
                a[i] = best_val
                # Renormalize
                a = a / a.sum()
                c = _full_convolve(a, a)
                sc = float(2.0 * n * c.max())
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_score

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t**2 / 0.3)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_clipped":
        t = np.linspace(-1, 1, n)
        a = np.sinc(t * 2) ** 2  # sinc^2 is non-negative
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Improved multi-phase optimizer with frequency-domain fixed-point, deflation,
and aggressive beta annealing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_weights(c, beta):
    cm = float(c.max())
    e = np.exp((c - cm) / beta)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    c_max_init = float(c_init.max())
    L = 4.0 * n / max(beta, c_max_init * 0.001)
    step = 1.0 / L
    best_score = 2.0 * n * c_max_init
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        w, c_max = _compute_weights(c_y, beta)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 60 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final

    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        w, cm = _compute_weights(c, beta)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.90)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

    from scipy.optimize import minimize
    n = len(a_init)

        s = float(e.sum())
        if s < 1e-300:

    def grad_fn(z):

        c = _full_convolve(a, a)
        cm = float(c.max())

            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(80, int(max_time * 40)), 'gtol': 1e-13, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _freq_fixed_point(a_init, deadline, max_iter=300):
    """Frequency-domain fixed-point: target |A[k]|^2 ~ 1/|W[k]|^2 where W = DFT(w)."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        beta_fp = 0.005 * float(c.max())
        w, cm = _compute_weights(c, beta_fp)
        W = np.fft.rfft(w, N)
        W_abs2 = np.abs(W) ** 2
        W_abs2 = np.maximum(W_abs2, 1e-20)
        target_power = 1.0 / W_abs2
        target_power = target_power / target_power.sum()
        A_curr = np.fft.rfft(a, N)
        # Use current phase
        A_new = np.abs(A_curr).copy()
        # Blend power spectrum
        alpha = 0.3 + 0.5 * (it / max(1, max_iter))
        new_magnitude = np.sqrt((1 - alpha) * np.abs(A_curr) ** 2 + alpha * target_power * float(np.abs(A_curr) ** 2).sum() / target_power.sum())
        phase = np.angle(A_curr)
        A_new = new_magnitude * np.exp(1j * phase)
        A_new[0] = np.abs(A_new[0])  # DC is real
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()
        # Blend for stability
        blend = 0.4
        a_new = (1 - blend) * a + blend * a_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _deflation(a_init, deadline, max_rounds=100):
    """Targeted peak reduction: reduce values contributing most to peak lags."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Find top contributing lags
        top_k = max(1, len(c) // 50)
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[top_k - 1]
        active_lags = np.where(c >= threshold)[0]
        # For each active lag k, the contribution to a[i] is a[k-i]
        # Build a "pressure" on each coordinate
        pressure = np.zeros(n)
        for k in active_lags:
            # c[k] involves a[i]*a[k-i] for valid i
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    pressure[i] += a[j]
        # Reduce high-pressure coordinates, boost low-pressure ones
        eta = 0.02 * (1.0 - 0.01 * r)
        a_shifted = a - eta * pressure
        a_shifted = np.maximum(a_shifted, 1e-15)
        a_new = a_shifted / a_shifted.sum()
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_shifted = a - 0.5 * eta * pressure
            a_shifted = np.maximum(a_shifted, 1e-15)
            a_new2 = a_shifted / a_shifted.sum()
            sc2 = _score_norm(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
            else:
                break

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t ** 2 / 0.25)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_sq":
        t = np.linspace(-1, 1, n)
        x = t * np.pi
        a = np.sin(x) / x
        a = a ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 18.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))
    np.random.seed(int(seed) % (2 ** 32 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "raised_cosine")
        sc0 = _score(a0)
        print(f"No parent, raised_cosine n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: FISTA with geometric beta annealing ===
    print("Phase 1: FISTA geometric beta annealing")
    phase1_end = min(deadline, t_start + 320.0)
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    n_phases = 35
    beta_start = 0.08 * max_c0
    beta_end = 0.00005 * max_c0

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 5:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        remaining = phase1_end - time.monotonic() - 3
        if remaining < 1:
            break
        per_time = min(remaining / max(1, n_phases - phase), 28.0)
        max_it = max(3000, int(per_time * 5000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 7 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Frequency-domain fixed point ===
    print("Phase 2: Frequency-domain fixed point")
    phase2_end = min(deadline, t_start + 520.0)
    if time.monotonic() < phase2_end - 10:
        a_fp, _ = _freq_fixed_point(a_work, phase2_end, max_iter=400)
        sc_fp = _score(a_fp)
        if sc_fp < best_score:
            best_score = sc_fp
            best_seq = a_fp.copy()
            print(f"  Freq FP: score={sc_fp:.10f} t={time.monotonic()-t_start:.1f}s")
        # Also try deflation from FISTA result
        a_def, _ = _deflation(a_work, time.monotonic() + 60.0, max_rounds=80)
        sc_def = _score(a_def)
        if sc_def < best_score:
            best_score = sc_def
            best_seq = a_def.copy()
            print(f"  Deflation: score={sc_def:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = a_def

    # === Phase 3: Adam fine optimization (multiple betas) ===
    print("Phase 3: Adam fine optimization")
    phase3_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase3_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0004 * max_cc, 0.0001 * max_cc, 0.00002 * max_cc, 0.000004 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 10) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase3_end - 10:
                break
            per_t = max(5.0, min(per_beta, phase3_end - time.monotonic() - 10 - bi * 5))
            a_work, _ = _adam_opt(a_work, time.monotonic() + per_t,
                                  beta, lr_init=0.004, max_iter=500000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS refinement ===
    print("Phase 4: L-BFGS refinement")
    phase4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase4_end - 20:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00001 * max_cc
        a_lb, _ = _lbfgs_opt(a_work, phase4_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 5: Deflation + re-optimize ===
    print("Phase 5: Deflation + re-optimize")
    phase5_end = min(deadline, t_start + 1420.0)
    if time.monotonic() < phase5_end - 20:
        a_def2, _ = _deflation(a_work, time.monotonic() + (phase5_end - time.monotonic()) * 0.3, max_rounds=60)
        c_d = _full_convolve(a_def2, a_def2)
        max_cd = float(c_d.max())
        b_r = 0.0001 * max_cd
        a_def2, _ = _fista_opt(a_def2, b_r, time.monotonic() + (phase5_end - time.monotonic()) * 0.3, 40000)
        a_def2, _ = _adam_opt(a_def2, time.monotonic() + (phase5_end - time.monotonic()) * 0.35, b_r * 0.2, 0.003, 80000)
        sc_d2 = _score(a_def2)
        if sc_d2 < best_score:
            best_score = sc_d2
            best_seq = a_def2.copy()
            a_work = a_def2
            print(f"  Deflate+reopt: score={sc_d2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Multi-restart exploration ===
    print("Phase 6: Multi-restart")
    phase6_end = min(deadline, t_start + 1640.0)
    if time.monotonic() < phase6_end - 25:
        n_restarts = 6
        per_r = (phase6_end - time.monotonic() - 20) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            if i == 0:
                p = _perturb(a_work, 0.008, rng)
            elif i == 1:
                p = _make_profile(n_main, "raised_cosine")
                p = 0.75 * a_work + 0.25 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                p = _make_profile(n_main, "gaussian")
                p = 0.5 * a_work + 0.5 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 3:
                p = _make_profile(n_main, "sinc_sq")
                p = 0.4 * a_work + 0.6 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 4:
                shift = rng.integers(1, max(2, n_main // 2048))
                p = np.roll(a_work, shift).copy()
                p = p / p.sum()
            else:
                p = _perturb(a_work, 0.02, rng)
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.00015 * max_cp
            t_r = time.monotonic() + per_r * 0.4
            p, _ = _fista_opt(p, b_r, t_r, 30000)
            t_r2 = time.monotonic() + per_r * 0.4
            p, _ = _adam_opt(p, t_r2, b_r * 0.25, 0.003, 60000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-polish ===
    print("Phase 7: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            # Very fine Adam
            t_adam = time.monotonic() + remaining * 0.4
            a_pol, _ = _adam_opt(a_pol, t_adam, 0.000002 * max_cp, 0.0008, 300000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # L-BFGS on remaining
            if time.monotonic() < deadline - 15:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000001 * float(c_f.max())
                a_pol2, _ = _lbfgs_opt(a_pol, deadline - 8.0, b_f)
                sc_f = _score(a_pol2)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol2.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")
            # Final deflation
            if time.monotonic() < deadline - 10:
                a_def_f, _ = _deflation(best_seq, deadline - 3.0, max_rounds=30)
                sc_df = _score(a_def_f)
                if sc_df < best_score:
                    best_score = sc_df
                    best_seq = a_def_f.copy()
                    print(f"  Final Deflate: score={sc_df:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "raised_cosine")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
