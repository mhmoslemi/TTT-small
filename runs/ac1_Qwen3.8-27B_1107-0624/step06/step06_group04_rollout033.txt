Improved optimizer using LP-based constraint generation (from state-of-the-art LP approach),
combined with FISTA smoothed optimization and Adam log-space fine-tuning.
Key innovation: uses scipy linprog to solve the LP max sum(g) s.t. f*g <= M, g>=0 at each

(n=4096, n=8192) and aggressive final polishing.

import numpy as np
import time
import scipy.sparse
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _build_conv_matrix_rows(f, active_lags):

    n = len(f)
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(active_lags):

        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(row_idx)
            cols.append(j)
            vals.append(f[k - j])
    if len(rows) == 0:
        return scipy.sparse.csr_matrix((1, n))
    A = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    return A.tocsr()

    t_start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() - t_start > max_time:

        conv_ff = _full_convolve(f, f)
        M = float(conv_ff.max())

        # Identify active lags (top-k closest to M)
        k_top = min(top_k, 2 * n - 1)
        active_lags = np.argsort(conv_ff)[::-1][:k_top]
        # Only include lags that are within 95% of max (to keep LP tight)

            active_lags = np.argsort(conv_ff)[::-1][:20]

        # Build constraint matrix
        A = _build_conv_matrix_rows(f, active_lags)
        b = np.full(len(active_lags), M)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_time - (time.monotonic() - t_start)})

        if res.status != 0 or res.x is None:
            break

        g = res.x.copy()
        sum_g = float(g.sum())

        if sum_g <= 1.0 + 1e-8:
            # No improvement from LP
            t_mix *= 0.8
            if t_mix < 0.01:
                break
            continue

        # Rescale g to sum 1 and mix
        g = g / sum_g
        f_new = (1.0 - t_mix) * f + t_mix * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        sc_new = _score_norm(f_new)
        if sc_new < _score_norm(f) - 1e-12:
            f = f_new
            if sc_new < best_score:
                best_score = sc_new
                best_f = f.copy()
            t_mix = min(0.4, t_mix * 1.05)
        else:
            t_mix *= 0.7
            if t_mix < 0.005:
                break

    return best_f, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=8000, lr_decay_factor=0.97, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 40 and t > 500:
                current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_nesterov(a_init, n_steps, deadline, lower_bound=None):
    """Nesterov-accelerated projected subgradient with top-K averaging."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)

    if lower_bound is None:
        lower_bound = 1.0 / (2 * n - 1)

    v = np.zeros(n)
    t_val = 1.0
    step_scale = 1.0 / np.sqrt(n_steps + 1)

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - (a - v / t_val)) if t_val > 1 else a
        # Nesterov: y = a + (t-1)/(t+1) * (a - a_prev) where a_prev = a - v/t
        a_prev = a - v / t_val if t_val > 0 else a
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = _proj_simplex(y)

        conv_y = _full_convolve(y, y)
        M = float(conv_y.max())

        # Top-K gradient averaging
        K = min(30, 2 * n - 1)
        top_idx = np.argsort(conv_y)[::-1][:K]
        # Weight by closeness to max
        weights = (conv_y[top_idx] - conv_y[top_idx].min() + 1e-10) ** 4
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for idx, wi in zip(top_idx, weights):
            a_rev = y[::-1]
            ei = np.zeros(2 * n - 1)
            ei[idx] = 1.0
            wc = _full_convolve(ei, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad += wi * 2.0 * h

        # Polyak-like step size
        gap = M - lower_bound
        if gap < 1e-12:
            break
        step_size = step_scale * gap / (np.linalg.norm(grad) ** 2 + 1e-10)
        step_size = min(step_size, 0.1)

        a_new = _proj_simplex(y - step_size * grad)
        v = (t_val / t_next) * (v + t_next * (a_new - a))
        a = a_new
        t_val = t_next

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None

    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")

        a0 = np.ones(n_main) / n_main

        best_seq = a0.copy()

    max_cc = float(c_curr.max())

    a_work = best_seq.copy()

    print(f"End Phase 1: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    print("Phase 2: LP-based constraint generation on n=" + str(n_main))

    # === Phase 3: FISTA + LP alternating for more improvement ===
    print("Phase 3: Alternating FISTA+LP")
    phase3_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase3_end - 20:
        c_work = _full_convolve(a_work, a_work)

                a_tmp = a_tmp / a_tmp.sum()

            sc_tmp = _score(a_tmp)
            if sc_tmp < best_score:
                best_score = sc_tmp
                best_seq = a_tmp.copy()
                a_work = a_tmp
                print(f"  Alt round {round_i}: score={sc_tmp:.10f}")
            else:
                a_work = a_tmp  # still use it as starting point

    a_work = best_seq / best_seq.sum()
    print(f"End Phase 3: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

            a_8k_opt = a_8k_opt / a_8k_opt.sum()
