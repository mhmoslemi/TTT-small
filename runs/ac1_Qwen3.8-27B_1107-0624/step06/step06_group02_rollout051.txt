import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _lp_direction(a, active_indices, M):
    """
    Solve LP: maximize sum(g)
    Subject to: (a*g)[k] <= M for k in active_indices
    g >= 0
    
    Returns: g (the solution) or None if the LP is infeasible/unbounded
    """
    n = len(a)
    K = len(active_indices)
    
    # Objective: minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraint matrix: K x n
    # Row k (for active index active_indices[k]):
    #   sum_j a[active_indices[k] - j] * g[j] <= M
    #   (where a[m] = 0 if m < 0 or m >= n)
    
    A_rows = []
    for k in active_indices:
        row = np.zeros(n)
        # g[j] appears with coefficient a[k-j] for j in [max(0,k-n+1), min(k,n-1)]
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            row[j] = a[k - j]
        A_rows.append(row)
    
    A_mat = np.array(A_rows)
    b_vec = np.full(K, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        g = np.maximum(g, 0)  # numerical safety
        return g
    else:
        return None

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 500 == 0:

            if sc < best_sc:

        a_prev = a_curr

        t_val = t_next
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if it_in_phase % 5000 == 0:

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    a = a / float(np.sum(a))
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.98, 0.99, 1.01, 1.02]:
                new_val = old_val * factor
                a[j] = new_val

                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
                    break
        a = a / float(np.sum(a))
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc

    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n0 = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n0}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n0 = 4096
        a0 = np.ones(n0) / n0
        sc0 = _score(a0)
        print(f"No parent, uniform n={n0}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # Phase 1: LP optimization at parent n (4096)
    print("Phase 1: LP optimization at n={}".format(len(a0)))
    phase1_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase1_end - 10:
        a_lp, sc_lp = _lp_optimize(a0, phase1_end, K_init=200, stall_limit=100)
        sc_full = _score(a_lp)
        print(f"  LP n={len(a_lp)}: score={sc_full:.10f} (t={time.monotonic()-t_start:.1f}s)")
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_lp.copy()
        a_work = a_lp
    else:
        a_work = a0

    # Phase 2: Interpolate to n=8192 and LP optimize
    print("Phase 2: LP at n=8192")
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end - 60:
        n_big = 8192
        a_big = _interp_profile(a_work, n_big)
        sc_big_init = _score(a_big)
        print(f"  Interpolated n={n_big}: score={sc_big_init:.10f}")
        if sc_big_init < best_score:
            best_score = sc_big_init
            best_seq = a_big.copy()
        a_big2, sc_big2 = _lp_optimize(a_big, phase2_end, K_init=250, stall_limit=80)
        sc_big_full = _score(a_big2)
        print(f"  LP n={n_big}: score={sc_big_full:.10f} (t={time.monotonic()-t_start:.1f}s)")
        if sc_big_full < best_score:
            best_score = sc_big_full
            best_seq = a_big2.copy()
        a_work = a_big2
    else:
        a_work = best_seq.copy()
        a_work = a_work / float(np.sum(a_work))

    # Phase 3: Adam fine-tuning on best sequence
    print("Phase 3: Adam fine-tuning")
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end - 30:
        aAdam = a_work.copy()
        aAdam = aAdam / float(np.sum(aAdam))
        c_ad = _convolve(aAdam, aAdam)
        max_cd = float(c_ad.max())
        betas_fine = [0.001 * max_cd, 0.0005 * max_cd, 0.0002 * max_cd,
                      0.0001 * max_cd, 0.00005 * max_cd, 0.00001 * max_cd]
        per_t = phase3_end - time.monotonic() - 15
        if per_t > 30:
            a_adam, sc_adam = _adam_logspace(
                aAdam, time.monotonic() + per_t,
                betas_fine, lr_init=0.008, max_iter_total=500000
            )
            sc_adam_full = _score(a_adam)
            print(f"  Adam: score={sc_adam_full:.10f} (t={time.monotonic()-t_start:.1f}s)")
            if sc_adam_full < best_score:
                best_score = sc_adam_full
                best_seq = a_adam.copy()

    # Phase 4: Multi-restart with different initializations
    print("Phase 4: Multi-restart")
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end - 60 and best_seq is not None:
        n_w = len(best_seq)
        a_ref = best_seq.copy()
        a_ref = a_ref / float(np.sum(a_ref))
        n_restarts = 4
        per_restart = (phase4_end - time.monotonic() - 30) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            if i == 0:
                p_start = _perturb(a_ref, 0.03, rng)
            elif i == 1:
                # Cosine envelope
                j = np.arange(n_w)
                p_start = (1.0 + np.cos(2.0 * np.pi * j / n_w)) / 2.0
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 2:
                # Exponential decay + wrap
                j = np.arange(n_w)
                sigma_n = n_w / 8.0
                p_start = np.exp(-j**2 / (2 * sigma_n**2))
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # Mix of parent and random
                p_rand = rng.random(n_w)
                p_rand = np.maximum(p_rand, 1e-10)
                p_rand = p_rand / p_rand.sum()
                p_start = 0.5 * a_ref + 0.5 * p_rand
                p_start = p_start / p_start.sum()

            sc_init = _score(p_start)
            if sc_init < best_score:
                best_score = sc_init
                best_seq = p_start.copy()

            # Quick LP + Adam on this restart
            rem = phase4_end - time.monotonic() - 20
            if rem > 40:
                lp_frac = rem * 0.6
                adam_frac = rem * 0.35
                p_lp, _ = _lp_optimize(p_start, time.monotonic() + lp_frac, K_init=200, stall_limit=50)
                sc_lp = _score(p_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_seq = p_lp.copy()
                if time.monotonic() < time.monotonic() + adam_frac - 5:
                    c_r = _convolve(p_lp, p_lp)
                    max_r = float(c_r.max())
                    bs_r = [0.001 * max_r, 0.0005 * max_r, 0.0001 * max_r, 0.00005 * max_r]
                    p_ad, _ = _adam_logspace(
                        p_lp, time.monotonic() + adam_frac,
                        bs_r, lr_init=0.006, max_iter_total=200000
                    )
                    sc_ad = _score(p_ad)
                    if sc_ad < best_score:
                        best_score = sc_ad
                        best_seq = p_ad.copy()
                        print(f"  Restart {i}: score={sc_ad:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # Phase 5: Try n=16384 interpolation
    print("Phase 5: n=16384 exploration")
    phase5_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        n_xl = 16384
        a_xl = _interp_profile(best_seq, n_xl)
        sc_xl = _score(a_xl)
        print(f"  Interpolated n={n_xl}: score={sc_xl:.10f}")
        if sc_xl < best_score:
            best_score = sc_xl
            best_seq = a_xl.copy()
        if time.monotonic() < phase5_end - 20:
            rem5 = phase5_end - time.monotonic() - 15
            if rem5 > 30:
                a_xl2, _ = _lp_optimize(a_xl, time.monotonic() + rem5, K_init=300, stall_limit=40)
                sc_xl2 = _score(a_xl2)
                print(f"  LP n={n_xl}: score={sc_xl2:.10f} (t={time.monotonic()-t_start:.1f}s)")
                if sc_xl2 < best_score:
                    best_score = sc_xl2
                    best_seq = a_xl2.copy()

    # Phase 6: Final FISTA + coordinate descent polish
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / float(np.sum(a_pol))
        n_pol = len(a_pol)

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        beta_fin = 0.0001 * max_cp
        rem6 = deadline - time.monotonic() - 5
        if rem6 > 20:
            a_f1, _ = _fista_smoothed(a_pol, beta_fin, time.monotonic() + rem6 * 0.5, 100000)
            a_f1 = a_f1 / float(np.sum(a_f1))
            sc_f1 = _score(a_f1)
            if sc_f1 < best_score:
                best_score = sc_f1
                best_seq = a_f1.copy()
            if time.monotonic() < deadline - 5:
                a_cd, _ = _coord_descent(best_seq, deadline - 3, max_rounds=200)
                sc_cd = _score(a_cd)
                if sc_cd < best_score:
                    best_score = sc_cd
                    best_seq = a_cd.copy()

    # Final validation
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_final = float(np.sum(final))
        if s_final < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
