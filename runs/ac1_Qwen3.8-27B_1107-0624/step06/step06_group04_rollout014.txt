So the gradient of the smoothed objective is:
`∂f/∂a[i] = sum_m w[m] * 2*a[m-i] * [0 ≤ m-i < n]`

This is very structured! It's like a weighted sum of `a` shifted by different amounts. We can compute it efficiently using FFT or direct convolution.

Specifically, let `r[i] = a[n-1-i]` (reversal). Then:
`∂f/∂a[i] = 2 * sum_m w[m] * a[m-i] * [0 ≤ m-i < n]`

Let `j = m-i`:
`= 2 * sum_j w[i+j] * a[j] * [0 ≤ i+j < 2n-1, 0 ≤ j < n]`

This is a cross-correlation of `w` and `a`, which can be computed via FFT.

Actually, let me define it differently. Let `g[i] = ∂f/∂a[i] / 2 = sum_m w[m] * a[m-i] * [0 ≤ m-i < n]`.

This is `(w ∘ a)[i]` where `∘` denotes cross-correlation: `(w ∘ a)[i] = sum_m w[m] * a[m-i]`.

We can compute this via FFT: `g = ifft(fft(w, N) * conj(fft(a, N)))` (with appropriate length N ≥ 2n-1).

Great, so we can compute the gradient efficiently.

"""
Optimized step function search using L-BFGS-B with smoothed max objective,
multi-scale optimization, and coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _objective_and_grad(x, beta):
    """Smoothed max of conv(a,a) and its gradient."""
    n = len(x)
    c = _full_convolve(x, x)
    cm = float(c.max())
    
    # Compute softmax weights
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s_w = float(e.sum())
    if not np.isfinite(s_w) or s_w < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_w
    
    f = cm + (1.0 / beta) * np.log(s_w)
    
    # Gradient computation
    w_rev = w[::-1]
    N = _next_pow2(2 * n - 1)
    conv = irfft(rfft(x, N) * rfft(w_rev, N), N)[:2*n-1]
    h = conv[n-1:2*n-1][::-1]
    grad = 2.0 * h
    
    return f, grad


def _lbfgs_optimize(a_init, beta, max_iter=5000, tol=1e-10):
    """Run L-BFGS-B optimization with given beta."""
    n = len(a_init)
    
    def func_and_grad(x):
        f, g = _objective_and_grad(x, beta)
        return f, g
    
    result = minimize(
        func_and_grad, a_init, method='L-BFGS-B',
        jac=True,
        bounds=[(0, None)] * n,
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8, 'maxls': 50}
    )
    return result.x, result.fun


def _coord_descent(a_init, n_sweeps=5, deadline=None):
    """Coordinate descent: minimize max(conv(a,a)) by adjusting pairs of coordinates."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_score = _score(a)
    best_a = a.copy()
    
    N_fft = _next_pow2(2 * n - 1)
    
    for sweep in range(n_swepes):
        if deadline and time.monotonic() > deadline:
            break
        for i in range(n):
            if deadline and time.monotonic() > deadline:
                break
            # Try adjusting a[i] (keeping sum constant by adjusting a uniformly? No, that's expensive)
            # Instead, try: reduce a[i] and see if score improves
            # Or: swap mass between coordinates
            
            # Simple approach: try setting a[i] to 0 (if it's small) and renormalize
            if a[i] > 1e-10:
                a_old_i = a[i]
                a[i] = 0
                s_new = a.sum()
                if s_new > 0.01:
                    a_norm = a / s_new
                    sc = _score(a_norm)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_norm.copy()
                    else:
                        a[i] = a_old_i
                        s = a.sum()
                        a = a / s
    
    return best_a, best_score

"""
Optimized step function search using L-BFGS-B with adaptive beta-continuation,
multi-scale optimization (4096 → 8192), subgradient polishing with momentum,
and coordinate-level local search.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _obj_grad(x, beta):
    """Compute smoothed_max(conv(x,x))/(sum(x))^2 and its gradient."""
    n = len(x)
    S = float(x.sum())
    if S < 1e-30:
        return 1.0, np.zeros(n)

    c = _full_convolve(x, x)
    cm = float(c.max())

    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s_w = float(e.sum())
    if not np.isfinite(s_w) or s_w < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_w

    f_sm = cm + (1.0 / beta) * np.log(s_w)

    # h[i] = sum_d x[d] * w[i+d]  (gradient of smoothed_max_c is 2*h)
    w_rev = w[::-1]
    N = _next_pow2(2 * n - 1)
    conv = irfft(rfft(x, N) * rfft(w_rev, N), N)[:2 * n - 1]
    h = conv[n - 1:2 * n - 1][::-1]

    obj = f_sm / (S * S)
    grad = 2.0 * (h * S - f_sm) / (S * S * S)

    return obj, grad


def _lbfgs_run(a_init, beta, max_iter, deadline):
    """Run L-BFGS-B optimization with given beta, respecting deadline."""
    n = len(a_init)

    def func(x):
        f, _ = _obj_grad(x, beta)
        return f

    def func_grad(x):
        return _obj_grad(x, beta)

    count = [0]

    def callback(xk):
        count[0] += 1
        if count[0] % 50 == 0 and time.monotonic() > deadline:
            raise TimeoutError()

    try:
        result = minimize(
            func_grad, a_init, method='L-BFGS-B',
            jac=True, bounds=[(0.0, None)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9,
                     'maxls': 30, 'maxfun': max_iter * 3}
        )
        x_best = result.x
    except (TimeoutError, Exception):
        x_best = a_init

    # Normalize
    s = x_best.sum()
    if s > 0:
        x_best = x_best / s
    return x_best


def _fista_smoothed(a_init, beta, deadline, max_iter=10000):
    """FISTA on the simplex with smoothed max objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    # Estimate Lipschitz constant
    c0 = _full_convolve(a_curr, a_curr)
    L = 4.0 * float(c0.max()) / (beta * (a_curr.sum() ** 2))
    if L < 1:
        L = 10.0
    step = 1.0 / L

    best_score = float(_score(a_curr))
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

        y = np.maximum(y, 0.0)
        s_y = y.sum()

            y = np.ones(n) / n

        s_next = a_next.sum()
        if s_next > 1e-30:

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

        if it % 200 == 0:

            if sc < best_score:

                best_a = a_curr.copy()

    sc_final = _score(a_curr)
    if sc_final < best_score:
