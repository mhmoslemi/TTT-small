                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)
    factors = [0.90, 0.95, 0.98, 1.02, 1.05, 1.10]

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            for factor in factors:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
            # Also try setting to a small value
            if old_val > 1e-3:
                a[j] = old_val * 0.5
                s = a.sum()
                if s > 1e-30:
                    sc = _score_norm(a / s)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                    else:
                        a[j] = old_val

        a = best_a.copy()
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return np.exp(z - z.max()) / np.exp(z - z.max()).sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.15f}")

    # === Phase 1: FISTA with aggressive adaptive beta continuation ===
    print("Phase 1: FISTA adaptive beta continuation")
    phase1_end = min(deadline, t_start + 500.0)
    a_curr = a0.copy()

    # Generate beta schedule adaptively
    beta = 0.02 * max_c0
    while beta > max(1e-15 * max_c0, 1e-20):
        per_time = min(30.0, phase1_end - time.monotonic() - 3)
        if per_time < 3:
            break
        max_it = max(2000, int(per_time * 2000))
        # Use top_k for stability at moderate beta
        ck = _convolve(a_curr, a_curr)
        cmax_k = float(ck.max())
        # Detect how many values are near the max
        threshold = cmax_k - 2.0 * beta
        n_active = int(np.sum(ck >= threshold))
        tk = min(max(3, n_active), 64) if n_active > 2 else None

        a_curr, sc = _fista_smoothed(a_curr, beta, time.monotonic() + per_time, max_it, top_k=tk)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _convolve(a_curr, a_curr)
        max_c_curr = float(c_curr.max())
        beta *= 0.35

    print(f"End Phase 1: score={best_score:.10f}")

    # === Phase 2: Subgradient + Nesterov on exact max ===
    print("Phase 2: Subgradient Nesterov")
    phase2_end = min(deadline, t_start + 750.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    if time.monotonic() < phase2_end - 20:
        a_sg, sc_sg = _subgrad_nesterov(
            a_work, time.monotonic() + min(120.0, phase2_end - time.monotonic() - 10),
            max_iter=50000, step_scale=0.3
        )
        a_sg = a_sg / a_sg.sum()
        sc_sg_full = _score(a_sg)
        print(f"  Subgrad: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()

    # === Phase 3: Adam log-space fine optimization ===
    print("Phase 3: Adam log-space")
    phase3_end = min(deadline, t_start + 1200.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    c_curr = _convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    betas_fine = []
    beta = 0.0005 * max_cc
    while beta > 1e-16 * max_c0:
        betas_fine.append(beta)
        beta *= 0.3
    if len(betas_fine) > 20:
        betas_fine = betas_fine[::len(betas_fine) // 20] + [1e-16 * max_c0]
    if not betas_fine:
        betas_fine = [1e-10 * max_cc]

    per_time = phase3_end - time.monotonic() - 15
    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.008, max_iter_total=3000000
        )
        a_opt = a_opt / a_opt.sum()
        sc_opt_full = _score(a_opt)
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()

    # === Phase 4: Coordinate descent polish ===
    print("Phase 4: Coordinate descent")
    if time.monotonic() < deadline - 150:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 100, time.monotonic() + 120)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=200)
        a_cd = a_cd / a_cd.sum()
        sc_cd_full = _score(a_cd)
        print(f"  Coord: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 5: Multi-restart with diverse perturbations ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase5_end - 60 and best_seq is not None:
        n_restarts = 6
        per_restart = (phase5_end - time.monotonic() - 30) / n_restarts

            c_s = _convolve(p_start, p_start)

            # Quick FISTA polish

    print("Phase 6: n=8192 exploration")
