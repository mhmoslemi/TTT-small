"""
LP-based iterative optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal update directions, combined with
gradient-based polishing for fine-tuning.
"""
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_matrix(f, n):
    """Build the sparse constraint matrix for the LP.
    A[j, k] = f[k] if 0 <= j-k <= n-1, else 0
    A is (2n-1) x n
    """
    rows = []
    cols = []
    data = []
    for k in range(n):
        j_start = k
        j_end = min(k + n, 2 * n - 1)
        # A[j, k] = f[k] for j in [j_start, j_end)
        js = np.arange(j_start, j_end)
        rows.append(js)
        cols.append(np.full(len(js), k))
        data.append(np.full(len(js), f[k]))
    
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A


def _solve_lp_direction(f, M, n):
    """Solve the LP: max sum(g) s.t. A @ g <= M, g >= 0.
    Returns the optimal g (unnormalized).
    """
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    b = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None


def _lp_iterate(f, M, n, t=1.0):
    """One LP iteration. Returns updated f."""
    g = _solve_lp_direction(f, M, n)
    if g is None:
        return f
    s = float(g.sum())
    if s < 1e-15:
        return f
    g = g / s  # Normalize to sum 1
    f_new = (1 - t) * f + t * g
    return f_new


def _line_search_lp(f, g_unnorm, n):
    """Line search for the LP update. Returns best f and best t."""
    g = g_unnorm / float(g_unnorm.sum())
    best_f = f.copy()
    best_score = _score_norm(f)
    
    for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
        f_trial = (1 - t) * f + t * g
        sc = _score_norm(f_trial)
        if sc < best_score:
            best_score = sc
            best_f = f_trial.copy()
    
    return best_f


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.97)
    
    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

2*n*max(conv(a,a))/sum(a)^2. Uses linear programming to find optimal

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    cols = np.concatenate(cols_list)

        j_start = k
        j_end = min(k + n, 2 * n - 1)
        valid = np.intersect1d(np.arange(j_start, j_end), active_indices, assume_sorted=False)
        if len(valid) > 0:
            rows_list.append(valid.astype(np.int32))
            cols_list.append(np.full(len(valid), k, dtype=np.int32))
            data_list.append(np.full(len(valid), f[k], dtype=np.float64))
    if not rows_list:
        return None
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    return csr_matrix((data, (rows, cols)), shape=(len(active_indices), n))


def _solve_lp(f, M, n, active_indices=None):
    """Solve LP: max sum(g) s.t. (f*g)_j <= M for active j, g >= 0."""
    if active_indices is not None:
        A = _build_reduced_lp_matrix(f, n, active_indices)
        if A is None or A.nnz == 0:
            A = _build_lp_matrix(f, n)
        b = np.full(A.shape[0], M)
    else:
        A = _build_lp_matrix(f, n)
        b = np.full(2 * n - 1, M)

    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 60})

    if res.success:
        return res.x
    return None


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_count = 0
        while it_count < max_iter:
            if time.monotonic() > deadline:
                break
            it_count += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_count % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15 * (t - 0.3) ** 2) + np.exp(-15 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent for fine polishing."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                a[i] = 1e-4
                a = a / a.sum()
                continue

            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            idx_top = np.argsort(c)[-top_k_w:]
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5,
                           a_i_old * 0.7, a_i_old * 0.3]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 5.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)

            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 10) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = _score_norm(a)
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    return a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"Initialized: n={n_parent}, score={best_score:.10f}")

    # ============ Phase 1: LP-based optimization (n=4096) ============
    n_lp = 4096
    if len(parent) == n_lp:
        f = parent.copy()
    else:
        f = np.interp(np.linspace(0, 1, n_lp), np.linspace(0, 1, len(parent)), parent)
        f = np.maximum(f, 1e-12)
        f = f / f.sum()

    print(f"\n=== LP Phase: n={n_lp}, initial score={_score(f):.10f} ===")

    lp_budget = 850
    lp_end = t_start + lp_budget
    best_f_lp = f.copy()
    best_score_norm = _score_norm(f)
    num_iters = 0
    prev_best = best_score_norm
    stagnation_count = 0

    # Track active constraints for reduced LP
    use_reduced = False
    active_indices = None

    while time.monotonic() < lp_end:
        it_t0 = time.monotonic()

        c = _full_convolve(f, f)
        M = float(c.max())

        # Identify active constraints (for reduced LP after warmup)
        tol = 0.01 * M
        if num_iters >= 10:
            active = np.where(c >= M - tol)[0]
            if len(active) < 200:
                active = np.argsort(c)[-200:]
            active_indices = active
            use_reduced = True
        else:
            active_indices = None
            use_reduced = False

        # Solve LP
        g = _solve_lp(f, M * 1.00001, n_lp, active_indices if use_reduced else None)
        if g is None:
            print(f"  LP solve failed at iter {num_iters}, falling back to full LP")
            use_reduced = False
            g = _solve_lp(f, M * 1.0001, n_lp, None)
            if g is None:
                print(f"  Full LP also failed, stopping LP phase")
                break

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_norm = g / g_sum

        # Line search for step size
        best_f_today = f
        best_sc_today = _score_norm(f)
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
            f_trial = (1.0 - t) * f + t * g_norm
            sc = _score_norm(f_trial)
            if sc < best_sc_today:
                best_sc_today = sc
                best_f_today = f_trial

        f = best_f_today
        num_iters += 1
        it_time = time.monotonic() - it_t0

        if best_sc_today < best_score_norm:
            best_score_norm = best_sc_today
            best_f_lp = f.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        full_sc = _score(f)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = f.copy()

        if num_iters % 5 == 0:
            print(f"  LP iter {num_iters}: score={full_sc:.10f}, "
                  f"norm={best_score_norm:.10f}, active={len(active_indices) if active_indices is not None else 'all'}, "
                  f"it={it_time:.1f}s, total={time.monotonic()-t_start:.0f}s")

        # Convergence: stop if no improvement for many iterations
        if stagnation_count >= 30 and num_iters > 50:
            print(f"  LP stagnated at iter {num_iters} (no improvement in 30 iters)")
            break
        if num_iters > 500:
            print(f"  LP max iterations reached")
            break

    print(f"  LP done: {num_iters} iters, best_norm={best_score_norm:.10f}, "
          f"best_full={best_score:.10f}")

    # ============ Phase 2: Gradient polish ============
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Gradient Polish (remaining={remaining:.0f}s) ===")
        a_work = best_f_lp.copy()
        a_work = a_work / a_work.sum()

        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())

        # Multi-beta Adam schedule
        betas = [0.005 * max_cw, 0.002 * max_cw, 0.001 * max_cw,
                 0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw]

        polish_budget = min(remaining * 0.4, 300.0)
        polish_end = time.monotonic() + polish_budget

        a_pol, sc_pol = _adam_logspace(a_work, polish_end, betas,
                                       lr_init=0.01, top_k=100, max_iter=300000)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  After Adam polish: score={sc_pol_full:.10f}")

        # Coordinate descent polish
        if time.monotonic() < deadline - 60:
            cd_end = min(deadline - 30, time.monotonic() + 120)
            a_cd, sc_cd = _coord_descend_polish(a_pol, n_passes=2,
                                                top_k_w=50, deadline=cd_end)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  After coord descent: score={sc_cd_full:.10f}")
            a_work = a_cd

    # ============ Phase 3: Diversity restarts ============
    remaining = deadline - time.monotonic()
    if remaining > 80:
        print(f"\n=== Diversity Phase (remaining={remaining:.0f}s) ===")
        n_div = 4096

        # Get best sequence as base
        a_base = best_f_lp.copy()
        a_base = a_base / a_base.sum()

        n_starts = min(4, int(remaining / 100))
        per_start = (remaining - 20) / max(1, n_starts)

        for si in range(max(1, n_starts)):
            if time.monotonic() > deadline - 20:
                break
            print(f"  Diversity start {si}...", end=" ")

            if si == 0:
                p = a_base + 0.02 * rng.standard_normal(n_div)
            elif si == 1:
                shift = rng.integers(1, 32)
                p = np.roll(a_base, shift).copy()
            elif si == 2:
                # Spectrum perturbation
                freq = rng.uniform(1, 5)
                p = a_base + 0.03 * np.cos(2 * np.pi * freq * np.arange(n_div) / n_div)
            else:
                # Random reweighting of blocks
                block_size = n_div // 16
                p = a_base.copy()
                for b in range(16):
                    p[b * block_size:(b + 1) * block_size] *= rng.uniform(0.7, 1.3)

            p = np.maximum(p, 1e-10)
            p = p / p.sum()

            # Quick LP step
            if time.monotonic() < deadline - 40:
                c_p = _full_convolve(p, p)
                M_p = float(c_p.max())
                g_p = _solve_lp(p, M_p * 1.0001, n_div, None)
                if g_p is not None and float(g_p.sum()) > 1e-15:
                    g_p = g_p / float(g_p.sum())
                    # Line search blend
                    best_p = p
                    best_sc_p = _score_norm(p)
                    for t in [1.0, 0.5, 0.3]:
                        p_t = (1 - t) * p + t * g_p
                        sc_t = _score_norm(p_t)
                        if sc_t < best_sc_p:
                            best_sc_p = sc_t
                            best_p = p_t
                    p = best_p

            # Gradient refinement
            c_p2 = _full_convolve(p, p)
            max_cp2 = float(c_p2.max())
            betas_r = [0.003 * max_cp2, 0.001 * max_cp2, 0.0005 * max_cp2]
            start_time = time.monotonic()

            p_best, sc_best = _adam_logspace(p, time.monotonic() + max(10, per_start - 5),
                                             betas_r, lr_init=0.008, top_k=80,
                                             max_iter=50000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
            print(f"score={sc_full:.10f}, improved={sc_full < best_score}")

    # ============ Final ============
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
