"""
Multi-phase optimizer: FISTA (softmax-smoothed max), LP-based Frank-Wolfe,
Adam log-space, peak-shaving coordinate descent, and large-n exploration.
Targets sub-1.5027 score for non-negative step functions.
"""
import numpy as np
import time
import sys

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(a*a)"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on the softmax-smoothed max of convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta  # Lipschitz constant estimate
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient of smoothed max
        y_rev = y[::-1]
        wc = _convolve(w, y_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 2000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lp_frank_wolfe(a, M, deadline, max_rounds=100):
    """
    LP-based Frank-Wolfe: find g >= 0 maximizing sum(g) subject to (a*g)_k <= M.
    Uses cutting plane method.
    Returns g (not rescaled).
    """
    n = len(a)
    # Active set
    c = _convolve(a, a)
    k_star = int(np.argmax(c))
    active_set = [k_star]
    
    g_result = None
    
    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Build constraint matrix for active set
        S = len(active_set)
        A_sub = np.zeros((S, n))
        for idx, k in enumerate(active_set):
            # Row k: A[k, i] = a[k-i] if max(0, k-n+1) <= i <= min(k, n-1)
            i_lo = max(0, k - n + 1)
            i_hi = min(k, n - 1)
            if i_lo <= i_hi:
                A_sub[idx, i_lo:i_hi+1] = a[k - i_hi : k - i_lo + 1][::-1]
        
        # LP: maximize sum(g) subject to A_sub @ g <= M, g >= 0
        c_obj = -np.ones(n)  # minimize -sum(g)
        b_ub = np.full(S, M)
        
        try:
            result = linprog(c_obj, A_ub=A_sub, b_ub=b_ub, 
                           bounds=[(0, None)] * n, method='highs')
        except Exception:
            break
        
        if not result.success:
            break
        
        g = result.x
        
        # Check all constraints
        c_full = _convolve(a, g)
        violations = c_full - M
        max_viol_idx = int(np.argmax(violations))
        max_viol = float(violations[max_viol_idx])
        
        if max_viol < 1e-10:
            # Feasible, this is optimal
            g_result = g
            break
        
        # Add most violated constraint
        if max_viol_idx not in active_set:
            active_set.append(max_viol_idx)
    
    if g_result is None:
        # Use last result even if not fully converged
        try:
            S = len(active_set)
            A_sub = np.zeros((S, n))
            for idx, k in enumerate(active_set):
                i_lo = max(0, k - n + 1)
                i_hi = min(k, n - 1)
                if i_lo <= i_hi:
                    A_sub[idx, i_lo:i_hi+1] = a[k - i_hi : k - i_lo + 1][::-1]
            b_ub = np.full(S, M)
            result = linprog(-np.ones(n), A_ub=A_sub, b_ub=b_ub,
                           bounds=[(0, None)] * n, method='highs')
            if result.success:
                g_result = result.x
        except Exception:
            g_result = a.copy()
    
    return g_result

def _frank_wolfe_iteration(a, t_candidates, deadline):
    """One Frank-Wolfe iteration: solve LP, line search, return new a."""
    n = len(a)
    c = _convolve(a, a)
    M = float(c.max())
    
    # Solve LP
    g = _lp_frank_wolfe(a, M, deadline, max_rounds=50)
    g_sum = float(g.sum())
    if g_sum < 1e-10:
        return a.copy(), M
    
    # Rescale g to have sum = 1
    g_rescaled = g / g_sum
    
    # Line search
    best_a = a.copy()
    best_sc = _score_norm(a)
    for t in t_candidates:
        a_trial = (1.0 - t) * a + t * g_rescaled
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s > 1e-30:
            a_trial = a_trial / s
        sc = _score_norm(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
    
    return best_a, best_sc

def _peak_shaving(a, deadline, max_iters=5000):
    """Iteratively reduce values at positions contributing most to peak."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a, a)
        k_star = int(np.argmax(c))
        
        # Find positions j that contribute most to c[k_star]
        # c[k_star] = sum_j a_j * a_{k_star - j}
        # Contribution of a_j is a_j * a_{k_star - j}
        i_lo = max(0, k_star - n + 1)
        i_hi = min(k_star, n - 1)
        contributions = np.zeros(n)
        if i_lo <= i_hi:
            contributions[i_lo:i_hi+1] = a[i_lo:i_hi+1] * a[k_star - i_hi : k_star - i_lo + 1][::-1]
        
        # Pick the top few contributors and reduce them
        top_k = min(32, n)
        top_indices = np.argsort(contributions)[-top_k:]
        
        improved = False
        for j in top_indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            # Try reducing
            for factor in [0.95, 0.9, 0.85]:
                new_val = old_val * factor
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
            else:
                a[j] = old_val
        
        if not improved and it > 50:
            break
    
    return best_a, best_sc

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
