"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA + Adam in log-space with adaptive top-K smoothing, direct subgradient peak reduction,
coordinate descent polishing, and multi-length exploration (4096, 8192).
Key improvements: adaptive step sizes via Barzilai-Borwein, multi-peak subgradient,
and more aggressive fine-tuning with very small beta values.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_phase(a_init, deadline, beta_schedule, top_k_schedule):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for bi, (beta, top_k) in enumerate(zip(beta_schedule, top_k_schedule)):
        max_iter = 20000
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            c_y = _full_convolve(y, y)
            w = _topk_weights(c_y, beta, top_k)
            grad = _compute_grad_conv(y, w)
            a_next = _proj_simplex(y - step * grad)
            if it % 500 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter_per_beta):

    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    prev_grad_norm = None
    prev_update_norm = None
    prev_grad = None
    prev_update = None

    for beta in beta_schedule:
        if time.monotonic() > deadline:

            it_in_phase += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)

            z = z - update
            z = np.clip(z, -60.0, 60.0)

            if t % 200 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            if it_in_phase % 2000 == 0:
                current_lr = max(0.0002, current_lr * 0.95)
        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

def _subgradient_peak(a_init, deadline, n_peaks=5, momentum=0.9):
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    vel = np.zeros(n)
    step_size = 0.01
    count_since_improve = 0

    for it in range(50000):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        thresh = c_max - 0.05 * c_max
        peak_mask = c >= thresh
        peak_indices = np.where(peak_mask)[0]
        if len(peak_indices) > n_peaks:
            top_thresh = np.partition(c, -n_peaks)[-n_peaks]
            peak_indices = np.where(c >= top_thresh)[0]
        w = np.zeros(2 * n - 1)
        c_peaks = c[peak_indices]
        w_peaks = (c_peaks - (c_max - 0.1 * c_max)) / max(1e-15, 0.1 * c_max)
        w[peak_indices] = w_peaks
        w = w / max(w.sum(), 1e-15)
        grad = _compute_grad_conv(a_curr, w)
        mean_grad = float(np.dot(grad, a_curr))
        grad_centered = grad - mean_grad
        vel = momentum * vel + (1.0 - momentum) * grad_centered
        update = step_size * vel
        a_trial = a_curr + update
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-15:
            a_trial = a_curr.copy()
        else:
            a_trial = a_trial / s_trial
        sc = _score_norm(a_trial)
        if sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            a_curr = a_trial.copy()
            count_since_improve = 0
        else:
            count_since_improve += 1
            if count_since_improve > 500:
                step_size *= 0.9
                count_since_improve = 0
                if step_size < 1e-6:
                    break
    return best_a, best_score


def _coord_descend(a_init, deadline, n_passes=5, top_k=30):
    n = len(a_init)
    a = a_init.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(999)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old
            idx_top = np.argsort(c)[-top_k:]
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5, a_i_old * 0.3, a_i_old * 0.7, a_i_old * 1.2]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    for mult in [0.7, 0.8, 1.0, 1.2, 1.3]:
                        tc = t_star * mult
                        if 0 <= tc <= 5.0 * max(a_i_old, 0.05):
                            t_candidates.append(tc)
            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]
            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "chirp":
        t = np.arange(n) / n
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * (t ** 2) * 3.0)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.8
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(7)
        a = rng.standard_normal(n)
        kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
        a = np.convolve(a, kernel, mode='same')
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n

    max_cd = float(c_d.max())
    betas_fine = [0.0003 * max_cd, 0.0001 * max_cd, 0.00005 * max_cd, 0.00002 * max_cd]
    phase_d_end = time.monotonic() + min(120, (deadline - time.monotonic()) * 0.15)
    a_d, sc_d = _adam_logspace(a_c, phase_d_end, betas_fine, lr_init=0.003, top_k=50, max_iter_per_beta=30000)
    a_d = a_d / a_d.sum()
    sc_d = _score(a_d)
    print(f"    Fine Adam: {sc_d:.10f} ({time.monotonic():.1f}s)")

    # Phase E: Coordinate descent
    a_e = a_d.copy()
    phase_e_end = time.monotonic() + min(80, (deadline - time.monotonic()) * 0.1)
    a_e, sc_e = _coord_descend(a_e, phase_e_end, n_passes=3, top_k=40)
    sc_e = _score(a_e)
    print(f"    CoordDesc: {sc_e:.10f} ({time.monotonic():.1f}s)")

    # Phase F: Final subgradient polish
    a_f = a_e.copy()
    if time.monotonic() < deadline - 30:
        phase_f_end = min(deadline - 10, time.monotonic() + 60)
        a_f, sc_f = _subgradient_peak(a_f, phase_f_end, n_peaks=3, momentum=0.95)
        sc_f = _score(a_f)
        print(f"    Final Subgrad: {sc_f:.10f} ({time.monotonic():.1f}s)")
    else:
        sc_f = sc_e
        a_f = a_e

    candidates = [(a_a, sc_a), (a_b, sc_b), (a_c, sc_c), (a_d, sc_d), (a_e, sc_e), (a_f, sc_f)]
    candidates = [(a, s) for a, s in candidates if np.isfinite(s)]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        parent = _structured_init(4096, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Init: n=4096, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try n=4096 first (from parent), then n=8192
    n_parent = len(parent) if parent is not None else 4096

    # === n=4096 optimization ===
    n1 = 4096
    remaining = deadline - time.monotonic()
    print(f"\n=== n={n1} (remaining: {remaining:.0f}s) ===")
    if remaining > 200:
        a1 = parent.copy() if n_parent == n1 else _interp_profile(parent, n1)
        a1 = a1 / a1.sum()
        sc1 = _score(a1)
        print(f"  Initial: {sc1:.10f}")
        if sc1 < best_score:
            best_score = sc1
            best_seq = a1.copy()

        # Run main optimization
        a_opt, sc_opt = _multi_init_optimize(a1, n1, deadline - 60)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  BEST n={n1}: {sc_opt:.10f}")

    # === n=8192 optimization ===
    n2 = 8192
    remaining = deadline - time.monotonic()
    print(f"\n=== n={n2} (remaining: {remaining:.0f}s) ===")
    if remaining > 200:
        a2 = _interp_profile(parent, n2)
        a2 = a2 / a2.sum()
        sc2 = _score(a2)
        print(f"  Initial: {sc2:.10f}")
        if sc2 < best_score:
            best_score = sc2
            best_seq = a2.copy()

        a_opt2, sc_opt2 = _multi_init_optimize(a2, n2, deadline - 60)
        if sc_opt2 < best_score:
            best_score = sc_opt2
            best_seq = a_opt2.copy()
            print(f"  BEST n={n2}: {sc_opt2:.10f}")

    # === Diversity restarts ===
    remaining = deadline - time.monotonic()
    if remaining > 100 and best_seq is not None:
        print(f"\n=== Diversity restarts (remaining: {remaining:.0f}s) ===")
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_best = len(a_base)

        restart_strategies = [
            "noise", "shift", "freq", "noise2"
        ]
        per_restart = (remaining - 30) / len(restart_strategies)

        for ri, strat in enumerate(restart_strategies):
            if time.monotonic() > deadline - 30:
                break
            if strat == "noise":
                p = a_base + 0.01 * rng.standard_normal(n_best)
            elif strat == "shift":
                shift = rng.integers(1, max(2, n_best // 512))
                p = np.roll(a_base, shift).copy()
            elif strat == "freq":
                freq = rng.uniform(1.0, 5.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p = a_base + pert
            else:
                p = a_base + 0.005 * rng.standard_normal(n_best)

            p = np.maximum(p, 1e-12)
            p = p / p.sum()

            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            betas_r = [0.003 * max_cp, 0.001 * max_cp, 0.0005 * max_cp, 0.0002 * max_cp]
            per_end = time.monotonic() + max(10, per_restart - 5)
            p_best, sc_r = _adam_logspace(p, per_end, betas_r, lr_init=0.006, top_k=80, max_iter_per_beta=20000)
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {strat}: {sc_full:.10f}")

    # === Final polish on best ===
    remaining = deadline - time.monotonic()
    if remaining > 50 and best_seq is not None:
        print(f"\n=== Final polish (remaining: {remaining:.0f}s) ===")
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        c_fin = _full_convolve(a_fin, a_fin)
        max_cfin = float(c_fin.max())
        betas_fin = [0.0002 * max_cfin, 0.0001 * max_cfin, 0.00005 * max_cfin]
        fin_end = min(deadline - 10, time.monotonic() + remaining * 0.6)
        a_fin_opt, sc_fin = _adam_logspace(a_fin, fin_end, betas_fin, lr_init=0.002, top_k=30, max_iter_per_beta=10000)
        a_fin_opt = a_fin_opt / a_fin_opt.sum()
        sc_fin_full = _score(a_fin_opt)
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin_opt.copy()

        # Final coord descent
        if time.monotonic() < deadline - 20:
            cd_end = min(deadline - 5, time.monotonic() + remaining * 0.3)
            a_cd, sc_cd = _coord_descend(best_seq, cd_end, n_passes=2, top_k=30)
            sc_cd = _score(a_cd)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"RETURNING: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
