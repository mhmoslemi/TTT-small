"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines FISTA, Adam in log-space, top-k subgradient with averaging,
and pair coordinate descent. Uses backtracking line search and
frequency-domain initialization for diversity.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    """Compute gradient of sum_k w_k * c_k w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "frequency":
        if rng is None:
            rng = np.random.default_rng(99)
        N = _next_pow2(2 * n)
        # Create a flat-spectrum signal
        freq = np.fft.rfftfreq(N)
        magnitude = np.ones(N // 2 + 1)
        # Suppress DC to avoid concentration
        magnitude[0] = 0.5
        phase = rng.random(N // 2 + 1) * 2 * np.pi
        spectrum = magnitude * np.exp(1j * phase)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:

        return a
    else:

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines FISTA with adaptive step, Adam in log-space with tuned hyperparameters,
top-k subgradient with iterate averaging, frequency-domain initialization,
and targeted pair coordinate descent. Key differences from previous: backtracking
step size, exact subgradient phase, pair CD targeting hot convolution positions,
and more aggressive beta/LR schedules.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "frequency":
        if rng is None:
            rng = np.random.default_rng(777)
        N = _next_pow2(2 * n)
        magnitude = np.ones(N // 2 + 1)
        magnitude[0] = 0.1
        phase = rng.random(N // 2 + 1) * 2.0 * np.pi
        spectrum = magnitude * np.exp(1j * phase)
        a_full = np.fft.irfft(spectrum, N)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif kind == "multi_sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3.0 * t) + 0.1 * np.cos(2.0 * np.pi * 7.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()

def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / (beta * 0.5)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    no_improve = 0
    plateau_threshold = 500

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
                no_improve = 0
            else:
                no_improve += 300
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / (beta * 0.5)
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

        if no_improve > plateau_threshold and it % 1000 == 0:
            step *= 0.8
            no_improve = 0

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, clip_bound=40.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.8:
                update = update * (0.8 / update_norm)

            z = z - update
            z = np.clip(z, -clip_bound, clip_bound)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 100

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

            if no_improve > 10000:
                z = z + 0.01 * np.random.default_rng(t).standard_normal(n)
                z = z - z.mean()
                no_improve = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_average(a_init, deadline, n_steps=5000, step_scale=0.1):
    """Subgradient method with iterate averaging using top-k subgradients."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_avg = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())

        # Top-k subgradient
        k_top = min(5, len(c))
        top_indices = np.argsort(c)[::-1][:k_top]
        w = np.zeros_like(c)
        w[top_indices] = 1.0 / k_top

        grad = _compute_grad(a_curr, w)
        grad = grad / (n * cm + 1e-15)

        step_size = step_scale / np.sqrt(t)
        a_trial = _proj_simplex(a_curr - step_size * grad)
        a_curr = a_trial
        a_avg = (a_avg * (t - 1) + a_curr) / t

        if t % 500 == 0:
            for candidate in [a_curr, a_avg]:
                sc = _score_norm(candidate)
                if sc < best_score:
                    best_score = sc
                    best_a = candidate.copy()

    return best_a, best_score


def _pair_coord_descent(a, deadline, n_iterations=50000):
    """Targeted pair coordinate descent: move mass from hot to cool positions."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        cm = float(c.max())

        # Find the argmax position in the convolution
        max_pos = int(np.argmax(c))

        # Find contributing indices for this position
        # c[max_pos] = sum_i a[i] * a[max_pos - i] for valid i
        lo = max(0, max_pos - n + 1)
        hi = min(max_pos, n - 1)
        contributors = list(range(lo, hi + 1))

        if len(contributors) < 2:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            # Pick two contributors and try to transfer mass
            idx = rng.integers(0, len(contributors))
            i = contributors[idx]
            # Pick j from contributors or randomly
            if rng.random() < 0.5:
                j = contributors[(idx + 1) % len(contributors)]
            else:
                j = rng.integers(0, n)

        if i == j:
            continue

        # Try transferring small mass from i to j (if a[i] > a[j])
        # or from j to i (if a[j] > a[i])
        deltas = [0.01, 0.005, 0.002, 0.02]
        for delta in deltas:
            if best_a[i] < best_a[j]:
                a_try = best_a.copy()
                a_try[i] += delta
                a_try[j] -= delta
            else:
                a_try = best_a.copy()
                a_try[j] += delta
                a_try[i] -= delta

            if a_try[i] < 0 or a_try[j] < 0:
                continue
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            sc_try = _score_norm(a_try)
            if sc_try < best_score - 1e-12:
                best_score = sc_try
                best_a = a_try.copy()
                break

        # Also try single coordinate scaling
        if it % 3 == 0:
            i = rng.integers(0, n)
            for factor in [0.85, 0.92, 1.08, 1.15, 0.75, 1.25]:
                a_try = best_a.copy()
                a_try[i] *= factor
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score_norm(a_try)
                if sc_try < best_score - 1e-12:
                    best_score = sc_try
                    best_a = a_try.copy()
                    break

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            break
        phase_deadline = min(deadline - 120, time.monotonic() + 40.0)
        a_work, sc = _fista_smoothed(a_work, beta, phase_deadline, 50000)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={beta:.6f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam fine optimization (main) ===
    print(f"\n=== Phase 2: Adam fine optimization ===")
    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    # Aggressive beta schedule
    adam_betas = []
    beta = 0.003 * max_cc
    while beta > 1e-6 * max_cc:
        adam_betas.append(beta)
        beta *= 0.55
    if len(adam_betas) > 12:
        adam_betas = adam_betas[:12]

    adam_deadline = min(deadline - 180, time.monotonic() + 400.0)
    a_opt, sc_opt = _adam_logspace(
        a_work, adam_deadline,
        adam_betas, lr_init=0.012, max_iter_total=200000, clip_bound=35.0
    )
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        a_work = a_opt.copy()
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Adam: score={sc_opt_full:.10f} (not better) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Subgradient with averaging ===
    print(f"\n=== Phase 3: Subgradient averaging ===")
    if time.monotonic() < deadline - 120:
        sg_deadline = min(deadline - 120, time.monotonic() + 80.0)
        a_sg, sc_sg = _subgradient_average(a_work, sg_deadline, n_steps=8000, step_scale=0.15)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            a_work = a_sg.copy()
            print(f"  Subgradient: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Subgradient: score={sc_sg_full:.10f} (not better) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multiple Adam restarts with different params ===
    print(f"\n=== Phase 4: Adam restarts ===")
    n_restarts = 5
    per_restart = max(20.0, (deadline - time.monotonic() - 80.0) / n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > deadline - 60:
            break

        if i == 0:
            p_start = _perturb(a_work, 0.015, rng)
        elif i == 1:
            p_start = _perturb(a_work, 0.04, rng)
        elif i == 2:
            sz = max(1, n_main // 2048)
            shift = rng.integers(1, max(2, sz))
            p_start = np.roll(a_work, shift).copy()
            p_start = p_start / p_start.sum()
        elif i == 3:
            alt = _make_profile(n_main, "tapered")
            p_start = 0.8 * a_work + 0.2 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        else:
            alt = _make_profile(n_main, "frequency", rng)
            p_start = 0.6 * a_work + 0.4 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

        c_s = _full_convolve(p_start, p_start)
        max_cs = float(c_s.max())

        # Vary learning rates
        lrs = [0.008, 0.005, 0.015, 0.003, 0.01]
        lr_i = lrs[i % len(lrs)]

        bs = []
        beta = 0.001 * max_cs
        while beta > 2e-6 * max_cs:
            bs.append(beta)
            beta *= 0.5
        if len(bs) > 8:
            bs = bs[:8]

        p_best, sc_best = _adam_logspace(
            p_start, time.monotonic() + per_restart,
            bs, lr_init=lr_i, max_iter_total=100000, clip_bound=35.0
        )
        sc_full = _score(p_best)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_best.copy()
            a_work = p_best.copy()
            print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Restart {i}: score={sc_full:.10f} (not better) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Ultra-fine Adam polish ===
    print(f"\n=== Phase 5: Ultra-fine polish ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        betas_ult = []
        beta = 3e-5 * max_cp
        while beta > 1e-8 * max_cp:
            betas_ult.append(beta)
            beta *= 0.4
        if len(betas_ult) < 3:
            betas_ult = [3e-5 * max_cp, 1e-5 * max_cp, 4e-6 * max_cp]
        if len(betas_ult) > 6:
            betas_ult = betas_ult[:6]

        per_t = min(60.0, deadline - time.monotonic() - 20)
        a_pol, sc_pol = _adam_logspace(
            a_pol, time.monotonic() + per_t,
            betas_ult, lr_init=0.002, max_iter_total=80000, clip_bound=30.0
        )
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Pair coordinate descent ===
    print(f"\n=== Phase 6: Pair coordinate descent ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_cd, sc_cd = _pair_coord_descent(best_seq, time.monotonic() + min(30.0, deadline - time.monotonic() - 5), n_iterations=30000)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Pair CD: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
