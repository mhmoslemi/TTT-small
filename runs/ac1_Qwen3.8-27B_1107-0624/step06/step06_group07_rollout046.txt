def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _taylor_window(n, order=5, sigma_db=30.0):
    """Generate a Taylor window."""
    N = order
    t = np.linspace(0, 1, n)
    a = np.ones(n)

            [(2*m - 2*j + 1) * (m - j)] for j in range(1, m))

        return a / a.sum()
    elif kind == "tapered_steep":
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_topped":
        a = 1.0 - 0.1 * (2 * t - 1) ** 6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        a = 1.0 + 0.15 * np.cos(4 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        a = 1.0 - 0.15 * (2 * t - 1) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "taylor":
        return _taylor_window(n)
    else:
        return np.ones(n) / n


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0, fine=False):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    if not fine:
        betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.0125, 0.006,
                                  0.003, 0.0015, 0.00075, 0.0004])
    else:
        betas = max_c * np.array([0.0002, 0.0001, 5e-5, 2.5e-5, 1.25e-5,
                                  6.25e-6, 3e-6, 1.5e-6, 7.5e-7])

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_iter_time = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(6000, max(200, int(remaining / (est_iter_time * 12))))
        max_iter = min(max_iter, 8000)

        a_new, success = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_polyak(a_init, deadline, tol_frac=1e-4, max_steps=20000,
                    f_star_est=None):
    """Subgradient method with Polyak step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    f_star = f_star_est if f_star_est is not None else best_sc * 0.999
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        sc = float(2.0 * n * c.max() / (a.sum() ** 2))

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            f_star = min(f_star, sc * 0.9995)
            stall = 0
        else:
            stall += 1

        if stall > 100:
            break

        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Polyak step size
        gap = sc - f_star
        if gap < 1e-12:
            step = 1e-4
        else:
            step = gap / (grad_norm ** 2 + 1e-10)
            step = min(step, 0.1)

        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_tight(a_init, deadline, tol_frac=1e-5, max_steps=15000, lr_init=0.03):
    """Tight subgradient with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.001)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.05
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.6
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _pairwise_cd(a_init, deadline, max_rounds=5000):
    """Pairwise coordinate descent with exact 1D optimization."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    improved_count = 0
    total_rounds_no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n

        # Try to redistribute between i and j
        s_pair = a[i] + a[j]
        if s_pair < 1e-15:
            continue

        # Current fraction
        frac_cur = a[i] / s_pair

        # Try several fractions
        best_local_sc = best_sc
        best_frac = frac_cur

        for frac_try in [frac_cur + 0.05, frac_cur - 0.05,
                         frac_cur + 0.02, frac_cur - 0.02,
                         frac_cur + 0.01, frac_cur - 0.01,
                         frac_cur + 0.005, frac_cur - 0.005]:
            if frac_try < 0.001 or frac_try > 0.999:
                continue
            a_try = a.copy()
            a_try[i] = s_pair * frac_try
            a_try[j] = s_pair * (1.0 - frac_try)
            sc_try = _score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_frac = frac_try

        if best_frac != frac_cur:
            a[i] = s_pair * best_frac
            a[j] = s_pair * (1.0 - best_frac)
            if best_local_sc < best_sc - 1e-14:
                best_sc = best_local_sc
                best_a = a.copy()
            improved_count += 1
            total_rounds_no_improve = 0
        else:
            total_rounds_no_improve += 1

        if total_rounds_no_improve > 500 and rnd > 500:
            # Try larger perturbations
            for _ in range(10):
                i2 = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i2 == j2:
                    continue
                s2 = a[i2] + a[j2]
                if s2 < 1e-15:
                    continue
                frac_try = rng.uniform(0.05, 0.95)
                a_try = a.copy()
                a_try[i2] = s2 * frac_try
                a_try[j2] = s2 * (1.0 - frac_try)
                sc_try = _score(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    total_rounds_no_improve = 0
                    break

            if total_rounds_no_improve > 2000:
                break

    return best_a, best_sc


def _triple_cd(a_init, deadline, max_rounds=2000):
    """Triple coordinate descent."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(54321)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        idx = np.random.default_rng(rnd).permutation(n)[:3]
        s_trip = a[idx[0]] + a[idx[1]] + a[idx[2]]
        if s_trip < 1e-15:
            continue

        # Try random redistribution
        for _ in range(5):
            weights = rng.dirichlet([1, 1, 1])
            a_try = a.copy()
            for k, ix in enumerate(idx):
                a_try[ix] = s_trip * weights[k]
            if np.any(a_try < 0):
                continue
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                break

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.003 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp smoothing,
    Polyak subgradient, tight subgradient, pairwise coordinate descent,
    diverse spectral initializations, and multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev-best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n_init = 4096
        a0 = _make_profile(n_init, "hamming")
        print(f"No parent. Hamming n={n_init}, score={_score(a0):.10f}")
        try_update(a0, "init_hamming")

    # === Phase 1: Diverse initializations ===
    print(f"\n=== Phase 1: Diverse initializations (t={time.monotonic()-t_start:.1f}s) ===")
    n_main = len(best_seq) if best_seq is not None else 4096

    init_profiles = [
        ("hamming", "hamming"),
        ("hann", "hann"),
        ("blackman", "blackman"),
        ("tapered", "tapered"),
        ("tapered_steep", "tapered_steep"),
        ("flat_topped", "flat_topped"),
        ("quartic", "quartic"),
        ("bimodal", "bimodal"),
        ("taylor", "taylor"),
    ]

    candidates = [best_seq.copy()] if best_seq is not None else []
    for name, kind in init_profiles:
        if time.monotonic() > deadline - 300:
            break
        a_p = _make_profile(n_main, kind)
        sc = _score(a_p)
        print(f"  Profile [{name}]: score={sc:.10f}")
        candidates.append(a_p)

    # Also try perturbations of parent
    if parent is not None:
        for sigma in [0.05, 0.1, 0.2]:
            a_p = _perturb(parent, sigma, rng)
            candidates.append(a_p)
        for modes in [3, 5, 8]:
            a_p = _perturb_fourier(parent, rng, modes)
            candidates.append(a_p)

    # Score all and keep top candidates
    scored = []
    for c in candidates:
        sc = _score(c)
        scored.append((sc, c))
    scored.sort(key=lambda x: x[0])

    print(f"  Top 3 initializations: {[f'{s:.10f}' for s, _ in scored[:3]]}")

    # === Phase 2: Coarse L-BFGS on top candidates ===
    print(f"\n=== Phase 2: Coarse L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    n_cands = min(5, len(scored))
    for idx in range(n_cands):
        if time.monotonic() > deadline - 500:
            break
        sc_init, a_init = scored[idx]
        print(f"  Coarse L-BFGS [{idx}]: init_score={sc_init:.10f}")
        a_opt = _lbfgs_schedule(a_init, deadline, min_time_remaining=400.0, fine=False)
        try_update(a_opt, f"coarse_{idx}")

    # === Phase 3: Fine L-BFGS on best ===
    print(f"\n=== Phase 3: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        a_fine = _lbfgs_schedule(best_seq, deadline, min_time_remaining=200.0, fine=True)
        try_update(a_fine, "fine_lbfgs")

    # === Phase 4: Polyak subgradient ===
    print(f"\n=== Phase 4: Polyak subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        f_star_est = best_score_val * 0.999
        a_pg, sc_pg = _subgrad_polyak(best_seq, deadline - 200,
                                      tol_frac=1e-4, max_steps=20000,
                                      f_star_est=f_star_est)
        try_update(a_pg, "polyak_sg")

    # === Phase 5: Tight subgradient ===
    print(f"\n=== Phase 5: Tight subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_ts, sc_ts = _subgrad_tight(best_seq, deadline - 150,
                                     tol_frac=1e-5, max_steps=15000, lr_init=0.02)
        try_update(a_ts, "tight_sg")

    # === Phase 6: Very fine L-BFGS ===
    print(f"\n=== Phase 6: Very fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        very_fine_betas = mb * np.array([5e-6, 2e-6, 1e-6, 5e-7, 2e-7])
        for i, beta in enumerate(very_fine_betas):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - 80 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(4000, max(100, int(remaining / 0.003)))
            a_vf, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_vf, f"vf_{i}")

    # === Phase 7: Pairwise coordinate descent ===
    print(f"\n=== Phase 7: Pairwise CD (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_cd, sc_cd = _pairwise_cd(best_seq, deadline - 80, max_rounds=8000)
        try_update(a_cd, "pairwise_cd")

    # === Phase 8: Triple CD ===
    print(f"\n=== Phase 8: Triple CD (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_tc, sc_tc = _triple_cd(best_seq, deadline - 50, max_rounds=3000)
        try_update(a_tc, "triple_cd")

    # === Phase 9: Multi-restart from best ===
    print(f"\n=== Phase 9: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restart_configs = [
            ("r_gauss_003", 0.003, "gauss"),
            ("r_gauss_010", 0.010, "gauss"),
            ("r_gauss_030", 0.030, "gauss"),
            ("r_fourier_2", 0, "fourier2"),
            ("r_fourier_6", 0, "fourier6"),
        ]
        for name, sigma, ptype in restart_configs:
            if time.monotonic() > deadline - 35:
                break
            if ptype == "gauss":
                p_start = _perturb(best_seq, sigma, rng)
            elif ptype == "fourier2":
                p_start = _perturb_fourier(best_seq, rng, 2)
            else:
                p_start = _perturb_fourier(best_seq, rng, 6)

            sc_s = _score(p_start)
            print(f"  Restart [{name}]: start={sc_s:.10f}")
            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=30.0, fine=True)
            try_update(a_r, f"restart_{name}")
            if time.monotonic() < deadline - 30:
                a_r2, _ = _subgrad_tight(a_r, deadline - 30,
                                         tol_frac=1e-5, max_steps=3000, lr_init=0.01)
                try_update(a_r2, f"restart_{name}_sg")

    # === Phase 10: Try different n ===
    print(f"\n=== Phase 10: Different n (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 35:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=20.0, fine=False)
            try_update(a_try, f"n_{target_n}")

    # === Phase 11: Final aggressive polish ===
    print(f"\n=== Phase 11: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            max_iter = min(3000, max(100, int((deadline - 5 - time.monotonic()) / 0.003)))
            a_f, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_tight(best_seq, deadline - 5,
                                        tol_frac=1e-6, max_steps=5000, lr_init=0.008)
                try_update(a_s, f"final_sg_{rnd}")

            # Pairwise CD
            if time.monotonic() < deadline - 5:
                a_p, _ = _pairwise_cd(best_seq, deadline - 5, max_rounds=3000)
                try_update(a_p, f"final_cd_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "hamming")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
