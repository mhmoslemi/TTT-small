import scipy.sparse as sp
import scipy.optimize as opt
import time

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def build_conv_matrix(f):
    """Build the matrix A such that (A @ g) = f * g (convolution)"""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def lp_step(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0"""
    n = len(f)
    A = build_conv_matrix(f)
    c_obj = -np.ones(n)  # minimize -sum(g)
    res = opt.linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')

"""
LP-guided optimizer with Adam fine-tuning for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal update directions (exploiting asymmetry),
combined with log-space Adam for final polishing. Key insight: optimal sequences
are asymmetric, and the LP finds directions that better flatten auto-convolution
peaks than subgradient methods.
"""
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _build_conv_indices(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n, k + 1)
        for i in range(i_start, i_end):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def _lp_direction(f, M, row_idx, col_idx):
    n = len(f)
    vals = np.array([f[i] for i in col_idx], dtype=np.float64)
    A = sp.csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = opt.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_simplex(a, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total,
                   lr_decay_every=8000, lr_decay_factor=0.96, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            ez = np.exp(z - z.max())
            a = ez / ez.sum()
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros(2 * n - 1)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > grad_clip:
                update = update * (grad_clip / un)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                ez_chk = np.exp(z - z.max())
                a_chk = ez_chk / ez_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)
            if no_improve > 80 and t > 500:
                current_lr = max(0.0003, current_lr * 0.92)
                no_improve = 0

    ez_f = np.exp(z - z.max())
    a_final = ez_f / ez_f.sum()
    sc_f = _score_norm(a_final)
    if np.isfinite(sc_f) and sc_f < best_score:
        best_score = sc_f
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=500, deadline=None):
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        if avg_mag < 1e-30:
            break
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_f = _score_norm(a)
    if sc_f < best_score:
        best_score = sc_f
        best_a = a.copy()
    return best_a, best_score


def _subgradient_polish(a_init, n_steps=3000, deadline=None):
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        top_k = min(30, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        w_vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(2 * n - 1)
        w_vec[top_idx] = w_vals ** 2
        w_vec = w_vec / w_vec.sum()

        grad = _compute_grad_simplex(a, w_vec, n)
        step_size = 0.3 / (1.0 + 0.005 * it)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            a_new2 = _proj_simplex(a - step_size * 0.5 * grad)
            if _score_norm(a_new2) < _score_norm(a):
                a = a_new2

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n = 4096
    if parent is not None:
        a = parent.copy()
        a = np.clip(a, 0, 1000)
        a = a / a.sum()
        best_seq = a.copy()
        best_score = _score(a)
        print(f"Parent loaded: n={n}, score={best_score:.10f}")
    else:
        a = np.ones(n) / n
        best_seq = a.copy()
        best_score = _score(a)
        print(f"Uniform init: n={n}, score={best_score:.10f}")

    # Precompute LP matrix indices
    print("Building LP indices...")
    row_idx, col_idx = _build_conv_indices(n)
    print(f"LP matrix: {len(row_idx)} nonzeros")

    # === Phase 1: LP-based optimization ===
    print("Phase 1: LP-based optimization")
    lp_end = min(deadline, t_start + 700.0)
    t_step = 0.15
    no_improve = 0
    lp_iters = 0

    while time.monotonic() < lp_end - 5:
        c = _full_convolve(a, a)
        M = float(c.max())

        g = _lp_direction(a, M, row_idx, col_idx)
        if g is None:
            print("  LP failed, stopping.")
            break

        g = np.maximum(g, 0)
        s_g = float(g.sum())
        if s_g < 1e-15:
            break
        g = g / s_g

        # Line search
        accepted = False
        candidates = [t_step, t_step * 0.7, t_step * 0.5, t_step * 0.3,
                      t_step * 0.1, t_step * 0.05, t_step * 0.02]
        best_t = 0.0
        best_obj = M
        for tv in candidates:
            a_test = (1.0 - tv) * a + tv * g
            c_test = _full_convolve(a_test, a_test)
            obj = float(c_test.max())
            if obj < best_obj:
                best_obj = obj
                best_t = tv

        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            accepted = True
            t_step = min(0.4, best_t * 1.3)
        else:
            t_step *= 0.5
            if t_step < 0.002:
                break

        lp_iters += 1
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            no_improve = 0
            if lp_iters <= 5 or best_score < 1.504 or lp_iters % 20 == 0:
                print(f"  LP iter {lp_iters}: score={sc:.10f}, t={best_t:.4f}, time={time.monotonic()-t_start:.1f}s")
        else:
            no_improve += 1

        if no_improve > 80:
            print(f"  LP stalled at iter {lp_iters}, score={best_score:.10f}")
            break

    print(f"End Phase 1: score={best_score:.10f}, {lp_iters} iters, time={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam fine-tuning on best ===
    print("Phase 2: Adam fine-tuning")
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    phase2_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase2_end - 20:
        c_w = _full_convolve(a_work, a_work)
        max_cw = float(c_w.max())
        betas = []
        b = 0.001 * max_cw
        while b > 0.000005 * max_cw:
            betas.append(b)
            b *= 0.5
        if len(betas) > 15:
            betas = betas[:15]

        per_t = phase2_end - time.monotonic() - 10
        a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + per_t, betas,
                                       lr_init=0.01, max_iter_total=300000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f}, time={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 3: Subgradient polish ===
    print("Phase 3: Subgradient polish")
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end - 10:
        a_sg, _ = _subgradient_polish(a_work, n_steps=4000, deadline=phase3_end)
        sc_sg = _score(a_sg)

            print(f"  Subgradient: score={sc_sg:.10f}, time={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 4: Try n=8192 ===
    print("Phase 4: n=8192 exploration")
    phase4_end = min(deadline, t_start + 1550.0)

        print(f"  Interpolated n=8192: score={sc_8192:.10f}")

        if sc_8192 < best_score + 0.01:  # Only worth trying if competitive
            c_8 = _full_convolve(a_8192, a_8192)

                                        lr_init=0.008, max_iter_total=200000)

            if sc_8_opt < best_score:

    if best_seq is not None and time.monotonic() < deadline - 20:
