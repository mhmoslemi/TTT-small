import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:

Let me also think about the "flat spectrum" approach as an initialization. The idea is:
- In the frequency domain, the autocorrelation is the IFFT of the power spectrum |A|^2
- We want |A|^2 to be as flat as possible (to make the autocorrelation flat)
- But a >= 0 constrains what spectra are achievable

One approach:
1. Start with a random phase φ[k]
2. A[k] = exp(i*φ[k]) (flat magnitude)
3. a = IFFT(A)
4. Project to non-negative: a = max(a, 0)
5. Renormalize
6. Repeat with updated phase

This is similar to the Gerchberg-Saxton algorithm. Let me include this as an initialization option.

Actually, for the non-circular case (linear convolution), the relationship between the spectrum and the autocorrelation is different. The linear autocorrelation corresponds to the circular autocorrelation of a zero-padded version. So we'd need to work with a zero-padded sequence.

"""
Improved multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-k active set subgradient with backtracking, adaptive L-BFGS with
log-sum-exp smoothing, structured coordinate descent, and guided restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgradient_topK(a, n, k):
    """Compute subgradient using top-k active set."""
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a), c_max
    # Get top-k indices
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 2.0 * n * (h - W)
    return grad, c_max

def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

def _lbfgs_schedule(a_start, deadline, min_time_remaining=300.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    # More aggressive beta schedule
    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002, 0.00001,
                              0.000005, 0.000002, 0.000001])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break
        
        # Estimate time per iteration
        est_iter_time = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(20000, max(500, int(remaining / (est_iter_time * 10))))
        
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            print(f"    L-BFGS beta={beta:.2e}: score={sc:.10f} (iter {i})")
    
    return current_a

def _subgrad_backtrack(a_init, deadline, k_values=(1, 5, 10, 50, 100), max_steps=50000):
    """Subgradient with backtracking line search and adaptive k."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.05
    stall = 0
    k_idx = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        # Cycle through k values
        k = k_values[k_idx % len(k_values)]
        grad, c_max = _subgradient_topK(a, n, k)
        
        if np.linalg.norm(grad) < 1e-15:
            break
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            z_try = z - step * grad
            z_try = np.clip(z_try, -80.0, 80.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            
            if sc_try < best_sc - 1e-15:
                z = z_try
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True
                lr = min(lr * 1.05, 0.3)
                stall = 0
                break
            step *= 0.5
        
        if not improved:
            stall += 1
            lr *= 0.8
            if lr < 1e-8:
                break
            if stall > 50:
                # Try a different k
                k_idx += 1
                stall = 0
                lr = 0.02
        else:
            k_idx = (k_idx + 1) % len(k_values)  # Cycle k values
        
    return best_a, best_sc

def _coord_descent(a_init, deadline, n_pairs=None):
    """Pair-wise coordinate descent."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)
    
    if n_pairs is None:
        n_pairs = n * 4
    
    step_frac = 0.3
    
    for i in range(n_pairs * 20):
        if time.monotonic() > deadline:
            break
        
        # Pick a random pair
        idx = rng.integers(0, n)
        jdx = rng.integers(0, n)
        while jdx == idx:
            jdx = rng.integers(0, n)
        
        s = a[idx] + a[jdx]
        if s < 1e-15:
            continue
        
        # Try several splits
        for frac in [step_frac, 1.0 - step_frac, 0.5, 0.3, 0.7]:
            new_i = s * frac
            new_j = s * (1.0 - frac)
            if new_i < 0 or new_j < 0:
                continue
            
            a_try = a.copy()
            a_try[idx] = new_i
            a_try[jdx] = new_j
            sc_try = _score(a_try)
            
            if sc_try < best_sc - 1e-14:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
        
        # Periodically recompute best
        if i % 1000 == 0:
            sc_check = _score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()
    
    return best_a, best_sc

def _perturb_structured(a, c, rng, scale=0.05):
    """Perturb based on the autocorrelation structure."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    
    # Find the peak lags and perturb in a way that reduces them
    c_max = c.max()
    peak_idx = np.argsort(c)[::-1][:20]
    
    # Add structured perturbation
    t = np.arange(n)
    for freq in [2, 3, 5, 7, 11]:
        amp = scale * rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * freq * t / n + phase)
    
    # Also add noise concentrated near the peaks
    for pi in peak_idx[:10]:
        # The lag pi corresponds to elements that contribute to c[pi]
        # Perturb those elements
        left = max(0, pi - n + 1)
        right = min(n - 1, pi)
        if right >= left:
            pert = rng.uniform(-scale, scale, size=right - left + 1)
            z[left:right+1] += pert * 0.5
    
    z = z - z.mean()
    return _softmax(z)

def _interpolate_to_n(a, target_n):

    a_interp = np.maximum(a_interp, 1e-10)
    return a_interp / a_interp.sum()
