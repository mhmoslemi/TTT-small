"""
LP-Convolution Optimizer: Minimizes 2*n*max(conv(a,a))/sum(a)^2 using
exact LP-based direction finding for self-convolution peak reduction,
combined with multi-scale interpolation and gradient-based polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _build_A(f, n):
    """Build sparse (2n-1) x n matrix A where (A*b)[k] = convolve(f,b)[k]."""
    total = n * n
    i = np.arange(total, dtype=np.int32)
    row_idx = (i // n + i % n).astype(np.int32)
    col_idx = (i // n).astype(np.int32)
    data = f[i % n].copy()
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
