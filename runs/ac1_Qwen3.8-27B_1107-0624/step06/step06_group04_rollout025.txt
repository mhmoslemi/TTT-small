"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses sinusoidal low-dim pre-optimization, FISTA with top-K subgradient,
log-space Adam, frequency-domain shaping, and multi-restart diversification.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Assume sum(a) = 1, return 2n*max(c)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _topk_subgrad(a, k=10):
    """Compute subgradient using top-k peaks of autocorrelation."""
    n = len(a)
    c = _full_convolve(a, a)
    # Get top k values
    top_idx = np.argsort(c)[::-1][:k]
    # Weight by proximity to max
    max_c = c[top_idx[0]]
    min_c = c[top_idx[-1]]
    if max_c - min_c < 1e-15:
        weights = np.ones(k) / k
    else:
        weights = (c[top_idx] - min_c + 1e-10)
        weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        # c[ki] = sum_i a[i]*a[ki-i] for valid i
        # d(c[ki])/d(a[j]) = a[ki-j] if 0 <= ki-j < n
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        # grad[j] += wi * a[ki-j] for j in [lo, hi]
        # a[ki-j] for j in [lo,hi] means indices ki-lo down to ki-hi
        # ki-lo = ki - max(0, ki-n+1)
        # ki-hi = ki - min(n-1, ki)
        a_slice = a[ki - hi : ki - lo + 1][::-1]
        grad[lo:hi+1] += wi * a_slice
    return grad, float(c.max())

def _fista_topk(a_init, deadline, max_iter, topk=10, step_scale=1.0):
    """FISTA with top-K subgradient on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Initial step size
    c_init = _full_convolve(a_curr, a_curr)
    L = 4.0 * n * step_scale  # Lipschitz constant estimate
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        grad, _ = _topk_subgrad(y, topk)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

log-space Adam with adaptive beta, spectral upscaling, and multi-restart

import time

    while k < n:

    c = _full_convolve(a, a)

def _proj_simplex(v):

    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)

def _upscale(a, n_new):
    """Upscale sequence to new length via interpolation + light smoothing."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA with softmax-smoothed max on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid k)
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=8000, lr_decay_factor=0.96, grad_clip=1.0):
    """Adam in log-space with softmax parameterization."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            # Gradient in log-space
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 80 and t > 2000:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _sinusoidal_optimize(n, deadline, n_harmonics=25, lr=0.01, max_iter=100000):
    """
    Optimize a sinusoidal parameterization: a[i] = max(0, 1 + sum_k (A_k cos + B_k sin)).
    Returns the best a found.
    """
    rng_local = np.random.default_rng(12345)
    K = n_harmonics
    n_params = 2 * K

    # Initialize parameters randomly (small)
    theta = 0.02 * rng_local.standard_normal(n_params)

    # Precompute basis matrices
    i = np.arange(n)
    basis = np.zeros((n, n_params))
    for k in range(1, K + 1):
        basis[:, 2 * (k - 1)] = np.cos(2.0 * np.pi * k * i / n)
        basis[:, 2 * (k - 1) + 1] = np.sin(2.0 * np.pi * k * i / n)

    best_a = None
    best_score = np.inf

    m_buf = np.zeros(n_params)
    v_buf = np.zeros(n_params)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        # Compute a from parameters
        a = 1.0 + basis @ theta
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

        # Compute gradient of smoothed max objective
        c = _full_convolve(a, a)
        beta = 0.005 * float(c.max())
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * a * (h - W)

        # Chain rule: d(score)/d(theta) = basis.T @ grad_a
        grad_theta = basis.T @ grad_a

        # Handle the max(0, .) nonlinearity (approximate)
        mask = (1.0 + basis @ theta) > 0
        grad_theta_masked = np.zeros(n_params)
        for j in range(n_params):
            grad_theta_masked[j] = np.sum(mask * grad_a * basis[:, j])

        m_buf = b1 * m_buf + (1.0 - b1) * grad_theta_masked
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_theta_masked * grad_theta_masked)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 5.0:
            update = update * (5.0 / update_norm)
        theta = theta - update

        if it % 200 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        if t % 5000 == 0:
            current_lr = max(0.001, current_lr * 0.95)

        if no_improve > 20 and t > 5000:
            current_lr = max(0.0005, current_lr * 0.9)

    if best_a is None:
        best_a = np.ones(n) / n
    return best_a, best_score


def _subgradient_polish(a_init, n_steps, deadline):
    """Subgradient method with exact max and adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    step_scale = 1.0

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        f_curr = 2.0 * n * cm

        # Top-K subgradient
        top_k = min(20, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        vals = c[top_idx]
        w_raw = vals - vals[-1] + 1e-10
        w_raw = w_raw / w_raw.sum()

        grad = np.zeros(n)
        for wi, ki in zip(w_raw, top_idx):
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            a_slice = a[ki - hi:ki - lo + 1][::-1]
            grad[lo:hi + 1] += wi * a_slice

        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-30:
            break

        # Polyak-like step size
        polyak_step = (f_curr - f_lb) / grad_norm_sq
        eta = step_scale * min(polyak_step, 0.5 / np.sqrt(it + 1))

        a_new = _proj_simplex(a - eta * grad)
        sc = _score_norm(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_scale *= 0.9

    return best_a, best_score


def _spectral_flatten(a_init, n_iter, deadline):
    """Iterative spectral magnitude flattening (Gerchberg-Saxton style)."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Flattening: push magnitudes toward the mean
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)
