```
    grad[i] = a[k_star - i]  (for valid i)
    
    # Update with adaptive step

        a_new = a_new / a_new.sum()  # re-normalize
        if score(a_new) < score(a):
            a = a_new
            break
```

```python
"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy: Start from parent, use adaptive FISTA with Nesterov restart, 
top-k subgradient refinement, log-space Adam, and multi-scale exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k

def _full_conv(a, b):
    """Full convolution using FFT."""
    n = len(a)
    m = len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]

def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score2(a):
    """Score assuming sum=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_conv(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v, target_sum=1.0):
    """Project onto the simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_vec(z, beta):
    """Stable softmax: exp(z/beta) / sum(exp(z/beta))."""
    z = z - z.max()
    e = np.exp(z / beta)
    s = e.sum()
    if s < 1e-300:
        out = np.zeros_like(z)
        out[z.argmax()] = 1.0
        return out
    return e / s

def _fista_adaptive(a_init, beta, deadline, max_iter, restart_interval=50):
    """FISTA with adaptive step and Nesterov restart."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_conv(a_curr, a_curr)
    max_c = float(c_curr.max())
    
    # Adaptive step: start with 1/L where L is an estimate
    step = min(1.0, 0.5 * beta / max(1e-15, max_c))
    
    best_a = a_curr.copy()
    best_f = _score2(a_curr)
    f_curr = _score2(a_curr)
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n
        
        # Compute softmax weights
        c_y = _full_conv(y, y)
        cm = float(c_y.max())
        w = _softmax_vec(c_y, beta)
        
        # Gradient: 2 * (w * y^{-1}) in the appropriate indexing
        # grad[i] = 2 * sum_k w[k] * y[k-i] (for the full conv, restricted to [0,n-1])
        y_rev = y[::-1]
        wc = _full_conv(w, y_rev)
        h = wc[n-1:2*n-1]  # This gives sum_k w[k] * y[k-i] for i=0..n-1
        grad = 2.0 * h
        
        # Adaptive step with backtracking
        a_next = None
        for trial in range(5):
            a_candidate = _proj_simplex(y - step * grad)
            f_cand = _score2(a_candidate)
            if f_cand < f_curr * (1 + 1e-10):  # sufficient decrease (relaxed)
                a_next = a_candidate
                f_next = f_cand
                break
            step *= 0.5
        
        if a_next is None:
            # Reset with smaller step
            step = max(0.01 * beta / max(1e-15, max_c), 1e-15)
            a_next = _proj_simplex(y - step * grad)
            f_next = _score2(a_next)
        
        # Nesterov restart check
        if f_next > f_curr + 1e-12:
            # Not improving, restart momentum
            t_val = 1.0
            a_prev = a_curr.copy()
        else:
            a_prev = a_curr.copy()
        
        a_curr = a_next
        f_curr = f_next
        t_val = t_next
        
        if f_curr < best_f:
            best_f = f_curr
            best_a = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > restart_interval:
            # Restart with fresh momentum
            t_val = 1.0
            a_prev = a_curr.copy()
            no_improve = 0
        
        if it % 200 == 0:
            max_c = float(_full_conv(a_curr, a_curr).max())
    
    return best_a, best_f

def _topk_subgrad(a_init, k_list, deadline, step_init, min_step):
    """Top-k subgradient method with geometric step decay."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    
    best_a = a.copy()
    best_f = _score2(a)
    step = step_init
    
    c = _full_conv(a, a)
    
    while time.monotonic() < deadline:
        cm = float(c.max())
        M = cm
        
        # Get top-k lags
        top_idx = np.argsort(c)[::-1][:max(1, len(k_list))]
        
        # Weighted subgradient
        w_vals = c[top_idx] - c[top_idx[-1]] + 1e-10
        w = np.zeros(len(c))
        w[top_idx] = w_vals
        w = w / w.sum()
        
        # Compute gradient
        a_rev = a[::-1]
        wc = _full_conv(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h - 4.0 * M * a  # Full gradient on simplex
        
        # Update
        a_new = _proj_simplex(a - step * grad)
        
        f_new = _score2(a_new)
        if f_new < best_f:
            best_f = f_new
            best_a = a_new.copy()
            a = a_new
            c = _full_conv(a, a)
            # Slightly increase step if we made progress
            step = min(step * 1.05, 10.0)
        else:
            step *= 0.7
            if step < min_step:
                break
        
    return best_a, best_f

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter):
    """Adam optimization in log-space with beta continuation."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    best_f = _score2(a_norm)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter and time.monotonic() < deadline:
            t += 1
            a = _softmax_vec(z, 1.0)  # Standard softmax
            # Actually, for log-space, a = exp(z) / sum(exp(z))
            z_shifted = z - z.max()
            e = np.exp(z_shifted)
            a = e / e.sum()
            
            c = _full_conv(a, a)
            cm = float(c.max())
            w = _softmax_vec(c, beta)
            
            # Gradient in log-space: a * (2*h - 2*M) where h[i] = sum_k w[k]*a[k-i]
            a_rev = a[::-1]
            wc = _full_conv(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1**t)
            vh = v_buf / (1.0 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 1.0:
                update = update * (1.0 / un)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 20 == 0:
                z_s = z - z.max()
                e_s = np.exp(z_s)
                a_chk = e_s / e_s.sum()
                f_chk = _score2(a_chk)
                if f_chk < best_f:
                    best_f = f_chk
                    best_a = a_chk.copy()
            
            if t % 5000 == 0:
                current_lr *= 0.95
    
    # Final
    z_s = z - z.max()
    e_s = np.exp(z_s)
    a_final = e_s / e_s.sum()
    f_final = _score2(a_final)
    if f_final < best_f:
        best_f = f_final
        best_a = a_final.copy()
    
    return best_a, best_f

def _sps_flatten(a_init, n_iter=200, strength=0.5, deadline=None):
    """Soft spectral flattening."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_f = _score2(a)
    N_fft = _next_pow2(2*n-1)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        # Soft flatten: interpolate between current and flat
        mag_new = (1-strength) * mag + strength * avg_mag
        phase = A / (mag + 1e-30)
        A_new = phase * mag_new
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        
        if it % 10 == 0:
            f = _score2(a)
            if f < best_f:
                best_f = f
                best_a = a.copy()
    
    f_final = _score2(a)
    if f_final < best_f:
        best_f = f_final
        best_a = a.copy()
    
    return best_a, best_f
```
