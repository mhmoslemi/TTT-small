2. Compute `A = FFT(a_padded)`
3. `|A|^2` gives the DFT of the linear auto-convolution

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, c, T, threshold_frac=0.0):
    """
    Compute gradient using all peaks above threshold_frac * max,
    or top-k if threshold not specified.
    """
    n = len(a)

    if c_max < 1e-30:
        return np.zeros(n)

        if not np.any(active):
            active = np.zeros_like(c, dtype=bool)
            active[np.argmax(c)] = True
        p = np.zeros_like(c)
        p[active] = (c[active] - threshold) / (c[active].sum() - threshold * active.sum() + 1e-30)
    else:

def _spectral_flatten_init(n, n_iters=200, seed=42):
    """
    Initialize using spectral flattening approach.
    Try to find a non-negative sequence with flat auto-convolution spectrum.
    """
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)
    a = np.ones(n) / n
    
    for it in range(n_iters):
        # Zero-pad and FFT
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A) ** 2
        
        # Target: flat power spectrum
        target = power.mean()
        
        # Update: modify A to be flatter
        # A_new = A * sqrt(target / |A|) = A * sqrt(target) / |A|
        # But we need to handle zeros
        safe_power = np.maximum(power, 1e-30)
        A_new = A * np.sqrt(target / safe_power)
        
        # Inverse FFT
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        
        # Normalize
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a_new / s
    
    return a

    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses spectral flattening initialization, adaptive threshold active-set gradients,
multi-scale search (n=4096, 8192, 16384), periodic shaking, and extended polishing.
Key improvements over previous: spectral init, adaptive threshold gradient,
larger n exploration, better annealing, and diverse restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_soft(a, c, T):
    """Temperature-softmax gradient for the max of convolution."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _gradient_threshold(a, c, frac):
    """Gradient using all lags above frac*max, weighted by excess above threshold."""
    n = len(a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    threshold = frac * c_max
    excess = np.maximum(c - threshold, 0.0)
    s = float(excess.sum())
    if s < 1e-30:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = excess / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _gradient_active(a, c, k):
    """Gradient using top-k lags uniformly weighted."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / k
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,

                  threshold_frac=0.3, eval_interval=200, clip_norm=0.8,
                  shake_interval=4000, shake_mag=0.03, seed=0):
    n = len(a_init)
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if grad_mode == 'soft':
            g = _gradient_soft(a, c, T)
        elif grad_mode == 'active':
            g = _gradient_active(a, c, active_k)
        else:
            g = _gradient_threshold(a, c, threshold_frac)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Periodic shaking to escape local minima (only in early-mid phase)
        if shake_interval > 0 and t % shake_interval == 0 and frac < 0.7:
            noise = shake_mag * (1.0 - frac / 0.7) * rng.standard_normal(n)
            z += noise
            z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                lr_mult = min(1.0, lr_mult * 1.02)
            else:
                patience += 1
            if patience > 12 and lr_mult > 0.15:
                lr_mult *= 0.65
                patience = 0
            if patience > 25:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0
                lr_mult = max(0.3, lr_mult * 0.8)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _spectral_flatten_init(n, n_iters=300, seed=42):
    """
    Initialize using iterative spectral flattening.
    Tries to find a non-negative sequence with flat auto-convolution spectrum.
    """
    rng = np.random.default_rng(seed)
    N = _next_pow2(2 * n - 1)
    a = np.ones(n) / n

    for it in range(n_iters):
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        power = np.abs(A) ** 2

        # Adaptive target: blend flat with current
        target = power.mean()
        # Exponential moving average of the flattening strength
        blend = 0.3 + 0.7 * (it / n_iters)

        safe_power = np.maximum(power, target * 0.01)
        # Flattening: reduce peaks, boost valleys
        ratio = np.sqrt(target / safe_power)
        ratio = np.clip(ratio, 1.0 - blend, 1.0 + blend * 0.5)
        A_new = A * ratio

        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            break
        a = a_new / s

    return a


def _cos_modulated_init(n, alpha=0.5, phase=0.0):
    """Initialize with cosine-modulated uniform: a[j] = max(0, 1 + alpha*cos(2*pi*j/n + phase))."""
    j = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * j / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _geometric_init(n, beta=0.5):
    """Initialize with one-sided geometric decay."""
    j = np.arange(n)
    a = np.exp(-beta * j / n)
    s = a.sum()
    return a / s


def _two_hump_init(n, ratio=0.3):
    """Initialize with two concentrated humps."""
    a = np.zeros(n)
    center1 = int(n * 0.3)
    center2 = int(n * 0.7)
    width = int(n * ratio / 4)
    for c in [center1, center2]:
        for j in range(max(0, c - width), min(n, c + width + 1)):
            a[j] += np.exp(-0.5 * ((j - c) / max(1, width // 2)) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _local_refine(a, deadline, n_trials=200):

    Simple coordinate descent: try small perturbations on individual elements.
    Keeps the best perturbation at each step.

    a_cur = a.copy()
    sc_cur = _score_norm(a_cur)
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Pick a random position
        i = rng.integers(0, n)
        # Try scaling by a factor
        for factor in [0.9, 0.95, 1.05, 1.1]:
            a_try = a_cur.copy()
            a_try[i] *= factor
            # Re-normalize to sum=1
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc_try = _score_norm(a_try)
            if sc_try < sc_cur - 1e-12:
                a_cur = a_try
                sc_cur = sc_try
                break

            eval_interval=250, clip_norm=0.7,
            shake_interval=4000, shake_mag=0.025, seed=seed + 2)
        full_sc = _score(a_large_opt)
        print(f"Phase 3 (n={n_large}, threshold): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_large_opt.copy()
        else:
            # Interpolate back and try at n_base
            a_back = _interp_to_length(a_large_opt, n_base)
            c_back = _fft_convolve(a_back, a_back)
            max_cback = float(c_back.max())
            a_back_opt, sc_back = _adam_z_space(
                a_back, phase3_end - 10,
                T_start=0.003 * max_cback, T_end=0.0002 * max_cback,
                lr_start=0.025, lr_end=0.002,
                grad_mode='threshold', threshold_frac=0.15,
                eval_interval=150, clip_norm=0.5,
                seed=seed + 3)
            full_sc_back = _score(a_back_opt)
            print(f"Phase 3b (back to n={n_base}): score={full_sc_back:.10f}")
            if full_sc_back < best_score:
                best_score = full_sc_back
                best_a = a_back_opt.copy()

    # ---- Phase 4: Continue at best n with refined params (600s - 950s) ----
    phase4_end = min(t_start + 950.0, deadline)
    if time.monotonic() < phase4_end - 20:
        n_cur = len(best_a)
        c_cur = _fft_convolve(best_a, best_a)
        max_cc = float(c_cur.max())

        # Use threshold gradient with adapting threshold
        a_cont, sc_cont = _adam_z_space(
            best_a, phase4_end - 10,
            T_start=0.001 * max_cc, T_end=0.00005 * max_cc,
            lr_start=0.02, lr_end=0.001,
            grad_mode='threshold', threshold_frac=0.1,
            eval_interval=150, clip_norm=0.5,
            shake_interval=5000, shake_mag=0.015, seed=seed + 4)
        full_sc = _score(a_cont)
        print(f"Phase 4 (refine n={n_cur}, thresh=0.1): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_cont.copy()

    # ---- Phase 5: Try n=16384 (950s - 1300s) ----
    phase5_end = min(t_start + 1300.0, deadline)
    if time.monotonic() < phase5_end - 40:
        n_xl = 16384
        a_xl = _interp_to_length(best_a, n_xl)
        c_xl = _fft_convolve(a_xl, a_xl)
        max_cxl = float(c_xl.max())

        a_xl_opt, sc_xl = _adam_z_space(
            a_xl, phase5_end - 15,
            T_start=0.003 * max_cxl, T_end=0.0001 * max_cxl,
            lr_start=0.025, lr_end=0.002,
            grad_mode='threshold', threshold_frac=0.15,
            eval_interval=400, clip_norm=0.6,
            shake_interval=5000, shake_mag=0.02, seed=seed + 5)
        full_sc = _score(a_xl_opt)
        print(f"Phase 5 (n={n_xl}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_xl_opt.copy()

        # Also try spectral init at this scale
        a_sf_xl = _spectral_flatten_init(n_xl, n_iters=100, seed=seed + 10)
        c_sf = _fft_convolve(a_sf_xl, a_sf_xl)
        max_csf = float(c_sf.max())
        a_sf_opt, sc_sf = _adam_z_space(
            a_sf_xl, phase5_end - 15,
            T_start=0.003 * max_csf, T_end=0.0001 * max_csf,
            lr_start=0.025, lr_end=0.002,
            grad_mode='threshold', threshold_frac=0.2,
            eval_interval=400, clip_norm=0.6, seed=seed + 11)
        full_sc_sf = _score(a_sf_opt)
        print(f"Phase 5b (n={n_xl}, spectral init): score={full_sc_sf:.10f}")
        if full_sc_sf < best_score:
            best_score = full_sc_sf
            best_a = a_sf_opt.copy()

    # ---- Phase 6: Diverse perturbation restarts (1300s - 1550s) ----
    phase6_end = min(t_start + 1550.0, deadline)
    n_best = len(best_a)
    if time.monotonic() < phase6_end - 20:
        configs = [
            (0.01, 'active', 3, 0.0),
            (0.02, 'threshold', 0, 0.1),
            (0.03, 'active', 5, 0.0),
            (0.05, 'threshold', 0, 0.05),
            (0.08, 'soft', 0, 0.0),
        ]
        for idx, (sigma, gmode, ak, tfrac) in enumerate(configs):
            if time.monotonic() > phase6_end - 45:
                break
            z_pert = np.log(np.maximum(best_a, 1e-15))
            z_pert = z_pert + sigma * rng.standard_normal(n_best)
            z_pert = z_pert - z_pert.mean()
            a_pert = _softmax_stable(z_pert)
            c_pert = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_pert.max())

            kwargs = {'grad_mode': gmode, 'eval_interval': 120, 'clip_norm': 0.5,
                      'seed': seed + 20 + idx}
            if gmode == 'active':
                kwargs['active_k'] = ak
            elif gmode == 'threshold':
                kwargs['threshold_frac'] = tfrac

            a_pert_opt, sc_pert = _adam_z_space(
                a_pert, min(phase6_end - 10, time.monotonic() + 55),
                T_start=0.001 * max_cp, T_end=0.00005 * max_cp,
                lr_start=0.02, lr_end=0.001, **kwargs)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{idx}] (sigma={sigma:.2f}, {gmode}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # ---- Phase 7: Final ultra-fine polish (1550s - deadline) ----
    if time.monotonic() < deadline - 20:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())

        # Very low temperature, very small learning rate
        a_final, sc_final = _adam_z_space(
            best_a, deadline - 15,
            T_start=0.0003 * max_cf, T_end=0.000005 * max_cf,
            lr_start=0.004, lr_end=0.0002,
            grad_mode='threshold', threshold_frac=0.05,
            eval_interval=80, clip_norm=0.2,
            shake_interval=0, shake_mag=0.0, seed=seed + 30)
        full_sc = _score(a_final)
        print(f"Phase 7 (polish thresh=0.05): score={full_sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # One more pass with top-3 active
        if time.monotonic() < deadline - 10:
            a_final2, sc_final2 = _adam_z_space(
                best_a, deadline - 5,
                T_start=0.0002 * max_cf, T_end=0.000003 * max_cf,
                lr_start=0.003, lr_end=0.0001,
                grad_mode='active', active_k=3,
                eval_interval=60, clip_norm=0.15,
                shake_interval=0, shake_mag=0.0, seed=seed + 31)
            full_sc2 = _score(a_final2)
            print(f"Phase 7b (polish active k=3): score={full_sc2:.10f}")
            if full_sc2 < best_score:
                best_score = full_sc2
                best_a = a_final2.copy()

    # ---- Phase 8: Local coordinate refinement (last few seconds) ----
    if time.monotonic() < deadline - 3:
        a_local, sc_local = _local_refine(best_a, deadline - 2, n_trials=100)
        full_sc_local = _score(a_local)
        print(f"Phase 8 (local refine): score={full_sc_local:.10f}")
        if full_sc_local < best_score:
            best_score = full_sc_local
            best_a = a_local.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
