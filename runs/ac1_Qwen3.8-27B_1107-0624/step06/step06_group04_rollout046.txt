"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses binomial-shape initialization, FISTA with beta-continuation,
Adam log-space fine-tuning, peak-targeted subgradient, and multi-n exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1, so just 2*n*max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _binomial_shape(n):
    """Generate the binomial coefficient shape a[k] = C(2k,k)/4^k"""
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)  # C(2k,k)/4^k = C(2k-2,k-1)/4^{k-1} * (2k-1)/(2k)
    return a / a.sum()

def _symmetric_binomial(n):

    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2*k-1) / (2*k)
    # Mirror from the other end
    a[n-1] = 1.0
    for k in range(1, n):
        a[n-1-k] += a[k-1] * (2*(n-1-k)-1) / (2*(n-1-k)) if n-1-k > 0 else 0
    # Actually let me redo this
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):

    a = a + b
    return a / a.sum()

def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        # Gradient of the smoothed max
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            # Gradient in log-space
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _peak_targeted_steps(a_init, n_steps, deadline):
    """Targeted subgradient steps on the top peaks"""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Focus on top K peaks
        top_k = min(10, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        
        # Weight by how much above the mean
        mean_c = c.sum() / len(c)
        w_vec = np.zeros(len(c))
        for idx in top_idx:
            w_vec[idx] = (c[idx] - mean_c + 1e-10) ** 2
        w_vec /= w_vec.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w_vec, a_rev)
        h = wc[n-1:2*n-1]
        grad = 2.0 * h

        # Adaptive step size
        step_size = 0.1 / (1.0 + 0.005 * it)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = _proj_simplex(a - 0.1 * step_size * grad)
            if _score_norm(a_new2) < _score_norm(a):
                a = a_new2

    return best_a, best_score

def _interpolate(a_old, n_new):

    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()
