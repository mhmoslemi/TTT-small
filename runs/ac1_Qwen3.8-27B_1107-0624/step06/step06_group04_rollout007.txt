                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad_a = _compute_grad_a(a, w)
            dot_ag = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - dot_ag)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -clip_val, clip_val)
            z = z - z.mean()

            if t % check_interval == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                    phase_no_improve = 0
                else:
                    no_improve += 1
                    phase_no_improve += 1

            if t % 3000 == 0:
                lr = max(0.001, lr * 0.96)

            if phase_no_improve > 40 and t > 1000 and noise_restart:
                z = z + 0.05 * np.random.standard_normal(n)
                z = z - z.mean()
                m_buf *= 0.5
                v_buf *= 0.5
                phase_no_improve = 0
                lr = max(0.001, lr * 1.2)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten_steps(a, n_steps=200, deadline=None):
    """GERG-style spectral flattening followed by non-negativity projection."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = _next_pow2(2 * n - 1)

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean() + 1e-30
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _beta_schedule(max_c, n_phases=10, start_frac=0.05, end_frac=1e-5):
    """Generate a geometric beta schedule from start_frac*max_c to end_frac*max_c."""
    start = start_frac * max_c
    end = end_frac * max_c
    if start <= 0:
        start = 1e-3
    if end <= 0:
        end = 1e-10
    ratios = (end / start) ** (1.0 / max(1, n_phases - 1))
    betas = []
    b = start
    for _ in range(n_phases):
        betas.append(b)
        b *= ratios
    return betas


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        sc_parent = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple target sizes
    target_sizes = [8192, 16384]

    # Allocate time: 60% to 8192, 35% to 16384, 5% final polish
    time_8192 = 0.55 * (deadline - t_start)
    time_16384 = 0.35 * (deadline - t_start)
    time_polish = 0.10 * (deadline - t_start)

    for n_target, time_alloc in [(8192, time_8192), (16384, time_16384)]:
        phase_deadline = time.monotonic() + time_alloc
        if time.monotonic() > phase_deadline - 50:
            continue

        print(f"\n=== Optimizing n={n_target} (budget {time_alloc:.0f}s) ===")

        # Create diverse initializations
        inits = []
        names = []

        # 1. Interpolated parent
        a_interp = _interp_profile(parent, n_target)
        inits.append(a_interp)
        names.append("interp_parent")

        # 2. Tilted
        a_tilt = _make_tilted(n_target, p=1.5)
        inits.append(a_tilt)
        names.append("tilted")

        # 3. Two-bump
        a_bump = _make_two_bump(n_target, sep=0.6, width=0.12)
        inits.append(a_bump)
        names.append("two_bump")

        # 4. Tapered
        a_taper = _make_tapered(n_target, power=0.3)
        inits.append(a_taper)
        names.append("tapered")

        # 5. Mix of parent and tilted
        a_mix = 0.5 * _interp_profile(parent, n_target) + 0.5 * _make_tilted(n_target, p=2.0)
        a_mix = a_mix / a_mix.sum()
        inits.append(a_mix)
        names.append("mix_tilt")

        # Evaluate initializations
        for name, a_init in zip(names, inits):
            sc = _score(a_init)
            print(f"  Init '{name}': score={sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_init.copy()

        # Optimize each initialization (time-limited)
        n_inits = len(inits)
        per_init_time = time_alloc / (n_inits + 1)  # +1 for final best optimization

        for i, (name, a_init) in enumerate(zip(names, inits)):
            if time.monotonic() > phase_deadline - 30:
                break
            init_deadline = min(phase_deadline - 20, time.monotonic() + per_init_time)
            if time.monotonic() > init_deadline - 5:
                continue

            # Pre-process with spectral flatten for a few steps
            a_pre, _ = _spectral_flatten_steps(a_init, n_steps=50,
                                               deadline=min(init_deadline - 10, time.monotonic() + 15))
            sc_pre = _score(a_pre)
            if sc_pre < best_score:
                best_score = sc_pre
                best_seq = a_pre.copy()

            # Adam optimization with beta continuation
            c_pre = _full_convolve(a_pre, a_pre)
            max_cp = float(c_pre.max())
            betas = _beta_schedule(max_cp, n_phases=8, start_frac=0.03, end_frac=1e-5)

            a_opt, sc_opt_norm = _adam_logspace(a_pre, betas, init_deadline,
                                                lr_init=0.015, max_iter=100000)
            sc_opt = _score(a_opt)
            print(f"  '{name}' -> n={n_target}: score={sc_opt:.10f} (t={time.monotonic()-t_start:.0f}s)")

            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()

        # Final optimization of the best found at this n
        if time.monotonic() < phase_deadline - 30:
            a_work = best_seq.copy()
            if len(a_work) != n_target:
                a_work = _interp_profile(a_work, n_target)
            a_work = a_work / a_work.sum()

            c_work = _full_convolve(a_work, a_work)
            max_cw = float(c_work.max())
            betas_fine = _beta_schedule(max_cw, n_phases=6, start_frac=0.01, end_frac=1e-6)

            a_final, sc_fn = _adam_logspace(a_work, betas_fine, phase_deadline - 10,
                                            lr_init=0.008, max_iter=200000)
            sc_final = _score(a_final)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_final.copy()
                print(f"  Final n={n_target}: score={sc_final:.10f}")

    # === Phase: Final polish on best sequence ===
    print(f"\n=== Final polish (budget {time_polish:.0f}s) ===")
    polish_deadline = deadline - 5
    a_pol = best_seq.copy()
    a_pol = np.maximum(a_pol, 0)
    a_pol = a_pol / a_pol.sum()

    # Spectral flatten
    if time.monotonic() < polish_deadline - 20:
        a_sf, _ = _spectral_flatten_steps(a_pol, n_steps=300,
                                          deadline=min(polish_deadline - 20, time.monotonic() + 30))
        sc_sf = _score(a_sf)
        if sc_sf < best_score:
            best_score = sc_sf
            best_seq = a_sf.copy()
            print(f"  Spectral polish: score={sc_sf:.10f}")
            a_pol = a_sf.copy()

    # Fine Adam
    if time.monotonic() < polish_deadline - 10:
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = _beta_schedule(max_cp, n_phases=5, start_frac=0.005, end_frac=1e-7)
        a_full, sc_fn = _adam_logspace(a_pol, betas_ult, polish_deadline - 5,
                                       lr_init=0.004, max_iter=300000,
                                       check_interval=100)
        sc_full = _score(a_full)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_full.copy()
            print(f"  Adam polish: score={sc_full:.10f}")

    # Try a few small-perturbation restarts
    if time.monotonic() < polish_deadline - 10 and best_seq is not None:
        n_final = len(best_seq)
        for restart_i in range(3):
            if time.monotonic() > polish_deadline - 10:
                break
            sigma = 0.02 + 0.01 * restart_i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z + sigma * rng.standard_normal(n_final)
            z = z - z.mean()
            a_pert = _softmax(z)
            c_pert = _full_convolve(a_pert, a_pert)
            max_cpt = float(c_pert.max())
            betas_r = _beta_schedule(max_cpt, n_phases=4, start_frac=0.005, end_frac=1e-6)
            a_r, _ = _adam_logspace(a_pert, betas_r, min(polish_deadline - 5, time.monotonic() + 20),
                                    lr_init=0.003, max_iter=50000)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart_i}: score={sc_r:.10f}")

    # Final validation and return
    print(f"\nAll done: best_score={best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            s = final.sum()
            final = final / s
        sc_final_check = _score(final)
        print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(512) / 512
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
