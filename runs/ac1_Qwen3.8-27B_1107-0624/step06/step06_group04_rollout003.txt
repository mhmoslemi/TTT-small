def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _softmax(z, beta):
    z_shifted = (z - z.max()) / beta
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(z)
        w[int(np.argmax(z))] = 1.0
        return w
    return e / s

def _topk_weights(c, k, temperature=0.01):
    """Create weights concentrated on top-k peaks."""
    n = len(c)
    w = np.zeros(n)
    if k >= n:
        return _softmax(c, temperature * c.max())
    idx = np.argpartition(c, -k)[-k:]
    c_top = c[idx]
    w_top = _softmax(c_top, temperature * (c_top.max() - c_top.min() + 1e-15))
    w[idx] = w_top
    return w

def _compute_grad(a, w):
    """Compute gradient of smoothed max-convolution w.r.t. a on simplex."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _multiplicative_descent(a_init, deadline, beta_schedule, topk_schedule, lr_init=0.1, max_iter=100000):
    """
    Multiplicative weights descent on the simplex.
    Uses adaptive peak targeting and backtracking.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    c = _full_convolve(a, a)
    max_c = float(c.max())
    
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Determine which schedule values to use
        frac = it / max_iter
        bi = min(int(frac * len(beta_schedule)), len(beta_schedule) - 1)
        beta = beta_schedule[bi]
        ki = min(int(frac * len(topk_schedule)), len(topk_schedule) - 1)
        k = topk_schedule[ki]
        
        # Recompute c if needed (every 10 iters or first)
        if it % 10 == 0:
            c = _full_convolve(a, a)
            max_c = float(c.max())
        
        # Compute weights
        w = _topk_weights(c, k, temperature=beta / max_c)
        
        # Compute gradient
        grad = _compute_grad(a, w)
        
        # Multiplicative update: a *= exp(-lr * grad) / sum
        # Clip grad to prevent numerical issues
        grad_clipped = np.clip(grad, -5.0, 5.0) / max_c
        a_new = a * np.exp(-lr * grad_clipped)
        s_new = a_new.sum()
        if s_new < 1e-30 or not np.isfinite(s_new):
            break
        a_new = a_new / s_new
        
        # Check improvement
        if it % 20 == 0 or it < 10:
            sc = _score_norm(a_new)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
            else:
                no_improve += 1
                a = a_new  # still accept for exploration
                # Reduce learning rate if stuck
                if no_improve > 30:
                    lr *= 0.9
                    no_improve = 0
        
        # Adaptive lr
        if it % 500 == 0:
            lr = max(0.001, min(lr_init, lr * 1.02))  # slowly increase
    
    # Final check
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _fista_peak_shave(a_init, deadline, beta, max_iter=50000, topk=50):
    """
    FISTA-style accelerated gradient with top-k peak shaving.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_curr.max())
    L = 4.0 * n * topk / (n + topk)  # approximate Lipschitz
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        
        # Use top-k weights
        if topk < len(c_y):
            idx = np.argpartition(c_y, -topk)[-topk:]
            w = np.zeros(len(c_y))
            c_top = c_y[idx]
            beta_eff = beta * cm
            w_top = _softmax(c_top, beta_eff)
            w[idx] = w_top
        else:
            w = _softmax(c_y, beta * cm)
        
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_logspace_v2(a_init, deadline, beta_schedule, lr_init, max_iter_total, topk_schedule):
    """
    Improved Adam optimizer in log-space with top-k peak targeting.
    """
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0
    
    while t < max_iter_total:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Determine top-k
        frac = t / max_iter_total
        ki = min(int(frac * len(topk_schedule)), len(topk_schedule) - 1)
        k = topk_schedule[ki]
        
        # Determine beta
        bi = min(int(frac * len(beta_schedule)), len(beta_schedule) - 1)
        beta = beta_schedule[bi]
        
        # Compute weights
        if k < len(c):
            idx = np.argpartition(c, -k)[-k:]
            w = np.zeros(len(c))
            c_top = c[idx]
            w_top = _softmax(c_top, beta * cm)
            w[idx] = w_top
        else:
            w = _softmax(c, beta * cm)
        
        # Gradient in log-space
        grad_conv = _compute_grad(a, w)
        grad = 2.0 * a * (grad_conv - float(np.dot(w, c)))
        
        # Hmm, that's not right. Let me use the standard log-space gradient.
        # f(a) = max_k c[k], dc[k]/da[i] = a[k-i] (approximately)
        # df/dz[i] = df/da[i] * da[i]/dz[i] = grad_conv[i] * a[i] (since a = softmax(z))
        # But we also need to account for the normalization...
        # Actually for the score 2*n*max(c), the gradient w.r.t. z is:
        # d/dz[i] [2n * max_k sum_j a[j]*a[k-j]] = 2n * sum_k w[k] * dc[k]/da[i] * a[i]
        # = 2n * a[i] * (w * a_rev)[i] * 2 (from the _compute_grad which already has factor 2)
        # Wait, _compute_grad already returns 2*h where h is the cross-correlation.
        # So grad_conv[i] = 2 * (w ⋆ a)[i]
        # And the log-space gradient is a[i] * grad_conv[i] (by chain rule for softmax)
        # But we need to subtract the mean for the normalization constraint.
        
        grad_log = a * grad_conv
        grad_log = grad_log - grad_log.mean()  # center for sum constraint
        
        if not np.all(np.isfinite(grad_log)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_log
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_log * grad_log)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % 20 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        if t % 5000 == 0:
            current_lr = max(0.001, current_lr * 0.97)
        
        if no_improve_count > 100 and t > 500:
            current_lr = max(0.001, current_lr * 0.95)
            no_improve_count = 0
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adam_simplex(a_init, deadline, beta_schedule, lr_init, max_iter_total, topk_schedule):
    """
    Adam optimizer directly on the simplex (with projection).
    This avoids the log-space reparametrization issues.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0
    
    while t < max_iter_total:
        if time.monotonic() > deadline:
            break
        t += 1
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        frac = t / max_iter_total
        ki = min(int(frac * len(topk_schedule)), len(topk_schedule) - 1)
        k = topk_schedule[ki]
        bi = min(int(frac * len(beta_schedule)), len(beta_schedule) - 1)
        beta = beta_schedule[bi]
        
        if k < len(c):
            idx = np.argpartition(c, -k)[-k:]
            w = np.zeros(len(c))
            c_top = c[idx]
            w_top = _softmax(c_top, beta * cm)
            w[idx] = w_top
        else:
            w = _softmax(c, beta * cm)
        
        grad = _compute_grad(a, w)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 2.0:
            update = update * (2.0 / update_norm)
        
        a_new = _proj_simplex(a - update * cm)
        
        if t % 30 == 0:
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
                a = a_new
            else:
                no_improve += 1
                a = a_new
        else:
            a = a_new
        
        if t % 3000 == 0:
            current_lr = max(0.0005, current_lr * 0.95)
        
        if no_improve > 80 and t > 500:
            current_lr = max(0.0005, current_lr * 0.9)
            no_improve = 0
    
    sc_final = _score_norm(a)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score

def _spectral_iterations(a_init, deadline, n_iter=5000):
    """
    Spectral domain optimization: try to flatten the convolution spectrum.
    Uses iterative projection between time and frequency domains.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _scoreNorm(a)
    
    N_fft = _next_pow2(2 * n - 1)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Forward FFT
        A = np.fft.fft(a, N_fft)
        
        # In freq domain: we want |A^2| to be as flat as possible
        # A^2 = C (FFT of convolution)
        C = A * A
        
        # Try to make |C| more uniform (while keeping DC component)
        # DC: C[0] = (sum(a))^2 = 1
        # We want other components to be small
        # Strategy: reduce non-DC components
        mag = np.abs(C)
        mag[0] = 0  # protect DC
        if mag.max() > 0:
            # Soft threshold
            threshold = 0.1 * mag.mean()
            C_new = C * np.where(mag > threshold, 1.0, threshold / (mag + 1e-30))
            # But we need C = A^2, so A = sqrt(C)
            # This is tricky because of phase...
            pass
        
        # Alternative: direct time-domain peak shaving via spectral filtering
        # Compute current convolution
        c = np.fft.ifft(C, N_fft).real[:2*n-1]
        c = np.maximum(c, 0)  # numerical fix
        
        # Find peaks
        cm = c.max()
        if cm < 1e-15:
            break
        
        # Create a "target" spectrum: reduce peaks in time domain
        # by reducing corresponding frequency components
        # Simple approach: reduce high-magnitude frequency components
        mag_A = np.abs(A)
        mag_A[0] = 0
        if mag_A.max() > 0:
            # Reduce non-DC by a factor
            reduction = 0.999
            A_new = A.copy()
            A_new[1:] *= reduction
            # But this just shrinks everything...
            # Better: reduce the components that contribute most to peaks
            pass
        
        # Actually, let's use a different spectral approach:
        # The peak of c is at the center (for non-negative a, it's at k=n-1)
        # c[n-1] = sum(a_i^2) ... wait no, c[n-1] = sum a_i * a_{n-1-i}
        # For non-negative a, the max is NOT necessarily at n-1
        
        # Let me just do a simple spectral flattening:
        # Reduce the variance of |A|^2 (which correlates with peakiness of c)
        power = np.abs(A) ** 2
        power[0] = 0
        if power.sum() > 0:
            avg_power = power.mean()
            # Compress dynamic range
            compression = 0.5
            power_new = avg_power * (power / (avg_power + 1e-30)) ** compression
            power_new[0] = 0
            # Recompute A with new magnitude, keeping phase
            phase = np.angle(A)
            A_new = np.sqrt(power_new + 1e-30) * np.exp(1j * phase)
            # But we need A to come from a non-negative sequence...
            a_new = np.fft.ifft(A_new, N_fft).real[:n]
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                a = a_new
        
        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score
