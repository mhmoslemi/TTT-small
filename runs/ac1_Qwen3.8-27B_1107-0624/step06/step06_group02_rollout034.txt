                update = update * (max_update / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if lr_decay < 1.0 and t % lr_decay_interval == 0:
                current_lr = max(0.0001, current_lr * lr_decay)

            if t % 200 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)

    rng = np.random.default_rng(777)
    factors = [0.92, 0.96, 0.99, 1.01, 1.04, 1.08]

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            n_coords = min(2048, n)
            indices = rng.choice(n, size=n_coords, replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-18:
                continue
            best_local_sc = _score_norm(a)
            best_local_val = old_val
            for f in factors:
                a[j] = old_val * f
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local_val = a[j]
                else:
                    a[j] = old_val
            a[j] = best_local_val

            if best_local_sc < best_sc - 1e-14:
                s = a.sum()
                if s > 1e-30:
                    a = a / s
                best_sc = _score_norm(a)
                best_a = a.copy()
                improved = True

        if not improved and r > 40:
            break

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _make_beta_schedule(max_c, n_betas=25, start_frac=0.005, end_frac=1e-8):
    start_beta = start_frac * max_c
    end_beta = end_frac * max_c
    if start_beta <= end_beta:
        return [start_beta]
    ratios = (end_beta / start_beta) ** (1.0 / (n_betas - 1))
    return [start_beta * (ratios ** i) for i in range(n_betas)]


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        sc_parent = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # === Phase 1: Explore different lengths with quick optimization ===
    print("\n=== Phase 1: Length exploration ===")
    lengths = [n_parent]
    if n_parent * 2 <= 16384:
        lengths.append(n_parent * 2)
    if n_parent * 4 <= 32768:
        lengths.append(n_parent * 4)

    candidates = {}
    for n_try in lengths:
        if time.monotonic() > deadline - 400:
            break

        if n_try == n_parent:
            a_try = parent.copy()
        else:
            a_try = _interp_profile(parent, n_try)

        sc_try = _score(a_try)
        print(f"  n={n_try}: initial score={sc_try:.10f}")

        c_try = _convolve(a_try, a_try)
        max_c = float(c_try.max())
        betas_quick = _make_beta_schedule(max_c, n_betas=8, start_frac=0.002, end_frac=1e-6)

        per_time = min(90.0, (deadline - time.monotonic() - 400) / max(1, len(lengths)))
        a_opt, _ = _adam_optimize(a_try, time.monotonic() + per_time, betas_quick, lr=0.01, max_iter=1000000)
        a_opt = a_opt / a_opt.sum()
        sc_full = _score(a_opt)
        print(f"  n={n_try}: after Adam score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")

        candidates[n_try] = (a_opt, sc_full)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()

    # === Phase 2: Deep optimization on the best length ===
    print("\n=== Phase 2: Deep Adam optimization ===")
    n_best = len(best_seq)
    a_work = best_seq.copy()
    a_work = np.clip(a_work, 0, 1000)
    a_work = a_work / a_work.sum()

    c_work = _convolve(a_work, a_work)
    max_cw = float(c_work.max())

    # Very gradual beta schedule for deep optimization
    betas_deep = _make_beta_schedule(max_cw, n_betas=35, start_frac=0.001, end_frac=1e-10)

    phase2_deadline = min(deadline, t_start + 850.0)
    per_time = phase2_deadline - time.monotonic() - 10
    if per_time > 30:
        a_ext, _ = _adam_optimize(a_work, time.monotonic() + per_time, betas_deep,
                                  lr=0.006, max_iter=5000000, lr_decay=0.95, lr_decay_interval=8000)
        a_ext = a_ext / a_ext.sum()
        sc_ext = _score(a_ext)
        print(f"  Deep Adam n={n_best}: score={sc_ext:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_ext < best_score:
            best_score = sc_ext
            best_seq = a_ext.copy()

    # === Phase 3: Second round of Adam with different init ===
    print("\n=== Phase 3: Second Adam round ===")
    if time.monotonic() < deadline - 300:
        # Perturb slightly and re-optimize
        z_pert = np.log(np.maximum(best_seq, 1e-15)) + 0.01 * rng.standard_normal(len(best_seq))
        z_pert = z_pert - z_pert.mean()
        a_pert = _softmax(z_pert)

        c_p = _convolve(a_pert, a_pert)
        max_cp = float(c_p.max())
        betas_pert = _make_beta_schedule(max_cp, n_betas=20, start_frac=0.0005, end_frac=1e-9)

        per_time = min(200.0, deadline - time.monotonic() - 280)
        if per_time > 20:
            a_p2, _ = _adam_optimize(a_pert, time.monotonic() + per_time, betas_pert,
                                     lr=0.004, max_iter=1000000)
            a_p2 = a_p2 / a_p2.sum()
            sc_p2 = _score(a_p2)
            print(f"  Second round: score={sc_p2:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_p2 < best_score:
                best_score = sc_p2
                best_seq = a_p2.copy()

    # === Phase 4: Coordinate descent polish ===
    print("\n=== Phase 4: Coordinate descent ===")
    if time.monotonic() < deadline - 150:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 130, time.monotonic() + 200)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=500)
        a_cd = a_cd / a_cd.sum()
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 5: Multi-restart ===
    print("\n=== Phase 5: Multi-restart ===")
    if time.monotonic() < deadline - 80:
        n_restarts = 4
        per_restart = (deadline - time.monotonic() - 50) / n_restarts
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            if i % 4 == 0:
                z = np.log(np.maximum(a_base, 1e-15)) + 0.04 * rng.standard_normal(len(a_base))
                z = z - z.mean()
                p_start = _softmax(z)
            elif i % 4 == 1:
                shift = rng.integers(1, max(2, len(a_base) // 128))
                p_start = np.roll(a_base, shift).copy()
                p_start = p_start / p_start.sum()
            elif i % 4 == 2:
                # Tapered perturbation
                n_b = len(a_base)
                p_start = a_base.copy()
                center = n_b // 2
                width = n_b // 4
                for j in range(max(0, center - width), min(n_b, center + width)):
                    dist = abs(j - center) / width
                    p_start[j] *= 1.0 + 0.1 * np.cos(np.pi * dist) * rng.standard_normal()
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()
            else:
                # Two-block scaling
                n_b = len(a_base)
                half = n_b // 2
                p_start = a_base.copy()
                factor1 = 0.85 + 0.3 * rng.random()
                factor2 = 2.0 / factor1
                p_start[:half] *= factor1
                p_start[half:] *= factor2
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = _make_beta_schedule(max_cs, n_betas=10, start_frac=0.0003, end_frac=1e-8)

            p_best, _ = _adam_optimize(p_start, time.monotonic() + max(5.0, per_restart * 0.75),
                                       bs, lr=0.004, max_iter=300000)
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)

            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = _make_beta_schedule(max_cp, n_betas=15, start_frac=1e-5, end_frac=1e-11)
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, _ = _adam_optimize(a_pol, time.monotonic() + per_t, betas_ult,
                                      lr=0.001, max_iter=2000000)
            a_pol = a_pol / a_pol.sum()
            sc_pol = _score(a_pol)
            print(f"  Final polish: score={sc_pol:.10f}")
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()

    # Final validation
    print(f"\nBest score: {best_score:.10f}, n={len(best_seq)}, time={time.monotonic()-t_start:.1f}s")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
