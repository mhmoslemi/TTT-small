"""
Hybrid LP-based Frank-Wolfe + gradient polishing optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP oracle for exact direction finding, Adam in log-space for polishing,
and coordinate descent for final refinement.

import numpy as np
import time
from scipy.sparse import csr_matrix, lil_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)

def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    n = len(f)
    n_rows = 2 * n - 1
    rows_list = []
    cols_list = []
    vals_list = []
    for d in range(n):
        k_start = d
        k_end = min(d + n - 1, 2 * n - 2)
        ks = np.arange(k_start, k_end + 1)
        is_ = ks - d
        rows_list.append(ks)
        cols_list.append(is_)
        vals_list.append(np.full(len(ks), f[d]))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    C = csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    return C

def _lp_direction(f, M, max_time=15.0):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for all k."""
    n = len(f)
    try:
        import cvxpy as cp
    except ImportError:
        return None

    t0 = time.monotonic()
    C = _build_conv_matrix(f)
    if time.monotonic() - t0 > max_time * 0.3:
        return None

    g = cp.Variable(n, nonneg=True)
    constraints = [C @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    solvers_to_try = [
        cp.MOSEK,
        cp.GUROBI,
        cp.CLARABEL,
        cp.PDLP,
        cp.SCS,
        cp.ECOS,
        cp.GLPK,
        cp.CVXOPT,
    ]

    for solver in solvers_to_try:
        if time.monotonic() - t0 > max_time:
            return None
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                g_val = g.value
                if g_val is not None and np.all(np.isfinite(g_val)):
                    g_val = np.maximum(g_val, 0.0)
                    gs = g_val.sum()
                    if gs > 1e-15:
                        return g_val / gs
        except Exception:
            continue

    return None

def _lp_frank_wolfe(a_init, deadline, max_iters=500):
    """LP-based Frank-Wolfe optimization."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    t_start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() > deadline - 5:
            break

        c = _full_convolve(a_curr, a_curr)
        M = float(c.max())

        g = _lp_direction(a_curr, M, max_time=min(20.0, deadline - time.monotonic() - 2))
        if g is None:
            print(f"  LP failed at iter {it}, stopping LP phase")
            break

        # Line search: find best t in [0, 1]
        best_t = 0.0
        best_sc = _score_norm(a_curr)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1.0 - t) * a_curr + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t <= 0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.05]:
                a_trial = (1.0 - t) * a_curr + t * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t

        if best_t <= 0:
            # No improvement possible with this direction
            break

        a_curr = (1.0 - best_t) * a_curr + best_t * g
        a_curr = np.maximum(a_curr, 0.0)
        a_curr = a_curr / a_curr.sum()

        if best_sc < best_score:
            best_score = best_sc
            best_a = a_curr.copy()
            if it < 20 or it % 20 == 0:
                print(f"  LP-FW iter {it}: score={_score(a_curr):.10f}, t={best_t:.3f}")

    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 2000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        m_buf *= 0.5
        v_buf *= 0.5

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _subgrad_method(a_init, deadline, n_steps=50000):
    """Subgradient method using exact active set."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    step_init = 0.05 / n
    momentum = 0.9
    v = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        M = float(c.max())
        # Find active set (within small tolerance)
        tol = 0.001 * M
        active = c >= (M - tol)
        if not np.any(active):
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
        w = active.astype(float) / active.sum()

        grad = _compute_grad_conv(a_curr, w)
        # Project to stay on simplex: subtract mean weighted by a
        mean_grad = float(np.dot(grad, a_curr))
        grad_centered = grad - mean_grad

        step_size = step_init / (1.0 + 0.01 * step)
        v = momentum * v + step_size * grad_centered
        a_curr = a_curr + v
        a_curr = np.maximum(a_curr, 0.0)
        s = a_curr.sum()
        if s < 1e-15:
            a_curr = best_a.copy()
            break
        a_curr = a_curr / s

        if step % 50 == 0:
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    return best_a, best_score

def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Coordinate descent polishing."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            c = _full_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                a[i] = 1e-5
                a = a / a.sum()
                continue

            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            idx_top = np.argsort(c)[-top_k_w:]
            t_candidates = [0.0, a_i_old * 0.3, a_i_old * 0.5, a_i_old * 0.7,
                           a_i_old, a_i_old * 1.3, a_i_old * 1.5, a_i_old * 2.0]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 5.0 * max(a_i_old, 0.01):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.8))
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 7]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        edge = max(2, n // 32)
        a[:edge] = np.linspace(0.3, 1.0, edge)
        a[-edge:] = np.linspace(1.0, 0.3, edge)
        return a / a.sum()
    else:
        return np.ones(n) / n

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Primary optimization at parent length
    n_main = n_parent
    a_work = parent.copy()
    a_work = a_work / a_work.sum()
    sc_work = _score(a_work)
    print(f"Working sequence: n={n_main}, score={sc_work:.10f}")

    # Phase 1: LP-based Frank-Wolfe (try for a while)
    phase1_deadline = min(t_start + 300, deadline - 200)
    if time.monotonic() < phase1_deadline:
        print(f"\n=== Phase 1: LP-based Frank-Wolfe (budget: {phase1_deadline - time.monotonic():.0f}s) ===")
        a_lp, sc_lp = _lp_frank_wolfe(a_work, phase1_deadline, max_iters=200)
        sc_lp_full = _score(a_lp)
        if sc_lp_full < best_score:
            best_score = sc_lp_full
            best_seq = a_lp.copy()
            print(f"  LP-FW result: score={sc_lp_full:.10f}")
        a_work = a_lp

    # Phase 2: Adam log-space with various beta schedules
    phase2_deadline = min(deadline - 150, time.monotonic() + 400)
    if time.monotonic() < phase2_deadline:
        print(f"\n=== Phase 2: Adam log-space (budget: {phase2_deadline - time.monotonic():.0f}s) ===")
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())

        # Coarse-to-fine beta schedule
        betas = []
        beta = 0.01 * max_cc
        while beta > 0.000005 * max_cc:
            betas.append(beta)
            beta *= 0.5
        if len(betas) > 10:
            betas = betas[:10]

        a_opt, sc_opt = _adam_logspace(a_work, phase2_deadline, betas,
                                       lr_init=0.015, top_k=150, max_iter=200000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam result: score={sc_opt_full:.10f}")
        a_work = a_opt

    # Phase 3: Subgradient with exact active set
    phase3_deadline = min(deadline - 100, time.monotonic() + 150)
    if time.monotonic() < phase3_deadline:
        print(f"\n=== Phase 3: Subgradient method (budget: {phase3_deadline - time.monotonic():.0f}s) ===")
        a_sg, sc_sg = _subgrad_method(a_work, phase3_deadline, n_steps=30000)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            print(f"  Subgradient result: score={sc_sg_full:.10f}")
        a_work = a_sg

    # Phase 4: Fine Adam polish
    phase4_deadline = min(deadline - 60, time.monotonic() + 200)
    if time.monotonic() < phase4_deadline:
        print(f"\n=== Phase 4: Fine Adam polish (budget: {phase4_deadline - time.monotonic():.0f}s) ===")
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas_fine = []
        beta = 0.001 * max_cc
        while beta > 0.000001 * max_cc:
            betas_fine.append(beta)
            beta *= 0.5
        if len(betas_fine) > 8:
            betas_fine = betas_fine[:8]

        a_fine, sc_fine = _adam_logspace(a_work, phase4_deadline, betas_fine,
                                         lr_init=0.005, top_k=80, max_iter=100000)
        sc_fine_full = _score(a_fine)
        if sc_fine_full < best_score:
            best_score = sc_fine_full
            best_seq = a_fine.copy()
            print(f"  Fine Adam result: score={sc_fine_full:.10f}")
        a_work = a_fine

    # Phase 5: Coordinate descent
    phase5_deadline = min(deadline - 30, time.monotonic() + 120)
    if time.monotonic() < phase5_deadline:
        print(f"\n=== Phase 5: Coordinate descent (budget: {phase5_deadline - time.monotonic():.0f}s) ===")
        a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=3, top_k_w=60,
                                            deadline=phase5_deadline)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  CoordDescend result: score={sc_cd_full:.10f}")
        a_work = a_cd

    # Phase 6: Try larger length
    n_large = 8192
    if time.monotonic() < deadline - 120 and best_seq is not None:
        print(f"\n=== Phase 6: Try n={n_large} (budget: {deadline - time.monotonic() - 120:.0f}s) ===")
        a_large = _interp_profile(best_seq, n_large)
        sc_large_init = _score(a_large)
        print(f"  Interpolated init: score={sc_large_init:.10f}")
        if sc_large_init < best_score:
            best_score = sc_large_init
            best_seq = a_large.copy()

        # Quick Adam on larger sequence
        phase6_deadline = min(deadline - 60, time.monotonic() + 200)
        c_l = _full_convolve(a_large, a_large)
        max_cl = float(c_l.max())
        betas_l = [0.005 * max_cl, 0.002 * max_cl, 0.001 * max_cl, 0.0005 * max_cl]
        a_l_opt, sc_l = _adam_logspace(a_large, phase6_deadline, betas_l,
                                       lr_init=0.01, top_k=200, max_iter=50000)
        sc_l_full = _score(a_l_opt)
        if sc_l_full < best_score:
            best_score = sc_l_full
            best_seq = a_l_opt.copy()
            print(f"  Large n result: score={sc_l_full:.10f}")

    # Phase 7: Diversity restarts on best
    if time.monotonic() < deadline - 40 and best_seq is not None:
        remaining = deadline - time.monotonic() - 40
        n_restarts = min(3, int(remaining // 40))
        if n_restarts > 0:
            print(f"\n=== Phase 7: {n_restarts} diversity restarts ===")
            n_best = len(best_seq)
            a_base = best_seq.copy() / best_seq.sum()

            for ri in range(n_restarts):
                if time.monotonic() > deadline - 40:
                    break
                per_restart = (deadline - time.monotonic() - 30) / (n_restarts - ri)

                if ri == 0:
                    p_start = a_base + 0.01 * rng.standard_normal(n_best)
                elif ri == 1:
                    shift = rng.integers(1, max(2, n_best // 512))
                    p_start = np.roll(a_base, shift).copy()
                else:
                    freq = rng.uniform(0.3, 5.0)
                    pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                    p_start = a_base + pert

                p_start = np.maximum(p_start, 1e-12)
                p_start = p_start / p_start.sum()

                c_s = _full_convolve(p_start, p_start)
                max_cs = float(c_s.max())
                betas_r = [0.003 * max_cs, 0.001 * max_cs, 0.0005 * max_cs]
                p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                                 betas_r, lr_init=0.008, top_k=100,
                                                 max_iter=30000)
                sc_full = _score(p_best)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = p_best.copy()
                    print(f"  Restart {ri}: score={sc_full:.10f}")

    # Final evaluation
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        else:
            # Scale to a reasonable range
            final = final * (1.0 / s)  # normalize to sum=1
        sc_final = _score(final)
        print(f"FINAL RETURNED: score={sc_final:.10f}, n={len(final)}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
