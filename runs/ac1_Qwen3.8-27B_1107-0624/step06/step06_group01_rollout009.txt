"""
Optimized step-function height sequence minimizer.
Uses multi-stage FISTA with top-k focused smoothing, spectral initialization,
peak-targeted refinement, and adaptive length selection to minimize
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements over prior approach: sharper peak focusing, backtracking
step sizes, spectral shaping initializations, and peak-targeted polish.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return _score_norm(a_norm)


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_softmax_weights(c, k, beta):
    """Compute weights focusing on top-k peaks with temperature beta.
    Returns a probability vector w over the 2n-1 convolution values."""
    c_max = float(c.max())
    # Use a "sharp" softmax that emphasizes the top values
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _hinge_smoothing(c, k, mu):
    """Smooth max using a hinge/soft-top-k approach.
    Returns (smoothed_value, weights) where weights are on the convolution indices."""
    c_max = float(c.max())
    # Soft threshold: values within mu of max get weight
    threshold = c_max - mu
    active = c >= threshold
    if not np.any(active):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    # Within the active set, use softmax
    c_active = np.where(active, c - c_max, 0.0) / (mu + 1e-15)
    e = np.exp(c_active)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return c_max, w


def _fista_topk(a_init, deadline, beta_schedule, k_list):
    """FISTA with top-k focused smoothing.
    For each beta in beta_schedule, use the corresponding k from k_list."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    best_score = float(2.0 * n * max_c)
    best_a = a_curr.copy()
    step = 1.0 / (4.0 * n * max_c + 1e-15)

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for k_focus in k_list:
            if time.monotonic() > deadline:
                break
            # Use the smoothing parameter
            mu = beta
            for _ in range(2000):
                if time.monotonic() > deadline:
                    break
                t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
                y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
                y = np.maximum(y, 0.0)
                s_y = y.sum()
                if s_y > 1e-30:
                    y = y / s_y
                else:
                    y = np.ones(n) / n

                c_y = _full_convolve(y, y)
                c_max_y = float(c_y.max())

                # Compute weights focusing on top peaks
                w = _topk_softmax_weights(c_y, k_focus, mu)

                grad = _compute_grad(y, w)
                a_next = _proj_simplex(y - step * grad)

                # Adaptive step update
                if _full_convolve(a_next, a_next).max() > max_c:
                    step *= 0.8
                    a_next = _proj_simplex(y - step * grad)

                c_new = _full_convolve(a_next, a_next)
                max_new = float(c_new.max())
                sc = float(2.0 * n * max_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

                # Update Lipschitz estimate
                L = 4.0 * n * max_new / mu
                step = 1.0 / L

                a_prev = a_curr.copy()
                a_curr = a_next
                t_val = t_next

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while True:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_targeted_polish(a, deadline):
    """Directly target and reduce the top few peaks of the convolution.
    Uses coordinate updates focused on coordinates that contribute to peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(77)
    c = _full_convolve(best_a, best_a)

    while time.monotonic() < deadline:
        # Find the top peaks
        top_indices = np.argsort(c)[::-1][:min(10, len(c) // 2)]
        c_max = float(c[top_indices[0]])

        # For each peak, find which coordinates contribute most
        improved = False
        for k in top_indices[:5]:
            # Compute contribution of each a_j to c_k
            # c_k = sum_j a_j * a_{k-j}
            contributions = np.zeros(n)
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    contributions[j] = a[j] * a[m]
                    if 2 * j == k:
                        contributions[j] += a[j] * a[m]

            # Find the coordinate with largest contribution
            j_max = int(np.argmax(contributions))
            # Try reducing a[j_max] slightly and compensating elsewhere
            # (to keep sum constant, we reduce a[j_max] and increase a random other)
            for _ in range(50):
                if time.monotonic() > deadline:
                    break
                j_red = j_max if rng.random() < 0.7 else int(rng.integers(0, n))
                j_inc = int(rng.integers(0, n))
                if j_red == j_inc:
                    continue
                delta = best_a[j_red] * 0.01
                if delta < 1e-15:
                    continue
                a_try = best_a.copy()
                a_try[j_red] -= delta
                a_try[j_inc] += delta
                if a_try[j_red] < 0:
                    continue
                s = a_try.sum()
                a_try = a_try / s
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(2.0 * n * c_try.max())
                if sc_try < best_score - 1e-10:
                    best_score = sc_try
                    best_a = a_try.copy()
                    c = c_try
                    improved = True
                    break

        if not improved:
            # Try random perturbations
            for _ in range(200):
                if time.monotonic() > deadline:
                    break
                sigma = 0.005
                z = np.log(np.maximum(best_a, 1e-15))
                z += sigma * rng.standard_normal(n)
                z = z - z.mean()
                a_try = _softmax_compute(z)
                sc_try = _score_norm(a_try)
                if sc_try < best_score - 1e-10:
                    best_score = sc_try
                    best_a = a_try.copy()
                    c = _full_convolve(best_a, best_a)

    return best_a, best_score


def _spectral_init(n, rng):
    """Initialize using spectral methods to get near-flat autocorrelation.
    Use random phases in frequency domain, then enforce non-negativity."""
    N = _next_pow2(2 * n - 1)
    # Create a flat-magnitude spectrum with random phases
    mag = np.ones(N // 2 + 1)
    phase = rng.random(N // 2 + 1) * 2 * np.pi
    # Try a few different magnitude profiles
    best_a = None
    best_sc = np.inf
    for trial in range(20):
        freq = np.fft.rfftfreq(N)
        # Various magnitude profiles
        if trial == 0:
            mag_t = np.ones(N // 2 + 1)
        elif trial == 1:
            mag_t = 1.0 + 0.3 * np.cos(2 * np.pi * freq * N * rng.random())
        elif trial == 2:
            mag_t = np.exp(-0.5 * (freq * N * 0.3) ** 2)
        else:
            mag_t = np.ones(N // 2 + 1) * (1.0 + 0.2 * rng.random(N // 2 + 1))

        ph_t = rng.random(N // 2 + 1) * 2 * np.pi
        spec = mag_t * np.exp(1j * ph_t)
        # Ensure Hermitian symmetry for real signal
        a_full = np.fft.irfft(spec, N)
        a_try = a_full[:n]
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s > 0.1 * n * mag_t[0]:
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
    if best_a is None:
        best_a = np.ones(n) / n
    return best_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 - np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "chirp_envelope":
        # Envelope of a chirp - should give good spectral flatness
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.sin(4 * np.pi * t) * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

                    betas_fine.append(beta)
                    beta *= 0.5
                k_list_fine = [2, 3, 5]

                phase2_deadline = min(deadline - 80, time.monotonic() + max(10, time_left * 0.2))
                a_opt2, sc2 = _fista_topk(a_work2, phase2_deadline, betas_fine, k_list_fine)
                sc2_full = _score(a_opt2)
                if sc2_full < best_score_norm_for_n:
                    best_score_norm_for_n = sc2_full
                    best_for_n = a_opt2.copy()

            # Phase 3: Adam log-space
            if time.monotonic() < deadline - 80:
                a_work3 = best_for_n.copy() / best_for_n.sum()
                c2 = _full_convolve(a_work3, a_work3)
                max_c2 = float(c2.max())
                betas_adam = []
                beta = 0.002 * max_c2
                for _ in range(8):
                    betas_adam.append(beta)
                    beta *= 0.5

                adam_deadline = min(deadline - 80, time.monotonic() + max(10, time_left * 0.25))
                a_opt3, sc3 = _adam_logspace(a_work3, adam_deadline, betas_adam,
                                             lr_init=0.012, max_iter_total=200000)
                sc3_full = _score(a_opt3)
                if sc3_full < best_score_norm_for_n:
                    best_score_norm_for_n = sc3_full
                    best_for_n = a_opt3.copy()

            if init_idx == 0:
                print(f"  Init 0 (main): best_score_norm={best_score_norm_for_n:.8f}")

        # Update global best
        if best_score_norm_for_n < best_score:
            best_score = best_score_norm_for_n
            best_seq = best_for_n.copy()
            full_score = _score(best_seq)
            print(f"  New global best: score={full_score:.10f}, n={n_target}")

    # Peak-targeted polish on the best found
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Peak-targeted polish (score={_score(best_seq):.10f}) ===")
        polish_deadline = min(deadline - 15, time.monotonic() + 60)
        a_pol, sc_pol = _peak_targeted_polish(best_seq, polish_deadline)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  After polish: score={sc_pol_full:.10f}")

    # Multi-restart with different strategies
    if best_seq is not None and time.monotonic() < deadline - 40:
        print(f"\n=== Multi-restart (score={_score(best_seq):.10f}) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        time_per = (deadline - time.monotonic() - 20.0) / 4

        for i in range(4):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(a_base, 0.015, rng)
            elif i == 1:
                # Spectral init at this length
                p_start = _spectral_init(n_best, rng)
            elif i == 2:
                # Blend with raised cosine
                alt = _make_profile(n_best, "raised_cosine")
                p_start = 0.8 * a_base + 0.2 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # Larger perturbation
                p_start = _perturb(a_base, 0.04, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0004 * max_cs, 0.00015 * max_cs, 0.00005 * max_cs]
            dl = min(deadline - 20, time.monotonic() + max(5, time_per))
            p_best, sc_best = _adam_logspace(p_start, dl, bs, lr_init=0.006, max_iter_total=100000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f}")

    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final ultra-fine polish ===")
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00002 * max_cp, 0.000005 * max_cp, 0.000001 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(a_pol, time.monotonic() + per_t,
                                           betas_ult, lr_init=0.001, max_iter_total=80000)
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f}")

    # Final validation
    final_score = _score(best_seq) if best_seq is not None else np.inf
    print(f"\n=== FINAL: score={final_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        print(f"FINAL: score={_score(final):.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
