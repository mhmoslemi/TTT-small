"""
Optimized step function sequence generator using spectral initialization, 
FISTA coarse optimization, L-BFGS fine-tuning, and coordinate descent polish.
Targets the minimum of 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_optimize(a_init, deadline, max_iter=50000):
    """FISTA with adaptive beta for softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    beta = 0.02 * max_c
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: 2 * convolve(w, y_reversed) centered
        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        a_next = _proj_simplex(y - step * grad)

                best_a = a_next.copy()

            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _lbgfs_refine(a_init, deadline, beta):
    """Use L-BFGS-B in log-space for fine optimization."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    # Precompute for efficiency
    n_conv = 2 * n - 1
    
    def objective_and_grad(z):
        a = _softmax_safe(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Objective: 2n * (smoothed max)
        obj = 2.0 * n * (c_max + beta * np.log(w_sum))
        
        # Gradient with respect to a
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        
        # Chain rule: grad_z = diag(a) @ (grad_a - sum(grad_a * a) * 1)
        # Actually for softmax: d(softmax(z))/dz = diag(a) - outer(a,a)
        # So grad_z = a * (grad_a - dot(grad_a, a))
        grad_z = a * (grad_a - float(np.dot(grad_a, a)))
        
        return obj, grad_z
    
    # Run L-BFGS-B
    result = scipy_minimize(
        objective_and_grad,
        z0,
        method='L-BFGS-B',
        jac=True,
        options={
            'maxiter': 5000,
            'maxfun': 5000,
            'ftol': 1e-12,
            'gtol': 1e-8,
            'maxls': 50,
        }
    )
    
    z_best = result.x
    a_best = _softmax_safe(z_best)
    return a_best

def _make_spectral_init(n, rng, n_modes=5):
    """Create initialization with anti-symmetric spectral perturbations."""
    base = np.ones(n) / n
    for _ in range(n_modes):
        k = rng.integers(n//4 + 1, 3*n//4)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.01, 0.05)
        base += amp * np.cos(2*np.pi*k*np.arange(n)/n + phase)
    base = np.maximum(base, 1e-10)
    return base / base.sum()
