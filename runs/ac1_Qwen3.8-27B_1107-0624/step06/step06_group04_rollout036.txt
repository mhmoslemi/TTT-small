"""
Multi-strategy optimizer combining LP-based iterations, coordinate descent,
FISTA with softmax smoothing, and Adam log-space fine-tuning.
Key new idea: LP-based projected steps that find the best g maximizing sum(g)
subject to ||f*g||_inf <= M, providing a theoretically motivated update direction.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_simplex(a, w):
    """Gradient of sum_w * (a*a) on the simplex, where w is a distribution over lag indices."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _lp_step(f, M, n):
    """
    Solve LP: max sum(g) s.t. (f*g)_j <= M for all j, g >= 0.
    Returns the optimal g (unnormalized).
    """
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)_j <= M for j = 0, ..., 2n-2
    # (f*g)_j = sum_k f_k * g_{j-k} where g_k = 0 for k outside [0, n-1]
    # For a given j, g_k appears with coefficient f_{j-k} (if 0 <= j-k < n)
    
    # Build constraint matrix A_ub (sparse would be better but let's try dense for n <= 8192)
    # A_ub has shape (2n-1, n)
    # A_ub[j, k] = f[j-k] if 0 <= j-k < n, else 0
    
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2*n - 1, n))
    for j in range(2*n - 1):
        # g_k appears in (f*g)_j with coefficient f[j-k] if 0 <= j-k < n
        # So k must satisfy: 0 <= j-k < n => j-n < k <= j
        # And also 0 <= k < n
        k_min = max(0, j - n + 1)
        k_max = min(n - 1, j)
        if k_min <= k_max:
            ks = np.arange(k_min, k_max + 1)
            A_ub[j, ks] = f[j - ks]
    
    b_ub = np.full(2*n - 1, M)
    
    # Bounds: g_i >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return np.maximum(result.x, 0)
    else:
        return f.copy()  # fallback

def _lp_iteration(f, n_max_lp=8, deadline=None):
    """
    Perform LP-based iterations:
    1. Compute M = max(f*f)
    2. Solve LP to find g maximizing sum(g) s.t. f*g <= M
    3. Update f <- (1-t)f + t*g/sum(g)
    """
    n = len(f)
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score_norm(f)
    
    t = 0.1  # mixing parameter
    
    for it in range(n_max_lp):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(f, f)
        M = float(c.max())
        
        # Solve LP
        g = _lp_step(f, M, n)
        s_g = float(g.sum())
        if s_g < 1e-10:
            break
        g = g / s_g
        
        # Check if g is better
        score_g = _score_norm(g)
        
        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new > 0:
            f_new = f_new / s_new
        
        score_new = _score_norm(f_new)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        
        f = f_new
        
        # Decrease t gradually
        t = max(0.01, t * 0.8)
    
    return best_f, best_score

def _coordinate_descent(a, n_sweeps=5, deadline=None):

    Coordinate descent: for each coordinate i, optimize a_i while keeping others fixed.
    The objective as a function of a_i (with sum normalized) is:

    n = len(a)
    a = a.copy() / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

        if deadline is not None and time.monotonic() > deadline:
            break
        
        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            
            old_val = a[i]
            if old_val < 1e-10:
                continue

            # The optimal a_i should be in [0, old_val * 3] or so
            candidates = [old_val * f for f in [0.1, 0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]]
            candidates.append(0.0)
            
            S_other = a.sum() - old_val
            best_local = _score_norm(a)
            best_local_val = old_val
            
            for new_val in candidates:
                a[i] = new_val

                a_norm = a / s

def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, lr_decay_every=10000, lr_decay_factor=0.97, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0005, current_lr * lr_decay_factor)

            if no_improve_count > 40 and t > 500:
                current_lr = max(0.0005, current_lr * 0.9)
                if current_lr < 0.001:
                    break

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_steps(a_init, n_steps=3000, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    
    step_size = 1.0

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Use top-k approach for the subgradient
        top_k = min(50, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w_vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals ** 2
        w_vec = w_vec / w_vec.sum()

        grad = _compute_grad_simplex(a, w_vec)
        
        a_new = _proj_simplex(a - step_size * grad)
        sc = _score_norm(a_new)
        
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.9
            if step_size < 1e-6:
                step_size = 0.5
                # Try a random restart of step size
                a_new = _proj_simplex(a - step_size * grad)
                if _score_norm(a_new) < _score_norm(a):
                    a = a_new

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    n = len(a_init)
    a = a_init.copy()

    s = a.sum()

        mag = np.abs(A)
        avg_mag = mag.mean()
        A_flat = A / (mag + 1e-30) * avg_mag

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _block_descent(a, n_blocks=100, deadline=None, max_rounds=5):
    """
    Block coordinate descent: divide the sequence into blocks and optimize each block
    using a small local FISTA or direct optimization.
    """
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    block_size = max(1, n // n_blocks)
    n_actual_blocks = (n + block_size - 1) // block_size
    
    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        for b in range(n_actual_blocks):
            if deadline is not None and time.monotonic() > deadline:
                break
            
            start = b * block_size
            end = min(start + block_size, n)
            
            # Try scaling the block
            block = a[start:end].copy()
            s_block = block.sum()
            if s_block < 1e-10:
                continue
            
            # Try different scalings of the block
            best_local_scale = 1.0
            best_local_score = _score_norm(a)
            
            for scale in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
                a[start:end] = block * scale
                sc = _score_norm(a / a.sum())
                a[start:end] = block
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_scale = scale
            
            if best_local_scale != 1.0:
                a[start:end] = block * best_local_scale
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_a, best_score

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key new ideas: LP-based feasibility-preserving steps (guaranteed descent),
coordinate descent polishing, and adaptive multi-phase strategy.
The LP step finds g maximizing sum(g) s.t. (f*g)_j <= M for all j,
providing a theoretically grounded update that never increases the peak.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

        vals.append(f[j - ks])

    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return f.copy()


def _lp_iteration(f, n_max_lp=20, deadline=None, t_init=0.15):
    """LP-based iterations: find g via LP, update f <- (1-t)
    best_f = f.copy()
    best_score = _score_norm(f)
    n = len(f)
    t = t_init

    for it in range(n_max_lp):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(f, f)
        M = float(c.max())

        g = _lp_step_single(f, M, n)
        s_g = float(g.sum())
        if s_g < 1e-10:
            break
        g = g / s_g

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-10:
            break
        f_new = f_new / s_new

        score_new = _score_norm(f_new)
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Try smaller t
            t = max(0.02, t * 0.7)
            f_new2 = (1.0 - t) * f + t * g
            f_new2 = np.maximum(f_new2, 0.0)
            s_new2 = f_new2.sum()
            if s_new2 > 1e-10:
                f_new2 = f_new2 / s_new2
                score_new2 = _score_norm(f_new2)
                if score_new2 < best_score:
                    best_score = score_new2
                    best_f = f_new2.copy()
                    f = f_new2

        # Anneal t
        t = max(0.05, t_init * (0.85 ** it))

    return best_f, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=10000, lr_decay_factor=0.97, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0005, current_lr * lr_decay_factor)

            if no_improve_count > 40 and t > 500:
                current_lr = max(0.0005, current_lr * 0.9)
                if current_lr < 0.001:
                    break

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_exact(a_init, n_steps=5000, deadline=None):
    """Subgradient method using exact max (top-k smoothing)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_size = 0.3

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Top-k smoothing for better gradient approximation
        top_k = min(100, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w_vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals ** 2
        ws = w_vec.sum()
        if ws > 0:
            w_vec = w_vec / ws
        else:
            w_vec[top_idx[0]] = 1.0

        grad = _compute_grad_simplex(a, w_vec)
        a_new = _proj_simplex(a - step_size * grad)
        sc = _score_norm(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            step_size = min(1.0, step_size * 1.05)
        else:
            step_size *= 0.92
            if step_size < 0.01:
                # Reboot
                step_size = 0.3
                a_new = _proj_simplex(a - step_size * grad)
                if _score_norm(a_new) < _score_norm(a):
                    a = a_new

    return best_a, best_score


def _coordinate_descent_scaled(a, n_sweeps=3, deadline=None):
    """Block coordinate descent: optimize scaling of random blocks."""
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    block_size = max(8, n // 256)
    n_blocks = (n + block_size - 1) // block_size

    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break

        block_order = rng.permutation(n_blocks)
        for b in block_order:
            if deadline is not None and time.monotonic() > deadline:
                break

            start = b * block_size
            end = min(start + block_size, n)
            if start >= n:
                continue

            block = a[start:end].copy()
            s_block = block.sum()
            if s_block < 1e-10:
                continue

            best_local_score = _score_norm(a)
            best_scale = 1.0

            # Try a range of scales
            for scale in [0.2, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0, 5.0]:
                a[start:end] = block * scale
                s_total = a.sum()
                if s_total < 1e-10:
                    a[start:end] = block
                    continue
                sc = _score_norm(a / s_total)
                if sc < best_local_score:
                    best_local_score = sc
                    best_scale = scale
            a[start:end] = block

            if best_scale != 1.0:
                a[start:end] = block * best_scale
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=300, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
