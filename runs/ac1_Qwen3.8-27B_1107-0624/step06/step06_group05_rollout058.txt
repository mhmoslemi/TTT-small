def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)

    rho = np.where(cond)[0][-1]

    return np.maximum(v - theta, 0.0)

        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)

        t_val = t_next
    
    sc_final = _score_norm(a_curr)

def _adam_soft(a_init, beta, deadline, lr_init=0.005, max_iter=100000):
    """Adam on log-sum-exp smoothed objective, working in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * h  # gradient of 2n * soft_max
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        if t % 100 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        
        if it % 5000 == 0:
            current_lr = max(0.0001, current_lr * 0.9)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_sequence(n, profile="uniform", rng=None):
    if profile == "uniform":
        return np.ones(n) / n
    elif profile == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif profile == "cosine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif profile == "exponential":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        return a / a.sum()
    elif profile == "random":
        if rng is None:
            rng = np.random.default_rng(42)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: direct subgradient with Nesterov acceleration, FISTA with log-sum-exp smoothing,
Adam in log-space, multi-length search, and diverse restarts.
Key improvements over previous: direct (unsmoothed) subgradient, multiple n values,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-5.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "exponential":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _grad_from_weights(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] with respect to a on simplex."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _nesterov_direct(a_init, deadline, max_iter=200000, tol_frac=1e-9):
    """Nesterov accelerated direct subgradient on the simplex.
    Uses adaptive step size based on observed progress."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0

    f_curr = _score_norm(a_curr)
    best_a = a_curr.copy()
    best_f = f_curr

    # Lower bound: 2n/(2n-1)
    lb = 2.0 * n / (2.0 * n - 1.0)
    # Use a tighter practical lower bound
    lb_eff = min(best_f, 1.49)

    # Adaptive step
    step = (f_curr - lb_eff) / (4.0 * n)
    step = min(step, 1.0 / (2.0 * n) * 10.0)
    no_improve_count = 0
    patience = 2000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        f_y = 2.0 * n * c_max

        # Subgradient: concentrate on near-max positions
        tol = max(tol_frac * c_max, 1e-12)
        active = np.where(c_y >= c_max - tol)[0]
        w = np.zeros_like(c_y)
        w[active] = 1.0 / len(active)

        grad = _grad_from_weights(y, w)
        g_norm_sq = float(np.dot(grad, grad))

        if g_norm_sq < 1e-30:
            break

        # Polyak-like step with damping
        polyak_step = 0.5 * (f_y - lb_eff) / g_norm_sq
        polyak_step = min(polyak_step, step)

        a_next = _proj_simplex(y - polyak_step * grad)

        # Track progress
        if it % 200 == 0:
            f_next = _score_norm(a_next)
            if f_next < best_f - 1e-10:
                best_f = f_next
                best_a = a_next.copy()
                no_improve_count = 0
                # Tighten the effective lower bound
                lb_eff = max(lb, best_f - 0.01)
            else:
                no_improve_count += 200

            if no_improve_count >= patience:
                step *= 0.7
                no_improve_count = 0

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    f_final = _score_norm(a_curr)
    if f_final < best_f:
        best_f = f_final
        best_a = a_curr.copy()

    return best_a, best_f


def _fista_soft(a_init, beta, deadline, max_iter=50000):
    """FISTA on log-sum-exp smoothed objective on simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    if max_c < 1e-15:
        max_c = 1.0 / n
    L = 4.0 * n / max(beta, 1e-15)
    if L > 1e10:
        L = 1e10
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())

        shifted = (c_y - c_max) / beta
        e = np.exp(shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _grad_from_weights(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, beta, deadline, lr_init=0.004, max_iter=100000):
    """Adam optimizer in log-space on the smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e_vals = np.exp((c - cm) / beta)
        s = float(e_vals.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e_vals / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * h

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.4:
            update = update * (0.4 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()

        if t % 150 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if it % 4000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lbfgs_refine(a_init, deadline, beta):
    """Scipy L-BFGS-B refinement in log-space."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e_vals = np.exp(c_sh)
        s = float(e_vals.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e_vals = np.exp(c_sh)
        s = float(e_vals.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e_vals / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        a_init_norm = a_init / a_init.sum()
        return a_init_norm, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 40)), 'gtol': 1e-13, 'maxls': 12}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score

            elif i == 3:

                n_r = n_work
                p = n_r // 32 if n_r % 32 == 0 else n_r // 31
                m = n_r // p
                b = _make_profile(p, "cosine", rng)
                a_r = np.tile(b, m)[:n_r]
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
            elif i == 4:
                # Shift
                shift = rng.integers(1, max(2, n_work // 512))
                a_r = np.roll(best_seq, shift)
                a_r = a_r / a_r.sum()
            elif i == 5:
                # Two-scale: blend coarse and fine
                n_r = n_work
                a_coarse = _make_profile(n_r // 8, "cosine", rng)
                a_coarse_up = np.interp(
                    np.linspace(0, 1, n_r),
                    np.linspace(0, 1, len(a_coarse)),
                    a_coarse
                )
                a_fine = 0.5 + 0.5 * rng.random(n_r)
                a_r = 0.5 * a_coarse_up + 0.5 * a_fine
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
            elif i == 6:
                # Power iteration style: start from best, apply inverse gradient
                a_r = best_seq.copy()
                c_r = _full_convolve(a_r, a_r)
                # Decrease where convolution is high, increase where low
                a_r = a_r * (1.0 + 0.05 * (c_r[n_work - 1:2 * n_work - 1] - c_r.max()))
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()
            else:
                # Random with structure
                n_r = n_work
                a_r = rng.random(n_r)
                # Apply smoothing
                for _ in range(3):
                    a_r = 0.5 * a_r + 0.25 * np.roll(a_r, 1) + 0.25 * np.roll(a_r, -1)
                a_r = np.maximum(a_r, 1e-12)
                a_r = a_r / a_r.sum()

            # Optimize the restart
            c_r = _full_convolve(a_r, a_r)
            max_cr = float(c_r.max())
            beta_r = 0.0005 * max_cr
            t_r_end = time.monotonic() + max(3.0, per_r * 0.5)
            a_r, _ = _fista_soft(a_r, beta_r, t_r_end, max_iter=20000)
            a_r = a_r / a_r.sum()

            # Adam polish
            if time.monotonic() < phase5_end - 10:
                beta_r2 = 0.00005 * max_cr
                t_r2_end = time.monotonic() + max(2.0, per_r * 0.4)
                a_r, _ = _adam_logspace(a_r, beta_r2, t_r2_end, lr_init=0.002, max_iter=30000)
                a_r = a_r / a_r.sum()

            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {i} (n={len(a_r)}): score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i} (n={len(a_r)}): score={sc_r:.10f} (no improvement)")

    # === Phase 6: Final ultra-fine polish ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            # Ultra-fine Adam
            t_end = time.monotonic() + remaining * 0.5
            a_pol, _ = _adam_logspace(a_pol, 0.000002 * max_cp, t_end, lr_init=0.0008, max_iter=100000)
            a_pol = a_pol / a_pol.sum()
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")

            # Direct Nesterov on remaining
            if time.monotonic() < deadline - 8:
                a_pol, _ = _nesterov_direct(a_pol, deadline - 3.0, max_iter=200000, tol_frac=1e-11)
                a_pol = a_pol / a_pol.sum()
                sc_p = _score(a_pol)
                if sc_p < best_score:
                    best_score = sc_p
                    best_seq = a_pol.copy()
                    print(f"  Final Nesterov: score={sc_p:.10f}")

            # L-BFGS if time permits
            if time.monotonic() < deadline - 6:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000001 * float(c_f.max())
                a_pol, _ = _lbfgs_refine(a_pol, deadline - 2.0, b_f)
                a_pol = a_pol / a_pol.sum()
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"\nDone: best_score={best_score:.10f}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(4096, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
