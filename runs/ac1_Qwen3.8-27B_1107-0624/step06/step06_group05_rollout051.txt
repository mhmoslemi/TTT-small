"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive FISTA, alternating subgradient Adam, L-BFGS polish, coordinate descent,
and multi-length exploration to beat the previous bound of 1.505780.

import numpy as np
import time
import sys

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_from_w(a, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _logsumexp_smooth(c, beta):
    cm = float(c.max())
    if beta < 1e-15:
        return cm, cm
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        return cm, cm
    return cm + beta * np.log(s), cm

def _fista_adaptive(a_init, beta_start, deadline, max_iter, beta_decay=0.5, n_phases=20):

    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    beta = beta_start
    per_phase_iter = max(100, max_iter // n_phases)
    
    for phase in range(n_phases):
        if time.monotonic() > deadline - 1:
            break
        L = 4.0 * n / max(beta, 1e-30)
        step = 1.0 / L
        for it in range(per_phase_iter):
            if time.monotonic() > deadline - 1:
                break
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:

            else:

            c_y = _full_convolve(y, y)
            sm, cm = _logsumexp_smooth(c_y, beta)
            if beta > 1e-10:

def _adam_alternating(a_init, deadline, beta, lr_init, max_iter, alpha_switch=0.5):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline - 0.5:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        if (it % 10) < int(10 * alpha_switch):
            # Softmax-weighted subgradient
            e = np.exp((c - cm) / beta)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            grad_w = 1.0
        else:
            # Exact subgradient (uniform over active set)
            tol = max(1e-9 * cm, 1e-12)
            active = np.where(c >= cm - tol)[0]
            w = np.zeros_like(c)
            w[active] = 1.0 / len(active)
            grad_w = 1.0
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * grad_w * (h - W * a)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

            a_chk = _softmax_compute(z)

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

        sm, _ = _logsumexp_smooth(c, beta)
        return 2.0 * n * sm
    
    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n * h - 4.0 * n * W * a
        return a * (grad_a - np.dot(a, grad_a))
    
    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 0.5:
        return a_init, best_score
    
    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 40)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _coordinate_descent(a_init, deadline, max_rounds=50):
    """Local coordinate descent: try small changes to individual coordinates."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 2:
            break
        improved = False
        # Random subset of coordinates to try
        n_try = max(10, n // 50)
        indices = rng.choice(n, size=n_try, replace=False)
        for idx in indices:
            if time.monotonic() > deadline - 2:
                break
            # Try a few perturbation sizes
            for delta in [0.001, 0.005, 0.01, 0.02]:
                a_try = a.copy()
                a_try[idx] -= delta * a[idx]
                if a_try[idx] < 0:
                    continue
                # Redistribute to random other coordinates
                others = rng.choice([j for j in range(n) if j != idx], size=min(10, n-1), replace=False)
                redist = delta * a[idx] / len(others)
                for o in others:
                    a_try[o] += redist
                a_try = a_try / a_try.sum()
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        if not improved and rnd > 10:
            break
    
    return best_a, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.25)**2 / 0.1) + np.exp(-4.0 * (t - 0.75)**2 / 0.1) + 0.01
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def _interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    n = len(a)
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s > 0:
        new_a = new_a / s
    else:
        new_a = np.ones(new_n) / new_n
    return new_a

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    
    a_work = a0.copy()
    
    # === Phase 1: FISTA adaptive continuation ===
    print("Phase 1: FISTA adaptive beta continuation")
    phase1_end = min(deadline, t_start + 350.0)
    if time.monotonic() < phase1_end - 5:
        c_curr = _full_convolve(a_work, a_work)
        max_c0 = float(c_curr.max())
        beta_start = 0.05 * max_c0
        a_work, _ = _fista_adaptive(a_work, beta_start, phase1_end, max_iter=200000, beta_decay=0.55, n_phases=22)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"  After FISTA: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 2: Adam alternating subgradient (multiple betas) ===
    print("Phase 2: Adam alternating subgradient")
    phase2_end = min(deadline, t_start + 750.0)
    if time.monotonic() < phase2_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.001 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc]
        per_beta = (phase2_end - time.monotonic() - 5) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase2_end - 8:
                break
            a_work, _ = _adam_alternating(a_work, time.monotonic() + max(3.0, per_beta),
                                          beta, lr_init=0.005, max_iter=300000, alpha_switch=0.6)
            a_work = a_work / a_work.sum()
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 3: L-BFGS refinement ===
    print("Phase 3: L-BFGS refinement")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 10:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        for beta_frac in [0.0001, 0.00003, 0.00001]:
            if time.monotonic() > phase3_end - 8:
                break
            beta_lb = beta_frac * max_cc
            a_work, _ = _lbfgs_smoothed(a_work, time.monotonic() + max(5.0, (phase3_end - time.monotonic() - 5) / 3), beta_lb)
            a_work = a_work / a_work.sum()
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  L-BFGS beta={beta_lb:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 4: Coordinate descent local search ===
    print("Phase 4: Coordinate descent")
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 10:
        a_cd, _ = _coordinate_descent(best_seq, phase4_end, max_rounds=100)
        a_cd = a_cd / a_cd.sum()
        sc = _score(a_cd)
        if sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 5: Try longer sequence (n*2) ===
    print("Phase 5: Longer sequence exploration")
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end - 20 and n_main <= 8192:
        n_long = n_main * 2
        a_long = _interpolate_sequence(best_seq, n_long)
        sc_long_init = _score(a_long)
        print(f"  Long init: n={n_long}, score={sc_long_init:.10f}")
        
        if sc_long_init < best_score + 0.05:  # Only continue if competitive
            c_l = _full_convolve(a_long, a_long)
            ml = float(c_l.max())
            a_long, _ = _fista_adaptive(a_long, 0.02 * ml, time.monotonic() + 60, max_iter=50000, beta_decay=0.5, n_phases=8)
            a_long = a_long / a_long.sum()
            sc_long = _score(a_long)
            if sc_long < best_score:
                best_score = sc_long
                best_seq = a_long.copy()
                print(f"  Long FISTA: score={sc_long:.10f} t={time.monotonic()-t_start:.1f}s")
            
            if time.monotonic() < phase5_end - 15:
                c_l = _full_convolve(a_long, a_long)
                ml = float(c_l.max())
                a_long, _ = _adam_alternating(a_long, phase5_end - 10, 0.0002 * ml, 0.003, 100000, alpha_switch=0.5)
                a_long = a_long / a_long.sum()
                sc_long = _score(a_long)
                if sc_long < best_score:
                    best_score = sc_long
                    best_seq = a_long.copy()
                    print(f"  Long Adam: score={sc_long:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 6: Multi-restart ===
    print("Phase 6: Multi-restart exploration")
    phase6_end = min(deadline, t_start + 1680.0)
    if time.monotonic() < phase6_end - 15:
        n_restarts = 5
        per_r = (phase6_end - time.monotonic() - 10) / n_restarts
        if per_r > 5:

                if i == 0:
                    p = _perturb(best_seq, 0.01, rng)
                elif i == 1:
                    p = _perturb(best_seq, 0.03, rng)
                elif i == 2:

                    p = np.roll(best_seq, shift).copy()
                    p = p / p.sum()
                elif i == 3:

                    alt = _make_profile(n_main, "raised_cosine")
                    p = 0.7 * best_seq + 0.3 * alt
                    p = np.maximum(p, 1e-12)
                    p = p / p.sum()
                else:
                    # Bimodal perturbation
                    alt = _make_profile(n_main, "bimodal")
                    p = 0.8 * best_seq + 0.2 * alt
                    p = np.maximum(p, 1e-12)
                    p = p / p.sum()
                
                c_p = _full_convolve(p, p)
                max_cp = float(c_p.max())
                b_r = 0.0005 * max_cp
                p, _ = _fista_adaptive(p, b_r, time.monotonic() + per_r * 0.5, 30000, beta_decay=0.5, n_phases=6)
                p = p / p.sum()
                p, _ = _adam_alternating(p, time.monotonic() + per_r * 0.4, b_r * 0.1, 0.003, 50000, alpha_switch=0.5)
                p = p / p.sum()
                sc_r = _score(p)
                if sc_r < best_score:

                    best_seq = p.copy()
                    print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 7: Final ultra-polish ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            # Very fine Adam
            a_pol, _ = _adam_alternating(a_pol, time.monotonic() + remaining * 0.5,
                                         0.000005 * max_cp, 0.001, 100000, alpha_switch=0.4)
            a_pol = a_pol / a_pol.sum()
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            
            if time.monotonic() < deadline - 6:
                # L-BFGS on remaining time
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000003 * float(c_f.max())
                a_pol, _ = _lbfgs_smoothed(a_pol, deadline - 3.0, b_f)

                sc_f = _score(a_pol)

                    best_seq = a_pol.copy()

    return [float(x) for x in fallback]
