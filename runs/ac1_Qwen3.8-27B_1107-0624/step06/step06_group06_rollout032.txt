"""
Optimized step function sequence optimizer using adaptive log-sum-exp L-BFGS,
multiplicative weight polish, multi-n search, and sparsity-promoting shrinkage.
Key improvements: adaptive beta from correlation spread, MWU refinement, 
frequency-domain perturbations, and aggressive multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_smoothed_max(c, beta):
    """Compute a smoothed approximation of max(c) and its weights."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max, None
    smoothed = c_max + beta * np.log(s)
    w = e / s
    return smoothed, w


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    smoothed, _ = _logsumexp_smoothed_max(c, beta)
    return 2.0 * n * smoothed


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    _, w = _logsumexp_smoothed_max(c, beta)
    if w is None:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=0.001):
    """Compute subgradient using active set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adapt_beta(c, floor=1e-10):
    """Adaptively compute beta from the spread of correlation values."""
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    # Use the gap between max and 95th percentile
    idx = min(10, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    return max(gap, c_max * 0.005, floor)


def _mweu_polish(a_init, deadline, n_steps=200, eta_init=0.01):
    """Multiplicative weight updates for fine-tuning."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive tolerance
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        
        # Compute gradient (subgradient of max)
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        
        # MWU update: p_i *= exp(-eta * (h_i - W))
        eta = eta_init / (1.0 + 0.01 * step)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * 4.0 * n * (h - W)  # factor of 4n from gradient
        log_a = log_a - log_a.max()
        a = np.exp(np.clip(log_a, -700, 700))
        a = a / a.sum()
        
        if step % 20 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

                    patience = 0

    n = len(a_init)
    current_a = a_init.copy()

            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 8))))
        
        def fun(z_local):
            return _smoobj(z_local, beta, n)
        
        def jac(z_local):
            return _smograd(z_local, beta, n)

                    'maxcor': 40,
                    'maxls': 50,

                current_sc = sc_new

                z = z - z.mean()
        except Exception:
            pass

        new_beta = _adapt_beta(c_cur)
        if new_beta >= beta * 0.8:

def _shrinkage(a, deadline):
    """Try zeroing out smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bowed":
        # Large at edges, small in middle
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 5.0 * np.sin(np.pi * t) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        # Modulated cosine
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.2
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_fourier(a, rng, n_modes=5, amp_scale=0.03):
    """Perturb in log-frequency domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_noise(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

"""
Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive log-sum-exp L-BFGS, multiplicative weight updates (MWU),
multi-n exploration, support refinement, and diverse frequency-domain restarts.
Key innovations: adaptive beta from correlation spread, MWU polish,
iterative support reduction, and concurrent multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_max(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max, None
    return c_max + beta * np.log(s), e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    smoothed, _ = _smoothed_max(c, beta)
    return 2.0 * n * smoothed


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    _, w = _smoothed_max(c, beta)
    if w is None:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_subgrad(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adapt_beta(c):
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    idx = min(10, len(c_sorted) - 1)
    gap = c_sorted[0] - c_sorted[idx]
    return max(gap, c_max * 0.005, 1e-12)


def _mweu_polish(a_init, deadline, n_steps=300, eta_init=0.008):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.003 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        eta = eta_init / (1.0 + 0.005 * step)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * 4.0 * n * (h - W)
        log_a = log_a - log_a.max()
        a = np.exp(np.clip(log_a, -700, 700))
        a = a / a.sum()
        if step % 25 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.015):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_subgrad(a, c, n, tol_frac=0.002)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step_size = lr / (1.0 + 0.0002 * i)
        z = z - step_size * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if i % 40 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.04, lr_init)
            else:
                patience += 1
                lr *= 0.82
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_adaptive_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    c_init = _full_convolve(current_a, current_a)
    beta = _adapt_beta(c_init)
    max_beta_steps = 25
    for step in range(max_beta_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 10))))
        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                    'maxcor': 40,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
        c_cur = _full_convolve(current_a, current_a)
        new_beta = _adapt_beta(c_cur)
        if new_beta >= beta * 0.7:
            new_beta = beta * 0.5
        beta = max(new_beta, 1e-12)
    return current_a, current_sc


def _shrinkage(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    fractions = [0.01, 0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bowed":
        t = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 4.0 * np.sin(np.pi * t) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.15
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_fourier(a, rng, n_modes=5, amp_scale=0.025):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_noise(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _carry_carry_optimize(a_init, deadline):

    n = len(a_init)

        improved = False
        for bi in range(0, n_blocks - 1, 2):
            if time.monotonic() > deadline:

            end = (bi + 2) * block_size
            if end > n:
                end = n
            block = a_init[start:end].copy()

        update_best(a_opt, sc_opt, f"n_{n_try}")
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_mwu3, sc_mwu3 = _mweu_polish(a_opt, min(deadline - 30, time.monotonic() + 50), n_steps=80)
            update_best(a_mwu3, sc_mwu3, f"n_{n_try}_mwu")

    # Phase 4: Diverse restarts
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        perturbations = [
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=6, amp_scale=0.02)),
            ("noise_small", lambda: _perturb_noise(best_seq, 0.03, rng)),
            ("noise_med", lambda: _perturb_noise(best_seq, 0.08, rng)),
            ("bowed_modify", None),
        ]
        for name, fn in perturbations:
            if time.monotonic() > deadline - 25:
                break
            if name == "bowed_modify":
                # Mix with a bowed profile
                n_b = len(best_seq)
                bowed = _make_profile(n_b, "bowed")
                mix = 0.7 * best_seq + 0.3 * bowed
                mix = mix / mix.sum()
                p_start = mix
            else:
                p_start = fn()
            sc_start = _score(p_start)
            a_r, sc_r = _run_adaptive_lbfgs(p_start, deadline, min_time_remaining=20.0)
            update_best(a_r, sc_r, f"restart_{name}")

    # Phase 5: Support refinement
    print(f"\n[Phase 5] Support refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_sup, sc_sup = _support_refine(best_seq, min(deadline - 5, time.monotonic() + 60))
        update_best(a_sup, sc_sup, "support")

    # Phase 6: Final polish combination
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very fine adaptive L-BFGS
        c_best = _full_convolve(best_seq, best_seq)
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = _adapt_beta(c_best) * 0.1
        remaining = deadline - 3 - time.monotonic()
        if remaining > 10:
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_local: _smoobj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smograd(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 40}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass
        # Final MWU
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _mweu_polish(best_seq, deadline - 1, n_steps=200, eta_init=0.004)
            update_best(a_fin2, sc_fin2, "final_mwu")
        # Final subgrad
        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 1, n_max_steps=500, lr_init=0.006)
            update_best(a_fin3, sc_fin3, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
