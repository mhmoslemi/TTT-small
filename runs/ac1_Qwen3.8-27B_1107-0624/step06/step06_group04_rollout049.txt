Convex QP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses constraint generation (cutting plane) with cvxpy to solve the QP exactly
for growing active sets, combined with gradient-based polishing.

import time
import cvxpy as cp
from scipy.sparse import lil_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _mk_matrix(k, n):
    """Build the symmetric matrix M_k such that a^T M_k a = sum_i a_i * a_{k-i}."""
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    rows = []
    cols = []
    vals = []
    for i in range(i_min, i_max + 1):
        j = k - i
        if i == j:
            rows.append(i)
            cols.append(j)
            vals.append(1.0)
        else:
            rows.append(i)
            cols.append(j)
            vals.append(0.5)
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
    from scipy.sparse import coo_matrix
    return coo_matrix((vals, (rows, cols)), shape=(n, n)).tocsr()

    a_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [cp.sum(a_var) == 1, a_var >= 0]
    
    for k in active_set:
        M = _mk_matrix(k, n)
        # Use quad_form with PSD assumption
        P = cp.psd_wrap(M)
        constraints.append(cp.quad_form(a_var, P) <= t_var)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)

        prob.solve(solver=cp.SCS, verbose=False, max_iters=10000)
    except Exception as e:
        print(f"  QP solve failed: {e}")
        return None, None
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        print(f"  QP status: {prob.status}")
        return None, None
    
    a_opt = np.asarray(a_var.value).flatten()

"""
Convex QP constraint-generation optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses cvxpy to solve QP subproblems with growing active constraint sets,
combined with subgradient polishing and Adam log-space refinement.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import coo_matrix, csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _mk_matrix(k, n):
    """Build sparse symmetric M_k: a^T M_k a = sum_{i} a_i * a_{k-i} (valid indices)."""
    i_min = max(0, k - n + 1)
    i_max = min(n - 1, k)
    rows = []
    cols = []
    vals = []
    for i in range(i_min, i_max + 1):
        j = k - i
        if i == j:
            rows.append(i)
            cols.append(j)
            vals.append(1.0)
        else:
            rows.append(i)
            cols.append(j)
            vals.append(0.5)
            rows.append(j)
            cols.append(i)
            vals.append(0.5)
    return coo_matrix((vals, (rows, cols)), shape=(n, n)).tocsr()


def _qp_solve(a_init, active_list, n, solver=cp.SCS):
    """Solve: min t s.t. a^T M_k a <= t for k in active_list, sum(a)=1, a>=0."""
    a_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [cp.sum(a_var) == 1, a_var >= 0]

    for k in active_list:
        M = _mk_matrix(k, n)
        P = cp.psd_wrap(M)
        constraints.append(cp.quad_form(a_var, P) <= t_var)

    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=solver, verbose=False, max_iters=20000, eps=1e-8)
    except Exception:
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=20000)
        except Exception:
            return None, None

    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, None

    a_opt = np.asarray(a_var.value).flatten()
    t_opt = float(t_var.value)
    a_opt = np.clip(a_opt, 0, None)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt / s
    else:
        return None, None
    return a_opt, t_opt


def _qp_constraint_generation(a_init, deadline, max_active=80, initial_k=8, batch_size=5):
    """Constraint generation: grow active set, solve QP, check violations."""
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    c = _full_convolve(a, a)

    active_set = set(top_idx.tolist())

    best_a = a.copy()
    best_score = _score_norm(a)

    iteration = 0
    stall_count = 0

    while time.monotonic() < deadline - 10:
        iteration += 1
        active_list = sorted(active_set)

        # Cap active set size
        if len(active_list) > max_active:
            c_cur = _full_convolve(a, a)
            keep = np.argsort(c_cur)[::-1][:max_active]
            active_set = set(keep.tolist())
            active_list = sorted(active_set)

        t0 = time.monotonic()
        a_opt, t_opt = _qp_solve(a, active_list, n)
        solve_time = time.monotonic() - t0

        if a_opt is None:
            print(f"    QP iter {iteration}: solve failed ({solve_time:.1f}s)")
            stall_count += 1
            if stall_count > 5:
                break
            continue

        a = a_opt
        c = _full_convolve(a, a)
        cur_max = float(c.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            stall_count = 0
            if iteration % 3 == 0 or cur_score < 1.503:
                print(f"    QP iter {iteration}: score={cur_score:.10f}, active={len(active_set)}, t={t_opt:.6e}, time={solve_time:.1f}s")
        else:
            stall_count += 1
            if stall_count > 15:
                # Try adding more constraints
                violations = np.where(c > (t_opt if t_opt else cur_max) + 1e-10)[0]
                if len(violations) > 0:
                    worst = violations[np.argsort(c[violations])][-batch_size:]
                    active_set.update(worst.tolist())
                else:
                    break

        # Find violations
        threshold = (t_opt if t_opt and np.isfinite(t_opt) else cur_max) + 1e-10
        violations = np.where(c > threshold)[0]

        if len(violations) == 0:
            if iteration > 3:
                print(f"    QP iter {iteration}: converged (no violations)")
            break

        # Add worst violators to active set
        worst_viol = violations[np.argsort(c[violations])][-batch_size:]
        active_set.update(worst_viol.tolist())

    return best_a, best_score


def _subgradient_method(a_init, deadline, n_steps=5000, batch=1):
    """Direct subgradient of max convolution, projected onto simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_base = 0.5
    t_val = 1.0

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Subgradient: use top-batch positions with softmax weighting
        top_k = min(20, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        beta_sg = max((c_top[0] - c_top[-1]) * 0.1, 1e-8)
        w = np.exp((c_top - c_top[0]) / beta_sg)
        w = w / w.sum()

        # Gradient = sum_k w_k * 2 * M_k * a
        grad = np.zeros(n)
        for idx, k in enumerate(top_idx[:batch]):
            if w[idx] < 1e-10:
                break
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                grad[i] += w[idx] * 2.0 * a[j]
                if i != j:
                    grad[j] += w[idx] * 2.0 * a[i] * 0.5 * 2  # account for symmetry

        # Simpler: grad = 2 * (a reversed and shifted for each k, weighted)
        # Let's redo this more carefully
        grad = np.zeros(n)
        for idx, k in enumerate(top_idx[:batch]):
            wi = w[idx]
            if wi < 1e-12:
                break
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                # d/d(a_i) of sum_l a_l * a_{k-l} = a_{k-i} + (if k-i != i) a_{k-i} appears once
                # Actually d/d(a_i) [sum_l a_l * a_{k-l}] = a_{k-i} (when i is the first index)
                # But also if k-l = i, i.e., l = k-i, then d/d(a_i) includes a_{k-i} * 1
                # So total: 2 * a_{k-i} if i != k-i, or a_i if i = k-i (i.e., k=2i)
                if i == k - i:
                    grad[i] += wi * 2.0 * a[i]
                else:
                    grad[i] += wi * 2.0 * a[k - i]

        # Projected gradient step with decaying step size
        step = step_base / (1.0 + 0.05 * it)
        a_new = _proj_simplex(a - step * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step *= 0.7
            a_new2 = _proj_simplex(a - step * grad)
            sc2 = _score_norm(a_new2)
            if sc2 < _score_norm(a):
                a = a_new2

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init=0.01, max_iter=200000):
    """Adam in log-space with softmax parameterization."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 1.0:
                update *= 1.0 / un

            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
            if no_improve > 40 and t > 500:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA with softmax smoothing of max."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_ci = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_ci
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        ws = float(e.sum())
        if ws < 1e-300 or not np.isfinite(ws):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / ws

        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_f = _score_norm(a_curr)
    if sc_f < best_score:
        best_score = sc_f
        best_a = a_curr.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        if sc_qp and _score(a_qp) < best_score:

            best_seq = a_qp.copy()
            n_main = len(best_seq)
            print(f"Phase 1 result: score={best_score:.10f}")
        else:
            print(f"Phase 1: no improvement (qp_norm_score={sc_qp:.10f})")

    # === Phase 2: QP on n=8192 ===
    print(f"\n=== Phase 2: QP on n=8192 ===")
    phase2_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase2_end - 30:
        n_8192 = 8192
        a_8192 = _interp_profile(best_seq, n_8192)
        sc_8192_init = _score(a_8192)
        print(f"  Interpolated n=8192 initial: score={sc_8192_init:.10f}")

        a_qp2, sc_qp2 = _qp_constraint_generation(
            a_8192, phase2_end, max_active=50, initial_k=6, batch_size=3
        )
        if sc_qp2 is not None:
            sc_8192_qp = _score(a_qp2)
            print(f"  QP n=8192: score={sc_8192_qp:.10f}")
            if sc_8192_qp < best_score:
                best_score = sc_8192_qp
                best_seq = a_qp2.copy()
                n_main = len(best_seq)

    # === Phase 3: Subgradient polish on best ===
    print(f"\n=== Phase 3: Subgradient polish (n={n_main}) ===")
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end - 10:
        a_sg, sc_sg = _subgradient_method(best_seq, phase3_end, n_steps=3000, batch=8)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            n_main = len(best_seq)
            print(f"  Subgradient: score={sc_sg_full:.10f}")

    # === Phase 4: Adam log-space fine-tuning ===
    print(f"\n=== Phase 4: Adam log-space (n={n_main}) ===")
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end - 20:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas = [0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw, 0.00005 * max_cw]
        a_adam, sc_adam = _adam_logspace(a_work, phase4_end, betas, lr_init=0.008, max_iter=300000)
        sc_adam_full = _score(a_adam)
        if sc_adam_full < best_score:
            best_score = sc_adam_full
            best_seq = a_adam.copy()
            n_main = len(best_seq)
            print(f"  Adam: score={sc_adam_full:.10f}")

    # === Phase 5: FISTA polish ===
    print(f"\n=== Phase 5: FISTA polish ===")
    if time.monotonic() < deadline - 30:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        c_w = _full_convolve(a_work, a_work)
        mc = float(c_w.max())
        betas_f = [0.0005 * mc, 0.0002 * mc, 0.0001 * mc]
        for b in betas_f:
            if time.monotonic() > deadline - 20:
                break
            a_f, sc_f = _fista_smoothed(a_work, b, time.monotonic() + 30, 50000)
            a_work = a_f / a_f.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
                n_main = len(best_seq)
                print(f"  FISTA beta={b:.2e}: score={sc_full:.10f}")

    # === Final validation ===
    print(f"\n=== Final: score={best_score:.10f}, n={len(best_seq)} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL internal score: {sc_final:.10f}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(512) / 512
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
