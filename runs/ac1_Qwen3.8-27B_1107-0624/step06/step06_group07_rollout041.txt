"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find descent directions, combined with
subgradient polishing and multi-restart strategies.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(2, N)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(a):
    """Build sparse matrix T where (T @ b)_k = (a*b)_k."""
    n = len(a)
    n_rows = 2 * n - 1
    # T[k, i] = a[k-i] if 0 <= k-i < n
    rows = np.zeros(n * n, dtype=np.int32)
    cols = np.zeros(n * n, dtype=np.int32)
    vals = np.zeros(n * n, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        count = i_max - i_min + 1
        rows[idx:idx+count] = k
        cols[idx:idx+count] = np.arange(i_min, i_max + 1)
        vals[idx:idx+count] = a[k - np.arange(i_min, i_max + 1)]
        idx += count
    T = csr_matrix((vals[:idx], (rows[:idx], cols[:idx])), shape=(n_rows, n))
    return T

def _solve_lp_direction(a, M, max_iter=10000):
    """Solve LP to find best direction b."""
    n = len(a)
    T = _build_conv_matrix(a)
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    A_ub = T
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'maxiter': max_iter, 'presolve': True})
        if res.success:
            return res.x
        else:
            return None
    except Exception as e:
        print(f"  LP error: {e}")
        return None

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):

Uses linear programming to find exact descent directions (from the paper's LP approach),
combined with L-BFGS log-sum-exp smoothing, subgradient polishing, and multi-restart.
The LP approach finds the globally best direction to move at each step, giving
monotone descent in the max-convolution.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix_sparse(a):
    n = len(a)
    n_rows = 2 * n - 1
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        rows[idx:idx + count] = k
        i_vals = np.arange(i_start, i_end + 1)
        cols[idx:idx + count] = i_vals
        vals[idx:idx + count] = a[k - i_vals]
        idx += count
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n))


def _solve_lp_full(a, M, timeout_s=180):
    """Solve full LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    t0 = time.monotonic()
    T = _build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
        if res.status == 0:
            return res.x
        elif time.monotonic() - t0 < timeout_s:
            return None
        else:
            return None
    except Exception:
        return None


def _solve_lp_reduced(a, M, top_k=200, timeout_s=60):

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _perturb(a, rng, kind, param):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    if kind == 'gauss':
        z = z + param * rng.standard_normal(n)
    elif kind == 'fourier':
        t = np.arange(n)
        for _ in range(int(param)):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.005 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif kind == 'block':
        n_blocks = int(param)
        block_size = max(4, n // max(1, n_blocks))
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
            z[start:end] += rng.uniform(-0.02, 0.02)
    elif kind == 'random_flip':
        n_flips = int(n * param)
        idx = rng.choice(n, size=n_flips, replace=False)
        z[idx] += rng.uniform(-0.05, 0.05, size=n_flips)
    z = z - z.mean()
    return _softmax(z)

"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Primary method: Linear programming to find exact descent directions per the paper's LP approach.
Secondary: L-BFGS with log-sum-exp smoothing, subgradient polishing, multi-restart.
The LP at each step finds the globally best non-negative direction b maximizing sum(b)
subject to (a*b)_k <= M for all k, guaranteeing monotone descent in max convolution.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix_sparse(a):
    n = len(a)
    n_rows = 2 * n - 1
    total_nnz = n * n
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(n_rows):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        count = i_end - i_start + 1
        rows[idx:idx + count] = k
        i_vals = np.arange(i_start, i_end + 1)
        cols[idx:idx + count] = i_vals
        vals[idx:idx + count] = a[k - i_vals]
        idx += count
    return csr_matrix((vals, (rows, cols)), shape=(n_rows, n))


def _solve_lp_full(a, M, timeout_s=150):
    n = len(a)
    T = _build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _solve_lp_reduced(a, M, top_k=200, timeout_s=60):
    n = len(a)
    c_aa = _full_convolve(a, a)
    k = min(top_k, len(c_aa))
    top_idx = np.argsort(c_aa)[-k:]
    top_idx = np.sort(top_idx)
    T_full = _build_conv_matrix_sparse(a)
    T_red = T_full[top_idx, :].tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=T_red, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxcor': 50})
        return _softmax(result.x)
    except Exception:
        return a_init


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    betas = max_c * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(8000, max(100, int(remaining / (est_time * 15))))
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + i * 0.0003)
        z = z - step * grad
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.05
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 4:
                    lr *= 0.7
                    if lr < 1e-6:
                        break
                else:
                    lr *= 0.92
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _subgrad_direct(a_init, deadline, tol_frac=1e-5, max_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()
        if np.linalg.norm(grad) < 1e-14:
            break
        a_try = a - step_size * grad
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
        if sc_try < _score(a) - 1e-15:
            a = a_try
            step_size = min(step_size * 1.01, 0.1)
        else:
            stall += 1
            step_size *= 0.9
        step_size *= (1.0 - 1e-5)
        step_size = max(step_size, 1e-12)
        if stall > 150:
            break
    return best_a, best_sc


def _perturb(a, rng, kind, param):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    if kind == 'gauss':
        z = z + param * rng.standard_normal(n)
    elif kind == 'fourier':
        t = np.arange(n)
        for _ in range(int(param)):
            freq = int(rng.integers(1, max(2, n // 8)))
            amp = 0.005 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif kind == 'block':
        n_blocks = int(param)
        block_size = max(4, n // max(1, n_blocks))
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
            z[start:end] += rng.uniform(-0.02, 0.02)
    elif kind == 'random_flip':
        n_flips = max(1, int(n * param))
        idx = rng.choice(n, size=n_flips, replace=False)
        z[idx] += rng.uniform(-0.05, 0.05, size=n_flips)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _local_search(a, deadline, max_rounds=300):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.01
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        idx = int((rnd * 7919 + 13) % n)
        jdx = int((rnd * 7919 + 13 + 37) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n
        for delta_frac in [0.25, 0.4, 0.5, 0.6, 0.75]:
            s_pair = best_a[idx] + best_a[jdx]
            new_val_i = s_pair * (delta_frac + step)
            new_val_j = s_pair * (1.0 - delta_frac - step)
            if new_val_i < 0 or new_val_j < 0:
                continue
            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True
        if not improved and rnd > 30:
            step *= 0.7
            if step < 1e-7:
                break
    return best_a, best_sc


def _lp_step(a, deadline, use_full=True, timeout_s=120, reduced_k=300):
    """Perform one LP-based step. Returns updated a or None if failed."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    c_cur = _full_convolve(a, a)
    M = float(c_cur.max())
    if M < 1e-15:
        return a

    b = None
    if use_full:
        b = _solve_lp_full(a, M, timeout_s=timeout_s)
    if b is None:
        b = _solve_lp_reduced(a, M, top_k=reduced_k, timeout_s=min(timeout_s, 60))
    if b is None:
        b = _solve_lp_reduced(a, M, top_k=150, timeout_s=30)
    if b is None:
        return None

    b_sum = float(b.sum())
    if b_sum < 1e-15:
        return None
    g = b * (target_sum / b_sum)

    best_t = 0.0
    best_a_out = a.copy()
    best_M_out = M
    for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
        if time.monotonic() > deadline:
            break
        a_try = (1.0 - t) * a + t * g
        a_try = np.maximum(a_try, 0.0)
        s_try = float(a_try.sum())
        if s_try < 1e-15:
            continue
        c_try = _full_convolve(a_try, a_try)
        M_try = float(c_try.max())
        if M_try < best_M_out - 1e-15:
            best_M_out = M_try
            best_a_out = a_try.copy()
            best_t = t

    if best_t > 0:
        return best_a_out
    return None
