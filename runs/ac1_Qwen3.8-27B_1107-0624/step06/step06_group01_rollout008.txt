Optimizer combining GS initialization, FISTA with adaptive smoothing, LP-based refinement,
and Adam log-space polishing to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gs_init(n, n_iter=200, rng=None):
    """Gerchberg-Saxton initialization for flat autocorrelation."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    N = _next_pow2(2 * n - 1)
    
    # Start with random non-negative
    a = rng.random(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_iter):
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        
        mag = np.abs(A)
        # Target: flat magnitude
        target_mag = 1.0 / np.sqrt(N)
        
        # Relaxed update (avoid oscillation)
        alpha = 0.3 + 0.7 * (it / max(1, n_iter))  # Start more relaxed, become more aggressive
        new_mag = alpha * target_mag + (1.0 - alpha) * mag
        phase = np.angle(A)
        
        A_new = new_mag * (A / (mag + 1e-30))
        a_new = np.fft.irfft(A_new, N)[:n]
        
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a_new / s
        
        if it % 20 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score

def _fista_smoothed(a_init, beta, deadline, max_iter):

    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

def _lp_step(a, M, max_iter=200):
    """
    Find b >= 0, sum(b)=1, minimizing max_k (a*b)[k] using projected gradient.
    Uses subgradient method on the max of linear functions.
    """
    n = len(a)
    b = np.ones(n) / n
    
    for it in range(max_iter):
        # Compute conv(a, b)
        ab = _full_convolve(a, b)
        k_star = int(np.argmax(ab))
        max_val = ab[k_star]
        
        # Subgradient of max_k (a*b)[k] w.r.t. b
        # d/db[j] of (a*b)[k] = a[k-j] (with boundary handling)
        # (a*b)[k] = sum_i a[i]*b[k-i]
        # d/db[j] = a[k-j] if 0 <= k-j < n, else 0
        # But also: (a*b)[k] = sum_i a[k-i]*b[i]
        # d/db[j] = a[k-j] if 0 <= k-j < n, else 0
        
        # Actually, let's be more careful:
        # (a*b)[k] = sum_{i} a[i] * b[k-i] where 0<=i<n and 0<=k-i<n
        # = sum_{j} a[k-j] * b[j] where 0<=k-j<n and 0<=j<n
        # So d/db[j] (a*b)[k] = a[k-j] if 0<=k-j<n, else 0
        
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        for j in range(lo, hi + 1):
            idx = k_star - j
            if 0 <= idx < n:
                sg[j] = a[idx]
        
        # Projected subgradient step
        eta = 2.0 / (max_val + 1e-10) / (it + 1)
        b_new = b - eta * sg
        b_new = np.maximum(b_new, 0.0)
        s = b_new.sum()
        if s > 1e-15:
            b_new = b_new / s
        else:
            b_new = b
        b = b_new
    
    return b

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3

        return a / a.sum()
    elif kind == "bimodal":

        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)

        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n
