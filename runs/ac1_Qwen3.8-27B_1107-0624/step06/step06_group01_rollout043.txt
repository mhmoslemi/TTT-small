import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_flat(a):
    """Score with flattening: 2n*max(c)/s^2 + lambda*var(c)/max(c)"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    cm = c.max()
    # Objective is still 2n*cm/s^2, but we track flatness for diagnostics
    return float(2.0 * n * cm / (s * s))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w, n):
    """Gradient of sum(w_k * c_k) w.r.t. a, where c = conv(a,a)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _softmax_normalize(z, beta):

    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(z)
        w[int(np.argmax(z))] = 1.0
        return w
    return e / s

def _adam_optimize(a_init, deadline, beta_schedule, lr_init, var_lambda=0.0, max_iter=200000):

                grad = grad + var_lambda * grad_var

def _peak_shave(a, deadline, n_peaks=10):
    """Targeted peak reduction: identify top peaks and try to reduce them."""
    n = len(a)
    best_a = a.copy()
    c = _full_convolve(best_a, best_a)
    best_score = float(c.max())
    
    rng = np.random.default_rng(7)
    
    iterations = 0
    while time.monotonic() < deadline and best_score > 0:
        iterations += 1
        c = _full_convolve(best_a, best_a)
        cm = float(c.max())
        best_score = min(best_score, cm)
        
        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        # For each peak, try to reduce it
        for pk in peak_indices[:5]:
            # The peak at index pk involves pairs (i, pk-i)
            # To reduce it, we can decrease a[i] for i contributing to this peak
            # and increase a[j] for j not contributing
            # Simple heuristic: decrease the largest contributor
            
            contribs = np.zeros(n)
            lo = max(0, pk - (n - 1))
            hi = min(n - 1, pk)
            for i in range(lo, hi + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] += best_a[i] * best_a[j]
                    if i != j:
                        contribs[j] += best_a[i] * best_a[j]
            
            # Find the index with the largest contribution that we can decrease
            top_contrib = np.argsort(contribs)[::-1][:3]
            
            for i in top_contrib:
                for factor in [0.95, 0.9, 0.85]:
                    a_try = best_a.copy()
                    old_val = a_try[i]
                    a_try[i] *= factor
                    # Compensate by increasing a random other index
                    # to keep sum roughly the same
                    # Actually, we'll renormalize
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try = a_try / s
                        c_try = _full_convolve(a_try, a_try)
                        if c_try.max() < best_score - 1e-15:
                            best_score = float(c_try.max())
                            best_a = a_try.copy()
                            break
        
        if iterations % 10 == 0:
            print(f"    Peak shave iter {iterations}: score={best_score:.10f}")
    
    return best_a, best_score

    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, power)
    return a / a.sum()

def _make_modulated_profile(n, power=0.5, freq=1.0, amp=0.1):
    """Modulated 1/i^power profile."""
    i = np.arange(n, dtype=np.float64)
    base = 1.0 / np.power(i + 1, power)
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    a = base * mod
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Minimizer for 2n*max(conv(a,a))/sum(a)^2 using adaptive-temperature Adam with
flatness regularization, spectral initialization, peak-shaving, and multi-restart.
Key improvements: adaptive beta based on top-k gap, cosine LR schedule,
variance penalty for flat convolution, and targeted peak reduction.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _max_conv(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_weights(c, beta):
    shifted = (c - c.max()) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _topk_weights(c, k, beta):
    """Weights emphasizing top-k values with temperature beta."""
    shifted = (c - c.max()) / beta
    e = np.exp(shifted)
    # Boost top-k
    sorted_idx = np.argsort(-e)
    top_k = sorted_idx[:min(k, len(c))]
    e_boosted = e.copy()
    e_boosted[top_k] *= 2.0
    s = e_boosted.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[top_k[0]] = 1.0
        return w
    return e_boosted / s


def _adam_core(a_init, deadline, beta_schedule, lr_init, flat_lambda=0.0,
               lr_decay=0.98, decay_every=3000, grad_clip=2.0, check_every=20):
    """Core Adam optimizer in log-space with adaptive features."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_mc = _max_conv(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while time.monotonic() < deadline:
            t += 1
            # Compute a from log-space
            z_max = z.max()
            e_z = np.exp(z - z_max)
            a = e_z / e_z.sum()

            c = _full_convolve(a, a)
            cm = float(c.max())

            # Adaptive: if top values are close, use effective smaller beta
            sorted_c = np.sort(c)[::-1]
            if len(sorted_c) >= 2:
                gap = sorted_c[0] - sorted_c[min(4, len(c)-1)]
                if gap < 0.01 * cm:
                    effective_beta = beta * 0.5
                else:
                    effective_beta = beta
            else:
                effective_beta = beta

            w = _softmax_weights(c, effective_beta)
            grad = _compute_grad(a, w, n)

            # Flatness regularizer
            if flat_lambda > 0:
                d = c - c.mean()
                grad_flat = _compute_grad(a, d / c.size, n)
                grad = grad + flat_lambda * grad_flat

            # Check for finite gradient
            if not np.all(np.isfinite(grad)):
                break

            # Adam update
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            # Periodic score check
            if t % check_every == 0:
                z_m = z.max()
                e_ch = np.exp(z - z_m)
                a_ch = e_ch / e_ch.sum()
                mc = _max_conv(a_ch)
                if np.isfinite(mc) and mc < best_mc:
                    best_mc = mc
                    best_a = a_ch.copy()

            # Learning rate decay
            if t % decay_every == 0:
                current_lr *= lr_decay

    # Final evaluation
    z_m = z.max()
    e_f = np.exp(z - z_m)
    a_f = e_f / e_f.sum()
    mc_f = _max_conv(a_f)
    if np.isfinite(mc_f) and mc_f < best_mc:
        best_mc = mc_f
        best_a = a_f.copy()

    return best_a, best_mc


def _peak_shave(a, deadline, n_top=5, n_try=3):
    """Targeted peak reduction via coordinate adjustments."""
    n = len(a)
    best_a = a.copy()
    best_mc = _max_conv(a)
    rng = np.random.default_rng(123)
    it = 0

    while time.monotonic() < deadline and it < 5000:
        it += 1
        c = _full_convolve(best_a, best_a)
        cm = float(c.max())
        if cm < best_mc:
            best_mc = cm

        # Find top peaks
        peak_indices = np.argsort(c)[::-1][:n_top]

        improved = False
        for pk in peak_indices:
            # Identify contributing indices
            lo = max(0, pk - (n - 1))
            hi = min(n - 1, pk)
            contribs = np.zeros(n)
            for i in range(lo, hi + 1):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] += best_a[i] * best_a[j]

            # Try reducing top contributors
            top_idx = np.argsort(-contribs)[:n_try]
            for i in top_idx:
                for factor in [0.92, 0.85, 0.78]:
                    a_try = best_a.copy()
                    a_try[i] *= factor
                    s = a_try.sum()
                    if s > 1e-15:
                        a_try /= s
                        mc_try = _max_conv(a_try)
                        if mc_try < best_mc - 1e-14:
                            best_mc = mc_try
                            best_a = a_try.copy()
                            improved = True
                            break

        if not improved and it > 100:
            # Try random perturbations of small indices
            for _ in range(10):
                i = rng.integers(0, n)
                a_try = best_a.copy()
                a_try[i] *= (0.5 + rng.random())
                s = a_try.sum()
                if s > 1e-15:
                    a_try /= s
                    mc_try = _max_conv(a_try)
                    if mc_try < best_mc - 1e-14:
                        best_mc = mc_try
                        best_a = a_try.copy()
                        break

    return best_a, best_mc


def _proj_simplex_fast(v):
    """Faster projection onto simplex using bisection."""
    n = len(v)
    if n == 0:
        return v
    # Bisection on threshold
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.maximum(v - mid, 0.0).sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2.0, 0.0)


def _fista_smoothed(a_init, beta, deadline, max_iter=100000):
    """FISTA with softmax-smoothed max on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_mc = _max_conv(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = _softmax_weights(c_y, beta)
        grad = _compute_grad(y, w, n)

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            mc = _max_conv(a_next)
            if mc < best_mc:
                best_mc = mc
                best_a = a_next.copy()
                # Adapt step size
                c_new = _full_convolve(a_next, a_next)
                max_c_new = float(c_new.max())
                L = 4.0 * n * max_c_new / beta
                step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    mc_final = _max_conv(a_curr)
    if mc_final < best_mc:
        best_mc = mc_final
        best_a = a_curr.copy()
    return best_a, best_mc


def _make_inv_sqrt_profile(n, power=0.5):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(i, power)
    return a / a.sum()


def _make_modulated_profile(n, power=0.5, freq=1.0, amp=0.1):
    i = np.arange(n, dtype=np.float64)
    base = 1.0 / np.power(i + 1.0, power)
    mod = 1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)
    a = base * mod
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_profile(n):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _spectral_init(n, rng):
    """Flat-spectrum initialization with random phases."""
    N = _next_pow2(2 * n - 1)
    magnitude = np.ones(N // 2 + 1)
    magnitude[0] = 0.0
    phase = rng.random(N // 2 + 1) * 2.0 * np.pi
    spectrum = magnitude * np.exp(1j * phase)
    a_full = np.fft.irfft(spectrum, N)
    a = np.maximum(a_full[:n], 0.0)
    s = a.sum()
    if s > 1e-15:
        return a / s
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()

                best_seq = a_ps2.copy()
                print(f"  Final shave: mc={mc_ps2:.10f}, obj={2.0*best_n*mc_ps2:.10f}")

    # Final output
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_final = float(final.sum())
        if s_final < 0.01:
            final = _make_inv_sqrt_profile(256)
        final_n = len(final)
        sc = _score(final)
        print(f"\n=== FINAL: n={final_n}, score={sc:.10f}, max_conv={_max_conv(final):.10f}, time={time.monotonic()-t_start:.1f}s ===")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_inv_sqrt_profile(4096)
    print(f"FALLBACK: n={len(fallback)}, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
