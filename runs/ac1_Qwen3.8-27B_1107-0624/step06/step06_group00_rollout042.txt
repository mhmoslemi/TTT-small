"""
Convex optimization approach to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient descent with FTRL, adaptive active set, 
and multi-scale exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _subgradient(a, active_set):
    """Compute exact subgradient of max_k (a*a)_k at point a."""
    n = len(a)
    grad = np.zeros(n)
    inv_s = 2.0 / len(active_set)
    for k in active_set:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            grad[j_start:j_end+1] += inv_s * a[j_end:j_start-1:-1]
    return grad

def _find_active_set(c, n, tolerance=None, top_k=None):
    """Find the active set: indices where c is close to max."""
    M = c.max()
    if tolerance is not None:
        active = np.where(c >= M - tolerance)[0]
        active = active[active < 2*n-1]  # ensure valid
    elif top_k is not None:
        active = np.argsort(c)[::-1][:top_k]
    else:
        # Use a default tolerance
        tol = 0.01 * M
        active = np.where(c >= M - tol)[0]
        if len(active) > 50:
            active = np.argsort(c)[::-1][:50]
    return active

    """FTRL optimization with adaptive active set."""
    n = len(a_init)
    a = a_init.copy()
    w = np.zeros(n)  # FTRL accumulator
    
    # Adaptive parameters
    eta = 0.01
    eta_decay = 0.9999
    active_tolerance = 0.05
    
    best_a = a.copy()
    best_score = _score_norm(a)  # assuming sum(a) = 1
    patience = 0
    best_patience_score = best_score
    
    t = 0
    while t < max_iter and time.monotonic() < deadline:
        t += 1
        c = _fft_convolve(a, a)
        M = float(c.max())
        
        # Find active set
        active = np.where(c >= M - active_tolerance * M)[0]
        if len(active) > 100:
            active = np.argsort(c)[::-1][:100]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute subgradient
        g = _subgradient(a, active)
        
        # FTRL update: w += g, then a = softmax(w)
        w += eta * g
        w -= w.mean()  # centering
        a = np.exp(w)
        a /= a.sum()
        
        # Decay eta
        eta *= eta_decay
        
        # Check improvement
        if t % 100 == 0:
            sc = float(2.0 * n * c.max())  # c was computed before update
            # Actually let's recompute

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())

        return np.inf

    return float(2.0 * n * c.max() / (s * s))

def _subgradient(a, active_set):
    """
    Compute exact subgradient of f(a) = max_k (a*a)_k at point a.
    active_set: array of indices k where (a*a)_k = max (or close to max).

    inv = 2.0 / S

        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            grad[j_start:j_end+1] += inv * a[j_end:j_start-1:-1]
    return grad

For the last mile, the active set is well-determined (only a few constraints are truly active), and we can use a more precise method:

**Active Set QP approach:**
1. Identify the active set S (indices where c[k] = M)
2. Solve: min_a (1/2) * sum_{k in S} (a*a)_k subject to sum(a) = 1, a >= 0
3. Check if the solution satisfies (a*a)_k <= M for all k not in S
4. If yes, we're at the optimum (of the relaxed problem)
5. If no, add the violated constraint to S and repeat

The subproblem in step 2 is a QP (quadratic objective, linear constraints). For |S| active constraints, the objective is:
(1/2) * sum_{k in S} (a*a)_k = (1/2) * a^T (sum_{k in S} H_k) a

where H_k is the matrix with 1s at positions (i,j) where i+j=k.

The Hessian is H = sum_{k in S} H_k, which is a banded matrix (each H_k is a single anti-diagonal). For |S| = 10, the Hessian has at most 10 non-zero anti-diagonals, so it's very sparse!

We can solve this QP efficiently using:
- scipy.optimize.minimize with method='trust-constr' or 'SLSQP'
- Or a custom active-set QP solver

But actually, since the problem is convex and we're on the simplex, we can use the KKT conditions:
At the optimum of the QP:
H * a = λ * 1 (where 1 is the all-ones vector, from the constraint sum(a)=1)
sum(a) = 1
a >= 0

The matrix H = sum_{k in S} H_k is symmetric and PSD (sum of PSD matrices). If it's positive definite on the subspace {x: sum(x)=0}, then the solution is unique.

We can solve H * a = λ * 1, sum(a) = 1 using a linear system:
[H, 1; 1^T, 0] [a; λ] = [0; 1]

This is a (n+1) x (n+1) system, but H is very sparse (banded with bandwidth |S|).

For n=4096 and |S|=10, H has about 10*4096 = 40K nonzeros. Solving a sparse linear system of this size should take milliseconds.

This is much faster than iterative first-order methods for the "last mile"!

    """
    Solve the active-set QP: min a^T H a / 2 s.t. sum(a)=1, a>=0
    where H = sum_{k in active_set} H_k (H_k has 1s on anti-diagonal k)
    """
    n = len(a_init)
    # Build H (sparse)
    # H_{j,l} = number of k in active_set such that j+l = k
    rows = []
    cols = []
    vals = []
    for k in active_set:
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            l = k - j
            rows.append(j)
            cols.append(l)
            vals.append(1.0)
    
    H = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(n, n))
    # H is symmetric, but might have both (j,l) and (l,j); that's fine for CSR.
    # Actually, we need H to be the Hessian of (1/2)*sum_k (a*a)_k
    # (a*a)_k = sum_i a[i]*a[k-i] = a^T H_k a where (H_k)_{i,j} = 1 if i+j=k
    # So the gradient is 2*H_k*a (as derived earlier)
    # And the Hessian is 2*H_k
    # The objective is (1/2)*sum_k (a*a)_k, so the Hessian of the objective is sum_k H_k
    # Wait: (1/2)*(a*a)_k = (1/2)*a^T H_k a, gradient = H_k * a, Hessian = H_k
    # (since H_k is symmetric)
    # So for the combined objective (1/2)*sum_k (a*a)_k, the Hessian is sum_k H_k

    # H * a = mu * 1  (from L = (1/2)*a^T H a - mu*(sum(a)-1), dL/da = H*a - mu*1 = 0)
    # sum(a) = 1
    
    # [H, 1] [a]   [0]
    # [1^T, 0] [mu] = [1]
    
    # But we also need a >= 0. If the unconstrained solution has a < 0,
    # we need to project onto the simplex or use a QP solver.
    
    # For now, let's solve the equality-constrained problem and check non-negativity.
    import scipy.sparse as sp
    import scipy.sparse.linalg as spla
    
    ones = sp.csr_matrix(np.ones((1, n)))
    zero = sp.csr_matrix(np.zeros((1, 1)))
    
    H_aug = sp.vstack([
        sp.hstack([H, sp.csr_matrix(np.ones((n, 1)))]),
        sp.hstack([ones, zero])
    ])
    b_aug = np.zeros(n + 1)
    b_aug[-1] = 1.0
    
    try:
        sol = spla.spsolve(H_aug, b_aug)
        a_sol = sol[:n]
        if np.all(a_sol >= -1e-10):
            a_sol = np.maximum(a_sol, 0)
            a_sol /= a_sol.sum()
            return a_sol
    except Exception:
        pass
    
    # Fallback: use scipy minimize
    from scipy.optimize import minimize
    H_dense = H.toarray()
    def obj(a):
        return 0.5 * a @ H_dense @ a
    def grad(a):
        return H_dense @ a
    res = minimize(obj, a_init, method='SLSQP', jac=grad,
                   constraints=[{'type': 'eq', 'fun': lambda a: a.sum() - 1,
                                  'jac': lambda a: np.ones(n)}],
                   bounds=[(0, None)]*n)
    if res.success:
        return res.x
    return a_init

"""
Convex optimization approach to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient mirror descent (FTRL) with adaptive active set,
active-set KKT solving, multi-scale exploration, and local polishing.
Key insight: after normalizing sum(a)=1, the objective max_k (a*a)_k is convex,
so first-order methods converge to the global optimum.
"""
import numpy as np
import time
import scipy.sparse as sp
import scipy.sparse.linalg as spla


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _subgradient(a, active_set):
    """Exact subgradient of max_k (a*a)_k at point a, given active set S."""
    n = len(a)
    S = len(active_set)
    grad = np.zeros(n)
    inv = 2.0 / S
    for k in active_set:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += inv * a[j_end:j_start - 1:-1]
    return grad


def _subgradient_fft(a, active_set):
    """FFT-based subgradient for large active sets."""
    n = len(a)
    S = len(active_set)
    L = _next_pow2(2 * n - 1)
    p_pad = np.zeros(L)
    for k in active_set:
        p_pad[int(k)] = 1.0 / S
    a_pad = np.zeros(L)
    a_pad[:n] = a
    H = np.fft.fft(p_pad) * np.conj(np.fft.fft(a_pad))
    h = np.fft.ifft(H).real[:n]
    return 2.0 * h


def _find_active_set(c, n, rel_tol=0.01, max_size=100):
    """Find active set from convolution values."""
    M = float(c.max())
    if M < 1e-30:
        return np.array([np.argmax(c)])
    threshold = M * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    active = active[active < 2 * n - 1]
    if len(active) > max_size:
        active = np.argsort(c)[::-1][:max_size]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    return active


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    p_new /= p_new.sum()
    return p_new


def _mirror_descent(a_init, deadline, eta_start=0.05, eta_end=0.001,
                    rel_tol=0.01, max_active=80, eval_interval=30,
                    max_iter=2000000, use_fft_grad=False):
    """
    Mirror descent (FTRL) with exact subgradient.
    Minimizes max_k (a*a)_k s.t. sum(a)=1, a>=0.
    """
    n = len(a_init)
    w = np.log(np.maximum(a_init, 1e-15))
    w -= w.mean()
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    patience = 0

    t = 0
    consecutive_no_improve = 0

    while t < max_iter and time.monotonic() < deadline:
        t += 1
        frac = min(1.0, t / max_iter)
        eta = eta_start * (eta_end / eta_start) ** frac

        c = _fft_convolve(a, a)
        M = float(c.max())
        active = _find_active_set(c, n, rel_tol=rel_tol, max_size=max_active)

        if use_fft_grad or len(active) > 20:
            g = _subgradient_fft(a, active)
        else:
            g = _subgradient(a, active)

        # Check for numerical issues
        if not np.all(np.isfinite(g)):
            g = np.zeros(n)

        # Mirror descent step: w -= eta * g
        w -= eta * g
        w -= w.mean()
        w = np.clip(w, -60, 60)

        a = np.exp(w)
        a_sum = a.sum()
        if a_sum < 1e-300 or not np.isfinite(a_sum):
            a = best_a.copy()
            w = np.log(np.maximum(a, 1e-15))
            w -= w.mean()
            continue
        a /= a_sum

        # Evaluate periodically
        if t % eval_interval == 0:
            c_cur = _fft_convolve(a, a)
            sc_cur = float(2.0 * n * c_cur.max())
            if sc_cur < best_sc - 1e-15:
                best_sc = sc_cur
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if patience > 40:
                patience = 0
                eta = max(eta_end, eta * 0.5)
                w = np.log(np.maximum(best_a, 1e-15))
                w -= w.mean()
                a = best_a.copy()

    c_final = _fft_convolve(a, a)
    sc_final = float(2.0 * n * c_final.max())
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _active_set_qp(a_init, active_set, max_iter_inner=50):
    """
    Solve the active-set KKT conditions:
    min (1/2)*sum_{k in S} (a*a)_k  s.t.  sum(a)=1, a>=0
    KKT: sum_{k in S} H_k * a = mu * 1,  sum(a) = 1
    """
    n = len(a_init)

    # Build H = sum_{k in S} H_k as sparse matrix
    # H_k has 1s at positions (i, j) where i + j = k
    rows_list = []
    cols_list = []
    vals_list = []
    for k in active_set:
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            l = k - j
            rows_list.append(j)
            cols_list.append(l)
            vals_list.append(1.0)

    if len(rows_list) == 0:
        return a_init

    H = sp.csr_matrix((vals_list, (rows_list, cols_list)), shape=(n, n))
    # Make symmetric (should already be, but ensure)
    H = (H + H.T) / 2

    # Build augmented KKT system:
    # [H   1] [a]   [0]
    # [1^T 0] [mu] = [1]
    ones_col = sp.csr_matrix(np.ones((n, 1)))
    ones_row = sp.csr_matrix(np.ones((1, n)))
    zero_scalar = sp.csr_matrix(np.zeros((1, 1)))

    H_aug = sp.vstack([
        sp.hstack([H, ones_col]),
        sp.hstack([ones_row, zero_scalar])
    ], format='csc')

    b_aug = np.zeros(n + 1)
    b_aug[-1] = 1.0

    try:
        sol = spla.spsolve(H_aug, b_aug)
        a_sol = sol[:n].real
        if np.all(np.isfinite(a_sol)) and np.all(a_sol >= -1e-8):
            a_sol = np.maximum(a_sol, 0.0)
            s = a_sol.sum()
            if s > 1e-10:
                a_sol /= s
                return a_sol
    except Exception:
        pass

    # Fallback: try with slight regularization
    try:
        H_reg = H + 1e-10 * sp.eye(n, format='csc')
        H_aug_reg = sp.vstack([
            sp.hstack([H_reg, ones_col]),
            sp.hstack([ones_row, zero_scalar])
        ], format='csc')
        sol = spla.spsolve(H_aug_reg, b_aug)
        a_sol = sol[:n].real
        if np.all(np.isfinite(a_sol)) and np.all(a_sol >= -1e-8):
            a_sol = np.maximum(a_sol, 0.0)
            s = a_sol.sum()
            if s > 1e-10:
                a_sol /= s
                return a_sol
    except Exception:
        pass

    return a_init


def _kkt_refine(a_init, deadline, max_rounds=100):
    """
    Iterate: find active set -> solve KKT -> check feasibility -> add violated constraints.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a, a)
        M = float(c.max())

        # Active set: all k where c[k] is very close to M
        tol = max(1e-10, 1e-6 * M)
        active = np.where(c >= M - tol)[0]
        if len(active) > 200:
            active = np.argsort(c)[::-1][:200]

        # Solve KKT
        a_new = _active_set_qp(a, active)

        # Check if the solution is feasible (all convolution values <= M + small tol)
        c_new = _fft_convolve(a_new, a_new)
        M_new = float(c_new.max())

        sc_new = float(2.0 * n * M_new)
        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()

        # Check for violated constraints
        violations = np.where(c_new > M + tol)[0]
        if len(violations) == 0:
            # No violations, we might be at optimum
            a = a_new
            break
        else:
            # Add violated constraints to active set and redo
            new_active = np.union1d(active, violations)
            if len(new_active) > 300:
                new_active = np.argsort(c_new)[::-1][:300]
            a = _active_set_qp(a, new_active)

    return best_a, best_sc


def _local_polish(a_init, deadline, max_steps=5000):

    """
    n = len(a_init)
    a = a_init.copy()

    best_a = a.copy()

        g = _subgradient(a, active)

        w -= w.mean()
        w = np.clip(w, -60, 60)
        a = np.exp(w)
        a /= a.sum()

        if t % 10 == 0:
            sc = _score_norm(a)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a.copy()
            else:
                eta *= 0.95

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase convex optimizer:
    Phase 1: Mirror descent with exact subgradient (main optimization)
    Phase 2: KKT/active-set refinement (solve linear system)
    Phase 3: Multi-n exploration
    Phase 4: Local polishing
    Phase 5: Final KKT + mirror descent combination
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_init = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        a_init = np.ones(4096) / 4096

    n_init = len(a_init)
    best_a = a_init.copy()
    best_score = _score(a_init)
    print(f"Initial: n={n_init}, score={best_score:.10f}")

    # Phase 1: Mirror descent (0 - 800s)
    phase1_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase1_end:
        a1, sc1 = _mirror_descent(a_init, phase1_end,
                                   eta_start=0.05, eta_end=0.002,
                                   rel_tol=0.01, max_active=80,
                                   eval_interval=20, max_iter=3000000)
        full_sc = _score(a1)
        print(f"Phase 1 MD (n={n_init}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a1.copy()

    # Phase 2: KKT active-set refinement (800s - 1100s)
    phase2_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase2_end:
        a2, sc2 = _kkt_refine(best_a, phase2_end, max_rounds=200)
        full_sc = _score(a2)
        print(f"Phase 2 KKT (n={len(best_a)}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a2.copy()

    # Phase 3: Continue mirror descent with tighter params (1100s - 1350s)
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end:
        a3, sc3 = _mirror_descent(best_a, phase3_end,
                                   eta_start=0.01, eta_end=0.0005,
                                   rel_tol=0.003, max_active=50,
                                   eval_interval=20, max_iter=1000000)
        full_sc = _score(a3)
        print(f"Phase 3 MD-tight (n={len(best_a)}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a3.copy()

    # Phase 4: Multi-n exploration (1350s - 1550s)
    phase4_end = min(deadline, t_start + 1550.0)
    for n_try in [3072, 5120]:
        if time.monotonic() > phase4_end - 80:
            break
        a_try = _interp_to_length(best_a, n_try)
        phase_end = min(phase4_end, time.monotonic() + 90)
        a_opt, _ = _mirror_descent(a_try, phase_end,
                                    eta_start=0.03, eta_end=0.001,
                                    rel_tol=0.005, max_active=60,
                                    eval_interval=20, max_iter=500000)
        full_sc = _score(a_opt)
        print(f"  n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # Phase 5: KKT on best (1550s - 1700s)
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end:
        a5, sc5 = _kkt_refine(best_a, phase5_end, max_rounds=100)
        full_sc = _score(a5)
        print(f"Phase 5 KKT (n={len(best_a)}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a5.copy()

    # Phase 6: Final local polish (1700s - deadline)
    if time.monotonic() < deadline - 3:
        a6, sc6 = _local_polish(best_a, deadline - 2, max_steps=20000)
        full_sc = _score(a6)
        print(f"Phase 6 polish (n={len(best_a)}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a6.copy()

        # One more KKT pass
        if time.monotonic() < deadline - 1:
            a7, sc7 = _kkt_refine(best_a, deadline - 0.5, max_rounds=20)
            full_sc7 = _score(a7)
            print(f"Phase 6b KKT: score={full_sc7:.10f}")
            if np.isfinite(full_sc7) and full_sc7 < best_score:
                best_score = full_sc7
                best_a = a7.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
