def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_nonneg_sum1(v):
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return v / s


def _softmax_topk_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _adam_optimize(a_init, deadline, betas, lr_init, top_k, max_iter_per_beta):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_beta:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _proj_nonneg_sum1(np.exp(z - z.max()))
            c = _full_convolve(a, a)
            w = _softmax_topk_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad_z = n * a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 0.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _proj_nonneg_sum1(np.exp(z - z.max()))
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr *= 0.95

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _proj_nonneg_sum1(np.exp(z - z.max()))
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_exact(a, n_passes, deadline, top_m=200):
    """Coordinate descent with exact 1D optimization using the quadratic structure."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        n_improved = 0

        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            # v_k = a_{k-i} if 0 <= k-i < n else 0
            v = np.zeros(2 * n - 1)
            lo = max(0, i)
            hi = min(2 * n - 1, i + n)
            v[lo:hi] = a[i - i:i + n - i]  # a[0:n] shifted by i
            # Actually: v[k] = a[k-i] for 0 <= k-i < n, i.e., i <= k < i+n
            v = np.zeros(2 * n - 1)
            v[i:i+n] = a

            c = _full_convolve(a, a)
            delta = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                delta[2 * i] = 1.0

            # b_k(t) = c_k + 2*(t - a_i_old)*v_k + (t^2 - a_i_old^2)*delta_k
            # S_new = 1 + (t - a_i_old) = 1 + u where u = t - a_i_old
            # f(u) = max_k [c_k + 2*u*v_k + u*(2*a_i_old + u)*delta_k] / (1+u)^2
            # u in [-a_i_old, some_upper_bound]

            S0 = 1.0 - a_i_old  # sum without a_i

            # Find top-M active k's
            idx_top = np.argsort(c)[-top_m:]

            # Candidate u values
            u_candidates = [0.0, -a_i_old, -0.5 * a_i_old, 0.5 * a_i_old,
                           a_i_old, 2.0 * a_i_old, 0.1 * a_i_old, -0.9 * a_i_old]

            # Critical points from active constraints
            for j in idx_top:
                vj = v[j]
                cj = c[j]
                dj = delta[j]
                if vj > 1e-15:
                    if dj > 0.5:
                        denom = 1.0 - vj - a_i_old
                        if abs(denom) > 1e-10:
                            u_star = (cj - vj - a_i_old) / denom
                            if -a_i_old * 1.5 < u_star < 5.0 * max(a_i_old, 0.01):
                                u_candidates.append(u_star)
                                u_candidates.append(u_star * 0.7)
                                u_candidates.append(u_star * 1.3)
                    else:
                        u_star = 1.0 - cj / vj
                        if -a_i_old * 1.5 < u_star < 5.0 * max(a_i_old, 0.01):
                            u_candidates.append(u_star)
                            u_candidates.append(u_star * 0.7)
                            u_candidates.append(u_star * 1.3)

            u_candidates = [max(-a_i_old + 1e-15, min(3.0, u)) for u in u_candidates]
            u_candidates = list(set([round(u, 12) for u in u_candidates]))

            best_u = 0.0
            best_f = np.inf
            for u in u_candidates:
                t_new = a_i_old + u
                if t_new < 0:
                    continue
                S_new = S0 + t_new
                if S_new < 1e-15:
                    continue
                b_trial = c + 2.0 * u * v + u * (2.0 * a_i_old + u) * delta
                max_b = float(b_trial.max())
                f_val = 2.0 * n * max_b / (S_new * S_new)
                if f_val < best_f:
                    best_f = f_val
                    best_u = u

            if abs(best_u) > 1e-12:
                a[i] = a_i_old + best_u
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                n_improved += 1

            if n_improved % 500 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

        sc_end = _score_norm(a)
        if sc_end < best_score:
            best_score = sc_end

    return a, best_score


def _antisym_init(n, freq=1, alpha=0.3):
    """Uniform + anti-symmetric sine perturbation."""
    i = np.arange(n)
    a = 1.0 + alpha * np.sin(2.0 * np.pi * freq * (i + 0.5) / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _multi_freq_antisym_init(n, freqs, alphas):
    """Uniform + sum of anti-symmetric sine perturbations."""
    i = np.arange(n)
    a = np.ones(n)
    for f, al in zip(freqs, alphas):
        a += al * np.sin(2.0 * np.pi * f * (i + 0.5) / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _try_optimize(a0, deadline, tag=""):
    """Run Adam + coord descent pipeline on a single initialization."""
    n = len(a0)
    best_a = a0.copy()
    best_score = _score_norm(a0)
    print(f"  {tag}: initial score={_score(a0):.10f}")

    # Phase 1: Adam with decreasing beta
    c_curr = _full_convolve(a0, a0)
    max_cc = float(c_curr.max())

    betas = []
    beta = 0.01 * max_cc
    while beta > 0.00001 * max_cc:
        betas.append(beta)
        beta *= 0.5
    if len(betas) > 10:
        betas = betas[:10]

    phase1_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
    a1, sc1 = _adam_optimize(a0, phase1_end, betas, lr_init=0.008, top_k=80,
                             max_iter_per_beta=30000)
    sc1_full = _score(a1)
    if sc1_full < best_score:
        best_score = sc1_full
        best_a = a1.copy()
        print(f"  {tag} Phase1 (Adam): score={sc1_full:.10f}")

    # Phase 2: Adam fine-tune
    c_fine = _full_convolve(a1, a1)
    max_cf = float(c_fine.max())
    betas_fine = []
    beta = 0.002 * max_cf
    while beta > 0.000005 * max_cf:
        betas_fine.append(beta)
        beta *= 0.5
    if len(betas_fine) > 8:
        betas_fine = betas_fine[:8]

    phase2_end = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.25)
    a2, sc2 = _adam_optimize(a1, phase2_end, betas_fine, lr_init=0.004, top_k=50,
                             max_iter_per_beta=20000)
    sc2_full = _score(a2)
    if sc2_full < best_score:
        best_score = sc2_full
        best_a = a2.copy()
        print(f"  {tag} Phase2 (Adam fine): score={sc2_full:.10f}")

    # Phase 3: Coordinate descent polishing
    a3 = best_a.copy()
    phase3_end = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) * 0.35))
    a3, sc3 = _coord_descend_exact(a3, n_passes=5, deadline=phase3_end, top_m=300)
    sc3_full = _score(a3)
    if sc3_full < best_score:
        best_score = sc3_full
        best_a = a3.copy()
        print(f"  {tag} Phase3 (CoordDesc): score={sc3_full:.10f}")

    # Phase 4: Short final Adam polish
    if time.monotonic() < deadline - 10:
        c_pol = _full_convolve(a3, a3)
        max_cp = float(c_pol.max())
        betas_final = [0.0005 * max_cp, 0.0002 * max_cp, 0.0001 * max_cp]
        a4, sc4 = _adam_optimize(a3, deadline, betas_final, lr_init=0.002, top_k=30,
                                 max_iter_per_beta=15000)
        sc4_full = _score(a4)
        if sc4_full < best_score:
            best_score = sc4_full
            best_a = a4.copy()
            print(f"  {tag} Phase4 (Final Adam): score={sc4_full:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _multi_freq_antisym_init(n_parent, [1, 2, 3], [0.4, 0.2, 0.1])
        sc_parent = _score(parent)
        print(f"No parent. Using antisym init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple lengths and initialization strategies
    strategies = []

    # Strategy 1: Parent-based
    strategies.append(("parent_4096", 4096, parent))
    if len(parent) != 8192:
        strategies.append(("parent_8192", 8192, _interp_profile(parent, 8192)))
    else:
        strategies.append(("parent_4096b", 4096, _interp_profile(parent, 4096)))

    # Strategy 2: Anti-symmetric inits
    strategies.append(("antisym_f1", 4096, _antisym_init(4096, freq=1, alpha=0.5)))
    strategies.append(("antisym_f2", 4096, _antisym_init(4096, freq=2, alpha=0.4)))
    strategies.append(("antisym_multi", 4096, _multi_freq_antisym_init(4096, [1, 2, 3, 5], [0.4, 0.25, 0.15, 0.1])))
    strategies.append(("antisym_f1_8192", 8192, _antisym_init(8192, freq=1, alpha=0.5)))
    strategies.append(("antisym_multi_8192", 8192, _multi_freq_antisym_init(8192, [1, 2, 3], [0.4, 0.2, 0.1])))

    # Strategy 3: Perturbed parent
    n_p = len(parent)
    pert1 = parent + 0.01 * rng.standard_normal(n_p)
    pert1 = np.maximum(pert1, 1e-10)
    pert1 = pert1 / pert1.sum()
    strategies.append(("perturbed_parent", n_p, pert1))

    # Sort strategies: put best-scoring inits first
    def strat_key(s):
        return _score(s[2])
    strategies.sort(key=strat_key)

    # Allocate time: give more time to promising strategies
    n_strats = len(strategies)
    # Give 40% to first 2, 30% to next 2, rest split
    elapsed = time.monotonic() - t_start
    remaining = deadline - time.monotonic()

    # Group strategies by time allocation
    time_allocs = []
    if n_strats >= 1:
        time_allocs.append(remaining * 0.30)
    if n_strats >= 2:
        time_allocs.append(remaining * 0.20)
    if n_strats >= 3:
        time_allocs.append(remaining * 0.15)
    remaining_alloc = remaining - sum(time_allocs)
    for i in range(3, n_strats):
        time_allocs.append(remaining_alloc / max(1, n_strats - 3))

    for idx, (tag, n_try, a_init) in enumerate(strategies):
        if time.monotonic() > deadline - 20:
            break
        alloc = time_allocs[idx] if idx < len(time_allocs) else 30
        strat_deadline = min(deadline - 20, time.monotonic() + alloc)
        if strat_deadline <= time.monotonic() + 10:
            break

        print(f"\n=== Strategy: {tag} (n={n_try}, budget={alloc:.0f}s) ===")
        a_init = np.maximum(a_init, 0.0)
        a_init = a_init / a_init.sum()

        a_best, sc_best = _try_optimize(a_init, strat_deadline, tag=tag)
        sc_best_full = _score(a_best)
        if sc_best_full < best_score:
            best_score = sc_best_full
            best_seq = a_best.copy()
            print(f"  *** New best: score={sc_best_full:.10f} ***")

    # Final: try a few diverse restarts from the best
    if time.monotonic() < deadline - 30 and best_seq is not None:
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 20) / max(1, n_restarts)

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if ri == 0:
                p_start = a_base + 0.015 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 2048))
                p_start = np.roll(a_base, shift).copy()
            else:
                freq = rng.uniform(1.0, 5.0)
                pert = 0.02 * np.sin(2 * np.pi * freq * (np.arange(n_best) + 0.5) / n_best)
                p_start = a_base + pert

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            print(f"\n=== Restart {ri} (budget={per_restart:.0f}s) ===")
            a_r, sc_r = _try_optimize(p_start, time.monotonic() + per_restart, tag=f"restart{ri}")
            sc_r_full = _score(a_r)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = a_r.copy()
                print(f"  *** New best from restart: score={sc_r_full:.10f} ***")

    # Final output
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL VERIFY: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        # Double check with evaluate_sequence
        try:
            ev_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {ev_score:.10f}")
        except Exception as e:
            print(f"Warning: evaluate_sequence check failed: {e}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
