"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses power-mean smoothing with L-BFGS in log-space, DFT variance regularization,
symmetric optimization, and coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -80, 80))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lbgfs_optimize(a_init, p, deadline, dft_reg=0.0, max_iter=500):
    """Run L-BFGS-B in log-space on the power-mean objective."""
    n = len(a_init)
    a_init = np.maximum(a_init, 1e-15)
    z0 = np.log(a_init) - np.log(a_init).mean()

    def obj(z):
        val, _ = _power_mean_score_and_grad(z, p, n, dft_reg)
        return val

    def grad(z):
        _, g = _power_mean_score_and_grad(z, p, n, dft_reg)
        return g

    result = minimize(
        obj, z0, jac=grad, method='L-BFGS-B',
        options={'maxiter': max_iter, 'gtol': 1e-8, 'maxls': 20,
                 'functol': 1e-12, 'iprint': -1}
    )
    a_out = _softmax(result.x)
    a_out = np.maximum(a_out, 0.0)
    a_out = a_out / a_out.sum()
    return a_out


def _lbgfs_symmetric(a_init, p, deadline, dft_reg=0.0, max_iter=300):
    """L-BFGS on symmetric sequence (only first half is free)."""
    n = len(a_init)
    half = (n + 1) // 2
    a_init = np.maximum(a_init, 1e-15)

    # Extract first half
    z0_full = np.log(a_init) - np.log(a_init).mean()
    z0 = z0_full[:half].copy()

    def _full_from_half(zh):
        # Symmetrize: a[i] = a[n-1-i]
        z_full = np.zeros(n)
        z_full[:half] = zh
        z_full[half:] = zh[:n - half][::-1]
        return z_full

    def obj(zh):
        z_full = _full_from_half(zh)
        val, _ = _power_mean_score_and_grad(z_full, p, n, dft_reg)
        return val

    def grad(zh):
        z_full = _full_from_half(zh)
        _, g_full = _power_mean_score_and_grad(z_full, p, n, dft_reg)
        # Fold gradient back to half
        gh = g_full[:half].copy()
        if n - half > 0:
            gh += g_full[half:][::-1]
        return gh

    result = minimize(
        obj, z0, jac=grad, method='L-BFGS-B',
        options={'maxiter': max_iter, 'gtol': 1e-8, 'maxls': 20,
                 'functol': 1e-12, 'iprint': -1}
    )

    z_full = _full_from_half(result.x)
    a_out = _softmax(z_full)
    a_out = np.maximum(a_out, 0.0)
    a_out = a_out / a_out.sum()
    return a_out


def _coord_descend(a, n_passes=2, deadline=None):
    """Coordinate descent with exact max evaluation."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_sc = _score_norm(a)

    if deadline is None:
        deadline = time.monotonic() + 60

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]
            # Try different values of a[i]
            # The convolution changes as: c_j_new = c_j + (t - a_i) * a_{j-i}
            c = _fft_convolve(a, a)

            # Build delta vector: d_j = a_{j-i} for valid j
            d = np.zeros(2 * n - 1)
            for j in range(i, min(i + n, 2 * n - 1)):
                d[j] = a[j - i]

            S0 = 1.0 - a_i_old  # sum of rest (a is normalized to sum=1)

            # Objective: 2n * max_j (c_j - a_i*d_j + t*d_j) / (S0+t)^2
            # = 2n * max_j (c_j + d_j*(t - a_i)) / (S0+t)^2

            def f_of_t(t):
                c_trial = c + d * (t - a_i_old)
                max_c = float(c_trial.max())
                denom = (S0 + t) ** 2
                if denom < 1e-30:
                    return np.inf
                return 2.0 * n * max_c / denom

            # Golden section search on [0, 5*a_i_old + 0.01]
            lo = 0.0
            hi = max(5.0 * a_i_old, 0.01)

            result = minimize_scalar(f_of_t, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-10, 'maxiter': 50})

            if result.success and result.fun < best_sc:
                a[i] = result.x
                a = a / a.sum()
                best_sc = _score_norm(a)

    return a, best_sc

                print(f"  * IMPROVED")
            else:
                print()

    # === Phase 6: Final coordinate descent on best ===
    print(f"\n=== Phase 6: Final polish ===")
    if time.monotonic() < deadline - 5:
        a_final, sc_final = _coord_descend(best_a, n_passes=1, deadline=deadline)
        sc_final_full = _score(a_final)
        if sc_final_full < best_score:
            best_score = sc_final_full
            best_a = a_final.copy()
            print(f"  Final CD: score={sc_final_full:.10f} * IMPROVED")
        else:
            print(f"  Final CD: score={sc_final_full:.10f} (no improvement)")

    # === Final output ===
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_a)} ===")
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (1.0 / 256)
    # Scale to ensure sum > 0.01
    if s < 0.01:
        final = final * (1.0 / s)
    sc_final = _score(final)
    print(f"Output: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
