def simple_convolve(a):
    """Compute full linear self-convolution"""
    n = len(a)
    out_n = 2 * n - 1
    fa = np.fft.rfft(a, 2 * n)
    fb = np.fft.rfft(a, 2 * n)
    fc = np.fft.irfft(fa * fb)
    return fc[:out_n] if len(fc) > out_n else fc[:out_n+1]

def compute_score(a):
    """Compute evaluationscore"""
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    conv = simple_convolve(a)
    max_conv = float(conv.max())
    return 2 * len(a) * max_conv / (s ** 2)
```

def perturb_gaussian(a, variance=0.15):
    """Gaussian perturbation"""
    n = len(a)
    z = np.log(np.clip(a, 1e-15, 1000))
    z += variance * np.random.standard_normal(n)
    z -= np.mean(z)
    return np.exp(z)

def perturb_fourier(a, modes=3):
    """Fourier-based perturbation"""
    n = len(a)
    base = np.linspace(0, 1, n)
    z = np.log(np.clip(a, 1e-15, 1000))
    z -= np.mean(z)
    for _ in range(modes):
        freq = int(np.random.randint(1, n//4))
        shift = np.random.uniform(0.5, 3.0)
        phi = np.random.uniform(0, 2 * np.pi)
        z[:n] += shift * np.cos(2 * np.pi * freq * base[:n] + phi)
    return np.exp(z)

def perturb_block(a, blocks=4):
    """Block-based perturbation"""
    n = len(a)
    indices = np.random.choice(range(n), blocks)
    for i in range(blocks):
        start = indices[i] - 10
        end = indices[i] + 10
        if start < 0: start = 0
        if end > n: end = n
        factor = np.random.uniform(0.5, 1.5)
        a[start:end] *= factor
        a *= 1 / a.sum()
    return a
```
