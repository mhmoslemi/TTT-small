"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Simplicial subgradient with Nesterov momentum, (2) L-BFGS smoothing,
(3) multi-n exploration, (4) simulated annealing restarts, (5) peak-shaving polish.
Key novelty: direct simplex projection (no log-softmax) for subgradient phase,
exponential peak weighting with adaptive temperature, and frequency-domain
informed perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(max(out_n, 2))
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    """Project v onto probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _peaked_weight(c, temp_frac=0.03):
    """Exponential peak weighting: heavily favors top peaks."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.ones_like(c) / len(c)
    temp = max(temp_frac * c_max, 1e-15)
    shifted = (c - c_max) / temp
    e = np.exp(np.clip(shifted, -50.0, 0.0))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _subgrad_from_weights(w, a, n):
    """Compute subgradient of weighted sum of autocorrelation peaks.
    Returns gradient w.r.t. a (before projection).
    grad[i] = sum_k w[k] * (a[k-i] + a[i-k]) for valid indices.
    """
    N = _next_pow2(3 * n + 10)
    w_fft = np.fft.rfft(w, N)
    a_fft = np.fft.rfft(a, N)

    # S2[i] = sum_j w[i-j] * a[j] = (w * a)[i] for i in 0..n-1
    conv_wa = np.fft.irfft(w_fft * a_fft, N)
    S2 = conv_wa[:n]

    # S1[i] = sum_j w[i+j] * a[j]
    # Let w_rev[m] = w[2n-2-m], then S1[i] = (w_rev * a)[2n-2-i]
    w_rev = w[::-1]
    w_rev_fft = np.fft.rfft(w_rev, N)
    conv_wra = np.fft.irfft(w_rev_fft * a_fft, N)
    S1 = conv_wra[n - 1:2 * n - 1][::-1]

    return S1 + S2


def _subgrad_nesterov(a_init, deadline, n_max_steps=20000, lr_init=0.015,
                      temp_start=0.08, temp_end=0.01, momentum=0.9):

    Directly operates on the simplex (no reparameterization).

    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    v = np.zeros_like(a)
    lr = lr_init
    patience = 0
    mu = momentum

    for i in range(n_max_steps):
        if time.monotonic() > deadline:

        y = a + mu * v

        y = _proj_simplex(y)

        c = _full_convolve(y, y)
        c_max = float(c.max())
        if c_max < 1e-30:
            break

        w = _peaked_weight(c, temp)

        # Project gradient to tangent space (orthogonal to 1)

def _smoobj_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_a = _subgrad_from_weights(w, a, n)
    ga = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - ga)
    return grad_z


def _lbfgs_smooth(a_init, deadline, min_remaining=30.0):
    """L-BFGS on smoothed objective with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    betas = base_beta * np.array([1.0, 0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - elapsed
        if remaining < 5:
            break

        est_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_per_iter * 5))))

        try:
            result = minimize(
                _smoobj_fn, z, args=(beta, n),
                jac=_smograd_fn,
                method='L-BFGS-B',

            sc_new = _score(a_new)

                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

        except Exception:
            pass

            for _ in range(n_modes):
                freq = rng.integers(1, max(2, n // 8))
                amp = T * 3.0 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2 * np.pi)
                t_arr = np.arange(n)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
            z = z - z.mean()
            a_try = _softmax(z)
        else:
            # Block swap
            a_try = a_current.copy()
            n_blocks = rng.integers(1, 4)
            for _ in range(n_blocks):
                block_size = max(4, n // rng.integers(4, 20))
                if block_size >= n // 2:
                    continue
                i = rng.integers(0, n - 2 * block_size)
                j = rng.integers(0, n - 2 * block_size)
                if abs(i - j) < block_size:
                    continue
                tmp = a_try[i:i + block_size].copy()
                a_try[i:i + block_size] = a_try[j:j + block_size]
                a_try[j:j + block_size] = tmp
            a_try = _proj_simplex(a_try)

        sc_try = _score(a_try)
        delta = sc_try - sc_current

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a_current = a_try
            sc_current = sc_try
            if sc_try < sc_best:
                sc_best = sc_try
                a_best_local = a_try.copy()
                print(f"    SA new best: {sc_best:.10f} at iter {i}, T={T:.6f}")

        T = max(T * cooldown, T_end)

    return a_best_local, sc_best


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * 3.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "super_tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * 5.0 * t) * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _peak_shave(a_init, deadline, n_iter=200):
    """Targeted peak shaving: iteratively reduce the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    lr = 0.01
    for i in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Use very tight peak weighting (only top few peaks)
        w = _peaked_weight(c, temp_frac=0.005)
        grad = _subgrad_from_weights(w, a, n)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        step = lr / (1.0 + 0.005 * i)
        a_try = a - step * (grad / gnorm)
        a_try = _proj_simplex(a_try)

        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: subgradient+Nesterov, L-BFGS, multi-n, annealing, peak-shave.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent, using tapered n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Subgradient + Nesterov on parent ===
    print(f"\n[Phase 1] Subgradient+Nesterov (t={time.monotonic()-t_start:.1f}s)")
    sg_deadline = min(deadline - 600, t_start + 400)
    a_sg, sc_sg = _subgrad_nesterov(parent, sg_deadline, n_max_steps=30000,
                                    lr_init=0.02, temp_start=0.06, temp_end=0.008,
                                    momentum=0.85)
    update_best(a_sg, sc_sg, "subgrad_nesterov")

    # === Phase 2: L-BFGS polish from Phase 1 result ===
    print(f"\n[Phase 2] L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        lbfgs_deadline = min(deadline - 400, time.monotonic() + 250)
        a_lb = _lbfgs_smooth(best_seq, lbfgs_deadline, min_remaining=200)
        update_best(a_lb, _score(a_lb), "lbfgs_phase2")

    # === Phase 3: Peak shaving ===
    print(f"\n[Phase 3] Peak shaving (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        ps_deadline = min(deadline - 300, time.monotonic() + 100)
        a_ps, sc_ps = _peak_shave(best_seq, ps_deadline, n_iter=500)
        update_best(a_ps, sc_ps, "peak_shave")

    # === Phase 4: Multi-n exploration ===
    print(f"\n[Phase 4] Multi-n exploration (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192]
    # Remove n_parent if already covered
    n_candidates = [x for x in n_candidates if x != n_parent] + [n_parent]
    # Sort by closeness to parent
    n_candidates.sort(key=lambda x: abs(x - n_parent))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        # Initialize
        if n_try == n_parent:
            a_init = best_seq.copy() if best_seq is not None else parent.copy()
        else:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            a_init = _proj_simplex(a_init)

        sc_init = _score(a_init)
        print(f"    n={n_try}: init score={sc_init:.10f}")

        # Quick subgradient
        algo_deadline = min(deadline - 80, time.monotonic() + 120)
        a_opt, sc_opt = _subgrad_nesterov(a_init, algo_deadline, n_max_steps=15000,
                                          lr_init=0.015, temp_start=0.05,
                                          temp_end=0.005, momentum=0.8)
        update_best(a_opt, sc_opt, f"subgrad_n{n_try}")

        # Quick L-BFGS
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 60:
            lb_deadline = min(deadline - 60, time.monotonic() + 80)
            a_opt2 = _lbfgs_smooth(a_opt, lb_deadline, min_remaining=30)
            update_best(a_opt2, _score(a_opt2), f"lbfgs_n{n_try}")

    # === Phase 5: Try different profiles at best n ===
    print(f"\n[Phase 5] Profile diversity (t={time.monotonic()-t_start:.1f}s)")
    n_best = len(best_seq) if best_seq is not None else 4096
    profiles = ["cos_mod", "super_tapered", "ramp"]
    for prof in profiles:
        if time.monotonic() > deadline - 60:
            break
        a_prof = _make_profile(n_best, prof)
        # Blend with best
        blend = 0.3 * a_prof + 0.7 * (best_seq / best_seq.sum())
        blend = _proj_simplex(blend)
        sc_blend = _score(blend)
        if sc_blend < best_score:
            # Optimize from this point
            algo_deadline = min(deadline - 40, time.monotonic() + 60)
            a_opt, sc_opt = _subgrad_nesterov(blend, algo_deadline, n_max_steps=8000,
                                              lr_init=0.01, temp_start=0.04,
                                              temp_end=0.005, momentum=0.8)
            update_best(a_opt, sc_opt, f"profile_{prof}")

    # === Phase 6: Simulated annealing ===
    print(f"\n[Phase 6] Simulated annealing (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        sa_deadline = min(deadline - 20, time.monotonic() + 200)
        a_sa, sc_sa = _simulated_annealing(best_seq, sa_deadline, rng, n_iter=8000)
        update_best(a_sa, sc_sa, "annealing")

    # === Phase 7: Final L-BFGS + peak shave ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Final L-BFGS with very small beta
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_fin = mb * 0.0005
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            max_iter = min(1000, max(30, int(remaining / 0.05)))
            try:
                result = minimize(
                    _smoobj_fn, z, args=(beta_fin, n),
                    jac=_smograd_fn,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 25}
                )
                a_fin = _softmax(result.x)
                a_fin = a_fin / a_fin.sum()
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final peak shave
        if time.monotonic() < deadline - 3:
            a_ps2, sc_ps2 = _peak_shave(best_seq, deadline - 3, n_iter=100)
            update_best(a_ps2, sc_ps2, "final_peak_shave")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
