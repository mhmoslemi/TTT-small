Uses tracking gradient descent with adaptive steps, peak-deflation, and multi-start.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _conv_grad_at_k(a, k):
    """Gradient of conv(a,a)[k] with respect to a.
    conv(a,a)[k] = sum_i a[i]*a[k-i]
    d/da[j] = a[k-j] (for valid k-j)
    """
    n = len(a)
    g = np.zeros(n)
    # For j in [0, n-1], the gradient component is a[k-j] if 0 <= k-j <= n-1
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        g[j_start:j_end+1] = a[k-j_end:k-j_start+1][::-1]
    return 2.0 * g  # Factor of 2 from the quadratic form

def _tracking_descent(a, deadline, max_iter, init_lr=0.01):
    """Gradient descent on simplex with tracking of the argmax."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    lr = init_lr
    prev_c_max = None
    prev_grad = None
    prev_a = None

        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))

        top_k = 5
        idx = np.argsort(c)[::-1][:top_k]

            grad += w * _conv_grad_at_k(a, k)

            y = grad - prev_grad
            ss = float(np.dot(s, s))
            sy = float(np.dot(s, y))

def _peak_deflation(a, deadline, max_iter):
    """Specifically target the peak of the convolution."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for it in range(max_iter):

        c_max = float(c.max())
        k_star = int(np.argmax(c))
        
        # The peak is at k_star. To reduce it, we want to reduce
        # the product a[i]*a[k_star-i] for the terms that contribute most.
        # Compute the "contribution" of each pair
        i_start = max(0, k_star - n + 1)
        i_end = min(k_star, n - 1)
        if i_start > i_end:
            break
        products = a[i_start:i_end+1] * a[k_star-i_end:k_star-i_start+1][::-1]
        
        # Find the top contributors
        top_idx = np.argsort(products)[::-1][:max(1, len(products)//4)]
        top_local = top_idx + i_start
        
        # Reduce a at those positions (and their reflections)
        # by redistributing to positions that don't contribute to the peak
        reduce_mask = np.zeros(n, dtype=bool)
        for i in top_local:
            reduce_mask[i] = True
            if 0 <= k_star - i <= n - 1:
                reduce_mask[k_star - i] = True
        
        # Find positions NOT in the reduce mask where we can add
        add_mask = ~reduce_mask
        
        # Do a small perturbation: move mass from reduce to add positions
        if reduce_mask.any() and add_mask.any():
            mass_to_move = 0.01 * a[reduce_mask].sum()
            a[reduce_mask] -= mass_to_move * a[reduce_mask] / max(a[reduce_mask].sum(), 1e-30)
            a[reduce_mask] = np.maximum(a[reduce_mask], 0)
            a[add_mask] += mass_to_move * a[add_mask] / max(a[add_mask].sum(), 1e-30)
            a = a / a.sum()
        
        sc = _score_norm(a)
        if sc < best_score:

        if t % 50 == 0:
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
        
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)
    
    sc_final = _score_norm(a_curr)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _softmax_z(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lbfgs_smoothed(a_init, deadline, beta):
    """L-BFGS on smoothed objective."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    best_score = _score_norm(a_init)
    best_z = z0.copy()
    
    def obj_fn(z):
        a = _softmax_z(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))
    
    def grad_fn(z):
        a = _softmax_z(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)
    
    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        return a_init, best_score
    
    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 25)), 'gtol': 1e-14, 'maxls': 10}
        )
        z_opt = result.x
        a_opt = _softmax_z(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    a_best = _softmax_z(best_z)
    return a_best, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = t + 0.1
        return a / a.sum()
    elif kind == "asymmetric_bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.4) ** 2) / (2 * 0.25 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "step":
        a = np.zeros(n)
        a[n//4:] = 1.0
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_z(z)
