Now, for the objective J = 2*n*max(c)/(sum(a))^2, if we fix sum(a) = 1, we minimize 2*n*max(c).

The gradient of max(c) is the gradient of c(k*) where k* = argmax(c).

For the top-k variant, the gradient is the sum of gradients of the top k values of c.

This gives us a very efficient way to compute the gradient! We don't need FFT for the gradient - we just need to know which lags are at the peak.

**Algorithm: "Peak Gradient Descent"**

1. Normalize a to sum = 1
2. Compute c = conv(a, a)
3. Find the top k peaks: k_1, k_2, ..., k_k
4. Compute gradient: grad[m] = sum_i 2*a(k_i - m) (for valid k_i - m)
5. Take a step: a -= lr * grad (in log-space or with projection)
6. Re-normalize
7. Repeat

This is O(n*k) per iteration (for the gradient) + O(n log n) for the FFT (for computing c).

For the log-space version:
- Work with z = log(a)
- The gradient of the objective with respect to z is: grad_z[m] = a[m] * grad_a[m]
- Use Adam on z

This should be quite efficient! And it's different from the previous approach which used the full softmax-smoothed convolution.

1. **Learning rate schedule**: Start with a larger lr, decrease over time.
2. **Peak selection**: Start with k=1 (just the max), increase to k=5, 10 as we refine.
3. **Regularization**: Add a small regularization to prevent the sequence from becoming too sparse.
4. **Momentum**: Use Adam or Nesterov momentum.
5. **Clipping**: Clip z to prevent extreme values.

Peak-gradient descent optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direct gradient of top-k convolution peaks in log-space with Adam,
active-set peak management, asymmetric initializations, and multi-length

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _peak_grad_active(c, a, active_indices):
    """
    Compute gradient of sum_{k in active} c(k) w.r.t. a.
    grad[m] = 2 * sum_{k active} a(k-m) for valid k-m.
    """
    n = len(a)
    grad = np.zeros(n)
    for ki in active_indices:
        ki = int(ki)
        if ki < 0 or ki >= 2 * n - 1:
            continue
        m_low = max(0, ki - (n - 1))
        m_high = min(n - 1, ki)
        if m_low <= m_high:
            indices = np.arange(m_low, m_high + 1)
            grad[m_low:m_high + 1] += 2.0 * a[ki - indices]
    return grad


def _soft_topk_grad(a, k, temperature):
    """
    Compute smooth gradient using softmax-weighted top-k peaks.
    Returns (objective_value, gradient_wrt_a).
    """
    n = len(a)
    c = _full_convolve(a, a)
    
    # Softmax weights over all c values (with temperature)
    c_max = float(c.max())
    shifted = (c - c_max) / temperature
    exps = np.exp(shifted)
    s = float(exps.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exps / s
    
    obj = float(np.dot(w, c))
    
    # Gradient: sum_k w[k] * d(c[k])/d(a[m])
    grad = np.zeros(n)
    # Only compute for lags with significant weight
    threshold = w.max() * 1e-4
    active = np.where(w > threshold)[0]
    for ki in active:
        ki = int(ki)
        wk = w[ki]
        m_low = max(0, ki - (n - 1))
        m_high = min(n - 1, ki)
        if m_low <= m_high:
            indices = np.arange(m_low, m_high + 1)
            grad[m_low:m_high + 1] += 2.0 * wk * a[ki - indices]
    
    return obj, grad

def _make_zadoff_chu_like(n, rng):
    """
    Create a Zadoff-Chu-inspired real sequence.
    Zadoff-Chu sequences have perfect periodic autocorrelation.
    We take the real part of a Zadoff-Chu sequence and rectify.
    """
    root = 1
    m = n
    t = np.arange(n)
    # Zadoff-Chu: x(k) = exp(-j*pi*root*k*(k+1)/m) for root odd
    phase = -np.pi * root * t * (t + 1) / m
    x_real = np.cos(phase)
    x_imag = np.sin(phase)
    # Use real part, rectify
    a = np.maximum(x_real, 0.0) + 0.1
    a = a / a.sum()
    # Add random perturbation
    a = a + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.001)
    return a / a.sum()

    block_size = max(1, n // n_blocks)
    rng = np.random.default_rng(7)

        # Pick a random block

        start = b * block_size
        end = min(n, (b + 1) * block_size)
        
        # Try different multipliers for this block
        best_block_score = best_score
        best_block_mult = 1.0
        
        a_curr = best_a.copy()
        for _ in range(50):
            # Random perturbation of the block
            mult = 1.0 + 0.1 * rng.standard_normal(end - start)
            mult = np.clip(mult, 0.5, 1.5)
            a_try = a_curr.copy()
            a_try[start:end] *= mult
            a_try = np.maximum(a_try, 0.0)

            if s > 1e-15:

                    a_curr = a_try.copy()

            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient of smoothed max on simplex
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 500 == 0:
            sc = _compute_score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _compute_score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    z = np.clip(z, -30, 30)
    a_new = np.exp(z)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    best_seq = None
    best_score = np.inf
    
    if parent is not None:
        best_seq = parent.copy()
        best_score = _compute_score(parent)
        print(f"Parent: n={len(parent)}, score={best_score:.10f}")
    else:
        n0 = 4096
        a0 = np.ones(n0) / n0
        best_seq = a0.copy()
        best_score = _compute_score(a0)
        print(f"No parent, uniform n={n0}, score={best_score:.10f}")

    # ============================================================

    a_work = best_seq.copy()

    t_remaining = deadline - time.monotonic() - 120.0

    if t_remaining > 60:
        schedule_coarse = [
            {'k': 1, 'lr': 0.3, 'temperature': 0.01, 'max_iter': 50000},
            {'k': 1, 'lr': 0.1, 'temperature': 0.005, 'max_iter': 50000},
            {'k': 3, 'lr': 0.05, 'temperature': 0.003, 'max_iter': 50000},

            print(f"  After coarse: score={sc_full:.10f}")

        t_rem2 = deadline - time.monotonic() - 90.0
        schedule_fine = [
            {'k': 3, 'lr': 0.03, 'temperature': 0.002, 'max_iter': 80000},
            {'k': 5, 'lr': 0.01, 'temperature': 0.001, 'max_iter': 80000},
            {'k': 5, 'lr': 0.005, 'temperature': 0.0005, 'max_iter': 80000},
            {'k': 10, 'lr': 0.002, 'temperature': 0.0003, 'max_iter': 80000},

        a_work, sc = _logspace_adam_peak(a_work, dl2, schedule_fine)

            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  After fine: score={sc_full:.10f}")
    
    # Phase 1c: Ultra-fine
    if time.monotonic() < deadline - 60:
        t_rem3 = min(120.0, deadline - time.monotonic() - 60.0)
        schedule_ultra = [
            {'k': 5, 'lr': 0.003, 'temperature': 0.0002, 'max_iter': 100000},
            {'k': 10, 'lr': 0.001, 'temperature': 0.0001, 'max_iter': 100000},
            {'k': 20, 'lr': 0.0005, 'temperature': 0.00005, 'max_iter': 100000},
        ]
        dl3 = time.monotonic() + t_rem3
        a_work, sc = _logspace_adam_peak(a_work, dl3, schedule_ultra)
        sc_full = _compute_score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  After ultra-fine: score={sc_full:.10f}")
    
    # ============================================================
    # PHASE 2: FISTA refinement (different optimization landscape)
    # ============================================================
    if time.monotonic() < deadline - 60:

        a_fista = best_seq.copy()
        c_curr = _full_convolve(a_fista, a_fista)

            schedule_alt = [
