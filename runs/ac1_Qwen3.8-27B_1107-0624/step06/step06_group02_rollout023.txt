import numpy as np
import time
from scipy.optimize import minimize

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k

def safe_convolve(a, b):
    n, m = len(a), len(b)
    N = next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()

def lc_p(a, p):
    """Compute L_p norm of a*a"""
    n = len(a)
    v = safe_convolve(a, a)
    if p == np.inf:
        return 2.0 * n * v.max()
    return 2.0 * n * np.sum(v**p)**(1.0/p)

def lc_p_grad_z(z, p):
    """Compute gradient of 2*n*||a*a||_p with respect to z, where a=softmax(z)"""
    n = len(z)
    a = softmax(z)
    v = safe_convolve(a, a)  # length 2n-1
    
    if p == 2:
        # Special case: ||v||_2 = sqrt(sum(v^2)), grad w.r.t. a_j = 2*n * (1/2) * (sum v^2)^(-1/2) * 2 * sum_k v[k] * (a[j+k]+a[j-k])
        # = 2*n * (sum v^2)^(-1/2) * sum_k v[k] * (a[j+k]+a[j-k])
        u = v  # v^(p-1) = v^1 = v for p=2
        C = 2.0 * n / np.sqrt(np.sum(v**2))
    else:
        u = v**(p-1)
        C = 2.0 * n * np.sum(v**p)**(1.0/p - 1.0)
    
    # g_j = C * sum_k u[k] * (a[j+k] + a[j-k])
    # term2_j = sum_k u[k] * a[j-k] = (a * u)[j] for j=0..n-1
    # term1_j = sum_k u[k] * a[j+k] = (a_rev * u)[n-1-j] for j=0..n-1
    
    a_rev = a[::-1]
    N_conv = len(a) + len(u) - 1  # n + (2n-1) - 1 = 3n-2
    Nf = next_pow2(N_conv)
    
    A = np.fft.rfft(a, Nf)
    U = np.fft.rfft(u, Nf)
    Ar = np.fft.rfft(a_rev, Nf)
    
    conv_au = np.fft.irfft(A * U, Nf)[:N_conv]
    conv_ar_u = np.fft.irfft(Ar * U, Nf)[:N_conv]

    term1 = conv_ar_u[n-1::-1][:n]  # (a_rev*u)[n-1-j] for j=0..n-1
    
    g = C * (term1 + term2)
    
    # Convert to z-space gradient
    grad_z = a * (g - np.dot(g, a))
    return grad_z
