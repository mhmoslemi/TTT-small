        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient descent with line search, soft-gradient Adam,
multi-scale exploration (n=4096 -> n=8192), and diverse restarts.
Key improvement over parent: exact subgradient with proper line search
for faster local convergence, and exploration at n=8192.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _exact_subgradient(p, c, top_k=3):
    """Exact subgradient of max at top-k peak positions."""
    n = len(p)
    k_indices = np.argsort(c)[::-1][:top_k]
    g = np.zeros(n)
    for k_idx in k_indices:
        j_start = max(0, k_idx - n + 1)
        j_end = min(n - 1, k_idx)
        g[j_start:j_end+1] += 2.0 * p[k_idx - j_end : k_idx - j_start + 1][::-1]
    g /= top_k
    return g


def _soft_weights(c, T):
    """Compute softmax weights for the convolution values."""
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return w


def _soft_gradient_fft(p, w):
    """Soft gradient using FFT. w is the weight vector (length 2n-1)."""
    n = len(p)
    conv_len = 2 * n - 1
    p_ext = np.zeros(conv_len)
    p_ext[:n] = p
    p_ext_rev = p_ext[::-1]
    conv = _fft_convolve(w, p_ext_rev)
    return 2.0 * conv[conv_len - 1 : 2 * conv_len - 1]


def _line_search_subgradient(p, g, max_iter=50):
    """
    Find best step: p_new = project_simplex(p - alpha * g)
    Minimize _score_norm(p_new) over alpha.
    Returns (p_new, score_new, alpha).
    """
    best_p = p.copy()
    best_sc = _score_norm(p)
    best_alpha = 0.0
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-15:
        return best_p, best_sc, 0.0
    # Normalize gradient for step size
    g = g / g_norm
    # Try a range of alphas
    alphas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for alpha in alphas:
        p_candidate = _project_simplex(p - alpha * g)
        sc = _score_norm(p_candidate)
        if sc < best_sc:
            best_sc = sc
            best_p = p_candidate
            best_alpha = alpha
    # If no improvement with these, try smaller steps
    if best_alpha == 0.0:
        for alpha in [0.0001, 0.0005, 0.001, 0.002]:
            p_candidate = _project_simplex(p - alpha * g)
            sc = _score_norm(p_candidate)
            if sc < best_sc:
                best_sc = sc
                best_p = p_candidate
                best_alpha = alpha
    return best_p, best_sc, best_alpha
