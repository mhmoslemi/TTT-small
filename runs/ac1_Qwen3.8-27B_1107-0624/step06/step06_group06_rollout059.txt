import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = c.max()
    sh = (c - cm) / beta
    e = np.exp(np.clip(sh, -700, 0))

def _project_simplex(v):
    n = len(v)
    if n <= 1:
        return np.maximum(v, 0) / max(1.0, np.sum(np.maximum(v, 0)))
    u = np.sort(v)[::-1]
    csv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho = np.nonzero(u * idx > (csv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (csv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_grad_a(a, w):
    """Gradient of weighted sum of convolution wrt a. Returns 2*(P_w a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _pg_bb(a_init, deadline, min_time_remaining=5.0):
    """Projected gradient with BB step in a-space, adaptive smoothing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # Beta schedule: geometric from moderate to very small
    betas = c_max * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005])

    prev_grad = None
    prev_a = None

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break

        max_iter = 500
        for it in range(max_iter):
            if time.monotonic() > deadline - min_time_remaining:
                break

            c = _full_convolve(a, a)
            c_max = c.max()
            sh = (c - c_max) / beta
            e = np.exp(np.clip(sh, -700, 0))
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _conv_grad_a(a, w)
            # Project gradient onto tangent space of simplex
            grad_t = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_t)
            if gnorm < 1e-16:
                break

            # BB step
            if prev_grad is not None and prev_a is not None:
                s_vec = a - prev_a
                y_vec = grad_t - prev_grad
                yy = float(np.dot(y_vec, y_vec))
                if yy > 1e-16:
                    sy = float(np.dot(s_vec, y_vec))
                    alpha = max(sy / yy, 1e-12)
                    alpha = min(alpha, 5.0)
                else:
                    alpha = 1.0 / gnorm
            else:
                alpha = 1.0 / (gnorm + 1e-12)
                alpha = min(alpha, 5.0)

            # Backtracking
            f_old = c_max
            a_try = a - alpha * grad_t
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            f_try = c_try.max()

            backtrack = 0
            while f_try > f_old - 1e-14 * alpha * gnorm and backtrack < 20:
                alpha *= 0.5
                a_try = a - alpha * grad_t
                a_try = _project_simplex(a_try)
                c_try = _full_convolve(a_try, a_try)
                f_try = c_try.max()
                backtrack += 1

            prev_a = a.copy()
            prev_grad = grad_t.copy()
            a = a_try

            # Update best
            sc = 2.0 * n * a_try.max() if False else _score_fast(a_try, c_try)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_try.copy()

        # Decay beta
        prev_grad = None
        prev_a = None

    return best_a, best_sc


def _score_fast(a, c):
    n = len(a)
    s = float(a.sum())
    return float(2.0 * n * c.max() / (s * s))


def _run_lbfgs_phased(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with aggressive multi-phase beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    # Sort convolution to determine gap
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[19]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    # More aggressive beta schedule with many small-beta phases
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):

                _lse_obj, z, args=(beta, n),
                jac=_lse_grad,

                    'maxcor': 50,
                    'maxls': 100,

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

def _active_set_newton(a_init, deadline, max_iter=200):
    """Newton method to equalize the top-k active convolution positions."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Identify active set: positions within 1% of max
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Compute gradient of the "average active" objective
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        grad = _conv_grad_a(a, w)
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-14:
            break

        # Newton-like step with backtracking
        # Approximate Hessian action via finite difference of gradient
        # (expensive but accurate for final iterations)
        alpha = 1.0 / gnorm
        alpha = min(alpha, 2.0)

        f_old = c_max
        success = False
        for ls in range(40):
            a_try = a - alpha * grad_t
            a_try = _project_simplex(a_try)
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())

            if f_try < f_old - 1e-12 * alpha * gnorm * 0.1:
                success = True
                a = a_try
                break
            alpha *= 0.5

        if not success:
            # Try a smaller step
            a_try = a - 0.01 * grad_t / gnorm
            a_try = _project_simplex(a_try)
            sc_try = _score(a_try)
            if sc_try < _score(a):
                a = a_try
            else:
                break

        sc = _score(a)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bullseye":
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

    a_main, sc_main = _run_lbfgs_phased(parent, deadline, min_time_remaining=300.0)
    update_best(a_main, sc_main, "lbfgs_main")
    print(f"  Phase 1 done: {sc_main:.10f}, time={time.monotonic()-t1:.1f}s")

    # === Phase 2: Projected Gradient BB in a-space ===
    print(f"\n[Phase 2] PG-BB in a-space")
    t2 = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_pg, sc_pg = _pg_bb(best_seq, deadline, min_time_remaining=150.0)
        update_best(a_pg, sc_pg, "pg_bb")
    print(f"  Phase 2 done, time={time.monotonic()-t2:.1f}s")

    # === Phase 3: Active Set Newton ===
    print(f"\n[Phase 3] Active Set Newton")
    t3 = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_as, sc_as = _active_set_newton(best_seq, min(deadline - 50, time.monotonic() + 120), max_iter=300)
        update_best(a_as, sc_as, "active_set")
    print(f"  Phase 3 done, time={time.monotonic()-t3:.1f}s")

    # === Phase 4: Re-run L-BFGS from best with even smaller beta ===
    print(f"\n[Phase 4] Fine L-BFGS polish")
    t4 = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        # Very small betas for final precision
        fine_betas = mb * np.array([0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])
        for beta in fine_betas:
            if time.monotonic() > deadline - 40:
                break
            try:
                result = minimize(
                    _lse_obj, z, args=(beta, n),
                    jac=_lse_grad,
                    method='L-BFGS-B',
                    options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 60, 'maxls': 100}
                )
                a_new = _softmax(result.x)
                sc_new = _score(a_new)
                if sc_new < best_score - 1e-15:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    z = np.log(np.maximum(a_new, 1e-30))
                    z = z - z.mean()
                    print(f"    Fine L-BFGS improve: {sc_new:.10f} (beta={beta:.2e})")
            except Exception:
                pass
    print(f"  Phase 4 done, time={time.monotonic()-t4:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n search")
    t5 = time.monotonic()
    n_candidates = [2048, 3072, 5120, 6144, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 80:
            break
        if n_try == n_parent:
            continue  # Already done

        # Resample parent to new length
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            if a_init.sum() > 0:
                a_init = a_init / a_init.sum()
            else:
                a_init = _make_profile(n_try, "uniform")
        else:
            a_init = _make_profile(n_try, "tapered")

        sc_init = _score(a_init)
        print(f"    n={n_try}: init score={sc_init:.10f}")

        a_opt, sc_opt = _run_lbfgs_phased(a_init, deadline, min_time_remaining=60.0)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick PG-BB
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 30:
            a_pg, sc_pg = _pg_bb(a_opt, deadline, min_time_remaining=20.0)
            update_best(a_pg, sc_pg, f"n_{n_try}_pg")

    print(f"  Phase 5 done, time={time.monotonic()-t5:.1f}s")

    # === Phase 6: Diverse perturbations + re-optimization ===
    print(f"\n[Phase 6] Diverse restarts")
    t6 = time.monotonic()
    if best_seq is not None and time.monotonic() < deadline - 40:
        n = len(best_seq)
        for i in range(3):
            if time.monotonic() > deadline - 20:
                break
            # Different perturbation strategies
            if i == 0:
                # Small Gaussian noise in log-space
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z + 0.03 * rng.standard_normal(n)
                z = z - z.mean()
                p_start = _softmax(z)
            elif i == 1:
                # Fourier perturbation
                z = np.log(np.maximum(best_seq, 1e-30))
                t = np.arange(n)
                for _ in range(3):
                    freq = rng.integers(1, max(2, n // 8))
                    amp = 0.02 * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
                z = z - z.mean()
                p_start = _softmax(z)
            else:
                # Anti-symmetric perturbation (breaks palindromic structure)
                z = np.log(np.maximum(best_seq, 1e-30))
                z_asym = z - z[::-1]
                z = z + 0.05 * z_asym
                z = z - z.mean()
                p_start = _softmax(z)

            sc_start = _score(p_start)
            a_r, sc_r = _run_lbfgs_phased(p_start, deadline, min_time_remaining=20.0)
            update_best(a_r, sc_r, f"restart_{i}")

    print(f"  Phase 6 done, time={time.monotonic()-t6:.1f}s")

    # === Phase 7: Final finest polish ===
    print(f"\n[Phase 7] Final polish")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())

        # Active set Newton one more time
        if time.monotonic() < deadline - 5:
            a_as, sc_as = _active_set_newton(best_seq, deadline - 2, max_iter=100)
            update_best(a_as, sc_as, "final_as")

        # Tiny L-BFGS
        if time.monotonic() < deadline - 2:
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            try:
                result = minimize(
                    _lse_obj, z, args=(beta_final, n),
                    jac=_lse_grad,
                    method='L-BFGS-B',
                    options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-15, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        final = final / final.sum() * 1.0  # normalize to sum=1
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        # Validate with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"Validation error: {e}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
