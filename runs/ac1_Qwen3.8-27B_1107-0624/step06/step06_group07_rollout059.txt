import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _autocorr(a):
    n = len(a)
    return _full_convolve(a, a)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_asymmetric_profile(n, kind="ramp", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "ramp":
        a = t + 0.1
    elif kind == "ramp2":
        a = t**2 + 0.05
    elif kind == "exponential":
        rate = rng.uniform(1.0, 3.0)
        a = np.exp(-rate * t) + 0.05
    elif kind == "exponential_rev":

    elif kind == "bump_offcenter":

def _spectral_shaping(a, alpha, n_boost=10):
    """
    Novel spectral shaping step: boost DC component of power spectrum
    relative to other components, which flattens the autocorrelation.
    alpha in (0,1): how aggressive the shaping is.
    """
    n = len(a)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, L)
    P = np.abs(A) ** 2
    phase = np.angle(A)
    mag = np.abs(A)

    # Target: make power spectrum more concentrated at DC
    # P_target[0] = P[0], P_target[k] = P[k] * (1-alpha) for k>0
    # But we need to be careful: reducing non-DC power increases relative DC
    new_mag = mag.copy()
    if len(new_mag) > 1:
        decay = 1.0 - alpha
        # Geometric decay from DC
        freqs = np.arange(len(new_mag))
        decay_profile = np.ones(len(new_mag))
        decay_profile[1:] = decay ** (freqs[1:] / n_boost)
        new_mag = mag * decay_profile
        # Boost DC to compensate (keep total "energy" similar)
        new_mag[0] = mag[0] * (1.0 + alpha * 0.5)

    # Reconstruct
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def _spectral_waterfill(a, target_ratio, max_iters=20):
    """
    Iteratively shape the spectrum to make autocorrelation more flat.
    target_ratio: ratio of max autocorrelation to mean we're targeting.
    """
    n = len(a)
    L = _next_pow2(2 * n - 1)
    a_cur = a.copy()

    for it in range(max_iters):
        c = _full_convolve(a_cur, a_cur)
        c_mean = float(c.sum()) / len(c)
        c_max = float(c.max())
        if c_max / c_mean < target_ratio:
            break

        A = np.fft.rfft(a_cur, L)
        P = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute "excess" in autocorrelation domain
        excess = np.maximum(c - c_mean, 0)

        # Convert excess to frequency domain correction
        # The excess autocorrelation has DFT = |A|^2 - mean
        # We want to reduce |A|^2 where it contributes to excess
        # Simple approach: reduce non-DC power proportionally to its
        # contribution to the excess autocorrelation

        # A simpler heuristic: reduce power at frequencies that correlate
        # with the excess pattern
        excess_L = np.zeros(L)
        excess_L[:len(excess)] = excess
        excess_dft = np.fft.rfft(excess_L, L)

        # Reduce power where excess_dft has high magnitude (aligned with excess)
        # and boost where it's low
        p_rel = P / (P.sum() + 1e-30)
        excess_mag = np.abs(excess_dft)
        excess_mag = excess_mag / (excess_mag.sum() + 1e-30)

        # Damp frequencies that contribute to excess
        damping = 1.0 - 0.3 * (excess_mag / (excess_mag.max() + 1e-30))
        damping[0] = 1.0  # Don't touch DC

        mag = np.abs(A) * np.sqrt(damping)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_cur = a_new / s

    return a_cur


def _adaptive_subgrad(a_init, deadline, tol_frac=1e-4, max_steps=20000):
    """
    Subgradient method with per-coordinate adaptive learning rates (RMSProp-like).
    Works in log-space (z-space) for automatic non-negativity.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15)).astype(np.float64)
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # RMSProp state
    sq_avg = np.zeros(n)
    beta_sq = 0.9
    base_lr = 0.05
    eps = 1e-8
    stall = 0
    best_lr = base_lr

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient computation
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # RMSProp update
        sq_avg = beta_sq * sq_avg + (1 - beta_sq) * grad ** 2
        lr_t = base_lr / (np.sqrt(sq_avg) + eps)
        z = z - lr_t * grad
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        # Periodic evaluation
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                base_lr = min(base_lr * 1.05, 0.3)
            else:
                stall += 1
                if stall > 5:
                    base_lr *= 0.75
                    if base_lr < 1e-7:
                        break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smoothed(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _lbfgs_schedule(a_start, deadline, min_time_remaining=300.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.002, 0.001, 0.0004, 0.00015, 0.00005,
                              0.00002, 0.000008])
    current_a = a_start.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(10000, max(200, int(remaining / 0.004)))
        a_new = _lbfgs_smoothed(current_a, beta, n, max_iter)
        sc_new = _score(a_new)
        sc_cur = _score(current_a)
        if sc_new < sc_cur:
            current_a = a_new

    return current_a


def _iterative_linearization(a_init, deadline, max_rounds=200, active_tol=0.05):
    """
    Cutting-plane style: linearize the max constraint at active lags,
    solve the resulting LP to get a new point.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = active_tol * c_max
        active = np.where(c >= c_max - tol)[0]

        # Limit active set size for LP tractability
        if len(active) > 80:
            # Keep only the top 80
            idx = np.argsort(c[active])[-80:]
            active = active[idx]

        # Build LP:
        # minimize t
        # s.t. sum_j gk[j] * a'[j] - t <= bk  for each active k
        #      a' >= 0
        #      sum(a') = 1
        # where gk[j] = d/da[j] conv(a,a)_k evaluated at current a
        # and bk = conv(a,a)_k - sum_j gk[j]*a[j] + c_max (the RHS)

        # Actually, the linearization is:
        # conv(a',a')_k ≈ conv(a,a)_k + gk^T (a' - a)
        # We want: conv(a,a)_k + gk^T a' - gk^T a <= t
        # i.e., gk^T a' - t <= t - conv(a,a)_k + gk^T a
        # Hmm, let me just set up: minimize t s.t. gk^T a' - t <= c_k - gk^T a + c_max
        # No, simpler: we want gk^T a' <= t + (c_k - gk^T a)
        # where the RHS = c_k - gk^T a is the "intercept" of the linearization

        A_rows = []
        b_vals = []
        for k in active:
            # Compute gradient gk
            gk = np.zeros(n)
            if k < n:
                # conv_full[k] = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a[j]*a[k-j]
                # d/da[i] conv_full[k] = a[k-i] (if valid) + a[i-k] (if valid)
                # More carefully: conv_full[k] = sum_j a[j]*a[k-j]
                # d/da[i] = a[k-i] (from the a[j] term when j=i)
                #            + a[i-k]... no
                # Let's use: conv_full is np.convolve(a,a)
                # np.convolve(a,a)[k] = sum_j a[j] * a[k-j]
                # d/da[i] sum_j a[j]*a[k-j] = a[k-i] + a[i-k] (if both indices valid)
                # Wait, that's not right for the cross derivative.
                # d/da[i] of sum_j a[j]*a[k-j]:
                #   = d/da[i] of [a[i]*a[k-i] + sum_{j!=i} a[j]*a[k-j]]
                #   = a[k-i] + (if i appears in a[k-j], i.e., k-j=i, j=k-i)
                #     then d/da[i] of a[k-i] = ... no, a[k-j] is a different index
                # Actually: sum_j a[j]*a[k-j]. The variable a[i] appears:
                #   - as a[j] when j=i: contribution a[i]*a[k-i], derivative = a[k-i]
                #   - as a[k-j] when k-j=i, i.e., j=k-i: contribution a[k-i]*a[i], derivative = a[k-i]
                # So d/da[i] = a[k-i] + a[k-i] = 2*a[k-i] if i != k-i
                #    or a[k-i] = a[i] (one term) if i = k-i, i.e., k = 2i
                # Wait no. Let me be very careful.
                # f = sum_j a[j] * a[k-j]
                # df/da[i] = sum_j (da[j]/da[i]) * a[k-j] + a[j] * (da[k-j]/da[i])
                #         = a[k-i] + a[i-k]  ... no
                # df/da[i] = (da[i]/da[i])*a[k-i] + sum_{j != i} 0*a[k-j]
                #           + sum_j a[j] * (da[k-j]/da[i])
                # For the second sum: da[k-j]/da[i] = 1 when k-j=i, i.e., j=k-i
                # So second sum = a[k-i] (if k-i is a valid index in [0,n-1])
                # First term: a[k-i] (if k-i is valid)
                # Total: 2*a[k-i] if i != k-i (i.e., 2i != k)
                #        a[k-i] = a[i] if i = k-i (i.e., 2i = k)
                # But wait, this is for the "full" convolution where indices can be negative.
                # For np.convolve, the valid range is 0 <= j < n and 0 <= k-j < n.
                # So:
                # df/da[i] = a[k-i] (if 0<=k-i<n, from j=i term)
                #           + a[k-i] (if 0<=k-i<n, from j=k-i term)
                # But if i = k-i (2i=k), these are the same term, so it's just a[i].

                # Simpler: just compute numerically or use the formula
                # gk[i] = (a[k-i] if 0<=k-i<n else 0) + (a[k-i] if 0<=k-i<n and i!=k-i else 0)
                # Actually even simpler: gk = 2*a reversed and shifted, handling the diagonal
                # Let me just compute it directly:
                gk = np.zeros(n)
                for i in range(n):
                    val = 0.0
                    j1 = k - i  # from a[j] term, j=i, need a[k-i]
                    if 0 <= j1 < n:
                        val += a[j1]
                    j2 = i - k  # from a[k-j] term, k-j=i means j=k-i... 
                    # Wait I'm overcomplicating. Let me just use the fact that
                    # the gradient of autocorrelation at lag k is:
                    # gk[i] = a[(i+k) % ...] no, it's not circular.
                    # 
                    # Let me just use: the subgradient of max_k conv(a,a)_k
                    # is sum_{k in active} (shifted_a_k + reversed_shifted_a_k) / |active|
                    # where shifted_a_k[i] = a[i+k] if i+k < n else 0
                    # and reversed: a[i-k] if i-k >= 0 else 0
                    pass
                
                break  # This direct computation is too slow for large n
            break
        # The per-element gradient computation is O(n) per lag, and we have up to 80 lags.
        # Total: O(80*n) = O(80*4096) ≈ 327K operations. That's fine actually.
        # Let me redo without the break:
        
        A_rows = []
        b_vals = []
        for k in active:
            gk = np.zeros(n)
            # gk[i] = a[i+k] (if i+k < n) + a[i-k] (if i-k >= 0)
            if k == 0:
                gk[:] = 2.0 * a
            else:
                gk[:n - k] += a[k:]   # a[i+k] for i=0..n-1-k
                gk[k:] += a[:n - k]   # a[i-k] for i=k..n-1
            # Note: for k=0, gk[i] = a[i]+a[i] = 2*a[i] ✓
            
            fk = float(c[k])  # current value
            gk_a = float(np.dot(gk, a))  # gradient dot current point
            
            # Linearization: fk + gk^T(a' - a) <= t
            # => gk^T a' - t <= t - fk + gk^T a
            # Hmm, I want: gk^T a' <= t + (fk - gk^T a)
            # Let me just set: gk^T a' - t <= fk - gk^T a
            # The RHS: fk - gk^T a = fk - (sum of gk[i]*a[i])
            # For k > 0: gk^T a = sum_i (a[i+k]+a[i-k])*a[i] = 2*fk (approximately, ignoring boundaries)
            # So RHS ≈ fk - 2*fk = -fk
            # For k = 0: gk^T a = 2*sum a^2 = 2*fk, same thing.
            
            rhs = fk - gk_a
            
            A_row = np.zeros(n + 1)
            A_row[:n] = gk
            A_row[n] = -1.0
            A_rows.append(A_row)
            b_vals.append(rhs)

        if len(A_rows) == 0:
            break

        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t

        # Equality: sum(a') = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Bounds: a' >= 0, t is free
        bounds = [(0, None)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='hihps')
            if res.success:
                a_new = res.x[:n]
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-10:
                    a_new /= s
                    sc_new = _score(a_new)
                    if sc_new < best_sc:
                        best_sc = sc_new
                        best_a = a_new.copy()
                        a = a_new
            else:
                break
        except Exception:
            break

    return best_a, best_sc


def _multi_scale_init(target_n, rng, time_budget=30):
    """
    Find a good pattern at small n, then upsample to target_n.
    """
    small_n = 128
    t_end = time.monotonic() + time_budget
    best_small = None
    best_sc_small = np.inf

    for kind in ["uniform", "ramp", "exponential", "bump_offcenter", "sine_asym", "two_bumps"]:
        a_s = _make_asymmetric_profile(small_n, kind, rng)
        # Quick optimization at small scale
        c = _full_convolve(a_s, a_s)
        mb = float(c.max())
        for beta in [mb * 0.05, mb * 0.01, mb * 0.002]:
            a_s = _lbfgs_smoothed(a_s, beta, small_n, max_iter=500)
        sc = _score(a_s)
        if sc < best_sc_small:
            best_sc_small = sc
            best_small = a_s.copy()
        if time.monotonic() > t_end:
            break

    if best_small is None:
        best_small = np.ones(small_n) / small_n

    # Upsample
    t_old = np.linspace(0, 1, small_n)
    t_new = np.linspace(0, 1, target_n)
    a_up = np.interp(t_new, t_old, best_small)
    a_up = np.maximum(a_up, 1e-10)
    a_up /= a_up.sum()
    return a_up, best_sc_small


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _fine_polish(a, deadline, max_rounds=50):
    """
    Very fine polishing using tiny subgradient steps + spectral shaping.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Tiny subgradient step
        a_cur = _softmax(z)
        c = _full_convolve(a_cur, a_cur)
        c_max = float(c.max())
        tol = 1e-5 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a_cur[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_cur * (h - W)

        step = 0.01 / (1.0 + rnd * 0.1)
        z = z - step * grad
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        # Spectral shaping
        a_spectral = _spectral_shaping(_softmax(z), alpha=0.1, n_boost=5)
        z2 = np.log(np.maximum(a_spectral, 1e-15))
        z2 = z2 - z2.mean()
        # Blend
        z = 0.7 * z + 0.3 * z2
        z = z - z.mean()

        a_check = _softmax(z)
        sc = _score(a_check)
        if sc < best_sc:
            best_sc = sc
            best_a = a_check.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Novel multi-strategy optimizer combining spectral shaping, adaptive
    subgradient (RMSProp), iterative linearization (LP cutting plane),
    multi-scale initialization, and L-BFGS with log-sum-exp smoothing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s  # normalize to sum=1 (score is scale-invariant)
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s")
            return True
        return False

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_target = 4096
    if parent is not None:
        n_target = len(parent)
        print(f"Parent: n={n_target}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_asymmetric_profile(n_target, "uniform", rng)
        print(f"No parent. Uniform n={n_target}, score={_score(a0):.10f}")
        try_update(a0, "init")

    # === Phase 1: Multi-scale initialization to find diverse starting points ===
    print(f"\n=== Phase 1: Multi-scale init (t={time.monotonic()-t_start:.1f}s) ===")
    t_end_p1 = min(deadline - 600, t_start + 120)
    ms_a, ms_sc = _multi_scale_init(n_target, rng, time_budget=t_end_p1 - time.monotonic())
    try_update(ms_a, "multi_scale")

    # Try asymmetric profiles directly
    asym_profiles = ["ramp", "exponential", "bump_offcenter", "sine_asym",
                     "two_bumps", "log_ramp", "step_half"]
    for kind in asym_profiles[:4]:
        if time.monotonic() > t_end_p1:
            break
        a_p = _make_asymmetric_profile(n_target, kind, rng)
        # Quick L-BFGS to improve
        c = _full_convolve(a_p, a_p)
        mb = float(c.max())
        a_p = _lbfgs_smoothed(a_p, mb * 0.05, n_target, max_iter=500)
        a_p = _lbfgs_smoothed(a_p, mb * 0.01, n_target, max_iter=500)
        try_update(a_p, f"asym_{kind}")

    # === Phase 2: Main L-BFGS from best ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None:
        t_end_p2 = min(deadline - 400, t_start + 500)
        a_p2 = _lbfgs_schedule(best_seq, t_end_p2, min_time_remaining=200.0)
        try_update(a_p2, "lbfgs_main")

    # === Phase 3: Adaptive subgradient (RMSProp) ===
    print(f"\n=== Phase 3: Adaptive subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        t_end_p3 = min(deadline - 250, t_start + 800)
        a_sg, sc_sg = _adaptive_subgrad(best_seq, t_end_p3, tol_frac=2e-4, max_steps=15000)
        try_update(a_sg, "adaptive_sg")

    # === Phase 4: Spectral water-filling ===
    print(f"\n=== Phase 4: Spectral water-filling (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        t_end_p4 = min(deadline - 150, t_start + 1000)
        # Iterative spectral shaping
        a_sw = best_seq.copy()
        for it in range(30):
            if time.monotonic() > t_end_p4:
                break
            a_sw = _spectral_shaping(a_sw, alpha=0.05, n_boost=20)
            # also do a small subgradient correction
            c = _full_convolve(a_sw, a_sw)
            c_max = c.max()
            active = c >= c_max - 0.01 * c_max
            w = np.zeros_like(c)
            w[active] = 1.0 / active.sum()
            n = len(a_sw)
            a_rev = a_sw[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W_val = float(np.dot(w, c))
            grad = 4.0 * n * a_sw * (h - W_val)
            a_sw = a_sw - 0.01 * grad
            a_sw = np.maximum(a_sw, 0)
            a_sw /= a_sw.sum()
            if it % 5 == 0:
                try_update(a_sw, f"spec_wf_{it}")
        try_update(a_sw, "spec_wf_final")

    # === Phase 5: Iterative linearization (LP) ===
    print(f"\n=== Phase 5: Iterative linearization (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        t_end_p5 = min(deadline - 100, t_start + 1200)
        a_il, sc_il = _iterative_linearization(best_seq, t_end_p5, max_rounds=100, active_tol=0.03)
        try_update(a_il, "iter_lin")

    # === Phase 6: Fine L-BFGS ===
    print(f"\n=== Phase 6: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - 80 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(5000, max(200, int(remaining / 0.005)))
            a_new = _lbfgs_smoothed(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 7: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n = len(best_seq)
        restarts = [
            ("gauss_s", 0.01), ("gauss_m", 0.03), ("gauss_l", 0.08),
            ("spectral", None), ("asym_shift", None),
        ]
        for name, sigma in restarts:
            if time.monotonic() > deadline - 60:
                break
            if sigma is not None:
                z = np.log(np.maximum(best_seq, 1e-15))
                z += sigma * rng.standard_normal(n)
                z = z - z.mean()
                p_start = _softmax(z)
            elif name == "spectral":
                p_start = _spectral_shaping(best_seq, alpha=0.2, n_boost=5)
            else:  # asym_shift
                p_start = np.roll(best_seq, int(rng.integers(-20, 20)))
                p_start = np.maximum(p_start, 0)
                p_start /= p_start.sum()

            # Quick optimize
            c_s = _full_convolve(p_start, p_start)
            mb_s = float(c_s.max())
            p_start = _lbfgs_smoothed(p_start, mb_s * 0.01, n, max_iter=1000)
            try_update(p_start, f"restart_{name}")

            # If it's close to best, do more optimization
            sc_s = _score(p_start)
            if sc_s < best_score_val + 0.01 and time.monotonic() < deadline - 50:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=40.0)
                try_update(a_r, f"restart_{name}_full")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [2048, 3072, 5120, 6144]:
            if time.monotonic() > deadline - 50:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            c_i = _full_convolve(a_interp, a_interp)
            mb_i = float(c_i.max())
            a_interp = _lbfgs_smoothed(a_interp, mb_i * 0.02, target_n, max_iter=2000)
            a_interp = _lbfgs_smoothed(a_interp, mb_i * 0.005, target_n, max_iter=2000)
            try_update(a_interp, f"n_{target_n}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(6):
            if time.monotonic() > deadline - 5:
                break
            # Very fine L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-7
            a_f = _lbfgs_smoothed(best_seq, beta, n, max_iter=3000)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Fine subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _adaptive_subgrad(best_seq, deadline - 5,
                                           tol_frac=5e-5, max_steps=5000)
                try_update(a_s, f"final_sg_{rnd}")

            # Spectral shaping + subgradient blend
            if time.monotonic() < deadline - 5:
                a_sp = _spectral_shaping(best_seq, alpha=0.03, n_boost=30)
                # Small subgradient correction
                c_sp = _full_convolve(a_sp, a_sp)
                c_max_sp = c_sp.max()
                active_sp = c_sp >= c_max_sp - 1e-5 * c_max_sp
                w_sp = np.zeros_like(c_sp)
                w_sp[active_sp] = 1.0 / active_sp.sum()
                n_sp = len(a_sp)
                a_rev_sp = a_sp[::-1]
                wc_sp = _full_convolve(w_sp, a_rev_sp)
                h_sp = wc_sp[n_sp - 1:2 * n_sp - 1]
                W_sp = float(np.dot(w_sp, c_sp))
                grad_sp = 4.0 * n_sp * a_sp * (h_sp - W_sp)
                a_sp = a_sp - 0.005 * grad_sp
                a_sp = np.maximum(a_sp, 0)
                a_sp /= a_sp.sum()
                try_update(a_sp, f"final_spec_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
