    # Actually, for Adam/L-BFGS, the absolute scale of the gradient doesn't matter (only the direction and relative magnitudes). So the factor of 2 vs 4 just changes the effective learning rate.
    
    # Let me use factor 2n (the correct one) and adjust learning rates accordingly.
    N_fft = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N_fft)
    fp = np.fft.rfft(p, N_fft)
    # cross_corr[j] = sum_k a[k] * p[k+j] ...

    # Let A(k) = a[k] for k in 0..n-1, 0 otherwise.
    # Let P(k) = p[k] for k in 0..2n-2, 0 otherwise.
    # We want g[j] = sum_k P(k) * A(k-j) for j in 0..n-1.
    # This is the cross-correlation: g = corr(P, A) evaluated at lags 0..n-1.
    # Using FFT: g = ifft(FFT(P) * conj(FFT(A))) ... no.
    # Standard: corr(x, y)[lag] = sum_n x[n+lag] * y[n]
    # So corr(P, A)[j] = sum_n P[n+j] * A[n] = sum_n P[n+j] * a[n] (for valid n)

    # Actually: ifft(fft(P) * conj(fft(A))) gives sum_n P[n] * A[n+m] (circular)

    # grad[j] = 2n * conv_full[n-1-j] = 2n * conv_full[n-1:2n-1][::-1][j]

    # I'll use 2n for correctness.
    
    p_rev = p[::-1]

    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]

def _compute_active_gradient(a, c, delta=1e-10):
    """Compute subgradient using all positions where c_k >= max(c) - delta."""
    n = len(a)
    max_c = c.max()
    active = c >= (max_c - delta)
    if not active.any():
        active[np.argmax(c)] = True
    p = np.zeros_like(c)
    p[active] = 1.0 / active.sum()
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _fista_simplex(a_init, deadline, T_start, T_end, max_iter=200000,
                   eval_interval=200, verbose=False):
    """FISTA with soft gradient on the probability simplex."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    L = 8.0 * n  # Lipschitz constant upper bound
    step = 1.0 / L
    patience = 0
    best_patience_score = best_score

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max(1, max_iter))
        T = T_start * (T_end / T_start) ** frac
        
        c = _fft_convolve(y, y)
        g = _compute_soft_gradient(y, c, T)
        
        # Backtracking line search (simplified)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            step = min(2.0 / (g_norm + 1e-10), step * 2.0)
        step = max(step, 1e-12)
        
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        
        t_new_t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new_t) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new_t
        a = a_new
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if verbose and t % (eval_interval * 10) == 0:
                print(f"    FISTA t={t}, score={sc:.10f}, T={T:.2e}")
            if patience > 50 and t > 1000:
                # Reset momentum
                y = a.copy()
                t_k = 1.0
                patience = 0

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=300000, active_mode=False, active_delta=1e-10,
                  eval_interval=200, clip_norm=1.0, verbose=False):
    """Adam optimizer in z-space with temperature annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if active_mode:
            g = _compute_active_gradient(a, c, active_delta)
        else:
            g = _compute_soft_gradient(a, c, T)
        
        # Center the gradient (for simplex constraint)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                if verbose and t % (eval_interval * 5) == 0:
                    print(f"    Adam t={t}, score={sc:.10f}, T={T:.2e}, lr={lr:.4f}")
            else:
                patience += 1
            if patience > 10 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                # Reset to best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _anti_palindromic_init(n, epsilon, mode='cos', rng=None):
    """Generate anti-palindromic perturbation of uniform distribution."""
    j = np.arange(n)
    if mode == 'cos':
        # cos(pi*(j+0.5)/n) is anti-palindromic
        u = np.cos(np.pi * (j + 0.5) / n)
    elif mode == 'cos2':
        # cos(pi*3*(j+0.5)/n) is also anti-palindromic (odd harmonic)
        u = np.cos(3 * np.pi * (j + 0.5) / n)
    elif mode == 'sin2':
        # sin(2*pi*(j+0.5)/n) is anti-palindromic (even harmonic sine)
        u = np.sin(2 * np.pi * (j + 0.5) / n)
    else:
        u = np.zeros(n)
    
    # Ensure non-negativity
    vals = 1.0 + epsilon * u
    if vals.min() < 0:
        # Scale down epsilon
        epsilon = epsilon * 0.9 / abs(u.min()) if u.min() != 0 else epsilon
        vals = 1.0 + epsilon * u
    
    a = vals / vals.sum()
    return a


def _random_init(n, rng):
    """Random initialization on the simplex."""
    a = rng.dirichlet(np.ones(n))
    return a


def _gaussian_init(n, sigma_frac, rng):
    """Gaussian-like initialization."""
    j = np.arange(n)
    center = n * 0.3  # Off-center to break symmetry
    a = np.exp(-0.5 * ((j - center) / (n * sigma_frac)) ** 2)
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def _nesterov_gradient(a, g, step, momentum=0.9):
    """Nesterov-like step with momentum."""
    a_new = a + momentum * (a - a) - step * g  # Simplified
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer leveraging anti-palindromic initialization.
    Key phases: diverse init -> FISTA -> Adam soft -> Adam active -> polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    # Try to load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            parent_norm = parent / parent.sum()
            has_parent = True
        else:
            has_parent = False
    except Exception:
        has_parent = False
        parent_norm = None
    
    # Work with n=4096 as primary, also try n=2048
    n_primary = 4096
    
    best_a = None
    best_score = np.inf
    best_n = 0
    
    def update_best(a_cand):
        nonlocal best_a, best_score, best_n
        sc = _score(a_cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            best_n = len(a_cand)
            return True
        return False
    
    # Phase 0: Generate diverse initializations (0-10s)
    inits = []
    n = n_primary
    
    # Anti-palindromic initializations
    for eps in [0.3, 0.5, 0.7, 0.9]:
        a_init = _anti_palindromic_init(n, eps, mode='cos')
        inits.append(a_init)
    
    # Higher-order anti-palindromic
    a_init = _anti_palindromic_init(n, 0.5, mode='cos2')
    inits.append(a_init)
    a_init = _anti_palindromic_init(n, 0.5, mode='sin2')
    inits.append(a_init)
    
    # Parent
    if has_parent and len(parent_norm) == n:
        inits.append(parent_norm.copy())
    elif has_parent:
        # Interpolate parent to n
        x0 = np.linspace(0, 1, len(parent_norm))
        x1 = np.linspace(0, 1, n)
        a_interp = np.interp(x1, x0, parent_norm)
        a_interp = np.maximum(a_interp, 1e-10)
        a_interp /= a_interp.sum()
        inits.append(a_interp)
    
    # Uniform (baseline)
    inits.append(np.ones(n) / n)
    
    # A few random
    for _ in range(2):
        inits.append(_random_init(n, rng))
    
    print(f"Phase 0: Generated {len(inits)} initializations, t={time.monotonic()-t_start:.1f}s")
    
    # Phase 1: Quick FISTA on each initialization (10s-200s)
    phase1_end = min(deadline, t_start + 200.0)
    time_per_init = (phase1_end - t_start) / max(1, len(inits))
    
    fista_results = []
    for i, a_init in enumerate(inits):
        if time.monotonic() > phase1_end - 5:
            break
        alloc = min(time_per_init, phase1_end - time.monotonic() - 3)
        if alloc < 5:
            break
        sub_deadline = t_start + (200.0 / len(inits)) * (i + 1)
        sub_deadline = min(sub_deadline, phase1_end - 2)
        
        c = _fft_convolve(a_init, a_init)
        max_c = float(c.max())
        a_fista, sc_fista = _fista_simplex(
            a_init, sub_deadline,
            T_start=0.01 * max_c, T_end=0.001 * max_c,
            max_iter=50000, eval_interval=500
        )
        full_sc = _score(a_fista)
        print(f"  FISTA[{i}] score={full_sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        fista_results.append(a_fista)
        update_best(a_fista)
    
    # Phase 2: Adam soft gradient on top FISTA results (200s-600s)
    phase2_end = min(deadline, t_start + 600.0)
    # Sort by score and take top 3
    fista_results.sort(key=lambda x: _score_norm(x))
    top_k = min(3, len(fista_results))
    
    adam_results = []
    for i in range(top_k):
        if time.monotonic() > phase2_end - 30:
            break
        a_cur = fista_results[i]
        c = _fft_convolve(a_cur, a_cur)
        max_c = float(c.max())
        alloc_deadline = min(phase2_end - 5, t_start + 600.0 - 5)
        a_adam, sc_adam = _adam_z_space(
            a_cur, alloc_deadline,
            T_start=0.005 * max_c, T_end=0.0001 * max_c,
            lr_start=0.05, lr_end=0.005,
            max_iter=200000, eval_interval=200
        )
        full_sc = _score(a_adam)
        print(f"  Adam-soft[{i}] score={full_sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        adam_results.append(a_adam)
        if _score(a_adam) < best_score:
            best_score = _score(a_adam)
            best_a = a_adam.copy()
            best_n = n
    
    # Phase 3: Adam active-set on best (600s-1100s)
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 30:
        a_cur = best_a.copy()
        c = _fft_convolve(a_cur, a_cur)
