```python
Minimax-based optimizer for step function convolution peaks.
Uses the von Neumann minimax theorem to reformulate the problem as a
convex optimization over probability distributions on the convolution index set,
solved via iterative dual ascent with FFT-accelerated conjugate gradients.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
```

```python
def _cg_solve_hankel(w, b, tol=1e-10, max_iter=10000):
    """Solve Q(w) * x = b using Conjugate Gradient.
    Q(w) is the n x n Hankel matrix with Q[i,j] = w[i+j].
    w has length 2n-1, b has length n.
    Returns x."""
    n = len(b)
    x = np.zeros(n)
    r = b.copy()  # r = b - Q*x = b initially
    p = r.copy()
    rs = r @ r
    b_norm = np.linalg.norm(b)
    if b_norm < 1e-30:
        return x
    target = tol * tol * b_norm * b_norm
    
    for it in range(max_iter):
        Qp = _hankel_matvec(w, p, n)
        alpha = rs / (p @ Qp + 1e-30)
        x += alpha * p
        r -= alpha * Qp
        new_rs = r @ r
        if new_rs < target:
            break
        p = r + (new_rs / (rs + 1e-30)) * p
        rs = new_rs
    
    return x
```
