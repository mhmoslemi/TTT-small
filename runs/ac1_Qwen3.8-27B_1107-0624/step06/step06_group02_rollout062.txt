```python
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume sum(a) = 1, return 2*n*max(a*a)."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.ones(n) * target_sum / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA for minimizing smoothed max of a*a, with simplex constraint sum(a)=1, a>=0."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    # Lipschitz constant estimate: L = 4*max(a) / beta (rough)
    # For safety, use L = 4.0 / beta (since a is on simplex, max(a) <= 1)
    L = 4.0 / beta + 4.0
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Compute gradient
        w_rev = w[::-1]
        conv = _convolve(w_rev, y)
        grad = 2.0 * conv[n-1:2*n-1][::-1]
        
        a_next = _proj_simplex(y - step * grad, 1.0)
        
        # Check improvement periodically
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space for the normalized sequence."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        for _ in range(max_iter_total):
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Gradient computation
            w_rev = w[::-1]
            conv = _convolve(w_rev, a)
            h = conv[n-1:2*n-1][::-1]
            W = float(np.dot(w, c))
            # The gradient of the smoothed max with respect to a is 2*h
            # But we want the gradient with respect to z (log-space)
            # a = softmax(z), so da/dz = a*(e_j - a) (outer product form)
            # grad_z = sum_m (da_m/dz_i) * grad_a_m
            # For the purpose of Adam, we use the "unnormalized" gradient:
            grad = 2.0 * a * h  # This is the "log-space gradient" approximation
            
            # Actually, the correct log-space gradient is:
            # d(smoothed_max)/dz_i = sum_m (da_m/dz_i) * d(smoothed_max)/da_m
            # da_m/dz_i = a_m * (delta_mi - a_i)
            # d(smoothed_max)/da_m = 2 * h_m (from our computation)
            # So d/dz_i = sum_m a_m*(delta_mi - a_i) * 2*h_m = 2*a_i*h_i - 2*a_i*sum_m a_m*h_m
            # = 2*a_i*(h_i - sum_m a_m*h_m)
            W2 = float(np.dot(a, h))
            grad = 2.0 * a * (h - W2)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if t % 5000 == 0:
                current_lr = max(0.0001, current_lr * 0.9)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _coord_descent(a_init, deadline, max_rounds=200):
    """Coordinate descent polish."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.95, 0.98, 1.02, 1.05]:
                a[j] = old_val * factor
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
                    break
                else:
                    a[j] = old_val
        if not improved and r > 30:
            break
    
    return best_a, best_sc

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
```

```python
"""
Multi-phase optimizer for minimizing 2*n*max(a*a)/(sum(a))^2 over non-negative sequences.
Uses FISTA with aggressive beta continuation, Adam log-space fine polish,
multi-restart with diverse perturbations, and coordinate descent.
Primary working size: n=8192, with fallback to n=4096.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.ones(n) * target_sum / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    """Compute gradient of dot(w, a*a) with respect to a.
    Result: grad[m] = 2 * sum_j w[j+m] * a[j] (w zero-padded)."""
    n = len(a)
    w_rev = w[::-1]
    conv = _convolve(w_rev, a)
    return 2.0 * conv[n - 1:2 * n - 1][::-1]

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.2:
                update = update * (0.2 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 4000 == 0:
                current_lr = max(0.00005, current_lr * 0.92)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 2048:
            indices = rng.choice(n, size=min(2048, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.94, 0.97, 1.03, 1.06]:
                a[j] = old_val * factor
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
                    break
                else:
                    a[j] = old_val
        if not improved and r > 40:
            break

    return best_a, best_sc


def _block_descent(a_init, deadline, max_rounds=100, block_size=16):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(77777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        n_blocks = n // block_size
        if n_blocks < 1:
            break
        block_idx = rng.integers(0, n_blocks)
        start = block_idx * block_size
        end = start + block_size

        old_block = a[start:end].copy()
        best_block = old_block.copy()
        best_block_sc = best_sc

        for attempt in range(8):
            perturb = old_block * (1.0 + 0.05 * rng.standard_normal(block_size))
            perturb = np.maximum(perturb, 0.0)
            if perturb.sum() < 1e-30:
                continue
            a[start:end] = perturb
            s = a.sum()
            if s < 1e-30:
                a[start:end] = old_block
                continue
            a_norm = a / s
            sc = _score_norm(a_norm)
            if sc < best_block_sc:
                best_block_sc = sc
                best_block = perturb.copy()
                best_a = a_norm.copy()
                a = a_norm.copy()

        a[start:end] = best_block
        s = a.sum()
        if s > 1e-30:
            a = a / s
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _optimize_at_size(a0, t_start, deadline, n_target, label):
    """Run full optimization pipeline at a given size."""
    best_a = a0.copy()
    best_sc = _score(a0)
    print(f"  [{label}] Start: n={n_target}, score={best_sc:.10f}")

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())

    # Phase A: FISTA with beta continuation
    phase_a_end = t_start + 0.25 * (deadline - t_start)
    if time.monotonic() < phase_a_end:
        beta = 0.05 * max_c0
        betas = []
        while beta > 1e-9 * max_c0:
            betas.append(beta)
            beta *= 0.3
        betas.append(1e-9 * max_c0)
        if len(betas) > 18:
            step_len = len(betas) // 18
            betas = betas[::step_len] + [1e-9 * max_c0]

        a_curr = a0.copy()
        for i, beta in enumerate(betas):
            if time.monotonic() > phase_a_end - 2:
                break
            per_time = min(25.0, phase_a_end - time.monotonic() - 1)
            if per_time < 3:
                break
            max_it = max(3000, int(per_time * 4000))
            a_curr, _ = _fista_smoothed(a_curr, beta, time.monotonic() + per_time, max_it)
            a_curr = a_curr / a_curr.sum()
            sc = _score(a_curr)
            if sc < best_sc:
                best_sc = sc
                best_a = a_curr.copy()
        print(f"  [{label}] After FISTA: score={best_sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase B: Adam log-space
    phase_b_end = t_start + 0.55 * (deadline - t_start)
    if time.monotonic() < phase_b_end - 20:
        c_curr = _convolve(best_a, best_a)
        max_cc = float(c_curr.max())
        betas_fine = []
        beta = 0.001 * max_cc
        while beta > 1e-12 * max_c0:
            betas_fine.append(beta)
            beta *= 0.35
        betas_fine.append(1e-12 * max_c0)
        if len(betas_fine) > 14:
            step_len = len(betas_fine) // 14
            betas_fine = betas_fine[::step_len] + [1e-12 * max_c0]

        per_t = phase_b_end - time.monotonic() - 10
        if per_t > 20:
            a_opt, _ = _adam_logspace(
                best_a, time.monotonic() + per_t,
                betas_fine, lr_init=0.006, max_iter_total=500000
            )
            sc = _score(a_opt)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
                print(f"  [{label}] After Adam: score={best_sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase C: Multi-restart
    phase_c_end = t_start + 0.80 * (deadline - t_start)
    if time.monotonic() < phase_c_end - 30:
        n_restarts = 5
        per_restart = (phase_c_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase_c_end - 10:
                break
            if i % 3 == 0:
                p_start = _perturb(best_a, 0.008 + 0.004 * i, rng_global)
            elif i % 3 == 1:
                p_start = _perturb(best_a, 0.025, rng_global)
            else:
                block = n_target // 8
                shift = rng_global.integers(1, max(2, block))
                p_start = best_a.copy()
                for b in range(0, n_target, block):
                    end = min(b + block, n_target)
                    p_start[b:end] = np.roll(best_a[b:end], shift)
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            p_best, _ = _fista_smoothed(
                p_start, 0.0001 * max_cs,
                time.monotonic() + per_restart * 0.35, 15000
            )
            p_best = p_best / p_best.sum()

            c_pb = _convolve(p_best, p_best)
            max_cp = float(c_pb.max())
            bs = [0.00005 * max_cp, 0.00002 * max_cp, 0.000005 * max_cp]
            p_best, _ = _adam_logspace(
                p_best, time.monotonic() + per_restart * 0.5,
                bs, lr_init=0.002, max_iter_total=30000
            )
            p_best = p_best / p_best.sum()
            sc = _score(p_best)
            if sc < best_sc:
                best_sc = sc
                best_a = p_best.copy()
                print(f"  [{label}] Restart {i}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase D: Coordinate + Block descent
    if time.monotonic() < deadline - 30:
        a_cd = best_a.copy() / best_a.sum()
        cd_deadline = min(deadline - 20, time.monotonic() + 80)
        a_cd, _ = _coord_descent(a_cd, cd_deadline, max_rounds=100)
        sc = _score(a_cd)
        if sc < best_sc:
            best_sc = sc
            best_a = a_cd.copy()

        if time.monotonic() < deadline - 15:
            a_bd = best_a.copy() / best_a.sum()
            bd_deadline = min(deadline - 10, time.monotonic() + 40)
            a_bd, _ = _block_descent(a_bd, bd_deadline, max_rounds=50, block_size=16)
            sc = _score(a_bd)
            if sc < best_sc:
                best_sc = sc
                best_a = a_bd.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    global rng_global
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng_global = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                print(f"Loaded parent: n={len(parent)}")
    except Exception:
        parent = None

    # Determine working sizes
    sizes_to_try = [8192]
    if parent is not None and len(parent) == 4096:
        sizes_to_try = [8192, 4096]

    total_time = deadline - t_start
    time_per_size = total_time / len(sizes_to_try)

    for n_target in sizes_to_try:
        size_deadline = t_start + (sizes_to_try.index(n_target) + 1) * time_per_size
        size_deadline = min(size_deadline, deadline - 10)

        if parent is not None:
            a0 = _interp_profile(parent, n_target)
        elif n_target == 4096:
            a0 = np.ones(n_target) / n_target
        else:
            a0 = np.ones(n_target) / n_target

        sc0 = _score(a0)
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        label = f"n={n_target}"
        best_a, best_sc = _optimize_at_size(a0, t_start, size_deadline, n_target, label)

        if best_sc < best_score:
            best_score = best_sc
            best_seq = best_a.copy()

    # Final re-optimization of the best found
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        a_final = best_seq.copy() / best_seq.sum()
        label = f"final_n={n_best}"
        print(f"\nFinal re-optimization: {label}")
        reopt_deadline = deadline - 15
        a_final, sc_final = _optimize_at_size(a_final, time.monotonic(), reopt_deadline, n_best, label)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()

    # Validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
