from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.001)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _coord_descent(a_init, deadline, n_sweeps=200):
    """Coordinate descent: optimize each element individually."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    s = a.sum()

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, max(1, n // 500)):
            # Optimize a[i] by trying a few values
            old_val = a[i]
            # Try: 0, old_val*0.5, old_val*0.75, old_val*1.25, old_val*1.5
            candidates = [0.0, old_val * 0.5, old_val * 0.75, old_val, old_val * 1.25, old_val * 1.5]
            if old_val > 0:
                candidates = [max(0, c) for c in candidates]
            best_local = old_val
            best_local_sc = best_sc
            # Quick score evaluation using the fact that only 2 values change
            for cv in candidates:
                if cv == old_val:
                    continue
                a[i] = cv
                sc_try = _score(a)
                if sc_try < best_local_sc:
                    best_local_sc = sc_try
                    best_local = cv
            a[i] = best_local
            if best_local != old_val:
                improved = True
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    break
                a = a / s
                best_sc = _score(a)
                best_a = a.copy()
        if not improved:
            break
    return best_a, best_sc

def _shrinkage_optimized(a, deadline, max_fractions=20):
    """Aggressive support reduction with re-optimization after each step."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_work = a.copy()
    s = a_work.sum()
    a_work = a_work / s

    # Try zeroing elements iteratively, keeping the best
    for frac_target in np.linspace(0.05, 0.50, max_fractions):
        if time.monotonic() > deadline:
            break
        n_zero_target = int(n * frac_target)
        # Zero out the smallest elements
        for _ in range(min(10, n_zero_target)):
            if time.monotonic() > deadline:
                break
            idx = int(np.argmin(a_work))
            if a_work[idx] < 1e-10:
                continue
            a_work[idx] = 0.0
            s_try = a_work.sum()
            if s_try < 1e-10:
                a_work[idx] = 0.0
                continue
            sc_try = _score(a_work)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_a = a_work.copy()
            else:
                # Undo
                if idx >= 0:
                    # We need to restore - but we zeroed it
                    # Just skip
                    pass
                break
        # Re-optimize on the support
        mask = a_work > 1e-10
        if mask.sum() > 10:
            a_sub = a_work[mask]
            z_sub = np.log(np.maximum(a_sub, 1e-30))
            z_sub = z_sub - z_sub.mean()
            n_sub = len(a_sub)
            # Quick L-BFGS on the sub-problem
            beta = _full_convolve(a_sub, a_sub).max() * 0.01
            try:
                res = minimize(
                    lambda z_l: _smoobj(z_l, beta, n_sub),
                    z_sub,
                    jac=lambda z_l: _smograd(z_l, beta, n_sub),
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-10}
                )
                a_sub_new = _softmax(res.x)
                # Reconstruct full sequence
                a_full_new = np.zeros(n)
                a_full_new[mask] = a_sub_new
                s_full = a_full_new.sum()
                if s_full > 1e-10:
                    a_full_new = a_full_new / s_full
                    sc_full = _score(a_full_new)
                    if sc_full < best_sc - 1e-12:
                        best_sc = sc_full
                        best_a = a_full_new.copy()
                        a_work = a_full_new
            except Exception:
                pass
    return best_a, best_sc

def _make_periodic_modulation(a_mod, p, amplitude, rng):
    """Add periodic modulation to the log-space sequence."""
    n = len(a_mod)
    t = np.arange(n)
    freqs = rng.choice(np.arange(1, max(2, n // 2)), size=min(3, n // 4), replace=False)
    for f in freqs:
        amp = amplitude * rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        a_mod = a_mod + amp * np.cos(2.0 * np.pi * f * t / n + phase)
    return a_mod

def _run_lbfgs_improved(a_init, deadline, min_time_remaining=30.0, tag=""):
    """L-BFGS with very fine beta continuation and more iterations."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.005)

    # Finer beta schedule with more steps
    n_betas = 18
    betas = base_beta * np.exp(np.linspace(0, np.log(0.0003), n_betas))

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter based on remaining time
        est_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_per_iter * 15))))

        try:
            result = minimize(
                lambda z_l: _smoobj(z_l, beta, n),
                z,
                jac=lambda z_l: _smograd(z_l, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search with improved L-BFGS, aggressive shrinkage,
    coordinate descent polish, periodic perturbations, and multi-n search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        par_sc = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={par_sc:.10f}")
        update_best(parent, par_sc, "parent")
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        update_best(parent, _score(parent), "uniform_init")

    # === Phase 1: Main L-BFGS with improved schedule ===
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs_improved(parent, deadline, min_time_remaining=500.0, tag="main")
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 400:
        sg_deadline = min(deadline - 400, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=4000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_polish_1")

    # === Phase 2: Aggressive shrinkage ===
    print(f"\n[Phase 2] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_shr, sc_shr = _shrinkage_optimized(best_seq, min(deadline - 300, time.monotonic() + 300))
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score and time.monotonic() < deadline - 200:
            a_shr2 = _run_lbfgs_improved(a_shr, deadline, min_time_remaining=200.0, tag="post_shrink")
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n[Phase 3] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_cd, sc_cd = _coord_descent(best_seq, min(deadline - 200, time.monotonic() + 120), n_sweeps=100)
        update_best(a_cd, sc_cd, "coord_descent")

    # === Phase 4: Multi-n search ===
    print(f"\n[Phase 4] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 6144, 8192]
    if n_parent in (3072, 5120):
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if parent is not None:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = np.ones(n_try) / n_try
        else:
            a_init = np.ones(n_try) / n_try

        # Quick L-BFGS
        a_opt = _run_lbfgs_improved(a_init, deadline, min_time_remaining=100.0, tag=f"n{n_try}")
        update_best(a_opt, _score(a_opt), f"n_{n_try}")

        # Subgradient on this
        if time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _subgrad_polish(a_opt, min(deadline - 50, time.monotonic() + 60),
                                              n_max_steps=1500, lr_init=0.015)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_sg")

    # === Phase 5: Periodic perturbation restarts ===
    print(f"\n[Phase 5] Periodic perturbations (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_perturb = 4
        for i in range(n_perturb):
            if time.monotonic() > deadline - 40:
                break
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            # Different types of perturbations
            if i == 0:
                # Small random noise
                z = z + 0.03 * rng.standard_normal(len(z))
            elif i == 1:
                # Low-frequency modulation
                z = _make_periodic_modulation(z, 0, 0.05, rng)
            elif i == 2:
                # High-frequency modulation
                t = np.arange(len(z))
                z = z + 0.04 * np.cos(2 * np.pi * rng.integers(2, len(z) // 4) * t / len(z))
            else:
                # Block swap
                block = len(z) // 8
                for _ in range(3):
                    i1 = rng.integers(0, len(z) - 2 * block)
                    i2 = rng.integers(0, len(z) - 2 * block)
                    if abs(i1 - i2) > block:
                        tmp = z[i1:i1 + block].copy()
                        z[i1:i1 + block] = z[i2:i2 + block]
                        z[i2:i2 + block] = tmp
            z = z - z.mean()
            a_pert = _softmax(z)
            a_r = _run_lbfgs_improved(a_pert, deadline, min_time_remaining=30.0, tag=f"pert{i}")
            update_best(a_r, _score(a_r), f"perturb_{i}")

    # === Phase 6: Final fine polish ===
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()

        # Very fine L-BFGS
        beta_final = mb * 0.0001
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(5000, max(100, int(remaining / 0.04)))
            try:
                result = minimize(
                    lambda z_l: _smoobj(z_l, beta_final, n),
                    z,
                    jac=lambda z_l: _smograd(z_l, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs_fine")
            except Exception:
                pass

        # Final subgradient with very tight tolerance
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=500, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg_tight")

        # Final coordinate descent
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _coord_descent(best_seq, deadline - 2, n_sweeps=50)
            update_best(a_fin3, sc_fin3, "final_cd")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
