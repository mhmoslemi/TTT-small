"""
Refined multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses block coordinate descent (alternating a-optimization and peak-weight updates), FISTA,
Adam in log-space, diverse restarts with peak-aware perturbations, and multi-length exploration.
Key improvements: two-timescale optimization, active-set-aware perturbations, and aggressive
beta annealing for final polishing.
"""
import numpy as np
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) == 1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    """Gradient of sum_k w_k * (conv(a,a))_k with respect to a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    """Compute softmax weights over convolution values."""
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _fista_quadratic(a_init, w, deadline, max_iter=5000):
    """FISTA to minimize a^T Q a (i.e., sum_k w_k*(conv(a,a))_k) on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    # Estimate Lipschitz constant
    grad0 = _compute_grad(a_curr, w)
    L = 2.0 * float(np.max(np.abs(grad0))) + 1.0
    step = 1.0 / max(L, 1e-10)

    obj_prev = float(np.dot(w, _full_convolve(a_curr, a_curr)))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        # Periodic check and adaptive L
        if it % 500 == 0:
            c_new = _full_convolve(a_next, a_next)
            obj_new = float(np.dot(w, c_new))
            if it > 0 and obj_prev - obj_new > 1e-15:
                max_c = float(c_new.max())
                L = max(L * 1.5, 4.0 * n * max_c)
                step = 1.0 / L
            obj_prev = obj_new

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    return a_curr


def _block_coord_descent(a_init, deadline, n_beta_steps=8, iters_per_phase=2000):
    """Block coordinate descent: alternate a-optimization and w-update."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    c = _full_convolve(a, a)
    max_c = float(c.max())
    best_score = _score_norm(a)
    best_a = a.copy()

    for phase in range(n_beta_steps):
        if time.monotonic() > deadline:
            break
        beta = max_c * (0.05 / (phase + 1))
        beta = max(beta, 1e-8 * max_c)

        w = _compute_weights(c, beta)

        # Optimize a for fixed w using FISTA
        phase_deadline = min(deadline, time.monotonic() + 30.0)
        a = _fista_quadratic(a, w, phase_deadline, max_iter=iters_per_phase)
        a = a / a.sum()

        c = _full_convolve(a, a)
        max_c = float(c.max())
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max(max_iter_total - t, 0)
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            w = _compute_weights(c, beta)
            grad = _compute_grad(a, w)

            # Scale gradient to account for softmax parameterization
            # d/da_i [f(a)] -> d/dz_i [f(softmax(z))]
            grad_z = a * (grad - np.dot(a, grad))

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -30.0, 30.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 10000 == 0:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """Standard FISTA on the softmax-smoothed max objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / max(beta, 1e-10)
    step = 1.0 / max(L, 1e-10)

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _compute_weights(c_y, beta)
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = max(4.0 * n * max_c_new / max(beta, 1e-10), L * 0.9)
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.2) ** 2) + np.exp(-30.0 * (t - 0.8) ** 2) + 0.1
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(777)
        a = rng.random(n) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_gaussian(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _perturb_peak_aware(a, c, rng, n_peaks=5, strength=0.3):
    """Perturb a in a direction that targets the top peaks of the convolution."""
    n = len(a)
    # Find top peaks
    top_idx = np.argsort(c)[::-1][:n_peaks]
    # Compute gradient at each peak
    grad_sum = np.zeros(n)
    for k in top_idx:
        a_rev = a[::-1]
        # grad_i = 2 * a[k-i] (for the specific peak at k)
        # We want to reduce (conv(a,a))_k, so move opposite to gradient
        grad_k = np.zeros(n)
        valid = (k >= 0) & (k < 2 * n - 1)
        if valid:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                j = k - i
                if 0 <= j < n:
                    grad_k[i] = 2.0 * a[j]
        grad_sum += grad_k

    # Add noise to gradient direction + some random component
    noise = strength * 0.1 * rng.standard_normal(n)
    update = -strength * grad_sum / (np.linalg.norm(grad_sum) + 1e-10) + noise

    z = np.log(np.maximum(a, 1e-15))
    z = z + update
    z = z - z.mean()
    return _softmax_compute(z)


def _perturb_fourier(a, rng, low_frac=0.1, high_frac=0.5, amplitude=0.1):
    """Perturb in Fourier domain with controlled frequency content."""
    n = len(a)
    N = _next_pow2(n)
    spectrum = np.fft.rfft(a, N)
    # Add noise to spectral components in a frequency band
    freqs = np.fft.rfftfreq(N)
    mask = (freqs >= low_frac / N) & (freqs <= high_frac / N)
    noise = amplitude * (2.0 * rng.random(np.sum(mask)) - 1.0)
    noisy_spectrum = spectrum.copy()
    noisy_spectrum[mask] += noise * amplitude
    # Also add a small amount of high-freq noise
    mask_h = freqs > high_frac / N
    noisy_spectrum[mask_h] += 0.01 * amplitude * (2.0 * rng.random(np.sum(mask_h)) - 1.0)
    a_new = np.fft.irfft(noisy_spectrum, N)[:n]
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def _coordinate_descent_polish(a, deadline, n_steps=500):
    """Try individual coordinate modifications to reduce the max convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    c = _full_convolve(best_a, best_a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Pick coordinates near the top peaks
        top_regions = np.argsort(c)[::-1][:10]
        # Pick a coordinate that contributes to a top peak
        k = top_regions[rng.integers(0, len(top_regions))]
        # The coordinate i contributes to (conv(a,a))_k via a_i * a_{k-i}
        i_options = []
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            j = k - i
            if 0 <= j < n:
                i_options.append(i)
        if not i_options:
            i_options = [rng.integers(0, n)]
        i = i_options[rng.integers(0, len(i_options))]

        for factor in [0.85, 0.9, 1.1, 1.15, 0.8, 1.2]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                c_try = _full_convolve(a_try, a_try)
                sc = float(2.0 * n * c_try.max())
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    c = c_try
                    break

    return best_a, best_score


def _optimize_sequence(a_init, deadline, label=""):
    """Full optimization pipeline for a single sequence."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    print(f"  [{label}] Start: score={best_score:.10f}, n={n}")

    # Phase A: FISTA coarse-to-fine
    c = _full_convolve(a, a)
    max_c = float(c.max())

    betas_fista = []
    beta = 0.02 * max_c
    for _ in range(6):
        betas_fista.append(beta)
        beta *= 0.4

    for i, beta in enumerate(betas_fista):
        if time.monotonic() > deadline:
            break
        phase_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.15)
        a, sc = _fista_smoothed(a, beta, phase_dl, max_iter=10000)
        a = a / a.sum()
        sc_full = _score_norm(a)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a.copy()
        if i < 2:
            print(f"  [{label}] FISTA beta={beta:.2e}: score={best_score:.10f}")

    # Phase B: Block coordinate descent
    if time.monotonic() < deadline - 30:
        bcd_dl = min(deadline - 30, time.monotonic() + (deadline - time.monotonic()) * 0.2)
        a, sc = _block_coord_descent(a, bcd_dl, n_beta_steps=6, iters_per_phase=1500)
        a = a / a.sum()
        sc_full = _score_norm(a)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a.copy()
        print(f"  [{label}] After BCD: score={best_score:.10f}")

    # Phase C: Adam log-space fine optimization
    if time.monotonic() < deadline - 20:
        c = _full_convolve(a, a)
        max_c = float(c.max())
        betas_adam = []
        beta = 0.001 * max_c
        for _ in range(8):
            betas_adam.append(beta)
            beta *= 0.5

        adam_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        a, sc = _adam_logspace(a, adam_dl, betas_adam, lr_init=0.01, max_iter_total=200000)
        a = a / a.sum()
        sc_full = _score_norm(a)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a.copy()
        print(f"  [{label}] After Adam: score={best_score:.10f}")

    # Phase D: Ultra-fine Adam
    if time.monotonic() < deadline - 10:
        c = _full_convolve(a, a)
        max_c = float(c.max())
        betas_ultra = [0.0001 * max_c, 0.00003 * max_c, 0.00001 * max_c]
        ultra_dl = min(deadline, time.monotonic() + (deadline - time.monotonic()) * 0.3)
        a, sc = _adam_logspace(a, ultra_dl, betas_ultra, lr_init=0.003, max_iter_total=100000)
        a = a / a.sum()
        sc_full = _score_norm(a)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a.copy()
        print(f"  [{label}] After ultra-fine: score={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        sc_parent = _score(parent)
        print(f"No parent, using uniform n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # ============================================================
    # Phase 1: Optimize parent at its native length
    # ============================================================
    print("\n=== Phase 1: Optimize parent ===")
    p1_deadline = t_start + 250.0
    a1, sc1 = _optimize_sequence(parent, min(p1_deadline, deadline), label="parent")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()
    print(f"Phase 1 done: best_score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # ============================================================
    # Phase 2: Multi-restart at n=4096 with diverse strategies
    # ============================================================
    print("\n=== Phase 2: Multi-restart n=4096 ===")
    n_main = len(best_seq)
    p2_deadline = t_start + 700.0
    n_restarts = 4
    per_restart = (min(p2_deadline, deadline) - time.monotonic()) / max(n_restarts, 1)

    strategies = [
        ("gauss_small", lambda a: _perturb_gaussian(a, 0.03, rng)),
        ("peak_aware", None),  # Special handling below
        ("fourier_mid", lambda a: _perturb_fourier(a, rng, 0.05, 0.3, 0.15)),
        ("gauss_med", lambda a: _perturb_gaussian(a, 0.08, rng)),
    ]

    for i, (name, fn) in enumerate(strategies):
        if time.monotonic() > min(p2_deadline, deadline) - 20:
            break
        if per_restart < 10:
            break

        a_current = best_seq.copy()
        a_current = a_current / a_current.sum()

        if name == "peak_aware":
            c_cur = _full_convolve(a_current, a_current)
            p_start = _perturb_peak_aware(a_current, c_cur, rng, n_peaks=8, strength=0.2)
        else:
            p_start = fn(a_current)
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

        phase_dl = time.monotonic() + per_restart
        a_r, sc_r = _optimize_sequence(p_start, phase_dl, label=f"restart_{name}")
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  New best from {name}: {best_score:.10f}")
        else:
            print(f"  {name}: {sc_r:.10f} (no improvement)")

    print(f"Phase 2 done: best_score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # ============================================================
    # Phase 3: Try larger length n=8192
    # ============================================================
    print("\n=== Phase 3: Try n=8192 ===")
    n_large = 8192
    p3_deadline = t_start + 1300.0
    if time.monotonic() < p3_deadline - 60:
        parent_large = _interp_profile(best_seq, n_large)
        sc_large = _score_norm(parent_large)
        print(f"  Interpolated to n={n_large}: score={sc_large:.10f}")

        a3, sc3 = _optimize_sequence(parent_large, min(p3_deadline, deadline), label="n8192")
        sc3_full = _score(a3)
        if sc3_full < best_score:
            best_score = sc3_full
            best_seq = a3.copy()
            print(f"  New best from n=8192: {best_score:.10f}")
        else:
            print(f"  n=8192: {sc3_full:.10f} (no improvement over {best_score:.10f})")

    print(f"Phase 3 done: best_score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # ============================================================
    # Phase 4: Second round of restarts on the best found
    # ============================================================
    print("\n=== Phase 4: Final restarts ===")
    p4_deadline = deadline - 30
    if time.monotonic() < p4_deadline - 30:
        n_best = len(best_seq)
        a_current = best_seq.copy() / best_seq.sum()

        n_final_restarts = 3
        per_final = (p4_deadline - time.monotonic()) / max(n_final_restarts, 1)
        if per_final > 15:
            final_strategies = [
                ("tiny_noise", lambda a: _perturb_gaussian(a, 0.005, rng)),
                ("fft_low", lambda a: _perturb_fourier(a, rng, 0.0, 0.05, 0.08)),
                ("peak2", None),
            ]
            for i, (name, fn) in enumerate(final_strategies):
                if time.monotonic() > p4_deadline - 10:
                    break
                if per_final < 10:
                    break

                a_curr = best_seq.copy() / best_seq.sum()
                if name == "peak2":
                    c_cur = _full_convolve(a_curr, a_curr)
                    p_start = _perturb_peak_aware(a_curr, c_cur, rng, n_peaks=3, strength=0.1)
                else:
                    p_start = fn(a_curr)
                    p_start = np.maximum(p_start, 1e-10)
                    p_start = p_start / p_start.sum()

                phase_dl = time.monotonic() + per_final
                a_r, sc_r = _optimize_sequence(p_start, phase_dl, label=f"final_{name}")
                sc_r_full = _score(a_r)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_seq = a_r.copy()
                    print(f"  New best from {name}: {best_score:.10f}")

    print(f"Phase 4 done: best_score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # ============================================================
    # Phase 5: Coordinate descent polish
    # ============================================================
    if best_seq is not None and time.monotonic() < deadline - 5:
        print("\n=== Phase 5: Coordinate descent polish ===")
        a_cd, sc_cd = _coordinate_descent_polish(
            best_seq, deadline - 2, n_steps=300
        )
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Coord descent improved: {best_score:.10f}")

    # ============================================================
    # Final validation and return
    # ============================================================
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL internal score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
