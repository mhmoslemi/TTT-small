import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is already normalized to sum=1."""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_conv(a, w):
    """
    Compute gradient of sum_k w_k * (a*a)_k with respect to a.
    w has length 2n-1.
    Returns gradient of length n.
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax_weights(c, beta):

    exponents = (c - c_max) / beta
    np.clip(exponents, -500, 500, out=exponents)
    w = np.exp(exponents)

    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w = w / s
    return w

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _Qa(a_vec, w):
    """Compute (Q(w) @ a) where Q(w)_{ij} = w_{i+j}. Uses FFT."""
    n = len(a_vec)
    N = _next_pow2(3 * n)
    w_rev = w[::-1]
    fa = np.fft.rfft(a_vec, N)
    fwr = np.fft.rfft(w_rev, N)
    conv = np.fft.irfft(fa * fwr, N)
    # (a * w_rev)_k = sum_j a_j w_rev[k-j] = sum_j a_j w[2n-2-k+j]
    # We want sum_j a_j w[i+j] for i=0,...,n-1
    # Set 2n-2-k = i => k = 2n-2-i
    # So result[i] = conv[2n-2-i]
    # For i=0: conv[2n-2], for i=n-1: conv[n-1]
    return conv[n-1:2*n-1][::-1]


def _grad_smoothed(a, w):
    """Gradient of sum_k w_k (a*a)_k, where w has length 2n-1."""
    n = len(a)
    return _Qa(a, w) * 2.0

    exponents = (c - c_max) / beta

        w[int(c.argmax())] = 1.0
    else:
        w = w / s
    return w

        step = 2.0 / L
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n
        grad_y = _grad_quadratic(y, w)
        a_next = _proj_simplex(y - step * grad_y)
        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        a_prev = a
        a = a_next
        t = t_next
    sc_f = _score_norm(a)
    if sc_f < best_sc:
        best_sc = sc_f
        best_a = a.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, betas, lr=0.01, max_iter_total=500000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.max()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr
    for beta in betas:
        if time.monotonic() > deadline:
            break
        itc = 0
        while itc < max_iter_total:
            if time.monotonic() > deadline:
                break
            itc += 1
            t += 1
            a = np.exp(z - z.max())
            s = a.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a = a / s
            c = _convolve(a, a)
            w = _softmax_weights(c, beta)
            grad_c = _grad_quadratic(a, w)
            mean_g = float(np.dot(a, grad_c))
            grad_z = a * (grad_c - mean_g)
            if not np.all(np.isfinite(grad_z)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = cur_lr * mh / (np.sqrt(vh) + eps)
            mu = np.max(np.abs(update))
            if mu > 0.05:
                update = update * (0.05 / mu)
            z = z - update
            z = z - z.max()
            if t % 200 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        cur_lr *= 0.6
    a_f = np.exp(z - z.max())
    a_f = a_f / a_f.sum()
    sc_f = _score_norm(a_f)
    if sc_f < best_score:
        best_score = sc_f
        best_a = a_f.copy()
    return best_a, best_score


def _tangential_fista(a_init, deadline, max_iter=50000):
    n = len(a_init)
    theta = np.log(np.maximum(a_init, 1e-15))
    theta = theta - theta.mean()
    a = a_init.copy()
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = _score_norm(a)
    step = 1.0 / n
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _convolve(a, a)
        max_c = c.max()
        top_k = max(50, n // 100)
        top_idx = np.argsort(c)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / top_k
        grad = _grad_quadratic(a, w)
        grad_proj = grad - np.dot(a, grad)
        col_sums = np.array([w[i:i + n].sum() for i in range(n)])
        L = 4.0 * float(col_sums.max()) + 0.1
        step = 2.0 / (L + 1.0 / n)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_theta = theta + ((t - 1.0) / t_next) * (theta - a_prev)
        a_y = np.exp(y_theta - y_theta.max())
        a_y = a_y / a_y.sum()
        theta_new = y_theta - step * grad_proj
        theta_new = theta_new - theta_new.mean()
        a_new = np.exp(theta_new - theta_new.max())
        a_new = a_new / a_new.sum()
        if it % 500 == 0:
            sc = _score_norm(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        a_prev = a
        a = a_new
        theta = theta_new
        t = t_next
    sc_f = _score_norm(a)
    if sc_f < best_sc:
        best_sc = sc_f
        best_a = a.copy()
    return best_a, best_sc


def _block_cd(a_init, deadline, max_rounds=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        j = rng.integers(0, n)
        old = a[j]
        improved = False
        for f in [0.999, 0.9995, 0.998, 1.0, 1.001, 1.0005, 1.002]:
            nv = old * f
            if nv < 0:
                continue
            a[j] = nv
            s = a.sum()
            if s < 1e-30:
                a[j] = old
                continue
            sc = _score_norm(a / s)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                improved = True
            else:
                a[j] = old
        a = a / a.sum()
        if not improved and r > 500:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    a_new = np.exp(z - z.max())
    return a_new / a_new.sum()


def _frequency_perturb(a, rng):
    n = len(a)
    fa = np.fft.rfft(a)
    mask = np.ones(len(fa), dtype=bool)
    n_active = max(10, n // 20)
    freqs = rng.choice(len(fa), size=n_active, replace=False)
    fa[freqs] *= (1.0 + 0.01 * rng.standard_normal(n_active))
    a_new = np.fft.irfft(fa, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s
