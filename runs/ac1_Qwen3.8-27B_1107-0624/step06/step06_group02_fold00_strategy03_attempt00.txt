def next_power_of_two(n):
    if n <= 1:
        return 1
    k = 1
    while k < n:
        k <<= 1
    return k

def elaborate_convolution(a, b):
    n_a, n_b = len(a), len(b)
    total_length = n_a + n_b - 1
    F = next_power_of_two(total_length)
    # First compute FFT of a and b
    a_fft = np.fft.rfft(a.astype(np.complex_), F)
    b_fft = np.fft.rfft(b.astype(np.complex_), F)
    # Multiply
    prod = a_fft * b_fft
    # Then IFFT
    result = np.fft.irfft(prod, F)[:total_length]
    return result

def score_candidate(seq):
    n = len(seq)
    # If sequence empty, penalty
    if n == 0:
        return np.inf
    total = float(np.sum(seq))
    if total < 0.01:
        return np.inf
    conv = elaborate_convolution(seq, seq)
    max_conv_val = float(np.max(conv))
    return 2.0 * n * max_conv_val / (total ** 2)
