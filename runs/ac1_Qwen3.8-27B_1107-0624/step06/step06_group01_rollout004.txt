import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)

    rho = np.where(cond)[0][-1]

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses improved FISTA with adaptive step size, Adam in log-space with noise injection
and cosine LR schedule, targeted peak shaving via gradient analysis, and
diversified restarts with multiple perturbation strategies.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_from_z(z):
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _softmax_weights(c, beta):
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _compute_grad_time(a, w):
    """Gradient of sum_k w_k * c_k w.r.t. a, where c = conv(a,a).
    Result has same length as a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1: 2 * n - 1]


def _compute_grad_logspace(a, c, w, n):
    """Gradient of the smoothed objective in log-space (z = log(a))."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_adaptive(a_init, betas, deadline, max_iter_per_beta=5000):
    """FISTA with softmax-smoothed max, adaptive Lipschitz, on simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    for beta in betas:
        if time.monotonic() > deadline:
            break

        a_prev = a_curr.copy()
        t_val = 1.0
        c_curr = _full_convolve(a_curr, a_curr)
        max_c = float(c_curr.max())
        L = 4.0 * n * max_c / beta
        step = 1.0 / L
        n_iter = 0
        no_improve_count = 0

        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            n_iter += 1

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta)
            grad = _compute_grad_time(y, w)

            a_next = _proj_simplex(y - step * grad)

            if it % 50 == 0:
                sc = _score_norm(a_next)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_next.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

                # Adaptive Lipschitz
                if it % 200 == 0:
                    c_new = _full_convolve(a_next, a_next)
                    max_c_new = float(c_new.max())
                    L = 4.0 * n * max_c_new / beta
                    step = 1.0 / L

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

            # Early break if no improvement for a while
            if no_improve_count > 20 and it > 500:
                break

    a_curr = a_curr / a_curr.sum()
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace_enhanced(a_init, deadline, beta_schedule, lr_init,
                            max_iter_total, noise_init, rng, cosine_decay=True):
    """Adam in log-space with noise injection and cosine learning rate decay."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    n_phases = len(beta_schedule)
    iters_per_phase = max_iter_total // n_phases if n_phases > 0 else max_iter_total

    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break

        # Decay noise over phases
        noise_level = noise_init * (0.5 ** phase_idx)

        phase_iter = 0
        while phase_iter < iters_per_phase:
            if time.monotonic() > deadline:
                break
            phase_iter += 1
            t += 1

            a = _softmax_from_z(z)
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)
            grad = _compute_grad_logspace(a, c, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine learning rate
            if cosine_decay:
                progress = t / max_iter_total
                current_lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            else:
                current_lr = lr_init

            update = current_lr * mh / (np.sqrt(vh) + eps)

            # Gradient clipping
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            # Noise injection
            if noise_level > 1e-6:
                noise = noise_level * rng.standard_normal(n)
            else:
                noise = 0.0

            z_new = z - update + noise
            z_new = np.clip(z_new, -100.0, 100.0)
            z_new = z_new - z_new.mean()
            z = z_new

            # Periodic evaluation
            if t % 40 == 0:
                a_chk = _softmax_from_z(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax_from_z(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_shaving_grad(a, deadline, n_iterations=10000, rng=None):
    """Targeted peak shaving using gradient information.
    Identifies the max peak and the coordinates contributing most to it,
    then reduces those coordinates."""
    if rng is None:
        rng = np.random.default_rng(77)

    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iterations):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        k_max = int(np.argmax(c))

        # Find contributing indices
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if hi < lo:
            continue

        # Contribution of index i to c[k_max] is a[i] * a[k_max - i]
        indices = np.arange(lo, hi + 1)
        partner_idx = k_max - indices
        valid = (partner_idx >= 0) & (partner_idx < n)
        if not np.any(valid):
            continue

        contributions = best_a[indices[valid]] * best_a[partner_idx[valid]]
        local_max = int(np.argmax(contributions))
        i_perturb = indices[valid][local_max]

        # Try reducing this coordinate
        improved = False
        for factor in [0.97, 0.95, 0.92, 0.88]:
            a_try = best_a.copy()
            a_try[i_perturb] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_try.copy()
                improved = True
                break

        if not improved:
            # Try a different random coordinate weighted by its contribution to the peak
            if it % 5 == 0:
                # Sample from the contribution distribution
                probs = contributions / (contributions.sum() + 1e-30)
                sample_idx = rng.choice(len(probs), p=probs)
                i_perturb = indices[valid][sample_idx]
                for factor in [0.95, 0.9, 0.85]:
                    a_try = best_a.copy()
                    a_try[i_perturb] *= factor
                    a_try = np.maximum(a_try, 0.0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try = a_try / s
                    sc = _score_norm(a_try)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_try.copy()
                        break

    return best_a, best_score


def _block_coordinate_descent(a, deadline, n_blocks=32, n_rounds=50):
    """Block coordinate descent: partition into blocks, optimize each block."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(55)

    block_size = max(1, n // n_blocks)
    n_actual_blocks = (n + block_size - 1) // block_size

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        # Shuffle blocks
        block_order = rng.permutation(n_actual_blocks)

        for bi in block_order:
            if time.monotonic() > deadline:
                break

            start = bi * block_size
            end = min(n, start + block_size)
            block_len = end - start
            if block_len < 2:
                continue

            # Try different scaling of this block
            current_block = best_a[start:end].copy()
            block_sum = current_block.sum()
            if block_sum < 1e-15:
                continue

            for scale in [0.9, 0.95, 1.05, 1.1]:
                a_try = best_a.copy()
                a_try[start:end] = current_block * scale
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    current_block = best_a[start:end].copy()
                    break

    return best_a, best_score


def _spectral_init(n, rng):
    """Initialize with a flat-spectrum random sequence."""
    N = _next_pow2(2 * n - 1)
    # Random phases
    k_max = N // 2
    phases = rng.uniform(0, 2 * np.pi, k_max + 1)
    # Flat magnitude
    magnitudes = np.ones(k_max + 1)
    spectrum = magnitudes * np.exp(1j * phases)

    # Make Hermitian for real signal
    full_spectrum = np.zeros(N, dtype=np.complex128)
    full_spectrum[:k_max + 1] = spectrum
    for k in range(2, k_max + 1):
        full_spectrum[N - k] = np.conj(spectrum[k])
    if k_max + 1 <= N:
        full_spectrum[k_max] = np.real(spectrum[k_max])

    a = np.fft.ifft(full_spectrum).real[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-50.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_from_z(z)

            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
