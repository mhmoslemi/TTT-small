"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated gradient with soft active set, LP recentering,
adaptive temperature staircase, perturbation restarts, and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    if T < 1e-15:
        idx = np.argmax(c)
        p = np.zeros_like(c)
        p[idx] = 1.0
    else:
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=10):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _weighted_active_gradient(a, c, threshold_frac=0.95):
    """Gradient using all positions above threshold_frac * max, weighted by proximity."""
    n = len(a)
    max_c = c.max()
    if max_c < 1e-15:
        return np.ones(n) * (2.0 * n)
    threshold = threshold_frac * max_c
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    weights = (c[active] - threshold) / (max_c - threshold + 1e-15)
    weights = np.maximum(weights, 0.1)
    weights /= weights.sum()
    p = np.zeros_like(c)
    p[active] = weights
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=20):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    if M < 1e-15:
        return a, M
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        # Backtracking line search
        t_val = 1.0
        for _ in range(20):
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_t = t_val
                break
            t_val *= 0.7
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    v = np.maximum(v, 0.0)
    if v.sum() < 1e-300:
        return np.ones(n) / n
    return v / v.sum()


def _nesterov_z_space(a_init, deadline, T_schedule, lr, max_iter=500000,
                       eval_interval=200, top_k=10, use_weighted=False,
                       lp_every=0):
    """Nesterov-accelerated gradient descent in log-space with staircase temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y_z = z.copy()  # Nesterov momentum variable
    t_k = 1.0
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience = best_score
    consecutive_worse = 0

    rows_lp = None
    cols_lp = None
    if lp_every > 0:
        rows_lp, cols_lp = _build_lp_indices(n)

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Determine current temperature
        frac = min(1.0, t / max(1, max_iter))
        T = T_schedule[int(frac * (len(T_schedule) - 1))]

        # Recover current iterate
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Compute gradient
        if use_weighted and T < T_schedule[0] * 0.1:
            g = _weighted_active_gradient(a, c, threshold_frac=0.92)
        elif T < T_schedule[0] * 0.01:
            g = _compute_active_gradient(a, c, top_k)
        else:
            g = _compute_soft_gradient(a, c, T)

        # Center the gradient (remove mean to stay on simplex)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        # Momentum step: update y using current z and gradient at y
        beta = (1.0 - t_k) / t_new
        y_z = z + beta * (z - y_z)
        # Clamp y_z
        y_z = np.clip(y_z, -80.0, 80.0)

        # Gradient step on y
        y_z = y_z - lr * z_grad
        y_z = np.clip(y_z, -80.0, 80.0)
        y_z = y_z - y_z.mean()

        z = y_z

        # Periodic LP recentering
        if lp_every > 0 and t > 0 and t % lp_every == 0:
            if time.monotonic() < deadline - 30:
                a_lp, _ = _lp_step(a, rows_lp, cols_lp, time_limit=10)
                z_lp = np.log(np.maximum(a_lp, 1e-15))
                z_lp = z_lp - z_lp.mean()
                z = 0.5 * z + 0.5 * z_lp
                y_z = z.copy()
                t_k = 1.0
                sc_lp = _score_norm(a_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_a = a_lp.copy()
                    patience = 0

        t_k = t_new

        # Evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
                consecutive_worse = 0
            else:
                consecutive_worse += 1
                if consecutive_worse % 50 == 0:
                    patience += 1
            if patience > 8:
                # Reset momentum and restart from best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                y_z = z.copy()
                t_k = 1.0
                patience = 0
                consecutive_worse = 0

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, top_k=10, use_weighted=False,
                  eval_interval=200, clip_norm=0.5):
    """Adam optimizer in log-space with adaptive temperature."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_weighted and T < T_start * 0.01:
            g = _weighted_active_gradient(a, c, threshold_frac=0.93)
        elif T < T_start * 0.001:
            g = _compute_active_gradient(a, c, top_k)
        else:
            g = _compute_soft_gradient(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10 and lr_mult > 0.1:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=50, top_m=50):
    """Coordinate descent: optimize one coordinate at a time using 1D search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Select coordinates to update: those with highest a_i * |grad_i|
        c = _fft_convolve(a, a)
        g = _compute_active_gradient(a, c, top_k=20)
        importance = a * np.abs(g)
        importance -= importance.min()
        top_idx = np.argsort(importance)[::-1][:top_m]
        top_idx = top_idx[rng.permutation(len(top_idx))]

        for i in top_idx:
            if time.monotonic() > deadline:
                break
            # Try scaling a[i] by factors
            base = a[i]
            if base < 1e-12:
                continue
            best_local = a[i]
            best_local_score = best_score
            for factor in [0.5, 0.75, 1.25, 1.5, 0.85, 1.15, 0.95, 1.05]:
                a[i] = base * factor
                sc = _score(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = a[i]
            a[i] = best_local
            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage optimizer: LP -> Nesterov (staircase T) -> Adam (active set)
    -> Perturbation restarts -> Coordinate descent final polish.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Build LP indices (cached)
    rows_lp, cols_lp = _build_lp_indices(n)
    print(f"LP indices built: {len(rows_lp)} nnz, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP iterations (0 - 60s)
    phase1_end = min(deadline, t_start + 60.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0
    while time.monotonic() < phase1_end:
        a, M = _lp_step(a, rows_lp, cols_lp, time_limit=12)
        sc = _score_norm(a)
        lp_count += 1
        if sc < prev_sc - 1e-12:
            stall = 0
        else:
            stall += 1
        full_sc = _score(a)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()
        if lp_count % 2 == 0:
            print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
        if stall > 12:
            break
        prev_sc = sc
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Nesterov with staircase temperature (60s - 400s)
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        # Staircase: 5 levels, each getting progressively smaller
        T_schedule = [0.01 * max_c, 0.005 * max_c, 0.002 * max_c, 0.0005 * max_c, 0.0001 * max_c]
        a_nest, sc_nest = _nesterov_z_space(best_a, phase2_end - 5,
                                            T_schedule, lr=0.015,
                                            max_iter=300000, top_k=10,
                                            use_weighted=True,
                                            eval_interval=150)
        full_sc = _score(a_nest)
        print(f"Phase 2 Nesterov: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_nest.copy()

    # Phase 3: Adam with adaptive temperature, active set (400s - 700s)
    phase3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase3_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_adam, sc_adam = _adam_z_space(best_a, phase3_end - 5,
                                         T_start=0.003 * max_c, T_end=0.00003 * max_c,
                                         lr_start=0.03, lr_end=0.003,
                                         max_iter=300000, top_k=10,
                                         use_weighted=True,
                                         eval_interval=150)
        full_sc = _score(a_adam)
        print(f"Phase 3 Adam-soft: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # Phase 4: Adam with tight active set (700s - 1000s)
    phase4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase4_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_active, sc_active = _adam_z_space(best_a, phase4_end - 5,
                                             T_start=0.0005 * max_c, T_end=0.00002 * max_c,
                                             lr_start=0.015, lr_end=0.001,
                                             max_iter=300000, top_k=5,
                                             use_weighted=False,
                                             eval_interval=100)
        full_sc = _score(a_active)
        print(f"Phase 4 Adam-active: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_active.copy()

    # Phase 5: Perturbation restarts (1000s - 1400s)
    phase5_end = min(deadline, t_start + 1400.0)
    n_best = len(best_a)
    if time.monotonic() < phase5_end:
        # Try different perturbation strategies
        strategies = [
            (0.01, "Gaussian small"),
            (0.02, "Gaussian med"),
            (0.04, "Gaussian large"),
            (0.0, "LP-restart"),
            (0.015, "Structured"),
        ]
        for attempt, (sigma, desc) in enumerate(strategies):
            if time.monotonic() > phase5_end - 60:
                break
            per_budget = min(phase5_end - 5, time.monotonic() + 70)

            if sigma == 0.0:
                # LP restart: run one LP step then optimize
                a_pert = best_a.copy()
                a_pert, _ = _lp_step(a_pert, rows_lp, cols_lp, time_limit=8)
            elif desc == "Structured":
                # Perturb in a structured way: add a low-frequency component
                freq = rng.integers(1, 5)
                phase = rng.uniform(0, 2 * np.pi)
                modulation = 1.0 + sigma * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                a_pert = best_a * modulation
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()
            else:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + sigma * rng.standard_normal(n_best)
                z = z - z.mean()
                a_pert = _softmax_stable(z)

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            a_pert_opt, sc_pert = _adam_z_space(a_pert, per_budget,
                                                 T_start=0.001 * max_cp, T_end=0.00002 * max_cp,
                                                 lr_start=0.02, lr_end=0.001,
                                                 max_iter=100000, top_k=5,
                                                 use_weighted=True,
                                                 eval_interval=100)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] ({desc}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 6: Try different n (1400s - 1550s)
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end:
        # Try n=2048 (downsample)
        n_small = 2048
        a_small = _interp_to_length(best_a, n_small)
        c_small = _fft_convolve(a_small, a_small)
        max_cs = float(c_small.max())
        a_small_opt, sc_small = _adam_z_space(a_small, min(phase6_end - 30, time.monotonic() + 60),
                                               T_start=0.002 * max_cs, T_end=0.00002 * max_cs,
                                               lr_start=0.025, lr_end=0.002,
                                               max_iter=100000, top_k=5,
                                               use_weighted=True,
                                               eval_interval=100)
        full_sc = _score(a_small_opt)
        print(f"Phase 6 n=2048: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_small_opt.copy()

    # Phase 7: Coordinate descent polish (1550s - 1700s)
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end:
        a_cd, sc_cd = _coord_descent(best_a, phase7_end - 3, max_rounds=30, top_m=40)
        full_sc = _score(a_cd)
        print(f"Phase 7 CoordDescent: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_cd.copy()

    # Phase 8: Final ultra-fine polish (1700s - deadline)
    if time.monotonic() < deadline - 8:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_z_space(best_a, deadline - 5,
                                           T_start=0.0001 * max_cf, T_end=0.00001 * max_cf,
                                           lr_start=0.003, lr_end=0.0002,
                                           max_iter=200000, top_k=3,
                                           use_weighted=False,
                                           eval_interval=80)
        full_sc = _score(a_final)
        print(f"Phase 8 Final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # One more coordinate descent pass
        if time.monotonic() < deadline - 5:
            a_cd2, sc_cd2 = _coord_descent(best_a, deadline - 3, max_rounds=20, top_m=30)
            full_sc2 = _score(a_cd2)
            if np.isfinite(full_sc2) and full_sc2 < best_score:
                best_score = full_sc2
                best_a = a_cd2.copy()
            print(f"Final CoordDescent: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
