"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- Adaptive active-set size based on convolution value distribution
- Cosine annealing temperature schedule
- Coordinate descent polishing
- Structured sinusoidal perturbations for restarts
- Multiple problem sizes (2048, 3072, 4096)
- Exact gradient computation for active set positions

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adaptive_active_set(c, top_k=10, threshold_frac=0.95):
    """Adaptively determine which convolution positions are 'active'."""
    c_max = c.max()
    # Positions where c is within threshold_frac of max
    active = np.where(c >= threshold_frac * c_max)[0]
    # But also include the top_k regardless
    top_indices = np.argsort(c)[::-1][:top_k]
    active = np.unique(np.concatenate([active, top_indices]))
    return active, c_max


def _compute_adaptive_gradient(a, c, active_indices):
    """Compute gradient using the active set positions."""
    n = len(a)
    k_list = active_indices
    grad = np.zeros(n)
    for k in k_list:
        # d(conv(a,a)[k])/da[i] = a[k-i] for valid i
        # valid i: 0 <= i < n and 0 <= k-i < n
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            grad[i_min:i_max+1] += a[k - i_max:k - i_min + 1][::-1]
    grad /= len(k_list)
    return grad * 2.0 * n


def _compute_softmax_gradient(a, c, T):
    """Soft gradient using softmax weights on convolution values."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    # convolve a with reversed p to get gradient
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=200000, use_adaptive=True, top_k=10,
                   threshold_frac=0.95, eval_interval=100,
                   clip_norm=0.5, momentum_decay=0.9):
    """Adam optimizer with adaptive active-set or softmax gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        # Cosine annealing
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** (frac ** 0.7)  # slower initial decrease
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_adaptive:
            active, c_max = _adaptive_active_set(c, top_k, threshold_frac)
            g = _compute_adaptive_gradient(a, c, active)
        else:
            g = _compute_softmax_gradient(a, c, T)

        # Center the gradient (preserves sum)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr_mult > 0.1:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, max_passes=5, eps=1e-10):
    """Coordinate descent: for each coordinate, find optimal value given others."""
    n = len(a)
    a = a.copy()
    c = _fft_convolve(a, a)
    c_max = c.max()
    
    for pass_idx in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                return a
            # Try adjusting a[i]
            # conv(a,a)[k] depends on a[i] for k in [i, i+n-1]
            # Specifically: conv(a,a)[i+j] += 2*a[i]*a[j] + (a[i] change)
            # d(conv(a,a)[k])/da[i] = a[k-i] for k in [i, i+n-1]
            # (plus a[i]^2 term for k = 2i, but that's second order)
            
            # Current contribution of a[i] to each c[k]:
            # c[k] = sum_j a[j] * a[k-j], the terms involving a[i] are:
            #   a[i] * a[k-i] (for k != 2i) or a[i]^2 (for k = 2i)
            # So if we change a[i] to a[i] + delta:
            #   c[k] changes by: delta * (a[k-i] + a[k-i]) = 2*delta*a[k-i] (k != 2i)
            #   c[2i] changes by: 2*delta*a[i] + delta^2
            
            # To minimize max(c), we want to reduce the largest c[k] that involves a[i].
            # The k values involving a[i] are k in [i, i+n-1] (clipped to [0, 2n-2]).
            k_min = max(0, i)
            k_max = min(2*n-2, i+n-1)
            c_slice = c[k_min:k_max+1]
            c_max_local = c_slice.max()
            
            if c_max_local < c_max * 0.99:
                continue  # This coordinate doesn't affect the max
            
            # The k that achieves the local max
            k_best = k_min + np.argmax(c_slice)
            # Gradient w.r.t. a[i]: sum over active k of a[k-i]
            # For the dominant k:
            a_opp = a[k_best - i] if 0 <= k_best - i < n else 0.0
            
            # Try reducing a[i]
            for frac in [0.99, 0.97, 0.95, 0.9]:
                a_new = a.copy()
                a_new[i] = a[i] * frac
                c_new = _fft_convolve(a_new, a_new)
                if c_new.max() < c_max - eps:
                    a = a_new
                    c = c_new
                    c_max = c_new.max()
                    improved = True
                    break
        
        if not improved:
            break
    
    # Re-normalize
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def _sinusoidal_perturb(a, freq, amp, phase=0.0, seed=None):
    """Add a sinusoidal perturbation and re-normalize."""
    n = len(a)
    if seed is not None:
        rng = np.random.default_rng(seed)
        phase = rng.uniform(0, 2*np.pi)
    x = np.arange(n)
    pert = 1.0 + amp * np.sin(2*np.pi*freq*x/n + phase)
    pert = np.maximum(pert, 0.01)
    a_new = a * pert
    a_new = a_new / a_new.sum()
    return a_new


def _lp_polish(a, deadline, max_iters=20):
    """LP-based polishing step."""
    n = len(a)
    # Build constraint matrix for convolution
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)

    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        vals = a[rows - cols]
        A = sp.csc_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                         method='highs', options={'time_limit': 10})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            if g.sum() < 1e-15:
                break
            g = g / g.sum()
            # Line search
            best_t = 0.0
            best_sc = _score_norm(a)
            for t_val in [0.1, 0.3, 0.5, 0.7, 1.0]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * g
            else:
                break
        except Exception:
            break
    return a


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: adaptive active-set with dynamic threshold, Nesterov momentum,
LP polishing, structured sinusoidal perturbations, multiple problem sizes,
and ultra-fine final annealing.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_fft(a, weights):
    """Compute 2*n * sum_k weights[k] * a[k-i] for all i using FFT."""
    n = len(a)
    w_rev = weights[::-1]
    conv_full = _fft_convolve(a, w_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=10, threshold_frac=0.97):
    """Adaptive active-set gradient."""
    n = len(a)
    c_max = c.max()
    active = np.where(c >= threshold_frac * c_max)[0]
    top_indices = np.argsort(c)[::-1][:top_k]
    active = np.unique(np.concatenate([active, top_indices]))
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    return _compute_gradient_fft(a, w)


def _compute_softmax_gradient(a, c, T):
    """Softmax-weighted gradient."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return _compute_gradient_fft(a, w)


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, use_adaptive=True, top_k=10,
                   threshold_frac=0.97, eval_interval=100,
                   clip_norm=0.5, nesterov=False):
    """Adam optimizer in z-space with optional Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        # Cosine-like annealing
        T = T_start * 0.5 * (1.0 + np.cos(np.pi * frac)) + T_end * 0.5 * (1.0 - np.cos(np.pi * frac))
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult

        if nesterov and t > 10:
            # Nesterov lookahead
            a_look = _softmax_stable(z - lr * m_buf / (np.sqrt(v_buf) + eps))
        else:
            a_look = _softmax_stable(z)

        c = _fft_convolve(a_look, a_look)
        if use_adaptive:
            g = _compute_active_gradient(a_look, c, top_k, threshold_frac)
        else:
            g = _compute_softmax_gradient(a_look, c, T)

        a = _softmax_stable(z)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_polish(a, deadline, max_iters=30):
    """LP-based polishing."""
    n = len(a)
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)

    prev_sc = _score_norm(a)
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        vals = a[rows - cols]
        A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 15})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            if g.sum() < 1e-15:
                break
            g = g / g.sum()
            best_t = 0.0
            best_sc = _score_norm(a)
            for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
            if best_t == 0.0:
                for t_val in [0.02, 0.05, 0.1]:
                    a_cand = (1.0 - t_val) * a + t_val * g
                    sc = _score_norm(a_cand)
                    if sc < best_sc:
                        best_sc = sc
                        best_t = t_val
            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * g
            else:
                break
            if best_sc > prev_sc - 1e-14:
                break
            prev_sc = best_sc
        except Exception:
            break
    return a


def _coord_descent(a, deadline, max_passes=3, frac_vals=None):
    """Coordinate descent on the most impactful coordinates."""
    if frac_vals is None:

    n = len(a)
    a = a.copy()
    c = _fft_convolve(a, a)
    c_max = float(c.max())

    for _ in range(max_passes):
        if time.monotonic() > deadline:
            break
        # Focus on coordinates that affect the top convolution positions
        top_k = np.argsort(c)[::-1][:8]
        coords = set()
        for k in top_k:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            coords.update(range(i_min, i_max + 1))
        coords = list(coords)
        if len(coords) > 500:
            # Sample
            step = len(coords) // 500
            coords = coords[::step]

        improved = False
        for i in coords:
            if time.monotonic() > deadline:
                return a
            for frac in frac_vals:
                a_new = a.copy()
                a_new[i] = a[i] * frac
                a_new = a_new / a_new.sum()
                c_new = _fft_convolve(a_new, a_new)
                c_new_max = float(c_new.max())
                if c_new_max < c_max - 1e-12:
                    a = a_new
                    c = c_new
                    c_max = c_new_max
                    improved = True
                    break
        if not improved:
            break

    s = a.sum()

    return a


def _sinusoidal_perturb(a, freq, amp, phase):
    n = len(a)
    x = np.arange(n)
    pert = 1.0 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    pert = np.maximum(pert, 0.01)
    a_new = a * pert
    a_new = a_new / a_new.sum()
    return a_new


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive active-set Adam with cosine annealing, LP polishing,
coordinate descent, structured sinusoidal perturbations, Nesterov
momentum, and multi-scale exploration.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_fft(a, weights):
    n = len(a)
    w_rev = weights[::-1]
    conv_full = _fft_convolve(a, w_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=10, threshold_frac=0.97):
    n = len(a)
    c_max = c.max()
    active = np.where(c >= threshold_frac * c_max)[0]
    top_indices = np.argsort(c)[::-1][:top_k]
    active = np.unique(np.concatenate([active, top_indices]))
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    return _compute_gradient_fft(a, w)


def _compute_softmax_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return _compute_gradient_fft(a, w)

def _lp_polish(a, deadline, max_iters=30):
    n = len(a)
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)

    prev_sc = _score_norm(a)
    for _ in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = float(c.max())
        vals = a[rows - cols]
        A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 15})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            if g.sum() < 1e-15:
                break
            g = g / g.sum()
            best_t = 0.0
            best_sc = _score_norm(a)
            for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)

                    best_t = t_val

                break
            if best_sc > prev_sc - 1e-14:

    return a

    a_new = a * pert
    a_new = a_new / a_new.sum()
    return a_new

    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)

        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        if time.monotonic() < deadline - 5:
            a_cd = _coord_descent(best_a, deadline - 5, max_passes=2)
            full_sc2 = _score(a_cd)
            print(f"  Final CoordDesc: score={full_sc2:.10f}")
            if np.isfinite(full_sc2) and full_sc2 < best_score:
                best_score = full_sc2
                best_a = a_cd.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
