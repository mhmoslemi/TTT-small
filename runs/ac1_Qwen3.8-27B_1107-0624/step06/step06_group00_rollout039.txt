"""
Optimized step function search using FFT-based gradients, adaptive temperature
annealing in log-space, structured initializations, coordinate descent polish,
and multi-scale exploration to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, c, T):
    """
    Compute gradient of max(conv(a,a)) using soft approximation.
    g[j] = 2 * sum_k p[k] * a[k-j] (for valid k-j)
    where p[k] = exp((c[k]-max_c)/T) / sum exp(...)
    """
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_flip = p[::-1]
    h = _fft_convolve(a, p_flip)
    # h has length n + (2n-1) - 1 = 3n-2
    # g[j] = 2 * h[2n-2-j] for j=0,...,n-1
    idx = 2*(n-1) - np.arange(n)
    g = 2.0 * h[idx]
    return g

def _compute_active_gradient(a, c, top_k=3):
    """Compute gradient using only the top-k peaks (hard)."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_flip = p[::-1]
    h = _fft_convolve(a, p_flip)
    idx = 2*(n-1) - np.arange(n)
    g = 2.0 * h[idx]
    return g

def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, use_active_set=False, active_k=5,
                  eval_interval=200, clip_norm=0.8, adaptive_T=False):
    """Adam optimizer in z-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if use_active_set and T < T_start * 0.01:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_gradient(a, c, T)
        
        # Project gradient to simplex tangent space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.1:
                current_lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _lp_step(a, rows, cols, time_limit=20):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.02, 0.05, 0.08]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M

def _cosine_window(n, alpha=1.0):
    x = np.linspace(0, 1, n)
    w = np.cos(np.pi * alpha * (x - 0.5)) ** 2
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def _gaussian_window(n, sigma=0.2):
    x = np.linspace(0, 1, n)
    w = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    return w / w.sum()

def _sinc_window(n, alpha=2.0):
    x = np.linspace(-0.5, 0.5, n)
    w = np.sinc(alpha * x) ** 2  # sinc(x) = sin(pi*x)/(pi*x)
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def _m_sequence(n):
    """Generate an m-sequence (maximum length sequence) of length ~n, converted to non-negative."""
    # Use a simple LFSR to generate a binary sequence, then map to non-negative
    # For simplicity, use a pseudo-random binary sequence with good auto-correlation properties
    rng = np.random.default_rng(42)
    # Generate a binary sequence and add small random perturbation
    b = rng.integers(0, 2, size=n).astype(float)
    # Map to [0,1] with some structure
    a = b * 0.5 + 0.25 + 0.25 * rng.standard_normal(n)
    a = np.maximum(a, 1e-6)
    return a / a.sum()

def _random_init(n, rng):
    """Random initialization with various distributions."""
    mode = rng.integers(0, 4)
    if mode == 0:
        a = rng.uniform(0.1, 1.0, size=n)
    elif mode == 1:
        a = rng.exponential(1.0, size=n) + 0.01
    elif mode == 2:
        # Bimodal
        centers = rng.uniform(0, 1, size=2)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - centers[0])**2) / 0.02) + np.exp(-((x - centers[1])**2) / 0.02)
        a = np.maximum(a, 1e-6)
    else:
        # Sparse peaks
        a = np.ones(n) * 0.01
        n_peaks = rng.integers(3, 10)
        peak_positions = rng.choice(n, size=n_peaks, replace=False)
        a[peak_positions] += rng.uniform(0.5, 2.0, size=n_peaks)
    return a / a.sum()

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradients with weighted active-set, gradual temperature
annealing in z-space with Adam, LP structure discovery, multi-scale
exploration (n=4096 and n=8192), and perturbation restarts.
Key improvements: weighted active-set gradient, more gradual LR decay,
longer ultra-low-temperature polish phase.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_flip = p[::-1]
    h = _fft_convolve(a, p_flip)
    idx = 2 * (n - 1) - np.arange(n)
    g = 2.0 * h[idx]
    return g


def _compute_weighted_active_gradient(a, c, top_k=3, decay=0.5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    c_max = float(c[k_indices[0]])
    weights = np.zeros(len(k_indices))
    for i, ki in enumerate(k_indices):
        weights[i] = np.exp(-decay * (c_max - c[ki]) / (c_max + 1e-30))
    weights = weights / weights.sum()
    p = np.zeros_like(c)
    for i, ki in enumerate(k_indices):
        p[ki] = weights[i]
    p_flip = p[::-1]
    h = _fft_convolve(a, p_flip)
    idx = 2 * (n - 1) - np.arange(n)
    g = 2.0 * h[idx]
    return g


def _compute_active_gradient(a, c, top_k=3):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_flip = p[::-1]
    h = _fft_convolve(a, p_flip)
    idx = 2 * (n - 1) - np.arange(n)
    g = 2.0 * h[idx]
    return g


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=20):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.01, 0.03, 0.05]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, use_active_set=False, active_k=3,
                  use_weighted=False, decay=0.5,
                  eval_interval=150, clip_norm=0.8):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        if use_active_set:
            if T < T_start * 0.005:
                if use_weighted:
                    g = _compute_weighted_active_gradient(a, c, active_k, decay)
                else:
                    g = _compute_active_gradient(a, c, active_k)
            else:
                g = _compute_soft_gradient(a, c, T)
        else:
            g = _compute_soft_gradient(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _cosine_window(n, alpha=1.0):
    x = np.linspace(0, 1, n)
    w = np.cos(np.pi * alpha * (x - 0.5)) ** 2
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def _gaussian_window(n, sigma=0.2):
    x = np.linspace(0, 1, n)
    w = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    return w / w.sum()


def _sinc_window(n, alpha=2.0):
    x = np.linspace(-0.5, 0.5, n)
    w = np.sinc(alpha * x) ** 2
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def _random_init(n, rng):
    mode = rng.integers(0, 4)
    if mode == 0:
        a = rng.uniform(0.1, 1.0, size=n)
    elif mode == 1:
        a = rng.exponential(1.0, size=n) + 0.01
    elif mode == 2:
        centers = rng.uniform(0.2, 0.8, size=2)
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - centers[0]) ** 2) / 0.01) + np.exp(-((x - centers[1]) ** 2) / 0.01)
        a = np.maximum(a, 1e-6)
    else:
        a = np.ones(n) * 0.01
        n_peaks = rng.integers(3, 10)
        peak_positions = rng.choice(n, size=n_peaks, replace=False)
        a[peak_positions] += rng.uniform(0.5, 2.0, size=n_peaks)
    return a / a.sum()


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
