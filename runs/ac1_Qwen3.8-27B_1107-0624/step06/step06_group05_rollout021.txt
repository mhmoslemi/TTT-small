LP-based descent + Adam polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set LP to find descent directions, line search for step size,
and gradient methods for final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize_scalar, minimize

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _full_conv(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_len]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_conv(a, a)
    return 2.0 * n * c.max()

    Returns the optimal b (or None on failure).

    n = len(a)
    b = cp.Variable(n)
    
    constraints = [b >= 0]
    
    # Add convolution constraints for active indices
    # conv(a, b)[k] = sum_i a[i] * b[k-i] for valid i

    for k in active_indices:

        if i_min > i_max:

        # The coefficient of b[j] in this constraint is a[k-j] where j = k-i

        coeffs = a[i_min:i_max+1][::-1]  # a[i_max], a[i_max-1], ..., a[i_min]

"""
LP-based descent + Adam + coordinate refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set linear programming to find descent directions, 1D line search for step size,
and gradient-based polishing with adaptive scheduling.

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar
from scipy.sparse import coo_matrix, csr_matrix


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_conv(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_conv(a, a)
    return float(2.0 * n * c.max())


def _build_conv_constraint_matrix(a, active_indices):
    """Build sparse matrix where row j gives conv(a,b)[active_indices[j]] as linear func of b."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_indices):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min > j_max:
            continue
        j_range = np.arange(j_min, j_max + 1)
        a_idx = k - j_range
        rows.extend(np.full(len(j_range), idx, dtype=np.int32))
        cols.extend(j_range.astype(np.int32))
        vals.extend(a[a_idx].tolist())
    if not rows:
        return csr_matrix((0, n))
    A = coo_matrix((np.array(vals), (np.array(rows), np.array(cols))), shape=(len(active_indices), n))
    return A.tocsr()


def _lp_direction(a, M, active_indices, timeout_s=15.0):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k]<=M for k in active_indices."""
    n = len(a)
    A = _build_conv_constraint_matrix(a, active_indices)
    if A.nnz == 0:
        return np.ones(n) / n
    c = -np.ones(n)
    b_ub = np.full(A.shape[0], M)
    bounds = [(0, None)] * n
    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout_s, 'presolve': True})
        if res.status == 0 and res.x is not None:
            b_val = np.maximum(res.x, 0.0)
            if b_val.sum() > 1e-10:
                return b_val
    except Exception:
        pass
    return None


def _line_search_score(a, b, S):
    """Find t in [0,1] minimizing score of (1-t)*a + t*b, where both have sum S."""
    c_a = _full_conv(a, a)
    c_ab = _full_conv(a, b)
    c_b = _full_conv(b, b)

    def obj(t):
        c_t = (1 - t) ** 2 * c_a + 2 * t * (1 - t) * c_ab + t ** 2 * c_b
        return float(c_t.max())

    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-6})
    return max(0.0, min(1.0, res.x))


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):

    if s < 1e-300:

def _coordinate_refine(a, deadline, num_rounds=3):
    """Simple coordinate descent: perturb individual elements to reduce score."""
    n = len(a)
    a = a.copy()
    current_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    coords_per_round = min(n, 200)

    for rnd in range(num_rounds):
        if time.monotonic() > deadline:
            break
        idx = rng.choice(n, size=coords_per_round, replace=False)
        for i in idx:
            if time.monotonic() > deadline:
                break
            best_val = a[i]
            best_sc = current_score
            # Try multiplicative perturbations
            for factor in [0.8, 0.85, 0.9, 1.1, 1.15, 1.2]:
                new_val = a[i] * factor
                if new_val < 0:
                    continue
                a[i] = new_val
                sc = _score_norm(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_val = new_val
                a[i] = new_val  # keep this trial
            a[i] = best_val
            current_score = best_sc

    return a, current_score

        a_new_sum = a_new.sum()
        if a_new_sum < 1e-10:
            a_new = a
        else:
            a_new = a_new / a_new_sum

        sc_new = _score_norm(a_new)
        if sc_new < score_now - 1e-10:
            a = a_new
            new_score = 2.0 * n * sc_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                stale_count = 0
            else:
                stale_count += 0.5
        else:
            # No improvement; try smaller step
            t_small = t_opt * 0.3
            a_new2 = (1 - t_small) * a + t_small * b
            a_new2 = np.maximum(a_new2, 0)
            if a_new2.sum() > 1e-10:
                a_new2 = a_new2 / a_new2.sum()
            sc_new2 = _score_norm(a_new2)
            if sc_new2 < score_now - 1e-12:
                a = a_new2
                new_score = 2.0 * n * sc_new2
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                stale_count = 0
            else:
                stale_count += 1

        lp_iter += 1
        if lp_iter % 5 == 0:
            print(f"  LP iter {lp_iter}: score={2.0*n*float(_full_conv(a,a).max()):.10f} "
                  f"active={len(active_indices)} t={t_opt:.4f} elapsed={time.monotonic()-t_start:.0f}s")

        # Adapt active threshold
        if stale_count > 5:
            active_threshold = max(0.7, active_threshold - 0.05)
            stale_count = 0
            print(f"  Widening active set: threshold={active_threshold:.2f}")
        elif lp_iter % 15 == 0:
            active_threshold = min(0.98, active_threshold + 0.01)

    print(f"Phase 1 done: best={best_score:.10f}, iters={lp_iter}")

    # === Phase 2: Adam optimization ===
    print("Phase 2: Adam optimization")
    phase2_end = min(deadline, t_start + 0.82 * (deadline - t_start))
    if time.monotonic() < phase2_end - 10:
        c_curr = _full_conv(a, a)
        max_cc = float(c_curr.max())
        betas = [0.002 * max_cc, 0.0005 * max_cc, 0.0001 * max_cc, 0.00003 * max_cc]
        per_beta = max(5.0, (phase2_end - time.monotonic() - 5) / len(betas))
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase2_end - 5:
                break
            a, sc_adam = _adam_opt(a, time.monotonic() + per_beta, beta, lr_init=0.005, max_iter=300000)
            full_sc = 2.0 * n * sc_adam
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
            print(f"  Adam beta={beta:.2e}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: L-BFGS polish ===
    print("Phase 3: L-BFGS polish")
    phase3_end = min(deadline, t_start + 0.93 * (deadline - t_start))
    if time.monotonic() < phase3_end - 8:
        c_curr = _full_conv(a, a)
        max_cc = float(c_curr.max())
        beta_lb = 0.00005 * max_cc
        a, sc_lb = _lbfgs_polish(a, phase3_end, beta_lb)
        full_sc = 2.0 * n * sc_lb
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()
        print(f"  L-BFGS: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 4: Coordinate refinement ===
    print("Phase 4: Coordinate refinement")
    phase4_end = deadline - 5.0
    if time.monotonic() < phase4_end - 20:
        a_copy = best_seq.copy()
        a_copy = a_copy / a_copy.sum()
        a_coord, sc_coord = _coordinate_refine(a_copy, phase4_end, num_rounds=5)
        full_sc = 2.0 * n * sc_coord
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_coord.copy()
        print(f"  Coord: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")

    # Final: verify and return
    final = np.clip(best_seq, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = np.ones(4096) / 4096
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
