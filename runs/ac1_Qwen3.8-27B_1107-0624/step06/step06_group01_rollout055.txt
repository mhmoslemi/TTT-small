"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA (coarse), L-BFGS with continuation (fine), and peak-directed
coordinate descent (polish). Tries multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp(x):
    m = np.max(x)
    if m == -np.inf:
        return -np.inf
    return m + np.log(np.sum(np.exp(x - m)))


def _smooth_obj_and_grad(z, beta):
    """
    Compute the smooth objective and its gradient in log-space.
    f(z) = 2n/beta * logsumexp(beta * conv(softmax(z), softmax(z)))
    """
    n = len(z)
    a = _softmax(z)
    b = _fft_convolve(a, a)  # length 2n-1
    
    # Smooth max
    scaled = beta * b
    lse = _logsumexp(scaled)
    f = (2.0 * n / beta) * lse
    
    # Softmax weights
    w = np.exp(scaled - lse)  # length 2n-1
    
    # Cross-correlation g[i] = sum_k w[k] * a[k-i]
    # Compute via FFT: g[i] = conv(w_rev, a)[2n-2-i]
    # w has length 2n-1, a has length n
    # conv length = (2n-1) + n - 1 = 3n-2
    N = _next_pow2(3 * n)
    w_rev = w[::-1]
    W = np.fft.rfft(w_rev, N)
    A = np.fft.rfft(a, N)
    conv_result = np.fft.irfft(W * A, N)[:3*n-2]
    g = conv_result[2*n-2::-1]  # g[i] = conv_result[2n-2-i], i=0..n-1
    # Wait, conv_result[2n-2-i] for i=0 gives conv_result[2n-2], i=1 gives conv_result[2n-3], etc.
    # So g = conv_result[2n-2, 2n-3, ..., n-1] = conv_result[2n-2][:n][::-1]
    # Actually: g[0] = conv_result[2n-2], g[1] = conv_result[2n-3], ..., g[n-1] = conv_result[n-1]
    g = conv_result[n-1:2*n-1][::-1]
    
    # grad_a[i] = 4n * g[i] (approximately, ignoring the k=2i correction)
    # grad_z[m] = 4n * a[m] * (g[m] - sum_j g[j]*a[j])
    dot = np.dot(g, a)
    grad_z = 4.0 * n * a * (g - dot)
    
    return f, grad_z


def _fista_smoothed(a_init, beta, deadline, max_iter=100000):
    """FISTA with softmax-smoothed max, operating on the simplex {a >= 0, sum(a)=1}."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _fft_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    best_score = 2.0 * n * max_c  # scaled score (sum=1)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _fft_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] ≈ 2 * cross_correlate(w, y)[...]
        # Use the same FFT trick
        N = _next_pow2(3 * n)
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, N)
        Y = np.fft.rfft(y, N)
        conv_res = np.fft.irfft(W * Y, N)[:3*n-2]
        g = conv_res[n-1:2*n-1][::-1]
        grad = 2.0 * g
        
        a_next = y - step * grad
        # Project onto simplex
        a_next = _proj_simplex(a_next)
        
        if it % 500 == 0:
            sc = 2.0 * n * float(_fft_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _fft_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = 2.0 * n * float(_fft_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lbfgs_phase(a_init, deadline, max_beta_steps=8):
    """
    L-BFGS optimization in log-space with continuation on beta.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    c_init = _fft_convolve(a_init, a_init)
    max_c = float(c_init.max())
    
    # Beta schedule: start moderate, increase
    betas = [0.05 * max_c, 0.02 * max_c, 0.01 * max_c, 0.005 * max_c,
             0.002 * max_c, 0.001 * max_c, 0.0005 * max_c, 0.0002 * max_c]
    betas = betas[:max_beta_steps]
    
    best_a = a_init.copy()
    best_score = 2.0 * n * max_c
    
    time_per_beta = max(5.0, (deadline - time.monotonic()) / len(betas))
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        phase_deadline = min(deadline, time.monotonic() + time_per_beta)
        
        # Define the objective and gradient for L-BFGS
        def func_and_grad(z_vec):
            f, grad = _smooth_obj_and_grad(z_vec, beta)
            if not np.isfinite(f):
                return 1e10, np.zeros_like(z_vec)
            if not np.all(np.isfinite(grad)):
                grad = np.zeros_like(z_vec)
            return f, grad
        
        # Run L-BFGS
        try:
            result = minimize(
                func_and_grad,
                z,
                jac=True,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-8,
                         'maxls': 50, 'maxfun': 1000}
            )
            z = result.x
        except Exception:
            pass
        
        # Evaluate
        a = _softmax(z)
        c = _fft_convolve(a, a)
        sc = 2.0 * n * float(c.max())
        a = a / a.sum()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        print(f"    L-BFGS beta={beta:.6f}: score={sc:.10f}")
        time_per_beta = max(5.0, (deadline - time.monotonic()) / max(1, len(betas) - min(int(time.monotonic() - (deadline + time_per_beta * len(betas)), 0), len(betas))))
    
    return best_a, best_score


def _peak_directed_cd(a, deadline, n_rounds=50):
    """
    Coordinate descent that specifically targets the peak lag.
    At each step, identify the max lag and try to reduce it by adjusting
    coordinates that contribute most to that lag.
    """
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * float(_fft_convolve(best_a, best_a).max())
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        b = _fft_convolve(best_a, best_a)
        k_star = int(np.argmax(b))
        current_peak = b[k_star]
        
        # Find the coordinates that contribute most to b[k_star]
        # b[k_star] = sum_i a[i] * a[k_star - i]
        # The contributing pairs are (i, k_star-i) for valid i
        contributions = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contributions[i] = a[i] * a[j]
        
        # Sort by contribution (descending)
        order = np.argsort(contributions)[::-1]
        
        # Try reducing the top contributors and increasing the bottom ones
        improved = False
        n_top = min(20, n // 100)
        n_bot = min(20, n // 100)
        
        for i in order[:n_top]:
            if time.monotonic() > deadline:
                break
            # Try reducing a[i]
            for factor in [0.95, 0.9, 0.85]:
                a_try = best_a.copy()
                a_try[i] *= factor
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try /= s
                    sc = 2.0 * n * float(_fft_convolve(a_try, a_try).max())
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_try.copy()
                        improved = True
                        break
        
        for i in order[-n_bot:]:
            if time.monotonic() > deadline:
                break
            # Try increasing a[i]
            for factor in [1.1, 1.15, 1.2]:
                a_try = best_a.copy()
                a_try[i] *= factor
                s = a_try.sum()
                if s > 1e-15:
                    a_try /= s
                    sc = 2.0 * n * float(_fft_convolve(a_try, a_try).max())
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_try.copy()
                        improved = True
                        break
        
        if not improved:
            break
    
    return best_a, best_score
