def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:

    return a_new / s

        # Step
        a_next = _proj_simplex(y - lr * grad)
        
        # Check improvement
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
        
        # Decay
        lr *= lr_decay
        beta = max(beta * beta_decay, 1e-15 * c_max)
        
        # Adaptive k: increase k as we converge (look at more peaks)
        if it % 500 == 0:
            k = min(len(c_y), max(4, int((top_k_frac * (1.0 + it / max_iter)) * len(c_y))))
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_sc


def _hybrid_optimize(a_init, deadline, top_k_frac=0.01):
    """
    Hybrid: alternate between subgradient steps and spectral flattening.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    c = _convolve(a, a)
    c_max = float(c.max())
    
    # Phase A: Subgradient with decreasing beta
    k = max(4, int(top_k_frac * len(c)))
    beta = 0.05 * c_max
    lr = 0.5 / n
    t_val = 1.0
    a_curr = a.copy()
    a_prev = a_curr.copy()
    
    it = 0
    max_iter = 5000000
    while time.monotonic() < deadline:
        # Subgradient step
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _convolve(y, y)
        grad = _topk_gradient(y, c_y, k, beta)
        a_next = _proj_simplex(y - lr * grad)
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
        it += 1
        
        # Adaptive parameters
        if it % 200 == 0:
            beta *= 0.95
            lr *= 0.99
            # Increase k slowly
            k = min(len(c_y), int(k * 1.05) + 1)
        
        # Periodic spectral flattening
        if it % 500 == 250:
            a_curr = _spectral_flatten(a_curr, gamma=0.1 / c_max, iters=1)
            t_val = 1.0
            a_prev = a_curr.copy()
        
        # Track best
        if it % 200 == 0:
            sc = _score_norm(a_curr)
            if sc < best_sc:
                best_sc = sc
                best_a = a_curr.copy()
        
        if beta < 1e-14 * c_max:
            # Reset beta, switch to finer
            beta = 1e-6 * c_max
            lr = 0.05 / n
            t_val = 1.0
            a_prev = a_curr.copy()
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_sc


def _fine_polish(a_init, deadline):
    """
    Very fine optimization: small lr, very small beta, top-1 or top-2 focus.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    c = _convolve(a, a)
    c_max = float(c.max())
    
    # Focus on top 1-2 peaks with very small beta
    for k in [1, 2, 3]:
        if time.monotonic() > deadline:
            break
        beta = 1e-5 * c_max
        lr = 0.01 / n
        t_val = 1.0
        a_curr = a.copy()
        a_prev = a_curr.copy()
        
        max_inner = 2000000
        for it in range(max_inner):
            if time.monotonic() > deadline - 5:
                break
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _convolve(y, y)
            grad = _topk_gradient(y, c_y, k, beta)
            a_next = _proj_simplex(y - lr * grad)
            
            a_prev = a_curr
            a_curr = a_next
            t_val = t_next
            
            if it % 50 == 0:
                sc = _score_norm(a_curr)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_curr.copy()
            
            # Decay
            lr *= 0.9998
            beta *= 0.9999
        
        a = a_curr.copy()
        sc_chk = _score_norm(a)
        if sc_chk < best_sc:
            best_sc = sc_chk
            best_a = a.copy()
    
    return best_a, best_sc


def _gen_cosine_start(n, phase_shift=0.0):
    """Generate a cosine-windowed starting sequence."""
    i = np.arange(n)
    t = i / n
    a = np.cos(np.pi * (t - 0.5)) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _gen_two_hump(n, ratio=0.6):
    """Two-hump structure: high on one side, lower on other."""
    i = np.arange(n)
    t = i / n
    a = np.exp(-4.0 * (t - ratio) ** 2) + 0.05 * np.exp(-4.0 * (t - (1-ratio)) ** 2) + 0.01
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _gen_random_structured(n, rng, sparsity=0.3):
    """Random with structured sparsity."""
    a = np.zeros(n)
    mask = rng.random(n) < sparsity
    a[mask] = rng.random(mask.sum()) + 0.01
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _gen_zadoff_chu_lp(n, rng):
    """Generate a non-negative sequence inspired by Zadoff-Chu structure."""
    # Use a binary pattern based on quadratic residue
    root = 2
    i = np.arange(n)
    # Phase pattern
    phase = (root * i * i) % n
    # Convert to amplitude pattern (non-negative)
    a = np.cos(np.pi * phase / n) ** 2 + 0.1
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

    # Prioritize parent and best-performing starts
