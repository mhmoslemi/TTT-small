def _lp_step(a, M):
    """One LP step: find b >= 0 maximizing sum(b) s.t. (a*b)_j <= M for all j.
    Returns the optimal b.
    """
    n = len(a)
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    
    # Build constraint matrix: A[j, k] = a_{j-k} if 0 <= j-k < n
    # A is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(n):
        # b_k appears in c_j for j = k to k+n-1 (i.e., j-k is 0 to n-1)
        # coefficient is a_{j-k}
        for j in range(k, k + n):
            if j < 2 * n - 1:
                rows.append(j)
                cols.append(k)
                vals.append(a[j - k])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: A @ b <= M
    b_ub = np.full(2*n-1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return np.zeros(n)

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses LP-based iterations, FISTA with full/top-K softmax, Adam in log-space,
and enhanced coordinate descent polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _cross_corr(a, w, n_out=None):
    """Compute cross-correlation: h[i] = sum_k a[k] * w[k+i]
    where w is of length 2n-1 (the convolution output).
    Result has length n.
    """
    n = len(a)
    if n_out is None:
        n_out = n
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n_out]
    return h


def _softmax_weights(c, beta, top_k=None):
    """Compute softmax weights over convolution values. Optionally restrict to top-k."""
    n_c = len(c)
    if top_k is not None and top_k < n_c:
        topk_thresh = np.partition(c, -top_k)[-top_k]
        mask = c >= topk_thresh
        if mask.sum() == 0:
            mask = np.zeros(n_c, dtype=bool)
            mask[np.argmax(c)] = True
        c_masked = np.where(mask, c, -np.inf)
    else:
        c_masked = c
    
    c_shift = c_masked - c_masked.max()
    exp_vals = np.exp(c_shift / beta)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros(n_c)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_vals / w_sum
    return w


def _grad_conv(a, w):
    """Gradient of sum_j w[j] * (a*a)[j] with respect to a.
    = 2 * cross_correlation(a, w)
    """
    return 2.0 * _cross_corr(a, w)


def _logspace_grad(a, w, n):
    """Gradient in log-space for normalized problem (sum(a)=1).
    dF/dz_i = 2n * a_i * (grad_i - mean_grad)
    """
    grad_a = _grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _fista_with_softmax(a_init, deadline, betas, top_k_list, max_iter=None):
    """FISTA on simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    total_iter = 0
    for bi, (beta, top_k) in enumerate(zip(betas, top_k_list)):
        if time.monotonic() > deadline:
            break
        c_curr = _full_convolve(a_curr, a_curr)
        max_c = float(c_curr.max())
        L = 8.0 * n / max(1e-10, max_c)
        step = 1.0 / L
        
        per_deadline = min(deadline, time.monotonic() + max(10, (deadline - time.monotonic()) / (len(betas) - bi)))
        
        for it in range(max_iter if max_iter else 100000):
            if time.monotonic() > per_deadline:
                break
            total_iter += 1
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta, top_k)
            grad = _grad_conv(y, w)
            
            a_next = _proj_simplex(y - step * grad)
            
            if total_iter % 500 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with softmax-smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        beta_iters = max_iter
        while it_in_phase < beta_iters:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
        
        m_buf *= 0.3
        v_buf *= 0.3
    
    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_iteration(a, M, relax=1.0):
    """One LP step: find b >= 0 maximizing sum(b) s.t. (a*b)_j <= M*relax."""
    n = len(a)
    # Build constraint matrix A[j, k] = a_{j-k} if 0 <= j-k < n
    rows = []
    cols = []
    vals = []
    
    for k in range(n):
        # b_k contributes to c_j for j in [k, k+n-1]
        j_start = k
        j_end = min(k + n, 2 * n - 1)
        for j in range(j_start, j_end):
            rows.append(j)
            cols.append(k)
            vals.append(a[j - k])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * relax)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 60})
        if result.success:
            b_opt = np.maximum(result.x, 0)
            s = b_opt.sum()
            if s > 1e-15:
                return b_opt / s
            else:
                return a.copy()
        else:
            return a.copy()
    except Exception as e:
        print(f"  LP error: {e}")
        return a.copy()


def _coord_descend_polish(a, deadline, max_passes=5, top_k_w=100):
    """Enhanced coordinate descent polishing."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    c = _full_convolve(a, a)
    rng = np.random.default_rng(42)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i_old = a[i]
            c = _full_convolve(a, a)
            
            # d_j = a_{j-i} for valid j (0 <= j-i < n)
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(max(0, i), min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            
            # c_j_new = c_j + (t - a_i_old) * d_j
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old  # since sum(a) = 1
            
            # Find top positions
            idx_top = np.argsort(c)[-top_k_w:]
            
            # Candidate t values
            t_candidates = [0.0, a_i_old * 0.5, a_i_old * 0.8, a_i_old,
                           a_i_old * 1.2, a_i_old * 1.5, a_i_old * 2.0]
            
            for j in idx_top:
                if d[j] > 1e-15:
                    # Optimal t for this constraint: minimize (c_base[j] + d[j]*t) / (S0+t)^2
                    # d/dt [(c_base[j] + d[j]*t) / (S0+t)^2] = 0
                    # d[j]*(S0+t)^2 - 2*(c_base[j] + d[j]*t)*(S0+t) = 0
                    # (S0+t) * [d[j]*(S0+t) - 2*(c_base[j] + d[j]*t)] = 0
                    # d[j]*S0 + d[j]*t - 2*c_base[j] - 2*d[j]*t = 0
                    # d[j]*S0 - 2*c_base[j] - d[j]*t = 0
                    # t = (d[j]*S0 - 2*c_base[j]) / d[j] = S0 - 2*c_base[j]/d[j]
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 5.0 * max(a_i_old, 0.05):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.7))
                        t_candidates.append(t_star * 1.3)
            
            t_candidates = [max(0.0, tc) for tc in t_candidates]
            t_candidates = list(set([round(tc, 12) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]
            
            best_t = a_i_old
            best_f = 2.0 * n * float(c.max())
            
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            a[i] = best_t
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines LP-based iterations, FISTA with softmax smoothing, Adam in log-space,
and enhanced coordinate descent. Exploits the structure of the autocorrelation
problem to find sequences with minimally peaked self-convolution.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
try:
    from scipy.optimize import linprog
    HAS_LINPROG = True
except ImportError:
    HAS_LINPROG = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _cross_corr(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return h


def _softmax_weights(c, beta, top_k=None):
    n_c = len(c)
    if top_k is not None and top_k < n_c:
        topk_thresh = np.partition(c, -top_k)[-top_k]
        mask = c >= topk_thresh
        if mask.sum() == 0:
            mask = np.zeros(n_c, dtype=bool)
            mask[np.argmax(c)] = True
        c_masked = np.where(mask, c, -np.inf)
    else:
        c_masked = c.copy()
    
    c_max = float(c_masked.max())
    exp_vals = np.exp((c_masked - c_max) / beta)
    w_sum = float(exp_vals.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros(n_c)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_vals / w_sum
    return w


def _grad_conv(a, w):
    return 2.0 * _cross_corr(a, w)


def _logspace_grad(a, w, n):
    grad_a = _grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_with_softmax(a_init, deadline, betas, top_k_list):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    total_iter = 0
    n_betas = len(betas)

    for bi in range(n_betas):
        beta = betas[bi]
        top_k = top_k_list[bi] if bi < len(top_k_list) else None
        if time.monotonic() > deadline:
            break

        c_curr = _full_convolve(a_curr, a_curr)
        max_c = float(c_curr.max())
        L = 8.0 * n / max(1e-10, max_c)
        step = 1.0 / L

        remaining_betas = n_betas - bi
        per_deadline = min(deadline, time.monotonic() + max(5, (deadline - time.monotonic()) / remaining_betas))

        while time.monotonic() < per_deadline:
            total_iter += 1
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta, top_k)
            grad = _grad_conv(y, w)
            a_next = _proj_simplex(y - step * grad)

            if total_iter % 300 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.8:
                update = update * (0.8 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)

            if t % 80 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 4000 == 0:
                current_lr = max(0.0003, current_lr * 0

.93)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _build_lp_matrix(a):
    n = len(a)
    row_starts = np.arange(n, dtype=np.int64)
    lengths = np.minimum(n, 2 * n - 1 - np.arange(n), dtype=np.int64)
    total_nnz = int(lengths.sum())
    rows = np.zeros(total_nnz, dtype=np.int32)
    cols = np.zeros(total_nnz, dtype=np.int32)
    vals = np.zeros(total_nnz, dtype=np.float64)
    offset = 0
    for k in range(n):
        ln = int(lengths[k])
        rs = int(row_starts[k])
        end = offset + ln
        rows[offset:end] = np.arange(rs, rs + ln, dtype=np.int32)
        cols[offset:end] = k
        vals[offset:end] = a[:ln]
        offset = end
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_iteration(a, M, relax=1.0, timeout=45):
    if not HAS_LINPROG:
        return a.copy()
    n = len(a)
    t0 = time.monotonic()
    A = _build_lp_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * relax)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': timeout, 'presolve': True})
        if time.monotonic() - t0 > timeout + 5:
            return a.copy()
        if result.success:
            b_opt = np.maximum(result.x, 0.0)
            s = b_opt.sum()
            if s > 1e-15:
                return b_opt / s
        return a.copy()
    except Exception:
        return a.copy()


def _lp_phase(a, deadline, max_iters=15, relax_init=1.05):

    c_curr = _full_convolve(a, a)
    M_curr = float(c_curr.max())

    for it in range(max_iters):
        if time.monotonic() > deadline - 10:
            break
        relax = relax_init * (0.8 ** it)
        relax = max(relax, 1.0)
        t0 = time.monotonic()

def _coord_descend_polish(a, deadline, max_passes=8, top_k_w=150):

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 7]:
            a += 0.15 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

    phase_d_end = min(deadline - 60, time.monotonic() + 200)
    print(f"Phase D: Coord descent (budget: {phase_d_end - time.monotonic():.0f}s)")
    if phase_d_end > time.monotonic() + 10:
        a_cd, _ = _coord_descend_polish(a0, phase_d_end, max_passes=10, top_k_w=200)
        sc_cd = _score(a_cd)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  CD: score={sc_cd:.10f} (t={time.monotonic()-t_start:.1f}s)")
        else:
            print(f"  CD: score={sc_cd:.10f} (no improvement)")
        a0 = a_cd

    # Phase E: More LP on refined solution
    remaining = deadline - time.monotonic()
    if remaining > 80:
        phase_e_end = time.monotonic() + min(remaining * 0.4, 120)
        print(f"Phase E: LP refinement (budget: {phase_e_end - time.monotonic():.0f}s)")
        a0, sc_e = _lp_phase(a0, phase_e_end, max_iters=8, relax_init=1.0)
        sc_e_full = _score(a0)
        if sc_e_full < best_score:
            best_score = sc_e_full
            best_seq = a0.copy()
            print(f"  LP refine: score={sc_e_full:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # Phase F: Final ultra-fine Adam
    remaining = deadline - time.monotonic()
    if remaining > 40:
        phase_f_end = time.monotonic() + min(remaining * 0.6, 100)
        print(f"Phase F: Final Adam (budget: {phase_f_end - time.monotonic():.0f}s)")
        c_curr = _full_convolve(a0, a0)
        max_cc = float(c_curr.max())
        betas_ult = [0.0008 * max_cc, 0.0003 * max_cc, 0.0001 * max_cc]
        a_ult, _ = _adam_logspace(a0, phase_f_end, betas_ult,
                                  lr_init=0.004, top_k=50, max_iter=50000)
        sc_ult = _score(a_ult)
        if sc_ult < best_score:
            best_score = sc_ult
            best_seq = a_ult.copy()
            print(f"  Final Adam: score={sc_ult:.10f} (t={time.monotonic()-t_start:.1f}s)")
        # Quick CD on result
        if time.monotonic() < deadline - 15:
            a_cd2, _ = _coord_descend_polish(a_ult, min(deadline, time.monotonic() + 40),
                                             max_passes=3, top_k_w=100)
            sc_cd2 = _score(a_cd2)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = a_cd2.copy()
                print(f"  Final CD: score={sc_cd2:.10f} (t={time.monotonic()-t_start:.1f}s)")

    # Phase G: Try alternate length
    remaining = deadline - time.monotonic()
    if remaining > 150 and best_seq is not None:
        n_alt = 8192 if n_main <= 4096 else 2048
        phase_g_end = time.monotonic() + min(remaining * 0.5, 180)
        print(f"Phase G: Try n={n_alt} (budget: {phase_g_end - time.monotonic():.0f}s)")
        a_alt = _interp_profile(best_seq, n_alt)
        c_alt = _full_convolve(a_alt, a_alt)
        max_ca = float(c_alt.max())
        betas_alt = [0.01 * max_ca, 0.003 * max_ca, 0.001 * max_ca, 0.0003 * max_ca]
        a_opt, _ = _adam_logspace(a_alt, phase_g_end, betas_alt,
                                  lr_init=0.006, top_k=150, max_iter=80000)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  Alt n={n_alt}: score={sc_opt:.10f} (t={time.monotonic()-t_start:.1f}s)")
        else:
            print(f"  Alt n={n_alt}: score={sc_opt:.10f} (no improvement)")
        if time.monotonic() < deadline - 10:
            a_cda, _ = _coord_descend_polish(a_opt, min(deadline, time.monotonic() + 50),
                                             max_passes=3, top_k_w=100)
            sc_cda = _score(a_cda)
            if sc_cda < best_score:
                best_score = sc_cda
                best_seq = a_cda.copy()
                print(f"  Alt CD: score={sc_cda:.10f}")

    # Final output
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.05 / 256)
            s = float(final.sum())
        target_sum = np.sqrt(2.0 * len(final))
        final = final * (target_sum / s)
        sc_final = _score(final)
        print(f"OUTPUT: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
