import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Gradient of <w, conv(a,a)> wrt a. Using cross-correlation."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _structured_init(n, kind="cos_sum"):

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _lp_direction_f(a, M, n_iter=100):
    """Approximate LP direction: find g >= 0, sum(g) max, s.t. (f*g)_j <= M.
    Use a simple iterative method instead of solving the full LP.
    """
    n = len(a)
    g = a.copy()
    g = g / g.sum()
    
    for it in range(n_iter):
        fg = _full_convolve(a, g)
        # Find violates
        violates = fg > M
        if not np.any(violates):
            break
        # Reduce g where it contributes to violations
        # (a*g)_j = sum_k a[k]*g[k+j-n+1]... this is cross-correlation
        # grad_j = a cross-correlated with indicator of violations
        viol_indicator = np.zeros(2 * n - 1)
        viol_indicator[violates] = fg[violates] - M
        # grad_g = 2 * cross_corr(a, viol_indicator)
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fv = np.fft.rfft(viol_indicator, N)
        grad = np.fft.irfft(np.conj(fa) * fv, N)[:n]
        g = np.maximum(g - 0.1 * grad, 0.0)
        s = g.sum()
        if s < 1e-15:
            g = a.copy()
        else:
            g = g / s
    
    return g

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines adaptive FISTA, mirror descent (multiplicative weights), Adam in log-space,
LP-inspired direction finding, coordinate descent polishing, and multi-length exploration.
Key improvements over previous: adaptive top-K weight selection, multiplicative updates
that naturally preserve non-negativity, and LP-based descent directions.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _adaptive_topk_weights(c, beta, min_k=20, max_k=300):
    n_c = len(c)
    c_max = float(c.max())
    threshold = c_max - 3.0 * beta
    k_eff = int(np.sum(c >= threshold))
    k = max(min_k, min(max_k, k_eff))
    k_min = int(np.sum(c >= c_max - 0.001 * c_max))
    k = max(k, min(max_k, k_min))
    if k >= n_c:
        topk_indices = np.arange(n_c)
    else:
        topk_thresh = np.partition(c, n_c - k)[n_c - k]
        topk_mask = c >= topk_thresh
        topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent_v2(a, n_passes=5, deadline=None):
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)

    for p in range(n_passes):
        if deadline and time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if deadline and time.monotonic() > deadline:
                break
            a_i = a[i]
            if a_i < 1e-15:
                continue
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            c = _full_convolve(a, a)
            c_base = c - a_i * d
            S0 = 1.0 - a_i
            top_k = 20
            idx_top = np.argsort(c_base + d * a_i)[-top_k:]
            t_candidates = [0.0, a_i, 2.0 * a_i, a_i * 0.5, a_i * 1.5]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = 2.0 * c_base[j] / d[j] - S0
                    if -0.5 * a_i < t_star < 5.0 * max(a_i, 0.1):
                        t_candidates.append(max(0.0, t_star))
            t_candidates = sorted(set([max(0.0, float(tc)) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i]
            best_t = a_i
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max_c / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            a[i] = best_t
            s = a.sum()
            if s > 1e-15:
                a = a / s

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _lp_direction_f(a, M_target, n_iter=200):
    n = len(a)
    g = a.copy()
    g = g / g.sum()
    for it in range(n_iter):
        fg = _full_convolve(a, g)
        exceeds = fg - M_target
        violates = exceeds > 0
        if not np.any(violates):
            break
        viol_indicator = np.zeros(2 * n - 1)
        viol_indicator[violates] = exceeds[violates]
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fv = np.fft.rfft(viol_indicator, N)
        grad = np.fft.irfft(np.conj(fa) * fv, N)[:n]
        step = 0.3 / max(1.0, np.linalg.norm(grad))
        g = np.maximum(g - step * grad, 0.0)
        s = g.sum()
        if s < 1e-15:
            g = a.copy()
        else:
            g = g / s
    return g


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining adaptive FISTA, mirror descent, Adam in log-space,
    LP-inspired directions, coordinate descent, and diversity restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    n_main = n_parent
    a_work = parent.copy()

    # Phase 1: FISTA with adaptive top-K (coarse to medium)
    print(f"\n--- Phase 1: FISTA adaptive (n={n_main}) ---")
    c0 = _full_convolve(a_work, a_work)
    max_c0 = float(c0.max())
    betas_fista = []
    beta = 0.05 * max_c0
    while beta > 0.001 * max_c0:
        betas_fista.append(beta)
        beta *= 0.6
    betas_fista = betas_fista[:6]

    phase1_end = min(t_start + 280, deadline - 600)
    a_work, sc1 = _fista_adaptive(a_work, phase1_end, betas_fista, max_iter=50000)
    sc1_full = _score(a_work)
    print(f"  FISTA: score={sc1_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc1_full < best_score:
        best_score = sc1_full
        best_seq = a_work.copy()

    # Phase 2: Mirror descent (multiplicative weights)
    print(f"\n--- Phase 2: Mirror descent (n={n_main}) ---")
    c2 = _full_convolve(a_work, a_work)
    max_c2 = float(c2.max())
    betas_md = []
    beta = 0.01 * max_c2
    while beta > 0.0001 * max_c2:
        betas_md.append(beta)
        beta *= 0.55
    betas_md = betas_md[:8]

    phase2_end = min(t_start + 450, deadline - 450)
    a_md, sc2 = _mirror_descent(a_work, phase2_end, betas_md, lr=0.4)
    sc2_full = _score(a_md)
    print(f"  Mirror descent: score={sc2_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc2_full < best_score:
        best_score = sc2_full
        best_seq = a_md.copy()
        a_work = a_md.copy()

    # Phase 3: Adam log-space (fine tuning)
    print(f"\n--- Phase 3: Adam log-space (n={n_main}) ---")
    c3 = _full_convolve(a_work, a_work)
    max_c3 = float(c3.max())
    betas_adam = []
    beta = 0.004 * max_c3
    while beta > 0.00002 * max_c3:
        betas_adam.append(beta)
        beta *= 0.5
    betas_adam = betas_adam[:10]

    phase3_end = min(t_start + 800, deadline - 300)
    a_adam, sc3 = _adam_logspace_v2(a_work, phase3_end, betas_adam, lr_init=0.008, max_iter=200000)
    sc3_full = _score(a_adam)
    print(f"  Adam: score={sc3_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc3_full < best_score:
        best_score = sc3_full
        best_seq = a_adam.copy()
        a_work = a_adam.copy()

    # Phase 4: LP-inspired direction + re-optimize
    print(f"\n--- Phase 4: LP direction + Adam (n={n_main}) ---")
    c4 = _full_convolve(a_work, a_work)
    M4 = float(c4.max()) * 0.98
    g_dir = _lp_direction_f(a_work, M4, n_iter=300)
    print(f"  LP direction computed, t={time.monotonic()-t_start:.1f}s")

    for t_blend in [0.3, 0.5, 0.7, 1.0]:
        if time.monotonic() > deadline - 120:
            break
        a_blend = (1.0 - t_blend) * a_work + t_blend * g_dir
        a_blend = np.maximum(a_blend, 1e-15)
        a_blend = a_blend / a_blend.sum()
        sc_blend = _score(a_blend)
        if sc_blend < best_score:
            best_score = sc_blend
            best_seq = a_blend.copy()
            print(f"  LP blend t={t_blend}: score={sc_blend:.10f}")

        c_b = _full_convolve(a_blend, a_blend)
        max_cb = float(c_b.max())
        betas_b = [0.003 * max_cb, 0.001 * max_cb, 0.0005 * max_cb, 0.0002 * max_cb]
        phase4_sub_end = min(time.monotonic() + 40, deadline - 80)
        a_opt_b, sc_b = _adam_logspace_v2(a_blend, phase4_sub_end, betas_b, lr_init=0.004, max_iter=30000)
        sc_b_full = _score(a_opt_b)
        if sc_b_full < best_score:
            best_score = sc_b_full
            best_seq = a_opt_b.copy()
            print(f"  LP blend t={t_blend} optimized: score={sc_b_full:.10f}")
        if sc_b_full < _score(a_blend):
            a_work = a_opt_b.copy()

    # Phase 5: Coordinate descent polishing
    print(f"\n--- Phase 5: Coord descent (n={n_main}) ---")
    phase5_end = min(t_start + 1400, deadline - 150)
    a_cd, sc4 = _coord_descent_v2(a_work, n_passes=5, deadline=phase5_end)
    sc4_full = _score(a_cd)
    print(f"  Coord descent: score={sc4_full:.10f}, t={time.monotonic()-t_start:.1f}s")
    if sc4_full < best_score:
        best_score = sc4_full
        best_seq = a_cd.copy()
        a_work = a_cd.copy()

    # Phase 6: Ultra-fine Adam on best
    if time.monotonic() < deadline - 80:
        print(f"\n--- Phase 6: Ultra-fine Adam (n={n_main}) ---")
        c6 = _full_convolve(a_work, a_work)
        max_c6 = float(c6.max())
        betas_ult = [0.0005 * max_c6, 0.0002 * max_c6, 0.0001 * max_c6, 0.00005 * max_c6]
        phase6_end = min(t_start + 1550, deadline - 50)
        a_ult, sc6 = _adam_logspace_v2(a_work, phase6_end, betas_ult, lr_init=0.003, max_iter=100000)
        sc6_full = _score(a_ult)
        print(f"  Ultra-fine: score={sc6_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc6_full < best_score:
            best_score = sc6_full
            best_seq = a_ult.copy()

    # Phase 7: Diversity restarts from best
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n--- Phase 7: Diversity restarts ---")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        remaining = deadline - time.monotonic()
        per_restart = max(15.0, remaining / 6.0)

        for ri in range(5):
            if time.monotonic() > deadline - 30:
                break

            if ri == 0:
                p = a_base + 0.02 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 2048))
                p = np.roll(a_base, shift).copy()
            elif ri == 2:
                freq = rng.uniform(0.5, 4.0)
                pert = 0.03 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p = a_base + pert
            elif ri == 3:
                p = a_base[::-1].copy()
            else:
                p = a_base * np.exp(0.04 * rng.standard_normal(n_best))

            p = np.maximum(p, 1e-12)
            p = p / p.sum()
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            betas_r = [0.003 * max_cp, 0.001 * max_cp, 0.0005 * max_cp, 0.0002 * max_cp]
            p_end = time.monotonic() + per_restart
            p_opt, sc_r = _adam_logspace_v2(p, p_end, betas_r, lr_init=0.006, max_iter=30000)
            sc_r_full = _score(p_opt)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = p_opt.copy()
                print(f"  Restart {ri}: score={sc_r_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
