Uses Adam in log-space with adaptive softmax-smoothed max, FFT-based gradient computation,

Key improvements: aggressive temperature schedule, top-peak focused gradients,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_norm(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_w(w, a, n):
    """Compute grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1.

    fw = np.fft.rfft(w, N)
    fr = np.fft.rfft(rev_a, N)
    conv = np.fft.irfft(fw * fr, N)
    return 2.0 * conv[n - 1:2 * n - 1]

                sorted_idx = np.argsort(c)[::-1]
                top_k = sorted_idx[:top_k_focus]
                c_shifted = (c - cm) / beta
                e = np.exp(c_shifted)
                # Boost top K values
                e[top_k] *= 10.0
                s = float(e.sum())
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s
            else:
                c_shifted = (c - cm) / beta
                e = np.exp(c_shifted)
                s = float(e.sum())
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s

            # Compute gradient: dJ/da[i] = 2n * 2 * sum_j w[i+j]*a[j]
            grad_a = 4.0 * n * _compute_grad_w(w, a, n)

            # Convert to log-space gradient: dJ/dz[i] = a[i]*(grad_a[i] - sum(grad_a*a))
            dot = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - dot)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** (t + 1))
            vh = v_buf / (1.0 - b2 ** (t + 1))

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_norm(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)

    a_final = _softmax_norm(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_logspace_simple(a_init, deadline, beta, lr_init, max_iter):
    """Simpler Adam in log-space with fixed beta. More aggressive learning."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_norm(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        grad_a = 4.0 * n * _compute_grad_w(w, a, n)
        dot = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - dot)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** (t + 1))
        vh = v_buf / (1.0 - b2 ** (t + 1))

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)

        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()

        if t % 20 == 0:
            a_chk = _softmax_norm(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.95)

    a_final = _softmax_norm(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_simplex(a_init, beta, deadline, max_iter):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = 4.0 * n * _compute_grad_w(w, y, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_passes=50, candidates_per_coord=8):
    """Exact coordinate descent: try multiple values for each coordinate."""
    n = len(a_init)
    best_a = a_init.copy()
    best_a = best_a / best_a.sum()
    best_score = _score_norm(best_a)
    rng = np.random.default_rng(123)

    # Precompute FFT size
    N_fft = _next_pow2(2 * n - 1)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        # Random permutation of coordinates for this pass
        order = rng.permutation(n)
        improved = False
        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = best_a[i]
            if old_val < 1e-10:
                # Try small positive values
                candidates = [1e-4, 5e-4, 1e-3, 5e-3]
                base = best_a.copy()
                base[i] = 0.0
                base_s = base.sum()
                b_base = _full_convolve(base, base)
                for t_val in candidates:
                    trial = base.copy()
                    trial[i] = t_val
                    trial = trial / trial.sum()
                    c = _full_convolve(trial, trial)
                    sc = float(2.0 * n * c.max())
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = trial.copy()
                        old_val = t_val
                        improved = True
                        break
                continue

            # Try scaling the current value
            candidates = [old_val * f for f in
                          [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 0.3, 2.0]]
            # Also try a few absolute values based on the mean
            mean_val = 1.0 / n
            candidates.extend([mean_val * f for f in [0.5, 1.0, 2.0]])

            base = best_a.copy()
            base[i] = 0.0
            base_s = base.sum()
            b_base = _full_convolve(base, base)

            for t_val in candidates:
                trial = base.copy()
                trial[i] = t_val
                s_trial = base_s + t_val
                if s_trial < 1e-15:
                    continue
                trial = trial / s_trial
                c = _full_convolve(trial, trial)
                sc = float(2.0 * n * c.max())
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = trial.copy()
                    improved = True
                    break

        if not improved and p > 3:
            break

    return best_a, best_score


def _peak_equalization(a_init, deadline, n_iters=200):
    """Targeted peak reduction: identify the max peak and specifically reduce it."""
    n = len(a_init)
    best_a = a_init.copy()
    best_a = best_a / best_a.sum()
    best_score = _score_norm(best_a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        k_max = int(np.argmax(c))
        peak_val = c[k_max]

        # The peak at k_max is sum_j a[j]*a[k_max-j]
        # To reduce it, we should reduce a[i] where a[k_max-i] is large
        # Compute the "sensitivity": how much does reducing a[i] reduce the peak?
        # d(peak)/da[i] = 2*a[k_max-i] (if valid)
        
        sens = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                sens[i] = 2.0 * best_a[j]

        # Find the coordinate with highest sensitivity
        i_star = int(np.argmax(sens))
        
        # Reduce a[i_star] and redistribute
        old_ir = best_a[i_star]
        if old_ir < 1e-8:
            break
            
        # Try different reduction factors
        best_trial = None
        best_trial_sc = best_score
        for factor in [0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3]:
            trial = best_a.copy()
            reduction = old_ir * (1.0 - factor)
            trial[i_star] = old_ir * factor
            # Redistribute to a random coordinate with low sensitivity
            trial_s = trial.sum()
            
            # Add the reduction to the coordinate with lowest sensitivity
            i_add = int(np.argmin(sens))
            if i_add != i_star:
                trial[i_add] += reduction
            
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s > 1e-15:
                trial = trial / s
                c_t = _full_convolve(trial, trial)
                sc_t = float(2.0 * n * c_t.max())
                if sc_t < best_trial_sc:
                    best_trial_sc = sc_t
                    best_trial = trial.copy()

        if best_trial is not None:
            best_a = best_trial
            best_score = best_trial_sc
        else:
            if it > 20:
                break

    return best_a, best_score


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_norm(z)


def _blend_profiles(a1, a2, weight):
    result = weight * a1 + (1.0 - weight) * a2
    result = np.maximum(result, 1e-10)
    return result / result.sum()
