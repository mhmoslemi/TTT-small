import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _proj_sym_simplex(v):
    """Project v onto the symmetric probability simplex."""
    n = len(v)
    # Average with reverse
    v_sym = 0.5 * (v + v[::-1])
    return _proj_simplex(v_sym)

def _subgrad_conv_max(a, n_top=5):
    """Compute subgradient of max_k (a*a)_k at point a.
    Returns the average of gradients over the top n_top positions."""
    n = len(a)
    c = _full_convolve(a, a)
    # Find top positions
    top_idx = np.argsort(c)[::-1][:n_top]
    g = np.zeros(n)
    for k in top_idx:
        # grad_k = 2 * P_k * a, where (P_k * a)_i = a_{k-i} if 0 <= k-i < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        indices = np.arange(lo, hi + 1)
        g[indices] += 2.0 * a[k - indices]
    g /= n_top
    return g, c.max()

    n = len(a_init)
    a = _proj_sym_simplex(a_init.copy())
    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())
    
    lr0 = 0.1
    best_lr = lr0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cmax = float(c.max())
        if cmax < best_cmax:
            best_cmax = cmax
            best_a = a.copy()
        
        # Subgradient
        g, _ = _subgrad_conv_max(a, n_top=3)
        
        # Adaptive step size
        lr = lr0 / np.sqrt(it + 1)
        
        # Try the step
        a_new = _proj_sym_simplex(a - lr * g)
        
        # Check if it improved (using exact max)
        c_new = _full_convolve(a_new, a_new)
        cmax_new = float(c_new.max())
        
        if cmax_new > cmax * 1.01:
            # Step was too large, reduce
            lr *= 0.5
            a_new = _proj_sym_simplex(a - lr * g)
        else:
            lr0 = min(lr0 * 1.05, 0.5)
        
        a = a_new
    
    return best_a, best_cmax

f(a) = max_k (a*a)_k

The subgradient at a is:
∂f(a) = { sum_{k in S} λ_k * grad_k(a) : S ⊆ argmax_k (a*a)_k, λ_k ≥ 0, sum λ_k = 1 }

where grad_k(a)_i = d/da_i [(a*a)_k] = 2 * a_{k-i} (with a_m = 0 for m outside [0,n-1]).

Wait, I showed earlier that d/da_i [(a*a)_k] = 2 * a_{k-i} (with zero padding). Let me verify:

(a*a)_k = sum_j a_j a_{k-j}

d/da_i = sum_j [δ_{ji} a_{k-j} + a_j δ_{k-j,i}]
= a_{k-i} + a_{k-i}  (first term: j=i gives a_{k-i}; second term: k-j=i gives j=k-i gives a_{k-i})
= 2 a_{k-i}

where a_m is defined as 0 for m not in [0, n-1].

So the subgradient component for position i is:
g_i = sum_{k in S} λ_k * 2 * a_{k-i}

For a single active position k: g_i = 2 * a_{k-i}.

For the average of top-K active positions (equal weights):
g_i = (2/K) * sum_{k in topK} a_{k-i}

This can be computed as: g_i = (2/K) * sum_{k in topK} a_{k-i} = (2/K) * (sum of a over positions that "correlate" with i through the topK positions).

For i = 0, ..., n-1: g_i = (2/K) * sum_{k in topK} a_{k-i} (with a_m = 0 for m ∉ [0,n-1]).

This is a sum of at most K shifted copies of a. Can be computed in O(K*n) time, or more efficiently with FFT.

For K=3 and n=4096, O(K*n) = 12K operations, which is fast.

The self-convolution matrix T (where (Ta)_k = (a*a)_k) is a specific structured matrix. The problem is to minimize ||Ta||_∞ / (sum a)^2 over the simplex.

Another perspective: we want a such that Ta is as "flat" as possible. This is related to the concept of a "flat top" in the autocorrelation.

In the frequency domain: if A(ω) = DFT(a), then the autocorrelation r = IDFT(|A(ω)|^2). We want r to be flat.

|r(ω)| = |A(ω)|^2. The "flatness" of r is related to the "flatness" of |A(ω)|^2 in a specific sense (not just pointwise flatness, but in the sense that its IDFT is flat).

One approach: iteratively modify a to make |A(ω)|^2 more "flat" in the appropriate sense. This is like the " Griffin-Lim" algorithm or "error reduction" methods in phase retrieval.

**Error reduction approach:**
1. Compute r = IDFT(|A(ω)|^2) = conv(a, a)
2. The "error" is e = r - mean(r) (we want r to be constant)
3. Modify a to reduce the error: a_new = a - μ * grad(red(e)/red(a))
4. The gradient of the "peakiness" of r with respect to a can be computed

"""
Symmetric minimax autocorrelation optimizer using accelerated projected subgradient
with exact active-set gradients, coordinate descent polishing, and multi-n exploration.
Exploits the reversibility symmetry of the objective to halve the search space.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _symmetrize(a):
    return 0.5 * (a + a[::-1])


def _proj_sym_simplex(v):
    n = len(v)
    v_sym = 0.5 * (v + v[::-1])
    return _proj_simplex(v_sym)


def _make_sym_init(n, kind="cosine"):
    half = n // 2
    if kind == "cosine":
        t = np.linspace(0, 1, half + 1)
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif kind == "sin":
        t = np.linspace(0, 1, half + 1)
        a = np.sin(np.pi * t) ** 0.5
    elif kind == "uniform":
        a = np.ones(half + 1)
    else:
        a = np.ones(half + 1)
    a = np.maximum(a, 1e-10)
    full = np.zeros(n)
    full[:half + 1] = a
    full[half + 1:] = a[-2::-1]
    return full / full.sum()


def _subgrad_topk(a, topk=3):
    """Compute subgradient of max_k (a*a)_k using top-k active positions."""
    n = len(a)
    c = _full_convolve(a, a)
    cmax = float(c.max())
    top_idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in top_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= topk
    return g, c, cmax


def _fista_sym(a_init, deadline, topk=3, lr_init=None):
    """FISTA with exact subgradient on symmetric simplex."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    t_val = 1.0
    if lr_init is None:
        lr_init = 0.3 / n
    lr = lr_init
    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())
    stagnation = 0
    best_stag_score = best_cmax

    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        # Nesterov momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        sy = float(y.sum())
        if sy > 1e-30:
            y = y / sy
        else:
            y = _make_sym_init(n, "uniform")

        # Subgradient
        g, c, cmax = _subgrad_topk(y, topk)

        # Gradient step
        a_new = _proj_sym_simplex(y - lr * g)

        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        cmax_new = float(c_new.max())

        if cmax_new < best_cmax - 1e-12:
            best_cmax = cmax_new
            best_a = a_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        # Adaptive step size
        if cmax_new > cmax * 1.02:
            lr *= 0.7
        elif stagnation > 2000:
            lr *= 0.95
            if stagnation > 10000:
                lr = min(lr * 1.1, 0.5)

        a_prev = a.copy()
        a = a_new
        t_val = t_next

        if it % 5000 == 0:
            pass  # just let it run

    return best_a, best_cmax


def _coord_descent_sym(a_init, deadline, n_sweeps=50):
    """Coordinate descent on symmetric sequence."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())
    half = n // 2

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        # Iterate over first half coordinates
        for i in range(half):
            if time.monotonic() > deadline:
                break
            # Save current
            ai_old = a[i]
            if ai_old < 1e-15:
                # Try setting to small value
                candidates = [ai_old, 0.01 / n, 0.1 / n]
            else:
                # Try a range of values around current
                spread = max(ai_old * 0.3, 1.0 / (n * 10))
                candidates = [ai_old, ai_old + spread, ai_old - spread,
                              ai_old * 1.5, ai_old * 0.5, 0.0]
                candidates = [max(0.0, c) for c in candidates]

            best_val = ai_old
            best_score_local = float(_full_convolve(a, a).max())

            for cv in candidates:
                if cv == ai_old:
                    continue
                # Update a_i and renormalize
                a[i] = cv
                # By symmetry, a[n-1-i] = a[i]
                a[n - 1 - i] = cv
                # Renormalize (adjust other elements proportionally)
                s = float(a.sum())
                if s < 1e-15:
                    a[i] = ai_old
                    a[n - 1 - i] = ai_old
                    continue
                a = a / s
                c = _full_convolve(a, a)
                cn = float(c.max())
                if cn < best_score_local:
                    best_score_local = cn
                    best_val = cv

            # Set back to best
            a[i] = best_val
            a[n - 1 - i] = best_val
            s = float(a.sum())
            if s > 1e-15:
                a = a / s

        # Check global improvement
        c = _full_convolve(a, a)
        cmax = float(c.max())
        if cmax < best_cmax - 1e-10:
            best_cmax = cmax
            best_a = a.copy()

    return best_a, best_cmax


def _adam_sym(a_init, deadline, beta=1e-6, lr=0.003, topk=5):
    """Adam optimization in log-space for symmetric sequences."""
    n = len(a_init)
    a = a_init.copy()
    a = _symmetrize(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    z = np.log(a)
    z = z - z.mean()
    # Enforce symmetry in z-space
    z = _symmetrize(z)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())

    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        t += 1
        # Compute a from z
        a = _softmax_compute(z)
        a = _symmetrize(a)
        a = a / a.sum()

        # Compute subgradient
        g, c, cmax = _subgrad_topk(a, topk)

        # Smooth gradient with exponential weighting
        c_sh = (c - cmax) / (beta * max(cmax, 1e-10))
        e = np.exp(c_sh)
        s_e = float(e.sum())
        if np.isfinite(s_e) and s_e > 1e-30:
            w = e / s_e
        else:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0

        # Weighted gradient
        g_w = np.zeros(n)
        top_idx = np.argsort(c)[::-1][:topk * 2]
        for k in top_idx:
            wk = w[k]
            if wk < 1e-6:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g_w[lo:hi + 1] += 2.0 * wk * a[k - hi:k - lo + 1][::-1]

        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * g_w
        v_buf = b2 * v_buf + (1 - b2) * (g_w * g_w)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        # Clip update
        update_norm = np.linalg.norm(update)
        max_norm = 0.2
        if update_norm > max_norm:
            update = update * (max_norm / update_norm)

        z = z - update
        z = _symmetrize(z)
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            a_chk = _symmetrize(a_chk)
            a_chk = a_chk / a_chk.sum()
            sc = float(_full_convolve(a_chk, a_chk).max())
            if sc < best_cmax:
                best_cmax = sc
                best_a = a_chk.copy()

        if t % 5000 == 0:
            lr = max(0.0002, lr * 0.9)

    a_final = _softmax_compute(z)
    a_final = _symmetrize(a_final)
    a_final = a_final / a_final.sum()
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_cmax:
        best_cmax = sc_final
        best_a = a_final.copy()

    return best_a, best_cmax


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
