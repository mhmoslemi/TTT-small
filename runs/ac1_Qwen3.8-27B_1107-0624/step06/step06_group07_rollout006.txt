"""
LP-driven optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions,
combined with L-BFGS refinement, subgradient polishing, and multi-restart.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import minimize
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def _subgradient_of_max_conv(a, active_set):
    """
    Compute the subgradient of max_k (a*a)[k] at a,
    given the active set (indices where (a*a)[k] is near max).
    Returns gradient vector of length n.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    if len(active_set) == 0:
        active_set = np.array([np.argmax(c)])
    
    w = np.zeros(2 * n - 1)
    w[active_set] = 1.0 / len(active_set)
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    g = 2.0 * wc[n - 1:2 * n - 1]
    return g, c_max


def _solve_lp_direction(a, M, solver='ECOS'):
    """
    Solve the LP: maximize sum(b) subject to a*b <= M, b >= 0.
    Returns the optimal b (normalized to sum 1) or None if infeasible.
    """
    n = len(a)
    # Build the convolution matrix A where (A @ b)[k] = (a*b)[k]
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # A is (2n-1) x n
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        # (a*b)[k] = sum_j a[j] * b[k-j] where 0 <= k-j < n
        # So j ranges from max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    if not rows:
        return None
    
    A_mat = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    b_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [A_mat @ b_var <= M, b_var >= 0]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return None
    
    if b_var.value is None:
        return None
    
    b_val = np.asarray(b_var.value).ravel()
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smoothed(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):

def _lp_iteration(a, t=0.5, solver='ECOS'):
    """
    One LP iteration: solve for best b, then mix.
    """
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    b = _solve_lp_direction(a, M, solver=solver)
    if b is None:
        return a
    
    # Mix
    a_new = (1 - t) * a + t * b
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new = a_new / s
    return a_new


def _perturb(a, rng, method='gauss', sigma=0.03):
    n = len(a)
    if method == 'gauss':
        z = np.log(np.maximum(a, 1e-15))
        z += sigma * rng.standard_normal(n)
        z -= z.mean()
        return _softmax(z)
    elif method == 'fourier':
        z = np.log(np.maximum(a, 1e-15))
        t = np.arange(n)
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 16)))
            amp = 0.01 * rng.uniform(0.5, 3.0)
            phase = rng.uniform(0, 2.0 * np.pi)
            z += amp * np.cos(2 * np.pi * freq * t / n + phase)
        z -= z.mean()
        return _softmax(z)
    elif method == 'block':
        z = np.log(np.maximum(a, 1e-15))
        n_blocks = 6
        block_size = max(4, n // n_blocks)
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
            z[start:end] += rng.uniform(-0.02, 0.02)
        z -= z.mean()
        return _softmax(z)
    elif method == 'swap':
        # Random swap perturbation
        a_new = a.copy()
        n_swaps = max(1, n // 100)
        for _ in range(n_swaps):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = a_new[i] * 0.1 * rng.uniform(-1, 1)
                a_new[i] -= delta
                a_new[j] += delta
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        return a_new
    else:
        return a


def _targeted_coordinate_descent(a, deadline, max_rounds=10000):
    """
    Targeted coordinate descent: find the peak, reduce contributing elements.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        c_max = c[k_max]
        
        # Find the pair (i, j) with i+j = k_max that contributes most
        # (a*a)[k_max] = sum_i a[i] * a[k_max - i]
        # The dominant term is where a[i] * a[k_max-i] is largest
        i_range = range(max(0, k_max - n + 1), min(n, k_max + 1))
        if len(i_range) == 0:
            break
        
        contributions = np.array([a[i] * a[k_max - i] for i in i_range])
        i_best = i_range[int(np.argmax(contributions))]
        j_best = k_max - i_best
        
        # Try to reduce the contribution
        # Reduce a[i_best] and increase a[j_best] (or vice versa)
        # Or reduce both
        for frac in [0.01, 0.02, 0.05]:
            for direction in [0, 1, 2]:
                a_try = a.copy()
                if direction == 0:
                    # Reduce i_best, shift to random other
                    delta = a_try[i_best] * frac
                    a_try[i_best] -= delta
                    j_other = int((i_best + n // 3) % n)
                    a_try[j_other] += delta
                elif direction == 1:
                    # Reduce j_best
                    delta = a_try[j_best] * frac
                    a_try[j_best] -= delta
                    j_other = int((j_best + n // 3) % n)
                    a_try[j_other] += delta
                else:
                    # Reduce both
                    delta_i = a_try[i_best] * frac
                    delta_j = a_try[j_best] * frac
                    a_try[i_best] -= delta_i
                    a_try[j_best] -= delta_j
                    j_other = int((i_best + n // 2) % n)
                    a_try[j_other] += delta_i + delta_j
                
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try /= s
                
                sc = _score(a_try)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    break
        
    return best_a, best_score

LP-driven optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses scipy HiGHS LP solver for fast direction finding, combined with
L-BFGS refinement, subgradient polishing, and multi-restart diversification.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
