import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _softmax(z, beta=1.0):
    z = z / beta
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    Uses the formula: d(conv[k])/da[m] = 2*a[k-m] (if 0<=k-m<n)
    Computed as: 2 * (w_rev * a)[n-1 : 2n-1]

    # We want elements [n-1, 2n-2]
    wc = _fft_convolve(w_rev, a)
    h = wc[n-1 : 2*n-1]
    return 2.0 * h

                    symm_every=500, beta_anneal_factor=0.5):

    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    v_buf = np.zeros(n)

"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses Adam log-space with continuation, LP-based direction finding (KKT exploitation),
symmetrization, and multi-scale exploration (n=4096, 8192, 16384).
Key insight: the gradient of conv[k] w.r.t. a[m] is 2*a[k-m], enabling efficient computation.
LP polishing exploits the KKT structure: at optimum, weighted sum of shifted a's is constant.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_conv(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_probs(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_conv_given_w(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses: d(conv[k])/da[m] = 2*a[k-m] if 0<=k-m<n."""
    n = len(a)
    w_rev = w[::-1]
    wc = _fft_conv(w_rev, a)
    h = wc[n - 1: 2 * n - 1]
    return 2.0 * h


def _grad_logspace(a, w):
    """Gradient in log-space: df/dz[m] = a[m]*(grad_a[m] - a.grad_a)."""
    grad_a = _grad_conv_given_w(a, w)
    W = float(np.dot(a, grad_a))
    return a * (grad_a - W)


def _adam_optimize(a_init, deadline, beta_list, lr_init=0.01, max_iter=200000,
                   lr_decay_every=5000, lr_decay_factor=0.98, clip_norm=1.0,
                   symm_every=0, check_interval=20):
    """Adam in log-space with beta continuation."""
    n = len(a_init)
    a0 = a_init.copy()
    s = a0.sum()
    a0 = a0 / s if s > 0 else np.ones(n) / n

    z = np.log(np.maximum(a0, 1e-15))
    z -= z.mean()

    best_a = a0.copy()
    best_score = _score_norm(a0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr_init
    no_improve = 0

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1

            a = np.exp(z - np.max(z))
            a /= a.sum()

            c = _fft_conv(a, a)
            cm = float(c.max())

            c_shift = (c - cm) / beta
            e = np.exp(c_shift)
            sw = float(e.sum())
            if sw < 1e-300 or not np.isfinite(sw):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / sw

            gz = _grad_logspace(a, w)
            if not np.all(np.isfinite(gz)):
                break

            m_buf = b1 * m_buf + (1 - b1) * gz
            v_buf = b2 * v_buf + (1 - b2) * gz * gz
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = cur_lr * mh / (np.sqrt(vh) + eps)
            unorm = np.linalg.norm(update)
            if unorm > clip_norm:
                update *= clip_norm / unorm

            z -= update
            z = np.clip(z, -60.0, 60.0)
            z -= z.mean()

            if symm_every > 0 and t % symm_every == 0:

                ac /= ac.sum()
                ac = 0.5 * (ac + ac[::-1])
                ac = np.maximum(ac, 1e-15)
                ac /= ac.sum()
                z = np.log(ac)
                z -= z.mean()

            if t % check_interval == 0:
                ac = np.exp(z - np.max(z))
                ac /= ac.sum()
                sc = _score_norm(ac)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = ac.copy()
                    no_improve = 0
                else:
                    no_improve += 1

    af = np.exp(z - np.max(z))
    af /= af.sum()
    scf = _score_norm(af)
    if np.isfinite(scf) and scf < best_score:
        best_score = scf
        best_a = af.copy()

    return best_a, best_score


def _lp_direction(a, top_k=200):
    """Solve LP: max sum(b) s.t. conv(a,b)_k <= M for top-k largest conv values, b>=0."""
    n = len(a)
    c = _fft_conv(a, a)
    M = float(c.max())

    # Use only top-k constraints for speed
    k = min(top_k, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    threshold = c[top_idx[-1]]

    active = top_idx  # constraint indices

    # Build sparse constraint matrix A: A[local_k, j] = a[active[local_k] - j] if valid
    rows = []
    cols = []
    vals = []
    for li, ki in enumerate(active):
        j_start = max(0, ki - n + 1)
        j_end = min(ki + 1, n)
        for j in range(j_start, j_end):
            rows.append(li)
            cols.append(j)
            vals.append(a[ki - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(active), n))
    c_obj = -np.ones(n)
    b_ub = np.array([c[ki] for ki in active])

    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  method='highs', bounds=[(0, None)] * n)

    if res.status == 0 and res.x is not None:
        return res.x, M
    return a.copy(), M


def _lp_polish(a_init, deadline, n_iters=50, top_k=200):
    """Iterative LP-based polishing."""
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        b, M = _lp_direction(a, top_k=top_k)
        if b is None or not np.all(np.isfinite(b)):
            break

        sb = float(b.sum())
        if sb < 1e-15:
            break

        b = b / sb  # normalize to sum=1

        # Check if b is meaningfully different from a
        diff = np.linalg.norm(b - a)
        if diff < 1e-10:
            if it > 3:
                break
            continue

        # Mix: a_new = (1-t)*a + t*b
        t = 0.3
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            # Print progress every 10 iterations
            if it % 10 == 0:
                print(f"    LP iter {it}: score={sc:.10f}")
        else:
            # Try smaller step
            t = 0.1
            a_new = (1 - t) * a + t * b
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
            sc2 = _score_norm(a_new)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new.copy()
                a = a_new
            else:

                a_new = (1 - t) * a + t * b
                a_new = np.maximum(a_new, 0)

                    best_a = a_new.copy()
                    a = a_new

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_beta_schedule(max_c, start_frac=0.01, end_frac=0.00001, n_betas=10):
    """Geometric beta schedule from large (smooth) to small (sharp)."""
    b_start = start_frac * max_c
    b_end = end_frac * max_c
    if b_start <= b_end:
        return [b_start]
    log_space = np.geomspace(b_start, b_end, n_betas)
    return list(log_space)


def _adam_with_schedule(a_init, deadline, max_c, n_betas=12, lr_init=0.008,
                        max_iter=300000, symm_every=0):
    """Adam with automatic beta schedule."""
    betas = _make_beta_schedule(max_c, 0.005, 0.000005, n_betas)
    return _adam_optimize(a_init, deadline, betas, lr_init=lr_init,
                          max_iter=max_iter, symm_every=symm_every)

        sc_lp_full = _score(a_lp_opt)

        # If we downsized, interpolate back
        if len(a_lp_opt) != len(best_seq):
            a_lp_back = _interp_profile(a_lp_opt, len(best_seq))
            sc_lp_back = _score(a_lp_back)
            if sc_lp_back < best_score:
                best_score = sc_lp_back
                best_seq = a_lp_back.copy()
        elif sc_lp_full < best_score:
            best_score = sc_lp_full
            best_seq = a_lp_opt.copy()

        print(f"  LP polish: score={_score(a_lp_opt):.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 4: Final Adam polish with very small beta ===
    print(f"\n--- Phase 4: Final polish (deadline: {deadline - t_start:.0f}s remaining) ---")
    p4_end = min(deadline, t_start + 1700.0)

    a_work = best_seq.copy()
    a_work /= a_work.sum()
    c_work = _fft_conv(a_work, a_work)
    max_cw = float(c_work.max())

    if time.monotonic() < p4_end - 15:
        rem = p4_end - time.monotonic() - 10
        betas4 = [0.00001 * max_cw, 0.000005 * max_cw, 0.000002 * max_cw]
        a_pol, sc_pol = _adam_optimize(a_work, time.monotonic() + rem, betas4,
                                       lr_init=0.003, max_iter=200000,
                                       lr_decay_every=10000, lr_decay_factor=0.96,
                                       symm_every=0)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Final polish: score={sc_pol_full:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Try symmetrization + Adam on best ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = np.maximum(a_sym, 1e-15)
        a_sym /= a_sym.sum()
        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  Symmetrization: score={sc_sym:.10f}")

        # Quick Adam on symmetrized
        if time.monotonic() < deadline - 30:
            c_sym = _fft_conv(a_sym, a_sym)
            max_cs = float(c_sym.max())
            rem = deadline - time.monotonic() - 20
            if rem > 20:
                a_sym2, _ = _adam_optimize(a_sym, time.monotonic() + rem,
                                           [0.00001 * max_cs, 0.000005 * max_cs],
                                           lr_init=0.002, max_iter=100000)
                sc_sym2 = _score(a_sym2)
                if sc_sym2 < best_score:
                    best_score = sc_sym2
                    best_seq = a_sym2.copy()
                    print(f"  Sym+Adam: score={sc_sym2:.10f}")

    # === Final validation ===
    print(f"\nAll phases done. Best score: {best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            s = final.sum()
            final /= s

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(512) / 512
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
