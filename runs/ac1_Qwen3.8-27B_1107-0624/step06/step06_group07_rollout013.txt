def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()

        w = np.zeros_like(c)

    else:
        w = e / s
    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_gradient(a, c, n, tol_frac=1e-4):

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-15, gtol=1e-12):

        return _smoobj(z, beta, n)

        return _smograd(z, beta, n)

                                   'maxls': 40, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)

def _lp_direction(a, M, deadline, max_lp_iters=50, tol_frac=1e-3):
    """LP-based direction finding using CVXPY.
    Solve: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0
    """
    if not HAS_CVX:
        return None

    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    M = min(M, c_max * (1 + tol_frac))

    b_var = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix: (a*b)(k) = sum_j a_j * b_{k-j}
    # For each k, the coefficients of b_i in (a*b)(k) are:
    # (a*b)(k) = sum_{i} a[k-i] * b_i (where a is zero outside [0,n-1])
    # So coeff[k][i] = a[k-i] if 0 <= k-i < n else 0
    
    # This is a Toeplitz-like matrix. Let's build it.
    # Constraint: A @ b <= M * ones(2n-1)
    # where A[k][i] = a[k-i] if 0 <= k-i < n else 0
    
    # For efficiency, we only include "active" constraints (near the max)
    active_mask = c >= (c_max - tol_frac * c_max)
    
    # Build the constraint matrix for active constraints only
    active_indices = np.where(active_mask)[0]
    if len(active_indices) == 0:
        active_indices = np.array([int(np.argmax(c))])
    
    A_active = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        # A_active[idx][i] = a[k-i] if 0 <= k-i < n
        i_vals = np.arange(n)
        valid = (k - i_vals >= 0) & (k - i_vals < n)
        A_active[idx, valid] = a[k - i_vals[valid]]
    
    constraints = [A_active @ b_var <= M * np.ones(len(active_indices))]
    
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=10000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000)
        except Exception:
            return None
    
    if b_var.value is None or prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_sol = np.maximum(b_var.value, 0.0)
    s = float(b_sol.sum())
    if s < 1e-10:
        return None
    
    return b_sol / s


def _lp_polish(a_init, deadline, max_rounds=80):

    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(current_a, current_a)
        c_max = float(c.max())
        
        b_dir = _lp_direction(current_a, c_max, deadline)
        if b_dir is None:
            break
        
        # Try several step sizes
        best_t = 0
        best_sc_trial = current_sc
        for t in [0.1, 0.05, 0.02, 0.01, 0.2, 0.3]:
            a_try = (1 - t) * current_a + t * b_dir
            a_try = np.maximum(a_try, 0.0)
            s = float(a_try.sum())
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc_trial:
                best_sc_trial = sc_try
                best_t = t
        
        if best_t > 0:
            current_a = (1 - best_t) * current_a + best_t * b_dir
            current_a = np.maximum(current_a, 0.0)
            current_a = current_a / current_a.sum()
            current_sc = best_sc_trial
        else:
            break
    
    return current_a, current_sc

    """Subgradient method in z-space with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.02
    stall = 0
    best_stall_lr = lr
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break
        
        step = lr / (1.0 + i * 1e-5)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()
                stall = 0
                if lr < 0.05:
                    lr *= 1.1
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.75
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.9
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final
    
    return best_a, best_sc


def _perturb(a, rng, method="gauss", strength=0.03):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    if method == "gauss":
        z += strength * rng.standard_normal(n)
    elif method == "fourier":
        t = np.arange(n)
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 10)))
            amp = strength * 0.5 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif method == "block":
        n_blocks = int(rng.integers(3, 10))
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - 4)))
            length = int(rng.integers(4, max(5, n // 10)))
            end = min(n, start + length)
            z[start:end] += rng.uniform(-strength, strength)
    z = z - z.mean()
    return _softmax(z)


def _interpolate(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS, LP-based direction finding,
    coordinate descent, and multi-scale search to minimize
    2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score_val = np.inf
    
    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"t={elapsed:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(2048) / 2048
        try_update(a0, "uniform_2048")
    
    # === Phase 1: L-BFGS with aggressive beta continuation ===
    p1_deadline = deadline - 0.55 * (deadline - t_start)
    print(f"\n=== Phase 1: L-BFGS (until t+{p1_deadline - t_start:.0f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, p1_deadline, min_time_remaining=300.0)
    try_update(a_p1, "phase1_lbfgs")
    
    # === Phase 2: LP-based direction finding ===
    p2_deadline = deadline - 0.35 * (deadline - t_start)
    print(f"\n=== Phase 2: LP Direction ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < p2_deadline:
        a_lp, sc_lp = _lp_polish(best_seq, p2_deadline, max_rounds=60)
        try_update(a_lp, "phase2_lp")
    
    # === Phase 3: Subgradient polish ===
    p3_deadline = deadline - 0.22 * (deadline - t_start)
    print(f"\n=== Phase 3: Subgradient ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < p3_deadline:
        a_sg, sc_sg = _subgrad_polish(best_seq, p3_deadline,
                                       tol_frac=1e-5, max_steps=30000)
        try_update(a_sg, "phase3_sg")
    
    # === Phase 4: Coordinate descent ===
    p4_deadline = deadline - 0.12 * (deadline - t_start)
    print(f"\n=== Phase 4: Coord Descent ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < p4_deadline:
        a_cd, sc_cd = _coord_descent_polish(best_seq, p4_deadline, max_sweeps=50)
        try_update(a_cd, "phase4_cd")
    
    # === Phase 5: Fine L-BFGS ===
    p5_deadline = deadline - 0.06 * (deadline - t_start)
    print(f"\n=== Phase 5: Fine L-BFGS ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < p5_deadline:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > p5_deadline:
                break
            remaining = p5_deadline - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(10000, max(200, int(remaining / 0.001)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine5_{i}")
    
    # === Phase 6: Try larger n ===
    p6_deadline = deadline - 0.03 * (deadline - t_start)
    print(f"\n=== Phase 6: Larger n ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < p6_deadline:
        for target_n in [8192]:
            if time.monotonic() > p6_deadline:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate(best_seq, target_n)
            try_update(a_interp, f"interp_{target_n}")
            if time.monotonic() < p6_deadline - 30:
                a_big = _lbfgs_schedule(a_interp, p6_deadline,
                                         min_time_remaining=20.0)
                try_update(a_big, f"lbfgs_{target_n}")
    
    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 30:
        restart_configs = [
            ("g002", "gauss", 0.02),
            ("g005", "gauss", 0.05),
            ("g010", "gauss", 0.10),
            ("f3", "fourier", 0.03),
            ("f5", "fourier", 0.05),
            ("b4", "block", 0.02),
            ("b8", "block", 0.03),
            ("g015", "gauss", 0.15),
            ("g020", "gauss", 0.20),
        ]
        for name, ptype, strength in restart_configs:
            if time.monotonic() > deadline - 25:
                break
            p_start = _perturb(best_seq, rng, ptype, strength)
            sc_start = _score(p_start)
            if sc_start < best_score_val + 0.02:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=20.0)
                try_update(a_r, f"restart_{name}")
    
    # === Phase 8: Final ultra-fine polish ===
    print(f"\n=== Phase 8: Final Polish ({time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            
            n = len(best_seq)
            c_b = _full_convolve(best_seq, c_b if False else _full_convolve(best_seq, best_seq), best_seq) if False else _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            
            # Ultra-fine L-BFGS
            beta = mb * 1e-7
            a_f, _, _ = _lbfgs_run(best_seq, beta, n, max_iter=5000)
            try_update(a_f, f"final_fine_{rnd}")
            
            # Subgradient with very tight tolerance
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_polish(best_seq, deadline - 5,
                                          tol_frac=1e-6, max_steps=10000)
                try_update(a_s, f"final_sg_{rnd}")
            
            # LP if available
            if HAS_CVX and time.monotonic() < deadline - 5:
                a_l, _ = _lp_polish(best_seq, deadline - 5, max_rounds=20)
                try_update(a_l, f"final_lp_{rnd}")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
