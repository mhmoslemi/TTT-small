"""
Multi-stage convex optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Exploits the convexity of the problem (max of convex quadratics over the simplex).
Uses adaptive log-sum-exp smoothing with L-BFGS, Nesterov-accelerated gradient,
peak-targeting active-set subgradient, iterative support reduction, and multi-n search.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_value_and_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, w
    val = c_max + beta * np.log(s)
    w = e / s
    return val, w


def _grad_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    val, w = _lse_value_and_weights(c, beta)
    a_flipped = a[::-1]
    wc = _full_convolve(w, a_flipped)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * (h - a * W)
    grad_a_mean = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - grad_a_mean)
    return grad_z


def _obj_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    val, w = _lse_value_and_weights(c, beta)
    return 2.0 * n * val


def _active_set_grad(a, c, n, tol_frac=0.001):
    """Compute subgradient targeting the active (near-max) set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros(len(a))
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_flipped = a[::-1]
    wc = _full_convolve(w, a_flipped)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * (h - a * W)
    return grad_a


def _nesterov_gradient_desc(a_init, deadline, n_steps=2000, beta=0.01, lr_init=0.01):
    """Nesterov-accelerated gradient descent on the smoothed objective (in z-space)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()
    t_k = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)
    mom = 0.0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad_y = _grad_lse(y, beta, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        z_new = y - lr_init * grad_y
        y = z_new + alpha * (z_new - z)
        z = z_new
        t_k = t_new

        if i % 20 == 0:
            a_chk = _softmax(z)
            sc = _score(a_chk)
            if sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _subgrad_momentum(a_init, deadline, n_steps=3000, lr_init=0.015, tol_frac=0.002):
    """Subgradient method with momentum targeting the active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    mom = np.zeros(n)
    beta_mom = 0.9
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_set_grad(a, c, n, tol_frac=tol_frac)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        lr_current = lr / (1.0 + 0.001 * i)
        mom = beta_mom * mom - lr_current * grad
        z = z + mom
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_chk = _softmax(z)
            sc = _score(a_chk)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_chk.copy()
                patience = 0
            else:
                patience += 1
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    mom *= 0.5
                    patience = 0
                    lr *= 0.8

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _support_reduction(a_init, deadline):
    """Iteratively zero out small elements and re-optimize."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    current_a = a_init.copy()

    fractions = [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = current_a.copy()
        idx = np.argsort(a_try)
        a_try[idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            current_a = a_try.copy()

    return best_a, best_sc


def _run_lbfgs_staged(a_init, deadline, min_time_remaining=60.0):
    """Multi-stage L-BFGS with adaptive beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[20]
    else:
        gap = c_sorted[0] * 0.1
    base_beta = max(gap * 0.5, max_c * 0.005)

    # More stages, slower decay for better convergence
    ratios = [1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002,
              0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    betas = base_beta * np.array(ratios)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Estimate iterations based on time
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining * 0.7 / (est_time_per_iter * 15))))

        try:
            result = minimize(
                lambda z_local: _obj_lse(z_local, beta, n),
                z,
                jac=lambda z_local: _grad_lse(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _run_equilibrium_iteration(a_init, deadline, max_iter=100):
    """
    Alternating minimization:
    1. Given a, compute minimax weights q on the lags
    2. Given q, solve min a^T H q a s.t. sum(a)=1, a>=0 via bisection on lambda
    This exploits the convexity and KKT structure.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Find minimax q: the q that minimizes max_k (a^T T_k a) / q_k... 
        # Actually, the dual: max_q min_a max_k ... 
        # Let's use: q proportional to c (this is the "natural" choice)
        # Or: q uniform over active set
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        q = np.zeros_like(c)
        q[active] = 1.0 / active.sum()

        # Compute H*v where H = sum_k q[k] * T_k
        # (H*v)_i = sum_k q[k] * sum_j T_k(i,j) * v_j = sum_k q[k] * v[k-i] (valid range)
        # = sum_k q[k] * v[k-i] = (q * v_flipped)[n-1+i]... same computation as before
        # Actually (H*v)_i = 2 * sum_k q[k] * v[k-i] (factor of 2 from symmetry)
        # Wait: T_k(i,j) = 1 if i+j=k, so (T_k v)_i = v[k-i] (if valid)
        # (H v)_i = sum_k q[k] * v[k-i]
        # This is the same cross-correlation computation.

        # For the QP: min a^T H a s.t. sum(a)=1, a>=0
        # KKT: 2*H*a + lambda*1 - mu = 0, mu_i >= 0, a_i >= 0, mu_i * a_i = 0
        # For a_i > 0: 2*(H*a)_i + lambda = 0, so (H*a)_i = -lambda/2 (constant)
        # This means on the support, H*a is constant.
        # We can solve this by: given lambda, a_i = 0 if (H*a)_i + lambda/2 < 0
        # But this is circular. Use iterative method.

        # Simpler: use projected gradient on the QP
        # min a^T H a s.t. sum(a)=1, a>=0
        # grad = 2*H*a, project onto simplex

        def hv(v):
            v_flipped = v[::-1]
            qc = _full_convolve(q, v_flipped)
            return qc[n - 1:2 * n - 1]

        lr_q = 1.0 / (2.0 * float(np.max(q)) * 2.0 * n)  # rough Lipschitz estimate
        # Actually, the Lipschitz constant of the QP gradient is 2*lambda_max(H)
        # lambda_max(H) <= 2*sum(q)*... it's at most 2 (since H is a sum of weighted permutation matrices)
        # More precisely, ||H|| <= 2 * sum_k q[k] * ||T_k|| = 2 (since each T_k has spectral norm 1)
        # So Lipschitz constant of 2*H*a is 4. Use lr = 1/4.
        lr_q = 0.25

        a_qp = a.copy()
        for qp_iter in range(50):
            Ha = hv(a_qp)
            grad_qp = 2.0 * Ha
            a_qp = a_qp - lr_q * grad_qp
            # Project onto simplex: sum(a)=1, a>=0
            # Use the standard projection algorithm
            a_qp = _project_simplex(a_qp)
            a_qp = np.maximum(a_qp, 0)
            a_qp = a_qp / a_qp.sum()

        sc_new = _score(a_qp)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_qp.copy()
            a = a_qp.copy()
        else:
            # Dampen
            a = 0.5 * a + 0.5 * a_qp
            a = a / a.sum()

    return best_a, best_sc


def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_center":
        t = np.linspace(-1, 1, n)
        a = np.exp(-2.0 * t * t)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(-1, 1, n)
        a = np.exp(-8.0 * (t - 0.4) ** 2) + np.exp(-8.0 * (t + 0.4) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        z = z + amp * 0.5 * np.sin(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _resample_sequence(a, new_n):
    """Resample a sequence to a new length using interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search exploiting convexity:
    1. L-BFGS with staged beta reduction
    2. Nesterov-accelerated gradient
    3. Active-set subgradient with momentum
    4. Equilibrium iteration (dual approach)
    5. Support reduction
    6. Multi-n exploration
    7. Diverse restarts
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Main L-BFGS from parent ---
    print(f"\n[Phase 1] Main L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs_staged(parent, deadline, min_time_remaining=800.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # --- Phase 2: Nesterov-accelerated gradient ---
    print(f"\n[Phase 2] Nesterov gradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_nes = mb * 0.01
        a_nes, sc_nes = _nesterov_gradient_desc(
            best_seq, min(deadline - 600, time.monotonic() + 120),
            n_steps=3000, beta=beta_nes, lr_init=0.02
        )
        update_best(a_nes, sc_nes, "nesterov")

    # --- Phase 3: Subgradient with momentum ---
    print(f"\n[Phase 3] Subgradient momentum (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_sg, sc_sg = _subgrad_momentum(
            best_seq, min(deadline - 500, time.monotonic() + 150),
            n_steps=5000, lr_init=0.012, tol_frac=0.003
        )
        update_best(a_sg, sc_sg, "sg_momentum")

    # --- Phase 4: Equilibrium iteration ---
    print(f"\n[Phase 4] Equilibrium iteration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_eq, sc_eq = _run_equilibrium_iteration(
            best_seq, min(deadline - 400, time.monotonic() + 120), max_iter=80
        )
        update_best(a_eq, sc_eq, "equilibrium")

    # --- Phase 5: Support reduction ---
    print(f"\n[Phase 5] Support reduction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_shr, sc_shr = _support_reduction(best_seq, deadline - 300)
        update_best(a_shr, sc_shr, "shrinkage")
        # Re-optimize after shrinkage
        if sc_shr <= best_score + 1e-10 and time.monotonic() < deadline - 200:
            a_shr2 = _run_lbfgs_staged(a_shr, deadline, min_time_remaining=200.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # --- Phase 6: Multi-n search ---
    print(f"\n[Phase 6] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if n_try == len(best_seq) if best_seq is not None else False:
            continue

        if parent is not None:
            a_init = _resample_sequence(parent, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        sc_init = _score(a_init)
        a_opt = _run_lbfgs_staged(a_init, deadline, min_time_remaining=80.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick subgradient polish
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _subgrad_momentum(
                a_opt, min(deadline - 30, time.monotonic() + 60),
                n_steps=1500, lr_init=0.01, tol_frac=0.002
            )
            update_best(a_opt2, sc_opt2, f"n_{n_try}_sg")

    # --- Phase 7: Diverse restarts ---
    print(f"\n[Phase 7] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for i in range(4):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=6)
            elif i == 2:
                p_start = _perturb(best_seq, 0.1, rng)
            else:
                # Symmetric perturbation
                n = len(best_seq)
                half = n // 2
                p_try = best_seq.copy()
                noise = rng.standard_normal(half) * 0.02
                p_try[:half] += noise
                p_try[half:] = p_try[:half][::-1]
                if n % 2 == 1:
                    p_try[half + 1:] = p_try[half - 1::-1]
                p_try = np.maximum(p_try, 0)
                p_try = p_try / p_try.sum()
                p_start = p_try

            sc_start = _score(p_start)
            a_r = _run_lbfgs_staged(p_start, deadline, min_time_remaining=20.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

    # --- Phase 8: Final fine polish ---
    print(f"\n[Phase 8] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        remain = deadline - 5 - time.monotonic()
        if remain > 10:
            # Very fine L-BFGS
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = max(mb * 0.00005, 1e-10)
            max_iter = min(5000, max(100, int(remain * 0.5 / 0.005)))
            try:
                result = minimize(
                    lambda z_local: _obj_lse(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _grad_lse(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient with tight tolerance
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_momentum(
                best_seq, deadline - 2, n_steps=1000, lr_init=0.005, tol_frac=0.001
            )
            update_best(a_fin2, sc_fin2, "final_sg")

        # Final support reduction
        if time.monotonic() < deadline - 1:
            a_fin3, sc_fin3 = _support_reduction(best_seq, deadline - 1)
            update_best(a_fin3, sc_fin3, "final_shrink")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
