    n = len(a)
    N = _next_pow2(2*n - 1)
    a = a.copy()
    a = a / a.sum()
    
    for it in range(n_iter):
        # Fourier domain: flatten |A|^2
        fa = np.fft.rfft(a, N)
        power = np.abs(fa)**2
        
        # Flatten: reduce dynamic range
        # Target: make power more uniform
        # Use: power_new = power^alpha where alpha < 1 compresses dynamic range
        alpha = 0.8  # compression factor
        power_new = power ** alpha
        
        # Reconstruct with original phase
        phase = np.angle(fa)
        fa_new = np.sqrt(power_new) * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(fa_new, N)[:n]
        
        # Enforce non-negativity and normalization
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new
    
    return a

    out_n = n + m - 1

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_uniform(n):
    return np.ones(n) / n


def _make_tapered(n):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quad_residue(n):
    """Create a sequence based on quadratic residues for good autocorrelation."""
    # Find a prime p >= n
    p = n
    while True:
        if _is_prime(p):
            break
        p += 1
    # Legendre symbol sequence
    a = np.ones(n)
    for i in range(1, n):
        if pow(i, p - 1, p) == p - 1:
            a[i] = 0.5  # non-residue gets lower weight
    return a / a.sum()


def _is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True


def _gerchberg_saxton(a, n_iter=500, alpha_schedule=None, deadline=None):
    """Fourier-domain Gerchberg-Saxton algorithm to flatten autocorrelation peak."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a = a.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    if alpha_schedule is None:
        alpha_schedule = np.linspace(0.95, 0.3, n_iter)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        fa = np.fft.rfft(a, N)
        power = np.abs(fa) ** 2
        
        alpha = alpha_schedule[min(it, len(alpha_schedule) - 1)]
        power_new = power ** alpha
        
        # Reconstruct with original phase
        magnitude_new = np.sqrt(power_new)
        phase = np.angle(fa)
        fa_new = magnitude_new * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(fa_new, N)[:n]
        
        # Enforce constraints
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a


def _lp_iterative(a, max_iter=50, deadline=None, tol=1e-6):
    """LP-based iterative refinement from the state-of-the-art description."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        # Build the convolution matrix A (2n-1 x n)
        # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                data.append(a[k - j])
        
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
        
        # Solve LP: max sum(g) s.t. A*g <= M, g >= 0
        try:
            from scipy.optimize import linprog
            c_obj = -np.ones(n)
            b_ub = M * np.ones(2 * n - 1)
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
            if res.success:
                g = np.maximum(res.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Update
                    t = 0.1 + 0.1 * (1 - it / max(1, max_iter))
                    a_new = (1 - t) * a + t * g
                    a_new = np.maximum(a_new, 0)
                    a_new = a_new / a_new.sum()
                    sc = _score(a_new)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_new.copy()
                    a = a_new
        except Exception as e:
            print(f"  LP failed at iter {it}: {e}")
            break
        
        if it % 10 == 0:
            print(f"  LP iter {it}: score={best_sc:.10f}")
    
    return best_a


def _simulated_annealing(a, deadline, n_iter_max=50000, t_start=1.0, t_end=0.01):
    """Simulated annealing with structured moves."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    current_sc = best_sc
    
    rng = np.random.default_rng(12345)
    
    for it in range(n_iter_max):
        if time.monotonic() > deadline:
            break
        
        # Temperature schedule
        t = t_start * (t_end / t_start) ** (it / max(1, n_iter_max))
        
        # Choose a random move
        move_type = rng.integers(0, 4)
        a_try = a.copy()
        
        if move_type == 0:
            # Local perturbation: add noise to a random subset
            n_perturb = max(1, n // 10)
            idx = rng.integers(0, n, size=n_perturb)
            noise = rng.normal(0, 0.1, size=n_perturb)
            a_try[idx] = np.maximum(a_try[idx] + noise, 0)
        elif move_type == 1:
            # Block swap
            block_size = max(1, n // 8)
            i = rng.integers(0, n - 2 * block_size)
            j = rng.integers(0, n - 2 * block_size)
            if abs(i - j) >= block_size:
                tmp = a_try[i:i+block_size].copy()
                a_try[i:i+block_size] = a_try[j:j+block_size]
                a_try[j:j+block_size] = tmp
        elif move_type == 2:
            # Fourier perturbation
            fa = np.fft.rfft(a_try, _next_pow2(2*n-1))
            freq = rng.integers(0, len(fa))
            amp = 0.05 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2*np.pi)
            fa[freq] += amp * np.exp(1j * phase)
            a_try = np.fft.irfft(fa, _next_pow2(2*n-1))[:n]
            a_try = np.maximum(a_try, 0)
        else:
            # Coordinate swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                tmp = a_try[i]
                a_try[i] = a_try[j]
                a_try[j] = tmp
        
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        
        # Accept/reject
        if sc_try < current_sc:
            a = a_try
            current_sc = sc_try
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                if it % 1000 == 0:
                    print(f"  SA iter {it}: new best={best_sc:.10f}, t={t:.6f}")
        else:
            delta = sc_try - current_sc
            if delta / max(t, 1e-10) < -np.log(rng.random()):
                a = a_try
                current_sc = sc_try
    
    return best_a


def _subgradient_descent(a, deadline, n_max_steps=10000, lr_init=0.02):
    """Subgradient descent on the log-sum-exp smoothed objective."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    patience = 0
    
    # Adaptive beta
    c = _full_convolve(a, a)
    c_max = float(c.max())
    beta = c_max * 0.01
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax(z)
        c = _full_convolve(a_cur, a_cur)
        c_max = float(c.max())
        
        # Log-sum-exp objective
        shifted = (c - c_max) / beta
        e = np.exp(np.clip(shifted, -700, 0))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient
        a_rev = a_cur[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_cur * (h - W)
        
        # Directional derivative check
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-10:
                improved = best_sc - sc
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
                    # Also try reducing beta
                    beta *= 0.5
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a


def _coordinate_descent(a, deadline, max_iter=5000):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        improved = False
        # Cycle through coordinates
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try changing a[i]
            old_val = a[i]
            # The contribution of a[i] to the convolution:
            # b_k involves a[i] * a[k-i] and a[k-i] * a[i] (when i != k-i)
            # or 2*a[i]^2 (when i == k-i, i.e., k = 2i)
            
            # Binary search for optimal a[i] given other elements fixed
            # b_k = a[i]*a[k-i] + a[k-i]*a[i] + (other terms) for k where both i and k-i are valid
            # This is quadratic in a[i]: b_k = 2*a[i]*a[k-i] + const (for k != 2i)
            #                          b_k = a[i]^2 + 2*a[i]*a[k-i] + const (for k = 2i, if k-2i is valid and != 0)
            
            # Actually, let's just do a simple line search
            # Try a[i] = 0 and a[i] = 2*old_val (keeping sum constant by adjusting another element)
            
            # Simpler: just try a few candidate values
            candidates = [0.0, old_val * 0.5, old_val * 1.5, old_val * 0.1, old_val * 2.0]
            
            best_local = old_val
            best_local_sc = best_sc
            
            for cv in candidates:
                a_try = a.copy()
                diff = cv - old_val
                a_try[i] = cv
                # Keep sum constant: distribute difference to a random other element
                if abs(diff) > 1e-15:
                    j = (i + n // 2) % n
                    a_try[j] = max(0, a_try[j] - diff)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    sc_try = _score(a_try)
                    if sc_try < best_local_sc:
                        best_local_sc = sc_try
                        best_local = cv
                        # Update a
                        a = a_try.copy()
                        improved = True
            
            if best_local_sc < best_sc:
                best_sc = best_local_sc
                best_a = a.copy()
        
        if not improved and it > 0:
            break
    
    return best_a


def _multi_scale_optimize(a, deadline):
    """Optimize at multiple scales: coarse to fine."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    # Coarse: downsample by factor 4, optimize, upsample
    for factor in [8, 4, 2]:
        if time.monotonic() > deadline - 60:
            break
        
        n_coarse = n // factor
        if n_coarse < 16:
            continue
        
        # Downsample
        a_coarse = np.zeros(n_coarse)
        for i in range(n_coarse):
            a_coarse[i] = a[i*factor:(i+1)*factor].sum()
        a_coarse = a_coarse / a_coarse.sum()
        
        # Optimize coarse
        a_coarse = _subgradient_descent(a_coarse, deadline - 60, n_max_steps=2000, lr_init=0.05)
        sc_coarse = _score(a_coarse)
        
        # Upsample back
        a_fine = np.zeros(n)
        for i in range(n_coarse):
            a_fine[i*factor:(i+1)*factor] = a_coarse[i] / factor
        a_fine = a_fine / a_fine.sum()
        
        # Blend with current
        t = 0.5
        a_blend = (1-t) * best_a + t * a_fine
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()
        
        sc_blend = _score(a_blend)
        if sc_blend < best_sc:
            best_sc = sc_blend
            best_a = a_blend.copy()
            print(f"  Multi-scale factor={factor}: score={best_sc:.10f}")
    
    return best_a


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    a_sorted_idx = np.argsort(a)
    
    for frac in [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a


def _lbfgs_smooth(a_init, deadline, min_time_remaining=10.0):
    """L-BFGS with adaptive log-sum-exp smoothing."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 10))))
        
        def fun(z_local):
            a = _softmax(z_local)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                return 2.0 * n * c_max
            return 2.0 * n * (c_max + beta * np.log(s))
        
        def jac(z_local):
            a = _softmax(z_local)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            shifted = (c - c_max) / beta
            e = np.exp(np.clip(shifted, -700, 0))
            s = e.sum()
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            return 4.0 * n * a * (h - W)
        
        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining:
    1. Fourier Gerchberg-Saxton for global structure
    2. LP-based iterative refinement (theoretically motivated)
    3. Simulated annealing for diversification
    4. L-BFGS with adaptive smoothing for local optimization
    5. Multi-scale optimization
    6. Shrinkage and coordinate descent for final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 2048
        parent = _make_uniform(n_parent)
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: Fourier Gerchberg-Saxton (novel approach) ===
    print(f"\n[Phase 1] Fourier Gerchberg-Saxton (t={time.monotonic()-t_start:.1f}s)")
    n_gs = n_parent if parent is not None else 4096
    a_gs_init = parent.copy() if parent is not None else _make_uniform(n_gs)
    
    # Try different alpha schedules
    for trial in range(3):
        if time.monotonic() > deadline - 600:

            alpha_sched = np.linspace(0.95, 0.4, 300)
        elif trial == 1:

        else:

        a_gs = _gerchberg_saxton(a_gs_init, n_iter=len(alpha_sched),
                                  alpha_schedule=alpha_sched,
                                  deadline=min(deadline - 500, time.monotonic() + 200))
        update_best(a_gs, _score(a_gs), f"gerchberg_{trial}")
    
    # === Phase 2: L
