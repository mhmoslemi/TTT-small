    A = np.zeros(N, dtype=complex)

    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _active_grad_fft(a, weights):
    """
    Compute h[j] = sum_i a[i] * weights[i+j] for j=0..n-1.
    weights has length 2n-1.
    Uses FFT.
    """
    n = len(a)
    w_rev = weights[::-1]
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w_rev
    W = np.fft.rfft(w_pad)
    conv_res = np.fft.irfft(A * W, N)[:3*n-2]
    idx = 2*n-2 - np.arange(n)
    return conv_res[idx]


def _subgrad_active(a, m=5, tol_frac=0.01):
    """
    Compute subgradient of sum of top-m (or tolerance-based) active set.
    Returns (gradient, active_set_size).
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Determine active set
    tol = max(tol_frac * c_max, 1e-10)
    active_mask = c >= (c_max - tol)
    n_active = int(active_mask.sum())
    
    # Also enforce minimum and maximum active set size
    if n_active < m:
        # Use top m
        top_idx = np.argsort(c)[-m:]
        active_mask = np.zeros(2*n-1, dtype=bool)
        active_mask[top_idx] = True
        n_active = m
    elif n_active > 50:
        # Use top 50
        top_idx = np.argsort(c)[-50:]
        active_mask = np.zeros(2*n-1, dtype=bool)
        active_mask[top_idx] = True
        n_active = 50
    
    weights = np.zeros(2*n-1)
    weights[active_mask] = 1.0 / n_active
    
    h = _active_grad_fft(a, weights)
    grad = 2.0 * h
    return grad, n_active, c, c_max


def _nesterov_subgrad(a_init, deadline, max_iter=50000, m_start=5, m_end=20):
    """
    Nesterov-accelerated subgradient method with adaptive active set.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Nesterov state
    x = a_init.copy()  # current point
    y = a_init.copy()  # extrapolated point
    beta_nesterov = 1.0  # momentum parameter
    
    # Step size parameters
    # Use Polyak-type step: step = f(x) - f* / ||g||^2
    # Since we don't know f*, use adaptive: step = alpha / (1 + t)
    alpha0 = 0.05
    t = 0
    
    patience = 0
    max_patience = 200
    
    for i in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Adaptive active set size: start small, grow over time
        prog = min(1.0, i / (max_iter * 0.5))
        m_cur = int(m_start + (m_end - m_start) * prog)
        
        # Compute subgradient at extrapolated point
        grad, n_active, c, c_max = _subgrad_active(y, m=m_cur, tol_frac=0.005)
        
        # Project gradient onto tangent space of simplex
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break
        
        # Step size: adaptive with Polyak-like schedule
        step = alpha0 / (1.0 + 0.1 * t)
        
        # Take step
        x_new = y - step * grad_t
        
        # Nesterov extrapolation for next iteration
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_nesterov**2)) / 2.0
        y_new = x_new + (beta_nesterov / beta_new) * (x_new - x)
        
        # Project to simplex (clip and renormalize)
        y_new = np.maximum(y_new, 0.0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = best_a.copy()
        else:
            y_new = y_new / s
        
        x = x_new
        y = y_new
        beta_nesterov = beta_new
        t += 1
        
        # Check improvement periodically
        if i % 100 == 0 or i == max_iter - 1:
            a_check = y.copy()
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Reset momentum and reduce step size
                    x = best_a.copy()
                    y = best_a.copy()
                    beta_nesterov = 1.0
                    alpha0 *= 0.5
                    patience = 0
                    if alpha0 < 1e-6:
                        break
    
    # Final check
    sc_final = _score(y)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = y.copy()
    
    return best_a, best_sc


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # grad = 4*n*a*(h_w - C_w) where h_w[j] = sum_k w[k]*a[k-j]
    C_w = float(np.dot(w, c))
    h_w = _active_grad_fft(a, w)
    return 4.0 * n * a * (h_w - C_w)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Adaptive betas: start from gap-based estimate
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 5:
        gap = c_sorted[0] - c_sorted[5]
    else:
        gap = max_c * 0.1
    base_beta = max(gap, max_c * 0.005)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125, 0.00390625, 0.001953125, 0.0009765625])
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 15))))
        
        def fun_local(z_local):
            return _smoobj(z_local, beta, n)
        
        def jac_local(z_local):
            return _smograd(z_local, beta, n)
        
        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _make_uniform(n):
    return np.ones(n) / n


def _make_tapered(n):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_phase(n, rng):
    """Initialize with random phase in Fourier domain for flat autocorrelation."""
    N = _next_pow2(2 * n - 1)
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0
    for f in range(1, N//2 + 1):
        if N - f < N:
            A[f] = np.exp(1j * rng.uniform(0, 2*np.pi))
            A[N-f] = np.conj(A[f])
        else:
            A[f] = np.exp(1j * rng.uniform(0, 2*np.pi))
    a = np.fft.irfft(A, N)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return _make_uniform(n)
    return a / s


def _make_skewed(n):
    t = np.linspace(0, 1, n)
    a = np.exp(-2.0 * t) + 0.1
    return a / a.sum()


def _symmetrize(a):
    """Average with reversed version for symmetry."""
    return (a + a[::-1]) / 2.0


def _peak_repair(a, deadline, max_iters=5000):
    """
    Targeted repair: find the peak, identify contributing elements,
    and redistribute mass to reduce the peak.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a = a.copy()
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_k = int(np.argmax(c))
        
        # Find which pairs contribute most to the peak
        # c[peak_k] = sum_i a[i] * a[peak_k - i]
        valid_i = np.arange(max(0, peak_k - (n-1)), min(n-1, peak_k) + 1)
        contributions = np.array([a[i] * a[peak_k - i] for i in valid_i])
        
        if len(contributions) == 0:
            break
        
        # Find the top contributing pair
        max_contrib_idx = int(np.argmax(contributions))
        i_star = int(valid_i[max_contrib_idx])
        j_star = peak_k - i_star
        
        # Reduce the larger of a[i_star], a[j_star] and increase the smaller
        if a[i_star] >= a[j_star]:
            # Move mass from i_star to a random low-value position
            a[i_star] *= 0.99
        else:
            a[j_star] *= 0.99
        
        # Renormalize
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Check improvement
        if i % 50 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
            elif i > 100 and sc > best_sc * 1.001:
                a = best_a.copy()
    
    return best_a, best_sc
