"""
Multi-length optimizer with targeted peak reduction, FISTA with adaptive beta,
and progressive length scaling (4096 -> 8192 -> 16384 -> 32768).
Targets sub-1.5027 score by exploiting the scale advantage of longer sequences
and using a novel responsibility-based peak shaving method.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _convolve(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    """Gradient of smoothed max convolution at y, with weights w over convolution positions."""
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA with softmax-smoothed max convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _targeted_peak_reduction(a, deadline, max_iters=5000, n_peaks_start=50, n_peaks_end=200,
                             frac_start=0.01, frac_end=0.0001):
    """Novel peak shaving: compute responsibility of each a[i] for top peaks, then adjust."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_M = _convolve(a).max()
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Interpolate parameters
        frac = max_iters / (it + 1) if it < max_iters / 2 else 1.0
        progress = min(1.0, it / (0.8 * max_iters))
        n_peaks = int(n_peaks_start + (n_peaks_end - n_peaks_start) * progress)
        reduction_frac = frac_start * (frac_end / frac_start) ** progress

        c = _convolve(a)
        M = float(c.max())

        # Find top peaks
        if n_peaks >= len(c) - 1:
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argpartition(c, -n_peaks)[-n_peaks:]

        # Compute responsibility: resp[i] = sum_{j in top_idx} a[j-i] for valid j-i
        resp = np.zeros(n)
        for j in top_idx:
            i_lo = max(0, j - n + 1)
            i_hi = min(n - 1, j)
            if i_hi >= i_lo:
                # a[j - i] for i in [i_lo, i_hi] means indices j-i_hi to j-i_lo
                a_slice = a[j - i_hi:j - i_lo + 1][::-1]
                resp[i_lo:i_hi + 1] += a_slice

        # Normalize responsibility
        resp_mean = resp.mean()
        resp_std = resp.std()
        if resp_std < 1e-30:
            break
        resp_norm = (resp - resp_mean) / resp_std

        # Adjust: reduce high-responsibility, increase low-responsibility
        adjust = reduction_frac * resp_norm
        adjust = np.clip(adjust, -0.4, 0.4)

        a_new = a * (1.0 + adjust)
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            break
        a_new = a_new / a_new_sum

        c_new = _convolve(a_new)
        M_new = float(c_new.max())

        if M_new < M - 1e-15:
            a = a_new
            if M_new < best_M:
                best_M = M_new
                best_a = a_new.copy()
        else:
            # Try without projection (simple scaling)
            a_try = a * (1.0 - reduction_frac * resp_norm)
            a_try = np.maximum(a_try, 0.0)
            a_try_sum = a_try.sum()
            if a_try_sum > 1e-30:
                a_try = a_try / a_try_sum
                M_try = float(_convolve(a_try).max())
                if M_try < best_M:
                    a = a_try
                    best_M = M_try
                    best_a = a.copy()

    return best_a, best_M


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 40 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.94)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    """Interpolate profile to new length, preserving shape."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)

            a_r, M_r = _optimize_at_length(p_start, deadline, per_restart)
            sc_r = _score(a_r)
            print(f"  Restart {i}: score={sc_r:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()

    # === Phase 6: Final ultra-fine polish ===
    print(f"Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = np.maximum(a_pol, 0.0)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol)
        max_cp = float(c_pol.max())

        # Very fine targeted peak reduction
        t_pol_end = deadline - 5
        time_budget_pol = t_pol_end - time.monotonic()
        if time_budget_pol > 15:
            a_pol2, M_pol2 = _targeted_peak_reduction(
                a_pol, t_pol_end, max_iters=5000,
                n_peaks_start=200, n_peaks_end=1000,
                frac_start=0.002, frac_end=0.00005
            )
            a_pol2 = a_pol2 / a_pol2.sum()
            sc_pol2 = _score(a_pol2)
            print(f"  Polish target: score={sc_pol2:.10f}")
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_seq = a_pol2.copy()

            # Tiny Adam step
            if time.monotonic() < deadline - 5:
                c_p = _convolve(a_pol2)
                mc = float(c_p.max())
                betas_ult = [1e-6 * mc, 5e-7 * mc]
                a_pol3, _ = _adam_logspace(a_pol2, deadline - 2, betas_ult, lr_init=0.002,
                                           max_iter_total=20000)
                a_pol3 = a_pol3 / a_pol3.sum()
                sc_pol3 = _score(a_pol3)
                print(f"  Polish Adam: score={sc_pol3:.10f}")
                if sc_pol3 < best_score:
                    best_score = sc_pol3
                    best_seq = a_pol3.copy()

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL internal: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
