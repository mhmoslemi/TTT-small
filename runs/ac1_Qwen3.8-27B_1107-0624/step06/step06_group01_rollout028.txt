```python
import numpy as np
import time
import scipy.optimize
import scipy.sparse

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(a):
    """Build the matrix A such that A @ g = a * g (linear convolution)."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, i] = a[k-i] if 0 <= k-i <= n-1 else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            A[k, i] = a[k-i]
    return A
```

"""
LP-guided optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines linear programming for descent directions with Adam log-space fine-tuning.
"""
import numpy as np
import time
import scipy.optimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k + 1] = a[k::-1]
    for k in range(n, 2 * n - 1):
        A[k, k - n + 1:] = a[n - 1:k - n:-1]
    return A


def _lp_direction(a, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns rescaled g."""
    n = len(a)
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = scipy.optimize.linprog(
        c_obj, A_ub=A, b_ub=b,
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )
    
    if not res.success:
        return None
    
    g = res.x
    s = float(g.sum())
    if s < 1e-15:
        return None
    return g / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 8000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

"""
LP-guided optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses linear programming to find descent directions, combined with Adam log-space fine-tuning
and FISTA with softmax-smoothed max for hybrid optimization.
"""
import numpy as np
import time
import scipy.optimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(a):
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(n):
        A[k, :k + 1] = a[k::-1]
    for k in range(n, 2 * n - 1):
        A[k, k - n + 1:] = a[n - 1:k - n:-1]
    return A


def _lp_direction(a, M, time_limit=20.0):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns rescaled g or None."""
    n = len(a)
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None

    g = np.asarray(res.x, dtype=np.float64)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s < 1e-15:
        return None
    return g / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / beta
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 8000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lp_adam_hybrid(a_init, deadline, n_lp_max=60):
    """Hybrid LP + Adam optimization loop."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_sc = _score(a_curr)
    successive_fails = 0

    for it in range(n_lp_max):
        if time.monotonic() > deadline:
            break

        c_curr = _full_convolve(a_curr, a_curr)
        M = float(c_curr.max())

        g = _lp_direction(a_curr, M, time_limit=25.0)
        if g is None:
            print(f"    LP failed at iter {it}")
            break

        a_old = a_curr.copy()
        s_old = _score_norm(a_curr)
        improved = False

        for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_try = (1.0 - alpha) * a_old + alpha * g
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            sc_try = _score_norm(a_try)
            if sc_try < s_old - 1e-14:
                a_curr = a_try
                sc_full = _score(a_curr)
                if sc_full < best_sc:
                    best_sc = sc_full
                    best_a = a_curr.copy()
                improved = True
                break

        if not improved:
            # Try blending with current a to ensure some smoothness
            for blend in [0.15, 0.1, 0.08, 0.05]:
                a_try = (1.0 - blend) * a_old + blend * g
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    sc_try = _score_norm(a_try)
                    if sc_try < s_old - 1e-14:
                        a_curr = a_try
                        sc_full = _score(a_curr)
                        if sc_full < best_sc:
                            best_sc = sc_full
                            best_a = a_curr.copy()
                        improved = True
                        break

        if improved:
            successive_fails = 0
        else:
            successive_fails += 1
            if successive_fails >= 4:
                print(f"    LP stuck after {successive_fails} consecutive failures")
                break

        # Periodic Adam micro-polish
        if (it + 1) % 8 == 0 and time.monotonic() < deadline - 60:
            c_chk = _full_convolve(a_curr, a_curr)
            max_cc = float(c_chk.max())
            micro_betas = [0.0005 * max_cc, 0.0002 * max_cc]
            a_curr, _ = _adam_logspace(
                a_curr, time.monotonic() + min(20.0, deadline - time.monotonic() - 10),
                micro_betas, lr_init=0.008, max_iter_total=30000
            )
            a_curr = a_curr / a_curr.sum()
            sc_chk = _score(a_curr)
            if sc_chk < best_sc:
                best_sc = sc_chk
                best_a = a_curr.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n = 4096
        a0 = np.ones(n) / n
        best_seq = a0.copy()
        best_score = _score(a0)
        print(f"No parent, uniform n={n}, score={best_score:.10f}")

    # === Phase 1: FISTA warm-up to get a good starting point ===
    print(f"\n=== Phase 1: FISTA warm-up ===")
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_w = _full_convolve(a_work, a_work)
    max_wc = float(c_w.max())

    fista_deadline = min(deadline - 600, t_start + 120.0)
    a_work, _ = _fista_smoothed(a_work, 0.01 * max_wc, fista_deadline, max_iter=20000)
    a_work = a_work / a_work.sum()
    sc_w = _score(a_work)
    if sc_w < best_score:
        best_score = sc_w
        best_seq = a_work.copy()
        print(f"  FISTA: score={sc_w:.10f}")
    else:
        a_work = best_seq.copy()
        print(f"  FISTA: no improvement ({sc_w:.10f} vs {best_score:.10f})")

    # === Phase 2: LP-guided optimization (main phase) ===
    print(f"\n=== Phase 2: LP-guided optimization ===")
    a_lp = a_work.copy()
    a_lp = a_lp / a_lp.sum()

    lp_deadline = deadline - 350.0
    a_lp, sc_lp = _lp_adam_hybrid(a_lp, lp_deadline, n_lp_max=80)
    if sc_lp < best_score:
        best_score = sc_lp
        best_seq = a_lp.copy()
        print(f"  LP phase: score={sc_lp:.10f}")
    else:
        print(f"  LP phase: no improvement ({sc_lp:.10f} vs {best_score:.10f})")

    # === Phase 3: Adam log-space fine optimization ===
    print(f"\n=== Phase 3: Adam fine optimization ===")
    a_adam = best_seq.copy()
    a_adam = a_adam / a_adam.sum()

    c_ad = _full_convolve(a_adam, a_adam)
    max_ca = float(c_ad.max())

    betas_fine = []
    beta = 0.001 * max_ca
    while beta > 1e-7 * max_ca:
        betas_fine.append(beta)
        beta *= 0.5
    if len(betas_fine) > 15:
        betas_fine = betas_fine[:15]

    adam_deadline = deadline - 150.0
    a_opt, sc_opt = _adam_logspace(
        a_adam, adam_deadline,
        betas_fine, lr_init=0.012, max_iter_total=500000
    )
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam: score={sc_opt_full:.10f}")
    else:
        print(f"  Adam: no improvement ({sc_opt_full:.10f} vs {best_score:.10f})")

    # === Phase 4: Multi-restart diversification ===
    print(f"\n=== Phase 4: Multi-restart ===")
    n_restarts = 6
    per_restart = (deadline - time.monotonic() - 50.0) / n_restarts

    restart_strategies = [
        ("perturb_small", 0.008),
        ("perturb_med", 0.02),
        ("shift", 0),
        ("blend_sin", 0),
        ("perturb_large", 0.04),
        ("blend_taper", 0),
    ]

    for i, (strategy, param) in enumerate(restart_strategies):
        if time.monotonic() > deadline - 40:
            break

        n_best = len(best_seq)
        if strategy == "perturb_small":
            p_start = _perturb(best_seq, param, rng)
        elif strategy == "perturb_med":
            p_start = _perturb(best_seq, param, rng)
        elif strategy == "shift":
            sz = max(1, n_best // 4096)
            shift = rng.integers(1, max(2, sz))
            p_start = np.roll(best_seq, shift).copy()
            p_start = p_start / p_start.sum()
        elif strategy == "blend_sin":
            alt = _make_profile(n_best, "sinusoidal")
            p_start = 0.85 * best_seq + 0.15 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
        elif strategy == "perturb_large":
            p_start = _perturb(best_seq, param, rng)
        else:
            alt = _make_profile(n_best, "tapered")
            p_start = 0.85 * best_seq + 0.15 * alt
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

        c_s = _full_convolve(p_start, p_start)
        max_cs = float(c_s.max())
        bs = [0.0004 * max_cs, 0.00015 * max_cs, 0.00006 * max_cs, 0.00002 * max_cs]

        r_deadline = time.monotonic() + max(5.0, per_restart)
        p_best, _ = _adam_logspace(
            p_start, r_deadline,
            bs, lr_init=0.008, max_iter_total=100000
        )
        sc_full = _score(p_best)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = p_best.copy()
            print(f"  Restart {i} ({strategy}): score={sc_full:.10f}")
        else:
            print(f"  Restart {i} ({strategy}): {sc_full:.10f} (no improvement)")

    # === Phase 5: Final ultra-fine polish ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [2e-5 * max_cp, 8e-6 * max_cp, 3e-6 * max_cp, 1e-6 * max_cp]

        pol_deadline = min(deadline - 3, time.monotonic() + 25.0)
        a_pol, _ = _adam_logspace(
            a_pol, pol_deadline,
            betas_ult, lr_init=0.003, max_iter_total=60000
        )
        sc_pol = _score(a_pol)
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()
            print(f"  Final polish: score={sc_pol:.10f}")
        else:
            print(f"  Final polish: {sc_pol:.10f} (no improvement)")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL verified score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
