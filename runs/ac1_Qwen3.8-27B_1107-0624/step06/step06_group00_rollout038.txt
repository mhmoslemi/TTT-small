def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_half(a):

    n = len(a)

    # where (D_j)_{k,l} = 1 if k-l=j
    # a^T D_j a = sum_{k,l} a[k] (D_j)_{k,l} a[l] = sum_k a[k] a[k-j]
    
    # For the quadratic form, we need symmetric Q. Since D_j is not symmetric (for j>0),
    # we use: a^T D_j a = (1/2) a^T (D_j + D_j^T) a
    # (D_j + D_j^T)_{k,l} = 1_{k-l=j} + 1_{l-k=j} = 1_{|k-l|=j}
    
    # So Q = sum_{j in A} (D_j + D_j^T)/2 + D_0 (if 0 in A)
    # Wait, for j=0: D_0 is the identity, already symmetric.
    # a^T D_0 a = sum a[k]^2
    # (1/2)(D_0 + D_0^T) = D_0 = I
    
    # So Q = D_0 (if 0 in A) + sum_{j in A, j>0} (1/2)(D_j + D_j^T)
    # (Q)_{k,l} = 1_{k==l, 0 in A} + sum_{j in A, j>0} (1/2) * 1_{|k-l|=j}
    
    # For a sparse matrix, Q has at most (2*top_k + 1) non-zero diagonals.
    
    # Let's build Q as asparse matrix
    import scipy.sparse as sp
    Q = sp.lil_matrix((n, n))
    for j in top_idx:
        if j == 0:
            for k in range(n):
                Q[k, k] += 1.0
        else:
            for k in range(n - j):
                Q[k, k+j] += 0.5
                Q[k+j, k] += 0.5
    Q = Q.tocsr()
    
    def objective(x):
        return float(x @ Q @ x)
    
    def constraint_sum(x):
        return float(np.sum(x) - 1.0)
    
    bounds = [(0, None)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        res = sp_minimize(objective, a_init, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-10})
        if res.success or res.fun < objective(a_init):
            a_opt = np.maximum(res.x, 0)
            a_opt = a_opt / a_opt.sum()
            return a_opt
    except Exception as e:
        print(f"  QP polish error: {e}")
    
    return a_init

def _init_uniform(n):
    return np.ones(n) / n

def _init_gaussian(n, sigma_frac=0.3):
    x = np.arange(n)
    center = (n-1)/2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center)/sigma)**2)
    return a / a.sum()

def _init_bimodal(n, sep_frac=0.3):
    x = np.arange(n)
    center1 = (1-sep_frac) * (n-1)/2.0
    center2 = (1+sep_frac) * (n-1)/2.0
    sigma = 0.1 * n
    a = np.exp(-0.5*((x-center1)/sigma)**2) + np.exp(-0.5*((x-center2)/sigma)**2)
    return a / a.sum()

def _init_symmetric_triangular(n):
    half = n // 2
    if n % 2 == 0:
        a = np.concatenate([np.arange(1, half+1), np.arange(half-1, 0, -1)])
    else:
        a = np.concatenate([np.arange(1, half+1), np.arange(half, 0, -1)])
    return a / a.sum()

def _init_random_symmetric(n, rng, sigma=1.0):
    half = (n + 1) // 2
    h = np.abs(rng.standard_normal(half)) * sigma + 0.1
    if n % 2 == 0:
        a = np.concatenate([h[::-1], h])
    else:
        a = np.concatenate([h[::-1], h[1:]])
    return a / a.sum()

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _project_simplex(v):
    """Project v onto the simplex {x: x>=0, sum(x)=1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)
