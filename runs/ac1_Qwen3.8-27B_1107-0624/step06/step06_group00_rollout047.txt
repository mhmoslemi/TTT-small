"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improvements over baseline: better LP line search, flat-peak gradient,
spectral guidance, adaptive active-set, multi-scale exploration,
and enhanced perturbation strategies.

import numpy as np

import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) is close to 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_soft_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_idx = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_idx] = 1.0 / len(k_idx)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_flat_gradient(a, c, top_k=8, alpha=0.5):
    """Gradient that weights peaks by their excess over the mean of top-k peaks.
    This encourages flattening the peak profile."""
    n = len(a)
    k_sorted = np.argsort(c)[::-1][:top_k]
    c_top = c[k_sorted]
    c_mean = c_top.mean()
    c_max = c_top.max()
    if c_max - c_mean < 1e-15:
        return _compute_active_gradient(a, c, top_k)
    p = np.zeros_like(c)
    for idx in k_sorted:
        weight = (c[idx] - c_mean) / (c_max - c_mean)
        p[idx] = weight
    p_sum = p.sum()
    if p_sum < 1e-15:
        p[k_sorted] = 1.0 / len(k_sorted)
        p_sum = 1.0
    else:

    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=18):
    """LP step: find b maximizing sum(b) s.t. (a*b)(k) <= M, b >= 0.
    Then blend with line search."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()
        # Line search over t (improved: log-spaced + fine grid)
        best_t = 0.0
        best_sc = _score_norm(a)
        t_candidates = np.concatenate([
            np.linspace(0.05, 1.0, 16),
            np.array([0.01, 0.02, 0.03, 0.05, 0.08, 0.15, 0.25])
        ])
        for t_val in t_candidates:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            return a, M
        a = (1.0 - best_t) * a + best_t * g
        return a, M
    except Exception:
        return a, M


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _adam_z_space(a, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, grad_type='active', active_k=5,
                  flat_k=8, eval_interval=200, clip_norm=0.6,
                  patience_threshold=8):
    """Adam optimizer in log-space with adaptive learning rate."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        if grad_type == 'soft':
            g = _compute_soft_gradient(a_cur, c, T)
        elif grad_type == 'flat':
            g = _compute_flat_gradient(a_cur, c, flat_k)
        else:
            g = _compute_active_gradient(a_cur, c, active_k)
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        if t % eval_interval == 0:
            a_eval = _softmax_stable(z)
            sc = _score_norm(a_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
            if patience > patience_threshold and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _subgradient_step(a, deadline, max_iter=5000, alpha=0.5):
    """Polyak-style subgradient method on the simplex."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    # Lower bound: 2n/(2n-1) * (normalized)
    f_lb = 2.0 * n / (2.0 * n - 1.0)
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        f_cur = 2.0 * n * c_max
        if f_cur <= f_lb + 1e-10:
            break
        # Subgradient: equal weight on all max-attaching lags
        k_max = np.where(c >= c_max - 1e-10)[0]
        if len(k_max) == 0:
            k_max = np.array([np.argmax(c)])
        # Limit to top min(len, 20) for stability
        if len(k_max) > 20:
            k_max = k_max[np.argsort(c[k_max])[::-1][:20]]
        p = np.zeros(2 * n - 1)
        p[k_max] = 1.0 / len(k_max)
        p_rev = p[::-1]
        conv_full = _fft_convolve(a, p_rev)
        g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
        # Polyak step size
        gap = max(f_cur - f_lb, f_lb * 1e-3)
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-20:
            break
        eta = alpha * gap / (g_norm_sq * n)
        eta = min(eta, 0.1)
        a_new = a - eta * g
        a_new = _project_simplex(a_new)
        sc = _score_norm(a_new)
        a = a_new
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _spectral_flatten(a, deadline, max_iter=2000):
    """Try to flatten |FFT(a)|^2 by adjusting in log-space.
    The idea: if |FFT(a)|^2 has peaks, those correspond to periodicity in a,
    which creates structure in the autocorrelation. We try to reduce them."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = 0.01
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        sc = _score_norm(a_cur)
        # Compute gradient w.r.t. z for the spectral flatness objective
        # Use the active-set gradient as a proxy
        g = _compute_active_gradient(a_cur, c, top_k=5)
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        z = z - lr * z_grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        lr *= 0.999
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_a = a_final.copy()
        best_score = sc_final
    return best_a, best_score


def _generate_diverse_inits(n, rng):
    """Generate diverse initial sequences for multi-start."""
    inits = []
    # 1. Geometric decay (left-heavy)
    r = 0.98
    a = np.array([r ** j for j in range(n)])
    a = a / a.sum()
    inits.append(a)
    # 2. Geometric decay (right-heavy)
    a = a[::-1].copy()
    inits.append(a)
    # 3. Two-sided exponential
    mid = n // 2
    a = np.array([exp(-abs(j - mid) / (n * 0.3)) for j in range(n)])
    a = a / a.sum()
    inits.append(a)
    # 4. Sine window (anti-symmetric bias)
    a = np.array([max(0.0, 1.0 + 0.5 * sin(2 * pi * j / n)) for j in range(n)])
    s = a.sum()
    if s > 1e-10:
        inits.append(a / s)
    # 5. Random (log-normal)
    a = np.exp(rng.normal(0, 0.5, n))
    a = a / a.sum()
    inits.append(a)
    # 6. Sparse random
    a = np.zeros(n)
    n_sparse = n // 4
    idx = rng.choice(n, n_sparse, replace=False)
    a[idx] = 1.0
    a = a / a.sum()
    inits.append(a)
    # 7. Linear ramp (strongly asymmetric)
    a = np.arange(1, n + 1, dtype=float)
    a = a / a.sum()
    inits.append(a)
    # 8. Rectangular pulse (first half)
    a = np.zeros(n)
    a[:n // 2] = 1.0
    a = a / a.sum()
    inits.append(a)
    # 9. Triangular (centered, but shifted)
    center = n // 3
    a = np.array([max(0, 1 - abs(j - center) / (n / 3)) for j in range(n)])
    s = a.sum()
    if s > 1e-10:
        inits.append(a / s)
    # 10. Cosine window (main lobe only)
    a = np.array([max(0, cos(pi * j / n)) ** 2 for j in range(n)])
    s = a.sum()
    if s > 1e-10:
        inits.append(a / s)
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse initialization -> LP with line search ->
    Adam (soft/flat/active) with adaptive schedules -> subgradient ->
    multi-scale exploration -> perturbation restarts -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent if available
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Build LP indices (expensive, do once)
    rows, cols = _build_lp_indices(n)
    print(f"LP indices built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # === Phase 0: Quick diverse init test (0 - 30s) ===
    phase0_end = min(deadline, t_start + 30.0)
    if time.monotonic() < phase0_end:
        inits = _generate_diverse_inits(n, rng)
        for i, a_init in enumerate(inits):
            if time.monotonic() > phase0_end - 2:
                break
            sc = _score_norm(a_init)
            print(f"  Init[{i}]: norm_score={sc:.6f}")
            # Quick Adam run on this init
            a_test = a_init.copy()
            c_t = _fft_convolve(a_test, a_test)
            max_ct = float(c_t.max())
            a_test, sc_test = _adam_z_space(
                a_test, min(phase0_end - 1, t_start + 3.0),
                T_start=0.01 * max_ct, T_end=0.001 * max_ct,
                lr_start=0.05, lr_end=0.01,
                max_iter=2000, grad_type='active', active_k=5,
                eval_interval=100, patience_threshold=3)
            sc_full = _score(a_test)
            if np.isfinite(sc_full) and sc_full < best_score:
                best_score = sc_full
                best_a = a_test.copy()
                print(f"  Init[{i}] improved: score={sc_full:.10f}")

    # === Phase 1: LP iterations with line search (30s - 150s) ===
    phase1_end = min(deadline, t_start + 150.0)
    if time.monotonic() < phase1_end:
        a = best_a.copy()
        lp_count = 0
        prev_sc = _score_norm(a)
        stall = 0
        while time.monotonic() < phase1_end - 2:
            a, M = _lp_step(a, rows, cols, time_limit=12)
            sc = _score_norm(a)
            lp_count += 1
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            full_sc = _score(a)
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a.copy()
            if lp_count % 5 == 0:
                print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
            if stall > 20:
                break
            prev_sc = sc
        print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 2: Adam soft gradient, medium temperature (150s - 400s) ===
    phase2_end = min(deadline, t_start + 400.0)
    if time.monotonic() < phase2_end:
        a_work = best_a.copy()
        c_cur = _fft_convolve(a_work, a_work)
        max_c = float(c_cur.max())
        a_opt, sc_opt = _adam_z_space(
            a_work, phase2_end - 3,
            T_start=0.008 * max_c, T_end=0.0001 * max_c,
            lr_start=0.04, lr_end=0.004,
            max_iter=500000, grad_type='soft',
            eval_interval=200, clip_norm=0.6)
        full_sc = _score(a_opt)
        print(f"Phase 2 Adam-soft: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # === Phase 3: Adam flat-peak gradient (400s - 700s) ===
    phase3_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase3_end:
        a_work = best_a.copy()
        c_cur = _fft_convolve(a_work, a_work)
        max_c = float(c_cur.max())
        a_opt, sc_opt = _adam_z_space(
            a_work, phase3_end - 3,
            T_start=0.003 * max_c, T_end=0.00005 * max_c,
            lr_start=0.03, lr_end=0.003,
            max_iter=500000, grad_type='flat', flat_k=8,
            eval_interval=200, clip_norm=0.5)
        full_sc = _score(a_opt)
        print(f"Phase 3 Adam-flat: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

        # Also try active-set with smaller k
        a_work = best_a.copy()
        c_cur = _fft_convolve(a_work, a_work)
        max_c = float(c_cur.max())
        a_opt2, sc_opt2 = _adam_z_space(
            a_work, phase3_end - 3,
            T_start=0.002 * max_c, T_end=0.00002 * max_c,
            lr_start=0.02, lr_end=0.002,
            max_iter=300000, grad_type='active', active_k=3,
            eval_interval=200, clip_norm=0.4)
        full_sc2 = _score(a_opt2)
        print(f"Phase 3b Adam-active(k=3): score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_a = a_opt2.copy()

    # === Phase 4: Subgradient refinement (700s - 900s) ===
    phase4_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase4_end:
        a_work = best_a.copy()
        a_opt, sc_opt = _subgradient_step(a_work, phase4_end - 2, max_iter=3000, alpha=0.5)
        full_sc = _score(a_opt)
        print(f"Phase 4 Subgradient: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # === Phase 5: LP again on improved sequence (900s - 1100s) ===
    phase5_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase5_end:
        a = best_a.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp_count = 0
        while time.monotonic() < phase5_end - 2:
            a, M = _lp_step(a, rows, cols, time_limit=10)
            sc = _score_norm(a)
            lp_count += 1
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if stall > 15:
                break
            prev_sc = sc
        full_sc = _score(a)
        print(f"Phase 5 LP round 2: score={full_sc:.10f}, {lp_count} iters, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()

    # === Phase 6: Multi-scale exploration (1100s - 1400s) ===
    phase6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase6_end:
        # Try n=8192
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        a_large_opt, sc_large = _adam_z_space(
            a_large, min(phase6_end - 30, t_start + 1250.0),
            T_start=0.004 * max_cl, T_end=0.00004 * max_cl,
            lr_start=0.03, lr_end=0.003,
            max_iter=200000, grad_type='active', active_k=5,
            eval_interval=200, clip_norm=0.5)
        full_sc = _score(a_large_opt)
        print(f"Phase 6 n=8192 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_large_opt.copy()

        # Try interpolating back to n=4096
        if len(best_a) != n:
            a_interp = _interp_to_length(best_a, n)
            c_int = _fft_convolve(a_interp, a_interp)
            max_ci = float(c_int.max())
            a_interp_opt, sc_int = _adam_z_space(
                a_interp, phase6_end - 3,
                T_start=0.002 * max_ci, T_end=0.00002 * max_ci,
                lr_start=0.02, lr_end=0.002,
                max_iter=100000, grad_type='active', active_k=3,
                eval_interval=150, clip_norm=0.4)
            full_sc_int = _score(a_interp_opt)
            print(f"Phase 6 interp back n={n}: score={full_sc_int:.10f}")
            if np.isfinite(full_sc_int) and full_sc_int < best_score:
                best_score = full_sc_int
                best_a = a_interp_opt.copy()
        else:
            # Also try a fresh Adam run on current best
            a_work = best_a.copy()
            c_cur = _fft_convolve(a_work, a_work)
            max_c = float(c_cur.max())
            a_opt, sc_opt = _adam_z_space(
                a_work, phase6_end - 3,
                T_start=0.002 * max_c, T_end=0.00002 * max_c,
                lr_start=0.02, lr_end=0.002,
                max_iter=150000, grad_type='flat', flat_k=6,
                eval_interval=150, clip_norm=0.4)
            full_sc2 = _score(a_opt)
            print(f"Phase 6b Adam-flat n={n}: score={full_sc2:.10f}")
            if np.isfinite(full_sc2) and full_sc2 < best_score:
                best_score = full_sc2
                best_a = a_opt.copy()

    # === Phase 7: Perturbation restarts (1400s - 1650s) ===
    phase7_end = min(deadline, t_start + 1650.0)
    n_best = len(best_a)
    if time.monotonic() < phase7_end:
        restart_configs = [
            (0.02, 'active', 3),
            (0.015, 'flat', 6),
            (0.025, 'active', 5),
            (0.01, 'soft', 5),
            (0.03, 'active', 3),
        ]
        for attempt, (sigma, gtype, gk) in enumerate(restart_configs):
            if time.monotonic() > phase7_end - 40:
                break
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_deadline = min(phase7_end - 3, phase7_end - 40 * (len(restart_configs) - attempt))
            a_pert_opt, sc_pert = _adam_z_space(
                a_pert, sub_deadline,
                T_start=0.003 * max_cp, T_end=0.00003 * max_cp,
                lr_start=0.025, lr_end=0.002,
                max_iter=60000, grad_type=gtype,
                active_k=gk if gtype != 'flat' else 6,
                flat_k=6 if gtype == 'flat' else 0,
                eval_interval=150, clip_norm=0.4,
                patience_threshold=5)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}, {gtype}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # === Phase 8: Final ultra-fine polish (1650s - deadline) ===
    if time.monotonic() < deadline - 8:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_z_space(
            best_a, deadline - 5,
            T_start=0.0002 * max_cf, T_end=0.000005 * max_cf,
            lr_start=0.005, lr_end=0.0003,
            max_iter=300000, grad_type='active', active_k=3,
            eval_interval=100, clip_norm=0.3,
            patience_threshold=6)
        full_sc = _score(a_final)
        print(f"Phase 8 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # One more LP polish
        if time.monotonic() < deadline - 15 and len(best_a) == n:
            a_lp = best_a.copy()
            a_lp, _ = _lp_step(a_lp, rows, cols, time_limit=5)
            full_sc_lp = _score(a_lp)
            if np.isfinite(full_sc_lp) and full_sc_lp < best_score:
                best_score = full_sc_lp
                best_a = a_lp.copy()
                print(f"  Final LP polish: score={full_sc_lp:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
