        best_a = a_curr.copy()
    return best_a, best_sc


def _nesterov_subgrad_avg(a_init, deadline, max_iter=200000):
    """Nesterov subgradient with Polyak averaging."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    x_avg = a_curr.copy()
    
    c0 = _convolve_fft(a_curr, a_curr)
    f0 = float(c0.max())
    step0 = f0 / 2.0  # initial step size based on function value
    
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    best_sc_avg = _score_norm(x_avg)
    if best_sc_avg < best_sc:
        best_sc = best_sc_avg
        best_a = x_avg.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _convolve_fft(y, c_y if False else y)
        c_y = _convolve_fft(y, y)
        f_y = float(c_y.max())
        
        # Subgradient
        grad = _subgradient_at(y, c_y)
        
        # Step size: O(1/sqrt(t))
        step = step0 / np.sqrt(t_val + 1.0)
        
        a_next = y - step * grad
        a_next = np.maximum(a_next, 0.0)
        s = a_next.sum()
        if s > 1e-30:
            a_next = a_next / s
        else:
            a_next = np.ones(n) / n
        
        # Polyak averaging
        x_avg = ((t_val - 1.0) * x_avg + a_next) / t_val
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
            sc_avg = _score_norm(x_avg)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = x_avg.copy()
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
    
    # Final check
    sc = _score_norm(a_curr)
    if sc < best_sc:
        best_sc = sc
        best_a = a_curr.copy()
    sc_avg = _score_norm(x_avg)
    if sc_avg < best_sc:
        best_sc = sc_avg
        best_a = x_avg.copy()
    
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_safe(z)
            c = _convolve_fft(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            
            a_rev = a[::-1]
            wc = _convolve_fft(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            
            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax_safe(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)
    
    a_final = _softmax_safe(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_anti_palindromic(n, rng):
    """Create a sequence that's anti-palindromic in spirit.
    Make first half large, second half small (or vice versa) with some structure.
    """
    a = np.zeros(n)
    # Use a combination of sinusoidal and linear patterns
    x = np.linspace(0, 1, n)
    # Anti-palindromic: a[i] + a[n-1-i] should be roughly constant
    # This means a[i] = c - a[n-1-i], suggesting a ramp
    a = x + 0.3 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
    a = np.maximum(a, 0.01)
    # Add some randomness
    a = a + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def _make_periodic(n, period, rng):
    """Create a periodic sequence."""
    base = np.zeros(period)
    # Random pattern with some structure
    base = 1.0 + 0.5 * np.sin(2 * np.pi * np.arange(period) / period) 
    base += 0.2 * rng.standard_normal(period)
    base = np.maximum(base, 0.01)
    n_reps = n // period
    a = np.tile(base, n_reps)
    if n % period != 0:
        a = np.concatenate([a, base[:n % period]])
    a = a + 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def _make_ramp(n, rng):
    """Create a ramp-like sequence."""
    x = np.linspace(0, 1, n)
    a = x + (1 - x) * 0.3 + 0.1 * np.sin(10 * np.pi * x)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def _make_bimodal(n, rng):
    """Create a bimodal sequence (large at two points, small elsewhere)."""
    x = np.linspace(0, 1, n)
    center1 = 0.25
    center2 = 0.75
    width = 0.1
    a = np.exp(-((x - center1) / width) ** 2) + np.exp(-((x - center2) / width) ** 2)
    a += 0.01 * np.ones(n)
    a += 0.03 * rng.standard_normal(n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def _make_random_smooth(n, rng):
    """Create a random but smooth sequence."""
    # Generate in frequency domain for smoothness
    N = _next_pow2(n)
    freq = np.zeros(N // 2 + 1, dtype=complex)
    # Low frequency emphasis
    weights = 1.0 / (1.0 + np.arange(N // 2 + 1) * 0.1)
    freq = (rng.standard_normal(N // 2 + 1) + 1j * rng.standard_normal(N // 2 + 1)) * weights
    freq[0] = 0
    a = np.fft.irfft(freq, N)[:n]
    a = np.abs(a) + 0.1
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.001)
    return a / a.sum()


def _respond_and_modify(a_init, deadline, max_iter=50000):
    """
    Respond-and-modify: alternate between reducing the peak and maintaining feasibility.
    This is inspired by the "Matrix Image Algorithm" from PAPR literature.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    c = _convolve_fft(a, a)
    c_max = float(c.max())
    
    lam = 0.1  # relaxation parameter
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_fft(a, a)
        c_max = float(c.max())
        
        # Find where the peak is
        k_peak = int(np.argmax(c))
        
        # Compute the "excess" at the peak
        # We want to reduce a in a way that reduces c[k_peak]
        # c[k_peak] = sum_i a[i] * a[k_peak - i]
        # To reduce this, we should decrease a[i] where a[i] * a[k_peak-i] is large
        
        # Gradient-based approach: reduce a in direction of subgradient
        grad = _subgradient_at(a, c)
        
        # Step
        step_size = lam * c_max / (np.linalg.norm(grad) ** 2 + 1e-15)
        step_size = min(step_size, 0.1)
        
        a_new = a - step_size * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-30:
            a_new = a_new / s
        else:
            a_new = a.copy()
        
        # Check improvement
        c_new = _convolve_fft(a_new, a_new)
        c_max_new = float(c_new.max())
        
        if c_max_new < c_max * (1 - 1e-12):
            a = a_new
            c_max = c_max_new
            c = c_new
        else:
            # Try smaller step
            step_size *= 0.5
            a_new = a - step_size * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-30:
                a_new = a_new / s
            else:
                a_new = a.copy()
            c_new = _convolve_fft(a_new, a_new)
            c_max_new = float(c_new.max())
            if c_max_new < c_max:
                a = a_new
                c_max = c_max_new
                c = c_new
        
        if it % 500 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

                                         betas_ab, lr_init=0.004, max_iter_total=200000)
                sc_ab = _score(a_ab)
                print(f"  Optimized n=8192: score={sc_ab:.10f} t={time.monotonic()-t_start:.1f}s")
                if sc_ab < best_score:
                    best_score = sc_ab
                    best_seq = a_ab.copy()
    
    # === Phase 6: Also try n=2048 (sometimes smaller is better) [1650 - 1730s] ===
    print("Phase 6: n=2048 exploration")
    phase6_end = min(deadline, t_start + 1730.0)
    
    if time.monotonic() < phase6_end - 20 and best_seq is not None:
        n_small = 2048
        x0 = np.linspace(0, 1, len(best_seq))
        x1 = np.linspace(0, 1, n_small)
        a_small = np.interp(x1, x0, best_seq * np.sqrt(len(best_seq)))
        a_small = np.maximum(a_small, 1e-15)
        a_small = a_small / a_small.sum()
        
        sc_small = _score(a_small)
        print(f"  Interpolated n=2048: score={sc_small:.10f}")
        if sc_small < best_score:
            best_score = sc_small
            best_seq = a_small.copy()
        
        if time.monotonic() < phase6_end - 10:
            c_s = _convolve_fft(a_small, a_small)
            max_cs = float(c_s.max())
            betas_s = [0.003 * max_cs, 0.001 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs]
            t_fs = min(time.monotonic() + 20.0, phase6_end - 5)
            if time.monotonic() < t_fs - 3:
                for bs in betas_s:
                    if time.monotonic() > t_fs - 2:
                        break
                    a_small, _ = _fista_smooth(a_small, bs, time.monotonic() + min(5.0, t_fs - time.monotonic() - 1), 20000)
                    a_small = a_small / a_small.sum()
                sc_small2 = _score(a_small)
                if sc_small2 < best_score:
                    best_score = sc_small2
                    best_seq = a_small.copy()
    
    # === Phase 7: Final ultra-fine polish [1730 - 1775s] ===
    print("Phase 7: Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        
        c_pol = _convolve_fft(a_pol, a_pol)
        max_cp = float(c_pol.max())
        
        # Very fine FISTA
        t_p = min(time.monotonic() + 15.0, deadline - 8)
        if time.monotonic() < t_p - 2:
            betas_ult = [0.00001 * max_cp, 0.000003 * max_cp, 0.000001 * max_cp]
            for bu in betas_ult:
                if time.monotonic() > t_p - 1:
                    break
                a_pol, _ = _fista_smooth(a_pol, bu, time.monotonic() + min(5.0, t_p - time.monotonic() - 1), 30000)
                a_pol = a_pol / a_pol.sum()
                sc_p = _score(a_pol)
                if sc_p < best_score:
                    best_score = sc_p
                    best_seq = a_pol.copy()
        
        # Quick Adam
        if time.monotonic() < deadline - 5 and best_seq is not None:
            a_az = best_seq.copy() / best_seq.sum()
            c_az = _convolve_fft(a_az, a_az)
            max caz = float(c_az.max())
            betas_az = [0.000003 * max_caz, 0.000001 * max_caz]
            a_az, _ = _adam_logspace(a_az, time.monotonic() + max(2, deadline - time.monotonic() - 3),
                                     betas_az, lr_init=0.002, max_iter_total=50000)
            sc_az = _score(a_az)
            if sc_az < best_score:
                best_score = sc_az
                best_seq = a_az.copy()
    
    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
