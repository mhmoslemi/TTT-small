from scipy.optimize import linprog

def _lp_step(a, M, deadline):
    """Solve the LP to find direction b, return b and LP objective value."""
    n = len(a)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # This is a Toeplitz-like matrix
    # For efficiency, we can use the structure
    
    c = np.ones(n)  # objective: maximize sum(b) = c @ b
    # Constraints: A @ b <= M * ones(2n-1), b >= 0
    # In linprog form: minimize -c @ b s.t. A_ub @ b <= b_ub, b >= 0
    
    # Build A_ub (the 2n-1 x n matrix)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            A_ub[k, j] = a[i]
    
    b_ub = np.full(2*n-1, M)
    
    # Solve LP
    result = linprog(-c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        b = result.x
        lp_obj = -result.fun  # = sum(b)
        return b, lp_obj
    else:
        return None, None

def _lp_optimize(a_init, max_iter, deadline):

    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

        if s_b < 1e-10:

Actually, to save memory and time, let me use a sparse matrix:

```python
from scipy.sparse import csr_matrix

def _build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```
