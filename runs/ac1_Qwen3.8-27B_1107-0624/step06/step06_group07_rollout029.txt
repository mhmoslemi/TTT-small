```python
"""
Optimized sequence finder for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive log-sum-exp smoothing with L-BFGS, peak-targeted 
local search, multi-length exploration, and diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoothed_obj(z, beta, n)
    
    def jac(z):
        return _smoothed_grad(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _peak_reduction(a, deadline, max_steps=50000, top_k=5):
    """Targeted reduction of the top peaks in the convolution."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / (n * 10)
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Find top-k peaks
        top_indices = np.argsort(c)[-top_k:]
        # Weight: higher for peaks closer to max
        weights = np.zeros_like(c)
        for idx in top_indices:
            frac = c[idx] / c_max
            weights[idx] = frac ** 10  # Sharp weighting
        
        w = weights / weights.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()  # Maintain sum
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        a_try = a - step_size * grad / gnorm
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
        
        if sc_try < _score(a) - 1e-15:
            a = a_try
            step_size = min(step_size * 1.02, 0.05)
        else:
            step_size *= 0.9
            
        if step_size < 1e-12:
            break
    
    return best_a, best_sc
```

```python
"""
Optimized sequence finder for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines aggressive L-BFGS with log-sum-exp smoothing, fast coordinate
descent with incremental convolution updates, peak-targeted subgradient,
and multi-length/diversification exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smoothed_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoothed_obj(z, beta, n)

    def jac(z):
        return _smoothed_grad(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _lbfgs_schedule_aggressive(a_start, deadline, min_time_remaining=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    # More aggressive and longer beta schedule
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006,
                              0.003, 0.0015, 0.0007, 0.00035,
                              0.00018, 0.00009, 0.00004, 0.00002,
                              0.00001, 0.000005])
    current_a = a_start.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter based on remaining time
        est_time_per_iter = 0.002 * (n / 4096.0) ** 1.3
        max_iter = min(10000, max(200, int(remaining / (est_time_per_iter * 20))))

        a_new = _lbfgs_smooth(current_a, beta, n, max_iter)

            current_a = a_new

    lr = lr_init
    stall = 0

        if time.monotonic() > deadline:

        c = _full_convolve(a, a)
        c_max = c.max()

                best_a = a_check.copy()
                lr *= 1.05
                stall = 0
            else:
                stall += 1
                if stall > 8:
                    lr *= 0.75
                    if lr < 1e-9:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a, best_sc


def _direct_subgrad(a_init, deadline, max_steps=30000, tol_frac=1e-5):
    """Direct-space subgradient with active set."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * n * (h - W)
        grad = grad - grad.mean()

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        a_try = a - step_size * grad / gnorm
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        if sc_try < _score(a) - 1e-15:
            a = a_try
            step_size = min(step_size * 1.01, 0.1)
        else:
            stall += 1
            step_size *= 0.88

        step_size = max(step_size, 1e-12)
        if stall > 150:
            break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _perturb(a, rng, sigma=0.03, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 16)))
        amp = sigma * 0.3 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_window":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Load parent and quick L-BFGS polish
    2. Fast coordinate descent with incremental conv
    3. Peak-targeted subgradient polish
    4. Try different sequence lengths
    5. Diversified restarts with full pipeline
    6. Final deep optimization loop
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "tapered")
        print(f"No parent. Tapered n={n_parent}, score={_score(parent):.10f}")
        try_update(parent, "init")

    # === Phase 1: Quick L-BFGS polish ===
    print(f"\n=== Phase 1: L-BFGS polish (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 600:
        a_p1 = _lbfgs_schedule_aggressive(best_seq, deadline - 600, min_time_remaining=500.0)
        try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Fast coordinate descent ===
    print(f"\n=== Phase 2: Coordinate descent (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_cd, sc_cd = _coord_descent_fast(best_seq, deadline - 400,
                                          max_rounds=800000, topk=7)
        try_update(a_cd, "phase2_coord")

    # === Phase 3: Peak-targeted subgradient ===
    print(f"\n=== Phase 3: Peak subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _peak_subgrad(best_seq, deadline - 300,
                                    top_k=5, max_steps=30000, lr_init=0.015)
        try_update(a_sg, "phase3_peaks")

    # === Phase 4: Direct subgradient ===
    print(f"\n=== Phase 4: Direct subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_ds, sc_ds = _direct_subgrad(best_seq, deadline - 250,
                                      max_steps=40000, tol_frac=1e-5)
        try_update(a_ds, "phase4_direct")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    current_n = len(best_seq)
    for target_n in [2048, 8192]:
        if time.monotonic() > deadline - 150:
            break
        if abs(target_n - current_n) < 100:
            continue
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        print(f"    Interpolated score: {sc_interp:.10f}")
        # Quick optimize at this length
        if time.monotonic() < deadline - 100:
            a_opt = _lbfgs_schedule_aggressive(a_interp, deadline - 100,
                                               min_time_remaining=90.0)
            try_update(a_opt, f"n={target_n}")
        if time.monotonic() < deadline - 80:
            a_cd2, _ = _coord_descent_fast(a_opt, deadline - 80,
                                           max_rounds=300000, topk=5)
            try_update(a_cd2, f"n={target_n}_coord")

    # === Phase 6: Go back to best length and do deep optimization ===
    print(f"\n=== Phase 6: Deep optimization (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        # Cycle of optimizations
        for cycle in range(3):
            if time.monotonic() > deadline - 60:
                break
            print(f"  Cycle {cycle+1}:")

            # L-BFGS with very small beta
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            for beta_frac in [1e-4, 5e-5, 2e-5, 1e-5]:
                if time.monotonic() > deadline - 60:
                    break
                beta = mb * beta_frac
                remaining = deadline - 60 - time.monotonic()
                if remaining < 10:
                    break
                max_iter = min(5000, max(500, int(remaining / 0.005)))
                a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
                try_update(a_f, f"deep_lbfgs_{cycle}_{beta_frac}")

            # Coordinate descent
            if time.monotonic() < deadline - 50:
                a_cd3, _ = _coord_descent_fast(best_seq, deadline - 50,
                                               max_rounds=500000, topk=9)
                try_update(a_cd3, f"deep_coord_{cycle}")

            # Peak subgradient
            if time.monotonic() < deadline - 40:
                a_sg2, _ = _peak_subgrad(best_seq, deadline - 40,
                                         top_k=7, max_steps=10000, lr_init=0.008)
                try_update(a_sg2, f"deep_peak_{cycle}")

    # === Phase 7: Diversified restarts ===
    print(f"\n=== Phase 7: Diversified restarts (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            ("g002", 0.02, 2),
            ("g005", 0.05, 3),
            ("g010", 0.10, 2),
            ("g015", 0.15, 4),
            ("g020", 0.20, 3),
        ]
        for name, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 45:
                break
            p_start = _perturb(best_seq, rng, sigma=sigma, n_modes=modes)
            sc_start = _score(p_start)

            if sc_start < best_score_val + 0.02:
                # Worth optimizing
                a_r = _lbfgs_schedule_aggressive(p_start, deadline,
                                                 min_time_remaining=35.0)
                try_update(a_r, f"restart_{name}")
                if time.monotonic() < deadline - 30:
                    a_r2, _ = _coord_descent_fast(a_r, deadline - 30,
                                                  max_rounds=200000, topk=5)
                    try_update(a_r2, f"restart_{name}_coord")

    # === Phase 8: Final polish loop ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break
            # Tiny coordinate descent
            a_f, _ = _coord_descent_fast(best_seq, deadline - 5,
                                         max_rounds=200000, topk=10)
            try_update(a_f, f"final_cd_{rnd}")

            if time.monotonic() > deadline - 5:
                break
            # Tiny L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            beta = float(c_b.max()) * 5e-7
            a_f2 = _lbfgs_smooth(best_seq, beta, n, max_iter=2000)
            try_update(a_f2, f"final_lbfgs_{rnd}")

    # === Final ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
