```python
    while k < n:

    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s
```

```python
"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed L-BFGS, projected subgradient with Nesterov
acceleration, pairwise coordinate search, and diverse multi-restart.
"""
import numpy as np
import time
from scipy.optimize import minimize

# === Helper functions ===

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    sb = e.sum()
    if sb < 1e-300 or not np.isfinite(sb):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / sb
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    # G_i = 2*h[i] - w[2i]*a[i]
    two_i = 2 * np.arange(n)
    G = 2.0 * h - w[two_i] * a
    W_grad = float(np.dot(a, G))
    grad_z = a * (G - W_grad)
    return 2.0 * n * grad_z

def _subgrad_a(a, tol_frac=1e-4):
    """Compute subgradient of score wrt a (in a-space, not z-space)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    two_i = 2 * np.arange(n)
    G = 2.0 * h - w[two_i] * a
    return G

def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    # Find t such that sum(max(v - t, 0)) = S
    v_sorted = np.sort(v)[::-1]
    cv = np.cumsum(v_sorted)
    rho = np.where(v_sorted > 0)[0]
    if len(rho) == 0:
        return np.full(n, S/n)
    # b(rho) = (v[rho] - S/rho) / (1 - 1/rho) ... 
    # Standard algorithm:
    # t = (sum(v[rho:rho+1]) - S) / (rho+1)
    # Check if v[t_index] > t >= v[t_index+1]
    cs = np.cumsum(v_sorted)
    ss = (cs - S) / np.arange(1, n+1)
    # Find the largest i such that v_sorted[i-1] > ss[i-1]
    cond = v_sorted > ss
    # Find where cond changes from True to False
    # Actually, we want the largest i (1-indexed) such that v_sorted[i-1] > ss[i-1]
    idx = np.where(cond)[0]
    if len(idx) == 0:
        t = 0.0
    else:
        i = idx[-1] + 1  # 1-indexed
        t = (float(cs[i-1]) - S) / i
    result = np.maximum(v - t, 0.0)
    # Fix sum to be exactly S
    result = result * (S / result.sum()) if result.sum() > 0 else np.full(n, S/n)
    return result

def _project_simplex_unit(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    return _project_simplex(v, 1.0)

def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

def _lbfgs_continuation(a_start, deadline, min_remaining=60.0):
    """Run L-BFGS with decreasing beta schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    # Finer and longer beta schedule
    betas = max_c * np.array([0.1, 0.05, 0.025, 0.0125, 0.00625,
                              0.003125, 0.0015625, 0.00078125,
                              0.000390625, 0.0001953125,
                              0.00009765625, 0.000048828125,
                              0.0000244140625, 0.00001220703125])
    
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break
        # Estimate iterations based on remaining time
        # Each iteration takes roughly 0.5ms * (n/2048)^1.5 for n=2048
        est_time_per_iter = 0.0005 * (n / 2048.0) ** 1.5
        max_iter = min(10000, max(200, int(remaining / (est_time_per_iter * 15))))
        
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc
    
    return current_a

def _nesterov_subgrad(a_init, deadline, tol_frac=1e-4, max_steps=20000):
    """Subgradient method with Nesterov acceleration in a-space."""
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros_like(a)  # velocity for Nesterov
    best_a = a.copy()
    best_sc = _score(a)
    S = a.sum()
    lr = 0.01 / np.sqrt(n)
    gamma = 1.0  # Nesterov momentum
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        a_n = _project_simplex(a + gamma * v, S)
        g = _subgrad_a(a_n, tol_frac)
        g = g - g.mean()  # Remove mean (tangent to simplex)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        step = lr / (1.0 + 0.01 * i)
        v = (gamma * (a - a_n)) + step * g
        # Actually, standard Nesterov:
        # x_{k+1} = x_k - step * grad(x_k + gamma*(x_k - x_{k-1}))
        # Let me use a simpler version:
        # v = gamma*v - step*g
        # a_new = project(a + v)
        
        # Simpler: just use regular gradient with decreasing step
        a_try = _project_simplex(a - step * g, S)
        sc_try = _score(a_try)
        
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            v = a_try - a
            a = a_try
        else:
            step *= 0.9
            if v is not None and np.linalg.norm(v) > 0:
                # Take a small step in the Nesterov direction
                a_try2 = _project_simplex(a + 0.5 * v - 0.5 * step * g, S)
                sc_try2 = _score(a_try2)
                if sc_try2 < best_sc:
                    best_sc = sc_try2
                    best_a = a_try2.copy()
                    a = a_try2
                v *= 0.8
    
    return best_a, best_sc

def _perturb(a, rng, method='gauss', magnitude=0.05):
    """Perturb the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    if method == 'gauss':
        z += magnitude * rng.standard_normal(n)
    elif method == 'fourier':
        t = np.arange(n)
        n_modes = 3
        for _ in range(n_modes):
            freq = int(rng.integers(1, max(2, n // 4)))
            amp = magnitude * 0.5 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif method == 'block':
        n_blocks = 4
        for _ in range(n_blocks):
            start = int(rng.integers(0, n - 10))
            length = int(rng.integers(10, max(11, n // 10)))
            z[start:start+length] += rng.uniform(-magnitude, magnitude)
    elif method == 'endpoint':
        # Perturb the endpoints more
        t = np.arange(n)
        weight = np.ones(n)
        weight[:n//4] *= 3
        weight[3*n//4:] *= 3
        z += magnitude * rng.standard_normal(n) * weight / weight.mean()
    
    z = z - z.mean()
    return _softmax(z)

def _interpolate(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()
```
