    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 0.01:
        return np.inf

    M = float(c.max())

"""
Top-K adaptive gradient descent with spectral flattening and multi-scale n exploration
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive K schedule, backtracking steps, and diverse restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(1, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    M = float(c.max())
    return float(2.0 * n * M / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_topk(a, c, K):
    n = len(a)
    K = min(K, 2 * n - 1)
    top_k_idx = np.argpartition(c, -K)[-K:]
    grad = np.zeros(n)
    for k in top_k_idx:
        i_start = max(0, int(k) - n + 1)
        i_end = min(int(k), n - 1)
        start_idx = int(k) - i_start
        end_idx = int(k) - i_end - 1
        if end_idx < 0:
            seg = a[start_idx::-1]
        else:
            seg = a[start_idx:end_idx:-1]
        grad[i_start:i_end + 1] += seg
    grad *= 2.0 / K
    return grad, top_k_idx


def _topk_descent(a_init, deadline, K_start=5, K_end=200, K_ramp=300,
                  lr_init=0.05, lr_max=0.5, patience=150):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)

    lr = lr_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1
        frac = min(1.0, iter_count / K_ramp)
        K = max(3, int(K_start + (K_end - K_start) * frac * frac))

        c = _full_convolve(a, a)
        current_max = float(c.max())

        grad, top_k = _compute_grad_topk(a, c, K)

        # Try several step sizes (backtracking)
        a_best = a
        obj_best = current_max
        steps = [lr, lr * 0.5, lr * 0.25, lr * 0.1]
        for st in steps:
            a_cand = _proj_simplex(a - st * grad)
            c_cand = _full_convolve(a_cand, a_cand)
            obj_cand = float(c_cand.max())
            if obj_cand < obj_best:
                obj_best = obj_cand
                a_best = a_cand
            if obj_cand < current_max * 0.999:
                break

        if obj_best < current_max - 1e-12:
            a = a_best
            lr = min(lr * 1.03, lr_max)
            no_improve = 0
        else:
            lr = max(lr * 0.92, 1e-5)
            no_improve += 1

        if iter_count % 15 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if no_improve > patience:
            break

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score, iter_count

        grad = np.zeros(n)
        for k in top_k_idx:
            i_start = max(0, int(k) - n + 1)
            i_end = min(int(k), n - 1)
            start_idx = int(k) - i_start
            end_idx = int(k) - i_end - 1
            if end_idx < 0:
                seg = a[start_idx::-1]
            else:
                seg = a[start_idx:end_idx:-1]
            Ba = np.zeros(n)
            Ba[i_start:i_end + 1] = seg
            grad += a * (Ba - c[k])
        grad *= 2.0 / K

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)

        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)

        if update_norm > 1.0:

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
