"""
Hybrid optimizer combining LP-based iteration (cutting-plane style),
FISTA with softmax-smoothed max, Adam log-space refinement, and coordinate
descent. Uses FFT-based gradient computation for efficiency. Targets
sub-1.5027 score for non-negative step functions of length ~4096.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smoothed_fista(y, w):
    """Gradient of softmax-smoothed max for FISTA. w is the softmax weight on conv values."""
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _grad_exact_subgrad(f, peak_idx):
    """Exact subgradient of max(convolve(f,f)) at peak_idx."""
    n = len(f)
    k = int(peak_idx)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    for i in range(lo, hi + 1):
        j = k - i
        if j >= 0 and j < n:
            if i == j:
                grad[i] = f[i]
            else:
                grad[i] = 2.0 * f[j]
    return grad


def _lp_solve(f, M, n, max_constraints=None):
    """
    Solve: max sum(g) s.t. convolve(f,g)_k <= M for all k, g >= 0.
    Returns g normalized to sum 1, or None if infeasible.
    """
    if max_constraints is None:
        max_constraints = 2 * n - 1

    # Build constraint matrix rows (only central ones for efficiency)
    # convolve(f, g)_k = sum_j f_j * g_{k-j} for valid j, k-j
    # Row k: coefficients are f_j where 0 <= k-j < n, i.e., j in [max(0,k-n+1), min(n-1,k)]
    
    # Select which rows to include
    all_k = np.arange(2 * n - 1)
    # Prioritize central rows (where peaks are likely)
    center = n - 1
    dist = np.abs(all_k - center)
    row_order = np.argsort(dist)
    
    use_rows = row_order[:max_constraints]
    use_rows = np.sort(use_rows)

    # Build sparse constraint matrix
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for k in use_rows:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])

    A = coo_matrix((vals, (rows, cols)), shape=(len(use_rows), n)).tocsc()
    b_constraint = np.full(len(use_rows), M)

    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A, b_ub=b_constraint, bounds=[(0, None)] * n,
                  method='highs')

    if not res.success:
        return None

    g = res.x
    s = float(g.sum())
    if s < 1e-15:
        return None
    return g / s


def _lp_iteration(f, M, n, t, max_constraints=None):
    """One LP-based iteration: solve LP, update f."""
    g = _lp_solve(f, M, n, max_constraints)
    if g is None:
        return f, M
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f, M
    f_new = f_new / s
    c_new = _convolve(f_new, f_new)
    M_new = float(c_new.max())
    return f_new, M_new

    n = len(f_init)
    a_curr = f_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    if L_override is not None:
        L = L_override
    else:
        L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed_fista(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _fista_subgrad(f_init, deadline, max_iter):
    """FISTA with exact subgradient at the max point (non-smooth, uses adaptive step)."""
    n = len(f_init)
    a_curr = f_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = 0.01
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

        c_y = _convolve(y, y)
        peak_idx = int(np.argmax(c_y))
        grad = _grad_exact_subgrad(y, peak_idx)

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
            else:
                step = step * 0.8
        else:
            step = min(step * 1.05, 0.1)

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=500, rng=None):
    """Coordinate descent with adaptive step sizes."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    if rng is None:
        rng = np.random.default_rng(999)

    c = _convolve(a, a)
    c_max = float(c.max())

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 2048:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            a[j] = 0.0
            s = a.sum()
            if s < 1e-30:
                a[j] = a_init[j]
                continue
            a_norm = a / s
            c_chk = _convolve(a_norm, a_norm)
            sc = float(c_chk.max())
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_norm.copy()
                improved = True
                # Also try increasing from 0
                for val in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1]:
                    a[j] = val
                    s2 = a.sum()
                    a_try = a / s2
                    c_try = _convolve(a_try, a_try)
                    sc_try = float(c_try.max())
                    if sc_try < best_sc:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        a[j] = val
                        break
            else:
                a[j] = a_init[j]

        a = best_a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _coord_descent_multiplicative(a_init, deadline, max_rounds=300, rng=None):
    """Multiplicative coordinate descent (like exponent gradient)."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    if rng is None:
        rng = np.random.default_rng(777)

    c = _convolve(a, a)
    c_max = float(c.max())
    peak_indices = np.where(np.isclose(c, c_max, rtol=0.01))[0]

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Focus on indices that contribute to the peak
        contrib_indices = set()
        for k in peak_indices[:20]:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            contrib_indices.update(range(lo, hi))
        contrib_list = list(contrib_indices)
        if len(contrib_list) > 512:
            contrib_list = list(rng.choice(contrib_list, 512, replace=False))
        else:
            rng.shuffle(contrib_list)

        improved = False
        for j in contrib_list:
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-10:
                continue
            # Multiplicative update: a_j *= (a_j / c_j)^(eta) approximately
            # Try a few shrink factors
            for factor in [0.95, 0.9, 0.8, 0.7]:
                a[j] = old * factor
                s = a.sum()
                if s < 1e-30:
                    a[j] = old
                    continue
                a_n = a / s
                c_n = _convolve(a_n, a_n)
                sc = float(c_n.max())
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_n.copy()
                    improved = True
                    break
                else:
                    a[j] = old
                    break

        if not improved and r > 20:
            break
    return best_a, best_sc


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()
    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")

    # === Phase 1: LP-based iteration ===
    print("Phase 1: LP-based iteration")
    phase1_end = min(deadline, t_start + 600.0)
    f = a_work.copy()
    M = max_c0
    t_lp = 0.3

    # Use reduced constraints initially, expand over time
    n_constraints = 1024
    lp_iter = 0
    lp_stagnant = 0

    while time.monotonic() < phase1_end - 10:
        t_lp_start = time.monotonic()
        g = _lp_solve(f, M, n_main, max_constraints=n_constraints)
        t_lp_solve = time.monotonic() - t_lp_start

        if g is None:
            print(f"  LP infeasible at iter {lp_iter}, switching to FISTA")
            break

        # Update with line search
        t_try = t_lp
        f_try = (1.0 - t_try) * f + t_try * g
        f_try = np.maximum(f_try, 0.0)
        s_try = f_try.sum()
        if s_try < 1e-15:
            break
        f_try = f_try / s_try
        c_try = _convolve(f_try, f_try)
        M_try = float(c_try.max())

        if M_try <= M + 1e-14:
            f = f_try
            M = M_try
            lp_stagnant = 0
        else:
            # Try smaller step
            t_try = t_try * 0.5
            f_try = (1.0 - t_try) * f + t_try * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                break
            f_try = f_try / s_try
            c_try = _convolve(f_try, f_try)
            M_try = float(c_try.max())
            if M_try <= M + 1e-14:
                f = f_try
                M = M_try
                t_lp = t_try
            else:
                lp_stagnant += 1
                t_lp = max(0.05, t_lp * 0.8)

        # Increase constraints over time
        if lp_iter % 5 == 0:
            n_constraints = min(2 * n_main - 1, n_constraints + 512)

        sc = float(2.0 * n_main * M / 1.0)
        if sc < best_score:
            best_score = sc
            best_seq = f.copy()

        if lp_iter % 10 == 0:
            print(f"  LP iter {lp_iter}: M={M:.8f}, score={sc:.10f}, "
                  f"t_lp={t_lp:.3f}, constraints={n_constraints}, "
                  f"solve_time={t_lp_solve:.2f}s, t={time.monotonic()-t_start:.1f}s")

        if lp_stagnant > 10:
            print(f"  LP stagnant at iter {lp_iter}, switching to FISTA")
            break

        lp_iter += 1

        # If LP is too slow, switch early
        if t_lp_solve > 60:
            print(f"  LP too slow ({t_lp_solve:.1f}s), switching to FISTA")
            break

    print(f"End Phase 1: LP iters={lp_iter}, score={best_score:.10f}")
    a_work = best_seq.copy() if best_seq is not None else f

    # === Phase 2: FISTA with softmax smoothing ===
    print("Phase 2: FISTA softmax continuation")
    phase2_end = min(deadline, t_start + 1100.0)
    a_curr = a_work.copy()
    a_curr = a_curr / a_curr.sum()

    c_curr = _convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas = []
    beta = 0.02 * max_cc
    while beta > 0.0001 * max_cc:
        betas.append(beta)
        beta *= 0.5
    betas.append(0.0001 * max_cc)
    if len(betas) > 20:
        betas = betas[::max(1, len(betas) // 20)] + [0.0001 * max_cc]

    for i, beta in enumerate(betas):
        if time.monotonic() > phase2_end - 5:
            break
        per_time = min(30.0, phase2_end - time.monotonic() - 3)
        if per_time < 2:
            break
        max_it = max(2000, int(per_time * 5000))
        a_curr, sc = _fista_softmax(a_curr, beta, time.monotonic() + per_time, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())

    print(f"End Phase 2: score={best_score:.10f}")
    a_work = best_seq.copy()

    # === Phase 3: Adam log-space fine optimization ===
    print("Phase 3: Adam log-space fine")
    phase3_end = min(deadline, t_start + 1450.0)
    c_curr = _convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    betas_fine = []
    beta = 0.0005 * max_cc
    while beta > 1e-6 * max_cc:
        betas_fine.append(beta)
        beta *= 0.4
    betas_fine.append(1e-6 * max_cc)

    per_time = phase3_end - time.monotonic() - 10
    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.008, max_iter_total=5000000
        )
        sc_opt_full = _score(a_opt)
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()

    a_work = best_seq.copy()

    # === Phase 4: Multi-restart with diverse perturbations ===
    print("Phase 4: Multi-restart")
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end - 30:
        n_restarts = 4
        per_restart = (phase4_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            if i % 2 == 0:
                p_start = _perturb(a_work, 0.01 + 0.005 * i, rng)
            else:
                sz = n_main // 1024
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0001 * max_cs, 0.00005 * max_cs, 0.00001 * max_cs,
                  0.000005 * max_cs]
            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart * 0.6),
                bs, lr_init=0.005, max_iter_total=100000
            )
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()

    # === Phase 5: Coordinate descent polish ===
    print("Phase 5: Coordinate descent polish")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 60, time.monotonic() + 80)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=300, rng=rng)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

        # Multiplicative CD
        if time.monotonic() < deadline - 40:
            a_mc = best_seq.copy() / best_seq.sum()
            mc_deadline = min(deadline - 30, time.monotonic() + 40)
            a_mc, sc_mc = _coord_descent_multiplicative(a_mc, mc_deadline, max_rounds=200, rng=rng)
            sc_mc_full = _score(a_mc)
            print(f"  Mult CD: score={sc_mc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_mc_full < best_score:
                best_score = sc_mc_full
                best_seq = a_mc.copy()

    # === Phase 6: Try n=8192 ===
    print("Phase 6: n=8192 exploration")
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end - 30 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase6_end - 20:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            beta_big = 0.005 * max_cb
            per_t = min(25.0, phase6_end - time.monotonic() - 10)
            if per_t > 5:
                a_big_opt, sc_big_opt = _fista_softmax(
                    a_big, beta_big, time.monotonic() + per_t, 50000
                )
                a_big_opt = a_big_opt / a_big_opt.sum()
                sc_big_opt_full = _score(a_big_opt)
                if sc_big_opt_full < best_score:
                    best_score = sc_big_opt_full
                    best_seq = a_big_opt.copy()
                    print(f"  FISTA n=8192: score={sc_big_opt_full:.10f}")

    # === Phase 7: Final ultra-fine Adam polish ===
    print("Phase 7: Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.000005 * max_cp, 0.000002 * max_cp, 0.000001 * max_cp,
                     0.0000005 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.001, max_iter_total=1000000
            )
            sc_pol_full = _score(a_pol)
            print(f"  Final Adam: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

        # One more coordinate descent pass
        if time.monotonic() < deadline - 5:
            a_final_cd = best_seq.copy() / best_seq.sum()
            a_final_cd, _ = _coord_descent(a_final_cd, time.monotonic() + 10, max_rounds=50, rng=rng)
            sc_cd = _score(a_final_cd)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_final_cd.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        else:
            final = final / s  # Normalize: score is scale-invariant
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
