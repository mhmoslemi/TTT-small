Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses multiplicative weights, subgradient methods with softmax smoothing, Adam in log-space,
structured perturbations, and coordinate descent. Explores multiple sequence lengths.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.2:
                update = update * (0.2 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    factors = [0.95, 0.97, 0.99, 1.01, 1.03, 1.05]

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in factors:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    old_val = new_val
                else:
                    a[j] = old_val
                    break
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _mult_weight_update(a, c, beta, alpha):
    """Multiplicative weights: reduce entries that contribute to high convolution peaks."""
    n = len(a)
    cm = float(c.max())
    c_norm = (c - cm) / beta
    np.clip(c_norm, -500, 500, out=c_norm)
    w = np.exp(c_norm)
    ws = float(w.sum())
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws

    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    a_new = a * (1.0 - alpha * (h - W) / (a.sum() + 1e-30))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _structured_init(n, rng):
    """Generate structured initial sequences based on known low-autocorrelation ideas."""
    method = rng.integers(0, 5)
    if method == 0:
        # Cosine window
        t = np.linspace(0, np.pi, n)
        a = 0.5 + 0.5 * np.cos(t)
    elif method == 1:
        # Raised cosine
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif method == 2:
        # Bimodal
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.2 + 0.1 * rng.random()
        a = np.exp(-((t - center) ** 2) / (2 * sigma ** 2))
    elif method == 3:
        # Periodic with random phase
        freq = int(rng.integers(2, max(3, n // 100)))
        t = np.linspace(0, freq * 2 * np.pi, n)
        a = 1.0 + 0.5 * np.cos(t)
    else:
        # sinc-like
        t = np.linspace(-3, 3, n)
        a = np.sinc(t / np.pi) ** 2
        a = np.maximum(a, 1e-6)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        a = np.ones(n)
        s = n
    return a / s
