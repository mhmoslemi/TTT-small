"""
Multi-phase optimizer with adaptive-step FISTA, refined Adam log-space,
larger-n exploration (up to 16384), simulated annealing, and 2-point
coordinate descent. Targets sub-1.5027 score for non-negative step functions.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

        best_a = a_final.copy()
    return best_a, best_score


def _simulated_annealing(a_init, deadline, n_iters=50000, t_init=0.01, t_final=0.0001):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    current_score = best_score

    rng = np.random.default_rng(777)
    iters_per_sec = 500  # rough estimate for timing

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        progress = it / n_iters
        temperature = t_init * (t_final / t_init) ** progress

        # Perturb a random block
        block_size = max(1, n // 256)
        start = rng.integers(0, n - block_size)
        old_block = z[start:start + block_size].copy()
        z[start:start + block_size] += temperature * 0.1 * rng.standard_normal(block_size)
        z = z - z.mean()

        a = _softmax(z)
        sc = _score_norm(a)

        if sc < current_score or (np.isfinite(sc) and rng.random() < np.exp(-(sc - current_score) / temperature)):
            current_score = sc
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            z[start:start + block_size] = old_block
            z = z - z.mean()

        if it % 5000 == 0:
            a_chk = _softmax(z)
            sc_chk = _score_norm(a_chk)
            if sc_chk < best_score:
                best_score = sc_chk
                best_a = a_chk.copy()

    return best_a, best_score


def _coord_descent_2pt(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            pairs = rng.integers(0, n, size=(512, 2))
        else:
            pairs = rng.integers(0, n, size=(n // 2, 2))

        improved = False
        for i, j in pairs:
            if time.monotonic() > deadline:
                break
            if i == j:
                continue
            old_i, old_j = a[i], a[j]
            # Try moving mass from i to j and vice versa
            for frac in [0.05, 0.1, 0.2, -0.05, -0.1, -0.2]:
                delta = frac * min(old_i, old_j)
                a[i] = old_i - delta
                a[j] = old_j + delta
                if a[i] < 0 or a[j] < 0:
                    a[i], a[j] = old_i, old_j
                    continue
                a[j] = a[j] - 0  # keep sum same (delta moved)
                # Actually: a[i] decreases by delta, a[j] increases by delta, sum preserved
                s = a.sum()
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    old_i, old_j = a[i], a[j]
                else:
                    a[i], a[j] = old_i, old_j

        if not improved and r > 10:
            break

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
