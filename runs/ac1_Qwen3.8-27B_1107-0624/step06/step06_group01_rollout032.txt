Active-set and frequency-aware optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set gradient descent, mass-transfer moves, frequency-domain equalization,

import numpy as np
import time

_next_pow2_cache = {}

def _next_pow2(n):
    if n in _next_pow2_cache:
        return _next_pow2_cache[n]
    k = 1
    while k < n:
        k <<= 1
    result = max(2, k)
    _next_pow2_cache[n] = result
    return result

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) with sum(a)=1 assumed."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    """Compute the full score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

def _compute_grad_active(a, w):
    """Compute gradient of sum_k w_k * conv(a,a)_k w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _compute_grad_logspace(a, c, w, n):
    """Compute gradient in log-space (for softmax parameterization)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _get_active_weights(c, beta):
    """Get softmax weights for the convolution peaks."""
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w

def _fista_active(a_init, beta, deadline, max_iter):
    """FISTA with active-set-focused gradient."""

    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w = _get_active_weights(c_y, beta)
        grad = _compute_grad_active(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 100 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:

            max_c_new = float(c_new.max())

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax(z)
            c = _full_convolve(a, a)
            w = _get_active_weights(c, beta)
            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 30 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.4) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Hybrid active-set and softmax optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive active-set FISTA, multi-scale softmax Adam, smart mass-transfer moves,
frequency-domain equalization, and multi-length exploration.

import numpy as np
import time

_next_pow2_cache = {}

def _next_pow2(n):
    if n in _next_pow2_cache:
        return _next_pow2_cache[n]
    k = 1
    while k < n:
        k <<= 1
    result = max(2, k)
    _next_pow2_cache[n] = result
    return result

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _get_soft_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w

def _get_hard_weights(c, top_k=5):
    n_c = len(c)
    w = np.zeros(n_c)
    idx = np.argsort(c)[::-1][:top_k]
    for k in idx:
        w[int(k)] = 1.0 / top_k
    return w

def _fista_hybrid(a_init, beta_soft, top_k, deadline, max_iter):
    """FISTA alternating between softmax and hard active-set gradients."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta_soft
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)

        # Alternate between soft and hard weights
        if it < 50:
            w = _get_soft_weights(c_y, beta_soft)
        else:
            # Blend: 70% hard active set, 30% soft
            w_soft = _get_soft_weights(c_y, beta_soft * 0.1)
            w_hard = _get_hard_weights(c_y, top_k)
            w = 0.7 * w_hard + 0.3 * w_soft

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = max(4.0 * n * max_c_new / beta_soft, 1e-8)
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _adam_hybrid(a_init, deadline, beta_schedule, top_k_schedule, lr_init, max_iter_total):
    """Adam in log-space with hybrid soft/hard active-set gradients."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        top_k = top_k_schedule[min(phase_idx, len(top_k_schedule) - 1)]
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)

            # Hybrid weight: blend soft and hard
            if phase_idx < len(beta_schedule) // 2:
                w = _get_soft_weights(c, beta)
            else:
                w_soft = _get_soft_weights(c, beta * 0.05)
                w_hard = _get_hard_weights(c, top_k)
                w = 0.8 * w_hard + 0.2 * w_soft

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 4000 == 0:
                current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _smart_mass_transfer(a, deadline, max_steps=8000):
    """Targeted mass transfer: move mass from high-gradient to low-gradient positions."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    rng = np.random.default_rng(7)

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c_max = float(c.max())

        # Compute gradient for active set
        w_hard = _get_hard_weights(c, 5)
        grad = _compute_grad(best_a, w_hard)

        # Find high-gradient source and low-gradient destination
        # Source: positions with high gradient (reducing them helps)
        # Destination: positions with low gradient (adding there doesn't hurt)
        grad_sorted = np.argsort(grad)
        # Try moving from high-grad to low-grad
        for _ in range(3):
            src_candidates = grad_sorted[::-1][:max(1, n // 10)]
            dst_candidates = grad_sorted[:max(1, n // 10)]
            i = int(rng.choice(src_candidates))
            j = int(rng.choice(dst_candidates))
            if i == j:
                continue

            for frac in [0.005, 0.01, 0.02]:
                delta = frac * best_a[i]
                if delta < 1e-15:
                    continue
                a_try = best_a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_try.copy()
                    break

    return best_a, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.4) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            max_it = max(2000, int(phase_time * 3000))
            a_work, sc = _fista_hybrid(a_work, beta, top_k, phase_deadline, max_it)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_work.copy()
            print(f"  FISTA phase {i} (beta={beta:.1e}, top_k={top_k}): score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Adam hybrid (log-space)
    if time.monotonic() < deadline - 250:
        print(f"\n=== Phase 2: Adam hybrid (n={n_best}) ===")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())

        beta_schedule = [
            0.002 * max_cc, 0.001 * max_cc, 0.0005 * max_cc,
            0.0002 * max_cc, 0.0001 * max_cc, 0.00005 * max_cc
        ]
        top_k_schedule = [10, 7, 5, 4, 3, 2]

        adam_deadline = min(deadline - 200, time.monotonic() + 250)
        a_opt, sc_opt = _adam_hybrid(
            a_work, adam_deadline,
            beta_schedule, top_k_schedule,
            lr_init=0.015, max_iter_total=150000
        )
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 3: Smart mass transfer
    if time.monotonic() < deadline - 180:
        print(f"\n=== Phase 3: Smart mass transfer ===")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        mt_deadline = min(deadline - 140, time.monotonic() + 150)
        a_mt, sc_mt = _smart_mass_transfer(a_work, mt_deadline, max_steps=15000)
        sc_mt_full = _score(a_mt)
        if sc_mt_full < best_score:
            best_score = sc_mt_full
            best_seq = a_mt.copy()
            print(f"  Mass transfer: score={sc_mt_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Frequency perturbation
    if time.monotonic() < deadline - 130:
        print(f"\n=== Phase 4: Frequency perturbation ===")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        fp_deadline = min(deadline - 100, time.monotonic() + 80)
        a_fp, sc_fp = _freq_perturb_optimize(a_work, fp_deadline, n_iters=500)
        sc_fp_full = _score(a_fp)
        if sc_fp_full < best_score:
            best_score = sc_fp_full
            best_seq = a_fp.copy()
            print(f"  Freq perturb: score={sc_fp_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Multi-restart with diverse perturbations
    if time.monotonic() < deadline - 120:
        print(f"\n=== Phase 5: Multi-restart ===")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        n_best = len(a_work)

        n_restarts = 4
        per_restart_time = (deadline - 100 - time.monotonic()) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > deadline - 90:
                break

            if i == 0:
                p_start = _perturb(a_work, 0.01, rng)
            elif i == 1:
                # Roll
                sz = max(1, n_best // 2048)
                shift = int(rng.integers(1, max(2, sz * 2)))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif i == 2:
                # Frequency perturbation
                N = _next_pow2(2 * n_best)
                A = np.fft.rfft(a_work, N)
                phase = rng.standard_normal(len(A)) * 0.03
                A_new = A * np.exp(1j * phase)
                a_new = np.fft.irfft(A_new, N)[:n_best]
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                p_start = a_new / s if s > 1e-10 else _perturb(a_work, 0.02, rng)
            else:
                # Blend with asymmetric profile
                alt = _make_profile(n_best, "asymmetric")
                p_start = 0.85 * a_work + 0.15 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0003 * max_cs, 0.0001 * max_cs, 0.00005 * max_cs]
            tks = [5, 3, 2]

            p_best, sc_best = _adam_hybrid(
                p_start, time.monotonic() + max(10.0, per_restart_time * 0.5),
                bs, tks,
                lr_init=0.008, max_iter_total=60000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
            print(f"  Restart {i}: score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Multi-length exploration
    if time.monotonic() < deadline - 100:
        print(f"\n=== Phase 6: Multi-length ===")
        lengths_to_try = []
        if n_best <= 4096:
            lengths_to_try = [8192, 2048]
        elif n_best == 8192:
            lengths_to_try = [4096, 16384]
        else:
            lengths_to_try = [n_best // 2, n_best * 2]

        time_per_len = min(60.0, (deadline - 80 - time.monotonic()) / max(1, len(lengths_to_try)))

        for n_try in lengths_to_try:
            if time.monotonic() > deadline - 70:
                break
            if n_try < 256 or n_try > 16384:
                continue

            print(f"  Trying n={n_try}...")
            if parent is not None:
                a_try = _interp_profile(parent, n_try)
            else:
                a_try = _make_profile(n_try, "uniform")

            c_try = _full_convolve(a_try, a_try)
            max_try = float(c_try.max())

            # Quick FISTA
            betas_try = [0.01 * max_try, 0.005 * max_try, 0.002 * max_try]
            top_ks_try = [8, 5, 3]
            for b, tk in zip(betas_try, top_ks_try):
                if time.monotonic() > deadline - 60:
                    break
                a_try, _ = _fista_hybrid(
                    a_try, b, tk,
                    time.monotonic() + time_per_len / 3, 1500
                )
                a_try = a_try / a_try.sum()

            # Quick Adam
            if time.monotonic() < deadline - 50:
                c_try = _full_convolve(a_try, a_try)
                max_try = float(c_try.max())
                bs_try = [0.001 * max_try, 0.0005 * max_try]
                tk_try = [4, 2]
                a_opt, _ = _adam_hybrid(
                    a_try, min(deadline - 40, time.monotonic() + 30),
                    bs_try, tk_try,
                    lr_init=0.01, max_iter_total=50000
                )
                sc_try = _score(a_opt)
                if sc_try < best_score:
                    best_score = sc_try
                    best_seq = a_opt.copy()
                    print(f"    n={n_try}: score={sc_try:.10f} (NEW BEST)")
                else:
                    print(f"    n={n_try}: score={sc_try:.10f}")

    # Phase 7: Multi-scale polish
    if time.monotonic() < deadline - 40:
        print(f"\n=== Phase 7: Multi-scale polish ===")
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        ms_deadline = min(deadline - 20, time.monotonic() + 40)
        a_ms, sc_ms = _multi_scale_polish(a_work, ms_deadline)
        sc_ms_full = _score(a_ms)
        if sc_ms_full < best_score:
            best_score = sc_ms_full
            best_seq = a_ms.copy()
            print(f"  Multi-scale: score={sc_ms_full:.10f}")

    # Phase 8: Ultra-fine final polish
    if best_seq is not None and time.monotonic() < deadline - 8:
        print(f"\n=== Phase 8: Ultra-fine final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [3e-6 * max_cp, 1e-6 * max_cp, 5e-7 * max_cp]
        tks_ult = [2, 1, 1]
        per_t = deadline - time.monotonic() - 5

        if per_t > 10:
            a_pol, sc_pol = _adam_hybrid(
                a_pol, time.monotonic() + per_t,
                betas_ult, tks_ult,
                lr_init=0.001, max_iter_total=80000
            )
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Ultra-fine: score={sc_pol_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
