```python
    c[k] = sum_i a[i] * a[k-i] (valid indices)
    dc[k]/da[j] = a[k-j] + a[j-k] (with 0 for out-of-range)

    da_j/dz_m = -a_j * a_m  for m != j

    # = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]
    # First term: (w * a)[j] (circular conv, but we need linear)
    # Actually: sum_k w[k] * a[k-j] = (w_rev * a)[j] where w_rev[k] = w[-k]
    # This is getting complicated. Let me just do it directly for small arrays or use FFT.
    
    # Using FFT: grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # Let's define: for valid k and j, a[k-j] is a[k-j] if 0<=k-j<n, else 0
    # sum_k w[k] * a[k-j] = (w * a)[j] where * is cross-correlation
    # In numpy: np.correlate(w, a, mode='full') gives sum_k w[k] * a[k + (n_w-1) - j]
    # This is getting confusing. Let me use a different approach.
    
    # Direct approach using the structure:
    # c[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
    # dc/d a[j] for a specific k:
    #   = a[k-j] (if 0 <= k-j < n) + a[j-k] (if 0 <= j-k < n)
    #   = a[k-j] * (0 <= k-j < n) + a[j-k] * (0 <= j-k < n)

    # = sum_k w[k] * a[k-j] * (0<=k-j<n) + sum_k w[k] * a[j-k] * (0<=j-k<n)
    # = (w ⊛ a)[j] + (w ⊛ a_rev)[j]  ... this is cross-correlation
    
    # Let me just use FFT:
    # Define W = w padded to size 2n-1
    # Define A = a padded to size 2n-1
    # (W * A)[j] = sum_k W[k] * A[j-k]  (linear convolution)
    # But we want sum_k W[k] * A[k-j] = (W_rev * A)[j] where W_rev = W[::-1]
    
    # Actually, let's use the fact that:
    # sum_k w[k] * a[k-j] = correlation of w and a at lag j
    # = ifft(fft(w) * conj(fft(a)))[j]  (circular, with zero-padding)
    
    N = _next_pow2(2 * n)
    W = np.zeros(N)
    W[:len(c)] = weights
    A = np.zeros(N)
    A[:n] = a
    
    # cor[j] = sum_k W[k] * A[k-j] (circular correlation)
    # = ifft(FFT(W) * conj(FFT(A)))[j]
    cor = np.fft.irfft(np.fft.rfft(W) * np.conj(np.fft.rfft(A)), N)

    # Actually: sum_k w[k] * a[j-k] = sum_i w[j-i] * a[i] = (w * a)[j] (convolution)
    # Using FFT: ifft(fft(w) * fft(a))[j]
    conv_wa = np.fft.irfft(np.fft.rfft(W) * np.fft.rfft(A), N)
    
    grad_a = grad_a + conv_wa[:n]
    
    return grad_a
```

```python
def _topk_gradient(a, c, k_top):
    """Compute gradient of sum of top-k c[k] values w.r.t. a, then w.r.t. log-space z."""
    n = len(a)
    c_len = len(c)
    
    if k_top >= c_len:
        # Use all with softmax weights
        c_shifted = c - c.max()
        w = np.exp(c_shifted)
        w = w / w.sum()
    else:
        w = np.zeros(c_len)
        idx = np.argpartition(c, -k_top)[-k_top:]
        w[idx] = 1.0 / k_top
    
    # grad_a[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # Using FFT for efficiency
    N = _next_pow2(2 * n)
    W = np.zeros(N)
    W[:c_len] = w
    A = np.zeros(N)
    A[:n] = a
    
    # sum_k w[k] * a[k-j] = circular correlation = ifft(fft(W) * conj(fft(A)))
    FW = np.fft.rfft(W)
    FA = np.fft.rfft(A)
```

`(w*a_flip)[4] = w2*a0 + w3*a1 + w4*a2` → `grad_a[2] = 2*(w2*a0 + w3*a1 + w4*a2)` ✓

So the formula is confirmed: `grad_a[j] = 2 * convolve(w, a_flip)[j + n - 1]`.

```python
def _compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] using FFT."""
    n = len(a)
    a_flip = a[::-1]
    # Linear convolution of w (length 2n-1) and a_flip (length n)
    # Result has length 3n-2
    out_len = len(w) + n - 1  # = 3n-2
    N = _next_pow2(out_len)
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flip, N)
    conv = np.fft.irfft(W * AF, N)[:out_len]
    # Extract grad_a[j] = 2 * conv[j + n - 1] for j = 0..n-1
    return 2.0 * conv[n-1 : 2*n-1]
```

Wait, `j + n - 1` for j=0 gives n-1, and for j=n-1 gives 2n-2. So the slice is `conv[n-1 : 2*n-1]` which has length n. ✓

Now for the log-space gradient:
If `a = softmax(z)`, then the gradient of `f(a)` w.r.t. `z` is:
`grad_z = a * (grad_a - a^T * grad_a)` (where `grad_a` is the gradient w.r.t. `a`, treated as a vector).

This is because `df/dz_j = sum_i (df/da_i) * (da_i/dz_j)` and `da_i/dz_j = a_i * (delta_ij - a_j)`.

So `df/dz_j = grad_a_j * a_j * (1 - a_j) - a_j * sum_i grad_a_i * a_i = a_j * (grad_a_j - sum_i grad_a_i * a_i)`.

Great, so:
```python
def _compute_grad_z(a, w):
    grad_a = _compute_grad_a(a, w)
    return a * (grad_a - np.dot(grad_a, a))
```

```python
"""
Top-k peak gradient optimizer with Adam in log-space for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations: exact top-k subgradient (not full softmax), adaptive phase timing,
structured restarts, and multi-length exploration (n=4096, 8192).
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_a(a, w):
    """Compute grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m] using FFT."""
    n = len(a)
    a_flip = a[::-1]
    out_len = len(w) + n - 1
    N = _next_pow2(out_len)
    W = np.fft.rfft(w, N)
    AF = np.fft.rfft(a_flip, N)
    conv = np.fft.irfft(W * AF, N)[:out_len]
    return 2.0 * conv[n - 1:2 * n - 1]


def _compute_grad_z(a, w):
    """Compute gradient in log-space (z = log a, softmax parameterization)."""
    grad_a = _compute_grad_a(a, w)
    return a * (grad_a - np.dot(grad_a, a))


def _make_weights(c, k_top):
    """Create weights for the top-k peaks of convolution."""
    c_len = len(c)
    if k_top >= c_len:
        c_shift = c - c.max()
        w = np.exp(c_shift)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(c_len)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws
        return w
    idx = np.argpartition(c, -k_top)[-k_top:]
    w = np.zeros(c_len)
    w[idx] = 1.0 / k_top
    return w


def _adam_optimize(a_init, deadline, k_schedule, lr_init, verbose=True):
    """Adam optimization in log-space with top-k peak gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-20))
    z = z - z.mean()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    lr = lr_init

    n_phases = len(k_schedule)
    phase_idx = 0
    no_improve_count = 0
    score_at_last_improve = best_score

    while time.monotonic() < deadline and phase_idx < n_phases:
        k_top = k_schedule[phase_idx]
        remaining = deadline - time.monotonic()
        time_per_phase = remaining / (n_phases - phase_idx)
        max_iter = max(100, int(time_per_phase * 350))

        if verbose:
            print(f"    Phase {phase_idx}: k_top={k_top}, max_iter={max_iter}, lr={lr:.4f}")

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            w = _make_weights(c, k_top)
            grad_z = _compute_grad_z(a, w)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

def _coordinate_descent(a, deadline):
    """Polish with random coordinate descent."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    steps = 0
    stagnation = 0
    while time.monotonic() < deadline and steps < 100000:
        steps += 1
        i = rng.integers(0, n)
        factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]
        improved = False
        for factor in factors:
            a_try = best_a.copy()
            a_try[i] *= factor
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_try.copy()
                    improved = True
                    stagnation = 0
                    break
        if not improved:
            stagnation += 1
        if stagnation > 5000:
            break
    return best_a, best_score


def _make_init(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-40 * (t - 0.3) ** 2) + np.exp(-40 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(99)
        a = rng.random(n) + 0.05
        return a / a.sum()
    elif kind == "plateau":
        a = np.ones(n)
        mid = n // 4
        a[:mid] = 0.5
        a[n - mid:] = 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        print(f"  Coord descent result: score={sc_cd_full:.10f}")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # Final validation and output
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.10f}, n={len(best_seq)}")
    print(f"{'='*60}")

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
        s = 1.0

    # Normalize to ensure sum is reasonable (the evaluator handles this but let's be safe)
    final = final / s * np.sqrt(2 * len(final))

    sc_final = _score(final)
    print(f"Final score (internal): {sc_final:.10f}, n={len(final)}")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in final]
```
