"""
LP-based optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Key innovation: uses the LP relaxation where for fixed f and M=max(f*f),
we solve max sum(g) s.t. ||f*g||_inf <= M, g>=0. This gives a direction
g that, when blended into f, reduces the autocorrelation peak.
Combined with L-BFGS log-sum-exp smoothing, subgradient polishing,
and multi-restart diversification.
"""
import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(a):
    n = len(a)
    cols_lists = []
    rows_lists = []
    vals_lists = []
    for j in range(n):
        r_start = j
        r_end = min(j + n - 1, 2 * n - 2)
        length = r_end - r_start + 1
        cols_lists.append(np.full(length, j))
        rows_lists.append(np.arange(r_start, r_end + 1))
        vals_lists.append(np.full(length, a[j]))
    rows = np.concatenate(rows_lists)
    cols = np.concatenate(cols_lists)
    vals = np.concatenate(vals_lists)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M, solver=None):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    A = _build_conv_matrix(a)
    try:
        if solver is None:
            solvers = ['GUROBI', 'MOSEK', 'ECOS', 'SCS', 'CLARABEL']
        else:
            solvers = [solver]
        for s in solvers:
            try:
                prob = cp.Problem(cp.Maximize(cp.sum(g)), [A @ g <= M])
                prob.solve(solver=s)
                if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                    return np.array(g.value).flatten()
            except Exception:
                continue
    except Exception:
        pass
    return None


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002, 0.00001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        est_iter_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(10000, max(100, int(remaining / (est_iter_time * 15))))
        a_new, _ = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.75
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.95
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _gradient_active(a, c, n, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _subgrad_direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        g = _gradient_active(a, c, n, tol_frac)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        a_try = a - step_size * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)
        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
        cur_sc = _score(a)
        if sc_try < cur_sc - 1e-15:
            a = a_try
            step_size = min(step_size * 1.003, 0.05)
        else:
            stall += 1
            step_size *= 0.9
        step_size = max(step_size, 1e-12)
        if stall > 300:
            break
    return best_a, best_sc


def _lp_iteration(a, deadline, max_iters=50, t_init=0.3, t_decay=0.95,
                  M_shrink=0.99, solver=None):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    t = t_init
    a_cur = a.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())
        if M < 1e-30:
            break
        M_target = M * M_shrink
        g = _lp_solve(a_cur, M_target, solver=solver)
        if g is None:
            M_target = M * 1.0
            g = _lp_solve(a_cur, M_target, solver=solver)
        if g is None:
            break
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_norm = g / g_sum
        a_new = (1.0 - t) * a_cur + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            t *= 0.5
            continue
        a_new = a_new / s
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_sc = sc_new
            best_a = a_new.copy()
            a_cur = a_new
            t_min = 0.01
            if t > t_min:
                t = max(t * t_decay, t_min)
        else:
            t *= 0.7
            if t < 0.005:
                break
        if it % 10 == 0:
            print(f"    LP iter {it}: score={best_sc:.10f}, t={t:.4f}, "
                  f"M={M:.6f}, M_target={M_target:.6f}")
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t**2 / 0.3)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        fade = n // 10
        for i in range(fade):
            a[i] *= (i + 1) / fade
            a[n - 1 - i] *= (i + 1) / fade
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    if best_seq is not None and time.monotonic() < deadline - 40.0:
        for target_n in [3072, 8192]:
            if time.monotonic() > deadline - 35:
                break
            if target_n == n_best:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            if time.monotonic() < deadline - 30:
                a_try = _lbfgs_schedule(a_interp, deadline - 30,
                                        min_time_remaining=25.0)
                try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5.0:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-6
            a_f, _ = _lbfgs_run(best_seq, beta, n, 3000)
            try_update(a_f, f"final_lbfgs_{rnd}")
            # Quick subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 5,
                                            tol_frac=1e-5, max_steps=3000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")
            # Quick LP
            if time.monotonic() < deadline - 5:
                a_l, _ = _lp_iteration(best_seq, deadline - 5, max_iters=5,
                                        t_init=0.1, M_shrink=0.998, solver=None)
                try_update(a_l, f"final_lp_{rnd}")

    # === Phase 10: Active-set LP (constrain only near-peak positions) ===
    print(f"\n=== Phase 10: Active-set LP (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10.0:
        n = len(best_seq)
        c = _full_convolve(best_seq, best_seq)
        c_max = float(c.max())
        # Get top positions
        top_k = max(50, n // 4)
        top_indices = np.argsort(c)[-top_k:]
        M = c_max * 0.999

        g = cp.Variable(n, nonneg=True)
        A = _build_conv_matrix(best_seq)
        # Only constrain top positions
        A_top = A[top_indices, :]
        try:
            for solver in ['GUROBI', 'MOSEK', 'ECOS', 'SCS']:
                try:
                    prob = cp.Problem(cp.Maximize(cp.sum(g)), [A_top @ g <= M])
                    prob.solve(solver=solver)
                    if g.value is not None and prob.status in ('optimal', 'optimal_inaccurate'):
                        g_val = np.array(g.value).flatten()
                        g_sum = float(g_val.sum())
                        if g_sum > 1e-15:
                            g_norm = g_val / g_sum
                            for t_blend in [0.1, 0.2, 0.3, 0.5]:
                                a_new = (1 - t_blend) * best_seq + t_blend * g_norm
                                a_new = np.maximum(a_new, 0.0)
                                a_new = a_new / a_new.sum()
                                try_update(a_new, f"active_lp_t{t_blend}")
                        break
                except Exception:
                    continue
        except Exception:
            pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
