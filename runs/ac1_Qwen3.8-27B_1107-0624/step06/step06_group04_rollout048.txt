"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements over previous: random-phase spectral initialization, two-scale optimization,
slower beta continuation, adaptive learning rate with cosine annealing, and more diverse restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=10000, lr_decay_factor=0.97, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 80 and t > 2000:
                current_lr = max(0.0003, current_lr * 0.9)
                no_improve_count = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_logspace_cosine(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                          grad_clip=1.0):

    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr_min = lr_init * 0.01

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        local_t = 0
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            local_t += 1

            progress = min(1.0, t / max(1, max_iter_total))
            current_lr = lr_min + 0.5 * (lr_init - lr_min) * (1.0 + np.cos(np.pi * progress))

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            grad = _compute_grad_logspace(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=500, deadline=None, soft_clip_eps=0.0):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        avg_mag = mag.mean()
        A_flat = A / (mag + 1e-30) * avg_mag
        a_new = np.fft.irfft(A_flat, N_fft)[:n]
        if soft_clip_eps > 0:
            a_new = np.maximum(a_new, soft_clip_eps)
        else:
            a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _random_phase_init(n, rng, n_trials=20):
    """Generate candidates via random phase + spectral flattening + clipping."""
    best_a = None
    best_score = np.inf

    for trial in range(n_trials):
        # Random phase
        N_fft = _next_pow2(2 * n - 1)
        k_max = N_fft // 2
        phase = rng.standard_normal(k_max)
        # Use a broad spectrum to avoid too-concentrated time signals
        mag_target = np.ones(k_max)
        A = mag_target * np.exp(1j * phase)
        # Make it Hermitian-symmetric for real output
        a_new = np.fft.irfft(A, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s

        # Quick spectral flattening
        a_new, _ = _spectral_flatten(a_new, n_iter=30)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_score = _score_norm(best_a)

    return best_a, best_score


def _two_scale_optimize(a_init, n, deadline):
    """Optimize at n/4 first, then upscale and refine at n."""
    n_small = max(64, n // 4)
    # Downsample by averaging
    chunks = len(a_init) // n_small
    a_small = np.array([a_init[i * chunks: (i + 1) * chunks].mean()
                        for i in range(n_small)])
    a_small = np.maximum(a_small, 0)
    s = a_small.sum()
    if s > 0:
        a_small = a_small / s

    # Quick FISTA at small scale
    c_s = _full_convolve(a_small, a_small)
    max_cs = float(c_s.max())
    beta_s = 0.005 * max_cs
    a_small_opt, _ = _fista_smoothed(a_small, beta_s,
                                     time.monotonic() + min(10.0, (deadline - time.monotonic()) * 0.3),
                                     2000)
    a_small_opt = a_small_opt / a_small_opt.sum()

    # Upscale to n
    a_up = np.zeros(n)
    scale = n // n_small
    for i in range(n_small):
        a_up[i * scale: (i + 1) * scale] = a_small_opt[i]
    a_up = a_up / a_up.sum()

    return a_up


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _perturb_structured(a, rng, n_perturb=16):
    """Perturb only a random subset of positions for more focused exploration."""
    n = len(a)
    a_new = a.copy()
    z = np.log(np.maximum(a_new, 1e-15))
    mask = rng.random(n) < (n_perturb / n) * 5
    z[mask] += 0.05 * rng.standard_normal(mask.sum())
    z = z - z.mean()
    return _softmax_compute(z)


def _subgradient_steps(a_init, n_steps=2000, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        top_k = min(30, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        w_vals = c[top_idx] - cm + 1e-10
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_vals ** 2
        w_vec = w_vec / w_vec.sum()

        grad = _compute_grad_simplex(a, w_vec)
        step_size = 0.3 / (1.0 + 0.005 * it)
        a_new = _proj_simplex(a - step_size * grad)

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            step_size *= 0.9
            a_new = _proj_simplex(a - step_size * grad)
            if _score_norm(a_new) < _score_norm(a):
                a = a_new

    return best_a, best_score


def _combined_spectral_adam(a_init, n, deadline, max_iter=50000):
    """Alternate between spectral flattening and Adam steps."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.005

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c_init = _full_convolve(a_init, a_init)
    max_c = float(c_init.max())
    beta = 0.001 * max_c

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        if it % 500 == 0 and it > 0:
            # Spectral flattening step
            a_cur = _softmax_compute(z)
            a_sf, _ = _spectral_flatten(a_cur, n_iter=10)
            z_tmp = np.log(np.maximum(a_sf, 1e-15))
            z_tmp = z_tmp - z_tmp.mean()
            # Blend
            z = 0.5 * z + 0.5 * z_tmp

        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(0.000001 * max_c, cm * 0.001 / (1.0 + 0.01 * it))
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s_w = float(e.sum())
        if s_w < 1e-300 or not np.isfinite(s_w):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_w

        grad = _compute_grad_logspace(a, c, w, n)
        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if it % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if it % 5000 == 0 and it > 0:
            lr *= 0.95

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 0: Generate diverse candidates via random-phase spectral init ===
    print("Phase 0: Random-phase spectral initialization")
    phase0_end = min(deadline, t_start + 40.0)
    a0_new, sc0_new = _random_phase_init(n_main, rng, n_trials=15)
    if sc0_new / (a0_new.sum() ** 2) < best_score:
        best_score = sc0_new / (a0_new.sum() ** 2)
        a0 = a0_new / a0_new.sum()
        best_seq = a0.copy()
        print(f"  Spectral init improved: score={best_score:.10f}")
    else:
        print(f"  Spectral init not better: {sc0_new/(a0_new.sum()**2):.10f} vs {best_score:.10f}")

    # Quick spectral flatten on current best
    a0, _ = _spectral_flatten(a0, n_iter=50, deadline=phase0_end)
    sc0 = _score(a0)
    if sc0 < best_score:
        best_score = sc0
        best_seq = a0.copy()
        print(f"  After spectral flatten: score={best_score:.10f}")

    # === Phase 1: FISTA with slower beta continuation ===
    print("Phase 1: FISTA beta-continuation")
    a_curr = best_seq.copy()
    a_curr = a_curr / a_curr.sum()
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    phase1_end = min(deadline, t_start + 200.0)
    betas_list = []
    beta = 0.01 * max_cc
    while beta > 0.000002 * max_cc:
        betas_list.append(beta)
        beta *= 0.6  # Slower decay

    for i, b in enumerate(betas_list):
        if time.monotonic() > phase1_end - 5:
            break
        per_time = phase1_end - time.monotonic() - 3
        if per_time < 5:
            break
        max_it = max(2000, int(per_time * 2500))
        phase_timeout = time.monotonic() + min(25.0, per_time)
        a_curr, sc = _fista_smoothed(a_curr, b, phase_timeout, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={b:.8f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: score={best_score:.10f}")

    # === Phase 2: Two-scale + n exploration ===
    print("Phase 2: Two-scale and n exploration")

            b *= 0.6

                                              time.monotonic() + min(15.0, per_t), 15000)

            sc8_full = _score(a_8192_opt)
            if sc8_full < best_score:
                best_score = sc8_full
                best_seq = a_8192_opt.copy()
                print(f"  8192 FISTA beta={b:.8f}: score={sc8_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Adam on 8192

    # === Phase 3: Combined spectral+Adam optimization (new method) ===

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 4: Standard Adam with cosine annealing (long run) ===
    print("Phase 4: Adam cosine annealing (long)")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end - 30:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_fine = []
        b = 0.0003 * max_cw

    # === Phase 5: Multi-restart with diverse strategies ===
    print("Phase 5: Multi-restart diversification")
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end - 30:
        n_restarts = 10
        per_restart = (phase5_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:

                p_start = _perturb(a_work, 0.015 + 0.005 * i, rng)

                p_start = mix * p_rp + (1 - mix) * a_work
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # Antisymmetric perturbation
                p_start = a_work.copy()
                p_start[:n_best // 2] *= (1.0 + 0.03 * rng.standard_normal(n_best // 2))
                p_start[n_best // 2:] *= (1.0 - 0.03 * rng.standard_normal(n_best - n_best // 2))
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_fast = 0.0005 * max_cs
            p_opt, _ = _fista_smoothed(p_start, beta_fast,
                                       time.monotonic() + max(3.0, per_restart * 0.3), 5000)
            p_opt = p_opt / p_opt.sum()

            if time.monotonic() < phase5_end - 15:
                c_p = _full_convolve(p_opt, p_opt)
                max_cp = float(c_p.max())
                betas_r = [0.0003 * max_cp, 0.0001 * max_cp, 0.00005 * max_cp, 0.00002 * max_cp]
                per_t_r = phase5_end - time.monotonic() - 10
                p_opt, sc_r = _adam_logspace(p_opt, time.monotonic() + per_t_r,
                                             betas_r, lr_init=0.006, max_iter_total=50000)
                sc_r_full = _score(p_opt)
                if sc_r_full < best_score:
                    best_score = sc_r_full
                    best_seq = p_opt.copy()
                    n_best = len(best_seq)
                    print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")
                    a_work = best_seq.copy()
                    a_work = a_work / a_work.sum()

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 6: Subgradient polish ===
    print("Phase 6: Subgradient polishing")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end - 10:
        a_sg, sc_sg = _subgradient_steps(a_work, n_steps=3000, deadline=phase6_end)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            n_best = len(best_seq)
            print(f"  Subgradient: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 7: Ultra-fine final polish ===
    print("Phase 7: Ultra-fine final polish")
    if best_seq is not None and time.monotonic() < deadline - 40:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp,
                     0.000001 * max_cp, 0.0000005 * max_cp]
        per_t = deadline - time.monotonic() - 20
        if per_t > 15:
            a_pol, sc_pol = _adam_logspace(a_work, time.monotonic() + per_t,
                                           betas_ult, lr_init=0.002, max_iter_total=200000,
                                           lr_decay_every=20000, lr_decay_factor=0.95)
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                n_best = len(best_seq)
                print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Final spectral flatten attempt
        if time.monotonic() < deadline - 20:
            a_sf, sc_sf = _spectral_flatten(best_seq, n_iter=500,
                                            deadline=min(deadline - 10, time.monotonic() + 30))
            sc_sf_full = _score(a_sf)
            if sc_sf_full < best_score:
                best_score = sc_sf_full
                best_seq = a_sf.copy()
                n_best = len(best_seq)
                print(f"  Final spectral: score={sc_sf_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
            s = final.sum()

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(512) / 512
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
