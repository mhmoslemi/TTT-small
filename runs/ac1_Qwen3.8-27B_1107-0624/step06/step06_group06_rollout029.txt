Coordinate descent + L-BFGS hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: exact coordinate descent on the active set for final polish.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))

    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s
    a_rev = a[::-1]

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

"""
Hybrid L-BFGS + exact coordinate descent optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: coordinate descent on the active convolution support as a final polish,
combined with adaptive log-sum-exp L-BFGS for global structure, multi-n exploration,
and diverse perturbation restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.003):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _coord_desc(a_init, deadline, n_active=40, n_passes=8):
    """
    Exact coordinate descent on the active set.
    Minimizes max(conv(a,a)) with a normalized to sum=1.
    Uses incremental convolution updates for efficiency.
    """
    n = len(a_init)
    a_work = a_init.copy()
    a_work = a_work / a_work.sum()
    conv = _full_convolve(a_work, a_work)
    best_a = a_work.copy()
    best_sc = float(2.0 * n * conv.max())

    mean_a = float(np.mean(a_work))

    for p in range(n_passes):
        if deadline is not None and time.monotonic() > deadline:
            break

        c_max = float(conv.max())

        # Determine active set: top values within 0.5% of max, at least n_active
        threshold = c_max * 0.998
        active_mask = conv >= threshold
        n_act = int(active_mask.sum())
        if n_act < n_active:
            sorted_idx = np.argsort(conv)[::-1]
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[sorted_idx[:n_active]] = True
        elif n_act > 4 * n_active:
            sorted_idx = np.argsort(conv)[::-1]
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[sorted_idx[:n_active]] = True

        active_ks = np.where(active_mask)[0]
        M = len(active_ks)

        # For each coordinate i, we need B, p, q vectors of length M
        # B[m] = conv[k_m] - contribution of a[i] to conv[k_m]
        # p[m] = 2*a[k_m - i] if 0<=k_m-i<n and k_m!=2i
        # q[m] = 1 if k_m == 2i

        for i in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break

            ai = a_work[i]

            # Compute B, p_coeff, q_coeff for this coordinate
            ks = active_ks
            # Contribution of a[i] to conv[k]:
            # k==2i: ai^2
            # 0<=k-i<n and k!=2i: 2*ai*a[k-i]
            # else: 0

            is_2i = (ks == 2 * i)
            kij = ks - i
            valid_kij = (kij >= 0) & (kij < n) & (~is_2i)

            B = conv[ks].copy()
            p_coeff = np.zeros(M)
            q_coeff = np.zeros(M)

            B[is_2i] -= ai * ai
            q_coeff[is_2i] = 1.0

            if np.any(valid_kij):
                a_kij = a_work[kij[valid_kij]]
                B[valid_kij] -= 2.0 * ai * a_kij
                p_coeff[valid_kij] = 2.0 * a_kij

            # Minimize max_m(B + q*t^2 + p*t) over t >= 0
            f0 = float(np.max(B))
            act0 = (B >= f0 - 1e-15)
            fp0 = float(np.max(p_coeff[act0]))

            if fp0 >= -1e-13:
                t_opt = 0.0
            else:
                t_low = 0.0
                t_high = max(2.0 * max(ai, 1e-10), 3.0 * mean_a)

                # Expand upper bound
                for _ in range(25):
                    vals = B + q_coeff * t_high * t_high + p_coeff * t_high
                    am = int(np.argmax(vals))
                    fp = 2.0 * q_coeff[am] * t_high + p_coeff[am]
                    if fp >= 0.0:
                        break
                    t_high *= 2.0
                else:
                    t_opt = t_high
                    t_opt = None  # signal to use t_high

                if t_opt is None:
                    t_opt = t_high
                else:
                    # Bisection
                    for _ in range(55):
                        t_mid = 0.5 * (t_low + t_high)
                        vals = B + q_coeff * t_mid * t_mid + p_coeff * t_mid
                        am = int(np.argmax(vals))
                        fp = 2.0 * q_coeff[am] * t_mid + p_coeff[am]
                        if fp >= 0.0:
                            t_high = t_mid
                        else:
                            t_low = t_mid
                        if t_high - t_low < 1e-15:
                            break
                    t_opt = 0.5 * (t_low + t_high)

            # Update a_work and conv
            delta = t_opt - ai
            if abs(delta) > 1e-18 * max(1.0, abs(ai)):
                a_work[i] = t_opt
                # Update conv: only need to update active_ks for next iterations
                # But for correctness of the full conv, update all affected k
                # k = 2i: change = t_opt^2 - ai^2
                conv[2 * i] += t_opt * t_opt - ai * ai
                # k in [i, i+n-1], k != 2i: change = 2*delta*a[k-i]
                if i > 0:
                    # k from i to 2i-1
                    k_lo = i
                    k_hi = min(2 * i - 1, i + n - 1)
                    if k_lo <= k_hi:
                        conv[k_lo:k_hi + 1] += 2.0 * delta * a_work[k_lo - i:k_hi - i + 1]
                if 2 * i + 1 <= i + n - 1:
                    k_lo = 2 * i + 1
                    k_hi = i + n - 1
                    conv[k_lo:k_hi + 1] += 2.0 * delta * a_work[k_lo - i:k_hi - i + 1]

        # Renormalize
        s = float(a_work.sum())
        if s > 1e-15:
            a_work = a_work / s
            conv = conv / (s * s)
            mean_a = float(np.mean(a_work))

        sc = float(2.0 * n * conv.max())
        if sc < best_sc:
            best_sc = sc
            best_a = a_work.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.005 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(30, int(remaining / (est_time_per_iter * 8))))

        def _fun(z_local):
            return _smoobj(z_local, beta, n)

        def _jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                _fun, z, jac=_jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-11,
                    'maxcor': 40,
                    'maxls': 60,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_subgrad(a, deadline, n_steps=2000, lr_init=0.01, tol_frac=0.002):
    """Subgradient method focusing on active set with adaptive learning rate."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax(z)
        c = _full_convolve(a_cur, a_cur)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Subgradient: average gradient over active set
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a_cur[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a_cur * (h - W)

        # Project to tangent space (sum constraint)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break

        z = z - lr * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.7

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: L-BFGS with log-sum-exp smoothing (global structure)
    Phase 2: Coordinate descent on active set (exact polish)
    Phase 3: Multi-n exploration with L-BFGS + coord desc
    Phase 4: Diverse restarts
    Phase 5: Final coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Main L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = t_start + 0.45 * (budget_s - 25.0)
    a_main = _run_lbfgs(parent, lbfgs_deadline, min_time_remaining=300.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=2000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 2: Coordinate descent (KEY INNOVATION) ===
    print(f"\n[Phase 2] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        cd_deadline = min(deadline - 200, time.monotonic() + 200)
        a_cd, sc_cd = _coord_desc(best_seq, cd_deadline, n_active=50, n_passes=10)
        update_best(a_cd, sc_cd, "coord_desc_1")

        # Second round of coord desc with tighter active set
        if time.monotonic() < deadline - 100:
            cd_deadline2 = min(deadline - 100, time.monotonic() + 100)
            a_cd2, sc_cd2 = _coord_desc(a_cd, cd_deadline2, n_active=20, n_passes=10)
            update_best(a_cd2, sc_cd2, "coord_desc_2")

    # Active set subgradient on the result
    if best_seq is not None and time.monotonic() < deadline - 50:
        as_deadline = min(deadline - 50, time.monotonic() + 60)
        a_as, sc_as = _active_set_subgrad(best_seq, as_deadline, n_steps=1500, lr_init=0.008, tol_frac=0.001)
        update_best(a_as, sc_as, "active_sg")

        # Final coord desc after active SG
        if time.monotonic() < deadline - 30:
            a_cd3, sc_cd3 = _coord_desc(best_seq, min(deadline - 25, time.monotonic() + 50), n_active=30, n_passes=8)
            update_best(a_cd3, sc_cd3, "coord_desc_3")

    # === Phase 3: Multi-n exploration ===
    print(f"\n[Phase 3] Multi-n search (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 60:
            break

        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif n_parent == n_try:
            a_init = parent.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        remaining_for_lbfgs = max(30, (deadline - 60) - (time.monotonic() - t_start))
        est_time = remaining_for_lbfgs * 0.6
        lbfgs_dl = time.monotonic() + est_time
        a_opt = _run_lbfgs(a_init, lbfgs_dl, min_time_remaining=20.0)

        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Coordinate descent on this n
        if sc_opt < best_score + 0.02 and time.monotonic() < deadline - 30:
            cd_dl = min(deadline - 25, time.monotonic() + est_time * 0.4)
            a_cd_n, sc_cd_n = _coord_desc(a_opt, cd_dl, n_active=40, n_passes=6)
            update_best(a_cd_n, sc_cd_n, f"n_{n_try}_cd")

    # === Phase 4: Diverse restarts ===
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_restarts = 4
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.04, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            elif i == 2:
                # Add a smooth bump
                n = len(best_seq)
                t = np.arange(n)
                bump = 0.05 * np.exp(-((t - rng.integers(n // 4, 3 * n // 4)) ** 2) / (2 * (n // 8) ** 2))
                p_start = best_seq + bump
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            else:
                # Blocked perturbation
                n = len(best_seq)
                p_start = best_seq.copy()
                block = n // 4
                j = rng.integers(0, n - block)
                p_start[j:j + block] *= 0.7
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()

            sc_start = _score(p_start)
            # Quick L-BFGS
            if time.monotonic() < deadline - 15:
                a_r = _run_lbfgs(p_start, min(deadline - 10, time.monotonic() + 30), min_time_remaining=5.0)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{i}")
                # Coord desc on result
                if time.monotonic() < deadline - 5:
                    a_r_cd, sc_r_cd = _coord_desc(a_r, min(deadline - 3, time.monotonic() + 15), n_active=30, n_passes=4)
                    update_best(a_r_cd, sc_r_cd, f"restart_{i}_cd")

    # === Phase 5: Final intensive coordinate descent ===
    print(f"\n[Phase 5] Final coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Multiple rounds of coord desc with decreasing active set size
        for n_act in [60, 40, 30, 20, 15, 10]:
            if time.monotonic() > deadline - 2:
                break
            remaining_cd = deadline - 2 - time.monotonic()
            if remaining_cd < 5:
                break
            n_pass = max(2, int(remaining_cd / 10))
            a_fin, sc_fin = _coord_desc(best_seq, min(deadline - 2, time.monotonic() + remaining_cd * 0.8),
                                         n_active=n_act, n_passes=n_pass)
            update_best(a_fin, sc_fin, f"final_cd_{n_act}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
