    n = len(a)

    # p-norm value
    val = np.sum(c ** p) ** (1.0/p)
    # weights for gradient
    if p == 2:

    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    # Estimate Lipschitz constant
    # The gradient is 2 * convolve(w, a_rev)[n-1:2n-1]
    # ||grad||_inf <= 2 * ||w||_inf * ||a||_1 (rough estimate)
    # Better: use the Hessian norm. For p=2, Hessian = 2 * (c^T ⊗ I + ...) 
    # Let's just use backtracking.
    
    # Initial step size
    max_c = float(c_curr.max())
    if p == 2:
        L = 4.0 * n * max_c  # rough estimate
    else:
        L = 4.0 * n * max_c ** p / (np.sum(c_curr ** p) ** 0.5)
    step = 1.0 / L
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)

        if s_y > 1e-30:

        else:
            y = np.ones(n) / n
        
        # Compute gradient at y
        val_y, grad_y = _pnorm_value_and_grad(y, p)
        
        # Gradient step with backtracking
        a_candidate = y - step * grad_y
        # Project onto simplex
