"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: FFT-based convolution, Adam in log-space with adaptive top-K softmax,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _cross_corr_fft(a, w, n_out=None):
    """Compute cross-correlation: h[i] = sum_m a[m] * w[m+i] for i=0..n_out-1."""
    n = len(a)
    if n_out is None:
        n_out = n
    N = _next_pow2(n + n_out)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n_out]
    return h


def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a. Returns 2 * cross_corr(a, w)."""
    n = len(a)
    h = _cross_corr_fft(a, w, n)
    return 2.0 * h


def _topk_softmax_weights(c, beta, k):
    """Compute softmax weights over top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k == 1:
        idx = int(np.argmax(c))
        w = np.zeros(n_c)
        w[idx] = 1.0
        return w
    topk_thresh = np.partition(c, n_c - k)[n_c - k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

    k = max(base_k, min(int(2.0 * close_to_max), n))

            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Coordinate descent polishing with incremental convolution updates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    rng = np.random.default_rng(999)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        c = _full_convolve(a, a)
        improved = 0

        for i in order:
            if time.monotonic() > deadline:
                break

            a_i_old = a[i]

            # Compute the "direction" d where conv changes
            # conv_new[k] = conv_old[k] + (new-old)*2*a[k-i] for k != 2i
            #              = conv_old[k] + (new^2 - old^2) for k = 2i
            d = np.zeros(2 * n - 1)
            # For k from i to i+n-1 (where k-i is in [0, n-1]):
            lo = max(0, i)
            hi = min(2 * n - 2, i + n - 1)
            d[lo:hi + 1] = 2.0 * a[max(0, lo - i):max(0, hi - i) + 1]
            # At k = 2i, the formula is different: change is new^2 - old^2
            # vs 2*(new-old)*a[i] = 2*(new-old)*old
            # So at k=2i: actual change = (new^2-old^2) = (new-old)(new+old)
            # Our formula gives: 2*(new-old)*old
            # Difference: (new-old)(new+old - 2*old) = (new-old)(new-old) = (new-old)^2
            # So: conv_new[2i] = c[2i] + 2*(new-old)*a[i] - (new-old)^2

            # Find top-K positions to check
            idx_top = np.argsort(c)[-top_k_w:]

            # Candidate values for a[i]
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5,
                           a_i_old * 0.3, a_i_old * 2.0, a_i_old * 0.7,
                           a_i_old * 1.2]

            # Critical points from active constraints
            for j in idx_top:
                if d[j] > 1e-15:
                    # F(t) = max_j (c[j] + d[j]*t) / 1 (since sum stays 1 after normalization)
                    # Actually we need to think about normalization...
                    # If we change a[i] to t, the new sum is 1 - a_i_old + t
                    # After renormalization, the conv scales by (1/(1-a_i_old+t))^2
                    # max_j conv_new[j] / (1-a_i_old+t)^2
                    # For constraint j: (c[j] + d[j]*t - (t-a_i_old)^2*delta_{j,2i}) / (1-a_i_old+t)^2
                    # Critical point: d(c[j] + d[j]*t)/dt * (1-a_i_old+t) - 2(c[j]+d[j]*t) = 0
                    # d[j]*(1-a_i_old+t) - 2(c[j] + d[j]*t) = 0
                    # d[j] - d[j]*a_i_old + d[j]*t - 2*c[j] - 2*d[j]*t = 0
                    # d[j]*(1-a_i_old) - d[j]*t - 2*c[j] = 0
                    # t = (d[j]*(1-a_i_old) - 2*c[j]) / d[j]
                    t_star = (1.0 - a_i_old) - 2.0 * c[j] / d[j]
                    if 0 <= t_star <= max(5.0 * a_i_old, 0.5):
                        t_candidates.append(t_star)
                        t_candidates.append(max(0, t_star * 0.8))
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(2.0, tc)) for tc in t_candidates]
            # Deduplicate
            t_candidates = list(set([round(tc, 8) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            S0 = 1.0 - a_i_old
            best_t = a_i_old
            best_f = np.inf

            for tc in t_candidates:
                # Update conv at top positions
                delta = tc - a_i_old
                if abs(delta) < 1e-15:
                    max_c_trial = c.max()
                else:
                    # Only check top positions (approximate)
                    c_top = c[idx_top] + d[idx_top] * delta
                    if 2 * i in idx_top:
                        c_top[np.where(idx_top == 2 * i)[0][0]] -= delta * delta
                    max_c_trial = max(c_top.max(), c.max() - abs(delta) * d.max())
                denom = (S0 + tc) ** 2
                f_val = 2.0 * n * max_c_trial / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                # Update conv incrementally
                delta = best_t * s_a - a_i_old  # Hmm, this is getting complicated
                # Just recompute periodically
                improved += 1

            if improved % 200 == 0:
                c = _full_convolve(a, a)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

        # Recompute conv at end of each pass
        c = _full_convolve(a, a)

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "cheb":
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.3 * np.cos(3 * np.arccos(x)) + 0.1 * np.cos(5 * np.arccos(x))
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _lp_direction(f, M, max_time=10.0):
    """Try to solve the LP to find a good direction. Returns None if too slow."""
    n = len(f)
    try:
        import cvxpy as cp
    except ImportError:
        return None

    t0 = time.monotonic()
    g = cp.Variable(n)
    # Build constraint matrix: (f*g)[k] <= M for all k
    # (f*g)[k] = sum_j f[j]*g[k-j]
    # Using the convolution structure, we can build this efficiently

    # For each k, the constraint is: sum_{j=0}^{n-1} f[j]*g[k-j] <= M
    # where g[m]=0 if m<0 or m>=n
    # This is equivalent to: the convolution matrix C times g <= M*ones

    # Build C as a sparse-ish operation using FFT structure
    # C[k, i] = f[k-i] if 0 <= k-i < n

    # For efficiency, let's just use the dense form for moderate n
    # Actually for n=4096, the matrix is 8191 x 4096 with n^2 = 16.7M nonzeros
    # This might be too large for cvxpy to handle quickly

    # Alternative: use the fact that C = toeplice-like structure
    # Let's try a different approach: solve the dual which has 2n-1 variables
    # Dual: min M * sum(y) s.t. C^T y >= 1, y >= 0
    # C^T y >= 1 means: for each i, sum_k C[k,i]*y[k] >= 1
    # C[k,i] = f[k-i] if 0<=k-i<n
    # So: sum_{k=i}^{i+n-1} f[k-i]*y[k] >= 1

    # This is also a large LP. Let's skip the LP for now and use it only for small n.
    if n > 2048:
        return None

    # For small n, build the constraint matrix
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            A[k, i] = f[k - i]

    constraints = [A @ g <= M, g >= 0]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)

    try:
        prob.solve(solver=cp.GUROBI, time_limit=max_time)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK, time_limit=max_time)
        except Exception:
            return None

    if time.monotonic() - t0 > max_time + 1:
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0)
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val / s


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple lengths, prioritizing 8192 and 4096
    lengths_to_try = [4096, 8192, 2048]
    if parent is not None and len(parent) == 8192:
        lengths_to_try = [8192, 4096, 2048]
    elif parent is not None and len(parent) == 2048:
        lengths_to_try = [2048, 4096, 8192]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 90:
            break

        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        print(f"\n=== n={n_try} (remaining: {remaining:.0f}s) ===")

        # Initialize
        if n_try == (len(parent)):
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)

        sc0 = _score(a0)
        print(f"  Init: score={sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        a_work = a0.copy()
        a_work = a_work / a_work.sum()

        # Phase A: FISTA (coarse) - 15% of remaining time
        phase_a_time = min(remaining * 0.15, 120.0)
        phase_a_end = time.monotonic() + phase_a_time

        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        a_best_a = a_work.copy()
        best_sc_a = _score_norm(a_work)

        # Beta schedule for FISTA
        betas_fista = []
        beta = 0.03 * max_cc
        while beta > 0.0001 * max_cc:
            betas_fista.append(beta)
            beta *= 0.5
        if len(betas_fista) > 10:
            betas_fista = betas_fista[:10]

        for bi, beta in enumerate(betas_fista):
            if time.monotonic() > phase_a_end - 2:
                break
            per_beta_time = min(30, phase_a_end - time.monotonic() - 1)
            top_k = max(30, min(300, int(30 * (1 + bi * 0.3))))
            a_curr, sc = _fista_simplex(a_best_a, time.monotonic() + per_beta_time,
                                         beta, max_iter=50000, top_k=top_k)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
            if sc < best_sc_a:
                best_sc_a = sc
                a_best_a = a_curr.copy()

        a_work = a_best_a.copy()
        print(f"  After FISTA: score={_score(a_work):.10f}")

        # Phase B: Adam log-space (fine) - 55% of remaining time
        phase_b_time = min(remaining * 0.55, remaining - (time.monotonic() - t_start) - 60)
        phase_b_end = time.monotonic() + max(30, phase_b_time)

        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())

        # More aggressive beta schedule
        betas_adam = []
        beta = 0.005 * max_cc
        while beta > 0.000005 * max_cc:
            betas_adam.append(beta)
            beta *= 0.55
        if len(betas_adam) > 15:
            betas_adam = betas_adam[:15]

        # Also add very small betas at the end
        final_beta = betas_adam[-1]
        for frac in [0.5, 0.25, 0.1, 0.05]:
            betas_adam.append(final_beta * frac)

        a_opt, sc_opt = _adam_logspace(a_work, phase_b_end, betas_adam,
                                        lr_init=0.015, top_k_base=80,
                                        max_iter=100000, adaptive_k=True)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
        print(f"  After Adam: score={sc_opt_full:.10f}")

        a_work = a_opt.copy()
        a_work = np.maximum(a_work, 0)
        a_work = a_work / a_work.sum()

        # Phase C: Coordinate descent polishing - 20% of remaining time
        phase_c_time = min(remaining * 0.20, remaining - (time.monotonic() - t_start) - 30)
        phase_c_end = time.monotonic() + max(5, phase_c_time)
        if phase_c_time > 10:
            n_cd_passes = max(2, int(phase_c_time / 30))
            a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=n_cd_passes,
                                                top_k_w=100, deadline=phase_c_end)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
            print(f"  After CD: score={sc_cd_full:.10f}")
            a_work = a_cd

        # Phase D: Ultra-fine Adam polish
        if time.monotonic() < deadline - 40:
            phase_d_end = min(deadline - 20, time.monotonic() + 60)
            c_work = _full_convolve(a_work, a_work)
            max_cw = float(c_work.max())
            betas_ult = [0.0003 * max_cw, 0.0001 * max_cw, 0.00003 * max_cw,
                         0.00001 * max_cw]
            a_ult, sc_ult = _adam_logspace(a_work, phase_d_end, betas_ult,
                                            lr_init=0.005, top_k_base=40,
                                            max_iter=100000, adaptive_k=True)
            sc_ult_full = _score(a_ult)
            if sc_ult_full < best_score:
                best_score = sc_ult_full
                best_seq = a_ult.copy()
            print(f"  After ultra-fine: score={sc_ult_full:.10f}")

    # Phase E: Diversity restarts
    if time.monotonic() < deadline - 40 and best_seq is not None:
        print(f"\n=== Diversity restarts (remaining: {deadline - time.monotonic():.0f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 20) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break
            if ri == 0:
                noise_scale = 0.01
                p_start = a_base + noise_scale * rng.standard_normal(n_best)
            elif ri == 1:
                # Block perturbation
                p_start = a_base.copy()
                block_size = max(1, n_best // 16)
                start = int(rng.integers(0, n_best - block_size))
                p_start[start:start + block_size] *= 1.05
                p_start[start:start + block_size] += 0.005 * rng.standard_normal(block_size)
            else:
                # Frequency domain perturbation
                F = np.fft.rfft(a_base)
                F *= (1.0 + 0.005 * rng.standard_normal(len(F)))
                p_start = np.fft.irfft(F, n_best)
                p_start = np.maximum(p_start, 1e-10)

            p_start = np.maximum(p_start, 1e-12)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.002 * max_cs, 0.001 * max_cs, 0.0004 * max_cs,
                       0.0002 * max_cs, 0.0001 * max_cs]

            p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                             betas_r, lr_init=0.01, top_k_base=60,
                                             max_iter=50000, adaptive_k=True)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f}")

    # Final validation
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
