def _lp_objective(z, p, n):
    """Objective in log-domain: 2*n*LP(conv(softmax(z),softmax(z))) with sum constraint."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0
    shifted = c / c_max
    val = c_max * (np.sum(shifted ** p)) ** (1.0 / p)
    return 2.0 * n * val

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lp_obj_and_grad(z, p, n):
    """Combined objective and gradient in log-domain."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 0.0, np.zeros_like(z)
    shifted = c / c_max
    lp_val = c_max * (np.sum(shifted ** p)) ** (1.0 / p)
    obj = 2.0 * n * lp_val
    
    # Gradient with respect to a
    denom = (np.sum(c ** p)) ** (1.0 / p)
    w = c ** (p - 1) / denom if p > 1 else np.ones_like(c) / c.size
    
    # Compute G (gradient of LP(c) w.r.t. a)
    N3 = _next_pow2(4 * n)
    A3 = np.fft.rfft(a, N3)
    W3 = np.fft.rfft(w, N3)
    Wf3 = np.fft.rfft(w[::-1], N3)
    conv_aw = np.fft.irfft(A3 * W3, N3)
    conv_awf = np.fft.irfft(A3 * Wf3, N3)
    T2 = conv_aw[:n]
    T1 = conv_awf[n-1:2*n-1][::-1]
    G = T1 + T2  # d LP(c) / da
    
    # Gradient w.r.t. z: a = softmax(z), so da/dz = a*(1-a/n... ) 
    # Actually da_i/dz_j = a_i*(delta_ij - a_j)
    # dz/da... we want d(obj)/dz = sum_m (d obj/da_m) * (da_m/dz_j)
    # = sum_m G_m * 2n * a_m * (delta_mj - a_j)
    # = 2n * (a_j * G_j - a_j * sum_m G_m * a_m)
    # = 2n * a_j * (G_j - a·G)
    aG = np.dot(a, G)
    grad_z = 2.0 * n * a * (G - aG)
    
    return obj, grad_z

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "lowpass_noise":
        rng = np.random.default_rng(12345)
        a = rng.standard_normal(n)
        # Apply low-pass filter in frequency domain
        A = np.fft.rfft(a)
        freqs = np.fft.rfftfreq(n)
        cutoff = 0.1  # keep only low frequencies
        mask = np.exp(-freqs / cutoff)
        A = A * mask
        a = np.fft.irfft(A, n)
        a = np.abs(a) + 0.1
        return a / a.sum()
    elif kind == "smooth_noise":
        rng = np.random.default_rng(67890)
        a = np.cumsum(rng.standard_normal(n))
        a = np.abs(a)
        # Smooth
        kernel = np.ones(10) / 10
        a = np.convolve(a, kernel, mode='same')
        a = a + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 3 * t / n) + 0.1 * np.sin(2 * np.pi * 7 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _lbfgs_lp(a_init, p, n, max_iter, deadline):
    """L-BFGS optimization of the L_p smoothed objective."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _lp_obj_and_grad(z, p, n)[0]
    
    def jac(z):
        return _lp_obj_and_grad(z, p, n)[1]
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init

def _active_set_weighted_opt(a_init, deadline, max_iter=200):
    """Iteratively optimize using active set weights."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 1e-6 * c_max
        active = c >= (c_max - tol)
        n_active = active.sum()
        
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1
        
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        
        # Compute gradient of sum_k w[k]*c[k] w.r.t. a
        N3 = _next_pow2(4 * n)
        A3 = np.fft.rfft(a, N3)
        W3 = np.fft.rfft(w, N3)
        Wf3 = np.fft.rfft(w[::-1], N3)
        conv_aw = np.fft.irfft(A3 * W3, N3)
        conv_awf = np.fft.irfft(A3 * Wf3, N3)
        T2 = conv_aw[:n]
        T1 = conv_awf[n-1:2*n-1][::-1]
        G = T1 + T2
        
        # Scaled gradient
        aG = np.dot(a, G)
        grad = G - aG  # tangent to simplex
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Line search
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            a_try = a - step * grad
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_sc:
                best_sc = sc
                best_a = a_try.copy()
                a = a_try
                break
    
    return best_a, best_sc

def _conv_flattening(a_init, deadline, max_iter=1000):
    """Directly try to flatten the convolution by adjusting a."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        c_min = c.min()
        
        if c_max < c_min * 1.01:
            break  # already flat
        
        # Compute the "excess" at each lag
        target = c_sum = c.sum() / len(c)  # average
        excess = c - target  # positive where c is above average
        
        # The excess at lag k contributes to the "pressure" on a
        # We want to reduce a where it contributes most to the high lags
        # Gradient of max(c) w.r.t. a (using active set)
        active = c >= c_max * 0.999
        n_act = active.sum()
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        
        N3 = _next_pow2(4 * n)
        A3 = np.fft.rfft(a, N3)
        W3 = np.fft.rfft(w, N3)
        Wf3 = np.fft.rfft(w[::-1], N3)
        conv_aw = np.fft.irfft(A3 * W3, N3)
        conv_awf = np.fft.irfft(A3 * Wf3, N3)
        T2 = conv_aw[:n]
        T1 = conv_awf[n-1:2*n-1][::-1]
        G = T1 + T2
        aG = np.dot(a, G)
        grad = a * G  # in log space: d log(a_i) direction
        
        # Move in log space
        z = np.log(np.maximum(a, 1e-15))
        z = z - grad * 0.001
        z = z - z.mean()
        a_new = _softmax(z)
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_a = a_new.copy()
            best_sc = sc_new
            a = a_new
        else:
            # Try smaller step
            z = np.log(np.maximum(a, 1e-15))
            z = z - grad * 0.0001
            z = z - z.mean()
            a_new = _softmax(z)
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_a = a_new.copy()
                best_sc = sc_new
                a = a_new
            else:
                break
    
    return best_a, best_sc

def _nelder_mead_polish(a_init, deadline, max_iter=500):
    """Derivative-free polish using Nelder-Mead on a subset of parameters."""
    # For large n, we can't do Nelder-Mead on all n params.
    # Instead, parametrize a small perturbation.
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # Use a Fourier parametrization of the perturbation
    n_modes = min(20, n // 4)
    t = np.arange(n)
    
    def _perturb(params):
        # params: [scale, freq1, amp1, phase1, freq2, amp2, phase2, ...]
        scale = params[0]
        a_try = a_init.copy()
        for i in range(n_modes):
            base = 1 + 3 * i
            if base + 2 >= len(params):
                break
            freq = params[base]
            amp = params[base + 1]
            phase = params[base + 2]
            a_try = a_try + scale * amp * np.cos(2 * np.pi * freq * t / n + phase)
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s < 1e-10:
            return np.inf
        a_try = a_try / s
        return _score(a_try)
    
    # Initial params
    x0 = np.zeros(1 + 3 * n_modes)
    x0[0] = 0.01  # initial scale
    
    try:
        result = minimize(_perturb, x0, method='Nelder-Mead',
                          options={'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-12})
        # Recover the perturbation
        a_try = a_init.copy()
        for i in range(n_modes):
            base = 1 + 3 * i
            if base + 2 >= len(result.x):
                break
            freq = result.x[base]
            amp = result.x[base + 1]
            phase = result.x[base + 2]
            a_try = a_try + result.x[0] * amp * np.cos(2 * np.pi * freq * t / n + phase)
        a_try = np.maximum(a_try, 0)
        s = a_try.sum()
        if s > 1e-10:
            a_try = a_try / s
            sc = _score(a_try)
            if sc < best_sc:
                best_a = a_try
                best_sc = sc
    except Exception:
        pass
    
    return best_a, best_sc

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

def _lp_schedule(a_start, deadline, min_time_remaining=100.0):
    """Schedule L_p optimization with increasing p."""
    n = len(a_start)
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    # Start with moderate p, increase
    p_values = [4, 8, 16, 32, 64, 128, 256]
    
    for p in p_values:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 30:
            break
        
        # Estimate iterations
        max_iter = min(3000, max(100, int(remaining / 0.01)))
        
        a_new = _lbfgs_lp(current_a, p, n, max_iter, deadline - 30)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            print(f"    p={p}: score={sc:.10f}")
    
    return current_a
