Optimized step function sequence generator using multi-length Adam optimization in log-space,
with peak-focused temperature scheduling, spectral rebalancing, and targeted coordinate refinement.
Primary focus: n=8192 and n=16384 with fine-grained optimization from upsampled parent.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_conv(a, c, w, n):
    """Compute gradient of sum_k w_k * c_k with respect to a, where c = a*a."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, w, n):
    """Gradient with respect to log-space variables z where a = softmax(z)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

    grad_a = 2.0 * h
    mean_grad = float(np.dot(a, grad_a))
    return a * (grad_a - mean_grad)

        best_a = a_final.copy()

    return best_a, best_score


def _adam_logspace_v2(a_init, deadline, lr_init, max_iter_total, beta_start, beta_end, n_phases=8, verbose=False):
    """Adam with geometric beta schedule from beta_start to beta_end."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init
    total_iter = max(100, max_iter_total)
    iter_per_phase = total_iter // n_phases

    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        it_in_phase = 0
        while it_in_phase < iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    if verbose and t % 2000 == 0:
                        print(f"    V2 t={t}, beta={beta:.8f}, score={sc:.10f}")

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0001, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _upsample(a, n_new):
    """Upsample sequence to new length using cubic-like interpolation."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a, left=a[0], right=a[-1])
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

def _targeted_peak_reduction(a, deadline, n_trials=5000):
    """Targeted coordinate updates to reduce the top peaks in convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a_best)
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        cm = float(c.max())
        # Find the peak location
        k_peak = int(np.argmax(c))

        # The gradient of c_{k_peak} wrt a_i is 2*a_{k_peak - i} (for valid i)
        # To reduce c_{k_peak}, decrease a_i where a_{k_peak-i} is large
        # i.e., decrease a_i for i in range where k_peak-i is in [0, n-1]

        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)

        if hi < lo:
            continue

        # Get the "kernel" for this peak
        idx = np.arange(lo, hi + 1)
        kernel = best_a[k_peak - idx]

        # Pick a coordinate to decrease where the kernel is large
        # Weighted random selection
        w_sel = kernel ** 2
        w_sel_sum = w_sel.sum()
        if w_sel_sum < 1e-30:
            continue
        probs = w_sel / w_sel_sum
        i_rel = rng.choice(len(idx), p=probs)
        i = idx[i_rel]

        # Try decreasing a[i]
        for factor in [0.95, 0.9, 0.85, 0.8, 1.05, 1.1]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_try.copy()
                break

    return best_a, best_score


def _fista_peaks(a_init, deadline, n_top=5, max_iter=50000):
    """FISTA optimization focusing on top n_top peaks of convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    cm = float(c_curr.max())
    step = 1.0 / (4.0 * n * cm + 1e-10)

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        # Use top-n peaks weighting
        cm_y = float(c_y.max())
        # Soft weighting: exponential with temperature based on gap to top
        shifted = (c_y - cm_y) / (0.001 * cm_y + 1e-10)
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros(len(c_y))
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Projected gradient step
        a_next = y - step * grad
        a_next = np.maximum(a_next, 0.0)
        s_next = a_next.sum()
        if s_next > 1e-30:
            a_next = a_next / s_next
        else:
            a_next = np.ones(n) / n

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

        if it % 500 == 0:
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
            # Adapt step size
            c_new = _full_convolve(a_curr, a_curr)
            cm_new = float(c_new.max())
            step = 1.0 / (4.0 * n * cm_new + 1e-10)

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _random_restart_optimize(a_base, n, deadline, rng, lr=0.005):
    """Do a short Adam optimization from a perturbed version of a_base."""
    sigma = rng.uniform(0.01, 0.05)
    a_pert = a_base.copy()
    z = np.log(np.maximum(a_pert, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    a_pert = _softmax_compute(z)

    c = _full_convolve(a_pert, a_pert)
    cm = float(c.max())
    betas = [0.001 * cm, 0.0005 * cm, 0.0002 * cm, 0.0001 * cm]

    a_opt, sc = _adam_logspace(a_pert, deadline, betas, lr, 50000)
    return a_opt, _score(a_opt)
