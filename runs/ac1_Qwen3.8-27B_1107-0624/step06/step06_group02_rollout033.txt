import numpy as np
import time

def _next_pow2(n):
    if n <= 0:
        return 1
    return 1 << (n - 1).bit_length()

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = mu * r > (cv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _power_mean_grad(a, p):
    """
    Compute gradient of (1/m * sum_k (a*a)[k]^p)^(1/p) w.r.t. a
    returns (grad, f_value)
    """
    n = len(a)
    c = _fft_convolve(a, a)  # length 2n-1
    m = len(c)
    
    # Clip to avoid issues with zeros
    c_safe = np.maximum(c, 1e-300)
    
    c_p = c_safe ** p
    c_p_minus_1 = c_safe ** (p - 1)
    
    f_val = (np.sum(c_p) / m) ** (1.0 / p)
    
    # grad[j] = (1/m) * f_val^(1/p - 1) * sum_k c[k]^(p-1) * a[k-j] (valid k-j)
    # The inner sum is a cross-correlation
    a_rev = a[::-1]
    wc = _fft_convolve(c_p_minus_1, a_rev)
    grad = wc[n-1 : 2*n]  # length n
    
    pre = (1.0 / m) * (f_val ** (1.0/p - 1.0))
    return pre * grad, f_val

Power-mean gradient optimization for minimizing max autocorrelation of non-negative sequences.
Uses L-BFGS-B in log-space with adaptive power-mean order for smooth approximation of the max.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _fft_conv(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    """
    n = len(a)
    c = _fft_conv(a, a)
    m = len(c)  # 2n-1
    
    c_safe = np.maximum(c, 1e-300)
    
    if p == 1.0:
        f_val = float(np.mean(c_safe))
        # grad[j] = (1/m) * sum_k a[k-j] for valid k-j
        # = (1/m) * (c_pm * a_rev)[n-1+j] where c_pm = ones
        a_rev = a[::-1]
        wc = _fft_conv(np.ones_like(c_safe), a_rev)
        grad = wc[n-1:2*n-1] / m

        # which is (1/m) * (number of valid k) * a[j]... no wait
        # c[k] = sum_j a[j]*a[k-j], dc[k]/da[j] = a[k-j]
        # grad[j] = (1/m) * sum_k a[k-j] = (1/m) * sum_{l valid} a[l] where l = k-j
        # For fixed j, l ranges over valid values of k-j where 0<=k<2n-1 and 0<=l<n
        # l = k-j, so k = l+j, need 0<=l+j<2n-1, i.e., l >= max(0, -j) and l <= min(n-1, 2n-2-j)
        # For 0<=j<n: l ranges from 0 to min(n-1, 2n-2-j)

        # If j = n-1: l ranges from 0 to n-1, same

        # k must be < 2n-1, so l+j < 2n-1, i.e., l < 2n-1-j

        grad = np.ones(n) / m * (a.sum())
        return f_val, grad

    pre = (1.0 / m) * (f_val ** (1.0/p - 1.0))
    grad = pre * grad_inner
    
    return f_val, grad

    a = _softmax(z)
    f_val, grad_a = _power_mean_func_and_grad(a, p)
    # Chain rule: grad_z = a * (grad_a - dot(a, grad_a))
    grad_z = a * (grad_a - float(np.dot(a, grad_a)))
    return f_val, grad_z

    """Optimize the power-mean objective using L-BFGS-B in z-space."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    result = minimize(
        _z_space_func_and_grad, z0, args=(p,),
        method='L-BFGS-B', jac=True,
        options={'maxiter': max_iter, 'gtol': tol, 'ftol': 1e-14}
    )
    
    a_opt = _softmax(result.x)
    a_opt = a_opt / a_opt.sum()
    return a_opt, result


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    r = np.arange(1, n + 1, dtype=np.float64)
    cond = mu * r > (cv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Solution: use log-space computation.

log(f) = (1/p) * log(1/m * sum_k exp(p * log(c[k])))

This is a log-sum-exp, which can be computed stably:
log(f) = (1/p) * (log(sum_k exp(p * log(c[k]))) - log(m))
= (1/p) * (max_k(p*log(c[k])) + log(sum_k exp(p*log(c[k]) - max_k(p*log(c[k]))) - log(m))
= log(c_max) + (1/p) * log(sum_k exp(p*(log(c[k]) - log(c_max)))) - log(m)/p

For the gradient, it's better to just be careful with the computation:
- Compute log_c = log(max(c, eps))
- Compute log_c_max = max(log_c)
- Compute shifted = p * (log_c - log_c_max)  # all <= 0
- Compute exp_shifted = exp(shifted)  # all in (0, 1]
- log_f = log_c_max + (1/p) * (log(sum(exp_shifted)) - log(m))
- f = exp(log_f)

For the gradient weights: c[k]^(p-1) = exp((p-1) * log_c[k]) = exp((p-1) * (log_c[k] - log_c_max)) * c_max^(p-1)

So the weights in the gradient are proportional to exp((p-1) * (log_c[k] - log_c_max)), and the overall scale factor is c_max^(p-1) * (1/m) * f^(1/p - 1).

This should be numerically stable as long as we use log-space for the computation.

```python
def _power_mean_func_and_grad(a, p):
    n = len(a)
    c = _fft_conv(a, a)
    m = len(c)
    
    eps = 1e-300
    c_safe = np.maximum(c, eps)
    log_c = np.log(c_safe)
    log_c_max = float(np.max(log_c))
    
    # f = (1/m * sum c^p)^(1/p)
    # log_f = (1/p) * (logsumexp(p * log_c) - log(m))
    shifted = p * (log_c - log_c_max)  # all <= 0
    exp_shifted = np.exp(shifted)
    log_sum_exp = log_c_max * p + np.log(float(np.sum(exp_shifted)))
    log_f = (log_sum_exp - np.log(m)) / p
    f_val = float(np.exp(log_f))
    
    if p == 1.0:
        grad = np.ones(n) * (float(a.sum()) / m)
        return f_val, grad
    
    # weights[k] = c[k]^(p-1) = exp((p-1) * log_c[k])
    # = c_max^(p-1) * exp((p-1) * (log_c[k] - log_c_max))
    log_weights = (p - 1.0) * (log_c - log_c_max)
    weights = np.exp(log_weights)  # proportional to c^(p-1), scaled by c_max^-(p-1)
    
    # grad[j] = (1/m) * f^(1/p-1) * sum_l a[l] * c[l+j]^(p-1)
    # = (1/m) * f^(1/p-1) * c_max^(p-1) * sum_l a[l] * weights[l+j]
    # = (1/m) * f^(1/p-1) * c_max^(p-1) * (weights * a_rev)[n-1+j] (conv)
    
    a_rev = a[::-1]
    wc = _fft_conv(weights, a_rev)
    grad_inner = wc[n-1:2*n-1]  # length n
    
    # Compute the pre-factor in log space
    # pre = (1/m) * f^(1/p-1) * c_max^(p-1)
    # log_pre = -log(m) + (1/p-1)*log(f) + (p-1)*log(c_max)
    # = -log(m) + (1/p-1)*log_f + (p-1)*log_c_max
    log_c_max_val = float(np.max(c))  # actual max value of c
    if log_c_max_val > 0:
        log_pre = -np.log(m) + (1.0/p - 1.0) * log_f + (p - 1.0) * np.log(log_c_max_val)
    else:
        log_pre = -np.log(m) + (1.0/p - 1.0) * log_f
    
    pre = float(np.exp(log_pre))
    grad = pre * grad_inner
    
    if not np.all(np.isfinite(grad)):
        # Fallback: use subgradient of max
        idx = int(np.argmax(c))
        a_rev2 = a[::-1]
        delta = np.zeros(2*n-1)
        delta[idx] = 1.0
        wc2 = _fft_conv(delta, a_rev2)
        grad = 2.0 * n * wc2[n-1:2*n-1]
    
    return f_val, grad
```

Actually, here's another idea: **use the KKT conditions to verify optimality**.

At the optimum of: min max_k (a*a)[k] s.t. sum(a) = 1, a >= 0

The KKT conditions are:
1. There exists λ (Lagrange multiplier for sum constraint) and μ_k >= 0 (for each max constraint) such that:
   2 * sum_k μ_k * a[k-j] = λ - ν_j for all j (where ν_j >= 0 is the multiplier for a[j] >= 0)
   with complementarity: ν_j * a[j] = 0

2. sum_k μ_k = 1 (normalization of the dual variables)
3. μ_k * (max - (a*a)[k]) = 0 (complementarity)

So at the optimum, the vector (a*a) should look like: (a*a)[k] = max for k in the active set S, and < max otherwise. The dual variables μ are supported on S.

The first KKT condition says: 2 * (a * μ_rev)[j] = λ - ν_j, where μ_rev is the reversed μ vector.

If a[j] > 0 for all j (interior solution), then ν_j = 0 and:
2 * (a * μ_rev)[j] = λ for all j

This means: (a * μ_rev)[j] is constant for all j!

This is a very strong condition. It says that the cross-correlation of a with the reversed dual variable is flat.

In the special case where only one index k* is active (μ = delta_{k*}):
2 * a[k*-j] = λ for all j, which means a is constant. But a constant sequence gives (a*a)[k] = 1/n for all k (within the valid range), so all are active. Contradiction.

For the uniform sequence: (a*a)[k] = (n-|k-(n-1)|) / n^2 for |k-(n-1)| < n, which has its max at k=n-1. So only the center is "nearly" active, but the values decrease linearly away from center.

For an optimal sequence, the active set might be {n-1} (just the center) or a few points around the center.

This KKT analysis suggests that at the optimum, the autocorrelation should be "flat" at the peak - i.e., several consecutive values should be equal to the max. This is the "flat-top" property.
