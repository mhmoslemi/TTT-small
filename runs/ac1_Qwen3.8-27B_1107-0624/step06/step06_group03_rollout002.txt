    N = 2 * n  # or next power of 2 >= 2n-1
    N = 1 << (2*n - 1).bit_length()  # next power of 2
    
    # Flat sequence of length 2n-1, zero-padded to N
    # DFT of flat sequence:
    # T(f) = sum_{k=0}^{2n-2} exp(-2πi*k*f/N)
    k = np.arange(2*n - 1)
    f = np.arange(N)
    # T(f) = exp(-πi*(2n-2)*f/N) * sin(π*(2n-1)*f/N) / sin(π*f/N)
    # But at f=0, T(0) = 2n-1
    
    # Actually, let's just compute it directly for clarity
    # T[f] = sum_{k=0}^{2n-2} exp(-2πi*k*f/N)
    # This is a geometric series: exp(-2πi*f/N) * (1 - exp(-2πi*(2n-1)*f/N)) / (1 - exp(-2πi*f/N))
    # At f=0: T(0) = 2n-1
    
    phase = np.exp(-2j * np.pi * np.outer(k, f) / N)
    T = phase.sum(axis=0)  # T[f] = sum_k exp(-2πi*k*f/N)
    
    # A(f) = sqrt(T(f))
    # Choose the branch so that A(0) is positive real
    A = np.sqrt(T + 0j)  # complex square root
    A[0] = np.sqrt(2*n - 1)  # ensure positive at f=0
    
    # Adjust phase: multiply by a constant phase to make a as non-negative as possible
    # Try phase = 0 first
    a_full = np.fft.ifft(A).real
    a = a_full[:n]
    
    # Project to non-negative
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    return a

```python
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    # Assumes sum(a) = 1
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _conv_sqrt_init(n):
    """Initialize using convolution square root idea."""
    N = _next_pow2(2 * n - 1)
    f = np.arange(N)
    k = np.arange(2 * n - 1)
    # T(f) = sum_k exp(-2πi*k*f/N)
    # Use the closed form: T(f) = exp(-πi*(2n-2)*f/N) * sin(π*(2n-1)*f/N) / sin(π*f/N)
    # At f=0: T(0) = 2n-1
    T = np.zeros(N, dtype=np.complex128)
    for fi in range(N):
        if fi == 0:
            T[fi] = 2 * n - 1
        else:
            num = np.sin(np.pi * (2*n-1) * fi / N)
            den = np.sin(np.pi * fi / N)
            if abs(den) < 1e-15:
                T[fi] = 2 * n - 1
            else:
                T[fi] = np.exp(-1j * np.pi * (2*n-2) * fi / N) * num / den
    
    # A(f) = sqrt(T(f))
    A = np.sqrt(T)
    # Make A[0] positive real
    A[0] = np.sqrt(2*n - 1)
    # Conjugate symmetry: A[N-f] = conj(A[f])
    # This is automatically satisfied if T has conjugate symmetry (which it does)
    
    a_full = np.fft.ifft(A).real
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_weights(c, beta):
    """Full softmax weights (numerically stable)."""
    c_max = float(c.max())
    w = np.exp((c - c_max) / beta)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w

def _grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _fista(a_init, deadline, betas, top_k=None):
    """FISTA with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    it = 0
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        # Reset some state for new beta
        t_val = 1.0
        a_prev = a_curr.copy()
        
        while time.monotonic() < deadline:
            it += 1
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta)
            grad = _grad_conv(y, w)
            
            a_next = _proj_simplex(y - step * grad)
            
            if it % 500 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
                    print(f"    FISTA it={it} beta={beta:.6f}: score={sc:.10f}")
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_per_beta):
    """Adam in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        m_buf *= 0.1  # decay momentum at phase boundary
        v_buf *= 0.1
        
        while it_in_phase < max_iter_per_beta:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = np.exp(z - z.max())
            a /= a.sum()
            
            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)
            grad_a = _grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad_z = n * a * (grad_a - mean_grad)  # gradient in z-space (for normalized problem)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 1.0
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 200 == 0:
                a_chk = np.exp(z - z.max())
                a_chk /= a_chk.sum()
                sc = _score_norm(a_chk)
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    print(f"    Adam it={t} beta={beta:.8f}: score={sc:.10f}")
            
            if it_in_phase % 5000 == 0:
                lr = max(0.0001, lr * 0.95)
    
    a_final = np.exp(z - z.max())
    a_final /= a_final.sum()
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _coord_descend(a, deadline, n_passes=5):
    """Coordinate descent with 1D optimization."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    c = _full_convolve(a, a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            c = _full_convolve(a, a)
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue
            
            # Affected positions: k = i, i+1, ..., i+n-1 (clipped to [0, 2n-2])
            k_start = max(0, i)
            k_end = min(2*n - 2, i + n - 1)
            affected = np.arange(k_start, k_end + 1)
            
            # d[k] = a[k - i] for k in affected
            d = a[affected - i]
            
            # b[k] = c[k] - a_i_old * d[k] (so c_new[k] = b[k] + t * d[k])
            b = c[affected] - a_i_old * d
            
            # Unaffected max
            if len(affected) < 2*n - 1:
                mask = np.ones(2*n - 1, dtype=bool)
                mask[affected] = False
                c_unaff_max = c[mask].max() if mask.any() else 0.0
            else:
                c_unaff_max = 0.0
            
            S0 = 1.0 - a_i_old  # sum of other elements
            
            # Try candidate t values
            t_candidates = [0.0, a_i_old, a_i_old * 0.5, a_i_old * 1.5]
            
            # Critical points: where b[k] + t*d[k] = c_unaff_max
            for j in range(len(affected)):
                if d[j] > 1e-15:
                    t_star = (c_unaff_max - b[j]) / d[j]
                    if -0.1 * a_i_old < t_star < 5.0 * a_i_old + 0.1:
                        t_candidates.append(max(0.0, t_star))
            
            # Also: where b[k1] + t*d[k1] = b[k2] + t*d[k2]
            # t = (b[k2] - b[k1]) / (d[k1] - d[k2])
            # Only check top few pairs
            top_idx = np.argsort(b + a_i_old * d)[-min(10, len(affected)):]
            for ii in range(len(top_idx)):
                for jj in range(ii+1, len(top_idx)):
                    k1, k2 = top_idx[ii], top_idx[jj]
                    dd = d[k1] - d[k2]
                    if abs(dd) > 1e-15:
                        t_cross = (b[k2] - b[k1]) / dd
                        if 0 <= t_cross <= 5.0 * a_i_old + 0.1:
                            t_candidates.append(t_cross)
            
            t_candidates = list(set([max(0.0, tc) for tc in t_candidates]))
            
            best_t = a_i_old
            best_val = np.inf
            for tc in t_candidates:
                c_aff_new = b + tc * d
                max_c = max(c_unaff_max, float(c_aff_new.max()))
                s_new = S0 + tc
                if s_new < 1e-15:
                    continue
                val = 2.0 * n * max_c / (s_new * s_new)
                if val < best_val:
                    best_val = val
                    best_t = tc
            
            if best_t != a_i_old:
                a[i] = best_t
                a = a / a.sum()
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score
