"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: adaptive active-set with top-k selection, dual-primal
coordinate updates, spectral pre-conditioning, aggressive beta continuation,
better subgradient polish with adaptive tolerance, and diverse asymmetric
initializations to break the symmetry barrier.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_active_gradient(z, a, c, n, top_k=None, tol_frac=0.005):
    """Compute gradient of smoothed max using active set weighting."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a), 0, c_max
    if top_k is not None:
        k = min(top_k, len(c))
        active = np.zeros(len(c), dtype=bool)
        top_indices = np.argsort(c)[::-1][:k]
        active[top_indices] = True
    else:
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W), active.sum(), c_max


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _dual_primal_update(a, c, n, n_iter=200, tol_frac=0.003):
    """Alternate between dual (active set weights) and primal (a) updates."""
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_cmax = float(c.max())

    for it in range(n_iter):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < best_cmax * (1 - 1e-12):
            best_cmax = c_max
            best_a = a.copy()

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_act = int(active.sum())

        # Primal update: reduce a[j] where it contributes to active peaks
        w = np.zeros_like(c)
        w[active] = 1.0 / n_act
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Multiplicative update: a[j] *= (W / h[j])^eta (push down where h is high)
        eta = 0.1 / (1 + 0.01 * it)
        ratio = W / np.maximum(h, 1e-30)
        a = a * np.power(ratio, eta)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

    a_final = a / a.sum()
    return a_final


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, aggressive=False):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    if aggressive:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    else:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.004 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 10))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp_up":
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif kind == "ramp_down":
        a = np.linspace(1.0, 0.1, n)
        return a / a.sum()
    elif kind == "asymmetric_bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.3) ** 2) + 0.05
        return a / a.sum()
    elif kind == "asymmetric_bell2":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.6) ** 2) + 0.05
        return a / a.sum()
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25) ** 2) + np.exp(-8.0 * (t - 0.75) ** 2) + 0.02
        return a / a.sum()
    elif kind == "sinusoidal_asym":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.sin(2.0 * np.pi * t + 0.5) + 0.1 * t
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "chirp_based":
        # Use a chirp in frequency domain to create structured time-domain signal
        N = _next_pow2(n)
        freqs = np.fft.rfftfreq(N)
        phase = np.pi * freqs * N
        spectrum = np.exp(1j * phase)
        a = np.fft.irfft(np.abs(np.fft.rfft(np.ones(N) / N)) * spectrum.real, N)[:n]
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _spectral_precondition(a, n_steps=50, alpha=0.5):
    """Iteratively flatten the auto-convolution spectrum."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    N = _next_pow2(2 * n - 1)

    for step in range(n_steps):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # In frequency domain, try to reduce the peak
        A = np.fft.rfft(a, N)
        C = A * A  # This is the DFT of the zero-padded auto-convolution
        # Reduce components that contribute to the peak
        c_ideal = (a.sum() ** 2) / (2 * n - 1)
        # Scale down frequencies where |C| is too large
        C_mag = np.abs(C)
        C_phase = np.angle(C)
        target = c_ideal * N
        C_new = np.where(C_mag > target, C * (target / C_mag) ** alpha, C)
        a_new = np.fft.irfft(C_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        sc_new = _score(a_new)
        sc_old = _score(a)
        if sc_new < sc_old:
            a = a_new
        else:
            alpha *= 0.7
    return a


def _shrinkage_reopt(a, deadline):
    """Try zeroing smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_asymmetric(a, rng):
    """Apply an asymmetric perturbation (breaks left-right symmetry)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.linspace(0, 1, n)
    # Add a linear trend to break symmetry
    z = z + 0.1 * rng.uniform(-1, 1) * t
    # Add a quadratic

    # Add small noise
    z = z + 0.02 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        # Very fine L-BFGS at last moment
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00005
        remaining = deadline - 5 - time.monotonic()
        if remaining > 3:
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_local: _smoobj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smograd(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient with very tight tolerance
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(
                best_seq, deadline - 2, n_max_steps=500, lr_init=0.005,
                tol_fracs=(0.001, 0.0005, 0.0002))
            update_best(a_fin2, sc_fin2, "final_sg")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
