def _next_pow2(n):

    while k < n:

    N = _next_pow2(out_n)

    fb = np.fft.rfft(b, N)

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

                continue

            c_try = _full_convolve(a_try, a_try)

            if sc_try < a_old_sc - 1e-12 * alpha * gnorm:

                improved = True
                break

            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a = a_try / s_try
                c_try = _full_convolve(a, a)

                    improved = True

                lr *= 0.8
            if no_improve > patience:

                lr = lr_init * 0.5

                    break
        
        # Periodically tighten tolerance
        if it % 1000 == 0 and tol > tol_end:
            tol = max(tol_end, tol * 0.95)
    
    a = best_a
    return a, _score(a)

    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time_per_iter * 15))))

        try:

                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))

            pass

def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    """
    Active-set gradient descent polishing.
    Works on normalized (sum=1) sequences.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = 2.0 * n * float(_full_convolve(a, a).max())
    
    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 300
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max
        
        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0
        
        # Active set
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
        
        # Gradient: 2 * sum_{k in active} a[k-m] / |active|
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        
        if gnorm < 1e-15:
            break
        
        # Line search
        alpha = lr
        a_old_sc = current_sc
        improved = False
        for _ in range(50):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-14:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            no_improve += 1
            if no_improve % 50 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.7)
                lr *= 0.7
            if no_improve >= max_stall:
                # Reset and try tighter tol
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                if tol_frac < tol_frac_end * 1.1:
                    break
        else:
            # Tighten tolerance slowly when making progress
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.98)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try  # continue from this point for next fraction
            # Re-sort
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.25)**2) + np.exp(-8.0 * (t - 0.75)**2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    """Resample sequence to new length using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with adaptive log-sum-exp smoothing, active-set gradient
descent with trust-region stepping, sparsity post-processing, and diverse
multi-start initializations with multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n

    def __call__(self, z):
        return _smoobj(z, self.beta, self.n)

    def grad(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < max_c * 0.001:
        gap = max_c * 0.001
    base_beta = max(gap, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.125, 0.0625,
                                  0.03, 0.015, 0.0075, 0.004,
                                  0.002, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(4000, max(20, int(remaining / (est_time_per_iter * 12))))

        prob = _LbfgsProb(beta, n)
        try:
            result = minimize(
                prob, z, jac=prob.grad, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 60,
                    'maxls': 40,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_polish(a_init, deadline, tol_frac_start=0.01,
                       tol_frac_end=0.00005, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    c0 = _full_convolve(a, a)
    best_sc = 2.0 * n * float(c0.max())

    lr = lr_init
    tol_frac = tol_frac_start
    no_improve = 0
    max_stall = 400
    stall_count = 0

    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        c_max = float(c.max())
        current_sc = 2.0 * n * c_max

        if current_sc < best_sc - 1e-14:
            best_sc = current_sc
            best_a = a.copy()
            no_improve = 0

        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            idx = int(np.argmax(c))
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = 1

        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)

        if gnorm < 1e-15:
            break

        alpha = min(lr, 1.0 / (gnorm + 1e-15))
        a_old_sc = current_sc
        improved = False
        for _ in range(60):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            sc_try = 2.0 * n * float(c_try.max())
            if sc_try < a_old_sc - 1e-13:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                tol_frac = max(tol_frac_end, tol_frac * 0.75)
                lr *= 0.7
            if no_improve >= max_stall:
                a = best_a.copy()
                tol_frac = max(tol_frac_end, tol_frac * 0.5)
                lr = lr_init * 0.3
                no_improve = 0
                stall_count += 1
                if stall_count > 8:
                    break
        else:
            if no_improve == 0 and tol_frac > tol_frac_end:
                tol_frac = max(tol_frac_end, tol_frac * 0.97)
            lr = min(lr_init, lr * 1.02)

    return best_a, _score(best_a)


def _shrinkage_optimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.05, 0.07, 0.10]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try
            a_sorted_idx = np.argsort(a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.arange(n)
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * 5.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-10.0 * (t - 0.2) ** 2) + np.exp(-10.0 * (t - 0.8) ** 2) + 0.05
        return a / a.sum()
    elif kind == "periodic":
        t = np.arange(n)
        a = 1.0 + 0.35 * np.sin(2.0 * np.pi * 13.0 * t / n) + 0.15 * np.cos(2.0 * np.pi * 7.0 * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 0.5 + t + 0.1 * np.sin(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase search: L-BFGS + active-set GD + shrinkage + multi-n + restarts."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        best_seq = parent.copy()
        best_score = parent_sc
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Main L-BFGS ===
    print(f"\n[P1] L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=500.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main.copy()
        best_score = sc_main
        print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [lbfgs_main] t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Active-set polish (coarse) ===
    print(f"\n[P2] Active-set polish coarse (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 350:
        pol_deadline = min(deadline - 350, time.monotonic() + 180)
        a_pol, sc_pol = _active_set_polish(best_seq, pol_deadline,
                                           tol_frac_start=0.008, tol_frac_end=0.0002,
                                           lr_init=0.04)
        if sc_pol < best_score:
            best_seq = a_pol.copy()
            best_score = sc_pol
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish_coarse] t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Active-set polish (fine) ===
    print(f"\n[P3] Active-set polish fine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        pol_deadline = min(deadline - 250, time.monotonic() + 150)
        a_pol2, sc_pol2 = _active_set_polish(best_seq, pol_deadline,
                                             tol_frac_start=0.002, tol_frac_end=0.00005,
                                             lr_init=0.015)
        if sc_pol2 < best_score:
            best_seq = a_pol2.copy()
            best_score = sc_pol2
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [polish_fine] t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Shrinkage ===
    print(f"\n[P4] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        a_shr, sc_shr = _shrinkage_optimize(best_seq, deadline - 200)
        if sc_shr < best_score:
            best_seq = a_shr.copy()
            best_score = sc_shr
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [shrinkage] t={time.monotonic()-t_start:.1f}s")
            # Re-optimize after shrinkage
            if time.monotonic() < deadline - 150:
                a_shr2 = _run_lbfgs(best_seq, deadline, min_time_remaining=120.0)
                sc_shr2 = _score(a_shr2)
                if sc_shr2 < best_score:
                    best_seq = a_shr2.copy()
                    best_score = sc_shr2
                    print(f"  NEW BEST: {best_score:.10f} [lbfgs_post_shrink] t={time.monotonic()-t_start:.1f}s")
                if time.monotonic() < deadline - 100:
                    a_shr3, sc_shr3 = _active_set_polish(best_seq, min(deadline - 100, time.monotonic() + 80),
                                                         tol_frac_start=0.001, tol_frac_end=0.00003,
                                                         lr_init=0.01)
                    if sc_shr3 < best_score:
                        best_seq = a_shr3.copy()
                        best_score = sc_shr3
                        print(f"  NEW BEST: {best_score:.10f} [polish_post_shrink] t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-n exploration ===
    print(f"\n[P5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 5120, 6144, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 100:
            break
        if n_try == len(best_seq) and time.monotonic() > deadline - 50:
            continue
        a_init = _resample(best_seq, n_try)
        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_seq = a_opt.copy()
            best_score = sc_opt
            print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n] t={time.monotonic()-t_start:.1f}s")
        elif sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_polish(a_opt, min(deadline - 35, time.monotonic() + 35),
                                                 tol_frac_start=0.003, tol_frac_end=0.0001,
                                                 lr_init=0.012)
            if sc_opt2 < best_score:
                best_seq = a_opt2.copy()
                best_score = sc_opt2
                print(f"  NEW BEST: {best_score:.10f} (n={n_try}) [multi_n_polish] t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Diverse restarts ===
    print(f"\n[P6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 50:
        n_best = len(best_seq)
        restart_starts = [
            ("pert3", _perturb(best_seq, 0.03, rng)),
            ("pert8", _perturb(best_seq, 0.08, rng)),
            ("fourier", _fourier_perturb(best_seq, rng, n_modes=4)),
            ("cos", _make_profile(n_best, "cos_mod")),
            ("bimo", _make_profile(n_best, "bimodal")),
            ("peri", _make_profile(n_best, "periodic")),
        ]
        for label, a_r in restart_starts:
            if time.monotonic() > deadline - 25:
                break
            a_opt = _run_lbfgs(a_r, deadline, min_time_remaining=15.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                best_seq = a_opt.copy()
                best_score = sc_opt
                print(f"  NEW BEST: {best_score:.10f} (n={n_best}) [restart_{label}] t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n[P7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 5:
        # Tightest active-set polish
        a_fin, sc_fin = _active_set_polish(best_seq, deadline - 3,
                                           tol_frac_start=0.0008, tol_frac_end=0.00001,
                                           lr_init=0.005)
        if sc_fin < best_score:
            best_seq = a_fin.copy()
            best_score = sc_fin
            print(f"  NEW BEST: {best_score:.10f} [final_polish] t={time.monotonic()-t_start:.1f}s")

        # Tiny L-BFGS with very small beta
        remaining = deadline - 3 - time.monotonic()
        if remaining > 8:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            beta_fin = mb * 0.00003
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            prob = _LbfgsProb(beta_fin, n)
            max_iter = min(2000, max(30, int(remaining / 0.04)))
            try:
                result = minimize(prob, z, jac=prob.grad, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 60})
                a_fin2 = _softmax(result.x)
                sc_fin2 = _score(a_fin2)
                if sc_fin2 < best_score:
                    best_seq = a_fin2.copy()
                    best_score = sc_fin2
                    print(f"  NEW BEST: {best_score:.10f} [final_lbfgs] t={time.monotonic()-t_start:.1f}s")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
