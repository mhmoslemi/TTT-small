"""
Optimized mirror descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive temperature annealing, active-set gradients, multi-scale
exploration, and perturbation restarts.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, T):
    """Compute soft gradient of 2n*max(c) w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    """Compute gradient using only top-k convolution coefficients."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _mirror_descent(a_init, deadline, T_start, T_end, lr_start, lr_end,
                    max_iter=200000, use_active_set=False, active_k=5,
                    eval_interval=200, clip_norm=1.0, seed=0):
    """
    Mirror descent in log-space (equivalent to multiplicative weights update).
    Handles the simplex constraint via softmax parameterization.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    # Adam buffers
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience_score = best_score
    lr_mult = 1.0
    
    # Precompute FFT of z-space stuff? No, a changes each iteration.
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        # Exponential temperature annealing
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        if use_active_set and T < T_start * 0.01:
            g = _compute_active_gradient(a, c, active_k)
        else:
            g = _compute_gradient(a, c, T)
        
        # Remove mean to maintain sum(a) = 1 (project gradient to tangent space)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()  # Maintain normalization
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            # Adaptive learning rate
            if patience > 3 and lr_mult > 0.05:
                lr_mult *= 0.6
                patience = 0
                # Reset to best
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_u_shaped(n, alpha=1.0):
    """Create a U-shaped profile (higher at edges)."""
    x = np.linspace(0, 1, n)
    a = 1.0 + alpha * (2*x - 1)**2
    return a / a.sum()


def _make_gaussian(n, sigma_frac=0.3):
    """Create a Gaussian profile."""
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma)**2)
    return a / a.sum()


def _make_lorentzian(n, gamma=0.2):
    """Create a Lorentzian (Cauchy) profile."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + ((x - 0.5) / gamma)**2)
    return a / a.sum()


def _make_power_edge(n, p=2.0):
    """Create a profile weighted towards edges: a_i ∝ (i+1)^p * (n-i)^p normalized."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p * (n - i) ** p
    # Actually let's do: a_i ∝ max(i, n-1-i) / n
    x = np.arange(n, dtype=np.float64) / n
    a = x ** p + (1 - x) ** p
    return a / a.sum()


def _make_sin(n, k=1):
    """Create a sinusoidal modulation."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(2 * np.pi * k * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()

"""
Multi-phase mirror descent optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive temperature annealing, active-set gradients, diverse initializations
(including U-shaped and bimodal profiles), multi-scale exploration (n=4096, n=8192),

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k=5):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_weighted_gradient(a, c, top_k=10, weight_beta=0.5):
    """Gradient weighted by c^beta for active peaks - emphasizes larger peaks more."""
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    c_top = c[k_indices]
    c_max = c_top[0]
    if c_max < 1e-15:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        w = (c_top / c_max) ** weight_beta
        w = w / w.sum()
        p = np.zeros_like(c)
        p[k_indices] = w
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

    z = z - z.mean()

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_profile(name, n):
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'u1':
        x = np.linspace(0, 1, n)
        a = 1.0 + 1.0 * (2.0 * x - 1.0) ** 2
        return a / a.sum()
    elif name == 'u2':
        x = np.linspace(0, 1, n)
        a = 1.0 + 3.0 * (2.0 * x - 1.0) ** 2
        return a / a.sum()
    elif name == 'u4':
        x = np.linspace(0, 1, n)
        a = 1.0 + 5.0 * (2.0 * x - 1.0) ** 4
        return a / a.sum()
    elif name == 'bimodal':
        x = np.linspace(0, 1, n)
        sigma = 0.15
        a = np.exp(-((x - 0.25) / sigma) ** 2) + np.exp(-((x - 0.75) / sigma) ** 2) + 0.05
        return a / a.sum()
    elif name == 'gauss':
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / 0.25) ** 2)
        return a / a.sum()
    elif name == 'lorentz':
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + ((x - 0.5) / 0.15) ** 2)
        return a / a.sum()
    elif name == 'w_shape':
        x = np.linspace(0, 1, n)
        a = (2.0 * x - 1.0) ** 4 + 0.1
        return a / a.sum()
    elif name == 'sin2':
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * x)

    else:
        return np.ones(n) / n

    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end - 30:
        n_large = 8192
        if n_large > n:
            a_large = _interp_to_length(best_a, n_large)
            c_large = _fft_convolve(a_large, a_large)
            max_cl = float(c_large.max())
            sub_dl_5a = min(phase5_end - 60, time.monotonic() + 120.0)
            a_large_opt, _ = _mirror_descent(
                a_large, sub_dl_5a,
                T_start=0.003 * max_cl, T_end=0.00005 * max_cl,
                lr_start=0.03, lr_end=0.002,
                max_iter=200000,
                use_active_set=True, active_k=5,
                grad_mode="weighted", weight_beta=0.5,
                eval_interval=200, clip_norm=0.8
            )
            full_sc_large = _score(a_large_opt)
            print(f"Phase 5 n=8192: score={full_sc_large:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_large) and full_sc_large < best_score:
                best_score = full_sc_large
                best_a = a_large_opt.copy()

            # Try compressing back to original n
            if time.monotonic() < phase5_end - 30:
                n_target = 4096
                a_comp = _interp_to_length(a_large_opt, n_target)
                c_comp = _fft_convolve(a_comp, a_comp)
                max_cc = float(c_comp.max())
                sub_dl_5b = min(phase5_end - 5, time.monotonic() + 60.0)
                a_comp_opt, _ = _mirror_descent(
                    a_comp, sub_dl_5b,
                    T_start=0.001 * max_cc, T_end=0.00002 * max_cc,
                    lr_start=0.02, lr_end=0.001,
                    max_iter=100000,
                    use_active_set=True, active_k=3,
                    eval_interval=150, clip_norm=0.6
                )
                full_sc_comp = _score(a_comp_opt)
                print(f"Phase 5 compress to n={n_target}: score={full_sc_comp:.10f}, t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc_comp) and full_sc_comp < best_score:
                    best_score = full_sc_comp
                    best_a = a_comp_opt.copy()
    else:
        print(f"Phase 5: skipped (not enough time)")

    # Phase 6: Perturbation restarts (1500s - 1700s)
    phase6_end = min(deadline, t_start + 1700.0)
    n_best = len(best_a)
    if time.monotonic() < phase6_end - 20:
        for attempt in range(8):
            if time.monotonic() > phase6_end - 25:
                break
            sigma = 0.008 + 0.004 * attempt
            z_tmp = np.log(np.maximum(best_a, 1e-15))
            z_tmp = z_tmp + sigma * rng.standard_normal(n_best)
            z_tmp = z_tmp - z_tmp.mean()
            a_pert = _softmax_stable(z_tmp)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_dl = min(phase6_end - 10, time.monotonic() + 22.0)
            a_pert_opt, _ = _mirror_descent(
                a_pert, sub_dl,
                T_start=0.001 * max_cp, T_end=0.00002 * max_cp,
                lr_start=0.02, lr_end=0.001,
                max_iter=40000,
                use_active_set=True, active_k=3,
                grad_mode="weighted", weight_beta=0.5,
                eval_interval=100, clip_norm=0.6
            )
            full_sc = _score(a_pert_opt)
            improved = " *" if full_sc < best_score else ""
            print(f"  Restart[{attempt}] sigma={sigma:.4f}: score={full_sc:.10f}{improved} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 7: Final ultra-fine polish (1700s - deadline)
    if time.monotonic() < deadline - 5:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _mirror_descent(
            best_a, deadline - 3,
            T_start=0.00005 * max_cf, T_end=0.0000005 * max_cf,
            lr_start=0.002, lr_end=0.0001,
            max_iter=300000,
            use_active_set=True, active_k=3,
            grad_mode="weighted", weight_beta=0.3,
            eval_interval=100, clip_norm=0.3
        )
        full_sc = _score(a_final)
        print(f"Phase 7 final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

    # Final validation and return
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
