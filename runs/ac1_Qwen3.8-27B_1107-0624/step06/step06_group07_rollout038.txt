from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_step(a, M):
    """Solve the LP: max sum(b) s.t. conv(a,b) <= M, b >= 0"""
    n = len(a)
    # Build constraint matrix A (2n-1 x n)
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    # This is a Toeplitz matrix
    # For efficiency, use sparse representation
    
    # Actually, for n=4096, building the full matrix would be 8191 x 4096 = 33M entries
    # That's about 264 MB as float64. Might be OK but let's use sparse.
    
    # Each row k has non-zeros at columns j where 0 <= k-j <= n-1
    # i.e., j is in [max(0, k-n+1), min(n-1, k)]
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max c^T b s.t. A b <= M, b >= 0
    # linprog minimizes, so: min -c^T b s.t. A b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if not res.success:
        return a  # Return original if LP fails
    
    b = res.x
    s = b.sum()
    if s < 1e-15:
        return a
    return b / s  # Normalize to sum 1

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = np.convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _lp_step(a, M, n_constraints=None):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M for all k, b >= 0"""
    n = len(a)
    
    # Build sparse constraint matrix
    # A[k, j] = a[k-j] if 0 <= k-j <= n-1
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            j_range = range(j_lo, j_hi + 1)
            rows.extend([k] * (j_hi - j_lo + 1))
            cols.extend(range(j_lo, j_hi + 1))
            vals.extend(a[k - j] for j in j_range)
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return a, False
    
    b = res.x
    s = b.sum()
    if s < 1e-15:
        return a, False
    return b / s, True
