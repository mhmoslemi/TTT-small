"""
Hybrid optimizer: LP-based direction finding + Adam log-space gradient descent + coordinate descent.
Minimizes 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Key improvements over previous: (1) LP polishing via scipy HiGHS, (2) full-position softmax smoothing,
(3) adaptive step sizes with line search, (4) multi-length exploration with spectral initialization.

import numpy as np
import time
import scipy.optimize
import scipy.sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1, returns 2n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a. Result has length n."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _softmax_weights(c, T):
    """Full softmax weights on convolution with temperature T."""
    c_max = float(c.max())
    exp_vals = np.exp((c - c_max) / T)
    w_sum = float(exp_vals.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return exp_vals / w_sum


def _logsumexp_score(a, T):
    """Compute log-sum-exp of convolution (smooth max) and its gradient."""
    c = _full_convolve(a, a)
    T = max(T, 1e-15)
    c_max = float(c.max())
    log_Z = c_max / T + np.log(float(np.exp((c - c_max) / T).sum()))
    w = np.exp((c - c_max) / T)
    w_sum = float(w.sum())
    w = w / w_sum
    grad = _compute_grad_conv(a, w)
    return log_Z, grad, c, w


def _adam_logspace_full(a_init, deadline, T_schedule, lr_init, max_iter_per_phase):
    """Adam in log-space with full-position softmax smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init

    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            log_Z, grad_c, c, w = _logsumexp_score(a, T)
            # d(log_Z)/d(a_i) = grad_c_i
            # d(log_Z)/d(z_i) = a_i * grad_c_i - <a * grad_c, a> (for simplex normalization)
            # Actually, since we're in log-space and a = softmax(z):
            # dF/dz_i = a_i * (dF/da_i - sum_j a_j * dF/da_j)
            mean_grad = float(np.dot(a, grad_c))
            grad_z = a * (grad_c - mean_grad)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        # Decay momentum at phase boundaries
        m_buf *= 0.5
        v_buf *= 0.5
        # Reduce learning rate between phases
        current_lr *= 0.8

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -40, 40))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _solve_lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Returns b (normalized to sum 1) or None if LP fails.
    Uses dual form for efficiency.
    Primal: max 1^T b s.t. A@b <= M*1, b >= 0
    where A[k,j] = a[k-j] if 0<=k-j<n else 0, k=0..2n-2, j=0..n-1
    
    Dual: min M * 1^T y s.t. A^T @ y >= 1, y >= 0
    A^T[j,k] = a[k-j] if 0<=k-j<n else 0
    Equivalently: sum_{i=0}^{n-1} a[i] * y[i+j] >= 1 for j=0..n-1
    """
    n = len(a)
    # Dual constraint matrix: C[j, k] = a[k-j] if 0 <= k-j < n, else 0
    # For dual variable y[k], k=0..2n-2:
    # Constraint j: sum_{k=0}^{2n-2} C[j,k] * y[k] >= 1
    # C[j,k] = a[k-j] if max(0,j) <= k <= min(2n-2, j+n-1)
    
    # Use scipy sparse for the constraint matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(2 * n - 1, j + n - 1)
        for k in range(k_start, k_end + 1):
            idx = k - j  # 0 to min(n-1, 2n-1-j)
            if 0 <= idx < n:
                rows.append(j)
                cols.append(k)
                vals.append(a[idx])
    
    C = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(n, 2 * n - 1))
    
    # Dual LP: min M * sum(y) s.t. C @ y >= 1, y >= 0
    # scipy linprog: min c^T x s.t. A_ub @ x <= b_ub, A_eq @ x = b_eq, bounds
    # Convert: min (M*1)^T y s.t. -C @ y <= -1, y >= 0
    c_obj = np.full(2 * n - 1, M)
    A_ub = -C.tocsc()
    b_ub = -np.ones(n)
    bounds = [(0, None)] * (2 * n - 1)
    
    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
            method='highs', options={'presolve': True, 'time_limit': 30}
        )
        if res.status == 0:
            y = res.x
            # Recover primal from dual: b_j is nonzero where dual constraint j is tight
            # Complementary slackness: b_j > 0 implies (C@y)_j = 1
            # For b, we use: b_j proportional to... actually we need to solve the primal
            # Alternatively, use the dual solution to get direction
            
            # The LP optimum value = M * sum(y)
            # The primal solution b can be recovered, but for our purpose,
            # we can use a simpler approach: the dual tells us which primal constraints are active.
            # Let's just solve the primal directly for b.
            
            # Actually, let's solve the primal LP directly instead:
            # max sum(b) s.t. A_p @ b <= M, b >= 0
            # where A_p[k, j] = a[k-j] if 0 <= k-j < n
            # This has 2n-1 constraints and n variables.
            
            # Convert to min form: min -sum(b) s.t. A_p @ b <= M, b >= 0
            c_p = -np.ones(n)
            # A_p is the transpose pattern: A_p[k,j] = a[k-j] if 0<=k-j<n
            # Build A_p as sparse
            rows_p = []
            cols_p = []
            vals_p = []
            for k in range(2 * n - 1):
                for j in range(max(0, k - n + 1), min(n, k + 1)):
                    idx = k - j  # 0 to n-1
                    rows_p.append(k)
                    cols_p.append(j)
                    vals_p.append(a[idx])
            
            A_p = scipy.sparse.csr_matrix((vals_p, (rows_p, cols_p)), shape=(2 * n - 1, n))
            
            res_p = scipy.optimize.linprog(
                c_p, A_ub=A_p, b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n,
                method='highs', options={'presolve': True, 'time_limit': 30}
            )
            if res_p.status == 0:
                b = res_p.x
                b_sum = float(b.sum())
                if b_sum > 1e-10:
                    return b / b_sum
                else:
                    return None
        return None
    except Exception as e:
        print(f"  LP error: {e}")
        return None


def _lp_polish(a, deadline, max_iters=15):
    """Polish using LP-based direction finding with line search."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        b = _solve_lp_direction(a, M)
        if b is None:
            print(f"  LP failed at iter {it}")
            break
        
        # Line search: find best t in [0,1]
        best_t = 0.0
        best_f = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            f_try = _score_norm(a_trial)
            if f_try < best_f:
                best_f = f_try
                best_t = t
        
        if best_t == 0.0:
            # Try smaller steps
            for t in [0.01, 0.02, 0.05]:
                a_trial = (1 - t) * a + t * b
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial = a_trial / s
                f_try = _score_norm(a_trial)
                if f_try < best_f:
                    best_f = f_try
                    best_t = t
        
        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"  LP iter {it}: t={best_t:.3f}, score={sc:.10f}")
        else:
            print(f"  LP iter {it}: no improvement, stopping")
            break
    
    return best_a, best_score


def _coord_descend_polish(a, deadline, n_passes=5, top_k_w=100):
    """Coordinate descent with thorough 1D optimization."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    rng = np.random.default_rng(9999)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i_old = a[i]
            c = _full_convolve(a, a)
            
            # d_j = a_{j-i} for valid j
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]
            
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old  # assuming sum(a) = 1
            
            # Golden section search for optimal t
            # F(t) = 2n * max(c_base + d*t) / (S0+t)^2
            lo = 0.0
            hi = max(10.0 * a_i_old, 1.0 / n * 10)
            
            # First, check if 0 is better
            f_0 = 2.0 * n * float(c_base.max()) / (S0 * S0) if S0 > 1e-15 else np.inf
            f_cur = 2.0 * n * float(c.max()) / 1.0
            
            # Golden section
            gr = (np.sqrt(5) - 1) / 2
            a_g = hi - gr * (hi - lo)
            b_g = lo + gr * (hi - lo)
            f_a = 2.0 * n * float((c_base + d * a_g).max()) / ((S0 + a_g) ** 2)
            f_b = 2.0 * n * float((c_base + d * b_g).max()) / ((S0 + b_g) ** 2)
            
            for _ in range(25):
                if f_a < f_b:
                    hi = b_g
                    b_g = a_g
                    f_b = f_a
                    a_g = hi - gr * (hi - lo)
                    f_a = 2.0 * n * float((c_base + d * a_g).max()) / ((S0 + a_g) ** 2)
                else:
                    lo = a_g
                    a_g = b_g
                    f_a = f_b
                    b_g = lo + gr * (hi - lo)
                    f_b = 2.0 * n * float((c_base + d * b_g).max()) / ((S0 + b_g) ** 2)
            
            best_t = (lo + hi) / 2
            best_f = min(f_a, f_b)
            
            # Compare with current and zero
            candidates = [(a_i_old, f_cur), (0.0, f_0), (best_t, best_f)]
            best_cand = min(candidates, key=lambda x: x[1])
            
            if best_cand[1] < f_cur - 1e-12:
                a[i] = best_cand[0]
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1
            
            if improved % 200 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"  CD pass {p}: improved {improved}, score={sc:.10f}")
    
    return best_a, best_score


def _spectral_init(n, rng):
    """Initialize using spectral method: try to equalize |A(k)|^2 in frequency domain."""
    # Start with random phase, then project to non-negative
    phases = rng.uniform(0, 2 * np.pi, size=n)
    # Create a signal with flat spectrum (random phase)
    spec = np.exp(1j * phases)
    a = np.fft.irfft(spec, n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s


def _structured_init(n, kind="multi_cos"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "multi_cos":
        a = np.ones(n, dtype=np.float64)
        x = np.arange(n, dtype=np.float64) / n
        for freq, amp in [(1, 0.4), (2, 0.3), (3, 0.2), (5, 0.1), (7, 0.05)]:
            a += amp * np.cos(2 * np.pi * freq * x + freq * 0.7)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3 * (t - 0.5) ** 2) + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "multi_cos")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple lengths
    lengths_to_try = [4096, 8192, 2048]
    if parent is not None and len(parent) == 8192:
        lengths_to_try = [8192, 4096, 2048]
    elif parent is not None and len(parent) == 2048:
        lengths_to_try = [2048, 4096, 8192]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 90:
            break

        remaining = deadline - time.monotonic()
        print(f"\n{'='*60}")
        print(f"=== Trying n={n_try} (remaining: {remaining:.1f}s) ===")

        # Initialize
        if n_try == n_parent and parent is not None:
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)

        sc0 = _score(a0)
        print(f"  Initial: score={sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # --- Phase 1: Adam log-space with full softmax (coarse to fine) ---
        phase1_budget = min(remaining * 0.40, 400.0)
        phase1_end = time.monotonic() + phase1_budget

        c_curr = _full_convolve(a0, a0)
        max_cc = float(c_curr.max())

        # Temperature schedule: from coarse to fine
        betas = []
        beta = 0.05 * max_cc
        while beta > 0.00001 * max_cc:
            betas.append(beta)
            beta *= 0.5
        if len(betas) > 10:
            betas = betas[:10]

        a_curr = a0.copy()
        lr = 0.02
        for bi, T_val in enumerate(betas):
            if time.monotonic() > phase1_end - 5:
                break
            per_budget = min(phase1_end - time.monotonic() - 2, 60.0)
            max_it = max(5000, int(per_budget * 2000))
            a_curr, sc = _adam_logspace_full(a_curr, time.monotonic() + per_budget,
                                             [T_val] * 3, lr, max_it)
            a_curr = np.maximum(a_curr, 0)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  Adam phase1 T={T_val:.6f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            lr = max(0.005, lr * 0.7)

        # --- Phase 2: LP polishing ---
        phase2_budget = min(remaining * 0.25, 200.0)
        phase2_end = min(time.monotonic() + phase2_budget, deadline - 60)
        if phase2_budget > 30:
            print(f"  Phase 2: LP polishing (budget {phase2_budget:.1f}s)")
            a_lp, sc_lp = _lp_polish(a_curr, phase2_end, max_iters=12)
            sc_lp_full = _score(a_lp)
            if sc_lp_full < best_score:
                best_score = sc_lp_full
                best_seq = a_lp.copy()
                print(f"  LP polish: score={sc_lp_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_curr = a_lp

        # --- Phase 3: Fine Adam with very low temperature ---
        phase3_budget = min(remaining * 0.20, 180.0)
        phase3_end = min(time.monotonic() + phase3_budget, deadline - 60)
        if phase3_budget > 20:
            c_work = _full_convolve(a_curr, a_curr)
            max_cw = float(c_work.max())
            fine_betas = [0.001 * max_cw, 0.0003 * max_cw, 0.0001 * max_cw, 0.00003 * max_cw]
            a_fine, sc_fine = _adam_logspace_full(a_curr, phase3_end, fine_betas,
                                                  lr_init=0.008, max_iter_per_phase=100000)
            sc_fine_full = _score(a_fine)
            if sc_fine_full < best_score:
                best_score = sc_fine_full
                best_seq = a_fine.copy()
                print(f"  Fine Adam: score={sc_fine_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_curr = a_fine

        # --- Phase 4: Coordinate descent ---
        phase4_budget = min(remaining * 0.10, 100.0)
        phase4_end = min(time.monotonic() + phase4_budget, deadline - 30)
        if phase4_budget > 15:
            print(f"  Phase 4: Coordinate descent (budget {phase4_budget:.1f}s)")
            a_cd, sc_cd = _coord_descend_polish(a_curr, phase4_end, n_passes=4, top_k_w=100)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_curr = a_cd

        # --- Phase 5: Final LP if time permits ---
        if time.monotonic() < deadline - 40:
            final_budget = min(60.0, deadline - time.monotonic() - 30)
            a_final_lp, sc_final_lp = _lp_polish(a_curr, time.monotonic() + final_budget, max_iters=5)
            sc_final_full = _score(a_final_lp)
            if sc_final_full < best_score:
                best_score = sc_final_full
                best_seq = a_final_lp.copy()
                print(f"  Final LP: score={sc_final_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_curr = a_final_lp

            # Quick CD after final LP
            if time.monotonic() < deadline - 35:
                a_fin_cd, sc_fin_cd = _coord_descend_polish(a_curr, deadline - 30, n_passes=2, top_k_w=50)
                sc_fin_full = _score(a_fin_cd)
                if sc_fin_full < best_score:
                    best_score = sc_fin_full
                    best_seq = a_fin_cd.copy()
                    print(f"  Final CD: score={sc_fin_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Diversity restarts if time remains ---
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\n=== Diversity restarts (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()

        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 30) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 35:
                break
            if ri == 0:
                p_start = a_base + 0.03 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 512))
                p_start = np.roll(a_base, shift).copy()
            else:
                freq = rng.uniform(2.0, 10.0)
                pert = 0.02 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.002 * max_cs, 0.0005 * max_cs, 0.0001 * max_cs]
            p_best, sc_best = _adam_logspace_full(p_start, time.monotonic() + max(10, per_restart),
                                                  betas_r, lr_init=0.01, max_iter_per_phase=50000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final: ensure valid output
    print(f"\n{'='*60}")
    print(f"=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL OUTPUT: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
