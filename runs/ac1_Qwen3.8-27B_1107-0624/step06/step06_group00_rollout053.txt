def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _subgradient(a, k):
    """Compute subgradient of (a*a)[k] w.r.t. a"""
    n = len(a)
    g = np.zeros(n)
    # g[m] = 2 * a[k-m] if 0 <= k-m < n
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        g[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return g

def _nesterov_subgrad(a_init, deadline, max_iter=5000000, alpha=None):
    """Nesterov accelerated projected subgradient method"""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score_norm(a)
    eval_count = 0
    
    for t in range(1, max_iter + 1):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution at y (Nesterov lookahead)
        c = _fft_convolve(y, y)
        k_star = int(np.argmax(c))
        
        # Subgradient at y
        g = _subgradient(y, k_star)
        
        # Step size
        if alpha is not None:
            step = alpha / np.sqrt(t)
        else:
            step = 2.0 / (t + 2)  # 2/(L*(k+2)) with L=2
        
        # Gradient step
        z = y - step * g
        
        # Project to simplex
        a_new = _project_simplex(z)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new
        
        a_prev = a
        a = a_new
        
        # Periodic evaluation
        if t % 1000 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Final evaluation
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

        p = e / s
    # g[m] = 2 * sum_k p[k] * a[k-m] (valid indices only)
    # = 2 * ifft(fft(p, N) * conj(fft(a, N)))[:n]
    N = _next_pow2(len(p) + n - 1)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(P * np.conj(A), N)[:n]
    return 2.0 * cross


def _hard_subgradient(a, c):
    """Compute hard subgradient at the argmax of c."""
    n = len(a)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]
    return g


def _nesterov_smoothed(a_init, deadline, T_start, T_end, lr_base=0.03,
                        momentum_decay=0.9, eval_interval=500):
    """Nesterov-momentum method with smoothed gradient and adaptive LR."""
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score_norm(a)
    t = 0
    max_iter = 10000000

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_base / (1.0 + 0.01 * t)

        c = _fft_convolve(a, a)
        g = _smoothed_gradient_fft(a, c, T)

        # Momentum update
        v = momentum_decay * v + (1.0 - momentum_decay) * g
        # Normalized momentum step
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            step = v / v_norm * lr
        else:
            step = np.zeros(n)

        a_new = _project_simplex(a - step)
        a = a_new

        if t % eval_interval == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                momentum_decay = min(0.99, momentum_decay + 0.005)

    sc = _score_norm(a)
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _adam_zspace(a_init, deadline, T_start, T_end, lr_start, lr_end,
                 max_iter=5000000, active_set=False, active_k=5,
                 eval_interval=300):
    """Adam optimizer in z-space (log-parameterization) with softmax gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    patience = 0
    best_patience = best_sc
    clip_norm = 0.8

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max(1, max_iter))
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _fft_convolve(a, a)
        if active_set and T < T_start * 0.01:
            top_idx = np.argsort(c)[::-1][:active_k]
            p = np.zeros(len(c))
            p[top_idx] = 1.0 / active_k
            N = _next_pow2(len(p) + n - 1)
            P = np.fft.rfft(p, N)
            A = np.fft.rfft(a, N)
            cross = np.fft.irfft(P * np.conj(A), N)[:n]
            g = 2.0 * cross
        else:
            g = _smoothed_gradient_fft(a, c, T)

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 20 and t > 1000:
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                patience = 0

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _backtracking_subgrad(a_init, deadline, lr_init=0.1, c_bt=0.5, rho=0.5):
    """Projected subgradient with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_k = 1.0
    lr = lr_init
    best_a = a.copy()
    best_sc = _score_norm(a)
    t = 0
    max_iter = 5000000

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        c_y = _fft_convolve(y, y)
        k_star = int(np.argmax(c_y))
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        if lo <= hi:
            g[lo:hi + 1] = 2.0 * y[k_star - hi:k_star - lo + 1][::-1]

        # Backtracking
        f_y = float(c_y.max())
        a_new = _project_simplex(y - lr * g)
        f_new = _score_norm(a_new)
        # Lipschitz estimate for backtracking
        while f_new > f_y - 0.5 * lr * float(np.dot(g, y - a_new)) and lr > 1e-10:
            lr *= rho
            a_new = _project_simplex(y - lr * g)
            f_new = _score_norm(a_new)
        lr = min(lr * c_bt * 2.0, 0.5)  # don't increase too fast

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        t_k = t_new
        a = a_new

        if t % 2000 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _random_init(n, rng, strategy='exponential_decay'):
    """Generate diverse random initializations."""
    if strategy == 'uniform':
        a = np.ones(n)
    elif strategy == 'linear_ramp':
        a = np.linspace(0.1, 1.0, n)
    elif strategy == 'trapezoid':
        mid = n // 2
        a = np.zeros(n)
        a[:mid] = np.linspace(0.5, 1.0, mid)
        a[mid:] = np.linspace(1.0, 0.5, n - mid)
    elif strategy == 'bimodal':
        a = np.zeros(n)
        a[:n // 3] = 1.0
        a[2 * n // 3:] = 1.0
        a = a + 0.1 * np.random.default_rng(0).standard_normal(n)
        a = np.maximum(a, 0.001)
    elif strategy == 'sine_mod':
        j = np.arange(n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * j / n * 3) + 0.1 * np.sin(2 * np.pi * j / n * 7)
        a = np.maximum(a, 0.01)
    elif strategy == 'exponential_decay':
        a = np.exp(-np.linspace(0, 2, n))
        a = np.maximum(a, 0.001)
    else:
        a = rng.standard_normal(n)
        a = np.maximum(a, 0.001)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining Nesterov subgradient, Adam z-space,
    backtracking methods, multi-n exploration, and structured restarts.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a_parent = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        a_parent = np.ones(4096) / 4096

    n_parent = len(a_parent)
    print(f"Parent: n={n_parent}, score={_score(a_parent):.10f}")

    best_a = a_parent.copy()
    best_score = _score(best_a)
    print(f"Initial best: {best_score:.10f}")

    # ============ PHASE 1: Multi-n exploration with Nesterov (0-500s) ============
    phase1_end = min(deadline, t_start + 500.0)
    n_candidates = [n_parent]
    # Add nearby n values for exploration
    for dn in [-512, -256, 256, 512, 1024]:
        nc = n_parent + dn
        if 512 <= nc <= 16384 and nc != n_parent:
            n_candidates.append(nc)
    # Sort by proximity to parent n
    n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_parent))
    # Keep at most 4 different n values, prioritize parent
    n_candidates = n_candidates[:4]

    time_per_n = max(60.0, (phase1_end - t_start) / len(n_candidates))
    for n_try in n_candidates:
        if time.monotonic() > phase1_end:
            break
        phase_deadline = min(phase1_end, t_start + (time.monotonic() - t_start) + time_per_n)

        if n_try == n_parent:
            a_cur = a_parent.copy()
        else:
            a_cur = _interp_to_length(a_parent, n_try)

        c_cur = _fft_convolve(a_cur, a_cur)
        max_c = float(c_cur.max())

        # Run Nesterov with smoothed gradient
        a_opt, sc_norm = _nesterov_smoothed(
            a_cur, phase_deadline - 5,
            T_start=0.01 * max_c, T_end=0.00005 * max_c,
            lr_base=0.05, momentum_decay=0.9,
            eval_interval=200)
        full_sc = _score(a_opt)
        print(f"Phase1 nesterov n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # ============ PHASE 2: Adam z-space on best (500-1000s) ============
    phase2_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase2_end:
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())

        # Run multiple Adam configurations
        configs = [
            dict(T_start=0.005 * max_c, T_end=0.00005 * max_c,
                 lr_start=0.04, lr_end=0.004, active_set=False, active_k=5),
            dict(T_start=0.002 * max_c, T_end=0.00002 * max_c,
                 lr_start=0.03, lr_end=0.002, active_set=True, active_k=3),
            dict(T_start=0.001 * max_c, T_end=0.00001 * max_c,
                 lr_start=0.02, lr_end=0.001, active_set=True, active_k=5),
        ]
        time_per_config = max(50.0, (phase2_end - t_start) / len(configs))
        for i, cfg in enumerate(configs):
            if time.monotonic() > phase2_end:
                break
            cfg_deadline = min(phase2_end, t_start + (time.monotonic() - t_start) + time_per_config)
            a_opt, sc_norm = _adam_zspace(best_a, cfg_deadline - 3, max_iter=3000000,
                                          eval_interval=200, **cfg)
            full_sc = _score(a_opt)
            print(f"Phase2 adam[{i}] n={len(a_opt)}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

    # ============ PHASE 3: Backtracking subgradient (1000-1300s) ============
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end:
        n_best = len(best_a)
        a_opt, sc = _backtracking_subgrad(best_a, phase3_end - 5,
                                          lr_init=0.05, c_bt=0.7, rho=0.7)
        full_sc = _score(a_opt)
        print(f"Phase3 backtrack n={n_best}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_opt.copy()

    # ============ PHASE 4: Diverse restarts (1300-1600s) ============
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end:
        n_best = len(best_a)
        strategies = ['sine_mod', 'trapezoid', 'bimodal', 'linear_ramp']
        time_per_restart = max(40.0, (phase4_end - t_start) / (len(strategies) + 2))

        # Try perturbed versions of best
        for i in range(3):
            if time.monotonic() > phase4_end:
                break
            sigma = 0.01 + 0.01 * i
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = np.exp(z - z.max())
            a_pert = a_pert / a_pert.sum()
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            rd = min(phase4_end, t_start + (time.monotonic() - t_start) + time_per_restart)
            a_opt, sc = _adam_zspace(a_pert, rd - 3,
                                     T_start=0.001 * max_cp, T_end=0.00001 * max_cp,
                                     lr_start=0.02, lr_end=0.001,
                                     max_iter=2000000, active_set=True, active_k=3,
                                     eval_interval=150)
            full_sc = _score(a_opt)
            print(f"Phase4 perturb[{i}] sigma={sigma:.3f}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

        # Try different structure initializations
        for strat in strategies[:2]:
            if time.monotonic() > phase4_end:
                break
            a_rnd = _random_init(n_best, rng, strategy=strat)
            c_r = _fft_convolve(a_rnd, a_rnd)
            max_cr = float(c_r.max())
            rd = min(phase4_end, t_start + (time.monotonic() - t_start) + time_per_restart)
            a_opt, sc = _adam_zspace(a_rnd, rd - 3,
                                     T_start=0.01 * max_cr, T_end=0.0001 * max_cr,
                                     lr_start=0.04, lr_end=0.003,
                                     max_iter=2000000, active_set=False, active_k=5,
                                     eval_interval=200)
            full_sc = _score(a_opt)
            print(f"Phase4 strategy={strat}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

    # ============ PHASE 5: Cross-scale transfer (1600-1750s) ============
    phase5_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase5_end:
        n_cur = len(best_a)
        # Try larger n
        n_large = n_cur * 2 if n_cur <= 4096 else n_cur
        if n_large != n_cur and n_large <= 16384:
            a_large = _interp_to_length(best_a, n_large)
            c_l = _fft_convolve(a_large, a_large)
            max_cl = float(c_l.max())
            a_lo, sc_l = _adam_zspace(a_large, phase5_end - 5,
                                      T_start=0.002 * max_cl, T_end=0.00002 * max_cl,
                                      lr_start=0.03, lr_end=0.002,
                                      max_iter=2000000, active_set=True, active_k=3,
                                      eval_interval=200)
            full_sc_l = _score(a_lo)
            print(f"Phase5 n={n_large}: score={full_sc_l:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_l) and full_sc_l < best_score:
                best_score = full_sc_l
                best_a = a_lo.copy()
            # Interpolate back
            a_back = _interp_to_length(a_lo, n_cur)
            c_b = _fft_convolve(a_back, a_back)
            max_cb = float(c_b.max())
            a_bo, sc_b = _adam_zspace(a_back, phase5_end - 5,
                                      T_start=0.001 * max_cb, T_end=0.00001 * max_cb,
                                      lr_start=0.02, lr_end=0.001,
                                      max_iter=1000000, active_set=True, active_k=3,
                                      eval_interval=150)
            full_sc_b = _score(a_bo)
            print(f"Phase5 interp back n={n_cur}: score={full_sc_b:.10f}")
            if np.isfinite(full_sc_b) and full_sc_b < best_score:
                best_score = full_sc_b
                best_a = a_bo.copy()

    # ============ PHASE 6: Final ultra-fine polish (1750-deadline) ============
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_f = _adam_zspace(best_a, deadline - 5,
                                     T_start=0.0001 * max_cf, T_end=0.000005 * max_cf,
                                     lr_start=0.005, lr_end=0.0003,
                                     max_iter=5000000,
                                     active_set=True, active_k=2,
                                     eval_interval=100)
        full_sc = _score(a_final)
        print(f"Phase6 final: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # One more Nesterov pass on the best
        if time.monotonic() < deadline - 15:
            a_nest, sc_n = _nesterov_smoothed(best_a, deadline - 10,
                                              T_start=0.0001 * max_cf,
                                              T_end=0.000002 * max_cf,
                                              lr_base=0.01, momentum_decay=0.99,
                                              eval_interval=200)
            full_sc_n = _score(a_nest)
            print(f"Phase6 nesterov: score={full_sc_n:.10f}")
            if np.isfinite(full_sc_n) and full_sc_n < best_score:
                best_score = full_sc_n
                best_a = a_nest.copy()

    # Final validation and return
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
