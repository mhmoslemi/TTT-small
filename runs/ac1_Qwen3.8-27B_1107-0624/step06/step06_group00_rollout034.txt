    fb = np.fft.rfft(b, N)

def _soft_gradient(a, c, T):
    """Soft gradient using temperature T."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    N = _next_pow2(3 * n)
    a_rev = a[::-1]
    Ar = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    result = np.fft.irfft(Ar * P, N)
    g = result[n-1:2*n-1]
    return 2.0 * n * g


def _optimize(a, deadline, T_start, T_end, lr_start, lr_end,
              max_iter=500000, k_active=5, momentum=0.9,
              eval_interval=200, symmetrize=False, symm_interval=500):
    """Heavy-ball momentum optimization in z-space with active-set gradient."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)  # momentum buffer
    best_a = a.copy()
    best_score = _score(a)
    patience = 0
    t = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        a_cur = _softmax_stable(z)
        c = _fft_convolve(a_cur, a_cur)
        
        # Compute gradient
        if T < T_start * 0.01:
            g = _active_gradient(a_cur, c, k=k_active)
        else:
            g = _soft_gradient(a_cur, c, T)
        
        # Center the gradient (remove mean for invariance to z-shifting)
        W = float(np.dot(a_cur, g))
        z_grad = a_cur * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        # Heavy-ball momentum
        v = momentum * v - lr * z_grad
        z = z + v
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Symmetrization
        if symmetrize and t % symm_interval == 0:
            z = (z + z[::-1]) / 2.0
            z = z - z.mean()
        
        # Periodic evaluation
        if t % eval_interval == 0:
            a_eval = _softmax_stable(z)
            sc = _score(a_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_eval.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and lr > lr_end * 1.1:
                # Restart momentum, reduce LR
                v = np.zeros(n)
                patience = 0
    
    a_final = _softmax_stable(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _lp_direction(a, M, n, time_limit=30):
    """
    Find a direction d >= 0 that reduces the max convolution.
    Solve: max sum(d) s.t. conv(a+d, a) <= M, d >= 0
    (This is a linear program in d.)
    """
    import scipy.sparse as sp
    from scipy.optimize import linprog
    
    # conv(a+d, a) = conv(a,a) + conv(d, a)
    # We want conv(a,a) + conv(d,a) <= M for all positions
    # i.e., conv(d, a) <= M - conv(a, a) for all positions
    
    c = _fft_convolve(a, a)
    rhs = M - c  # Must be non-negative for feasibility; if negative, infeasible
    
    if np.any(rhs < -1e-10):
        # Infeasible - return None
        return None
    
    rhs = np.maximum(rhs, 0)
    
    # Build constraint matrix: A @ d <= rhs
    # A[k, j] = a[k-j] + a[j-k] ... wait
    # conv(d, a)[k] = sum_j d[j] * a[k-j]
    # So A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # This is a (2n-1) x n matrix
    rows_list = []
    cols_list = []
    vals_list = []
    for j in range(n):
        for k in range(j, j + n):
            # conv(d,a)[k] includes d[j] * a[k-j]
            if k < 2*n - 1:
                rows_list.append(k)
                cols_list.append(j)
                vals_list.append(a[k-j])
    
    if not rows_list:
        return None
    
    A = sp.csc_matrix((vals_list, (rows_list, cols_list)), shape=(2*n-1, n))
    c_obj = -np.ones(n)  # minimize -sum(d) = maximize sum(d)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=rhs, bounds=bounds,
                     method='highs', options={'time_limit': time_limit})
        if res.status != 0:
            return None
        d = np.maximum(res.x, 0)
        if d.sum() < 1e-15:
            return None
        return d
    except:
        return None


def _lp_step(a, max_iter=50, time_limit=20):
    """Apply LP direction to reduce the score."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for _ in range(max_iter):
        c = _fft_convolve(a, a)
        M = float(c.max())
        d = _lp_direction(a, M, n, time_limit=time_limit)
        if d is None:
            break
        d = d / d.sum()  # normalize
        
        # Line search
        best_t = 0.0
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.05, 0.15]:
            a_cand = (1 - t) * a + t * d
            sc = _score(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
                best_a = a_cand.copy()
        
        if best_t == 0.0:
            break
        a = best_a
    
    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_parabolic(n, p=1.0):
    x = np.linspace(-1, 1, n)
    a = np.maximum(0, 1 - x**2) ** p
    a = a / a.sum()
    return a


def _make_sine(n):
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x)
    a = a / a.sum()
    return a


def _make_u_shape(n):
    x = np.linspace(-1, 1, n)
    a = 1 + x**2  # "U" shape - large at ends
    a = a / a.sum()
    return a


def _make_w_shape(n):
    x = np.linspace(0, 1, n)
    a = 1 + 0.5 * np.cos(4 * np.pi * x)  # "W" shape
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a
