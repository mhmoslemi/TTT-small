def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_subgradient(a, c, k_active=5):
    """Compute average subgradient of max(c) w.r.t. a, where c=conv(a,a)."""
    n = len(a)
    len_c = len(c)
    k = min(k_active, len_c)
    if k >= len_c:
        top_idx = np.arange(len_c)
    else:
        top_idx = np.argpartition(c, -(len_c - k))[-k:] if len_c - k > 0 else np.arange(len_c)
    
    grad = np.zeros(n)
    for ki in top_idx:
        ki = int(ki)
        i_start = max(0, ki - n + 1)
        i_end = min(ki + 1, n)
        if i_start < i_end:
            m = ki - np.arange(i_start, i_end)
            grad[i_start:i_end] += a[m]
    grad *= 2.0 / k
    return grad


def _compute_soft_gradient(a, c, T):
    """Compute soft gradient using softmax smoothing of max."""
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    
    # grad[i] = 2 * sum_k p[k] * a[k-i] (clipped)
    N = _next_pow2(4 * n)
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # We want: sum_k p[k]*a[k-i] for i=0..n-1
    # This is the cross-correlation: symmetric convolution
    # result[i] = sum_k p[k] * a[k-i] = (a * p_flipped)[i + offset]
    # p_flipped[k] = p[len_c-1-k], then convolve with a
    p_flipped = p[::-1]
    conv_result = np.fft.irfft(A * np.fft.rfft(p_flipped, N), N)
    # Index: result[i] = sum_m a[m] * p_flipped[i-m] = sum_m a[m] * p[len_c-1-i+m]
    # We want sum_k p[k]*a[k-i]. Let k = m, sum_m p[m]*a[m-i]

    a_rev = a[::-1]

    return grad


def _compute_subgradient_direct(a, c, k_active=5):

    n = len(a)
    len_c = len(c)
    k = min(k_active, len_c)

def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, use_active=True, active_k_start=10, active_k_end=3,
                  eval_interval=200, clip_norm=0.5, momentum=0.9):
    """Adam in z-space with adaptive active set subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    v_buf[:] = 1e-8  # Avoid division by zero
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience_score = best_score
    current_lr_mult = 1.0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        # Cosine annealing
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        
        # Adaptive active set size
        k_cur = int(active_k_start + (active_k_end - active_k_start) * frac)
        k_cur = max(1, k_cur)
        
        if use_active:
            g = _compute_subgradient_direct(a, c, k_cur)
        else:
            g = _compute_soft_gradient(a, c, T)
        
        # Project gradient to simplex tangent space
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 8 and current_lr_mult > 0.1:
                current_lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf[:] = 1e-8

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_step(a, rows, cols, time_limit=15):
    """LP-based direction finding."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t == 0.0:
            for t_val in [0.02, 0.05, 0.08, 0.15]:
                a_cand = (1.0 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _local_search(a, deadline, max_rounds=20, window=3, step_frac=0.002):
    """Local search: try shifting mass between nearby positions."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = 0
        # Try random pairs of positions
        num_trials = min(n * 2, 5000)
        for _ in range(num_trials):
            if time.monotonic() > deadline:
                break
            i = rng.integers(0, n)
            # Try moving to a neighbor
            j = int(rng.integers(max(0, i - window), min(n, i + window + 1)))
            if i == j:
                continue
            # Try small transfer
            eps = step_frac * best_a[i] * best_a[j] / (best_a[i] + best_a[j] + 1e-15)
            if eps < 1e-12 or eps > best_a[i] * 0.5:
                continue
            a_try = best_a.copy()
            a_try[i] -= eps
            a_try[j] += eps
            if a_try[i] < 0:
                continue
            sc = _score_norm(a_try)
            if sc < best_sc - 1e-12:
                best_a = a_try
                best_sc = sc
                improved += 1
        if improved == 0:
            # Try larger steps
            step_frac *= 0.7
        else:
            # Progress made, try slightly larger steps occasionally
            step_frac = min(step_frac * 1.1, 0.02)
    
    return best_a, best_sc


def _local_search_greedy(a, deadline, max_rounds=5):
    """Greedy local search: for each position, find best neighbor to exchange with."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, 4):  # Every 4th position for speed
            if time.monotonic() > deadline:
                break
            best_move = None
            best_move_sc = best_sc
            for dj in [-2, -1, 1, 2, 4, -4]:
                j = i + dj
                if j < 0 or j >= n:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1]:
                    eps = frac * best_a[i]
                    if eps >= best_a[i]:
                        continue
                    a_try = best_a.copy()
                    a_try[i] -= eps
                    a_try[j] += eps
                    sc = _score_norm(a_try)
                    if sc < best_move_sc - 1e-10:
                        best_move_sc = sc
                        best_move = (i, j, eps)
            if best_move is not None:
                i, j, eps = best_move
                best_a[i] -= eps
                best_a[j] += eps
                best_sc = best_move_sc
                improved = True
        if not improved:
            break
    
    return best_a, best_sc


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    if n0 == 0:
        return np.ones(n_new) / n_new
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _cosine_schedule(start, end, frac):
    return end + 0.5 * (start - end) * (1 + np.cos(np.pi * frac))

                best_score = full_sc
                best_a = a_ls2.copy()

    # Phase 4: Multi-restart exploration (650s-1250s)
    phase4_end = min(deadline, t_start + 1250.0)
    n_restarts = 4
    per_restart = (phase4_end - time.monotonic() - 10) / n_restarts if time.monotonic() < phase4_end - 10 else 0
    print(f"Phase 4: {n_restarts} restarts, ~{per_restart:.0f}s each")
    
    for ridx in range(n_restarts):
        if time.monotonic() > phase4_end - 10 or per_restart < 20:
            break
        restart_deadline = min(time.monotonic() + per_restart, phase4_end - 5)
        
        # Different initialization strategies
        if ridx == 0:
            # Sinusoidal modulation
            x = np.arange(n)
            a_init = np.ones(n) + 0.5 * np.sin(2 * np.pi * x / (n // 3)) + 0.3 * np.sin(2 * np.pi * x / (n // 7))
            a_init = np.maximum(a_init, 0.01)
        elif ridx == 1:
            # Asymmetric ramp
            x = np.arange(n) / n
            a_init = 1.0 + 0.8 * np.sin(np.pi * x ** 0.7)
            a_init = np.maximum(a_init, 0.01)
        elif ridx == 2:
            # Perturbed best
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + 0.02 * rng.standard_normal(n)
            z = z - z.mean()
            a_init = _softmax_stable(z)
        else:
            # Sparse peaks on uniform base
            a_init = np.full(n, 0.3)
            num_peaks = 7
            peak_positions = rng.choice(n, size=num_peaks, replace=False)
            for pp in peak_positions:
                a_init[pp] = 1.5
                if pp > 0:
                    a_init[pp-1] = 0.8
                if pp < n-1:
                    a_init[pp+1] = 0.8
        
        a_init = a_init / a_init.sum()
        
        c_init = _fft_convolve(a_init, a_init)
        max_ci = float(c_init.max())
        
        adam_deadline = restart_deadline - max(10, (restart_deadline - time.monotonic()) * 0.2)
        a_r, sc_r = _adam_z_space(a_init, adam_deadline,
                                   T_start=0.005 * max_ci, T_end=0.00005 * max_ci,
                                   lr_start=0.04, lr_end=0.003,
                                   max_iter=200000, use_active=True,
                                   active_k_start=10, active_k_end=3,
                                   eval_interval=150, clip_norm=0.4)
        full_sc = _score(a_r)
        
        # Quick local search
        a_r2, sc_r2 = _local_search(a_r, restart_deadline, max_rounds=5, window=3, step_frac=0.002)
        full_sc2 = _score(a_r2)
        
        best_r_sc = min(full_sc, full_sc2)
        best_r_a = a_r if full_sc < full_sc2 else a_r2
        print(f"  Restart[{ridx}]: adam={full_sc:.10f}, ls={full_sc2:.10f}, best={best_r_sc:.10f} t={time.monotonic()-t_start:.0f}s")
        
        if best_r_sc < best_score:
            best_score = best_r_sc
            best_a = best_r_a.copy()
            print(f"  NEW BEST: {best_score:.10f}")

    # Phase 5: Fine polish on best (1250s-1550s)
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end:
        c5 = _fft_convolve(best_a, best_a)
        max_c5 = float(c5.max())
        a_fin, sc_fin = _adam_z_space(best_a, phase5_end - 5,
                                       T_start=0.0005 * max_c5, T_end=0.000005 * max_c5,
                                       lr_start=0.015, lr_end=0.001,
                                       max_iter=300000, use_active=True,
                                       active_k_start=5, active_k_end=2,
                                       eval_interval=150, clip_norm=0.3)
        full_sc = _score(a_fin)
        print(f"Phase 5 FinePolish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fin.copy()
        
        # Local search after fine polish
        if time.monotonic() < phase5_end:
            a_ls, sc_ls = _local_search(best_a, phase5_end - 3, max_rounds=8, window=2, step_frac=0.001)
            full_sc = _score(a_ls)
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_ls.copy()
                print(f"Phase 5 FinalLS: score={full_sc:.10f}")

    # Phase 6: Try different sizes (1550s-1750s)
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end:
        # Try n=2048
        n2 = 2048
        a2 = _interp_to_length(best_a, n2)
        c2 = _fft_convolve(a2, a2)
        max_c2 = float(c2.max())
        adam_dl = min(time.monotonic() + 60, phase6_end - 30)
        a2_opt, _ = _adam_z_space(a2, adam_dl,
                                   T_start=0.001 * max_c2, T_end=0.00001 * max_c2,
                                   lr_start=0.02, lr_end=0.002,
                                   max_iter=100000, use_active=True,
                                   active_k_start=5, active_k_end=2,
                                   eval_interval=100, clip_norm=0.3)
        sc2 = _score(a2_opt)
        print(f"Phase 6 n={n2}: score={sc2:.10f}")
        if sc2 < best_score:
            best_score = sc2
            best_a = a2_opt.copy()
            print(f"  NEW BEST (n={n2}): {best_score:.10f}")
        
        # Try n=8192
        if time.monotonic() < phase6_end - 40:
            n3 = 8192
            a3 = _interp_to_length(best_a, n3)
            c3 = _fft_convolve(a3, a3)
            max_c3 = float(c3.max())
            adam_dl3 = min(time.monotonic() + 60, phase6_end - 5)
            a3_opt, _ = _adam_z_space(a3, adam_dl3,
                                       T_start=0.001 * max_c3, T_end=0.00001 * max_c3,
                                       lr_start=0.02, lr_end=0.002,
                                       max_iter=100000, use_active=True,
                                       active_k_start=5, active_k_end=2,
                                       eval_interval=100, clip_norm=0.3)
            sc3 = _score(a3_opt)
            print(f"Phase 6 n={n3}: score={sc3:.10f}")
            if sc3 < best_score:
                best_score = sc3
                best_a = a3_opt.copy()
                print(f"  NEW BEST (n={n3}): {best_score:.10f}")

    # Phase 7: Final ultra-fine polish (1750s-deadline)
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, _ = _adam_z_space(best_a, deadline - 8,
                                    T_start=0.0001 * max_cf, T_end=0.000001 * max_cf,
                                    lr_start=0.005, lr_end=0.0003,
                                    max_iter=200000, use_active=True,
                                    active_k_start=3, active_k_end=1,
                                    eval_interval=100, clip_norm=0.2)
        full_sc = _score(a_final)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()
        
        # One more local search
        if time.monotonic() < deadline - 3:
            a_ls, _ = _local_search(best_a, deadline - 2, max_rounds=3, window=2, step_frac=0.0005)
            full_sc = _score(a_ls)
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_ls.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
