- For all k in S: c_k = t (binding)

If we assume all ν_k are positive (all constraints binding), and if the active set is "small" (say |S| = m), we can try to find a such that:

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    """Compute gradient using active set (top elements of convolution)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_equalize(a, deadline, max_iter=2000, tol=1e-12, min_active=2, max_active=80):
    """
    Multiplicative update to equalize active convolution values.
    Solves the KKT conditions: for j in support, sum_{k in active} a[k-j] = constant.
    """
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    current_score = best_sc

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Adaptive active set: top elements within tolerance
        c_sorted = np.sort(c)[::-1]
        # Use a threshold that captures the "peaks"
        base_tol = 0.005 * c_max
        active = c >= (c_max - base_tol)
        n_active = int(active.sum())

        # Cap the active set size
        if n_active > max_active:
            idx = np.argsort(c)[::-1][:max_active]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = max_active
        elif n_active < min_active:
            idx = np.argsort(c)[::-1][:min_active]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            n_active = min_active

        # Compute h[j] = sum_{k in active} a[k-j] for j = 0..n-1
        h = np.zeros(n)
        active_indices = np.where(active)[0]
        for k in active_indices:
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min <= j_max:
                # h[j] += a[k-j] for j in [j_min, j_max]
                # a[k-j] for j=j_min gives a[k-j_min], for j=j_max gives a[k-j_max]
                # So the slice is a[k-j_max : k-j_min+1]
                h[j_min:j_max + 1] += a[k - j_max:k - j_min + 1]

        # Multiplicative update
        support = a > 1e-8
        if support.sum() < 2:
            break
        h_sup = h[support]
        target = h_sup.mean()
        if target < 1e-30:
            break

        ratio = target / np.maximum(h_sup, 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        a_new = a.copy()
        a_new[support] *= ratio
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        # Check convergence
        change = np.max(np.abs(a_new - a))
        a = a_new

        if it % 20 == 0:
            sc = _score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
            if change < tol:
                break

    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    return best_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient method with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.002)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=10.0, verbose=False):
    """L-BFGS with adaptive beta-continuation (improved schedule)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Gap-based initial beta
    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    # More aggressive beta schedule
    betas = base_beta * np.array([1.0, 0.5, 0.3, 0.15, 0.08, 0.04,
                                  0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        # Estimate time per L-BFGS iteration
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time * 15))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
                if verbose:
                    print(f"    beta={beta:.6f}, score={sc_new:.10f}, iter={result.nit}")
        except Exception:
            pass

    return current_a


def _shrinkage_aggressive(a, deadline, max_rounds=5):
    """Aggressive support reduction: zero small elements, re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    for round_num in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Try zeroing bottom 1%, 3%, 5%
        fractions = [0.01, 0.03, 0.05]
        for frac in fractions:
            if time.monotonic() > deadline:
                break
            n_zero = max(1, int(n * frac))
            a_try = best_a.copy()
            a_sorted_idx = np.argsort(a_try)
            a_try[a_sorted_idx[:n_zero]] = 0.0
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-10:
                best_sc = sc_try
                best_a = a_try.copy()

        # Re-optimize after shrinkage
        if time.monotonic() < deadline - 5:
            a_opt = _run_lbfgs(best_a, deadline, min_time_remaining=5.0)
            sc_opt = _score(a_opt)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt

    return best_a, best_sc


def _make_chirp(n, freq=1.0, rate=2.0):
    """Create a chirp-like initialization."""
    t = np.arange(n) / n
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t + rate * np.pi * t * t)
    a = np.maximum(a, 0.1)
    return a / a.sum()


def _make_cos_pattern(n, n_harmonics=3):
    """Create a multi-harmonic initialization."""
    t = np.arange(n) / n
    a = np.ones(n)
    for k in range(1, n_harmonics + 1):
        a += 0.15 / k * np.cos(2.0 * np.pi * k * t + k * 0.7)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def _make_random_structured(n, rng, n_blocks=8):
    """Create a block-structured random initialization."""
    block_size = n // n_blocks
    a = np.ones(n)
    for i in range(n_blocks):
        scale = rng.uniform(0.5, 1.8)
        start = i * block_size
        end = min(n, (i + 1) * block_size)
        a[start:end] = scale + rng.normal(0, 0.05, end - start)
    a = np.maximum(a, 0.05)
    return a / a.sum()


def _resample(a, n_new):
    """Resample a sequence to a new length using interpolation."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _nested_active_optimize(a, deadline):
    """
    Nested active-set optimization:
    Iteratively grow the active set and equalize.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    # Start with small active set, grow it
    for n_active_target in [2, 4, 8, 16, 32, 48, 64, 80]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        # Run active set equalization
        a_eq = _active_set_equalize(best_a, deadline, max_iter=500,
                                     min_active=n_active_target, max_active=n_active_target)
        sc_eq = _score(a_eq)
        if sc_eq < best_sc - 1e-12:
            best_sc = sc_eq
            best_a = a_eq
            # Quick L-BFGS polish
            if time.monotonic() < deadline - 10:
                a_pol = _run_lbfgs(best_a, deadline, min_time_remaining=5.0)
                sc_pol = _score(a_pol)
                if sc_pol < best_sc:
                    best_sc = sc_pol
                    best_a = a_pol

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses L-BFGS + active-set equalization + shrinkage + multi-n + restructuring.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Quick L-BFGS on parent ---
    print(f"\n[Phase 1] Quick L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = min(deadline, t_start + 300)
    a_p1 = _run_lbfgs(parent, phase1_deadline, min_time_remaining=500.0, verbose=True)
    update_best(a_p1, _score(a_p1), "p1_lbfgs")

    # --- Phase 2: Active-set equalization ---
    print(f"\n[Phase 2] Active-set equalization (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        phase2_deadline = min(deadline - 400, t_start + 600)
        a_p2, sc_p2 = _nested_active_optimize(best_seq, phase2_deadline)
        update_best(a_p2, sc_p2, "p2_active")

    # --- Phase 3: Shrinkage ---
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        phase3_deadline = min(deadline - 300, t_start + 800)
        a_p3, sc_p3 = _shrinkage_aggressive(best_seq, phase3_deadline, max_rounds=4)
        update_best(a_p3, sc_p3, "p3_shrink")

    # --- Phase 4: Subgradient polish ---
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        phase4_deadline = min(deadline - 200, t_start + 1000)
        a_p4, sc_p4 = _subgrad_polish(best_seq, phase4_deadline, n_max_steps=5000, lr_init=0.02)
        update_best(a_p4, sc_p4, "p4_sg")

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120, 6144]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates):
        if time.monotonic() > deadline - 150:
            break
        if n_try == n_parent:
            a_init = parent.copy()
        else:
            a_init = _resample(parent, n_try)

        # Try structured initializations too
        sc_init = _score(a_init)
        chirp_init = _make_chirp(n_try, freq=1, rate=3)
        sc_chirp = _score(chirp_init)
        cos_init = _make_cos_pattern(n_try, n_harmonics=4)
        sc_cos = _score(cos_init)

        # Pick best initialization
        if sc_chirp < sc_init:
            a_init = chirp_init
            sc_init = sc_chirp
        if sc_cos < sc_init:
            a_init = cos_init
            sc_init = sc_cos

        print(f"    n={n_try}, init_score={sc_init:.10f}")

        # L-BFGS optimization
        phase_deadline = min(deadline - 100, time.monotonic() + 120)
        a_opt = _run_lbfgs(a_init, phase_deadline, min_time_remaining=50.0)
        update_best(a_opt, _score(a_opt), f"n_{n_try}")

        # Active-set polish
        if time.monotonic() < deadline - 50:
            a_act, sc_act = _nested_active_optimize(best_seq if best_seq is not None and len(best_seq) == n_try else a_opt,
                                                     min(deadline - 30, time.monotonic() + 40))
            if len(a_act) == n_try:
                update_best(a_act, _score(a_act), f"n_{n_try}_act")

    # --- Phase 6: Diverse restarts ---
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n_best = len(best_seq)
        restarts = [
            ("chirp_mix", _make_chirp(n_best, freq=2, rate=1.5) * 0.3 + best_seq * 0.7),
            ("cos_mix", _make_cos_pattern(n_best, 5) * 0.3 + best_seq * 0.7),
            ("random_struct", _make_random_structured(n_best, rng, n_blocks=16)),
        ]
        for name, a_start in restarts:
            if time.monotonic() > deadline - 30:
                break
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
            sc_start = _score(a_start)
            print(f"    {name}: init_score={sc_start:.10f}")
            a_r = _run_lbfgs(a_start, min(deadline - 20, time.monotonic() + 80), min_time_remaining=15.0)
            update_best(a_r, _score(a_r), f"restart_{name}")

    # --- Phase 7: Final intensive polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very fine beta L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0003
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            max_iter = min(5000, max(100, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda zz: _smoobj(zz, beta_final, n),
                    z,
                    jac=lambda zz: _smograd(zz, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final active-set equalization
        if time.monotonic() < deadline - 3:
            a_eq_final = _active_set_equalize(best_seq, deadline - 3, max_iter=1000,
                                               min_active=4, max_active=100)
            update_best(a_eq_final, _score(a_eq_final), "final_active")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_sg_final, sc_sg_final = _subgrad_polish(best_seq, deadline - 2,
                                                       n_max_steps=500, lr_init=0.005)
            update_best(a_sg_final, sc_sg_final, "final_sg")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
