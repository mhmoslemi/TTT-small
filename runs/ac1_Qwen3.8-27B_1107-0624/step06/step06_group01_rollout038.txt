    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            e = np.exp(z - z.max())
            s = e.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a = e / s
            
            c = _fft_convolve(a, a)
            cm = float(c.max())
            shifted = (c - cm) / beta
            w = np.exp(shifted)
            ws = w.sum()
            if ws < 1e-300 or not np.isfinite(ws):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / ws
            
            # Gradient of 2n*sum_m w[m]*c[m] w.r.t. z
            # d(c[m])/d(a[j]) = a[m-j] (approx)
            # d(a[j])/d(z[j]) = a[j]*(1-a[j])
            # d(a[j])/d(z[l]) = -a[j]*a[l] for j != l
            # So d(c[m])/d(z) involves all z's
            # Simplify: grad_z[j] = a[j] * (1 - a[j]) * d(c[m])/d(a[j]) - a[j] * sum_l a[l] * d(c[m])/d(a[l])
            # = a[j] * (1 - a[j]) * g[j] - a[j] * sum_l a[l] * g[l]
            # where g = d(c[m])/d(a)
            
            g = np.zeros(n)
            for m in np.argsort(w)[::-1][:min(n, int(0.02 * 2 * n))]:
                if w[m] < 1e-10:
                    break
                i_min = max(0, m - n + 1)
                i_max = min(n - 1, m)
                lo = m - i_max
                hi = m - i_min
                g[i_min:i_max+1] += w[m] * a[lo:hi+1][::-1]
            
            W = float(np.dot(a, g))
            grad_z = 4.0 * n * a * ((1 - a) * g - W)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 1.0:
                update = update / un
            
            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                e2 = np.exp(z - z.max())
                s2 = e2.sum()
                if s2 > 1e-300:
                    a_chk = e2 / s2
                    sc = _score_unnorm(a_chk)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_chk.copy()
            
            if it_in_phase % 10000 == 0:
                lr = max(0.0003, lr * 0.9)
    
    e2 = np.exp(z - z.max())
    s2 = e2.sum()
    if s2 > 1e-300:
        a_final = e2 / s2
        sc = _score_unnorm(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
    
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        a = np.linspace(1.0, 0.1, n)
        return a / a.sum()
    elif kind == "ramp_reverse":
        a = np.linspace(0.1, 1.0, n)
        return a / a.sum()
    elif kind == "asymmetric_bump":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-30.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "linear_fade":
        a = np.linspace(1.0, 0.01, n)
        return a / a.sum()
    elif kind == "step":
        a = np.ones(n)
        a[:n//2] = 1.5
        a[n//2:] = 0.5
        return a / a.sum()
    elif kind == "sawtooth":
        a = np.arange(1, n+1, dtype=float)
        a = a / a.max()
        a[1:] = 1.0 - 0.8 * a[1:]
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()


def _coord_descent(a, deadline, n_steps=5000):
    """Coordinate descent: try scaling individual elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_unnorm(a)
    rng = np.random.default_rng(12345)
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        i = rng.integers(0, n)
        best_local = best_score
        best_factor = 1.0
        for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
            a_try = best_a.copy()
            a_try[i] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_unnorm(a_try)
                if sc < best_local:
                    best_local = sc
                    best_factor = factor
        if best_factor != 1.0:
            best_a[i] *= best_factor
            best_a = np.maximum(best_a, 0.0)
            s = best_a.sum()
            if s > 1e-15:
                best_a = best_a / s
            best_score = best_local
    
    return best_a, best_score


def _hot_spot_reduce(a, deadline, n_iters=5000):
    """Directly reduce the hot spots of the convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_unnorm(a)
    rng = np.random.default_rng(99)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(best_a, best_a)
        max_c = float(c.max())
        # Find the hottest convolution index
        m_star = int(np.argmax(c))
        
        # The gradient at m_star: grad[j] = a[m_star - j]
        # To reduce c[m_star], decrease a[j] where a[m_star-j] is large
        i_min = max(0, m_star - n + 1)
        i_max = min(n - 1, m_star)
        
        # Try reducing a random element in the hot region
        j = i_min + rng.integers(0, i_max - i_min + 1)
        for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
            a_try = best_a.copy()
            a_try[j] *= factor
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                sc = _score_unnorm(a_try)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_try.copy()
                    break
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}")
        best_seq = parent.copy()
        best_score = _score(parent)
    else:
        print("No parent available, starting from scratch.")
        a0 = _make_profile(4096, "uniform")
        best_seq = a0.copy()
        best_score = _score(a0)
        print(f"Initial uniform: score={best_score:.10f}")
    
    # Define lengths to try
    lengths_to_try = [4096]
    if parent is not None:
        n_p = len(parent)
        lengths_to_try = [n_p]
        if n_p <= 4096:
            lengths_to_try.extend([8192, 2048])
        elif n_p <= 8192:
            lengths_to_try.extend([n_p * 2, n_p // 2])
    else:
        lengths_to_try = [4096, 8192, 2048]
    
    # Deduplicate
    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    
    # Time allocation
    time_per_n = (deadline - t_start - 120.0) / len(lengths_to_try)
    
    for n_target in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break
        print(f"\n=== Optimizing n={n_target} (time_alloc={time_per_n:.0f}s, remaining={deadline-time.monotonic():.0f}s) ===")
        
        best_n = None
        best_n_score = np.inf
        
        # Initializations to try
        inits = []
        if parent is not None:
            inits.append(("parent", _interp_profile(parent, n_target)))
        inits.append(("ramp", _make_profile(n_target, "ramp")))
        inits.append(("asym_bump", _make_profile(n_target, "asymmetric_bump")))
        inits.append(("linear_fade", _make_profile(n_target, "linear_fade")))
        inits.append(("step", _make_profile(n_target, "step")))
        if n_target <= 8192:
            inits.append(("sawtooth", _make_profile(n_target, "sawtooth")))
        
        # Limit number of inits based on time
        max_inits = min(len(inits), max(2, int(time_per_n / 30)))
        
        for init_name, a_init in inits[:max_inits]:
            if time.monotonic() > deadline - 120:
                break
            
            a_work = _proj_nonneg_sum1(a_init.copy())
            
            # Phase A: FISTA with subgradient
            phase_deadline = min(deadline - 120, time.monotonic() + time_per_n * 0.4)
            a_opt, sc_opt = _fista_subgrad(a_work, phase_deadline, max_iter=30000)
            sc_full = _score(a_opt)
            if sc_full < best_n_score:
                best_n_score = sc_full
                best_n = a_opt.copy()
            print(f"  {init_name} FISTA: score={best_n_score:.10f}")
            
            # Phase B: Adam log-space fine optimization
            if time.monotonic() < deadline - 120 and best_n is not None:
                a_work = best_n.copy() if best_n is not None else _proj_nonneg_sum1(a_init.copy())
                c = _fft_convolve(a_work, a_work)
                max_c = float(c.max())
                betas = [0.003 * max_c, 0.001 * max_c, 0.0003 * max_c, 0.0001 * max_c]
                adam_deadline = min(deadline - 120, time.monotonic() + time_per_n * 0.35)
                a_opt2, sc_opt2 = _adam_logspace(a_work, adam_deadline, betas, lr_init=0.012, max_iter=100000)
                sc_full2 = _score(a_opt2)
                if sc_full2 < best_n_score:
                    best_n_score = sc_full2
                    best_n = a_opt2.copy()
                print(f"  {init_name} Adam: score={best_n_score:.10f}")
        
        # Update global
        if best_n is not None and best_n_score < best_score:
            best_score = best_n_score
            best_seq = best_n.copy()
            print(f"  NEW GLOBAL BEST: n={n_target}, score={best_score:.10f}")
    
    # Final polishing on the best found
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Final polishing (current best={best_score:.10f}) ===")
        a_pol = best_seq.copy()
        a_pol = _proj_nonneg_sum1(a_pol)
        
        # Ultra-fine Adam
        c = _fft_convolve(a_pol, a_pol)
        max_c = float(c.max())
        betas_ult = [0.00005 * max_c, 0.00002 * max_c, 0.000005 * max_c]
        adam_dl = min(deadline - 30, time.monotonic() + 120)
        a_pol2, sc_pol2 = _adam_logspace(a_pol, adam_dl, betas_ult, lr_init=0.003, max_iter=80000)
        sc_full = _score(a_pol2)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_pol2.copy()
            print(f"  Ultra-fine Adam: score={best_score:.10f}")
        
        # Hot spot reduction
        if time.monotonic() < deadline - 20:
            a_hs, sc_hs = _hot_spot_reduce(best_seq, min(deadline - 10, time.monotonic() + 60), n_iters=3000)
            sc_full = _score(a_hs)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_hs.copy()
                print(f"  Hot spot reduce: score={best_score:.10f}")
        
        # Coordinate descent
        if time.monotonic() < deadline - 10:
            a_cd, sc_cd = _coord_descent(best_seq, min(deadline - 5, time.monotonic() + 30), n_steps=3000)
            sc_full = _score(a_cd)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_cd.copy()
                print(f"  Coord descent: score={best_score:.10f}")
    
    # Final validation and return
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL verified score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
