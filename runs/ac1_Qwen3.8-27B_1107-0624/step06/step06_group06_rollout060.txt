            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _make_geometric(n, r):
    a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


def _make_cosine(n, freq, amp, phase=0.0):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_parabolic(n, amp):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * (t - 0.5) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_powerlaw(n, alpha):
    a = (np.arange(n) + 1.0) ** (-alpha)
    return a / a.sum()


def _make_two_cosine(n, f1, f2, amp1, amp2, phase=0.0):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + amp1 * np.cos(2.0 * np.pi * f1 * t / n + phase) + amp2 * np.cos(2.0 * np.pi * f2 * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _resample(a, n_new):
    """Resample sequence a to length n_new using linear interpolation."""
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def _shrinkage_postprocess(a, deadline):
    """Try zeroing small elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    sorted_idx = np.argsort(a)

    for frac in [0.01, 0.02, 0.05, 0.08, 0.12, 0.15, 0.2, 0.3]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = _score(a_try)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_try.copy()

    return best_a, best_sc


def _perturb_and_optimize(a, rng, deadline, method='log'):
    """Perturb and re-optimize."""
    n = len(a)
    sigma = rng.uniform(0.01, 0.08)
    z = np.log(np.maximum(a, 1e-30)) + sigma * rng.standard_normal(n)
    z = z - z.mean()
    a_pert = _softmax(z)

    if method == 'log':
        return _run_logspace_active(a_pert, deadline, max_iter=2000)
    else:
        return _run_direct_active(a_pert, deadline, max_iter=2000)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy active set optimizer.
    Phase 1: Diverse initializations
    Phase 2: Active set optimization (log-space + direct-space)
    Phase 3: L-BFGS smoothing refinement
    Phase 4: Post-processing (shrinkage, perturbations, multi-n)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "uniform_init")

    # --- Phase 1: Generate diverse candidates ---
    print(f"\n[Phase 1] Generating diverse candidates (t={time.monotonic() - t_start:.1f}s)")
    n_opt = n_parent if n_parent >= 2048 else 4096

    candidates = []
    labels = []

    # Parent-based
    candidates.append(parent.copy())
    labels.append("parent")

    # Geometric (various decay rates)
    for r in [0.999, 0.998, 0.997, 0.995, 0.99]:
        candidates.append(_make_geometric(n_opt, r))
        labels.append(f"geom_r{r}")
        candidates.append(_make_geometric(n_opt, 1.0 / r))
        labels.append(f"geom_inv_r{r}")

    # Cosine-modulated
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.1, 0.2, 0.3, 0.5]:
            candidates.append(_make_cosine(n_opt, freq, amp))
            labels.append(f"cos_f{freq}_a{amp}")

    # Two-cosine
    candidates.append(_make_two_cosine(n_opt, 1, 3, 0.2, 0.1))
    labels.append("cos2_1_3")
    candidates.append(_make_two_cosine(n_opt, 2, 5, 0.15, 0.15))
    labels.append("cos2_2_5")

    # Parabolic
    for amp in [0.3, 0.5, 0.8, 1.0]:
        candidates.append(_make_parabolic(n_opt, amp))
        labels.append(f"parab_a{amp}")

    # Power law
    for alpha in [0.05, 0.1, 0.2, 0.3]:
        candidates.append(_make_powerlaw(n_opt, alpha))
        labels.append(f"powlaw_a{alpha}")

    # Random perturbations of parent
    for _ in range(3):
        z = np.log(np.maximum(parent, 1e-30)) + 0.05 * rng.standard_normal(n_parent)
        z = z - z.mean()
        candidates.append(_softmax(z))
        labels.append("rand_perturb")

    # Evaluate all candidates
    scored_candidates = []
    for i, (cand, lab) in enumerate(zip(candidates, labels)):
        if time.monotonic() > deadline - 600:
            break
        sc = _score(cand)
        scored_candidates.append((sc, cand, lab))

    # Sort by score and keep top candidates
    scored_candidates.sort(key=lambda x: x[0])
    top_candidates = scored_candidates[:min(12, len(scored_candidates))]

    print(f"  Generated {len(candidates)} candidates, top scores: "
          f"{[f'{s:.6f}({l})' for s, _, l in top_candidates[:5]]}")

    # --- Phase 2: Active set optimization on top candidates ---
    print(f"\n[Phase 2] Active set optimization (t={time.monotonic() - t_start:.1f}s)")
    # Allocate time: give more time to better candidates
    n_top = len(top_candidates)
    time_per_candidate = (deadline - t_start - 600) / max(1, n_top)

    for rank, (sc_init, cand, lab) in enumerate(top_candidates):
        if time.monotonic() > deadline - 300:
            break
        alloc = min(time_per_candidate * (n_top - rank) * 0.3, deadline - t_start - 200)
        cand_deadline = t_start + alloc

        # Log-space active set
        a_opt, sc_opt = _run_logspace_active(cand, cand_deadline, max_iter=6000)
        update_best(a_opt, sc_opt, f"log_{lab}")

        # Direct-space active set from best so far
        if time.monotonic() < deadline - 200:
            start_a = a_opt if sc_opt < best_score else best_seq
            direct_deadline = min(cand_deadline, deadline - 150)
            a_opt2, sc_opt2 = _run_direct_active(start_a, direct_deadline, max_iter=4000)
            update_best(a_opt2, sc_opt2, f"direct_{lab}")

    # --- Phase 3: L-BFGS refinement on the best ---
    print(f"\n[Phase 3] L-BFGS refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        lbfgs_deadline = min(deadline - 60, time.monotonic() + 180)
        a_lbfgs, sc_lbfgs = _run_lbfgs_smoothed(best_seq, lbfgs_deadline, min_time_remaining=30)
        update_best(a_lbfgs, sc_lbfgs, "lbfgs_refine")

        # Follow up with active set on the L-BFGS result
        if sc_lbfgs < best_score and time.monotonic() < deadline - 60:
            a_act, sc_act = _run_logspace_active(a_lbfgs, deadline - 30, max_iter=3000)
            update_best(a_act, sc_act, "log_post_lbfgs")

    # --- Phase 4: Post-processing ---
    print(f"\n[Phase 4] Post-processing (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        # Shrinkage
        pp_deadline = min(deadline - 10, time.monotonic() + 120)
        a_shr, sc_shr = _shrinkage_postprocess(best_seq, pp_deadline)
        update_best(a_shr, sc_shr, "shrinkage")

        # Re-optimize after shrinkage
        if sc_shr < best_score + 0.001 and time.monotonic() < deadline - 20:
            a_shr_opt, sc_shr_opt = _run_logspace_active(a_shr, deadline - 5, max_iter=2000)
            update_best(a_shr_opt, sc_shr_opt, "shrinkage_reopt")

        # Perturbation restarts
        if time.monotonic() < deadline - 20:
            for i in range(3):
                if time.monotonic() > deadline - 5:
                    break
                method = 'log' if i % 2 == 0 else 'direct'
                a_pert, sc_pert = _perturb_and_optimize(best_seq, rng, deadline - 3, method)
                update_best(a_pert, sc_pert, f"perturb_{i}")

    # --- Phase 5: Multi-n exploration ---
    print(f"\n[Phase 5] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n_trials = [2048, 3072, 4096, 5120, 6144]
        n_trials = [n for n in n_trials if n != len(best_seq)]
        for n_try in n_trials[:3]:
            if time.monotonic() > deadline - 10:
                break
            a_resampled = _resample(best_seq, n_try)
            sc_res = _score(a_resampled)
            # Quick optimization
            a_opt_n, sc_opt_n = _run_logspace_active(a_resampled, deadline - 5, max_iter=1500)
            update_best(a_opt_n, sc_opt_n, f"n_{n_try}")

    # --- Phase 6: Final fine polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very fine active set with tiny tolerance
        a_fine, sc_fine = _run_logspace_active(best_seq, deadline - 1, max_iter=2000)
        update_best(a_fine, sc_fine, "final_log")

        a_fine2, sc_fine2 = _run_direct_active(best_seq, deadline - 1, max_iter=1000)
        update_best(a_fine2, sc_fine2, "final_direct")

    # --- Final output ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
