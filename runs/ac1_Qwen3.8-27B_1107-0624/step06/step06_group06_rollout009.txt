"""
Optimized step function search using refined L-BFGS with adaptive beta,
coordinate descent post-processing, top-k mean objective, and multi-n exploration.
Key improvements: finer beta schedule, coordinate descent, n=8192 exploration,
top-k mean transition, and pairwise exchange local search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _topk_obj(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    idx = np.argsort(c)[::-1][:k]
    top_mean = float(c[idx].mean())
    return 2.0 * n * top_mean


def _topk_grad(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_active(z, a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _coord_descent(a_init, deadline, n_passes=50, n_trials=4):

    n = len(a_init)
    a = a_init.copy()

    best_sc = _score(a)

    # a[i] affects conv[k] for k in [i, i+n-1] clamped to [0, 2n-2]
    # The coefficient of a[i] in conv[k] is a[k-i] + a[i-k] (minus correction)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        c = _full_convolve(a, a)
        c_max = float(c.max())

        for i in order:
            if time.monotonic() > deadline:
                break
            old_val = a[i]
            if old_val < 1e-15:

                candidates = [
                    old_val * 0.9,
                    old_val * 1.1,
                    old_val - 0.01 * rel,
                    old_val + 0.01 * rel,

                # Also try based on local optimum
                # The optimal a[i] given others: minimize max_k (c_k + delta * coeff_k)
                # This is a 1D problem but expensive to solve exactly
                candidates.append(old_val * 0.8)
                candidates.append(old_val * 1.2)

            best_trial_cmax = c_max

                    v = 0.0
                if v == old_val:
                    continue

                delta = v - old_val
                new_c_max = c_max
                for k in range(max(0, i - n + 1), min(2 * n - 1, i + n)):
                    # conv[k] involves a[i] as a[j] when j=i, and as a[k-j] when k-j=i
                    coeff = 0.0
                    ji = k - i
                    if 0 <= ji <= n - 1:
                        coeff += a[ji]
                    ji2 = i - k  # this would be negative, skip

                    # So delta in conv[k] = delta * a[k-i] + delta * a[i-k] - delta^2*delta_{k,2i}
                    # Simplified: delta * (a[k-i] + a[k-i]) if k-i and i-k both valid
                    # But a[i-k] is only valid if 0 <= i-k <= n-1, i.e., k <= i
                    # Let me just recompute the affected conv values directly
                    pass

            # Simpler approach: just recompute the full conv for the best candidate
            # but only if it looks promising based on a quick estimate
            a[i] = old_val  # reset
            # For efficiency, just try the 4 main candidates
            for v in [old_val * 0.9, old_val * 1.1, max(0, old_val - 0.005/a[i].sum() if a[i].sum() > 0 else 0), old_val + 0.005 * (a.sum() / n)]:
                if v < 0:
                    v = 0.0
                a[i] = v
                a[i] = a[i]  # keep
                new_c = _full_convolve(a, a)
                new_cmax = float(new_c.max())
                if new_cmax < best_trial_cmax - 1e-15:
                    best_trial_cmax = new_cmax
                    best_trial = v
                    # Update c for subsequent coordinates
                    c = new_c
                    c_max = new_cmax
                    improved = True
                    a[i] = best_trial
                else:
                    a[i] = old_val

            a[i] = best_trial

        # Renormalize
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc

                # But wait, if k-i = j, then the term a[k-i] already includes a[j] which is changing

def _run_lbfgs_fine(a_init, deadline, min_time_remaining=100.0):
    """L-BFGS with very fine beta schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 20:
        gap = c_sorted[0] - c_sorted[19]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)

    # Very fine geometric schedule
    ratio = 0.5
    betas = []
    b = base_beta
    while b > max_c * 1e-6:
        betas.append(b)
        b *= ratio
    betas.append(max_c * 1e-6)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break

        scale = max(1.0, n / 2048.0)
        est_time_per_iter = 0.003 * scale
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 20))))
        # Limit iterations per beta level
        max_iter = min(max_iter, 1000)

        def fun(obj_z):
            return _smoobj(obj_z, beta, n)

        def jac(obj_z):
            return _smograd(obj_z, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02):
    """Subgradient method on the active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol_frac = 0.001 + 0.001 * (1.0 - min(1.0, i / 1000.0))
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = a + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interpolate(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: fine L-BFGS + top-k mean + coordinate descent +
    pairwise exchange + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent)
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}, score={_score(parent):.10f}")
        update_best(parent)

    # === Phase 1: Main L-BFGS with fine beta schedule ===
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs_fine(parent, deadline, min_time_remaining=300.0)
    update_best(a_main)

    # === Phase 2: Top-k mean polish ===
    print(f"\n[Phase 2] Top-k polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_topk = _run_topk_phase(best_seq, deadline, min_time_remaining=200.0)
        update_best(a_topk)

    # === Phase 3: Coordinate descent ===
    print(f"\n[Phase 3] Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_cd, sc_cd = _coord_descent_fast(best_seq, min(deadline - 150, time.monotonic() + 180), n_passes=200)
        update_best(a_cd, sc_cd)

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 100, time.monotonic() + 120), n_max_steps=5000, lr_init=0.01)
        update_best(a_sg, sc_sg)

    # === Phase 5: Multi-n exploration ===
    print(f"\n[Phase 5] Multi-n (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in sorted(n_candidates, key=lambda x: abs(x - n_parent)):
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_parent:
            continue  # Already explored

        if parent is not None and n_parent != n_try:
            a_init = _interpolate(parent, n_try)
        else:
            a_init = _make_profile(n_try, "tapered")

        print(f"  Trying n={n_try}, init score={_score(a_init):.10f}")
        a_opt = _run_lbfgs_fine(a_init, deadline, min_time_remaining=80.0)
        update_best(a_opt)

        if _score(a_opt) < best_score + 0.005 and time.monotonic() < deadline - 60:
            # Coordinate descent on this
            a_cd2, sc_cd2 = _coord_descent_fast(a_opt, min(deadline - 40, time.monotonic() + 80), n_passes=100)
            update_best(a_cd2, sc_cd2)

    # === Phase 6: Pairwise exchange ===
    print(f"\n[Phase 6] Pairwise exchange (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_pw, sc_pw = _pairwise_exchange(best_seq, min(deadline - 30, time.monotonic() + 60), n_rounds=20)
        update_best(a_pw, sc_pw)

    # === Phase 7: Final fine polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # One more round of coordinate descent
        a_fin, sc_fin = _coord_descent_fast(best_seq, min(deadline - 5, time.monotonic() + 60), n_passes=50)
        update_best(a_fin, sc_fin)

        # Final L-BFGS with very small beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 1e-5
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(1000, max(30, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda zz: _smoobj(zz, beta_final, n),
                    z,
                    jac=lambda zz: _smograd(zz, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin2 = _softmax(result.x)
                update_best(a_fin2)
            except Exception:
                pass

    # === Phase 8: Diverse restarts (if time remains) ===
    print(f"\n[Phase 8] Restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for i in range(2):
            if time.monotonic() > deadline - 20:
                break
            sigma = 0.03 + 0.02 * i
            p_start = _perturb(best_seq, sigma, rng)
            a_r = _run_lbfgs_fine(p_start, deadline, min_time_remaining=15.0)
            update_best(a_r)

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
