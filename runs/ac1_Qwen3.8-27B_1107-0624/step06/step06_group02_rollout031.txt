"""
Sbrana-inspired frequency domain optimizer combined with gradient refinement
for minimizing the peak of the autocorrelation of non-negative step functions.
"""
import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max_conv"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}"""
    n = len(v)
    if n == 0:
        return v
    # Scale to work with target_sum
    v_scaled = v / target_sum
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v_scaled - theta, 0.0) * target_sum

def sbrana_optimize(a_init, deadline, max_iter=100000):
    """
    Sbrana-style frequency domain optimization.
    Minimizes max of autocorrelation for non-negative a with sum=1.
    """
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Precompute
    k_star = np.arange(N)  # All lags
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute FFT of a (zero-padded to N)
        a_pad = np.zeros(N)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        P = (np.abs(A) ** 2)
        
        # Compute autocorrelation in frequency domain
        c = np.fft.ifft(P).real
        c = c[:2*n-1]  # Only the linear part
        
        c_max = float(c.max())
        k_peak = int(np.argmax(c))
        
        # Compute score
        sc = 2.0 * n * c_max
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Sbrana step: reduce frequencies that contribute to the peak
        # The contribution of frequency f to c[k_peak] is:
        # (1/N) * P[f] * cos(2*pi*f*k_peak/N) (real part)
        # We want to reduce frequencies where this contribution is large and positive
        
        # Simple approach: if P[f] > threshold, reduce it
        # Better: use a multiplicative rule
        # P_new[f] = P[f] * exp(-eta * contribution[f])
        
        # Contribution to the peak
        phase = 2.0 * np.pi * np.arange(N) * k_peak / N
        contribution = (P / N) * np.cos(phase)
        
        # Adaptive step size
        eta = 0.1 * N / (P.max() + 1e-30)
        
        # Only reduce, don't increase
        P_new = P * np.exp(-np.maximum(eta * contribution, 0))
        
        # Preserve total power (related to sum(a)^2)
        # Actually, sum(a)^2 = c[0] = (1/N) * sum(P[f]) ... no
        # c[0] = sum_j a[j]^2 (for the linear case)
        # In circular: c_circ[0] = sum_j a_padded[j]^2 = sum_j a[j]^2
        # (1/N) * sum_f |A(f)|^2 = (1/N) * sum_f P[f] = sum_j |a_padded[j]|^2 = sum_j a[j]^2
        # So sum(P) / N = sum(a[j]^2), not (sum a)^2.
        # (sum a)^2 = c_circ[n-1] (for the circular case with the right indexing)
        # Actually, (sum a)^2 = (sum_j a[j])^2 = |A(0)|^2 / N = P[0] / N
        # So P[0] = N * (sum a)^2. Since we normalize sum(a)=1, P[0] should be N.
        
        # We don't strictly need to preserve total power since we renormalize a afterwards.
        
        # Recover A from P_new
        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a = a_new / s

"""
Multi-strategy optimizer combining Sbrana frequency-domain iterations,
p-norm smoothed gradient descent with Nesterov acceleration, adaptive
step sizes, and multi-scale exploration (n=4096, n=8192) to minimize
the peak autocorrelation ratio of non-negative step functions.

import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    v_s = v / target_sum
    u = np.sort(v_s)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v_s - theta, 0.0) * target_sum


def _xcorr_padded(a, cp):
    """Compute R[i] = sum_j a[j]*cp[i+j] for i=0..n-1 using FFT."""
    n = len(a)
    b = a[::-1]
    N = _next_pow2(2 * n - 1)
    fb = np.fft.rfft(b, N)
    fcp = np.fft.rfft(cp, N)
    conv = np.fft.irfft(fb * fcp, N)
    return conv[n - 1:2 * n - 1]


def _pnorm_grad(a, p):
    """Gradient of (sum(c^p))^(1/p) where c = convolve(a,a), assuming sum(a)=1."""
    n = len(a)
    c = _convolve(a, a)
    c_pos = np.maximum(c, 1e-300)
    cp = c_pos ** (p - 1)
    pnorm_val = float(np.sum(c_pos ** p)) ** (1.0 / p)
    if pnorm_val < 1e-300:
        return np.zeros(n), 0.0
    R = _xcorr_padded(a, cp)
    grad = pnorm_val ** (1.0 - p) * 2.0 * R * 2.0 * n
    return grad, pnorm_val


def _sbrana_step(a, N, strength=0.5):
    """One Sbrana-style frequency domain iteration."""
    n = len(a)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    A = np.fft.fft(a_pad)
    P = (np.abs(A) ** 2)
    c = np.fft.ifft(P).real
    c_lin = c[:2 * n - 1]
    k_peak = int(np.argmax(c_lin))
    c_max = c_lin[k_peak]
    if c_max < 1e-30:
        return a
    phase = 2.0 * np.pi * np.arange(N) * k_peak / N
    contribution = (P / N) * np.cos(phase)
    P_max = P.max()
    if P_max < 1e-30:
        return a
    eta = strength
    reduction = np.exp(-eta * np.maximum(contribution, 0) / (P_max / N + 1e-30))
    P_new = P * reduction
    P_new = np.maximum(P_new, 0)
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.fft.ifft(A_new).real
    a_new = a_new[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a
    return a_new / s


def _fista_pnorm(a_init, p_list, deadline, max_iter_per_phase=50000):
    """FISTA-like accelerated p-norm gradient descent."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for p in p_list:
        t_start_phase = time.monotonic()
        a_phase = a_curr.copy()
        a_prev_p = a_curr.copy()
        t_p = 1.0

        # Estimate Lipschitz constant
        grad0, _ = _pnorm_grad(a_phase, p)
        L = np.linalg.norm(grad0) + 1.0
        step = 1.0 / L

        for it in range(max_iter_per_phase):
            if time.monotonic() > deadline:
                break
            if time.monotonic() - t_start_phase > 120:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_phase + ((t_val - 1.0) / t_next) * (a_phase - a_prev_p)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            grad, pnorm_val = _pnorm_grad(y, p)
            if not np.all(np.isfinite(grad)):
                break

            # Backtracking line search (simplified)
            alpha = step
            for _bt in range(10):
                a_cand = _proj_simplex(y - alpha * grad)
                sc_cand = _score_norm(a_cand)
                if sc_cand < _score_norm(y) * 1.001 + 1e-15:
                    break
                alpha *= 0.5

            a_next = _proj_simplex(y - alpha * grad)
            a_prev_p = a_phase
            a_phase = a_next
            t_val = t_next

            if it % 200 == 0:
                sc = _score_norm(a_phase)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_phase.copy()

        a_curr = a_phase
        a_prev = a_curr.copy()
        sc = _score_norm(a_curr)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def _sbrana_optimize(a_init, deadline, max_iter=200000):
    """Sbrana-style frequency domain optimization."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        strength = 0.3 + 0.3 * (it / max(1, max_iter / 4))
        a_new = _sbrana_step(a, N, strength)
        if np.all(a_new == a):
            break
        a = a_new
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _adam_pnorm(a_init, deadline, p_schedule, lr_init=0.01, max_iter=300000):
    """Adam optimizer on p-norm smoothed objective in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        it_phase = 0
        lr = lr_init
        while it_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            a = np.exp(z - np.logaddexp.reduce(z))
            a = a / a.sum()

            grad, _ = _pnorm_grad(a, p)
            if not np.all(np.isfinite(grad)):
                break

            # Project gradient onto simplex tangent
            grad_mean = grad.mean()
            grad_t = grad - grad_mean

            m_buf = b1 * m_buf + (1.0 - b1) * grad_t
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_t * grad_t)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = np.exp(z - np.logaddexp.reduce(z))
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_phase % 5000 == 0:
                lr = max(0.0003, lr * 0.92)

    a_final = np.exp(z - np.logaddexp.reduce(z))
    a_final = a_final / a_final.sum()
    sc = _score_norm(a_final)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    """Coordinate descent with local search."""
    n = len(a_init)

    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.95, 0.98, 1.02, 1.05]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - np.max(z))
    return e / e.sum()

                            p_start[b1_i*block:(b1_i+1)*block] = p_start[b2_i*block:(b2_i+1)*block]
                            p_start[b2_i*block:(b2_i+1)*block] = tmp
                    p_start = p_start / p_start.sum()
                else:
                    # Raised cosine window blend
                    x = np.arange(len(best_seq))
                    raised = np.sin(np.pi * (x + 0.5) / len(best_seq)) ** 2
                    blend = 0.7 * best_seq + 0.3 * raised / raised.sum()
                    p_start = blend / blend.sum()

                # Quick optimization: Sbrana + FISTA
                c_s = _convolve(p_start, p_start)
                max_cs = float(c_s.max())

                if time.monotonic() < phase5_end - 15:
                    t_sub = time.monotonic() + max(5.0, per_restart * 0.4)
                    p_sbr, _ = _sbrana_optimize(p_start, t_sub, max_iter=20000)

                    t_sub2 = time.monotonic() + max(5.0, per_restart * 0.5)
                    p_opt, _ = _fista_pnorm(p_sbr, [32, 64, 128], t_sub2, max_iter_per_phase=10000)

                    sc_r = _score(p_opt)
                    if sc_r < best_score:
                        best_score = sc_r
                        best_seq = p_opt.copy()
                        print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final coordinate descent polish ===
    print("Phase 6: Final coordinate descent polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 10, time.monotonic() + 100)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=200)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
