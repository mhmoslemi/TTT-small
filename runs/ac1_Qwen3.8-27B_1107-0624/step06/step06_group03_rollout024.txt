Optimized step-function search using efficient coordinate descent with explicit
convolution tracking, gradient-based polishing with top-K smoothed max, and
multi-length exploration. Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

import numpy as np

def _next_pow2(n):
    k = 1

def _full_convolve(a, b):

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * float(c.max()) / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)

        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

            # Remove a_i contribution: 2*a_i*a[j] for j != i, a_i^2 for j == i
            base -= 2.0 * a_i * a[j_idx]
            # For j == i: we subtracted 2*a_i*a[i] but should only subtract a_i^2
            mask_i = (j_idx == i)
            if np.any(mask_i):
                base[mask_i] += a_i * a[i]  # add back a_i^2 (we subtracted 2*a_i^2, need -a_i^2)

            slope = 2.0 * a[j_idx]
            slope[mask_i] = 0.0  # quadratic term handled separately

            # Unaffected max
            if k_lo > 0:
                unaffected_left = c[:k_lo].max() if k_lo > 0 else 0.0
            else:
                unaffected_left = 0.0
            if k_hi < 2 * n - 2:
                unaffected_right = c[k_hi + 1:].max()
            else:
                unaffected_right = 0.0
            max_unaffected = max(unaffected_left, unaffected_right, 0.0)

            # Generate candidate t values
            candidates = [0.0, a_i, a_i * 0.5, a_i * 1.5]

            # Top-M affected positions by base + a_i*slope (current c values)
            current_affected = base + a_i * slope
            if len(k_idx) > top_m:
                top_positions = np.argpartition(-current_affected, top_m)[:top_m]
            else:
                top_positions = np.arange(len(k_idx))

            for pos in top_positions:
                s_val = slope[pos]
                b_val = base[pos]
                if s_val > 1e-15:
                    t_k = S0 - 2.0 * b_val / s_val
                    if t_k >= 0 and t_k < 100.0:
                        candidates.append(t_k)
                        candidates.append(max(0, t_k * 0.8))
                        candidates.append(t_k * 1.2)

            # Quadratic term at k = 2i
            if mask_i.any():
                c_2i = c[2 * i]
                if S0 > 1e-15:
                    t_quad = (c_2i - a_i * a_i) / S0
                    if t_quad >= 0 and t_quad < 100.0:
                        candidates.append(t_quad)

            # Deduplicate
            candidates = list(set([max(0.0, min(50.0, float(tc))) for tc in candidates]))

            # Evaluate score at each candidate
            best_t = a_i
            best_f = np.inf
            for tc in candidates:
                # Compute new max convolution value
                # New affected values: base + tc * slope (linear part)
                new_affected = base + tc * slope
                # Handle quadratic at k=2i
                if mask_i.any():
                    new_affected[mask_i] = base[mask_i] + tc * tc

                new_max = max(float(new_affected.max()), max_unaffected)
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * new_max / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i and best_f < best_score * 1.001:
                # Update a and c
                delta = best_t - a_i
                a[i] = best_t
                # Update convolution
                c[k_idx] = base + best_t * slope
                if mask_i.any():
                    c[k_idx[mask_i]] = base[mask_i] + best_t * best_t
                S = S0 + best_t
                best_score = min(best_score, best_f)
                improved += 1

        if improved == 0:
            break

    final_score = _score_from_conv(a, c)
    if final_score < best_score:
        best_score = final_score
    return a, c, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad = 2.0 * n * a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        sc_parent = _score(parent)
        print(f"No parent. Uniform init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # === PHASE 1: Coordinate Descent (primary, ~65% of time) ===
    phase1_budget = (deadline - time.monotonic()) * 0.65
    phase1_end = time.monotonic() + phase1_budget
    print(f"\n=== Phase 1: Coordinate Descent (budget: {phase1_budget:.1f}s) ===")

    n = len(parent)
    a = parent.copy()
    a = a / a.sum()
    c = _full_convolve(a, a)
    print(f"  Initial: n={n}, score={_score_from_conv(a, c):.10f}")

    # Run coordinate descent with progressively smaller top_m
    top_m_schedule = [60, 40, 30, 20, 15, 10]
    pass_counts = [50, 50, 80, 80, 100, 150]

    total_passes_done = 0
    for mi, (tm, pc) in enumerate(zip(top_m_schedule, pass_counts)):
        passes_this_round = min(pc, max(1, int((phase1_end - time.monotonic()) / 0.3)))
        if passes_this_round < 1:
            break
        a, c, sc = _coord_descent_pass(a, c, rng, top_m=tm, max_passes=passes_this_round,
                                        deadline=phase1_end)
        total_passes_done += passes_this_round
        score_norm = _score_from_conv(a, c)
        print(f"  Round {mi} (top_m={tm}, {passes_this_round} passes): score={score_norm:.10f} (t={time.monotonic()-t_start:.1f}s)")
        if score_norm < best_score:
            best_score = score_norm
            best_seq = a.copy()

    a = a / a.sum()
    c = _full_convolve(a, a)  # refresh for safety
    print(f"  Phase 1 done: score={_score_from_conv(a, c):.10f}, total passes={total_passes_done}")

    # === PHASE 2: Gradient polish with Adam in log-space (~20% of time) ===
    phase2_budget = min((deadline - time.monotonic()) * 0.20, 250.0)
    phase2_end = time.monotonic() + phase2_budget
    print(f"\n=== Phase 2: Adam log-space polish (budget: {phase2_budget:.1f}s) ===")

    if phase2_budget > 10:
        a_work = a.copy()
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas = []
        beta = 0.005 * max_cw
        while beta > 0.00005 * max_cw:
            betas.append(beta)
            beta *= 0.5
        if len(betas) > 10:
            betas = betas[:10]

        a_opt, sc_opt = _adam_logspace(a_work, phase2_end, betas,
                                       lr_init=0.008, top_k=80, max_iter=200000)
        sc_opt_full = _score(a_opt)
        print(f"  Adam: score={sc_opt_full:.10f} (t={time.monotonic()-t_start:.1f}s)")
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            a = a_opt.copy()
            c = _full_convolve(a, a)
            # Do a few more CD passes after Adam
            a, c, sc = _coord_descent_pass(a, c, rng, top_m=20, max_passes=30,
                                            deadline=min(deadline - 30, time.monotonic() + 60))
            sc2 = _score_from_conv(a, c)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a.copy()
                print(f"  Post-Adam CD: score={sc2:.10f}")

    # === PHASE 3: Try different lengths (~10% of time) ===
    phase3_budget = min(deadline - time.monotonic() - 30, 180.0)
    print(f"\n=== Phase 3: Multi-length exploration (budget: {phase3_budget:.1f}s) ===")

    if phase3_budget > 20:
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        for n_try in [2048, 8192]:
            if time.monotonic() > deadline - 30:
                break
            per_len_budget = min(phase3_budget / 2, 80.0)
            if per_len_budget < 15:
                break

            a_interp = _interp_profile(a_base, n_try)
            sc_interp = _score(a_interp)
            print(f"  n={n_try}: initial score={sc_interp:.10f}")
            if sc_interp < best_score:
                best_score = sc_interp
                best_seq = a_interp.copy()

            c_interp = _full_convolve(a_interp, a_interp)
            # Run coordinate descent
            a_new, c_new, sc_cd = _coord_descent_pass(a_interp, c_interp, rng,
                                                       top_m=30, max_passes=30,
                                                       deadline=time.monotonic() + per_len_budget - 10)
            sc_cd_full = _score_from_conv(a_new, c_new)
            print(f"  n={n_try}: after CD score={sc_cd_full:.10f}")
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_new.copy()

    # === PHASE 4: Final polish with very fine CD ===
    if time.monotonic() < deadline - 10 and best_seq is not None:
        print(f"\n=== Phase 4: Final CD polish ===")
        a_final = best_seq.copy()
        a_final = a_final / a_final.sum()
        c_final = _full_convolve(a_final, a_final)
        # Several passes with small top_m
        for tm in [10, 5, 3]:
            if time.monotonic() > deadline - 5:
                break
            a_final, c_final, sc = _coord_descent_pass(a_final, c_final, rng,
                                                        top_m=tm, max_passes=100,
                                                        deadline=deadline - 5)
            sc_f = _score_from_conv(a_final, c_final)
            print(f"  Final CD (top_m={tm}): score={sc_f:.10f}")
            if sc_f < best_score:
                best_score = sc_f
                best_seq = a_final.copy()

    # Final validation
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
