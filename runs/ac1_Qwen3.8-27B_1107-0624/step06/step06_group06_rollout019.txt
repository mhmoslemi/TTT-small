def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())

    s = e.sum()

def _kkt_polish(a_init, deadline, max_iter=200):

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)

        grad = -np.where(support, resid_t, 0.0)

        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()

            lr *= 0.5
            continue
        a_try = a_try / s_try

        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.82
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sin2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.exp(-8.0 * (t - 0.5) ** 2)
        a = a + 0.05
        return a / a.sum()
    else:
        return np.ones(n) / n


def _enforce_symmetry(a):
    """Make sequence symmetric: a[j] = a[n-1-j]."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS + Active-set Newton CG + KKT polish + symmetry + multi-n.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === PHASE 1: L-BFGS with improved beta schedule ===
    print(f"\n[Phase 1] L-BFGS improved (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 500.0
    a_main = _run_lbfgs_improved(parent, min(p1_deadline, deadline - 300), min_time_remaining=50.0)
    update_best(a_main, _score(a_main), "lbfgs_p1")

    # === PHASE 2: Active-set Newton with CG ===
    print(f"\n[Phase 2] Active-set Newton CG (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p2_deadline = min(t_start + 850.0, deadline - 200)
        a_newton, sc_newton = _active_newton_cg(best_seq, p2_deadline, max_iter=400)
        update_best(a_newton, sc_newton, "newton_cg")

        # Second pass with tighter tolerance
        if time.monotonic() < deadline - 150:
            a_newton2, sc_newton2 = _active_newton_cg(a_newton, min(t_start + 1000.0, deadline - 150), max_iter=200, tol_start=0.005, tol_end=0.00005)
            update_best(a_newton2, sc_newton2, "newton_cg2")

    # === PHASE 3: KKT Polishing ===
    print(f"\n[Phase 3] KKT Polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_kkt, sc_kkt = _kkt_polish(best_seq, min(t_start + 1100.0, deadline - 100), max_iter=300)
        update_best(a_kkt, sc_kkt, "kkt")

    # === PHASE 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(t_start + 1250.0, deadline - 80), n_max_steps=4000, lr_init=0.015)
        update_best(a_sg, sc_sg, "subgrad")

    # === PHASE 5: Symmetry exploitation ===
    print(f"\n[Phase 5] Symmetry (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_sym = _enforce_symmetry(best_seq)
        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            update_best(a_sym, sc_sym, "symmetric")
            # Re-optimize from symmetric point
            a_sym_opt = _run_lbfgs_improved(a_sym, min(t_start + 1400.0, deadline - 40), min_time_remaining=20.0)
            update_best(a_sym_opt, _score(a_sym_opt), "sym_lbfgs")
            a_sym_newton, sc_sym_newton = _active_newton_cg(a_sym_opt, min(t_start + 1500.0, deadline - 30), max_iter=100)
            update_best(a_sym_newton, sc_sym_newton, "sym_newton")
        else:
            print(f"  Symmetry worsened score: {sc_sym:.10f} vs {best_score:.10f}")

    # === PHASE 6: Second L-BFGS pass on best ===
    print(f"\n[Phase 6] Final L-BFGS refinement (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_fin = _run_lbfgs_improved(best_seq, deadline - 5, min_time_remaining=5.0)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

    # === PHASE 7: Micro-perturbation search ===
    print(f"\n[Phase 7] Micro-perturbations (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        for pert_sigma in [0.01, 0.005, 0.002]:
            if time.monotonic() > deadline - 8:
                break
            a_pert = _perturb(best_seq, pert_sigma, rng)
            # Quick Newton step
            a_pert2, sc_pert2 = _active_newton_cg(a_pert, min(deadline - 5, time.monotonic() + 20), max_iter=50)
            update_best(a_pert2, sc_pert2, f"pert_{pert_sigma}")

    # === PHASE 8: Try slightly different n ===
    print(f"\n[Phase 8] Alternative n values (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        n_cur = len(best_seq) if best_seq is not None else 4096
        n_alts = [n_cur, n_cur + 128, n_cur - 128]
        n_alts = [n for n in n_alts if 512 <= n <= 8192]
        for n_try in n_alts:
            if time.monotonic() > deadline - 15:
                break
            if best_seq is not None and n_try != len(best_seq):
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_try)
                a_init = np.interp(x_new, x_old, best_seq)
                a_init = np.maximum(a_init, 0)
                s_init = a_init.sum()
                if s_init > 1e-10:
                    a_init = a_init / s_init
            else:
                a_init = best_seq.copy() if best_seq is not None else _make_profile(n_try, "uniform")

            a_opt, sc_opt = _active_newton_cg(a_init, min(deadline - 10, time.monotonic() + 25), max_iter=80)
            update_best(a_opt, sc_opt, f"n_{n_try}")

    # === Final ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
