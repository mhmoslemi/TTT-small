"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive FISTA with backtracking, spectral initialization, 
multi-scale optimization, and peak-shaving polish.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm(a):
    """Assume sum(a) = 1, return 2n*max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad(a, c, w):
    """Gradient of smooth objective w.r.t. a, where c = a*a, w = softmax weights on c"""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_backtrack(a_init, beta, deadline, max_iter, use_backtrack=True):
    """FISTA with optional backtracking line search"""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    
    # Initial Lipschitz estimate
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        grad = _compute_grad(y, c_y, w)
        
        if use_backtrack:
            # Backtracking line search
            a_next = _proj_simplex(y - step * grad)
            # Simplified: just use fixed step (backtracking on the non-convex problem is tricky)
        else:
            a_next = _proj_simplex(y - step * grad)
        
        if it % 50 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, lr_decay_every=10000, lr_decay_factor=0.97, grad_clip=1.0):

    a_norm = a_init.copy()

    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n
    
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()
    
    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w
            
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            
            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)
            
            if no_improve_count > 100 and t > 500:
                current_lr = max(0.0003, current_lr * 0.85)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

def _spectral_flatten(a_init, n_iter=500, deadline=None):
    """Flatten the power spectrum to reduce autocorrelation peaks"""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        # Gentle flattening: interpolate between current magnitude and uniform
        alpha = 0.5 + 0.3 * (it / n_iter)  # Increase flattening over time
        avg_mag = mag.mean()
        new_mag = (1 - alpha) * mag + alpha * avg_mag
        phase = A / (mag + 1e-30)
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_a, best_score

def _peak_shaving(a_init, deadline, n_steps=5000):
    """Directly target the top peaks using subgradient estimation"""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Use top-K peaks with softmax weighting
        top_k = min(32, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        # Temperature for selecting among top peaks
        temp = 0.01 * (c[-1] - c_top[0] + 1e-10) + 1e-10
        w_top = np.exp((c_top - c_top[0]) / temp)
        w_top = w_top / w_top.sum()
        
        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_top
        
        grad = _compute_grad(a, c, w_vec)
        
        # Adaptive step size
        step_size = 1.0 / (n * (1.0 + 0.001 * it))
        
        a_new = _proj_simplex(a - step_size * grad)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = _proj_simplex(a - step_size * 0.5 * grad)
            if _score_norm(a_new2) < _score_norm(a):
                a = a_new2
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

from scipy.optimize import linprog

def _build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix of f"""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based iterations (state-of-the-art approach), FISTA with smooth approximation,
Adam log-space fine-tuning, spectral flattening, peak shaving, and multi-restart.
The key innovation is the reduced-constraint LP approach that directly optimizes
the non-smooth objective without beta approximation, combined with smooth methods for polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _build_reduced_constraint(f, active_indices):
    n = len(f)
    K = len(active_indices)
    A = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start > j_end:
            continue
        j_range = np.arange(j_start, j_end + 1)
        f_indices = k - j_range
        A[idx, j_range] = f[f_indices]
    return A


def _lp_iteration(f, M, active_indices, max_iter_s=5.0):
    n = len(f)
    A = _build_reduced_constraint(f, active_indices)
    c_obj = -np.ones(n)
    b = np.full(len(active_indices), M)
    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': max_iter_s, 'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def _lp_based_optimize(a_init, deadline, max_iterations=2000, top_k=50):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    current_max = float(_full_convolve(a, a).max())

    for it in range(max_iterations):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        current_max = float(c.max())

        # Select active set: top peaks
        n_active = min(top_k, len(c))
        active_indices = np.argsort(c)[::-1][:n_active]

        M = current_max * 1.0

        g = _lp_iteration(a, M, active_indices.tolist(), max_iter_s=3.0)
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break

        # Rescale g to match sum of a
        g_rescaled = g * (1.0 / g_sum)

        # Blend
        t_blend = 0.5
        a_new = (1.0 - t_blend) * a + t_blend * g_rescaled
        a_new = np.maximum(a_new, 0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score_norm(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            if it % 20 == 0:
                print(f"    LP iter {it}: score={sc_new:.8f}, max_conv={float(_full_convolve(a,a).max()):.6f}")
        else:
            # Try smaller blend
            a_new2 = (1.0 - 0.2) * a + 0.2 * g_rescaled
            a_new2 = np.maximum(a_new2, 0)
            s2 = float(a_new2.sum())
            if s2 > 0:
                a_new2 = a_new2 / s2
                sc2 = _score_norm(a_new2)
                if sc2 < _score_norm(a):
                    a = a_new2

    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta
        e = np.exp(c_shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=8000, lr_decay_factor=0.96, grad_clip=1.0):
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 0:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_w = float(e.sum())
            if s_w < 1e-300 or not np.isfinite(s_w):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_w

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 100 and t > 500:
                current_lr = max(0.0003, current_lr * 0.85)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=1000, deadline=None):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        alpha = 0.3 + 0.4 * (it / max(1, n_iter))
        avg_mag = mag.mean()
        new_mag = (1.0 - alpha) * mag + alpha * avg_mag
        phase = A / (mag + 1e-30)
        A_new = new_mag * phase
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _peak_shaving(a_init, deadline, n_steps=5000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score_norm(a)
    step_multiplier = 1.0

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        top_k = min(32, len(c))
        top_idx = np.argsort(c)[::-1][:top_k]
        c_top = c[top_idx]
        temp = max(0.005 * (c_top[0] - c_top[-1] + 1e-10), 1e-10)
        w_top = np.exp((c_top - c_top[0]) / temp)
        w_top = w_top / w_top.sum()

        w_vec = np.zeros(len(c))
        w_vec[top_idx] = w_top

        grad = _compute_grad(a, w_vec)
        step_size = 1.0 / (n * step_multiplier)

        a_new = _proj_simplex(a - step_size * grad)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            step_multiplier = max(0.5, step_multiplier * 0.98)
        else:
            step_multiplier = min(10.0, step_multiplier * 1.1)
            a_new2 = _proj_simplex(a - step_size * 0.3 * grad)
            if _score_norm(a_new2) < _score_norm(a):

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  FISTA beta={b:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 3: best_score={best_score:.10f}")
    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 4: Adam log-space fine-tuning ===
    print("\n=== Phase 4: Adam log-space ===")
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end - 30:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())

        betas_fine = []
        b = 0.001 * max_cw
        while b > 1e-9:
            betas_fine.append(b)
            b *= 0.5
        if len(betas_fine) > 12:
            betas_fine = betas_fine[:12]

        per_t = phase4_end - time.monotonic() - 10
        if per_t > 30:
            a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + per_t, betas_fine,
                                           lr_init=0.008, max_iter_total=400000,
                                           lr_decay_every=8000, lr_decay_factor=0.95)
            sc_opt_full = _score(a_opt)
            if sc_opt_full < best_score:
                best_score = sc_opt_full
                best_seq = a_opt.copy()
                print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 4: best_score={best_score:.10f}")
    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 5: Second LP pass ===
    print("\n=== Phase 5: Second LP pass ===")
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end - 30:
        a_lp2, sc_lp2 = _lp_based_optimize(a_work, deadline=phase5_end, max_iterations=2000, top_k=30)
        sc_lp2_full = _score(a_lp2)
        if sc_lp2_full < best_score:
            best_score = sc_lp2_full
            best_seq = a_lp2.copy()
            print(f"  LP pass 2: score={sc_lp2_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 6: Peak shaving ===
    print("\n=== Phase 6: Peak shaving ===")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end - 20:
        a_ps, sc_ps = _peak_shaving(a_work, deadline=phase6_end, n_steps=8000)
        sc_ps_full = _score(a_ps)
        if sc_ps_full < best_score:
            best_score = sc_ps_full
            best_seq = a_ps.copy()
            print(f"  Peak shaving: score={sc_ps_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()

    # === Phase 7: Ultra-fine Adam polish ===
    print("\n=== Phase 7: Ultra-fine polish ===")
    if time.monotonic() < deadline - 40:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = [0.00003 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 20
        if per_t > 15:
            a_pol, sc_pol = _adam_logspace(a_work, time.monotonic() + per_t, betas_ult,
                                           lr_init=0.002, max_iter_total=200000,
                                           lr_decay_every=8000, lr_decay_factor=0.93)
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final peak shaving if time permits
    if time.monotonic() < deadline - 20:
        a_work = best_seq.copy() / best_seq.sum()
        a_ps2, _ = _peak_shaving(a_work, deadline=deadline - 5, n_steps=3000)
        sc_ps2 = _score(a_ps2)
        if sc_ps2 < best_score:
            best_score = sc_ps2
            best_seq = a_ps2.copy()
            print(f"  Final peak shaving: score={sc_ps2:.10f}")

    # === Final validation ===
    print(f"\nAll phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            final = final / final.sum()
            s = final.sum()

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(512)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
