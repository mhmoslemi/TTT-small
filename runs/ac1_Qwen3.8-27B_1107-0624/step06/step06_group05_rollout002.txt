"""
Refined optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses parent sequence as starting point, with multi-strategy refinement:
- Fine-grained Adam with adaptive beta

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_and_conv(a):
    n = len(a)
    c = _full_convolve(a, a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf, c
    return float(2.0 * n * c.max() / (s * s)), c

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_weights(a, w):
    """Compute gradient of weighted sum of convolution values."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

def _softmax_grad(a, grad_a):
    """Convert gradient in a-space to gradient in z-space (log-space)."""
    correction = float(np.dot(a, grad_a))
    return a * (grad_a - correction)

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = _compute_grad_weights(a, w)
        grad_z = _softmax_grad(a, grad_a)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 20 == 0:
            a_chk = _softmax_compute(z)
            sc = _score(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = _compute_grad_weights(a, w)
        return _softmax_grad(a, grad_a)

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 20)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score(a_init)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad_weights(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 50 == 0:
            sc = _score(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _coordinate_descent(a, deadline, max_iter):
    """Coordinate descent in a-space (simplex)."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score(a)
    best_a = a.copy()
    c = _full_convolve(a, a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        # Find the position of max in convolution
        k_max = int(np.argmax(c))
        # Gradient at the max position
        # grad[i] = 2 * a[k_max - i] (with bounds)
        i_min = max(0, k_max - n + 1)
        i_max = min(n - 1, k_max)
        # Try to reduce the max by adjusting a
        # Move mass from a[i_max_pos] to a[1-i_max_pos] where this reduces the max
        # Simple approach: try all pairs and pick the best swap
        
        # Actually, let's do a simpler coordinate descent:
        # Pick the i that contributes most to c[k_max]
        contrib = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = 2.0 * a[j]
        
        # Try moving mass from high-contrib position to low-contrib position
        i_high = int(np.argmax(contrib))
        # Find a position to move mass to (one that doesn't increase other peaks)
        # For simplicity, try moving to a random low-value position
        if a[i_high] > 1e-10:
            # Try small moves
            delta = a[i_high] * 0.01
            # Find position to move to
            i_low = int(np.argmin(a))
            a_trial = a.copy()
            a_trial[i_high] -= delta
            a_trial[i_low] += delta
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = _score(a_trial)
            if sc_trial < best_score:
                a = a_trial
                best_score = sc_trial
                best_a = a.copy()
                c = c_trial
            else:
                # Try smaller move
                delta = a[i_high] * 0.001
                a_trial = a.copy()
                a_trial[i_high] -= delta
                a_trial[i_low] += delta
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                c_trial = _full_convolve(a_trial, a_trial)
                sc_trial = _score(a_trial)
                if sc_trial < best_score:
                    a = a_trial
                    best_score = sc_trial
                    best_a = a.copy()
                    c = c_trial
    
    return best_a, best_score

def _equalize_peaks(a, deadline, n_top=5, n_iter=1000):
    """Try to equalize the top-k peaks of the convolution."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score(a)
    best_a = a.copy()
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_sorted_idx = np.argsort(c)[::-1][:n_top]
        c_max = c[c_sorted_idx[0]]
        c_min_top = c[c_sorted_idx[-1]]
        
        if c_max - c_min_top < 1e-10:
            break
        
        # Compute gradient to equalize: weighted by 1/(c_k - mean) for top peaks
        mean_top = np.mean(c[c_sorted_idx])
        weights = np.zeros_like(c)
        for k in c_sorted_idx:
            weights[k] = 1.0 / (max(c[k] - mean_top, 1e-10))
        weights = weights / weights.sum()
        
        # Gradient step
        grad = _compute_grad_weights(a, weights)
        # Simple projected gradient step
        lr = 0.01
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = a
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.5
    
    return best_a, best_score

def _try_length_variation(a_parent, n_target, deadline):
    """Try to create a good sequence of length n_target by interpolating/restricting parent."""
    if n_target == len(a_parent):
        return a_parent, _score(a_parent)
    
    if n_target > len(a_parent):
        # Upsample
        old_n = len(a_parent)
        indices_old = np.linspace(0, old_n - 1, old_n)
        indices_new = np.linspace(0, old_n - 1, n_target)
        a_new = np.interp(indices_new, indices_old, a_parent)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
    else:
        # Downsample
        old_n = len(a_parent)
        indices_old = np.linspace(0, old_n - 1, n_target)
        a_new = np.interp(np.linspace(0, old_n - 1, old_n), np.linspace(0, old_n - 1, old_n), a_parent)
        # Actually, just take every other element or use interpolation
        step = old_n / n_target
        indices = np.arange(n_target) * step
        indices = np.clip(indices, 0, old_n - 1)
        a_new = np.interp(indices, np.arange(old_n), a_parent)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
    
    sc = _score(a_new)
    return a_new, sc

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)
