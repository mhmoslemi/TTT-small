def _next_pow2(n):
    k = 1

def _fft_convolve(a, b):

    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _interp_to_length(a, n_new):
    n0 = len(a)

    x0 = np.linspace(0, 1, n0)

    # Build LP indices
    rows, cols = _build_lp_indices(n)
    print(f"LP indices built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: LP refinement (0 - 200s) ===
    phase1_end = min(deadline, t_start + 200.0)
    t_lp = time.monotonic()
    a, sc = _lp_refine(a, rows, cols, time_per_lp=8, max_iters=80)
    full_sc = _score(a)
    print(f"Phase 1 LP: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_a = a.copy()

    # === Phase 2: L-BFGS smoothed (200s - 600s) ===
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        z_init = np.log(np.maximum(best_a, 1e-15))
        z_init = z_init - z_init.mean()
        a_lbfgs, sc_lbfgs = _lbfgs_z_space(z_init, phase2_end - 3,
                                            T_start=0.01 * max_c, T_end=0.00005 * max_c,
                                            n_stages=12, max_iter_per_stage=300)
        full_sc = _score(a_lbfgs)
        print(f"Phase 2 L-BFGS: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_lbfgs.copy()

    # === Phase 3: LP refinement again (600s - 800s) ===
    phase3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase3_end:
        remaining = phase3_end - time.monotonic()
        a, sc = _lp_refine(best_a, rows, cols, time_per_lp=6, 
                          max_iters=int(remaining / 7))
        full_sc = _score(a)
        print(f"Phase 3 LP: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()

    # === Phase 4: Projected gradient with BB steps (800s - 1000s) ===
    phase4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase4_end:
        a_pg, sc_pg = _grad_descent_simplex(best_a, phase4_end - 3,
                                            lr_start=0.03, lr_end=0.001,
                                            top_k=30, max_iter=30000, eval_interval=300)
        full_sc = _score(a_pg)
        print(f"Phase 4 GradDesc: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_pg.copy()

    # === Phase 5: Diverse restarts (1000s - 1400s) ===
    phase5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase5_end:
        n_best = len(best_a)
        n_restarts = 6
        time_per_restart = (phase5_end - time.monotonic()) / max(1, n_restarts)

        for attempt in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            t_restart = time.monotonic()
            
            # Different perturbation types
            if attempt == 0:
                # Small Gaussian perturbation in z-space
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + 0.01 * rng.standard_normal(n_best)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            elif attempt == 1:
                # Moderate perturbation
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + 0.03 * rng.standard_normal(n_best)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            elif attempt == 2:
                # Symmetrized + perturbation
                a_sym = _symmetrize(best_a)
                z = np.log(np.maximum(a_sym, 1e-15))
                z = z + 0.01 * rng.standard_normal(n_best)
                z = z - z.mean()
                a_pert = _softmax_stable(z)
            elif attempt == 3:
                # Weighted mix with uniform
                mix = 0.7 * best_a + 0.3 * (np.ones(n_best) / n_best)
                a_pert = mix / mix.sum()
            elif attempt == 4:
                # Frequency-domain perturbation
                A = np.fft.rfft(best_a, n_best)
                mask = np.ones_like(A)
                n_high = n_best // 4
                mask[n_high:] *= 0.5
                A_pert = A * mask
                a_pert = np.fft.irfft(A_pert, n_best)
                a_pert = np.maximum(a_pert, 0)
                if a_pert.sum() < 1e-10:
                    a_pert = best_a.copy()
                a_pert = a_pert / a_pert.sum()
            else:
                # Sparse perturbation: zero out random 10% and redistribute
                a_pert = best_a.copy()
                n_zero = n_best // 10
                zero_idx = rng.choice(n_best, n_zero, replace=False)
                redistribute = a_pert[zero_idx].sum()
                a_pert[zero_idx] = 0
                keep_idx = np.setdiff1d(np.arange(n_best), zero_idx)
                a_pert[keep_idx] += redistribute / len(keep_idx)
                a_pert = a_pert / a_pert.sum()

            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            z_p = np.log(np.maximum(a_pert, 1e-15))
            z_p = z_p - z_p.mean()
            
            # Quick L-BFGS polish
            a_opt, sc_opt = _lbfgs_z_space(z_p, min(phase5_end - 10, t_restart + time_per_restart),
                                            T_start=0.002 * max_cp, T_end=0.00002 * max_cp,
                                            n_stages=5, max_iter_per_stage=200)
            full_sc = _score(a_opt)
            print(f"  Restart[{attempt}]: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

    # === Phase 6: Symmetry-enforced optimization (1400s - 1550s) ===
    phase6_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase6_end:
        a_sym = _symmetrize(best_a)
        sc_sym_init = _score(a_sym)
        print(f"  Symmetrized init: score={sc_sym_init:.10f}")
        if sc_sym_init < best_score * 1.01:  # Only if not much worse
            c_sym = _fft_convolve(a_sym, a_sym)
            max_cs = float(c_sym.max())
            z_sym = np.log(np.maximum(a_sym, 1e-15))
            z_sym = z_sym - z_sym.mean()
            a_sym_opt, sc_sym = _lbfgs_z_space(z_sym, phase6_end - 3,
                                                 T_start=0.001 * max_cs, T_end=0.00001 * max_cs,
                                                 n_stages=6, max_iter_per_stage=200)
            # Symmetrize result
            a_sym_opt = _symmetrize(a_sym_opt)
            full_sc = _score(a_sym_opt)
            print(f"  Symmetry opt: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_sym_opt.copy()

    # === Phase 7: Final LP polish (1550s - 1700s) ===
    phase7_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase7_end:
        remaining = phase7_end - time.monotonic()
        a, sc = _lp_refine(best_a, rows, cols, time_per_lp=5,
                          max_iters=int(remaining / 6))
        full_sc = _score(a)
        print(f"Phase 7 Final LP: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()

    # === Phase 8: Ultra-fine L-BFGS polish (1700s - deadline) ===
    if time.monotonic() < deadline - 5:
        c_fin = _fft_convolve(best_a, _fft_convolve(best_a, best_a).max() * 0 + best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        z_fin = np.log(np.maximum(best_a, 1e-15))
        z_fin = z_fin - z_fin.mean()
        a_fin, sc_fin = _lbfgs_z_space(z_fin, deadline - 3,
                                         T_start=0.0001 * max_cf, T_end=0.000001 * max_cf,
                                         n_stages=6, max_iter_per_stage=150)
        full_sc = _score(a_fin)
        print(f"Phase 8 Final L-BFGS: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_fin.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
