3. Check if all a_i > 0. If some a_i <= 0, remove those indices from the active set and re-solve.
4. Repeat until convergence.

This is the standard active set method for QPs. The main cost is inverting W (or solving W x = b) at each step. Since W is n×n with n=4096, this is O(n^3) per step, which is about 64 billion operations. That's too slow.

Actually, y[i] = sum_j w_{i+j} x[j]. If we define w_full as a vector of length 2n-1 (zero-padded), then y is the "cross-correlation" of w_full and x:

Alternatively: y[i] = sum_j w_full[i+j] x[j]. Let j' = i+j, so j = j'-i:
y[i] = sum_{j'} w_full[j'] x[j'-i]

This is the cross-correlation of w_full and x, which can be computed via FFT.

So: y = ifft(fft(w_full) * conj(fft(x))) (with appropriate normalization and length).

W = sum_{k in S} (w_k / |S|) * P_k

where P_k[i,j] = 1 if i+j = k, 0 otherwise.

"""
Hybrid optimizer combining L-BFGS with log-sum-exp smoothing, QP-based active set
refinement using Hankel matrix structure, Nesterov-accelerated subgradient with
Barzilai-Borwein step sizes, and diverse restart strategies to minimize
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _hankel_matvec(w_lags, S, x, n):
    """Compute W @ x where W[i,j] = w_lags[i+j] (zero if i+j not in S)."""
    y = np.zeros(n)
    for idx, k in enumerate(S):
        wk = w_lags[k]
        if wk == 0:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        j_lo = k - i_hi
        j_hi = k - i_lo
        y[i_lo:i_hi + 1] += wk * x[j_lo:j_hi + 1][::-1]
    return y


def _solve_qp_active_set(w_lags, S, n, max_iter=2000, tol=1e-10):
    """
    Solve: min a^T W a s.t. sum(a)=1, a>=0
    where W[i,j] = w_lags[i+j].
    Uses active set method with conjugate gradient for inner solves.
    Returns optimal a and the optimal value.
    """
    a = np.ones(n) / n
    active = np.ones(n, dtype=bool)
    lam = 0.0

    for outer in range(max_iter):
        n_act = active.sum()
        if n_act < 2:
            break

        active_idx = np.where(active)[0]
        # Solve W_A,A x = 1 using CG
        def matvec(x_sub):
            x_full = np.zeros(n)
            x_full[active_idx] = x_sub
            y_full = _hankel_matvec(w_lags, S, x_full, n)
            return y_full[active_idx]

        b_sub = np.ones(n_act)
        try:
            x_sub, info = cg(matvec, b_sub, rtol=1e-10, atol=1e-14, maxiter=min(500, 3*n_act))
            if info < 0:
                # CG failed, use simple solution
                x_sub = np.ones(n_act) / n_act
        except Exception:
            x_sub = np.ones(n_act) / n_act

        s = x_sub.sum()
        if s < 1e-15:
            x_sub = np.ones(n_act) / n_act
            s = x_sub.sum()
        x_sub = x_sub / s

        a_new = np.zeros(n)
        a_new[active_idx] = x_sub

        # Compute Lagrange multiplier
        Wa = _hankel_matvec(w_lags, S, a_new, n)
        lam = 2.0 * np.mean(Wa[active])

        # Check KKT for inactive indices
        inactive_idx = np.where(~active)[0]
        if len(inactive_idx) > 0:
            min_val = Wa[inactive].min() if len(inactive_idx) > 0 else np.inf
            # KKT: for inactive i, Wa_i >= lam (with our convention)
            # Actually: at optimum, for i not in support, (Wa)_i >= lam/2 * 2 = lam
            # Let's use: (Wa)_i >= lam for inactive
            # If violated, add the most violated index
            cands = inactive_idx[Wa[inactive_idx] < lam - tol]
            if len(cands) > 0:
                worst = cands[np.argmin(Wa[cands])]
                active[worst] = True
                continue

        # Check if any active index should be removed (a_i <= 0)
        if x_sub.min() <= tol:
            worst_idx = active_idx[np.argmin(x_sub)]
            active[worst_idx] = False
            continue

        a = a_new
        # Check convergence
        Wa_check = _hankel_matvec(w_lags, S, a, n)
        grad_active = Wa_check[active] - lam / 2.0
        if np.abs(grad_active).max() < 1e-10:
            break

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a, float(np.dot(a, _hankel_matvec(w_lags, S, a, n)))


def _qp_refine(a_init, deadline, min_time=30.0):
    """Iterative QP-based active set refinement."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_sc = _score(a)
    prev_sc = best_sc

    for iteration in range(50):

            break

        c = _full_convolve(a, a)
        c_max = c.max()
        # Identify active lags
        tol = max(1e-6 * c_max, 1e-12)
        S = list(np.where(c >= c_max - tol)[0])

        if len(S) == 0:
            S = [int(np.argmax(c))]
        # Cap active set size for efficiency
        if len(S) > 200:
            S_sorted = np.argsort(c)[::-1][:200]
            S = list(S_sorted)

        w_lags = np.zeros(2 * n - 1)
        for k in S:
            w_lags[k] = 1.0 / len(S)

        # Solve QP

        sc_new = _score(a_new)

            best_a = a_new.copy()
            a = a_new
            prev_sc = sc_new
            print(f"    QP iter {iteration}: score={sc_new:.10f}, |S|={len(S)}")
        else:
            # Shrink active set
            c = _full_convolve(a, a)
            c_max = c.max()
            tol_strict = max(1e-5 * c_max, 1e-11)
            S_tight = list(np.where(c >= c_max - tol_strict)[0])
            if len(S_tight) < len(S):
                S = S_tight
                w_lags = np.zeros(2 * n - 1)
                for k in S:
                    w_lags[k] = 1.0 / len(S)
                a_new, _ = _solve_qp_active_set(w_lags, S, n, max_iter=300)
                sc_new2 = _score(a_new)
                if sc_new2 < best_sc:
                    best_sc = sc_new2
                    best_a = a_new.copy()
                    a = a_new
            else:
                # No improvement, break

def _subgrad_bb(a_init, deadline, tol_frac=1e-4, max_steps=5000, use_nesterov=True):
    """Subgradient method with Barzilai-Borwein step sizes and optional Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    z_prev = z.copy()
    a_prev = _softmax(z_prev)
    grad_prev = None
    beta_anchor = None

    momentum = 0.0 if not use_nesterov else 0.3
    beta = 0.0
    prev_z = None
    prev_grad = None

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad = grad - grad.mean()

        # Barzilai-Borwein step size
        if prev_grad is not None and prev_z is not None:
            y = grad - prev_grad
            s = z - prev_z
            yy = float(np.dot(y, y))
            sy = float(np.dot(s, y))
            ss = float(np.dot(s, s))
            if yy > 1e-30 and ss > 1e-30:
                if i % 2 == 0:
                    alpha_bb = sy / yy if yy > 1e-15 else 0.01
                else:
                    alpha_bb = ss / sy if sy > 1e-15 else 0.01
                alpha_bb = np.clip(alpha_bb, 1e-8, 0.5)
            else:
                alpha_bb = beta_anchor if beta_anchor is not None else 0.01
        else:
            alpha_bb = beta_anchor if beta_anchor is not None else 0.01
            beta_anchor = alpha_bb

        # Nesterov momentum
        if use_nesterov and prev_z is not None:
            z_y = z + momentum * (z - prev_z)
            z_prev_calc = z_y
        else:
            z_y = z

        # Evaluate gradient at momentum point
        a_y = _softmax(z_y)
        c_y = _full_convolve(a_y, a_y)
        c_max_y = c_y.max()
        tol_y = tol_frac * c_max_y
        active_y = c_y >= (c_max_y - tol_y)
        if active_y.sum() == 0:
            active_y = np.zeros_like(c_y, dtype=bool)
            active_y[int(np.argmax(c_y))] = True
        w_y = np.zeros_like(c_y)
        w_y[active_y] = 1.0 / active_y.sum()
        a_y_rev = a_y[::-1]
        wc_y = _full_convolve(w_y, a_y_rev)
        h_y = wc_y[n - 1:2 * n - 1]
        W_y = float(np.dot(w_y, c_y))
        grad_y = 4.0 * n * a_y * (h_y - W_y)
        grad_y = grad_y - grad_y.mean()

        z_new = z_y - alpha_bb * grad_y
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        prev_grad = grad.copy()
        prev_z = z.copy()
        z = z_new

        # Periodic evaluation
        if i % 50 == 0 or i == max_steps - 1:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        max_iter = min(6000, max(200, int(remaining / 0.003)))
        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * (2.0 * t - 1.0) ** 2
        return a / a.sum()
    elif kind == "u_shape_strong":
        t = np.linspace(0, 1, n)
        a = 1.0 + 1.5 * (2.0 * t - 1.0) ** 2
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "fourier_mix":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2 * np.pi * 3 * t) + 0.05 * np.cos(2 * np.pi * 7 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_structured(a, rng, mode="fourier"):
    """Apply structured perturbation to escape local minima."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)

    if mode == "fourier":
        n_modes = int(rng.integers(1, 4))
        for _ in range(n_modes):
            freq = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.005, 0.03)
            phase = rng.uniform(0, 2.0 * np.pi)
            z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    elif mode == "gauss":
        sigma = rng.uniform(0.01, 0.05)
        z = z + sigma * rng.standard_normal(n)
    elif mode == "block":
        n_blocks = int(rng.integers(2, 8))
        block_size = max(4, n // n_blocks)
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.01, 0.01)
    elif mode == "mirror":
        # Perturb in a way that breaks/create symmetry
        half = n // 2
        delta = rng.uniform(-0.01, 0.01)
        z[:half] += delta
        z[half:] -= delta
    else:
        z = z + 0.02 * rng.standard_normal(n)

    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp smoothing,
    QP-based active set refinement exploiting Hankel matrix PSD structure,
    Nesterov-accelerated subgradient with BB step sizes, and diverse
    restart strategies to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: L-BFGS from parent ===
    print(f"\n=== Phase 1: L-BFGS from parent (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=500.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: QP-based active set refinement ===
    print(f"\n=== Phase 2: QP refinement (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_qp, sc_qp = _qp_refine(best_seq, deadline - 400, min_time=30.0)
        try_update(a_qp, "phase2_qp")

    # === Phase 3: Nesterov subgradient with BB ===
    print(f"\n=== Phase 3: Nesterov subgrad+BB (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_ng, sc_ng = _subgrad_bb(best_seq, deadline - 250, tol_frac=1e-4,
                                  max_steps=8000, use_nesterov=True)
        try_update(a_ng, "phase3_ng_bb")

    # === Phase 4: QP refinement again (tighter) ===
    print(f"\n=== Phase 4: QP refinement round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_qp2, sc_qp2 = _qp_refine(best_seq, deadline - 150, min_time=20.0)
        try_update(a_qp2, "phase4_qp2")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - 80 - time.monotonic()
            if remaining < 15:
                break
            max_iter = min(4000, max(200, int(remaining / 0.004)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 6: Diversification ===
    print(f"\n=== Phase 6: Diversification (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        restart_configs = [
            ("fourier_a", "fourier"),
            ("fourier_b", "fourier"),
            ("gauss_a", "gauss"),
            ("block_a", "block"),
            ("mirror_a", "mirror"),
            ("gauss_b", "gauss"),
        ]
        for name, mode in restart_configs:
            if time.monotonic() > deadline - 100:
                break

            p_start = _perturb_structured(best_seq, rng, mode=mode)
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            # Quick L-BFGS
            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=60.0)
            try_update(a_r, f"restart_{name}")

            # QP refinement
            if time.monotonic() < deadline - 50 and _score(a_r) < best_score_val + 0.01:
                a_r2, _ = _qp_refine(a_r, deadline - 50, min_time=20.0)
                try_update(a_r2, f"restart_{name}_qp")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        cur_n = len(best_seq)
        for target_n in [2048, 8192, 3072]:
            if time.monotonic() > deadline - 60:
                break
            if target_n == cur_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=30.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 8: Diverse initializations ===
    print(f"\n=== Phase 8: Diverse inits (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_cur = len(best_seq)
        diverse_kinds = ["u_shape", "u_shape_strong", "bimodal", "fourier_mix"]
        for kind in diverse_kinds:
            if time.monotonic() > deadline - 40:
                break
            a_div = _make_profile(n_cur, kind)
            # Interpolate toward best to keep it in a good region
            a_div = 0.5 * a_div + 0.5 * best_seq
            a_div = np.maximum(a_div, 1e-10)
            a_div = a_div / a_div.sum()
            a_div_opt = _lbfgs_schedule(a_div, deadline, min_time_remaining=20.0)
            try_update(a_div_opt, f"div_{kind}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break

            # QP with very tight active set
            if time.monotonic() < deadline - 5:
                a_f, _ = _qp_refine(best_seq, deadline - 5, min_time=5.0)
                try_update(a_f, f"final_qp_{rnd}")

            # Nesterov with tight tol
            if time.monotonic() < deadline - 5:
                a_n, _ = _subgrad_bb(best_seq, deadline - 5, tol_frac=1e-5,
                                     max_steps=5000, use_nesterov=True)
                try_update(a_n, f"final_ng_{rnd}")

            # Tiny-beta L-BFGS
            if time.monotonic() < deadline - 10:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 1e-7
                max_iter = 2000
                remaining = deadline - 10 - time.monotonic()
                if remaining > 5:
                    max_iter = min(2000, int(remaining / 0.003))
                    a_l, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
                    try_update(a_l, f"final_lbfgs_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
