```python
from scipy.optimize import minimize, linprog
from scipy.signal import fftconvolve

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):

    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
```

```python
def lp_projection(a, M, max_iter=50):
    """
    Find g >= 0, sum(g) = 1, that minimizes max(conv(a, g))
    (or equivalently, find the point closest to a in some metric
     within the feasible set {g: conv(a,g) <= M componentwise})
    
    Actually, let's reformulate:
    We want to find g that maximizes sum(g) subject to:
      g >= 0
      conv(a, g)[k] <= M for all k
    Then rescale g to have sum 1.
    """
    n = len(a)
    # Build the constraint matrix A (sparse)
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # This is a banded matrix
    
    # Use scipy.sparse
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: maximize sum(g) subject to A@g <= M*ones, g >= 0
    # In standard form: minimize -sum(g) subject to A@g <= M*ones, g >= 0
    
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n - 1)
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if result.success:
        g = result.x
        s = g.sum()
        if s > 0:
            return g / s
    return None
```

```python
Multi-strategy optimizer combining L-BFGS with log-sum-exp smoothing,
LP-based constraint projection, subgradient polishing, spectral shaping,
and multi-restart diversification to minimize 2*n*max(conv(a,a))/sum(a)^2.

Key innovation: LP projection step that exactly projects onto the constraint

compared to pure subgradient methods.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _fft_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.0125,
                              0.00625, 0.003, 0.0015, 0.00075,
                              0.000375, 0.00019, 0.0001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        est_iter_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(10000, max(200, int(remaining / (est_iter_time * 15))))
        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _build_conv_matrix(a):
    """Build sparse (2n-1) x n matrix A where A[k,j] = a[j] if 0<=k-j<n."""
    n = len(a)
    total = 2 * n - 1
    arange_total = np.arange(total)
    j_starts = np.clip(arange_total - n + 1, 0, n)
    j_ends = np.clip(arange_total + 1, 0, n)
    lengths = j_ends - j_starts

    rows = np.repeat(arange_total, lengths)
    cols_list = []
    vals_list = []
    for k in range(total):
        js = j_starts[k]
        je = j_ends[k]
        cols_list.append(np.arange(js, je))
        vals_list.append(a[js:je])
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)

    A = coo_matrix((vals, (rows, cols)), shape=(total, n))
    return A.tocsr()


def _lp_projection(a, slack=1.0):
    """
    Solve LP: maximize sum(g) subject to conv(a,g) <= M*slack, g >= 0.
    Returns normalized g or None if LP fails.
    """
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max()) * slack
    if M < 1e-30:
        return None

    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
        if result.success:
            g = result.x
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None


def _lp_iteration(a, t, slack=1.0):
    """One LP projection iteration: find g, blend with a, rescale."""
    g = _lp_projection(a, slack=slack)
    if g is None:
        return a
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s


def _lp_phase(a_start, deadline, max_iters=200):
    """Run multiple LP projection iterations with adaptive step size."""
    n = len(a_start)
    current_a = a_start.copy()
    current_sc = _score(current_a)
    t = 0.3
    stall = 0

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        t_step = max(0.02, t * (0.98 ** i))
        a_new = _lp_iteration(current_a, t_step, slack=1.0)
        sc_new = _score(a_new)

        if sc_new < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc_new
            stall = 0
            if (i + 1) % 10 == 0:
                print(f"    LP iter {i+1}: score={sc_new:.10f}, t={t_step:.4f}")
        else:
            stall += 1
            if stall > 20:
                break

    return current_a, current_sc


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _fft_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + i * 0.0003)
        z = z - step * grad
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.7
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.93

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _subgrad_direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=30000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        c_max = c.max()
        if c_max < 1e-30:
            break
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _fft_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        g = 2.0 * n * (h - W)
        g = g - g.mean()
        if np.linalg.norm(g) < 1e-14:
            break
        a_try = a - step_size * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0
        sc_cur = _score(a)
        if sc_try < sc_cur - 1e-15:
            a = a_try
            step_size = min(step_size * 1.005, 0.1)
        else:
            stall += 1
            step_size *= 0.85
        step_size = max(step_size, 1e-12)
        if stall > 300:
            break

    return best_a, best_sc


def _spectral_equalize(a, n_iters=50, deadline=None):
    """
    Iterative spectral equalization: try to flatten |FFT(a)|^2
    while maintaining a >= 0 and sum(a) = 1.
    """
    n = len(a)
    current_a = a.copy()
    current_sc = _score(current_a)

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        N = 1
        while N < 2 * n - 1:
            N <<= 1
        F = np.fft.rfft(current_a, N)
        mag_sq = np.abs(F) ** 2
        mean_mag = mag_sq.mean()
        if mean_mag < 1e-30:
            break

        # Compute "equalization factor": push high-magnitude bins down,
        # low-magnitude bins up
        # Use a soft version: factor = 1 - alpha * (mag_sq - mean) / mean
        alpha = 0.1
        normalized = (mag_sq - mean_mag) / (mean_mag + 1e-30)
        factor = 1.0 - alpha * normalized
        factor = np.maximum(factor, 0.1)  # don't go too aggressive

        # Modify the spectrum magnitude
        new_mag = np.abs(F) * np.sqrt(factor)
        new_F = new_mag * np.exp(1j * np.angle(F))
        new_a = np.fft.irfft(new_F, N)[:n]
        new_a = np.maximum(new_a, 0.0)
        s = new_a.sum()
        if s < 1e-10:
            break
        new_a = new_a / s
        sc_new = _score(new_a)

        if sc_new < current_sc - 1e-13:
            current_a = new_a
            current_sc = sc_new
        else:
            alpha *= 0.8
            if alpha < 0.001:
                break

    return current_a, current_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.005 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(4, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
        z[start:end] += rng.uniform(-0.02, 0.02)
    z = z - z.mean()
    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _local_search(a, deadline, max_rounds=300):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.008

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        idx = int((rnd * 997 + 13) % n)
        jdx = int((rnd * 997 + 41) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        for delta_frac in [0.3, 0.5, 0.7]:
            s_pair = best_a[idx] + best_a[jdx]
            if s_pair < 1e-15:
                continue
            new_val_i = s_pair * (1.0 - delta_frac + step)
            new_val_j = s_pair * (delta_frac - step)
            if new_val_i < 0 or new_val_j < 0:
                continue
            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True

        if not improved and rnd > 30:
            step *= 0.7
            if step < 1e-6:
                break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp smoothing,
    LP-based constraint projection, spectral equalization, subgradient
    polishing, and multi-restart to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev-best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: L-BFGS with smoothing ===
    print(f"\n=== Phase 1: L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    p1_deadline = deadline - 1200.0
    a_p1 = _lbfgs_schedule(best_seq, p1_deadline, min_time_remaining=400.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: LP projection (key innovation) ===
    print(f"\n=== Phase 2: LP Projection (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 600:
        n = len(best_seq)
        if n <= 8192:
            lp_deadline = deadline - 500.0
            a_lp, sc_lp = _lp_phase(best_seq, lp_deadline, max_iters=300)
            try_update(a_lp, "phase2_lp")
        else:
            print("  Skipping LP: n too large")

    # === Phase 3: Spectral equalization ===
    print(f"\n=== Phase 3: Spectral Equalization (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_se, sc_se = _spectral_equalize(best_seq, n_iters=100,
                                         deadline=deadline - 350)
        try_update(a_se, "phase3_spectral")

    # === Phase 4: Subgradient polishing ===
    print(f"\n=== Phase 4: Subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 250,
                                        tol_frac=1e-4, max_steps=15000, lr_init=0.03)
        try_update(a_sg, "phase4_sg_z")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sg2, sc_sg2 = _subgrad_direct_polish(best_seq, deadline - 150,
                                               tol_frac=1e-5, max_steps=20000)
        try_update(a_sg2, "phase4_sg_direct")

    # === Phase 5: Fine L-BFGS ===
    print(f"\n=== Phase 5: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        c_b = _fft_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([3e-5, 1.5e-5, 7e-6, 3e-6, 1e-6])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(5000, max(200, int(remaining / 0.004)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 6: Another LP pass on the improved solution ===
    print(f"\n=== Phase 6: LP Pass 2 (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n = len(best_seq)
        if n <= 8192:
            a_lp2, sc_lp2 = _lp_phase(best_seq, deadline - 80, max_iters=150)
            try_update(a_lp2, "phase6_lp2")

    # === Phase 7: Multi-restart ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            ("gauss_003", "gauss", 0.03, 0),
            ("gauss_008", "gauss", 0.08, 0),
            ("fourier_3", "fourier", 0, 3),
            ("fourier_5", "fourier", 0, 5),
            ("block_4", "block", 0, 4),
            ("block_8", "block", 0, 8),
        ]
        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 60:
                break
            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, modes)
            else:
                p_start = _perturb_block(best_seq, rng, modes)

            sc_start = _score(p_start)
            if sc_start < best_score_val + 0.02:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=40.0)
                try_update(a_r, f"restart_{name}")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        current_n = len(best_seq)
        for target_n in [8192, 2048, 6144]:
            if time.monotonic() > deadline - 40:
                break
            if target_n == current_n:
                continue
            if target_n > 8192:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=25.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final aggressive cycle ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(3):
            if time.monotonic() > deadline - 5:
                break
            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _fft_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-7
            a_f, _, _ = _lbfgs_run(best_seq, beta, n, 2000)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Quick subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 5,
                                           tol_frac=1e-5, max_steps=3000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

            # Local search
            if time.monotonic() < deadline - 5:
                a_l, _ = _local_search(best_seq, deadline - 5, max_rounds=200)
                try_update(a_l, f"final_ls_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
