import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max_c."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_boundary_init(n, rng):
    """
    Initialize with boundary-aware profile.
    Boundary elements are O(1/sqrt(n)), interior is O(1/n).
    """
    a = np.ones(n) / n
    # Boost boundaries
    alpha = 1.0 / np.sqrt(2.0 * n)
    # Smoothly blend from boundary to interior
    blend_width = max(1, n // 64)
    for i in range(blend_width):
        t = 1.0 - i / blend_width
        a[i] = alpha * t + (1.0 / n) * (1.0 - t)
        a[n-1-i] = alpha * t + (1.0 / n) * (1.0 - t)
    a = np.maximum(a, 0.0)
    return a / a.sum()

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

def _fista_backtrack(a_init, beta, deadline, max_iter):
    """
    FISTA with backtracking line search on the softmax-smoothed max.
    Minimize: (1/beta) * log(sum_k exp(beta * c[k]))
    on the simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    # Compute initial Lipschitz constant
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        # Compute smoothed gradient
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient of smoothed objective
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        # Backtracking line search
        # Try step, halve until we get sufficient decrease
        f_curr = float(np.dot(w, c_y))  # approximate
        step_try = step
        a_next = _proj_simplex(y - step_try * grad)
        
        # Check if we need to backtrack (simplified: just check every 50 iters)
        if it % 50 == 0:
            c_check = _full_convolve(a_next, a_next)
            max_check = float(c_check.max())
            L_new = 4.0 * n * max_check / beta
            step = 1.0 / max(L_new, 1e-10)
        
        sc = _score_norm(a_next)
        if sc < best_score:
            best_score = sc
            best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """
    Adam optimization in log-space.
    Parameters:
    - a_init: initial sequence (sum=1)
    - deadline: time limit
    - beta_schedule: list of beta values to use (annealing)
    - lr_init: initial learning rate
    - max_iter_total: maximum total iterations
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        iter_per_phase = min(remain_iter, max(10000, max_iter_total // len(beta_schedule)))
        
        while it_in_phase < iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            # Gradient of the smoothed objective w.r.t. z
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine annealing for learning rate
            progress = min(1.0, it_in_phase / iter_per_phase)
            current_lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            current_lr = max(current_lr, lr_init * 0.01)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 0.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _subgradient_polish(a_init, deadline, n_steps=500):
    """
    Exact subgradient polishing: at each step, find the argmax of c,
    compute the exact subgradient, and take a projected step.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    # Adaptive step size
    step = 0.01 / n
    
    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a_curr, a_curr)
        max_idx = int(np.argmax(c))
        max_val = float(c[max_idx])
        
        # Subgradient of c[max_idx] w.r.t. a
        # c[k] = sum_j a[j] * a[k-j], gradient w.r.t. a[j] = a[k-j] + a[j-k] (if valid)
        # For the full convolution index k (0 to 2n-2):
        # c[k] = sum_j a[j] * a[k-j] where 0<=j<n and 0<=k-j<n
        # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n and j!=k-j)... 
        # Actually, since c[k] = sum_j a[j]*a[k-j], and both terms involve a:
        # dc[k]/da[j] = a[k-j] + a[j-k] (the second term is nonzero only if j-k >= 0, i.e., j >= k)
        # But wait, in the convolution sum, each pair (j, k-j) appears once.
        # c[k] = sum_j a[j] * a[k-j]
        # dc[k]/da[j] = a[k-j] (from the first factor) + a[j] * d(a[k-j])/d(a[j]) 
        # Hmm, this is a product of two a's, so:
        # dc[k]/da[j] = a[k-j] + a[j-k] (by symmetry, the "second" term)
        # But a[k-j] is defined only when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # And a[j-k] is defined only when 0 <= j-k < n, i.e., k <= j <= k+n-1
        
        # So: dc[k]/da[j] = (a[k-j] if 0<=k-j<n else 0) + (a[j-k] if 0<=j-k<n else 0) - (a[j] if j == k-j, i.e., 2j=k)
        # Wait, I'm overcomplicating. Let me just use the linearity:
        # c[k] = sum_j a[j] * a[k-j]
        # If we write this as a vector: c[k] = a^T P_k a where P_k is the permutation that maps j -> k-j
        # Then dc[k]/da = 2 * P_k a (restricted to valid indices)
        # More precisely: dc[k]/da[j] = a[k-j] + a[j-k] (with out-of-range = 0)
        # But when j = k-j (i.e., 2j = k), the two terms are the same, so we get 2*a[j] (not a[j] + a[j] = 2a[j], which is correct for the derivative of a[j]^2).
        
        # Actually, let's just compute it directly:
        # c[k] = sum_{j: valid} a[j] * a[k-j]
        # dc[k]/da[j] = a[k-j] (from differentiating the first factor) 
        #              + a[j-k] (from differentiating the second factor, where the "k-j" index equals j, so j = k-j' means j' = k-j, and the contribution is a[j] * 1 for the second factor when j' = k-j, i.e., when the second index equals j)
        
        # I'll just compute it numerically via the convolution:
        # grad[j] = sum over active k of w[k] * (a[k-j] + a[j-k])
        # For a single k: grad[j] = a[k-j] (if valid) + a[j-k] (if valid)
        
        k = max_idx
        grad = np.zeros(n)
        # a[k-j] for j in [max(0, k-n+1), min(n-1, k)]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            grad[j_start:j_end+1] += a[k - j_end:k - j_start + 1][::-1]
        # a[j-k] for j in [k, min(n-1, k+n-1)]
        j_start2 = max(0, k)
        j_end2 = min(n - 1, k + n - 1)
        if j_start2 <= j_end2:
            grad[j_start2:j_end2+1] += a[j_start2 - k:j_end2 - k + 1]
        
        # The factor of 2 is already accounted for (each term appears once in each direction)
        # Actually wait: c[k] = sum_j a[j]*a[k-j]. 
        # d/d(a[j]) of this = a[k-j] (if k-j in range) + [derivative of a[k-j] w.r.t. a[j]]
        # The second term: a[k-j] as a function of a[j] is a[j'] where j' = k-j. 
        # This equals a[j] only when j' = j, i.e., k-j = j, i.e., j = k/2.
        # So: dc[k]/da[j] = a[k-j] (if valid) + 1 (if j = k/2 and k is even)
        # Hmm, that's not quite right either.
        
        # Let me just do it cleanly:
        # c[k] = sum_{j=0}^{n-1} a[j] * a[k-j] (with a[m] = 0 if m not in [0,n-1])
        # dc[k]/da[j] for a specific j:
        # = a[k-j] (differentiating the first a[j] in the sum term where index = j)
        #   + a[j-k] (differentiating the second factor: a[k-j'] where j' = j-k... wait)
        
        # OK I'm confusing myself. Let me use a different approach:
        # c = conv(a, a). The gradient of c[k] w.r.t. a is:
        # (grad_a c[k])[j] = a[k-j] + a[j-k] with out-of-range = 0
        # This is because c[k] = sum_j a[j] a[k-j], and:
        # - The term a[j] a[k-j]: derivative w.r.t. a[j] gives a[k-j]
        # - The term a[j'] a[k-j'] where k-j' = j, i.e., j' = k-j: derivative w.r.t. a[j] gives a[k-j] * ... 
        # No wait. Let me index properly.
        # c[k] = sum_{j} a_j * a_{k-j}
        # d c[k] / d a_m = sum_j [delta_{j,m} * a_{k-j} + a_j * delta_{k-j,m}]
        # = a_{k-m} + a_{k-m} ... no
        # = a_{k-m} (from j=m in first factor) + a_{m} * (1 if k-m is a valid index and the second factor is a_m, i.e., k-j=m so j=k-m)
        # = a_{k-m} + a_{k-m} (if both terms are valid)
        # Wait: delta_{k-j, m} means k-j = m, so j = k-m.
        # So: d c[k]/d a_m = a_{k-m} + a_{k-m} (when j = k-m is valid)
        # = 2 * a_{k-m} (when 0 <= k-m < n)
        # But that's only if k-m != m (i.e., 2m != k). If 2m = k, then both terms give a_m, and the total is 2*a_m, which is correct (derivative of a_m^2 is 2*a_m).
        
        # So: grad[j] = 2 * a[k-j] (if 0 <= k-j < n), else 0
        
        grad = np.zeros(n)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # grad[j] = 2 * a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_start to k-j_end, reversed
            grad[j_start:j_end+1] = 2.0 * a[k - j_end:k - j_start + 1][::-1]
        
        # Normalize step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Take step with adaptive learning rate
        a_try = a_curr - step * grad
        a_try = _proj_simplex(a_try)
        
        c_try = _full_convolve(a_try, a_try)
        sc_try = float(c_try.max())
        
        if sc_try < max_val:
            a_curr = a_try
            step = min(step * 1.05, 0.05)  # increase step slightly
        else:
            step = max(step * 0.7, 1e-6)
        
        sc = _score_norm(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                sc_parent = _score(parent)
                print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
                best_seq = parent.copy()
                best_score = sc_parent
    except Exception:
        parent = None

    if parent is None:
        print("No parent available, starting from scratch.")
        n_init = 4096
        a0 = _make_profile(n_init, "boundary", rng)
        sc0 = _score(a0)
        print(f"Initial boundary profile n={n_init}: score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # Determine lengths to try
    if parent is not None:
        n_parent = len(parent)
        lengths_to_try = [n_parent]
        if n_parent <= 6000:
            lengths_to_try.append(8192)
        if n_parent >= 4096:
            lengths_to_try.append(4096)
    else:
        lengths_to_try = [4096, 8192]
    
    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    
    # Time allocation
    time_left = deadline - t_start
    # Give more time to the first (parent) length
    if len(lengths_to_try) == 1:
        time_alloc = {lengths_to_try[0]: time_left - 30}
    elif len(lengths_to_try) == 2:
        time_alloc = {lengths_to_try[0]: int(time_left * 0.6 - 30), 
                     lengths_to_try[1]: int(time_left * 0.3)}
    else:
        per = (time_left - 30) // len(lengths_to_try)
        time_alloc = {l: per for l in lengths_to_try}

    for n_target in lengths_to_try:
        if time.monotonic() > deadline - 30:
            break
        alloc = time_alloc.get(n_target, 300)
        print(f"\n=== Optimizing n={n_target} (alloc={alloc}s, time_left={deadline - time.monotonic():.1f}s) ===")

        # Initialize
        if parent is not None and n_target == len(parent):
            a_work = parent.copy()
        elif parent is not None:
            a_work = _interp_profile(parent, n_target)
        else:
            a_work = _make_profile(n_target, "boundary", rng)
        
        a_work = a_work / a_work.sum()
        sc_work = _score(a_work)
        print(f"  Starting: score={sc_work:.10f}")

        best_for_n = a_work.copy()
        best_score_for_n = sc_work

        # Phase A: FISTA with aggressive beta annealing
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())

        # More aggressive: start smaller beta
        betas_fista = []
        beta = 0.01 * max_cc
        while beta > 5e-7 * max_cc:
            betas_fista.append(beta)
            beta *= 0.35
        if len(betas_fista) > 15:
            betas_fista = betas_fista[:15]

        fista_deadline = min(deadline - 60, time.monotonic() + alloc * 0.5)
        
        for i, beta in enumerate(betas_fista):
            if time.monotonic() > fista_deadline:
                break
            remain = fista_deadline - time.monotonic()
            if remain < 5:
                break
            phase_dl = time.monotonic() + min(remain, max(5.0, alloc * 0.05))
            max_it = max(2000, int((phase_dl - time.monotonic()) * 5000))
            a_work, sc = _fista_backtrack(a_work, beta, phase_dl, max_it)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score_for_n:
                best_score_for_n = sc_full
                best_for_n = a_work.copy()
            if i == 0 or i == len(betas_fista) - 1:
                print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase B: Adam log-space
        if time.monotonic() < deadline - 40:
            c_curr = _full_convolve(a_work, a_work)
            max_cc = float(c_curr.max())
            betas_adam = []
            beta = 0.001 * max_cc
            while beta > 1e-6 * max_cc:
                betas_adam.append(beta)
                beta *= 0.5
            if len(betas_adam) > 12:
                betas_adam = betas_adam[:12]

            adam_deadline = min(deadline - 30, time.monotonic() + alloc * 0.35)
            a_opt, sc_opt = _adam_logspace(
                a_work, adam_deadline,
                betas_adam, lr_init=0.01, max_iter_total=500000
            )
            sc_opt_full = _score(a_opt)
            if sc_opt_full < best_score_for_n:
                best_score_for_n = sc_opt_full
                best_for_n = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase C: Subgradient polish
        if time.monotonic() < deadline - 10:
            sg_deadline = min(deadline - 5, time.monotonic() + 30)
            a_sg, sc_sg = _subgradient_polish(best_for_n, sg_deadline, n_steps=1000)
            sc_sg_full = _score(a_sg)
            if sc_sg_full < best_score_for_n:
                best_score_for_n = sc_sg_full
                best_for_n = a_sg.copy()
            print(f"  Subgrad: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Update global best
        if best_score_for_n < best_score:
            best_score = best_score_for_n
            best_seq = best_for_n.copy()
            print(f"  *** New global best: score={best_score:.10f}, n={n_target} ***")

    # Diversification on best found
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Diversification (score={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        n_restarts = 6
        per_restart = (deadline - time.monotonic() - 30.0) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            if i == 0:
                p_start = _perturb(a_work, 0.015, rng)
            elif i == 1:
                sz = max(1, n_best // 2048)
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif i == 2:
                p_start = _perturb(a_work, 0.04, rng)
            elif i == 3:
                # Blend with boundary profile
                alt = _make_profile(n_best, "boundary", rng)
                p_start = 0.8 * a_work + 0.2 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 4:
                # Local blip perturbation
                p_start = a_work.copy()
                center = rng.integers(n_best // 4, 3 * n_best // 4)
                width = n_best // 32
                p_start[center-width:center+width] *= 1.1
                p_start = p_start / p_start.sum()
            else:
                # Random mask perturbation
                p_start = a_work.copy()
                mask = rng.integers(0, n_best, size=n_best // 8)
                p_start[mask] *= (0.8 + 0.4 * rng.random(len(mask)))
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0005 * max_cs, 0.0002 * max_cs, 0.00005 * max_cs, 0.00001 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                bs, lr_init=0.005, max_iter_total=200000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [1e-5 * max_cp, 5e-6 * max_cp, 2e-6 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t * 0.6,
                betas_ult, lr_init=0.001, max_iter_total=100000
            )
            sc_pol_full = _score(a_pol)

"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
- Boundary-aware initialization exploiting the O(1/sqrt(n)) boundary structure
- FISTA with adaptive Lipschitz constant and aggressive beta annealing
- Adam in log-space with cosine-annealed learning rate
- Exact subgradient polishing on the true (unsmoothed) max
- Multi-length exploration (n=4096, 8192) with interpolated warm starts
- Diversified restarts with varied perturbation strategies

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_boundary_init(n):
    a = np.ones(n) / n
    alpha = 1.0 / np.sqrt(2.0 * n)
    blend_width = max(2, n // 32)
    for i in range(blend_width):
        t = 1.0 - i / blend_width
        a[i] = alpha * t + (1.0 / n) * (1.0 - t)
        if n - 1 - i != i:
            a[n - 1 - i] = alpha * t + (1.0 / n) * (1.0 - t)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _make_profile(n, kind="uniform", rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":

    elif kind == "tapered":
        t = np.linspace(0, 1, n)

        return np.maximum(a, 1e-10) / (np.maximum(a, 1e-10)).sum()
    elif kind == "bimodal":

        return np.maximum(a, 1e-10) / (np.maximum(a, 1e-10)).sum()
    elif kind == "random":
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "boundary":

    elif kind == "random_boundary":
        a = rng.random(n) + 0.01
        a[0] *= 4.0

    elif kind == "gaussian":

    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):

    z = z + sigma * rng.standard_normal(len(a))

def _fista_optimize(a_init, beta, deadline, max_iter):
    """FISTA on softmax-smoothed max over the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_curr.max())
    L = 4.0 * n * max_c_init / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    stagnant = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient: 2 * conv(w, a_rev)[n-1:2n-1]
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)

        a_next = _proj_simplex(y - step * grad)

        else:

                stagnant = 0

        a_curr = a_next

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with cosine LR schedule per phase."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    n_phases = len(beta_schedule)
    iter_per_phase = max(5000, max_iter_total // max(1, n_phases))

    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        remain = max_iter_total - t
        it_in_phase = min(iter_per_phase, remain)
        if it_in_phase <= 0:
            break

        for _ in range(it_in_phase):
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine annealing within this phase
            progress = min(1.0, t / (t + it_in_phase))
            current_lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            current_lr = max(current_lr, lr_init * 0.05)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -55.0, 55.0)
            z = z - z.mean()

            if t % 80 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

    """Exact subgradient descent on the true max of conv(a,a)."""
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_curr.copy()

        if grad_norm > 1e-15:
            grad = grad / grad_norm

        a_try = a_curr - step * grad
        a_try = _proj_simplex(a_try)

        c_try = _full_convolve(a_try, a_try)
        sc_try = float(c_try.max())

        if sc_try < max_val:
            a_curr = a_try
            if sc_try < max_val * 0.999:
                step = min(step * 1.1, 0.05)
                best_step = step
        else:
            step = max(step * 0.8, 1e-8)
            if step > best_step * 0.5:
                step = best_step * 0.5

        sc = _score_norm(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                sc_parent = _score(parent)
                print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
                best_seq = parent.copy()
                best_score = sc_parent
    except Exception:
        parent = None

    if parent is None:
        n_init = 4096
        a0 = _make_boundary_init(n_init)
        sc0 = _score(a0)
        print(f"No parent. Boundary init n={n_init}: score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # Determine lengths to try
    if parent is not None:
        n_parent = len(parent)
        lengths_to_try = [n_parent]
        if n_parent <= 5000:
            lengths_to_try.append(8192)
    else:
        lengths_to_try = [4096, 8192]

    lengths_to_try = list(dict.fromkeys(lengths_to_try))

    # Time allocation: prioritize parent length
    time_left_total = deadline - t_start
    if len(lengths_to_try) == 1:
        time_alloc = {lengths_to_try[0]: time_left_total - 25}
    else:
        time_alloc[lengths_to_try[0]] = int(time_left_total * 0.55 - 20)
        time_alloc[lengths_to_try[1]] = int(time_left_total * 0.30)

    for n_target in lengths_to_try:
        if time.monotonic() > deadline - 25:
            break
        alloc = time_alloc.get(n_target, 200)
        print(f"\n=== Optimizing n={n_target} (alloc={alloc}s, remaining={deadline - time.monotonic():.1f}s) ===")

        # Initialize
        if parent is not None and n_target == len(parent):
            a_work = parent.copy()
        elif parent is not None:
            a_work = _interp_profile(parent, n_target)
        else:
            a_work = _make_boundary_init(n_target)

        a_work = a_work / a_work.sum()
        sc_work = _score(a_work)
        print(f"  Start: score={sc_work:.10f}")

        best_for_n = a_work.copy()
        best_score_for_n = sc_work

        # Phase A: FISTA with aggressive beta annealing
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())

        betas_fista = []
        beta = 0.008 * max_cc
        while beta > 3e-7 * max_cc:
            betas_fista.append(beta)
            beta *= 0.3
        if len(betas_fista) > 14:
            betas_fista = betas_fista[:14]

        fista_deadline = min(deadline - 50, t_start + alloc * 0.55 + 5)
        for i, beta in enumerate(betas_fista):
            if time.monotonic() > fista_deadline:
                break
            remain = fista_deadline - time.monotonic()
            if remain < 3:
                break
            phase_dl = time.monotonic() + min(remain, max(5.0, alloc * 0.04))
            max_it = max(3000, int((phase_dl - time.monotonic()) * 6000))
            a_work, sc = _fista_optimize(a_work, beta, phase_dl, max_it)
            a_work = a_work / a_work.sum()
            sc_full = _score(a_work)
            if sc_full < best_score_for_n:
                best_score_for_n = sc_full
                best_for_n = a_work.copy()
            if i == 0 or i >= len(betas_fista) - 2:
                print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase B: Adam log-space
        if time.monotonic() < deadline - 35:
            c_curr = _full_convolve(a_work, a_work)
            max_cc = float(c_curr.max())
            betas_adam = []
            beta = 0.0008 * max_cc
            while beta > 5e-7 * max_cc:
                betas_adam.append(beta)
                beta *= 0.45
            if len(betas_adam) > 14:
                betas_adam = betas_adam[:14]

            adam_dl = min(deadline - 25, time.monotonic() + alloc * 0.35)
            a_opt, sc_opt = _adam_logspace(
                a_work, adam_dl,
                betas_adam, lr_init=0.008, max_iter_total=600000
            )
            sc_opt_full = _score(a_opt)
            if sc_opt_full < best_score_for_n:
                best_score_for_n = sc_opt_full
                best_for_n = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase C: Subgradient polish
        if time.monotonic() < deadline - 15:
            sg_dl = min(deadline - 8, time.monotonic() + min(40.0, alloc * 0.1))
            a_sg, sc_sg = _subgradient_polish(best_for_n, sg_dl, n_steps=3000)
            sc_sg_full = _score(a_sg)
            if sc_sg_full < best_score_for_n:
                best_score_for_n = sc_sg_full
                best_for_n = a_sg.copy()
            print(f"  Subgrad: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Update global
        if best_score_for_n < best_score:
            best_score = best_score_for_n
            best_seq = best_for_n.copy()
            print(f"  *** Global best: score={best_score:.10f}, n={n_target} ***")

    # Diversification
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Diversification (score={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        n_restarts = 6
        per_restart = (deadline - time.monotonic() - 25.0) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > deadline - 18:
                break

            if i == 0:
                p_start = _perturb(a_work, 0.012, rng)
            elif i == 1:
                sz = max(1, n_best // 1024)
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif i == 2:
                p_start = _perturb(a_work, 0.035, rng)
            elif i == 3:
                alt = _make_boundary_init(n_best)
                p_start = 0.85 * a_work + 0.15 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 4:
                p_start = a_work.copy()
                center = rng.integers(n_best // 4, 3 * n_best // 4)
                width = max(3, n_best // 25)
                p_start[center - width:center + width] *= 1.15
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()
            else:
                p_start = a_work.copy()
                n_mod = max(10, n_best // 6)
                mask = rng.choice(n_best, size=n_mod, replace=False)
                p_start[mask] *= (0.75 + 0.5 * rng.random(n_mod))
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0004 * max_cs, 0.00015 * max_cs, 0.00004 * max_cs, 0.00001 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(4.0, per_restart),
                bs, lr_init=0.004, max_iter_total=250000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

            # Quick subgradient polish on the restart result
            if time.monotonic() < deadline - 12:
                a_sg, sc_sg = _subgradient_polish(p_best, time.monotonic() + min(8.0, deadline - time.monotonic() - 6), n_steps=500)
                sc_sg_full = _score(a_sg)
                if sc_sg_full < best_score:
                    best_score = sc_sg_full
                    best_seq = a_sg.copy()
                    print(f"  Restart {i} SG: score={sc_sg_full:.10f}")

    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 12:
        print(f"\n=== Final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [8e-6 * max_cp, 4e-6 * max_cp, 1.5e-6 * max_cp]
        per_t = deadline - time.monotonic() - 6
        if per_t > 8:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t * 0.55,
                betas_ult, lr_init=0.0008, max_iter_total=80000
            )
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_pol_full:.10f}")

            if time.monotonic() < deadline - 5:
                a_sg2, sc_sg2 = _subgradient_polish(
                    best_seq, time.monotonic() + min(8.0, deadline - time.monotonic() - 3), n_steps=1000
                )
                sc_sg2_full = _score(a_sg2)
                if sc_sg2_full < best_score:
                    best_score = sc_sg2_full
                    best_seq = a_sg2.copy()
                    print(f"  Final SG: score={sc_sg2_full:.10f}")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.full(256, 0.01)
        sc_final = _score(final)
        print(f"\n=== FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s ===")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.full(256, 0.01)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
