def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_wrt_a(a, c, n, tol_frac=0.005):
    """Compute subgradient of max(c) w.r.t. a using active set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * (h - W) * 2.0


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cv - target_sum))[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _adam_optimize(a_init, deadline, max_steps=20000, lr=0.02, beta1=0.9, beta2=0.999):
    """Adam optimization on log-parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    t = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Gradient w.r.t. z (through softmax)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_a = 2.0 * (h - W)  # gradient of max(c) w.r.t. a (approximate)
        # Chain rule through softmax: grad_z = a * (grad_a - sum(a * grad_a))
        grad_z = a * (grad_a - np.dot(a, grad_a))

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr_t = lr * np.sqrt(1 - beta2 ** t) / (1 - beta1 ** t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = z - z.mean()

        if step % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _projected_subgrad(a_init, deadline, max_steps=50000):
    """Projected subgradient with Polyak-Ruppert averaging."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    a_avg = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)

    for t in range(1, max_steps + 1):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Subgradient w.r.t. a
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h  # simplified subgradient

        # Project gradient onto tangent space of simplex
        grad_t = grad - np.mean(grad)

        # Step size (decreasing)
        alpha = 1.0 / (np.sqrt(t) + 1.0)
        a_new = a - alpha * grad_t
        a_new = _project_simplex(a_new, 1.0)

        a_avg = a_avg + (a_new - a_avg) / t
        a = a_new

        if t % 500 == 0:
            sc = _score(a_avg)
            if sc < best_sc:
                best_sc = sc
                best_a = a_avg.copy()
            sc2 = _score(a)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a.copy()

    sc_final = _score(a_avg)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_avg.copy()
    return best_a, best_sc


def _lp_refine(a_init, deadline, n_lp_calls=5):
    """LP-based direction finding for refinement."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for lp_call in range(n_lp_calls):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        active_idx = np.where(active)[0]

        # Limit active set size for tractability
        if len(active_idx) > 200:
            # Keep only the most active constraints
            c_vals = c[active_idx]
            top_k = np.argsort(c_vals)[::-1][:200]
            active_idx = active_idx[top_k]

        m = len(active_idx)

        # Build constraint matrix A: A[i, j] = a[active_idx[i] - j] if valid
        A = np.zeros((m, n))
        for i, k in enumerate(active_idx):
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                A[i, j_min:j_max + 1] = a[j_max:j_min - 1:-1] if j_max > j_min else a[j_min]

        # Solve LP: max sum(b) s.t. A @ b <= c_max, b >= 0
        try:
            c_obj = -np.ones(n)
            b_ub = np.full(m, c_max)
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
            if res.success and res.x is not None:
                b = res.x
                b_sum = b.sum()
                if b_sum > 1e-10:
                    b = b / b_sum
                    # Blend: new_a = (1-t)*a + t*b
                    for t_blend in [0.3, 0.1, 0.05]:
                        a_try = (1 - t_blend) * a + t_blend * b
                        a_try = np.maximum(a_try, 0)
                        s_try = a_try.sum()
                        if s_try < 1e-10:
                            continue
                        a_try = a_try / s_try
                        sc_try = _score(a_try)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a_try.copy()
        except Exception:
            pass

        # Update a to best_a for next LP call
        a = best_a.copy()

    return best_a, best_sc


def _lbfgs_refine(a_init, deadline, min_remaining=30.0):
    """L-BFGS with adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time = 0.005 * max(1, n / 2048.0)
        max_iter = min(3000, max(20, int(remaining / (est_time * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(fun, z, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-15,
                                       'gtol': 1e-12, 'maxcor': 50, 'maxls': 60})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrink_and_refine(a_init, deadline, min_remaining=20):
    """Adaptive shrinkage: zero out small elements and re-optimize."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    a_sorted_idx = np.argsort(a_init)

    fractions = [0.005, 0.01, 0.02, 0.05, 0.08, 0.12, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline - min_remaining:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            a_sorted_idx = np.argsort(best_a)

    return best_a, best_sc


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos1":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.7 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos4":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.4 * np.cos(8.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "w_shape":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.6 * np.cos(2.0 * np.pi * t) + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_structured":
        if rng is None:
            rng = np.random.default_rng(42)
        # Mix of different frequency components
        t = np.arange(n)
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 7, 11]:
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "half_concentrated":
        # More mass on one side
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.1
        return a / a.sum()
    elif kind == "anti_palindrome":
        # a_i != a_{n-1-i} to reduce c_{n-1}
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient polish with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.002 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: Adam + L-BFGS + LP + projected subgradient + shrinkage.
    Focuses compute on the best candidates found early.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score, best_n
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            best_n = len(p)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init_uniform")

    n_main = n_parent if n_parent > 0 else 4096

    # === Phase 1: Diverse initialization (60s) ===
    print(f"\n[Phase 1] Diverse initialization (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = t_start + 60.0

    profiles_to_try = [
        ("cos1", 0.7), ("cos2", 0.5), ("tapered", 0.3),
        ("tapered2", 0.7), ("w_shape", None), ("random_structured", None),
        ("anti_palindrome", None), ("half_concentrated", None),
    ]

    for kind, _ in profiles_to_try:
        if time.monotonic() > phase1_deadline:
            break
        a_try = _make_profile(n_main, kind, rng)
        sc_try = _score(a_try)
        if sc_try < best_score:
            update_best(a_try, sc_try, f"profile_{kind}")

    # Quick Adam from parent
    if time.monotonic() < phase1_deadline - 20:
        a_adam, sc_adam = _adam_optimize(parent, phase1_deadline - 15, max_steps=5000, lr=0.015)
        update_best(a_adam, sc_adam, "adam_init")

    # === Phase 2: Main optimization from best (500s) ===
    print(f"\n[Phase 2] Main optimization (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = t_start + 550.0

    # Adam from best
    if best_seq is not None:
        a_adam2, sc_adam2 = _adam_optimize(best_seq, phase2_deadline - 200, max_steps=30000, lr=0.01)
        update_best(a_adam2, sc_adam2, "adam_main")

    # L-BFGS from best
    if best_seq is not None and time.monotonic() < phase2_deadline - 100:
        lbfgs_deadline = min(phase2_deadline - 20, time.monotonic() + 200)
        a_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, min_remaining=50)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_main")

    # Projected subgradient
    if best_seq is not None and time.monotonic() < phase2_deadline - 30:
        ps_deadline = min(phase2_deadline - 10, time.monotonic() + 100)
        a_ps, sc_ps = _projected_subgrad(best_seq, ps_deadline, max_steps=10000)
        update_best(a_ps, sc_ps, "proj_subgrad")

    # === Phase 3: LP refinement (200s) ===
    print(f"\n[Phase 3] LP refinement (t={time.monotonic() - t_start:.1f}s)")
    phase3_deadline = t_start + 750.0

    if best_seq is not None:
        a_lp, sc_lp = _lp_refine(best_seq, phase3_deadline, n_lp_calls=4)
        update_best(a_lp, sc_lp, "lp_refine")

    # L-BFGS after LP
    if best_seq is not None and time.monotonic() < phase3_deadline - 30:
        a_post = _lbfgs_refine(best_seq, phase3_deadline - 5, min_remaining=20)
        update_best(a_post, _score(a_post), "lbfgs_post_lp")

    # === Phase 4: Shrinkage and support reduction (150s) ===
    print(f"\n[Phase 4] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    phase4_deadline = t_start + 900.0

    if best_seq is not None:
        a_shr, sc_shr = _shrink_and_refine(best_seq, phase4_deadline, min_remaining=30)
        update_best(a_shr, sc_shr, "shrinkage")

        # Re-optimize after shrinkage
        if best_seq is not None and time.monotonic() < phase4_deadline - 30:
            a_shr2 = _lbfgs_refine(best_seq, phase4_deadline - 10, min_remaining=15)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # === Phase 5: Diverse restarts from best (200s) ===
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    phase5_deadline = t_start + 1100.0

    if best_seq is not None:
        restart_strategies = [
            ("perturb_small", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_medium", lambda: _perturb(best_seq, 0.08, rng)),
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=5)),
            ("fourier_big", lambda: _perturb_fourier(best_seq, rng, n_modes=8)),
        ]

        for name, gen in restart_strategies:
            if time.monotonic() > phase5_deadline - 40:
                break
            a_start = gen()
            sc_start = _score(a_start)
            # Quick Adam
            a_r, sc_r = _adam_optimize(a_start, min(phase5_deadline - 30, time.monotonic() + 60),
                                       max_steps=5000, lr=0.01)
            update_best(a_r, sc_r, f"restart_{name}")

    # === Phase 6: Different n exploration (150s) ===
    print(f"\n[Phase 6] Different n (t={time.monotonic() - t_start:.1f}s)")
    phase6_deadline = t_start + 1250.0

    n_alternatives = [2048, 8192]
    for n_alt in n_alternatives:
        if time.monotonic() > phase6_deadline - 40:
            break
        # Resample best to new length
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_alt)
            a_resamp = np.interp(x_new, x_old, best_seq)
            a_resamp = np.maximum(a_resamp, 0)
            a_resamp = a_resamp / a_resamp.sum()
        else:
            a_resamp = _make_profile(n_alt, "uniform")

        sc_resamp = _score(a_resamp)
        if sc_resamp < best_score:
            update_best(a_resamp, sc_resamp, f"resample_n{n_alt}")

        # Quick optimization
        if time.monotonic() < phase6_deadline - 20:
            a_opt = _lbfgs_refine(a_resamp, phase6_deadline - 10, min_remaining=10)
            update_best(a_opt, _score(a_opt), f"opt_n{n_alt}")

    # === Phase 7: Final intensive polish (300s) ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    phase7_deadline = deadline

    if best_seq is not None:
        # Adam with very small lr
        a_fin_adam, sc_fin_adam = _adam_optimize(best_seq, min(phase7_deadline - 120, time.monotonic() + 120),
                                                 max_steps=20000, lr=0.003)
        update_best(a_fin_adam, sc_fin_adam, "final_adam")

        # LP refinement again
        if time.monotonic() < phase7_deadline - 60:
            a_fin_lp, sc_fin_lp = _lp_refine(best_seq, min(phase7_deadline - 40, time.monotonic() + 80),
                                             n_lp_calls=3)
            update_best(a_fin_lp, sc_fin_lp, "final_lp")

        # L-BFGS with very small beta
        if time.monotonic() < phase7_deadline - 30:
            a_fin_lb = _lbfgs_refine(best_seq, min(phase7_deadline - 10, time.monotonic() + 80),
                                     min_remaining=5)
            update_best(a_fin_lb, _score(a_fin_lb), "final_lbfgs")

        # Subgradient polish
        if time.monotonic() < phase7_deadline - 5:
            a_fin_sg, sc_fin_sg = _subgrad_polish(best_seq, phase7_deadline - 2,
                                                  n_max_steps=5000, lr_init=0.005)
            update_best(a_fin_sg, sc_fin_sg, "final_sg")

        # One more shrinkage pass
        if time.monotonic() < phase7_deadline - 3:
            a_fin_shr, sc_fin_shr = _shrink_and_refine(best_seq, phase7_deadline - 1, min_remaining=0)
            update_best(a_fin_shr, sc_fin_shr, "final_shrink")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = final.sum()
        # Normalize to a reasonable scale
        final = final / s
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
