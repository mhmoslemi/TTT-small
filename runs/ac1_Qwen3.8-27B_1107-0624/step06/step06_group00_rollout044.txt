"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based smoothed gradient, Adam in log-space with backtracking,
coordinate descent polishing, multi-scale exploration (n=2048, 4096, 8192),
and adaptive temperature annealing with plateau phases.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_smoothed_gradient(a, c, T):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_weighted_active_gradient(a, c, top_k=5, decay=0.7):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    for i, idx in enumerate(k_indices):
        p[idx] = decay ** i
    p_sum = p.sum()
    if p_sum < 1e-300:
        return np.zeros(n)
    p = p / p_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _adam_z_space(a_init, deadline, T_start, T_end, lr_start, lr_end,
                  max_iter=500000, active_k=3, decay=0.6,
                  eval_interval=200, clip_norm=0.8, use_backtrack=True,
                  plateau_frac=0.3):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    best_patience_score = best_score
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        if frac < plateau_frac:
            T = T_start
        else:
            frac2 = (frac - plateau_frac) / (1.0 - plateau_frac)
            T = T_start * (T_end / T_start) ** frac2
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult
        a = _softmax_stable(z)
        c = _fft_convolve(a, a)
        if T < T_start * 0.01:
            g = _compute_weighted_active_gradient(a, c, active_k, decay)
        else:
            g = _compute_smoothed_gradient(a, c, T)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z_new = z - update
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        if use_backtrack and t % 50 == 0:
            a_cand = _softmax_stable(z_new)
            sc_cand = _score_norm(a_cand)
            if sc_cand > best_score * 1.01:
                z_new = z
            z = z_new
        else:
            z = z_new

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 4 and current_lr_mult > 0.05:
                current_lr_mult *= 0.6
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent_polish(a, deadline, max_rounds=5000):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _fft_convolve(best_a, best_a)
    top_k = 5
    k_indices = np.argsort(c)[::-1][:top_k]

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for j in range(n):
            a_cand = best_a.copy()
            old_val = a_cand[j]
            original_sum = a_cand.sum()
            for delta_frac in [0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 0.0]:
                new_val = old_val * delta_frac
                if new_val < 0:
                    continue
                a_cand[j] = new_val
                a_cand = a_cand / a_cand.sum()
                c_cand = _fft_convolve(a_cand, a_cand)
                max_cand = float(c_cand[top_k].max()) if len(c_cand) > max(k_indices) else float(c_cand.max())
                sc_cand = 2.0 * n * max_cand
                if sc_cand < best_score * (1.0 - 1e-10):
                    best_score = sc_cand
                    best_a = a_cand.copy()
                    improved = True
                    break
            if not improved:
                best_a[j] = old_val
                best_a = best_a / best_a.sum()
        if not improved:
            break

    return best_a, _score_norm(best_a)


def _fast_cd_polish(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    c = _fft_convolve(best_a, best_a)
    c_max = float(c.max())
    tol = c_max * 1.001
    active = np.where(c > tol)[0]

    round_num = 0
    while round_num < max_rounds:
        if time.monotonic() > deadline:
            break
        round_num += 1
        improved = False
        for k in active:
            if k < n:
                js = range(max(0, k - n + 1), min(k, n - 1) + 1)
            else:
                js = range(k - (n - 1), n)
            if len(list(js)) == 0:
                continue
            for j in js:
                for scale in [1.15, 1.3, 0.85, 0.7, 0.5]:
                    a_cand = best_a.copy()
                    a_cand[j] *= scale
                    s = a_cand.sum()
                    if s < 1e-15:
                        continue
                    a_cand /= s
                    c_cand = _fft_convolve(a_cand, a_cand)
                    sc = 2.0 * n * float(c_cand.max())
                    if sc < best_score * (1.0 - 1e-12):
                        best_score = sc
                        best_a = a_cand.copy()
                        improved = True
                        c = c_cand
                        c_max = float(c.max())
                        tol = c_max * 1.001
                        active = np.where(c > tol)[0]
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break

    return best_a, best_score


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _gradient_descent_simplex(a_init, deadline, max_iter=100000, lr=0.01):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    y = a.copy()
    t_k = 1.0
    step = lr
    eval_interval = 300

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(y, y)
        c_max = float(c.max())
        T = c_max * 0.005
        g = _compute_smoothed_gradient(y, c, T)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            step = min(step * 1.5, 2.0 / (g_norm + 1e-10))
        a_new = y - step * g
        a_new = _project_simplex(a_new)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = _project_simplex(y)
        t_k = t_new
        a = a_new
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score

                                     max_iter=200000, active_k=3,
                                     eval_interval=200, use_backtrack=True)
        full_sc_4k = _score(a_4k)
        print(f"Phase 4 Adam n=4096: score={full_sc_4k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_4k) and full_sc_4k < best_score:
            best_score = full_sc_4k
            best_a = a_4k.copy()

        # CD polish at n=4096
        if time.monotonic() < phase4_end - 30:
            a_4k_cd, sc_4k_cd = _fast_cd_polish(best_a, phase4_end - 5, max_rounds=80)
            full_sc_4k_cd = _score(a_4k_cd)
            print(f"Phase 4 CD n=4096: score={full_sc_4k_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_4k_cd) and full_sc_4k_cd < best_score:
                best_score = full_sc_4k_cd
                best_a = a_4k_cd.copy()

    # Phase 5: Try n=8192 (650s - 1000s)
    phase5_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase5_end:
        a_8k = _interp_to_length(best_a, 8192)
        c_8k = _fft_convolve(a_8k, a_8k)
        max_8k = float(c_8k.max())
        a_8k_opt, sc_8k = _adam_z_space(a_8k, phase5_end - 5,
                                         T_start=0.002 * max_8k, T_end=0.00002 * max_8k,
                                         lr_start=0.025, lr_end=0.002,
                                         max_iter=150000, active_k=3,
                                         eval_interval=200, use_backtrack=True)
        full_sc_8k = _score(a_8k_opt)
        print(f"Phase 5 n=8192: score={full_sc_8k:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_8k) and full_sc_8k < best_score:
            best_score = full_sc_8k
            best_a = a_8k_opt.copy()

        # Also try interpolating 8k result back to 4096
        if full_sc_8k < 2.0 and time.monotonic() < phase5_end - 60:
            a_back = _interp_to_length(a_8k_opt, 4096)
            c_back = _fft_convolve(a_back, a_back)
            max_back = float(c_back.max())
            a_back_opt, sc_back = _adam_z_space(a_back, phase5_end - 5,
                                                 T_start=0.001 * max_back, T_end=0.00001 * max_back,
                                                 lr_start=0.02, lr_end=0.0015,
                                                 max_iter=80000, active_k=3,
                                                 eval_interval=100, use_backtrack=True)
            full_sc_back = _score(a_back_opt)
            print(f"Phase 5 interp 8k->4k: score={full_sc_back:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_back) and full_sc_back < best_score:
                best_score = full_sc_back
                best_a = a_back_opt.copy()

    # Phase 6: Intensive polish on best (1000s - 1400s)
    phase6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase6_end:
        n_best = len(best_a)
        c_best = _fft_convolve(best_a, best_a)
        max_cb = float(c_best.max())
        # Very low temperature Adam
        mid6 = min(phase6_end, t_start + 1200.0)
        a_pol, sc_pol = _adam_z_space(best_a, mid6,
                                       T_start=0.0005 * max_cb, T_end=0.000005 * max_cb,
                                       lr_start=0.01, lr_end=0.0005,
                                       max_iter=300000, active_k=3,
                                       eval_interval=100, use_backtrack=True,
                                       plateau_frac=0.0)
        full_sc_pol = _score(a_pol)
        print(f"Phase 6 polish: score={full_sc_pol:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_pol) and full_sc_pol < best_score:
            best_score = full_sc_pol
            best_a = a_pol.copy()

        # CD polish
        if time.monotonic() < phase6_end - 20:
            a_cd2, sc_cd2 = _fast_cd_polish(best_a, phase6_end - 5, max_rounds=60)
            full_sc_cd2 = _score(a_cd2)
            print(f"Phase 6 CD2: score={full_sc_cd2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_cd2) and full_sc_cd2 < best_score:
                best_score = full_sc_cd2
                best_a = a_cd2.copy()

    # Phase 7: Perturbation restarts (1400s - 1650s)
    phase7_end = min(deadline, t_start + 1650.0)
    n_fin = len(best_a)
    if time.monotonic() < phase7_end:
        for attempt in range(4):
            if time.monotonic() > phase7_end - 60:
                break
            sigma = 0.02 + 0.015 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_fin)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            sub_deadline = min(phase7_end - 10, t_start + 1650.0)
            time_per = max(20.0, (phase7_end - time.monotonic()) / (4 - attempt))
            sub_dl = min(sub_deadline, time.monotonic() + time_per)
            a_pert_opt, sc_pert = _adam_z_space(a_pert, sub_dl,
                                                 T_start=0.001 * max_cp, T_end=0.00001 * max_cp,
                                                 lr_start=0.02, lr_end=0.001,
                                                 max_iter=100000, active_k=3,
                                                 eval_interval=100, use_backtrack=True)
            full_sc_pert = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc_pert:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_pert) and full_sc_pert < best_score:
                best_score = full_sc_pert
                best_a = a_pert_opt.copy()

    # Phase 8: Final ultra-fine polish (1650s - deadline)
    if time.monotonic() < deadline - 10:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_z_space(best_a, deadline - 8,
                                           T_start=0.0001 * max_cf, T_end=0.000002 * max_cf,
                                           lr_start=0.003, lr_end=0.0002,
                                           max_iter=200000, active_k=3,
                                           eval_interval=80, use_backtrack=True,
                                           plateau_frac=0.0)
        full_sc_final = _score(a_final)
        print(f"Phase 8 final: score={full_sc_final:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_final) and full_sc_final < best_score:
            best_score = full_sc_final
            best_a = a_final.copy()

        # Quick CD at the end
        if time.monotonic() < deadline - 3:
            a_cd_final, _ = _fast_cd_polish(best_a, deadline - 2, max_rounds=20)
            sc_cd_final = _score(a_cd_final)
            if sc_cd_final < best_score:
                best_score = sc_cd_final
                best_a = a_cd_final.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01
    sc_final_check = _score(final)
    print(f"FINAL: score={sc_final_check:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
