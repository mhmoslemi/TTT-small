    best_score = _score_norm(_softmax_compute(z_init))
    best_z = z_init.copy()

    try:
        result = minimize(obj_fn, z_init, jac=grad_fn, method='L-BFGS-B',
                          options={'maxiter': max(30, int(max_time * 20)),
                                   'gtol': 1e-14, 'maxls': 10})
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    return best_z, best_score


def _make_catalan_profile(n):
    """Generate C(2i,i)/4^i profile (flat-convolution basis)."""
    a = np.zeros(n)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-30:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_uniform(n):
    return np.ones(n) / n


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered(n, power=0.4):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_symmetric_catalan(n):
    """Symmetric version of Catalan profile."""
    cat = _make_catalan_profile(n)
    return (cat + cat[::-1]) / (cat.sum() + cat[::-1].sum())


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimization in log-space with correct gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = _compute_grad_correct(a, w)
        correction = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - correction)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)
        z = z - update
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _continuation_optimize(a_init, deadline, n_phases=20, use_correct=True):
    """Continuation method: progressively decrease beta."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    c = _full_convolve(a, a)
    max_c = float(c.max())
    beta_start = 0.1 * max_c
    beta_end = 0.00001 * max_c

    phase_times = np.linspace(0, 1, n_phases + 1)
    phase_deadlines = [time.monotonic() + (deadline - time.monotonic()) * (phase_times[i+1] - phase_times[i]) + (phase_times[i]) * (deadline - time.monotonic()) for i in range(n_phases)]
    # Simpler: divide time evenly
    total_time = deadline - time.monotonic()
    per_phase_time = total_time / n_phases

    for phase in range(n_phases):
        if time.monotonic() >= deadline - 2:
            break
        frac = phase / max(1, n_phases - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        phase_deadline = time.monotonic() + per_phase_time
        per_phase_time = (deadline - time.monotonic()) / max(1, n_phases - phase)

        a, _ = _nesterov_opt(a, beta, phase_deadline, use_correct_grad=use_correct)
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # Collect candidate starting points
    candidates = []

    if parent is not None:
        n_main = len(parent)
        candidates.append(('parent', parent.copy(), n_main))
        sc = _score(parent)
        print(f"Parent: n={n_main}, score={sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = parent.copy()
    else:
        n_main = 4096
        candidates.append(('uniform', _make_uniform(n_main), n_main))
        sc = _score(candidates[-1][1])
        print(f"No parent, uniform n={n_main}, score={sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = candidates[-1][1].copy()

    # Add diverse starting points
    n_test = 4096
    cat = _make_catalan_profile(n_test)
    candidates.append(('catalan', cat, n_test))
    sc = _score(cat)
    print(f"Catalan: n={n_test}, score={sc:.10f}")
    if sc < best_score:
        best_score = sc
        best_seq = cat.copy()

    sym_cat = _make_symmetric_catalan(n_test)
    candidates.append(('sym_catalan', sym_cat, n_test))
    sc = _score(sym_cat)
    print(f"Sym Catalan: n={n_test}, score={sc:.10f}")
    if sc < best_score:
        best_score = sc
        best_seq = sym_cat.copy()

    raised = _make_raised_cosine(n_test)
    candidates.append(('raised', raised, n_test))
    sc = _score(raised)
    print(f"Raised cosine: n={n_test}, score={sc:.10f}")
    if sc < best_score:
        best_score = sc
        best_seq = raised.copy()

    # Mix parent with catalan for diversity
    if parent is not None and len(parent) == n_test:
        mix = 0.5 * parent + 0.5 * cat
        mix = np.maximum(mix, 1e-12)
        mix = mix / mix.sum()
        candidates.append(('mix', mix, n_test))
        sc = _score(mix)
        print(f"Mix parent+catalan: score={sc:.10f}")
        if sc < best_score:
            best_score = sc
            best_seq = mix.copy()

    # Sort candidates by initial score
    scored_candidates = []
    for name, a, n in candidates:
        sc = _score(a)
        scored_candidates.append((sc, name, a, n))
    scored_candidates.sort(key=lambda x: x[0])
    print(f"\nCandidate scores: {[(name, f'{sc:.6f}') for sc, name, a, n in scored_candidates]}")

    # Allocate time to top candidates
    time_per_candidate = max(5.0, (deadline - time.monotonic() - 60.0) / min(4, len(scored_candidates)))

    for ci, (init_score, name, a_init, n) in enumerate(scored_candidates[:4]):
        if time.monotonic() > deadline - 60:
            break
        print(f"\nOptimizing candidate '{name}' (n={n}, init_score={init_score:.8f}), time_budget={time_per_candidate:.1f}s")
        c_deadline = min(deadline - 60, time.monotonic() + time_per_candidate)

        a_work = a_init.copy()
        a_work = a_work / a_work.sum()

        # Phase 1: Continuation with correct gradient
        sc_target = deadline - time.monotonic()
        if sc_target > 10:
            a_work, sc1 = _continuation_optimize(a_work, c_deadline, n_phases=15, use_correct=True)
            sc1 = _score(a_work)
            print(f"  After continuation (correct grad): score={sc1:.10f}")
            if sc1 < best_score:
                best_score = sc1
                best_seq = a_work.copy()

        # Phase 2: Fine Adam with correct gradient
        remaining = deadline - time.monotonic() - 30
        if remaining > 5:
            c_curr = _full_convolve(a_work, a_work)
            max_cc = float(c_curr.max())
            beta_fine = 0.0001 * max_cc
            a_work, _ = _adam_opt(a_work, time.monotonic() + remaining * 0.5,
                                  beta_fine, lr_init=0.005, max_iter=200000)
            sc2 = _score(a_work)
            print(f"  After Adam (correct grad): score={sc2:.10f}")
            if sc2 < best_score:
                best_score = sc2
                best_seq = a_work.copy()

        # Phase 3: L-BFGS polish
        if time.monotonic() < deadline - 15:
            z = np.log(np.maximum(a_work, 1e-15))
            z = z - z.mean()
            c_curr = _full_convolve(a_work, a_work)
            beta_lb = 0.00001 * float(c_curr.max())
            z, sc3 = _lbfgs_refine(z, deadline - 10, beta_lb)
            a_lb = _softmax_compute(z)
            sc3 = _score(a_lb)
            print(f"  After L-BFGS: score={sc3:.10f}")
            if sc3 < best_score:
                best_score = sc3
                best_seq = a_lb.copy()

    # Final polish on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\nFinal polish on best sequence (score={best_score:.10f})")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        remaining = deadline - time.monotonic() - 5

        # Ultra-fine Adam
        if remaining > 10:
            c_p = _full_convolve(a_pol, a_pol)
            max_cp = float(c_p.max())
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + remaining * 0.5,
                                 0.000005 * max_cp, lr_init=0.001, max_iter=100000)
            sc = _score(a_pol)
            print(f"  Final Adam: score={sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()

        # Final L-BFGS
        if time.monotonic() < deadline - 5:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            c_f = _full_convolve(best_seq, best_seq)
            beta_final = 0.000001 * float(c_f.max())
            z, sc = _lbfgs_refine(z, deadline - 2, beta_final)
            a_final = _softmax_compute(z)
            sc = _score(a_final)
            print(f"  Final L-BFGS: score={sc:.10f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_final.copy()

    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(4096)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(4096)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
