"""
Multi-phase optimizer for non-negative step function:
1. Adam log-space with aggressive beta continuation (n=4096)
2. Subgradient polishing with Polyak step size
3. Analytical coordinate descent
4. Upscale to n=8192 and re-optimize
5. Final ultra-fine polish

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_a(a, w, n):
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        iter_count = 0
        while iter_count < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            iter_count += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())

            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            w = np.exp(c_shifted)
            w_sum = float(w.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = w / w_sum

            grad_a = _compute_grad_a(a, w, n)
            W = float(np.dot(a, grad_a))

def _subgradient_polish(a_init, deadline, max_iter=200000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _convolve(a, a)
    f_lb = 2.0 * n / (2 * n - 1)  # Lower bound on score ~ 1
    mu = 1.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        cm = float(c.max())
        f_curr = 2.0 * n * cm

        # Active set: indices where c_k is within 0.5% of max
        threshold = cm * 0.995
        active = c >= threshold
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Subgradient: average of gradients over active set
        w = active.astype(np.float64) / n_active
        grad_a = _compute_grad_a(a, w, n)

        # Polyak step size
        gap = f_curr - f_lb
        grad_norm_sq = float(np.dot(grad_a, grad_a))
        if grad_norm_sq < 1e-30:
            break
        step = (0.9 * gap) / (grad_norm_sq + 1e-10)
        step = min(step, mu)
        mu = mu * 0.999

        a_new = a - step * grad_a
        a_new = _proj_simplex(a_new)

        sc_new = _score_norm(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
            c = _convolve(a, a)
        else:
            # Don't update a, but decay step
            mu *= 0.5

    return best_a, best_sc

        a_cd, sc_cd = _coord_descent(a_best, phase3_end - 10, max_rounds=500)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f}, t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            a_best = a_cd.copy()

    # === Phase 4: Try n=8192 ===
    print("Phase 4: n=8192 exploration")
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 60 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        # Optimize the big version
        if time.monotonic() < phase4_end - 30:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            betas_big = []
            beta = 0.003 * max_cb
            while beta > 1e-12 * max_cb:
                betas_big.append(beta)
                beta *= 0.5
            betas_big.append(1e-12 * max_cb)
            if len(betas_big) > 20:
                step = max(1, len(betas_big) // 20)
                betas_big = betas_big[::step]
                if betas_big[-1] != 1e-12 * max_cb:
                    betas_big.append(1e-12 * max_cb)

            per_t = phase4_end - time.monotonic() - 20
            if per_t > 30:
                a_big_opt, sc_bigo = _adam_logspace(
                    a_big, time.monotonic() + per_t,
                    betas_big, lr_init=0.012, max_iter_per_phase=200000
                )
                sc_big_opt_full = _score(a_big_opt)
                print(f"  Optimized n=8192: score={sc_big_opt_full:.10f}, t={time.monotonic()-t_start:.1f}s")
                if sc_big_opt_full < best_score:
                    best_score = sc_big_opt_full
                    best_seq = a_big_opt.copy()
                    a_best = a_big_opt.copy()

    # === Phase 5: Final ultra-fine polish ===
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cpol = float(c_pol.max())

        # Very fine Adam
        betas_final = []
        beta = 1e-5 * max_cpol
        while beta > 1e-15 * max_c0:
            betas_final.append(beta)
            beta *= 0.5
        betas_final.append(1e-15 * max_c0)
        if len(betas_final) > 12:
            betas_final = betas_final[::max(1, len(betas_final) // 12)]

        per_t = deadline - time.monotonic() - 8
        if per_t > 20:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t * 0.6,
                betas_final, lr_init=0.003, max_iter_per_phase=50000
            )
            sc_pol_full = _score(a_pol)
            print(f"  Final Adam: score={sc_pol_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

            # Final coord descent
            if time.monotonic() < deadline - 8:
                a_cd2, sc_cd2 = _coord_descent(
                    best_seq, deadline - 5, max_rounds=200
                )
                sc_cd2_full = _score(a_cd2)
                print(f"  Final coord: score={sc_cd2_full:.10f}, t={time.monotonic()-t_start:.1f}s")
                if sc_cd2_full < best_score:
                    best_score = sc_cd2_full
                    best_seq = a_cd2.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            print("WARNING: sum too small, using uniform fallback")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
