def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)

    elif kind == "cosine":

    elif kind == "bimodal":

    elif kind == "trimodal":

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 60, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _lbfgs_schedule(a_start, deadline, min_time_remaining=200.0):

    betas = max_c * 0.12
    beta_list = [betas]
    for i in range(29):
        betas *= ratio
        beta_list.append(betas)

    current_sc = _score(current_a)

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:

            current_sc = sc
            if i % 5 == 0:

def _subgrad_soft(a_init, deadline, beta_sg=None, max_steps=20000, lr_init=0.06):

    n = len(a_init)

    if beta_sg is None:

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0
    momentum = 0.9
    z_prev = z.copy()

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Soft active set: exponential weights
        shifted = (c - c_max) / beta_sg
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        # Nesterov: interpolate previous step
        z_nest = z + momentum * (z - z_prev)
        a_nest = _softmax(z_nest)
        c_nest = _full_convolve(a_nest, a_nest)
        c_max_nest = c_nest.max()
        shifted_n = (c_nest - c_max_nest) / beta_sg
        e_n = np.exp(shifted_n)
        s_n = e_n.sum()
        if s_n < 1e-300 or not np.isfinite(s_n):
            w_n = np.zeros_like(c_nest)
            w_n[int(np.argmax(c_nest))] = 1.0
        else:
            w_n = e_n / s_n
        a_rev_n = a_nest[::-1]
        wc_n = _full_convolve(w_n, a_rev_n)
        h_n = wc_n[n - 1:2 * n - 1]
        W_n = float(np.dot(w_n, c_nest))
        grad_n = 4.0 * n * a_nest * (h_n - W_n)

        step = lr / (1.0 + 0.0002 * i)
        z_prev = z.copy()
        z = z - step * grad_n
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Periodic evaluation
        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:

                best_a = a_check.copy()
                lr *= 1.02
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 4:
                    lr *= 0.75
                    momentum = max(0.5, momentum * 0.95)
                    if lr < 1e-7:
                        break
                else:
                    lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc

def _lp_primal_iteration(f, n, deadline, min_remaining=60.0):
    """Solve the primal LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    if time.monotonic() + min_remaining > deadline:
        return None

    c_f = _full_convolve(f, f)
    M = float(c_f.max())
    if M < 1e-30:
        return None

    # A is (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<n else 0
    # This is a convolution matrix - build it as sparse
    # For n up to 4096, this is 8191 x 4096 = ~33M entries (dense but structured)
    # Use the fact that it's a convolution matrix

def _perturb_fourier(a, rng, n_modes=3, amp_scale=0.008):

def _perturb_peak_targeted(a, rng, n_targets=3, strength=0.02):
    """Perturb specifically to reduce peaks in the convolution."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    # Find top peak positions
    peak_indices = np.argsort(c)[-n_targets:]

    z = np.log(np.maximum(a, 1e-15))
    for pk in peak_indices:
        # The peak at position pk is contributed by pairs (i, pk-i)
        # Reduce elements that contribute most to this peak
        # Move probability mass away from elements near the "center" of the peak
        for offset in range(-2, 3):
            i = int(pk // 2 + offset)
            j = int(pk - i)
            if 0 <= i < n and 0 <= j < n:
                z[i] -= strength * rng.uniform(0.5, 1.5)
                # Compensate at edges
                edge = int(rng.integers(0, n))
                z[edge] += strength * rng.uniform(0.5, 1.5) * 0.5

    z = z - z.mean()
    return _softmax(z)


def _local_search(a, deadline, max_rounds=300):
    """Gradient-informed local search with adaptive step."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.003

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Compute gradient
        c = _full_convolve(best_a, best_a)
        c_max = c.max()
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = best_a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * best_a * (h - W)
        grad = grad - grad.mean()  # Project to tangent space of simplex

        # Take a step
        a_try = best_a - step * grad
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc - 1e-14:
            best_a = a_try.copy()
            best_sc = sc_try
            step = min(step * 1.01, 0.02)
        else:
            step *= 0.8
            if step < 1e-8:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

    # === Phase 1: Quick diverse initializations ===
    print(f"\n=== Phase 1: Diverse initializations (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 1500:
        n_cur = len(best_seq)
        profiles = ["bimodal", "edge_boost", "trimodal", "cosine", "tapered"]
        for prof in profiles:
            if time.monotonic() > deadline - 1400:
                break
            a_p = _make_profile(n_cur, prof)
            # Quick L-BFGS with moderate smoothing
            c_p = _full_convolve(a_p, a_p)
            beta = float(c_p.max()) * 0.05
            a_quick = _lbfgs_run(a_p, beta, n_cur, max_iter=1000)
            a_quick2 = _lbfgs_run(a_quick, beta * 0.1, n_cur, max_iter=1000)
            try_update(a_quick2, f"init_{prof}")

    # === Phase 2: Main L-BFGS with fine schedule ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p2 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=600.0)
    try_update(a_p2, "main_lbfgs")

    # === Phase 3: Soft subgradient with Nesterov ===
    print(f"\n=== Phase 3: Soft subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 400:
        a_sg, _ = _subgrad_soft(best_seq, deadline - 400, beta_sg=None,
                                max_steps=15000, lr_init=0.05)
        try_update(a_sg, "subgrad_soft")

    # === Phase 4: LP-based iteration ===
    print(f"\n=== Phase 4: LP iteration (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 250:
        n_lp = len(best_seq)
        if n_lp <= 4096:
            g_lp = _lp_primal_iteration(best_seq, n_lp, deadline, min_remaining=100)
            if g_lp is not None:
                # Blend LP solution with current best
                for alpha in [1.0, 0.8, 0.6, 0.4]:
                    a_blend = (1 - alpha) * best_seq + alpha * g_lp
                    a_blend = np.maximum(a_blend, 0)
                    a_blend = a_blend / a_blend.sum()
                    try_update(a_blend, f"lp_blend_{alpha}")
                # Polish the best LP blend
                if best_score_val < np.inf and time.monotonic() < deadline - 80:
                    a_lp_pol = _lbfgs_schedule(best_seq, deadline, min_time_remaining=30)
                    try_update(a_lp_pol, "lp_polish")

    # === Phase 5: Targeted peak reduction ===
    print(f"\n=== Phase 5: Peak reduction (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        for attempt in range(3):
            if time.monotonic() > deadline - 150:
                break
            a_pt = _perturb_peak_targeted(best_seq, rng, n_targets=5,
                                          strength=0.015 * (attempt + 1))
            c_pt = _full_convolve(a_pt, a_pt)
            beta_pt = float(c_pt.max()) * 0.01
            a_pt2 = _lbfgs_run(a_pt, beta_pt, len(a_pt), max_iter=2000)
            a_pt3 = _subgrad_soft(a_pt2, deadline - 100, beta_sg=float(c_pt.max()) * 0.003,
                                  max_steps=3000, lr_init=0.03)
            try_update(a_pt3, f"peak_red_{attempt}")

    # === Phase 6: Local search ===
    print(f"\n=== Phase 6: Local search (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_ls, _ = _local_search(best_seq, deadline - 100, max_rounds=400)
        try_update(a_ls, "local_search")

    # === Phase 7: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 80:
        restart_configs = [
            ("gauss_003", "gauss", 0.03, 0),
            ("gauss_008", "gauss", 0.08, 0),
            ("fourier_3", "fourier", 0, 3),
            ("fourier_5", "fourier", 0, 5),
            ("fourier_8", "fourier", 0, 8),
            ("peak_targ", "peak", 0.02, 0),
        ]

        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 60:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, modes)
            else:
                p_start = _perturb_peak_targeted(best_seq, rng, n_targets=3,
                                                 strength=sigma)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f} t={time.monotonic()-t_start:.0f}s")

            # Quick L-BFGS
            c_r = _full_convolve(p_start, p_start)
            max_c_r = float(c_r.max())
            for beta_frac in [0.05, 0.01, 0.002]:
                if time.monotonic() > deadline - 50:
                    break
                beta_r = max_c_r * beta_frac
                max_iter = min(3000, max(500, int((deadline - 50 - time.monotonic()) / 0.005)))
                a_r = _lbfgs_run(p_start, beta_r, len(p_start), max_iter)
                try_update(a_r, f"restart_{name}_{beta_frac}")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 40:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            c_in = _full_convolve(a_interp, a_interp)
            beta_in = float(c_in.max()) * 0.03
            max_iter = min(5000, max(500, int((deadline - 40 - time.monotonic()) / 0.008)))
            a_try = _lbfgs_run(a_interp, beta_in, target_n, max_iter)
            a_try2 = _lbfgs_run(a_try, beta_in * 0.1, target_n, max_iter // 2)
            try_update(a_try2, f"n_{target_n}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())

            # Very small beta L-BFGS
            beta = mb * (1e-5 if rnd < 2 else 5e-6)
            max_iter = min(5000, max(500, int((deadline - 5 - time.monotonic()) / 0.004)))
            if max_iter > 100:
                a_f = _lbfgs_run(best_seq, beta, n, max_iter)
                try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_soft(best_seq, deadline - 5,
                                       beta_sg=mb * 0.002, max_steps=5000, lr_init=0.02)
                try_update(a_s, f"final_sg_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
