"""
LP-based active set optimizer with L-BFGS polish for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Uses actual LP solving (HiGHS) for the active set subproblem instead of
gradient approximation, giving exact directions. Combined with adaptive active set refinement,
Fourier-domain diverse restarts, and multi-scale post-processing.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_convolution_matrix_sparse(a):
    """Build sparse matrix A where A @ b = convolve(a, b)."""
    n = len(a)
    out_len = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(out_len):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            row_vals = a[k - indices]
            rows.extend([k] * len(indices))
            cols.extend(indices.tolist())
            vals.extend(row_vals.tolist())
    A = sp.csr_matrix((vals, (rows, cols)), shape=(out_len, n))
    return A

    Solve LP: maximize sum(b) subject to (a*b)[k] <= M for near-active k, b >= 0.
    Returns normalized b or None if LP fails.

    n = len(a)
    c = _full_convolve(a, a)
    # Identify near-active constraints
    threshold = M - tol_frac * M
    active = c >= threshold
    n_active = int(active.sum())
    if n_active == 0:
        idx = int(np.argmax(c))
        active[idx] = True
        n_active = 1
    # Cap number of constraints
    if n_active > max_constraints:
        # Keep the top max_constraints
        c_active_vals = c[active]
        top_k = max_constraints
        global_thresh = np.partition(c_active_vals, -top_k)[-top_k]
        active = (c >= global_thresh)
    
    active_indices = np.where(active)[0]
    A = _make_convolution_matrix_sparse(a)
    A_sub = A[active_indices]
    b_vec = np.ones(len(active_indices)) * M
    
    # LP: max sum(b) s.t. A_sub @ b <= b_vec, b >= 0
    # In linprog form: minimize -sum(b) s.t. A_sub @ b <= b_vec, -b <= 0
    c_obj = -np.ones(n)
    try:
        res = linprog(c_obj, A_ub=A_sub.tocoo(), b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs')
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A_sub.toarray(), b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            return None
    if res.status != 0 or res.x is None:
        return None
    b_sol = res.x
    s = float(b_sol.sum())
    if s < 1e-10:
        return None
    return b_sol / s


def _run_lp_optimization(a_init, deadline, max_iter=50, tol_init=0.05, max_constraints_init=500):
    """Iterative LP-based active set optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    tol = tol_init
    max_con = max_constraints_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        
        # Solve LP
        b = _solve_lp_active(a, M, tol, max_con)
        if b is None:
            # Try with relaxed tolerance
            b = _solve_lp_active(a, M, tol * 3, max_con * 2)
            if b is None:
                break
        
        # Check improvement
        sc_b = _score(b)
        if sc_b < best_sc - 1e-12:
            a = b
            best_a = a.copy()
            best_sc = sc_b
        else:
            # Try blending
            for t_blend in [0.5, 0.3, 0.1]:
                a_try = (1 - t_blend) * a + t_blend * b
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-12:
                    a = a_try
                    best_a = a.copy()
                    best_sc = sc_try
                    break
            else:
                # No improvement, tighten tolerance
                tol *= 0.7
                max_con = min(max_con + 200, 2 * n)
                if tol < 0.001:
                    break
        
        # Tighten tolerance gradually
        tol *= 0.8
        max_con = min(max_con + 100, 2 * n - 1)
    
    return best_a


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation on log-sum-exp smoothed objective."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,
                                   0.003, 0.001, 0.0005, 0.0002])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        
        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 10))))
        
        z = z.copy()
        try:
            result = minimize(
                _smoobj, z, args=(beta, n),
                jac=_smograd, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.025):
    """Subgradient descent with momentum on active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    momentum = np.zeros_like(z)
    mu = 0.9
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        momentum = mu * momentum - step * grad
        z = z + momentum
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    momentum = np.zeros_like(z)
                    patience = 0
                    lr = lr_init * 0.4
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage(a, deadline):
    """Zero out smallest elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)
    
    fractions = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.16, 0.20, 0.25, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    return best_a, best_sc


def _perturb_fourier(a, rng, n_modes=5, amp_range=(0.01, 0.08)):
    """Perturb in frequency domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=5, block_frac=0.05):
    """Swap random blocks."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, int(n * block_frac))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _perturb_random(a, rng, sigma=0.04):
    """Additive Gaussian noise in log domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bump":
        # Emphasize center, small edges (inspired by n=4 optimal [1/6, 1/3, 1/3, 1/6])
        t = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * np.sin(np.pi * t) ** 2
        return a / a.sum()
    elif kind == "wavy":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t) + 0.1 * np.cos(8 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _resample(a, n_new):
    """Resample sequence to new length using linear interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return np.ones(n_new) / n_new
    return a_new / s


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP active set + L-BFGS + diverse restarts + shrinkage.
    Phase 1 (55%): LP-based active set optimization (main innovation).
    Phase 2 (20%): L-BFGS polish with adaptive beta.
    Phase 3 (15%): Diverse restarts with Fourier/block perturbations.
    Phase 4 (10%): Shrinkage, multi-n, final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={elapsed:.1f}s")
    
    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass
    
    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "bump")
        print(f"No parent, using bump profile n={n_parent}")
        update_best(parent, _score(parent), "init")
    
    # === Phase 1: LP-based active set optimization ===
    print(f"\n[Phase 1] LP active set optimization (t={time.monotonic() - t_start:.1f}s)")
    lp_deadline = t_start + 0.55 * (float(budget_s) - 15)
    
    a_lp = _run_lp_optimization(parent, lp_deadline, max_iter=40, tol_init=0.08, max_constraints_init=300)
    sc_lp = _score(a_lp)
    update_best(a_lp, sc_lp, "lp_main")
    
    # Second LP pass with tighter initial tolerance
    if time.monotonic() < lp_deadline - 10 and best_seq is not None:
        a_lp2 = _run_lp_optimization(best_seq, min(lp_deadline, time.monotonic() + 120),
                                     max_iter=25, tol_init=0.03, max_constraints_init=200)
        update_best(a_lp2, _score(a_lp2), "lp_tight")
    
    # Subgradient polish after LP
    if best_seq is not None and time.monotonic() < lp_deadline - 5:
        a_sg, sc_sg = _subgrad_polish(best_seq, min(lp_deadline, time.monotonic() + 90),
                                      n_max_steps=3000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_after_lp")
    
    # === Phase 2: L-BFGS polish ===
    print(f"\n[Phase 2] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    lbfgs_deadline = t_start + 0.75 * (float(budget_s) - 15)
    
    if best_seq is not None and time.monotonic() < lbfgs_deadline - 30:
        a_lbfgs = _run_lbfgs(best_seq, min(lbfgs_deadline, time.monotonic() + 300),
                             min_time_remaining=60)
        update_best(a_lbfgs, _score(a_lbfgs), "lbfgs_main")
        
        # Extra subgradient
        if time.monotonic() < lbfgs_deadline - 10 and best_seq is not None:
            a_sg2, sc_sg2 = _subgrad_polish(best_seq, min(lbfgs_deadline, time.monotonic() + 60),
                                            n_max_steps=2000, lr_init=0.015)
            update_best(a_sg2, sc_sg2, "sg_polish_2")
    
    # === Phase 3: Diverse restarts ===
    print(f"\n[Phase 3] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    restart_deadline = t_start + 0.90 * (float(budget_s) - 15)
    
    if best_seq is not None:
        restarts_config = [
            ("fourier", lambda: _perturb_fourier(best_seq, rng, n_modes=6, amp_range=(0.02, 0.10))),
            ("block", lambda: _perturb_block(best_seq, rng, n_blocks=6, block_frac=0.04)),
            ("random", lambda: _perturb_random(best_seq, rng, sigma=0.05)),
            ("fourier_small", lambda: _perturb_fourier(best_seq, rng, n_modes=3, amp_range=(0.01, 0.04))),
        ]
        
        for i, (name, gen_fn) in enumerate(restarts_config):
            if time.monotonic() > restart_deadline - 20:
                break
            # Can't use lambda in the list due to rules, so use direct calls
            pass
        
        # Direct calls to avoid lambda/closure issues
        p1 = _perturb_fourier(best_seq, rng, n_modes=6, amp_range=(0.02, 0.10))
        a_r1 = _run_lp_optimization(p1, min(restart_deadline, time.monotonic() + 120),
                                    max_iter=20, tol_init=0.05, max_constraints_init=200)
        update_best(a_r1, _score(a_r1), "restart_fourier")
        
        if time.monotonic() < restart_deadline - 30:
            p2 = _perturb_block(best_seq, rng, n_blocks=6, block_frac=0.04)
            a_r2 = _run_lbfgs(p2, min(restart_deadline, time.monotonic() + 100), min_time_remaining=20)
            update_best(a_r2, _score(a_r2), "restart_block_lbfgs")
        
        if time.monotonic() < restart_deadline - 30:
            p3 = _perturb_random(best_seq, rng, sigma=0.06)
            a_r3 = _run_lp_optimization(p3, min(restart_deadline, time.monotonic() + 80),
                                        max_iter=15, tol_init=0.05, max_constraints_init=200)
            update_best(a_r3, _score(a_r3), "restart_random")
        
        if time.monotonic() < restart_deadline - 20:
            p4 = _perturb_fourier(best_seq, rng, n_modes=3, amp_range=(0.01, 0.04))
            a_r4, sc_r4 = _subgrad_polish(p4, min(restart_deadline - 5, time.monotonic() + 60),
                                          n_max_steps=2000, lr_init=0.02)
            update_best(a_r4, sc_r4, "restart_fsmall_sg")
    
    # === Phase 4: Shrinkage, multi-n, final ===
    print(f"\n[Phase 4] Shrinkage & multi-n (t={time.monotonic() - t_start:.1f}s)")
    final_deadline = deadline
    
    # Shrinkage
    if best_seq is not None and time.monotonic() < final_deadline - 60:
        a_shr, sc_shr = _shrinkage(best_seq, min(final_deadline - 30, time.monotonic() + 60))
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score and time.monotonic() < final_deadline - 30:
            a_shr2 = _run_lp_optimization(a_shr, min(final_deadline - 10, time.monotonic() + 50),
                                          max_iter=15, tol_init=0.03, max_constraints_init=150)
            update_best(a_shr2, _score(a_shr2), "shrink_lp")
    
    # Multi-n exploration
    if time.monotonic() < final_deadline - 40:
        n_targets = [3500, 4000, 4500]
        for n_try in n_targets:
            if time.monotonic() > final_deadline - 20:
                break
            if best_seq is not None:
                a_resample = _resample(best_seq, n_try)
            else:
                a_resample = _make_profile(n_try, "bump")
            a_opt = _run_lp_optimization(a_resample, min(final_deadline - 10, time.monotonic() + 50),
                                         max_iter=10, tol_init=0.05, max_constraints_init=150)
            update_best(a_opt, _score(a_opt), f"n_{n_try}")
    
    # Final polish
    if best_seq is not None and time.monotonic() < final_deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, min(final_deadline - 2, time.monotonic() + 30),
                                        n_max_steps=1000, lr_init=0.01)
        update_best(a_fin, sc_fin, "final_sg")
        
        # Very fine L-BFGS
        if time.monotonic() < final_deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.0005
            remaining = final_deadline - 3 - time.monotonic()
            if remaining > 5:
                max_iter = min(3000, max(100, int(remaining / 0.03)))
                try:
                    result = minimize(
                        _smoobj, z, args=(beta_final, n),
                        jac=_smograd, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 50}
                    )
                    a_fin2 = _softmax(result.x)
                    update_best(a_fin2, _score(a_fin2), "final_lbfgs")
                except Exception:
                    pass
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
