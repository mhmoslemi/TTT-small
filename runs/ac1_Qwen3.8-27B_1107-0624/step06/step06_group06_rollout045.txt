def build_conv_matrix(f, n):
    """Build the (2n-1) x n matrix A such that A @ g = conv(f, g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2*n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

LP-based alternating optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find the optimal "direction" at each step,
combined with subgradient polish for fine-tuning.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

                cols.append(j)
    return np.array(rows), np.array(cols), len(rows)


def solve_lp_step(f, n, precomputed_structure):
    """
    Solve: minimize t
           s.t. conv(f, g)_k <= t for all k
                sum(g) = 1
                g >= 0
    """
    rows_idx, cols_idx, nnz = precomputed_structure
    
    # Build A matrix for convolution constraint: A @ g = conv(f, g)
    # A[k, j] = f[k-j] for valid indices
    vals = np.zeros(nnz)
    for idx in range(nnz):
        k = rows_idx[idx]
        j = cols_idx[idx]
        i = k - j
        vals[idx] = f[i]
    
    A_conv = csr_matrix((vals, (rows_idx, cols_idx)), shape=(2*n-1, n))
    
    # Full inequality constraint: [A_conv | -1] @ [g; t] <= 0
    # We need to add the -t column
    from scipy.sparse import hstack, csr_matrix as cm
    neg_one = cm((np.ones(2*n-1), np.arange(2*n-1)), shape=(2*n-1, 1))
    A_ineq = hstack([A_conv, -neg_one])
    b_ineq = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    # Bounds: g >= 0, t unrestricted (but in practice t >= 0)
    bounds = [(0, None)] * (n+1)
    
    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    return None, None

"""
LP-based alternating optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Core idea: at each step, solve an LP that finds the probability vector g
minimizing the peak of conv(f,g), then update f toward g. Combined with
subgradient polish and log-sum-exp smoothed L-BFGS for fine-tuning.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _precompute_structure(n):
    """Precompute row/col indices for the convolution matrix A where A@g = conv(f,g)."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            i = k - j
            if 0 <= i < n:
                rows_list.append(k)
                cols_list.append(j)
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    nnz = len(rows)
    return rows, cols, nnz


def _fill_conv_values(f, n, rows, cols):
    """Fill values: A[k,j] = f[k-j] for precomputed structure."""
    i_vals = (rows - cols).astype(np.int32)
    return f[i_vals]


def _solve_lp(f, n, rows, cols, nnz):
    """
    Solve: minimize t
           s.t. conv(f,g)_k <= t, k=0..2n-2
                sum(g) = 1
                g >= 0, t >= 0
    Returns (g, t_opt) or (None, None) on failure.
    """
    vals = _fill_conv_values(f, n, rows, cols)
    
    # Build inequality constraint matrix: [A_conv | -1] @ [g; t] <= 0
    all_rows = np.concatenate([rows, np.arange(2 * n - 1, dtype=np.int32)])
    all_cols = np.concatenate([cols, np.full(2 * n - 1, n, dtype=np.int32)])
    all_vals = np.concatenate([vals, -np.ones(2 * n - 1)])
    A_ineq = csr_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n + 1))
    b_ineq = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    bounds = [(0, None)] * (n + 1)
    
    try:
        result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.success:
            return result.x[:n].copy(), float(result.x[-1])
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 30 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.05, lr_init)
            else:
                patience += 1
                lr *= 0.85
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_smooth(a_init, deadline, min_time_remaining=30.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 10:
        gap = c_sorted[0] - c_sorted[10]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break
        est_time = 0.004 * max(1, n / 2048.0)
        max_iter = min(3000, max(30, int(remaining / (est_time * 8))))

        try:
            result = minimize(
                lambda zz: _smoobj(zz, beta, n),
                z,
                jac=lambda zz: _smograd(zz, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline):
    """Try zeroing small elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc

    if best_seq is not None and time.monotonic() < deadline - 40:
        f2 = best_seq.copy()
        for it2 in range(30):
            if time.monotonic() > deadline - 30:
                break
            g2, t_opt2 = _solve_lp(f2, n, rows, cols, nnz)
            if g2 is None:
                break
            c2 = _full_convolve(f2, f2)
            M2 = float(c2.max())
            if t_opt2 >= M2 * 0.999:
                break
            f_try = 0.5 * f2 + 0.5 * g2
            f_try = np.maximum(f_try, 0)
            f_try = f_try / f_try.sum()
            sc_try = _score(f_try)
            if sc_try < best_score:
                f2 = f_try
                update_best(f2, sc_try, f"lp2_iter{it2}")
            else:
                break

    # --- Phase 6: Final fine polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fin, sc_fin = _subgrad_polish(best_seq, min(deadline - 5, time.monotonic() + 60),
                                         n_max_steps=3000, lr_init=0.008)
        update_best(a_fin, sc_fin, "final_sg")
        
        if best_seq is not None and time.monotonic() < deadline - 5:
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            n_fin = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 0.0001
            try:
                result = minimize(
                    lambda zz: _smoobj(zz, beta_fin, n_fin),
                    z,
                    jac=lambda zz: _smograd(zz, beta_fin, n_fin),
                    method='L-BFGS-B',
                    options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_flag = _softmax(result.x)
                update_best(a_flag, _score(a_flag), "final_lbfgs")
            except Exception:
                pass

    # --- Return ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
