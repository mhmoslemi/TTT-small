Let me also think about the spectral shaping more carefully.

The power spectrum is P(ω) = |A(ω)|^2 where A(ω) = sum_i a[i] * exp(-iωi).
The autocorrelation is c[k] = (1/2π) ∫ P(ω) * exp(iωk) dω.

If P(ω) is flat, c[k] = 0 for k ≠ 0 and c[0] = P/2π.
We want to minimize max_k c[k], which is equivalent to making P(ω) as flat as possible (but we can't make it perfectly flat with non-negative coefficients).

The "ideal" operation would be:
- Compute P(ω)
- Set P_new(ω) = 1/sqrt(P(ω)) (or P(ω)^(-α) for some α)
- Set A_new(ω) = sqrt(P_new(ω)) * exp(i*arg(A(ω)))
- a_new = IFFT(A_new)
- Project to non-negative

But we need to be careful: a_new might have negative values, and projecting to non-negative changes the spectrum.

A better approach: use a soft projection or iterative reweighting.

Actually, here's a cleaner spectral approach:

**Iterative spectral shaping:**
1. Start with a (sum=1, a >= 0)
2. Compute A = FFT(a, N) for N >= 2n-1
3. Compute P = |A|^2
4. For flattening: P_target = P^-α + ε (for small α like 0.1-0.5)
5. Scale: P_target = P_target * mean(P) / mean(P_target) (to preserve total power)
6. A_new = sqrt(P_target) * exp(i*angle(A))
7. a_new = real(IFFT(A_new, N))[:n]
8. Blend: a = (1-t)*a + t*max(a_new, 0)
9. Normalize: a = a / sum(a)
10. Repeat

This should work! The key parameter is α (how aggressively to flatten) and t (step size).

"""
Multi-strategy optimizer for non-negative step functions minimizing
2*n*max(autocorr)/(sum)^2. Combines FISTA, spectral shaping, and
active-set gradient methods.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_max_grad(a, beta):
    """Compute smoothed max of autocorrelation and its gradient.
    Assumes a is normalized (sum=1).
    Returns (smoothed_max_value, gradient).
    """
    n = len(a)
    c = _convolve(a, a)
    c_max = float(c.max())
    exponents = (c - c_max) / beta
    np.clip(exponents, -400, 400, out=exponents)
    w = np.exp(exponents)
    s = float(w.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(c.argmax())] = 1.0
    else:
        w = w / s
    
    # Gradient: grad[j] = 2 * sum_k w[k]*a[k-j]
    N = _next_pow2(2 * n + 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    xc = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * xc[:n]
    
    smoothed_val = float(np.dot(w, c))
    return smoothed_val, grad, c, w


def _fista(a_init, deadline, beta_schedule, alpha_schedule=None):
    """FISTA optimization with given beta schedule.
    a_init: normalized (sum=1) sequence
    Returns best sequence found.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_val = _score_norm(a_curr)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        L = 4.0 / beta  # Lipschitz constant estimate
        step = 1.0 / L
        a_curr = a_curr.copy()
        a_prev = a_curr.copy()
        t_val = 1.0
        
        max_iter = 200000
        for it in range(max_iter):
            if time.monotonic() > deadline:
                break
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            _, grad, _, _ = _smoothed_max_grad(y, beta)
            grad = np.maximum(grad, -1000)  # safety clip
            a_next = _proj_simplex(y - step * grad)
            
            if it % 500 == 0:
                val = _score_norm(a_next)
                if val < best_val:
                    best_val = val
                    best_a = a_next.copy()
            
            a_prev = a_curr
            a_curr = a_next
            t_val = t_next
    
    val = _score_norm(a_curr)
    if val < best_val:
        best_val = val
        best_a = a_curr.copy()
    return best_a


def _spectral_shaping(a, n_steps=1000, alpha=0.3, step_size=0.1, deadline=None):
    """Iterative spectral shaping to flatten power spectrum."""
    n = len(a)
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()
    N = _next_pow2(2 * n + 1)
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    for it in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        A = np.fft.rfft(a_curr, N)
        P = np.abs(A)**2
        P_mean = float(P.mean())
        
        # Flatten: P_target = P^(-alpha)
        P_target = np.power(np.maximum(P, 1e-30), -alpha)
        P_target = P_target / P_target.mean() * P_mean
        
        # New spectrum with same phase
        A_new = np.sqrt(P_target) * np.exp(1j * np.angle(A))
        # For rfft, angle is real, magnitude is real
        A_new = np.sqrt(P_target) * (A / np.maximum(np.abs(A), 1e-30))
        
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        # Blend
        a_curr = (1.0 - step_size) * a_curr + step_size * a_new
        
        if it % 100 == 0:
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
    
    sc = _score_norm(a_curr)
    if sc < best_score:
        best_score = sc
        best_a = a_curr.copy()
    return best_a


def _active_set_grad_descent(a, deadline, top_k=200, lr=0.001, max_iter=100000):
    """Gradient descent focusing on top-k autocorrelation values."""
    n = len(a)
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_val = _score_norm(a_curr)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a_curr, a_curr)
        c_max = float(c.max())
        
        # Focus on top-k values
        # Use a "top-k softmax" weighting
        # Sort and take top k
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:top_k]
        top_vals = c[top_idx]
        
        # Softmax over top-k
        t_max = top_vals.max()
        exponents = (top_vals - t_max) / (0.01 * c_max + 1e-15)
        np.clip(exponents, -200, 200, out=exponents)
        w_top = np.exp(exponents)
        w_top = w_top / w_top.sum()
        
        # Full w vector
        w = np.zeros(len(c))
        w[top_idx] = w_top
        
        # Gradient
        N = _next_pow2(2 * n + 1)
        W = np.fft.rfft(w, N)
        A = np.fft.rfft(a_curr, N)
        xc = np.fft.irfft(W * np.conj(A), N)
        grad = 2.0 * xc[:n]
        
        # Gradient step with simplex projection
        a_curr = _proj_simplex(a_curr - lr * grad)
        
        if it % 200 == 0:
            val = _score_norm(a_curr)
            if val < best_val:
                best_val = val
                best_a = a_curr.copy()
        
        # Adaptive lr
        if it % 5000 == 0 and it > 0:
            lr *= 0.95
    
    val = _score_norm(a_curr)
    if val < best_val:
        best_val = val
        best_a = a_curr.copy()
    return best_a


def _coord_descent(a, deadline, max_rounds=1000):
    """Coordinate descent optimization."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)
    
    # Pre-compute autocorrelation for efficiency
    c = _convolve(a, a)
    
    factors = [0.95, 0.97, 0.99, 1.01, 1.03, 1.05]
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        if n > 1024:
            indices = rng.choice(n, size=min(256, n), replace=False)
        else:
            indices = np.arange(n)
        
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in factors:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                # Quick score computation
                a_norm = a / s
                c_new = _convolve(a_norm, a_norm)
                sc = float(2.0 * n * c_new.max())
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
            # Renormalize
            a = a / a.sum()
        
        if not improved and r > 50:
            break
    
    return best_a


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _adam_logspace(a_init, deadline, betas, lr=0.01, max_iter=500000):
    """Adam optimizer in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        per_deadline = min(deadline, time.monotonic() + 120)
        it_count = 0
        while it_count < max_iter:
            if time.monotonic() > per_deadline:
                break
            it_count += 1
            t += 1
            
            a = _softmax(z)
            _, grad, _, _ = _smoothed_max_grad(a, beta)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            
            if it_count % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            # Decay lr
            if it_count % 2000 == 0:
                lr = max(0.0005, lr * 0.97)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a


def _softmax(z):
