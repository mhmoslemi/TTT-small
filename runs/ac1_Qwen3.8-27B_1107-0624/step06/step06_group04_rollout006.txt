    # = conv(w[::-1], a)[len(w)-1 + j] for j = 0,...,n-1
    w_rev = w[::-1]
    full = _full_convolve(w_rev, a)
    # full has length len(w_rev) + n - 1 = len(w) + n - 1
    # We want full[len(w)-1 : len(w)-1+n]
    grad = 2.0 * full[len(w)-1:len(w)-1+n]
    return grad

def _gradient_logspace(a, w, c):
    """
    Gradient of f(a(z)) = sum_k w[k]*conv(a,a)[k] with respect to z, where a = softmax(z).
    = 2 * (h - a * dot(w, c))
    where h[j] = sum_k w[k]*a[k-j] (cross-correlation)
    and c = conv(a,a)
    """
    n = len(a)
    w_rev = w[::-1]
    full = _full_convolve(w_rev, a)
    h = full[len(w)-1:len(w)-1+n]
    W = float(np.dot(w, c))
    return 2.0 * (h - a * W)

"""
Multi-phase optimizer using adaptive top-k smoothing, log-space Adam with aggressive
beta continuation, spectral flattening, and multi-length exploration (4096, 8192, 16384).
Aims to beat 1.5027 by combining sharper peak reduction with broader exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta, top_k):
    """Compute smoothed weights over convolution values."""
    n_c = len(c)
    if top_k is not None and top_k < n_c:
        part_idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[part_idx]
        cm = float(top_vals.max())
        shifted = (top_vals - cm) / beta
        e = np.exp(shifted)
        e_sum = float(e.sum())
        if e_sum < 1e-300 or not np.isfinite(e_sum):
            w = np.zeros(n_c)
            w[int(part_idx[np.argmax(top_vals)])] = 1.0
        else:
            e = e / e_sum
            w = np.zeros(n_c)
            w[part_idx] = e
    else:
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        e_sum = float(e.sum())
        if e_sum < 1e-300 or not np.isfinite(e_sum):
            w = np.zeros(n_c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / e_sum
    return w


def _grad_simplex_vec(a, w):
    """Gradient of dot(w, conv(a,a)) w.r.t. a on simplex.
    df/da[j] = 2 * sum_k w[k] * a[k-j]
    """
    n = len(a)
    w_rev = w[::-1]
    full = _full_convolve(w_rev, a)
    grad = 2.0 * full[len(w) - 1:len(w) - 1 + n]
    return grad


def _grad_logspace_vec(a, w, c):
    """Gradient in log-space: d/dz of dot(w, conv(a,a)) where a=softmax(z).
    = 2 * (h - a * dot(w, c)) where h[j] = sum_k w[k]*a[k-j]
    """
    n = len(a)
    w_rev = w[::-1]
    full = _full_convolve(w_rev, a)
    h = full[len(w) - 1:len(w) - 1 + n]
    W = float(np.dot(w, c))
    return 2.0 * (h - a * W)


def _adam_logspace_v2(a_init, beta_schedule, deadline, lr_init=0.008,
                      top_k_schedule=None, max_iter_total=300000,
                      lr_decay_every=5000, lr_decay_factor=0.95,
                      grad_clip=2.0, momentum_b1=0.9, momentum_b2=0.999):
    """Adam optimizer in log-space with adaptive top-k smoothing."""
    n = len(a_init)
    s = a_init.sum()
    if s > 0:
        a_norm = a_init / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = momentum_b1, momentum_b2, 1e-8
    current_lr = lr_init
    no_improve = 0

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        # Determine top_k for this beta
        if top_k_schedule is not None and bi < len(top_k_schedule):
            tk = top_k_schedule[bi]
        else:
            tk = None  # full softmax

        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            if beta > 0:
                w = _compute_weights(c, beta, tk)
            else:
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0

            grad = _grad_logspace_vec(a, w, c)

            if not np.all(np.isfinite(grad)):
                # Try with a larger beta
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve = 0
                else:
                    no_improve += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve > 40 and t > 500:
                current_lr = max(0.0003, current_lr * 0.92)
                no_improve = 0

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_v2(a_init, beta, deadline, max_iter=5000, top_k=None):
    """FISTA with adaptive top-k smoothing for the smoothed max-convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 0:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _compute_weights(c_y, beta, top_k)
        grad = _grad_simplex_vec(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _spectral_flatten_v2(a_init, n_iter=300, deadline=None, blend=0.5):
    """Spectral flattening with blending to preserve non-negativity."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = 2 * n - 1

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        avg_mag = mag.mean()
        # Blend: partially flatten magnitude
        mag_new = (1.0 - blend) * mag + blend * avg_mag
        A_new = phase * mag_new
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        # FISTA on 8192
        betas_8k = []
        b = 0.005 * max_8k
        while b > 0.00005 * max_8k:
            betas_8k.append(b)
            b *= 0.6
        if len(betas_8k) > 6:
            betas_8k = betas_8k[:6]

        a_8k_opt = a_8k.copy()
        for b in betas_8k:
            rem = phase3_end - time.monotonic() - 20
            if rem < 5:
                break
            a_8k_opt, _ = _fista_v2(a_8k_opt, b, time.monotonic() + min(rem * 0.7, 12.0), max_iter=3000)
            a_8k_opt = a_8k_opt / a_8k_opt.sum()

        # Adam on 8192
        if time.monotonic() < phase3_end - 15:
            c_8k = _full_convolve(a_8k_opt, a_8k_opt)
            max_8k = float(c_8k.max())
            betas_8k_fine = []
            b = 0.0005 * max_8k
            while b > 0.000005 * max_8k:
                betas_8k_fine.append(b)
                b *= 0.6
            if len(betas_8k_fine) > 8:
                betas_8k_fine = betas_8k_fine[:8]

            per_t = phase3_end - time.monotonic() - 10
            a_8k_opt, _ = _adam_logspace_v2(a_8k_opt, betas_8k_fine, time.monotonic() + per_t,
                                            lr_init=0.008, max_iter_total=300000)
            a_8k_opt = a_8k_opt / a_8k_opt.sum()

        sc_8k_final = _score(a_8k_opt)
        print(f"  n=8192 final: score={sc_8k_final:.10f}")
        if sc_8k_final < best_score:
            best_score = sc_8k_final
            best_seq = a_8k_opt.copy()
            n_best_updated = 8192

    print(f"Phase 3 done: best score={best_score:.10f} (n={len(best_seq)})")

    # === Phase 4: Fine-tune best with very small beta (700-1100s) ===
    print("=== Phase 4: Ultra-fine tuning ===")
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end - 20:
        a_work = best_seq.copy()
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_ult = []
        b = 0.00005 * max_cw
        while b > 0.0000005 * max_cw:
            betas_ult.append(b)
            b *= 0.55
        if len(betas_ult) > 10:
            betas_ult = betas_ult[:10]

        top_k_sch4 = [None] * len(betas_ult)
        for i in range(len(betas_ult)):
            if i >= len(betas_ult) - 5:
                top_k_sch4[i] = max(1, (len(betas_ult) - i + 1))

        per_t = phase4_end - time.monotonic() - 10
        a_opt, _ = _adam_logspace_v2(a_work, betas_ult, time.monotonic() + per_t,
                                     lr_init=0.005, top_k_schedule=top_k_sch4,
                                     max_iter_total=400000, lr_decay_every=5000,
                                     lr_decay_factor=0.97, grad_clip=1.0)
        sc = _score(a_opt)
        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  Ultra-fine: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    print(f"Phase 4 done: score={best_score:.10f}")

    # === Phase 5: Spectral flatten + Adam (1100-1350s) ===
    print("=== Phase 5: Spectral flatten + refine ===")
    phase5_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase5_end - 20:
        a_sf, _ = _spectral_flatten_v2(a_work, n_iter=200, deadline=time.monotonic() + 30, blend=0.4)
        sc_sf = _score(a_sf)
        if sc_sf < best_score:
            best_score = sc_sf
            best_seq = a_sf.copy()
            print(f"  Spectral flatten: score={sc_sf:.10f}")

        # Adam after spectral
        if time.monotonic() < phase5_end - 15:
            a_work2 = best_seq.copy() / best_seq.sum()
            c_w2 = _full_convolve(a_work2, a_work2)
            mc2 = float(c_w2.max())
            betas_sf = [0.00002 * mc2, 0.00001 * mc2, 0.000005 * mc2]
            per_t = phase5_end - time.monotonic() - 10
            a_sf2, _ = _adam_logspace_v2(a_work2, betas_sf, time.monotonic() + per_t,
                                         lr_init=0.004, max_iter_total=150000)
            sc_sf2 = _score(a_sf2)
            if sc_sf2 < best_score:
                best_score = sc_sf2
                best_seq = a_sf2.copy()
                print(f"  Spectral+Adam: score={sc_sf2:.10f}")

    print(f"Phase 5 done: score={best_score:.10f}")

    # === Phase 6: Multi-restart (1350-1600s) ===
    print("=== Phase 6: Multi-restart ===")
    phase6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase6_end - 20:
        a_base = best_seq.copy()
        s_b = a_base.sum()
        a_base = a_base / s_b
        n_base = len(a_base)

        n_restarts = 5
        per_restart = (phase6_end - time.monotonic() - 15) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break

            if i % 3 == 0:
                p_start = _perturb(a_base, 0.015 + 0.005 * i, rng)
            elif i % 3 == 1:
                shift = int(rng.integers(5, max(6, n_base // 256)))
                p_start = np.roll(a_base, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                p_struct = _make_profile(n_base, ["cosine", "tapered"][i % 2])
                mix = 0.2 + 0.1 * i
                p_start = mix * p_struct + (1 - mix) * a_base
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            mc_s = float(c_s.max())
            beta_q = 0.001 * mc_s
            p_opt, _ = _fista_v2(p_start, beta_q, time.monotonic() + max(2.0, per_restart * 0.3), max_iter=2000)
            p_opt = p_opt / p_opt.sum()

            # Quick Adam
            if time.monotonic() < phase6_end - 10:
                c_s2 = _full_convolve(p_opt, p_opt)
                mc_s2 = float(c_s2.max())
                betas_r = [0.0003 * mc_s2, 0.0001 * mc_s2, 0.00003 * mc_s2]
                per_t_r = phase6_end - time.monotonic() - 8
                p_opt, _ = _adam_logspace_v2(p_opt, betas_r, time.monotonic() + per_t_r,
                                             lr_init=0.006, max_iter_total=50000)
                p_opt = p_opt / p_opt.sum()

            sc_r = _score(p_opt)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_opt.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.0f}s")

    print(f"Phase 6 done: score={best_score:.10f}")

    # === Phase 7: Subgradient + final polish (1600-1770s) ===
    print("=== Phase 7: Final polish ===")
    phase7_end = min(deadline, t_start + 1760.0)
    if time.monotonic() < phase7_end - 10:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        # Subgradient
        a_sg, _ = _subgradient_v2(a_work, n_steps=2000, deadline=time.monotonic() + 40)
        sc_sg = _score(a_sg)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            a_work = a_sg.copy() / a_sg.sum()
            print(f"  Subgradient: score={sc_sg:.10f}")

        # Final Adam with very small beta
        if time.monotonic() < phase7_end - 10:
            c_f = _full_convolve(a_work, a_work)
            mc_f = float(c_f.max())
            betas_final = [0.00001 * mc_f, 0.000005 * mc_f, 0.000002 * mc_f, 0.000001 * mc_f]
            per_t = phase7_end - time.monotonic() - 5
            if per_t > 10:
                a_fin, _ = _adam_logspace_v2(a_work, betas_final, time.monotonic() + per_t,
                                             lr_init=0.003, max_iter_total=200000,
                                             lr_decay_every=8000, lr_decay_factor=0.97,
                                             grad_clip=0.5)
                sc_fin = _score(a_fin)
                if sc_fin < best_score:
                    best_score = sc_fin
                    best_seq = a_fin.copy()
                    print(f"  Final Adam: score={sc_fin:.10f}")

    # === Final validation and return ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = final.sum()

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.0f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
