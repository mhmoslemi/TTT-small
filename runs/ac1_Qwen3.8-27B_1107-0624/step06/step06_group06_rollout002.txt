    g = conv_result[n - 1:2 * n - 1] * 2.0
    return g


def _Ha_prefix(a, k_min, k_max, n):
    """Fast computation of 2*H_S*a for consecutive active set [k_min, k_max] using prefix sums."""
    ind = np.zeros(2 * n - 1)
    ind[k_min:k_max + 1] = 1.0
    a_rev = a[::-1]
    conv_result = _full_convolve(a_rev, ind)
    return conv_result[n - 1:2 * n - 1] * 2.0


def _proj_simplex(a):
    """Project onto {x: sum(x)=1, x>=0}."""
    a_clip = np.maximum(a, 0.0)
    s = a_clip.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a_clip / s


def _multiplicative_weight(a, g, alpha):
    """Multiplicative weight update: a_i <- a_i * exp(-alpha * g_i), renormalized."""
    z = np.log(np.maximum(a, 1e-30))
    g_centered = g - np.mean(g)
    z = z - alpha * g_centered
    z = np.clip(z, -80.0, 80.0)
    return _softmax(z)


def _active_set_optimize(a_init, deadline, max_iter=20000, verbose=True, label=""):
    """Main optimizer: active-set multiplicative weight with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = _score(a)

    # Compute initial active set
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_sorted = np.sort(c)[::-1]
    # Active set: lags within 1% of max
    gap = c_sorted[0] - c_sorted[min(len(c_sorted) - 1, 20)]
    tol_frac = max(0.005, gap / c_max * 0.5)
    threshold = c_max - tol_frac * c_max
    active_idx = np.where(c >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c))])

    # Adaptive parameters
    alpha = 0.1
    alpha_min = 0.001
    patience = 0
    max_patience = 500
    active_update_interval = 50
    expand_count = 0

    t0 = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute gradient from active set
        g = _Ha(a, active_idx, n)

        # Multiplicative weight update
        a_new = _multiplicative_weight(a, g, alpha)

        # Check improvement
        if it % 10 == 0:
            sc = _score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                patience = 0
                # Slowly increase step size when improving
                alpha = min(alpha * 1.02, 0.3)
            else:
                a = a_new  # Still accept (CG-like)
                patience += 1
                if patience > 50:
                    alpha *= 0.9
                    if alpha < alpha_min:
                        # Try expanding active set
                        c = _full_convolve(a, a)
                        c_max = float(c.max())
                        tol_frac = max(0.002, tol_frac * 0.7)
                        threshold = c_max - tol_frac * c_max
                        new_active = np.where(c >= threshold)[0]
                        if len(new_active) > len(active_idx):
                            active_idx = new_active
                            tol_frac = max(0.001, tol_frac)
                            expand_count += 1
                            patience = 0
                            alpha = 0.05
                        else:
                            # Try reducing step size further or break
                            alpha *= 0.7
                            if alpha < alpha_min * 0.1:
                                break

        # Periodically update active set
        if it % active_update_interval == 0 and it > 0:
            c = _full_convolve(a, a)
            c_max = float(c.max())
            c_sorted = np.sort(c)[::-1]
            gap = c_sorted[0] - c_sorted[min(len(c_sorted) - 1, 30)]
            tol_frac = max(0.003, 0.5 * gap / c_max)
            threshold = c_max - tol_frac * c_max
            new_active = np.where(c >= threshold)[0]
            if len(new_active) >= 1:
                active_idx = new_active

    # Final score check
    sc_final = _score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _qp_polish(a_init, deadline, max_iter=2000, verbose=True, label=""):
    """Polish using projected gradient on the active-set QP with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())
    threshold = c_max - 0.005 * c_max
    active_idx = np.where(c >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([int(np.argmax(c))])

    # Estimate Lipschitz constant
    g = _Ha(a, active_idx, n)
    g_proj = g - np.mean(g)
    gn = np.dot(g_proj, g_proj)
    if gn < 1e-30:
        return best_a, best_score
    # Use power method for eigenvalue estimate
    v = g_proj / np.sqrt(gn)
    for _ in range(5):
        w = _Ha(v, active_idx, n)
        w = w - np.mean(w)
        wn = np.linalg.norm(w)
        if wn < 1e-30:
            break
        v_new = w / wn
        v = v_new
    L = max(wn / np.sqrt(gn), 1.0) if gn > 1e-30 else 100.0
    L = min(L, 500.0)

    alpha = 1.0 / L
    patience = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = _Ha(a, active_idx, n)
        g_proj = g - np.mean(g)
        gn = np.dot(g_proj, g_proj)
        if gn < 1e-20:
            break

        # Backtracking line search on QP objective
        f_old = float(np.dot(a, g))
        step_accepted = False
        alpha_ls = alpha
        for ls in range(20):
            a_new = a - alpha_ls * g_proj
            a_new = _proj_simplex(a_new)
            g_new = _Ha(a_new, active_idx, n)
            f_new = float(np.dot(a_new, g_new))
            if f_new < f_old - 1e-14 * alpha_ls * gn:
                a = a_new
                step_accepted = True
                alpha = min(alpha_ls * 1.5, 2.0 / L)
                break
            alpha_ls *= 0.5

        if not step_accepted:
            alpha *= 0.5
            patience += 1
            if patience > 100:
                break
        else:
            sc = _score(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
                if patience > 200:
                    break

        # Update active set periodically
        if it % 200 == 0:
            c = _full_convolve(a, a)
            c_max = float(c.max())
            threshold = c_max - 0.005 * c_max
            new_active = np.where(c >= threshold)[0]
            if len(new_active) >= 1:
                active_idx = new_active

    sc_final = _score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_upgrade(a_init, deadline, betas=None):
    """L-BFGS with adaptive beta continuation (as backup/refinement)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    if betas is None:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.004, 0.001, 0.0003])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + 10 > deadline:
            break
        remaining = deadline - 10 - time.monotonic()
        if remaining < 3:
            break
        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 10))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11, 'maxcor': 40, 'maxls': 50}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "exp_decay":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t)
        return a / a.sum()
    elif kind == "exp_sym":
        t = np.linspace(0, 1, n)
        a = np.exp(-1.5 * (2 * t - 1) ** 2)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "water_fill":
        # Start with a "water-filled" profile based on expected active set
        a = np.ones(n)
        # Slightly boost middle, reduce edges
        t = np.linspace(0, 1, n)
        a = a + 0.1 * np.cos(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage(a, deadline, fractions=None):
    """Try zeroing small elements to reduce effective support."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)

    if fractions is None:
        fractions = [0.01, 0.02, 0.05, 0.08, 0.12, 0.15, 0.20, 0.30]

    a_sorted_idx = np.argsort(a)

    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _multi_start_optimize(a_init, deadline, n_starts=3, verbose=True):
    """Try multiple optimization strategies from the same starting point."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    # Strategy 1: Active set multiplicative weight
    if time.monotonic() < deadline - 60:
        a1, sc1 = _active_set_optimize(a_init, min(deadline - 30, time.monotonic() + (deadline - time.monotonic()) * 0.4), max_iter=10000)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1.copy()

    # Strategy 2: QP polish
    if time.monotonic() < deadline - 30:
        a2, sc2 = _qp_polish(best_a, min(deadline - 10, time.monotonic() + (deadline - time.monotonic()) * 0.3), max_iter=2000)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()

    # Strategy 3: L-BFGS with fine beta
    if time.monotonic() < deadline - 10:
        a3 = _lbfgs_upgrade(best_a, min(deadline - 5, time.monotonic() + (deadline - time.monotonic()) * 0.25))
        sc3 = _score(a3)
        if sc3 < best_sc:
            best_sc = sc3
            best_a = a3.copy()

    # Strategy 4: Final active set pass
    if time.monotonic() < deadline - 5:
        a4, sc4 = _active_set_optimize(best_a, min(deadline - 2, time.monotonic() + (deadline - time.monotonic()) * 0.15), max_iter=5000)
        if sc4 < best_sc:
            best_sc = sc4
            best_a = a4.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy active-set optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multiplicative weight (mirror descent) with active set exploitation,
    QP polishing, L-BFGS refinement, shrinkage, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Main optimization from parent ---
    print(f"\n[Phase 1] Main optimization from parent (t={time.monotonic() - t_start:.1f}s)")
    phase1_deadline = deadline - 600.0
    a_main, sc_main = _multi_start_optimize(parent, phase1_deadline)
    update_best(a_main, sc_main, "main_opt")

    # --- Phase 2: Shrinkage ---
    print(f"\n[Phase 2] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 400)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score:
            a_shr2, sc_shr2 = _multi_start_optimize(a_shr, deadline - 300)
            update_best(a_shr2, sc_shr2, "shrink_reopt")

    # --- Phase 3: Multi-n exploration ---
    print(f"\n[Phase 3] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192]
    if n_parent not in n_candidates:
        n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 200:
            break
        if n_try == n_parent and best_seq is not None and len(best_seq) == n_try:
            continue  # Already optimized at this n

        # Initialize
        if parent is not None and n_parent != n_try:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            if a_init.sum() > 1e-10:
                a_init = a_init / a_init.sum()
            else:
                a_init = _make_profile(n_try, "tapered")
        else:
            a_init = _make_profile(n_try, "tapered")

        remaining = deadline - time.monotonic() - 100
        if remaining < 30:
            break
        a_opt, sc_opt = _multi_start_optimize(a_init, time.monotonic() + remaining)
        update_best(a_opt, sc_opt, f"n_{n_try}")

    # --- Phase 4: Diverse restarts ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        perturbations = [
            (0.03, "gauss3"),
            (0.05, "gauss5"),
            (0.08, "gauss8"),
        ]
        for sigma, label in perturbations:
            if time.monotonic() > deadline - 40:
                break
            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            remaining = deadline - time.monotonic() - 30
            if remaining < 20:
                break
            a_r, sc_r = _multi_start_optimize(p_start, time.monotonic() + remaining)
            update_best(a_r, sc_r, f"restart_{label}")

    # --- Phase 5: Structured initializations ---
    print(f"\n[Phase 5] Structured initializations (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        profiles = ["exp_decay", "exp_sym", "cos_mod", "water_fill"]
        n_struct = len(best_seq) if best_seq is not None else 4096
        for prof in profiles:
            if time.monotonic() > deadline - 50:
                break
            a_init = _make_profile(n_struct, prof)
            remaining = deadline - time.monotonic() - 40
            if remaining < 30:
                break
            a_opt, sc_opt = _active_set_optimize(a_init, time.monotonic() + remaining * 0.6, max_iter=5000)
            sc_opt = _score(a_opt)
            if sc_opt < best_score:
                # Polish it
                a_pol, sc_pol = _qp_polish(a_opt, time.monotonic() + remaining * 0.3, max_iter=1000)
                update_best(a_pol, sc_pol, f"struct_{prof}")
            else:
                update_best(a_opt, sc_opt, f"struct_{prof}_raw")

    # --- Phase 6: Final fine polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Very fine L-BFGS
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        betas_fine = mb * np.array([0.001, 0.0003, 0.0001, 0.00003])
        remaining = deadline - 10 - time.monotonic()
        if remaining > 10:
            a_fin = _lbfgs_upgrade(best_seq, time.monotonic() + remaining * 0.5, betas=betas_fine)
            update_best(a_fin, _score(a_fin), "final_lbfgs")

        # Final active set
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _active_set_optimize(best_seq, deadline - 5, max_iter=3000)
            update_best(a_fin2, sc_fin2, "final_active")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        # Sanity check with evaluate_sequence
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            if eval_sc < sc_final:
                sc_final = eval_sc
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
