def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)

    step = 1.0 / (4.0 * n)
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _frank_wolfe_lp(a, M, deadline):
    """Solve the LP: max sum(g) s.t. g>=0, conv(a,g)[k] <= M for all k.
    Returns the optimal g (not normalized)."""
    from scipy.optimize import linprog
    n = len(a)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: conv(a, g)[k] <= M for k = 0, ..., 2n-2
    # conv(a, g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * g[k-i]
    # This is a sparse matrix with (2n-1) rows and n columns.
    
    # Build the constraint matrix A_user (sparse)
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i  # index of g
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    
    A_user = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_user, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            return result.x
        else:
            return None
    except Exception:
        return None

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad = _compute_grad_conv(a, w)
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        
        if t % 100 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = _compute_grad_conv(a, w)
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines FISTA (adaptive L), Frank-Wolfe with LP oracle, Adam, and L-BFGS-B.
Key improvements: adaptive Lipschitz estimation, LP-based FW steps, better beta schedules,
structured multi-restart, and extended final polish.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_conv(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    return 2.0 * h


def _weights_from_conv(c, beta):
    c_max = float(c.max())
    if beta < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, c_max
    e = np.exp((c - c_max) / beta)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, c_max
    return e / s, c_max


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()
    L = 4.0 * n
    step = 1.0 / L

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)

        # Adaptive L: backtracking on the quadratic model
        # Check if the objective decreased
        diff = a_next - y
        diff_norm_sq = float(np.dot(diff, diff))
        if diff_norm_sq > 1e-30 and (it % 50 == 0):
            c_next = _full_convolve(a_next, a_next)
            c_next_sub = float(np.dot(w, c_next))
            model = c_max + float(np.dot(w, grad * (a_next - y))) + 0.5 * L * diff_norm_sq
            if c_next_sub > model + 1e-10:
                L *= 2.0
                step = 1.0 / L
                a_next = _proj_simplex(y - step * grad)
            elif (it % 200 == 0) and L > 4.0 * n * 2.0:
                L = max(4.0 * n, L * 0.9)
                step = 1.0 / L

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _frank_wolfe_lp(a, M):
    """Solve LP: max sum(g) s.t. g>=0, conv(a,g)[k] <= M for all k."""
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    n = len(a)
    c_obj = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            vals.append(a[i])
    A_user = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=A_user, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': 60, 'presolve': True})
        if result.success:
            return result.x
        return None
    except Exception:
        return None


def _frank_wolfe_opt(a_init, deadline, max_iter=30):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline - 10:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        print(f"    FW iter {it}: M={M:.8f}, solving LP...")
        g = _frank_wolfe_lp(a, M)
        if g is None:
            print("    LP failed, stopping FW.")
            break
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g / g_sum

        best_t = 0.0
        best_sc = best_score
        a_filled = np.maximum(a, 0.0)
        for t in [1.0/(it+2), 0.5, 2.0/(it+3), 0.3, 0.7, 0.1, 0.9]:
            t = min(1.0, max(0.0, t))
            a_try = (1.0 - t) * a_filled + t * g
            a_try = np.maximum(a_try, 0.0)
            s_t = a_try.sum()
            if s_t < 1e-15:
                continue
            a_try = a_try / s_t
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            if no_improve >= 3:
                break
            continue

        a = (1.0 - best_t) * a_filled + best_t * g
        a = np.maximum(a, 0.0)
        s_a = a.sum()
        if s_a > 1e-15:
            a = a / s_a
        else:
            break

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            print(f"    FW iter {it}: score={sc:.10f}, t={best_t:.4f}")

    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    best_iter = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        w, cm = _weights_from_conv(c, beta)
        grad = _compute_grad_conv(a, w)
        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                best_iter = t
        if t % 8000 == 0:
            current_lr = max(0.0001, current_lr * 0.88)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        w, cm = _weights_from_conv(c, beta)
        grad_a = _compute_grad_conv(a, w)
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 80)), 'gtol': 1e-14, 'maxls': 25}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-12)

        t = np.linspace(0, 1, n)

    else:

            # One more FISTA pass with tiny beta
            if time.monotonic() < deadline - 8:
                c_g = _full_convolve(a_pol, a_pol)
                b_g = 1e-7 * float(c_g.max())
                a_pol, _ = _fista_opt(a_pol, b_g, deadline - 4.0, 100000)
                a_pol = np.maximum(a_pol, 0.0)
                s_pol = a_pol.sum()
                if s_pol > 1e-15:
                    a_pol = a_pol / s_pol
                sc_g = _score(a_pol)
                if sc_g < best_score:
                    best_score = sc_g
                    best_seq = a_pol.copy()
                    print(f"  Final FISTA: score={sc_g:.10f}")

    print(f"\nDone: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
