def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient_from_weights(a, weights):
    """Compute subgradient: g[k] = 2 * sum_j a[j] * w[j+k]"""
    n = len(a)
    w_rev = weights[::-1]
    conv_full = _fft_convolve(a, w_rev)
    g = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_soft_gradient(a, c, T):
    n = len(a)
    if T < 1e-12:
        idx = np.argmax(c)
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        c_shifted = (c - c.max()) / T
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w = e / s
    return _compute_gradient_from_weights(a, w)


def _compute_active_gradient(a, c, active_indices):
    n = len(a)
    w = np.zeros_like(c)
    w[active_indices] = 1.0 / len(active_indices)
    return _compute_gradient_from_weights(a, w)


def _compute_adversarial_gradient(a, c, active_indices, rng, n_adversarial=4):
    n = len(a)
    all_indices = list(active_indices)
    if n_adversarial > 0:
        rand_indices = rng.choice(2 * n - 1, size=n_adversarial, replace=False)
        all_indices = all_indices + list(rand_indices)
    w = np.zeros_like(c)
    w[all_indices] = 1.0 / len(all_indices)
    return _compute_gradient_from_weights(a, w)


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _cosine_window(n, alpha=0.5):
    """Tukey/Hann-like window."""
    x = np.arange(n) / (n - 1)
    if alpha >= 1.0:
        w = 0.5 * (1 - np.cos(2 * np.pi * x))
    else:
        w = np.ones(n)
        m = int(alpha * (n - 1))
        for i in range(m):
            w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
            w[n - 1 - i] = w[i]
    w = np.maximum(w, 1e-10)
    return w / w.sum()

        other_ns = [n for n in [2048, 3072, 4096, 5120] if n != n_best]
        for n_try in other_ns:
            if time.monotonic() > phase4_end - 60:
                break
            a_interp = _interp_to_length(best_a, n_try)
            c_i = _fft_convolve(a_interp, a_interp)
            max_ci = float(c_i.max())
            time_alloc = min(80.0, phase4_end - time.monotonic() - 30)
            a_opt, sc_opt = _adam_z_space(
                a_interp, t_start + time.monotonic() - t_start + time_alloc,
                T_start=0.001 * max_ci, T_end=0.00001 * max_ci,
                lr_start=0.025, lr_end=0.002,
                max_iter=100000, active_size=6,
                eval_interval=100, use_active_set=True,
                restart_patience=6
            )
            full_sc = _score(a_opt)
            print(f"Phase 4 n={n_try}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_opt.copy()

    # Phase 5: Perturbation restarts (1300s - 1550s)
    phase5_end = min(deadline, t_start + 1550.0)
    n_best = len(best_a)
    if time.monotonic() < phase5_end:
        for attempt in range(6):
            if time.monotonic() > phase5_end - 30:
                break
            sigma = 0.008 + 0.005 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            time_alloc = min(35.0, phase5_end - time.monotonic() - 10)
            a_pert_opt, sc_pert = _adam_z_space(
                a_pert, t_start + time.monotonic() - t_start + time_alloc,
                T_start=0.0005 * max_cp, T_end=0.000005 * max_cp,
                lr_start=0.015, lr_end=0.001,
                max_iter=60000, active_size=4,
                eval_interval=80, use_active_set=True,
                restart_patience=5
            )
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] (sigma={sigma:.4f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 6: Final ultra-fine polish (1550s - deadline)
    if time.monotonic() < deadline - 10:
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        
        # Very tight Adam
        a_final1, sc_final1 = _adam_z_space(
            best_a, deadline - 5,
            T_start=0.00005 * max_cf, T_end=0.000001 * max_cf,
            lr_start=0.003, lr_end=0.0002,
            max_iter=200000, active_size=3,
            eval_interval=50, use_active_set=True,
            restart_patience=12
        )
        full_sc = _score(a_final1)
        print(f"Phase 6a Adam-ultra: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final1.copy()

        # FISTA on top
        if time.monotonic() < deadline - 15:
            c_fin2 = _fft_convolve(best_a, best_a)
            max_cf2 = float(c_fin2.max())
            a_final2, sc_final2 = _fista_mirror_z(
                best_a, deadline - 5,
                T_start=0.00002 * max_cf2, T_end=0.0000005 * max_cf2,
                eta_start=0.002, eta_end=0.0001,
                max_iter=100000, active_size=3, eval_interval=50,
                use_adversarial=False,
                restart_patience=15
            )
            full_sc = _score(a_final2)
            print(f"Phase 6b FISTA-ultra: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_final2.copy()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * 0.01 / 256
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
