def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _grad_smooth_max(a, beta):

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous:
1. Hann-window and spectral initializations (better than uniform)
2. Multi-resolution coarse-to-fine optimization
3. Spectral flattening post-processing
4. Aggressive adaptive beta schedule with "restart on stagnation"
5. Larger n exploration (8192, 16384) via interpolation
6. Deflation-based peak reduction as auxiliary strategy
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_smooth_max(a, beta):
    """Gradient of smooth max of conv(a,a), assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _hann_init(n):
    t = np.arange(n) / max(1, n - 1)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _raised_cosine_init(n):
    t = np.arange(n) / max(1, n - 1)
    a = np.sin(np.pi * t) ** 1.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _bimodal_cos_init(n):
    t = np.arange(n) / max(1, n - 1)
    a = (1.0 + 0.4 * np.cos(4.0 * np.pi * t)) 
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _spectral_flatten(a, alpha=0.15):
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    mag_sq = np.abs(A) ** 2
    phase = np.angle(A)
    flat = np.full_like(mag_sq, mag_sq.mean())
    new_mag_sq = (1.0 - alpha) * mag_sq + alpha * flat
    new_mag = np.sqrt(np.maximum(new_mag_sq, 0))
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _adam_optimize(a_init, deadline, beta_schedule, lr_init=0.02, max_iters_per_beta=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        iters = min(max_iters_per_beta, max(500, int((deadline - time.monotonic()) / max(1, len(beta_schedule) - beta_idx) * 3000)))
        for it in range(iters):
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax(z)
            grad = _grad_smooth_max(a, beta)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            step = current_lr * mh / (np.sqrt(vh) + eps)
            sn = np.linalg.norm(step)
            if sn > 1.0:
                step = step / sn

            z = z - step
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        current_lr = max(0.001, current_lr * 0.7)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _deflation_step(a, n_iter=100, deadline=None):
    """Reduce peak by deflating elements that contribute to it."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # Find elements contributing most to the peak
        # contribution[i] = a[peak_idx - i] (for valid i)
        contrib = np.zeros(n)
        i_start = max(0, peak_idx - n + 1)
        i_end = min(n - 1, peak_idx)
        if i_start <= i_end:
            contrib[i_start:i_end + 1] = a[peak_idx - i_end:peak_idx - i_start + 1][::-1]
        
        # Deflate top contributors
        top_k = max(1, n // 64)
        top_indices = np.argsort(contrib)[::-1][:top_k]
        factor = 0.95
        for idx in top_indices:
            a[idx] *= factor
        
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        sc = _score_norm(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
        elif sc > best_score * 1.01:
            # Revert if got worse
            a = best_a.copy()
            break
    
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
