    k = 1
    while k < n:

"""
Coordinate descent + L-BFGS hybrid for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: exact coordinate-wise optimization via 1D root-finding,
combined with L-BFGS for global structure and multiple diverse initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _coord_update(a, m, S):
    """
    Find optimal a[m] to minimize the score, given all other elements fixed.
    Uses the analytic structure of the 1D subproblem.
    """
    n = len(a)
    x = a[m]
    
    # Compute c = conv(a_prime, a_prime) where a_prime = a with a[m]=0
    a_prime = a.copy()
    a_prime[m] = 0.0
    c = np.convolve(a_prime, a_prime)  # Use direct for simplicity (n is manageable)
    
    # b_k(xp) = c[k] + 2*a[k-m]*xp for valid k != 2m
    #           c[2m] + xp^2 for k = 2m
    # S_new = S - x + xp
    # score = 2n * max(b_k) / S_new^2
    
    # Precompute linear parts
    k_lo = m
    k_hi = min(m + n - 1, 2 * n - 2)
    ks = np.arange(k_lo, k_hi + 1)
    a_idx = ks - m  # ranges from max(0, k_lo-m) to min(n-1, k_hi-m)
    # Since k_lo = m and k_hi <= m+n-1, a_idx ranges from 0 to n-1
    # Filter: a_idx != m (since a_prime[m] = 0)
    
    lin_ks = ks[a_idx != m]
    c_lin = c[lin_ks]
    d_lin = 2.0 * a[a_idx[a_idx != m]]
    
    has_quad = (2 * m <= 2 * n - 2)
    c_quad = c[2 * m] if has_quad else 0.0
    
    # Define the max-conv function
    if len(c_lin) > 0:
        c_lin_max = float(c_lin.max())
    else:
        c_lin_max = 0.0
    
    def bmax(xp):
        if len(c_lin) > 0:
            lin_vals = c_lin + d_lin * xp
            lm = float(lin_vals.max())
        else:
            lm = -1e30
        qv = c_quad + xp * xp if has_quad else -1e30
        return max(lm, qv)
    
    def obj(xp):
        S_new = S - x + xp
        if S_new < 1e-15:
            return 1e30
        return 2.0 * n * bmax(xp) / (S_new * S_new)
    
    # Golden section / bounded search
    # Lower bound: 0
    # Upper bound: try a few multiples of x
    hi = max(3.0 * x + 0.01, 0.5 / n * 3)
    
    # If x is already very small, the optimum might be at 0
    if x < 1e-10:
        val_0 = obj(0.0)
        val_hi = obj(hi)
        if val_0 <= val_hi:
            return 0.0
    
    res = minimize_scalar(obj, bounds=(0.0, hi), method='bounded',
                          options={'xatol': 1e-10, 'maxiter': 50})
    return res.x

def _coord_descent_pass(a, order, S, damping=1.0):
    """One pass of coordinate descent."""
    n = len(a)
    for m in order:
        x_new = _coord_update(a, m, S)
        # Damping
        x_old = a[m]
        a[m] = (1.0 - damping) * x_old + damping * x_new
        S += a[m] - x_old
    return a, S

def _run_coord_descent(a_init, deadline, n_max_passes=200, damping=1.0, seed=42):
    """Run coordinate descent for multiple passes."""
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    a = a / S  # normalize
    S = 1.0
    
    best_a = a.copy()
    best_sc = _score(a)
    
    no_improve = 0
    for p in range(n_max_passes):
        if time.monotonic() > deadline:
            break
        order = list(rng.permutation(n))
        a, S = _coord_descent_pass(a, order, S, damping)
        a = a / a.sum()  # renormalize
        S = 1.0
        sc = _score(a)
        if sc < best_sc - 1e-12:
            improved = best_sc - sc
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
            if p % 10 == 0:
                print(f"    CD pass {p}: score={sc:.10f} (improve={improved:.2e})")
        else:
            no_improve += 1
            if no_improve >= 20:
                # Perturb and restart
                z = np.log(np.maximum(a, 1e-30))
                z += 0.05 * rng.standard_normal(n)
                z = z - z.mean()
                a = _softmax(z)
                a = a / a.sum()
                no_improve = 0
    
    return best_a, best_sc

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        max_iter = min(3000, max(50, int(remaining / 0.05)))
        
        result = minimize(
            lambda z_local: _smoobj(z_local, beta, n),
            z,
            jac=lambda z_local: _smograd(z_local, beta, n),
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
        )
        a_new = _softmax(result.x)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc_new
            z = np.log(np.maximum(a_new, 1e-30))
            z = z - z.mean()
    
    return current_a
