"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
- Top-K peak approximation for efficient gradients
- FISTA with geometric beta continuation
- Adam in log-space with adaptive learning rate
- Exploration of larger sequence lengths (up to 32768)

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_simplex(y, w):
    """Compute gradient of smoothed objective on simplex.
    w is a distribution over convolution lags (length 2n-1).
    Returns gradient of length n."""
    n = len(y)
    a_rev = y[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_logspace(a, w, n):
    """Compute gradient in log-space."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, _full_convolve(a, a)))
    return 2.0 * a * (h - W)


def _topk_weights(c, beta, top_k):
    """Compute weights using top-K approximation."""
    n_c = len(c)
    k = min(top_k, n_c)
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    shifted = (top_vals - top_vals.max()) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(n_c)
        w[top_idx[np.argmax(top_vals)]] = 1.0
        return w
    w = np.zeros(n_c)
    w[top_idx] = e / s
    return w


def _full_soft_weights(c, beta):
    """Compute full softmax weights."""
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter, top_k=0):
    """FISTA optimization on simplex with smoothed max objective.
    If top_k > 0, use top-K approximation; else use full softmax."""
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 1e-30:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c_init
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        if top_k > 0:
            w = _topk_weights(c_y, beta, top_k)
        else:
            w = _full_soft_weights(c_y, beta)

        grad = _compute_grad_simplex(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay_every=5000, lr_decay_factor=0.97, grad_clip=1.0, top_k=0):
    """Adam optimization in log-space with beta continuation."""
    n = len(a_init)
    a_norm = a_init.copy()
    s = a_norm.sum()
    if s > 1e-30:
        a_norm = a_norm / s
    else:
        a_norm = np.ones(n) / n

    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    best_a = a_norm.copy()
    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    no_improve_count = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total:
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            if top_k > 0:
                w = _topk_weights(c, beta, top_k)
            else:
                w = _full_soft_weights(c, beta)

            grad = _compute_grad_logspace(a, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > grad_clip:
                update = update * (grad_clip / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1

            if t % lr_decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay_factor)

            if no_improve_count > 40 and t > 500:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten(a_init, n_iter=200, deadline=None):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = _score_norm(a)
    N_fft = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        A = np.fft.rfft(a, N_fft)
        mag_sq = (A.real ** 2 + A.imag ** 2)
        mag_sq_avg = mag_sq.mean()
        # Flatten: reduce peaks in power spectrum
        ratio = mag_sq_avg / (mag_sq + 1e-30)
        new_A_mag = np.sqrt(mag_sq * ratio + 1e-30)
        # Preserve phase
        phase = A / (np.abs(A) + 1e-30)
        new_A = new_A_mag * phase
        a_new = np.fft.irfft(new_A, N_fft)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new
        if it % 10 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Adam fine-tuning on best ===
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    n_best = len(a_work)
    print(f"Phase 3: Adam fine-tuning on n={n_best}")
    phase3_end = min(deadline, t_start + 750.0)
    if time.monotonic() < phase3_end - 15:
        c_work = _full_convolve(a_work, a_work)
        max_cw = float(c_work.max())
        betas_fine = []
        b = 0.001 * max_cw
        while b > 0.000001 * max_cw:
            betas_fine.append(b)
            b *= 0.5
        if len(betas_fine) > 15:
            betas_fine = betas_fine[:15]

        per_t = phase3_end - time.monotonic() - 10
        a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + per_t, betas_fine,
                                       lr_init=0.01, max_iter_total=300000, top_k=50)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            n_best = len(best_seq)
            print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    n_best = len(a_work)
    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Peak shaving + spectral flatten ===
    print(f"Phase 4: Peak shaving + spectral on n={n_best}")
    phase4_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase4_end - 10:
        a_ps, sc_ps = _peak_shaving(a_work, n_steps=2000, deadline=phase4_end)
        sc_ps_full = _score(a_ps)
        if sc_ps_full < best_score:
            best_score = sc_ps_full
            best_seq = a_ps.copy()
            print(f"  Peak shaving: score={sc_ps_full:.10f} t={time.monotonic()-t_start:.1f}s")

        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        if time.monotonic() < phase4_end - 5:
            a_sf, sc_sf = _spectral_flatten(a_work, n_iter=500, deadline=phase4_end)
            sc_sf_full = _score(a_sf)
            if sc_sf_full < best_score:
                best_score = sc_sf_full
                best_seq = a_sf.copy()
                print(f"  Spectral flatten: score={sc_sf_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    n_best = len(a_work)
    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: Multi-restart diversification ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase5_end - 30:
        n_restarts = 10
        per_restart = (phase5_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            t_restart = time.monotonic()

            # Diverse initialization
            r = rng.random()
            if r < 0.15:
                p_start = _perturb(a_work, 0.01 + 0.005 * i, rng)
            elif r < 0.3:
                p_start = _perturb(a_work, 0.05, rng)
            elif r < 0.45:
                # Shift
                shift = int(rng.integers(1, max(2, n_best // 1024)))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif r < 0.6:
                # Mix with structured profile
                p_struct = _make_profile(n_best, ["cosine", "tapered", "two_block", "gaussian_center"][i % 4])
                mix = 0.2 + 0.1 * (i % 5)
                p_start = mix * p_struct + (1 - mix) * a_work
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif r < 0.75:
                # Spectral flatten of perturbed
                p_tmp = _perturb(a_work, 0.02, rng)
                p_start, _ = _spectral_flatten(p_tmp, n_iter=50, deadline=time.monotonic() + 5)
                p_start = p_start / p_start.sum()
            else:
                # Random new profile
                p_start = _make_profile(n_best, ["uniform", "cosine", "tapered"][i % 3])
                p_start = _perturb(p_start, 0.1, rng)

            # Quick FISTA
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            beta_fast = 0.002 * max_cs
            per_ft = max(4.0, per_restart * 0.4)
            p_opt, _ = _fista_smoothed(p_start, beta_fast, time.monotonic() + per_ft, 5000, top_k=50)
            p_opt = p_opt / p_opt.sum()

            # Quick Adam
            if time.monotonic() < phase5_end - 15:
                betas_r = [0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs]
                per_t_r = phase5_end - time.monotonic() - 10
                if per_t_r > 5:
                    p_opt, sc_r = _adam_logspace(p_opt, time.monotonic() + per_t_r, betas_r,
                                                 lr_init=0.008, max_iter_total=50000, top_k=50)
                    sc_r_full = _score(p_opt)
                    if sc_r_full < best_score:
                        best_score = sc_r_full
                        best_seq = p_opt.copy()
                        n_best = len(best_seq)
                        print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    n_best = len(a_work)
    print(f"End Phase 5: best={best_score:.10f}")

    # === Phase 6: Try n=32768 ===
    print("Phase 6: Try n=32768")
    phase6_end = min(deadline, t_start + 1550.0)
    n_32k = 32768
    if time.monotonic() < phase6_end - 60:
        a_32k = _interp_profile(best_seq, n_32k)
        sc_32k = _score(a_32k)
        print(f"  Interpolated n=32768: score={sc_32k:.10f}")
        if sc_32k < best_score:
            best_score = sc_32k
            best_seq = a_32k.copy()
            n_best = n_32k
            a_work = a_32k.copy()
        else:
            a_work = a_32k.copy()

        # FISTA on 32768
        c_32k = _full_convolve(a_work, a_work)
        max_c32 = float(c_32k.max())
        betas_32k = []
        b = 0.02 * max_c32
        while b > 0.0001 * max_c32:
            betas_32k.append(b)
            b *= 0.5
        if len(betas_32k) > 12:
            betas_32k = betas_32k[:12]

        for i, b in enumerate(betas_32k):
            if time.monotonic() > phase6_end - 10:
                break
            per_t = phase6_end - time.monotonic() - 5
            if per_t < 3:
                break
            tk = 100 if b > 0.003 * max_c32 else 50
            max_it = max(2000, int(per_t * 800))
            a_work, sc32 = _fista_smoothed(a_work, b, time.monotonic() + min(15.0, per_t), max_it, top_k=tk)
            a_work = a_work / a_work.sum()
            sc32_full = _score(a_work)
            if sc32_full < best_score:
                best_score = sc32_full
                best_seq = a_work.copy()
                n_best = n_32k
                print(f"  32k FISTA beta={b:.6f}: score={sc32_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 6: best={best_score:.10f}")

    # === Phase 7: Final ultra-fine polish ===
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    n_best = len(a_work)
    print(f"Phase 7: Ultra-fine polish on n={n_best}")
    phase7_end = min(deadline, t_start + 1720.0)
    if time.monotonic() < phase7_end - 10:
        c_pol = _full_convolve(a_work, a_work)
        max_cp = float(c_pol.max())
        betas_ult = [0.00005 * max_cp, 0.00002 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp]
        per_t = phase7_end - time.monotonic() - 5
        a_pol, sc_pol = _adam_logspace(a_work, time.monotonic() + per_t, betas_ult,
                                       lr_init=0.003, max_iter_total=300000,
                                       lr_decay_every=10000, lr_decay_factor=0.95, top_k=30)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Peak shaving after Adam
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        if time.monotonic() < phase7_end - 3:
            a_ps2, _ = _peak_shaving(a_work, n_steps=2000, deadline=phase7_end, top_k=10)
            sc_ps2 = _score(a_ps2)
            if sc_ps2 < best_score:
                best_score = sc_ps2
                best_seq = a_ps2.copy()
                print(f"  Final peak shave: score={sc_ps2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = final.sum()
            final = final / s

        sc_final = _score(final)
        print(f"FINAL internal: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
