"""
Mirror descent with FFT-accelerated subgradient for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses cross-correlation trick for O(n log n)
gradient computation, adaptive active-set size, and multi-phase strategy.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _grad_cross(weights, a):
    """Compute grad[i] = 2 * sum_k weights[k] * a[k-i] via FFT."""
    n = len(a)
    a_rev = a[::-1]
    # Linear convolution of weights (len 2n-1) and a_rev (len n) has length 3n-2
    N = _next_pow2(3 * n - 2)
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W * A, N)[:3 * n - 2]
    return 2.0 * conv[n - 1 : 2 * n - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

def _score_norm1(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    rho = np.where(cond)[0][-1] if np.any(cond) else 0
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_safe(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _fista_smoothed(a_init, beta, deadline, max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-30:
        a = a / s
    else:
        a = np.ones(n) / n
    a_prev = a.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a, a)
    max_c = float(c_init.max())
    if beta <= 0:
        beta = 1e-6 * max_c
    L = 4.0 * n / beta
    step = 1.0 / L
    
    best_score = _score_norm1(a)
    best_a = a.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a + ((t_val - 1.0) / t_next) * (a - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        c_shifted = (c_y - cm) / beta

        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        a_next = _proj_simplex(y - step * grad)

                best_a = a_next.copy()

        a = a_next
        t_val = t_next
    
    sc = _score_norm1(a)
    if sc < best_score:

        best_a = a.copy()
    return best_a, best_score
