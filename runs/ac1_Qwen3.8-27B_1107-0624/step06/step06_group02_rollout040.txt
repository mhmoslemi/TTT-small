    return np.maximum(v - theta, 0.0)


def _conv_grad(a, beta):
    """Compute convolution, smoothed value, weights, and gradient."""
    n = len(a)
    c = _convolve(a, a)
    c_max = float(c.max())
    exponents = (c - c_max) / beta
    np.clip(exponents, -500, 500, out=exponents)
    w = np.exp(exponents)
    w_sum = float(w.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    val = c_max + beta * np.log(max(w_sum, 1e-300))
    a_flip = a[::-1]
    wc = _convolve(w, a_flip)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return c, val, w, grad


def _fista_smooth(a_init, beta, deadline, max_iter):
    """FISTA for smoothed max-convolution with simplex constraint."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    c0, v0, w0, g0 = _conv_grad(a_curr, beta)
    L = max(4.0 / beta, 2.0 * np.linalg.norm(g0) / max(np.linalg.norm(a_curr), 1e-10) + 2.0 / beta)
    step = 1.0 / L

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        _, _, _, grad_y = _conv_grad(y, beta)
        a_next = _proj_simplex(y - step * grad_y)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_list, lr_init, max_iter_total):
    """Adam in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    global_it = 0

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            global_it += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_flip = a[::-1]
            wc = _convolve(w, a_flip)
            h = wc[n - 1:2 * n - 1]
            grad_a = 2.0 * h
            a_dot_grad = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - a_dot_grad)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 8000 == 0:
                current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    """Coordinate descent with multiplicative updates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [0.95, 0.98, 1.02, 1.05]:
                a[j] = old_val * factor
                sc = _score_norm(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    old_val = a[j]
                else:
                    a[j] = old_val
        if not improved and r > 30:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    scale = np.sqrt(n0 / n_new)
    p_new = np.interp(x1, x0, p * scale)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _make_initializations(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())
        inits.append(_perturb(parent, 0.03, rng))
        inits.append(_perturb(parent, 0.08, rng))
    x = np.linspace(0, 1, n)
    # Cosine window
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    # Raised cosine
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    # Gaussian centered
    a = np.exp(-2.0 * (x - 0.5) ** 2)
    inits.append(a / a.sum())
    # Bimodal
    a = np.exp(-3.0 * (x - 0.25) ** 2) + np.exp(-3.0 * (x - 0.75) ** 2)
    inits.append(a / a.sum())
    # Uniform
    inits.append(np.ones(n) / n)
    # Inverse cosine (peaked at edges)
    a = 1.0 / (0.1 + np.abs(x - 0.5) * 4.0)
    inits.append(a / a.sum())
    # Quartic depression
    a = 1.0 - 0.7 * (2 * x - 1) ** 4
    a = np.maximum(a, 1e-10)
    inits.append(a / a.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer targeting sub-1.5027 score.
    Phases: multi-start -> FISTA beta-continuation -> Adam log-space ->
    n=16384 upsample+optimize -> coordinate descent polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_main = 4096

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: Multi-start initialization ===
    print(f"Phase 1: Multi-start (n={n_main})")
    phase1_end = min(deadline, t_start + 200.0)
    inits = _make_initializations(n_main, parent if parent is not None and len(parent) == n_main else None, rng)

    for idx, a_init in enumerate(inits):
        if time.monotonic() > phase1_end - 10:
            break
        per_time = min(30.0, phase1_end - time.monotonic() - 5)
        if per_time < 5:
            break
        c_init = _convolve(a_init, a_init)
        max_c = float(c_init.max())
        beta = 0.02 * max_c
        a_opt, sc_opt = _fista_smooth(a_init, beta, time.monotonic() + per_time * 0.7, 20000)
        a_opt = a_opt / a_opt.sum()
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Init {idx}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    print(f"After Phase 1: score={best_score:.10f}")

    # === Phase 2: FISTA with aggressive beta continuation ===
    print("Phase 2: FISTA beta continuation")
    phase2_end = min(deadline, t_start + 600.0)
    c_curr = _convolve(a_work, a_work)
    max_c0 = float(c_curr.max())

    betas_list = []
    beta = 0.05 * max_c0
    while beta > 1e-10 * max_c0:
        betas_list.append(beta)
        beta *= 0.45
    betas_list.append(1e-10 * max_c0)
    if len(betas_list) > 30:
        step = len(betas_list) // 30
        betas_list = betas_list[::step] + [1e-10 * max_c0]

    a_curr = a_work.copy()
    for i, beta in enumerate(betas_list):
        if time.monotonic() > phase2_end - 5:
            break
        per_time = min(30.0, phase2_end - time.monotonic() - 3)
        if per_time < 2:
            break
        max_it = max(5000, int(per_time * 2000))
        a_curr, _ = _fista_smooth(a_curr, beta, time.monotonic() + per_time, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"After Phase 2: score={best_score:.10f}")

    # === Phase 3: Adam log-space fine optimization ===
    print("Phase 3: Adam log-space")
    phase3_end = min(deadline, t_start + 1000.0)
    a_base = best_seq.copy()
    a_base = a_base / a_base.sum()
    c_base = _convolve(a_base, a_base)
    max_cb = float(c_base.max())

    betas_fine = [0.005 * max_cb, 0.002 * max_cb, 0.001 * max_cb,
                  0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb,
                  0.00005 * max_cb, 0.00002 * max_cb]

    per_t = phase3_end - time.monotonic() - 10
    if per_t > 30:
        a_opt, _ = _adam_logspace(a_base, time.monotonic() + per_t,
                                   betas_fine, lr_init=0.008, max_iter_total=2000000)
        sc_opt = _score(a_opt)
        print(f"  Adam: score={sc_opt:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    print(f"After Phase 3: score={best_score:.10f}")

    # === Phase 4: Multi-restart ===
    print("Phase 4: Multi-restart")
    phase4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase4_end - 50:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 20) / n_restarts
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            if i % 3 == 0:
                p_start = _perturb(a_work, 0.015 + 0.01 * i, rng)
            elif i % 3 == 1:
                sz = n_main // 512
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                # Block shuffle
                block = n_main // 8
                p_start = a_work.copy()
                for b in range(7):
                    sh = rng.integers(1, block)
                    p_start[b*block:(b+1)*block] = np.roll(p_start[b*block:(b+1)*block], sh)
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs,
                  0.0001 * max_cs, 0.00005 * max_cs]
            p_best, _ = _adam_logspace(p_start, time.monotonic() + max(5.0, per_restart * 0.6),
                                        bs, lr_init=0.006, max_iter_total=200000)
            # FISTA polish
            c_pb = _convolve(p_best, p_best)
            max_cp = float(c_pb.max())
            p_best, _ = _fista_smooth(p_best, 0.00001 * max_cp,
                                       time.monotonic() + per_restart * 0.2, 30000)
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"After Phase 4: score={best_score:.10f}")

    # === Phase 5: n=16384 upsample and optimize ===
    print("Phase 5: n=16384 exploration")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 40 and best_seq is not None:
        n_big = 16384
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=16384: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()
            n_main = n_big

        if time.monotonic() < phase5_end - 20:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            # FISTA on big
            beta_big = 0.01 * max_cb
            per_fista = min(60.0, phase5_end - time.monotonic() - 20)
            if per_fista > 10:
                a_big_opt, _ = _fista_smooth(a_big, beta_big,
                                              time.monotonic() + per_fista, 100000)
                a_big_opt = a_big_opt / a_big_opt.sum()
                sc_bo = _score(a_big_opt)
                if sc_bo < best_score:
                    best_score = sc_bo
                    best_seq = a_big_opt.copy()
                    print(f"  FISTA n=16384: score={sc_bo:.10f} t={time.monotonic()-t_start:.1f}s")

                # Adam on big
                if time.monotonic() < phase5_end - 10:
                    c_bo2 = _convolve(a_big_opt, a_big_opt)
                    max_cbo = float(c_bo2.max())
                    bs_big = [0.002 * max_cbo, 0.001 * max_cbo, 0.0005 * max_cbo,
                              0.0002 * max_cbo, 0.0001 * max_cbo]
                    per_adam = phase5_end - time.monotonic() - 5
                    if per_adam > 15:
                        a_big_opt, _ = _adam_logspace(a_big_opt, time.monotonic() + per_adam,
                                                       bs_big, lr_init=0.005, max_iter_total=500000)
                        sc_bo2 = _score(a_big_opt)
                        if sc_bo2 < best_score:
                            best_score = sc_bo2
                            best_seq = a_big_opt.copy()
                            print(f"  Adam n=16384: score={sc_bo2:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"After Phase 5: score={best_score:.10f}")

    # === Phase 6: Final coordinate descent + ultra-fine Adam ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 30, time.monotonic() + 80)
        if cd_deadline > time.monotonic() + 10:
            a_cd, _ = _coord_descent(a_cd, cd_deadline, max_rounds=200)
            sc_cd = _score(a_cd)
            print(f"  Coord descent: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()

        # Ultra-fine Adam
        if time.monotonic() < deadline - 15:
            a_pol = best_seq.copy()
            a_pol = a_pol / a_pol.sum()
            c_pol = _convolve(a_pol, a_pol)
            max_cp = float(c_pol.max())
            betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000001 * max_cp,
                         0.0000005 * max_cp]
            per_t = deadline - time.monotonic() - 5
            if per_t > 10:
                a_pol, _ = _adam_logspace(a_pol, time.monotonic() + per_t,
                                           betas_ult, lr_init=0.002, max_iter_total=500000)
                sc_pol = _score(a_pol)
                print(f"  Final Adam: score={sc_pol:.10f} t={time.monotonic()-t_start:.1f}s")
                if sc_pol < best_score:
                    best_score = sc_pol
                    best_seq = a_pol.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
