```python
Multi-scale optimizer with top-k smoothing, diverse initializations,
and swap-based local search. Uses a bottom-up strategy from n=256 to n=8192.

import time

# Helper functions (top-level, no closures)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_full(a):
    """Compute the full objective score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))

    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)

    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Gradient of mean of top-k values of c w.r.t. a, where c=conv(a,a)."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)

        lo = max(0, k_lag - n + 1)
        hi = min(n - 1, k_lag)
        if lo <= hi:
            indices = k_lag - np.arange(lo, hi + 1)
            grad[lo:hi+1] += 2.0 * a[indices]

    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c = _full_convolve(a_curr, a_curr)
    c_max = float(c.max())
    # Lipschitz constant estimate for the top-k gradient
    L = 8.0 * n / k * step_scale  # rough estimate
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()

        else:

        c_y = _full_convolve(y, y)
        grad = _topk_grad(c_y, y, k)
        
        a_next = _proj_simplex(y - step * grad)
        
        # Backtracking if score increased too much
        if it > 10 and it % 20 == 0:
            sc = _score_norm(a_next)
            ref = _score_norm(a_curr)
            if sc > ref * 1.05:
                step *= 0.5
                a_next = _proj_simplex(y - step * grad)
            
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score
```

```python
"""
Multi-scale optimizer using top-k smoothing with provable Lipschitz constant,
FISTA with adaptive continuation, diverse initializations, Adam log-space
fine-tuning, and swap-based local search. Explores n=4096 and n=8192.
Key insight: the Hessian of c[k]=a^T Q_k a has operator norm 2, enabling
large FISTA steps (1/2). Uses top-k continuation for smooth-to-sharp transition.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _topk_grad_fft(c, a, k):
    n = len(a)
    k = min(k, 2 * n - 1)
    top_idx = np.argsort(c)[::-1][:k]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0
    N = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * g / k
    return grad, top_idx


def _fista_topk(a_init, k_list, deadline, max_iter_per_k=5000, step=0.45):
    n = len(a_init)
    a_curr = a_init.copy()
    s = a_curr.sum()
    if s > 1e-30:
        a_curr = a_curr / s
    else:
        a_curr = np.ones(n) / n

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for k_val in k_list:
        if time.monotonic() > deadline:
            break
        a_curr_f = a_curr.copy()
        a_prev = a_curr_f.copy()
        t_val = 1.0

        for it in range(max_iter_per_k):
            if time.monotonic() > deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr_f + ((t_val - 1.0) / t_next) * (a_curr_f - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            grad, _ = _topk_grad_fft(c_y, y, k_val)
            a_next = _proj_simplex(y - step * grad)

            if it % 200 == 0 or it < 3:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

            a_prev = a_curr_f.copy()
            a_curr_f = a_next
            t_val = t_next

        sc_end = _score_norm(a_curr_f)
        if sc_end < best_score:
            best_score = sc_end
            best_a = a_curr_f.copy()
        a_curr = a_curr_f

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_list, lr_init=0.01, max_iter=200000):

    a_norm = a_init.copy()
    s = a_norm.sum()

    best_score = _score_norm(a_norm)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

            c_shifted = (c - cm) / beta

            vh = v_buf / (1.0 - b2 ** t)

            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:

def _softmax_vec(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _softmax_grad(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _swap_local_search(a_init, n_trials, deadline):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 0:
        a = a / s
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Try transferring a small amount from i to j and vice versa
        delta = a[min(i, j)] * 0.1
        a_try1 = a.copy()
        a_try1[i] -= delta
        a_try1[j] += delta
        if a_try1[i] >= 0 and a_try1[j] >= 0:
            a_try1 = a_try1 / a_try1.sum()
            sc1 = _score_norm(a_try1)
            if sc1 < best_score:
                best_score = sc1
                best_a = a_try1.copy()
                a = a_try1
                continue

        delta2 = a[max(i, j)] * 0.1
        a_try2 = a.copy()
        a_try1[i] += delta2
        a_try1[j] -= delta2
        if a_try2[i] >= 0 and a_try2[j] >= 0:
            a_try2 = a_try2 / a_try2.sum()
            sc2 = _score_norm(a_try2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_try2.copy()
                a = a_try2

        # Also try swap
        if trial % 3 == 0:
            a_swap = a.copy()
            a_swap[i], a_swap[j] = a_swap[j], a_swap[i]
            a_swap = a_swap / a_swap.sum()
            sc_s = _score_norm(a_swap)
            if sc_s < best_score:
                best_score = sc_s
                best_a = a_swap.copy()
                a = a_swap

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_vec(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_full(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        best_seq = np.ones(n_parent) / n_parent
        best_score = _score_full(best_seq)
        print(f"No parent, using uniform n={n_parent}, score={best_score:.10f}")

    # === Phase 1: FISTA top-k continuation on parent ===
    print(f"\n=== Phase 1: FISTA on n={n_parent} ===")
    phase1_end = min(deadline, t_start + 250.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    k_list_p1 = [min(n_parent, 400), 200, 100, 50, 25, 10, 5, 3, 2, 1]
    a_opt, sc_opt = _fista_topk(a_work, k_list_p1, phase1_end, max_iter_per_k=3000, step=0.45)
    sc_full = _score_full(a_opt)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_opt.copy()
        print(f"  Phase 1 done: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Phase 1: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Try n=8192 ===
    print(f"\n=== Phase 2: n=8192 exploration ===")
    phase2_end = min(deadline, t_start + 550.0)
    n_big = 8192
    a_big = _interp_profile(best_seq, n_big)
    sc_big = _score_full(a_big)
    print(f"  Interpolated to n={n_big}: score={sc_big:.10f}")
    if sc_big < best_score:
        best_score = sc_big
        best_seq = a_big.copy()

    k_list_p2 = [min(n_big, 800), 400, 200, 100, 50, 25, 10, 5, 3, 2, 1]
    a_big_opt, sc_big_opt = _fista_topk(a_big, k_list_p2, phase2_end, max_iter_per_k=2000, step=0.45)
    sc_big_full = _score_full(a_big_opt)
    if sc_big_full < best_score:
        best_score = sc_big_full
        best_seq = a_big_opt.copy()
        print(f"  Phase 2 n=8192: score={sc_big_full:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Phase 2: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam fine-tuning on best ===
    print(f"\n=== Phase 3: Adam fine-tuning on n={len(best_seq)} ===")
    phase3_end = min(deadline, t_start + 900.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())
    betas = [0.001 * max_cw, 0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw, 0.00005 * max_cw, 0.00002 * max_cw]
    a_adam, sc_adam = _adam_logspace(a_work, phase3_end, betas, lr_init=0.008, max_iter=300000)
    sc_adam_full = _score_full(a_adam)
    if sc_adam_full < best_score:
        best_score = sc_adam_full
        best_seq = a_adam.copy()
        print(f"  Phase 3 Adam: score={sc_adam_full:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Phase 3: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse initializations ===
    print(f"\n=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1250.0)
    n_best = len(best_seq)
    a_ref = best_seq.copy()
    a_ref = a_ref / a_ref.sum()

    n_restarts = 6
    per_restart = max(5.0, (phase4_end - time.monotonic() - 20.0) / n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > phase4_end - 10:
            break

        r = rng.random()
        if r < 0.3:
            p_start = _perturb(a_ref, 0.02 + 0.01 * i, rng)
        elif r < 0.5:
            p_start = _perturb(a_ref, 0.05, rng)
        elif r < 0.7:
            # Blend with structured profile
            t_arr = np.linspace(0, 1, n_best)
            struct = np.sin(np.pi * t_arr) ** 0.5
            struct = np.maximum(struct, 1e-10)
            struct = struct / struct.sum()
            p_start = 0.4 * struct + 0.6 * a_ref
            p_start = np.maximum(p_start, 1e-12)
            p_start = p_start / p_start.sum()
        else:
            # Spectral flattening of perturbed
            p_tmp = _perturb(a_ref, 0.03, rng)
            N_fft = 2 * n_best - 1
            a_sf = p_tmp.copy()
            for _ in range(50):
                A = np.fft.rfft(a_sf, N_fft)
                mag = np.abs(A)
                avg_mag = mag.mean()
                A_flat = A / (mag + 1e-30) * avg_mag
                a_sf = np.fft.irfft(A_flat, N_fft)[:n_best]
                a_sf = np.maximum(a_sf, 0)
                s_sf = a_sf.sum()
                if s_sf < 1e-15:
                    break
                a_sf = a_sf / s_sf
            p_start = a_sf

        # Quick FISTA then check
        k_restart = [min(n_best, 50), 20, 10, 5, 3, 2, 1]
        p_end = min(time.monotonic() + per_restart * 0.7, phase4_end - 10)
        p_opt, _ = _fista_topk(p_start, k_restart, p_end, max_iter_per_k=1000, step=0.45)
        p_opt = p_opt / p_opt.sum()
        sc_r = _score_full(p_opt)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = p_opt.copy()
            print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Restart {i}: {sc_r:.10f} (no improve) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Swap-based local search ===
    print(f"\n=== Phase 5: Swap local search ===")
    phase5_end = min(deadline, t_start + 1500.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    n_trials = max(1000, int((phase5_end - time.monotonic()) * 30))
    a_ls, sc_ls = _swap_local_search(a_work, n_trials, phase5_end)
    sc_ls_full = _score_full(a_ls)
    if sc_ls_full < best_score:
        best_score = sc_ls_full
        best_seq = a_ls.copy()
        print(f"  Phase 5: score={sc_ls_full:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Phase 5: no improvement (best={best_score:.10f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine Adam polish ===
    print(f"\n=== Phase 6: Final polish ===")
    phase6_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase6_end - 20:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        c_final = _full_convolve(a_work, a_work)
        max_cf = float(c_final.max())
        betas_final = [0.00001 * max_cf, 0.000005 * max_cf, 0.000002 * max_cf, 0.000001 * max_cf]
        per_t = phase6_end - time.monotonic() - 10
        if per_t > 20:
            a_pol, sc_pol = _adam_logspace(a_work, phase6_end, betas_final, lr_init=0.003, max_iter=200000)
            sc_pol_full = _score_full(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Phase 6: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Phase 6: no improvement (best={best_score:.10f})")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = final.sum()
            final = final / s

        sc_final = _score_full(final)
        print(f"Final score (internal): {sc_final:.10f}")
        print(f"n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Final score (evaluator): {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(512)
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
```
