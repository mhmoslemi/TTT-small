    n = len(v)
    if n == 0:

    Uses FFT for O(n log n) computation.
    Returns a vector of length n.

    """Smoothed objective: 2n * LSE(c) using softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with adaptive log-sum-exp smoothing, active-set QP refinement via FISTA,
direct subgradient polishing, and multi-n exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hankel_mv(v, S_ind, n):
    """Compute (H_S * v)_{i} = sum_j S_ind[i+j] * v[j] for i in [0,n-1]."""
    N = _next_pow2(3 * n)
    A = S_ind
    B_flip = v[::-1]
    fa = np.fft.rfft(A, N)
    fb = np.fft.rfft(B_flip, N)
    conv = np.fft.irfft(fa * fb, N)
    return conv[n - 1:2 * n - 1]


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h = _hankel_mv(a, w, n)
    W = float(np.dot(w, c))
    g_a = 4.0 * n * a * (h - W)
    dot_ga_a = float(np.dot(g_a, a))
    g_z = a * (g_a - dot_ga_a)
    return g_z


def _gradient_direct(a, c, n, top_frac=0.01):
    """Subgradient using top fraction of active set."""
    c_max = float(c.max())
    n_top = max(1, int(len(c) * top_frac))
    top_idx = np.argsort(c)[::-1][:n_top]
    w = np.zeros_like(c)
    for idx in top_idx:
        w[idx] = 1.0 / n_top
    h = _hankel_mv(a, w, n)
    W = float(np.dot(w, c))
    g_a = 4.0 * n * a * (h - W)
    return g_a - np.mean(g_a)


def _subgrad_polish(a_init, deadline, n_max_steps=2000, lr_init=0.02, top_frac=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        g_a = _gradient_direct(a, c, n, top_frac)
        gnorm = np.linalg.norm(g_a)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * (g_a / a + np.mean(g_a / a))
        # Simpler: work in log-space directly
        # Actually let's just do projected gradient on a directly
        # Reset: use log-space
        break

    # Redo with proper log-space gradient
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        n_top = max(1, int(len(c) * top_frac))
        top_idx = np.argsort(c)[::-1][:n_top]
        w = np.zeros_like(c)
        for idx in top_idx:
            w[idx] = 1.0 / n_top
        h = _hankel_mv(a, w, n)
        W = float(np.dot(w, c))
        g_a = 4.0 * n * a * (h - W)
        dot_ga_a = float(np.dot(g_a, a))
        g_z = a * (g_a - dot_ga_a)
        gnorm = np.linalg.norm(g_z)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * g_z
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 20:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _active_set_fista(a_init, deadline, max_rounds=15, top_k_start=30, top_k_end=150):
    """Active-set QP refinement using FISTA."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    rng_local = np.random.default_rng(123)

    for round_idx in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())

        k = top_k_start + (top_k_end - top_k_start) * round_idx // max(1, max_rounds - 1)
        k = min(k, len(c))
        top_indices = np.argsort(c)[::-1][:k]
        S_ind = np.zeros(2 * n - 1, dtype=np.float64)
        S_ind[top_indices] = 1.0 / k

        # FISTA on: min a^T H_S a / 2 s.t. sum(a)=1, a>=0
        a_fista = a.copy()
        a_prev = a_fista.copy()
        t = 1.0

        # Estimate Lipschitz constant
        v = rng_local.standard_normal(n)
        v = v / np.linalg.norm(v)
        L_est = 1.0
        for _ in range(8):
            w_vec = _hankel_mv(v, S_ind, n)
            w_norm = np.linalg.norm(w_vec)
            if w_norm < 1e-15:
                break
            L_est = float(np.dot(v, w_vec / w_norm)) * 2.0
            v = w_vec / w_norm
        L = max(abs(L_est), 1e-6)
        step = 1.0 / L

        n_fista = 150
        for it in range(n_fista):
            if time.monotonic() > deadline:
                break
            if t > 1.0:
                y = a_prev + (t - 1.0) / (t + 1.0) * (a_prev - a_fista)
            else:
                y = a_prev.copy()
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y < 1e-15:
                y = np.ones(n) / n
            else:
                y = y / s_y

            grad = _hankel_mv(y, S_ind, n)
            grad_t = grad - np.mean(grad)

            a_new = _proj_simplex(y - step * grad_t, 1.0)

            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            a_prev = a_fista.copy()
            a_fista = a_new
            t = t_new

        sc = _score(a_fista)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a_fista.copy()
            a = a_fista.copy()

    return best_a, best_sc


class _LSEObj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def fun(self, z):
        return _smoobj(z, self.beta, self.n)

    def jac(self, z):
        return _smograd(z, self.beta, self.n)


def _run_lbfgs(a_init, deadline, min_time_remaining, n_beta=12):
    """Run L-BFGS with beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.008)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005])[:n_beta]

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.004 * max(1.0, n / 4096.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 10))))

        obj = _LSEObj(n, beta)
        try:
            result = minimize(
                obj.fun, z, jac=obj.jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.sin(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """Main search function: multi-strategy optimization."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # Phase 1: Main L-BFGS optimization
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=500, n_beta=12)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Phase 2: Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 400:
        print(f"\n[Phase 2] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
        a_sg, sc_sg = _subgrad_polish(best_seq, min(deadline - 400, time.monotonic() + 120),
                                       n_max_steps=3000, lr_init=0.015, top_frac=0.003)
        update_best(a_sg, sc_sg, "sg_polish_1")

    # Phase 3: Active-set FISTA refinement
    if best_seq is not None and time.monotonic() < deadline - 300:
        print(f"\n[Phase 3] Active-set FISTA (t={time.monotonic() - t_start:.1f}s)")
        a_fista, sc_fista = _active_set_fista(best_seq, min(deadline - 300, time.monotonic() + 180),
                                               max_rounds=15, top_k_start=20, top_k_end=100)
        update_best(a_fista, sc_fista, "fista_active")

    # Phase 4: L-BFGS after active-set refinement
    if best_seq is not None and time.monotonic() < deadline - 200:
        print(f"\n[Phase 4] L-BFGS post-refinement (t={time.monotonic() - t_start:.1f}s)")
        a_post = _run_lbfgs(best_seq, deadline, min_time_remaining=200, n_beta=8)
        update_best(a_post, _score(a_post), "lbfgs_post")

    # Phase 5: Multi-n exploration
    print(f"\n[Phase 5] Multi-n (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [3072, 4096, 5120, 6144]
    if n_parent in n_candidates:
        n_candidates.remove(n_parent)
    n_candidates.insert(0, n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        if n_try == n_parent and best_seq is not None:
            a_init = best_seq.copy()
        elif parent is not None:
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s < 1e-10:
                a_init = _make_profile(n_try, "tapered")
            else:
                a_init = a_init / s
        else:
            a_init = _make_profile(n_try, "tapered")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=100, n_beta=8)
        update_best(a_opt, _score(a_opt), f"n_{n_try}")

        # Quick FISTA
        if time.monotonic() < deadline - 60:
            a_f, sc_f = _active_set_fista(a_opt, min(deadline - 60, time.monotonic() + 60),
                                           max_rounds=8, top_k_start=20, top_k_end=60)
            update_best(a_f, sc_f, f"n_{n_try}_fista")

                    i2 = rng.integers(0, n - 2 * bs)
