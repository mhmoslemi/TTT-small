For coordinate descent on the objective f(a) = 2n * max_k (a*a)_k / (sum a)^2:

If I fix all coordinates except a_i and vary a_i, the objective becomes:
f(x) = 2n * max(ph(x), max_inactive) / (S + x - a_i_old)^2

where ph(x) = max_k (a*a)_k as a function of x = a_i, and max_inactive is the max over convolution terms that don't depend on a_i.

The terms that depend on a_i are (a*a)_k for k = i, i+1, ..., i+n-1 (where the term a_i * a_{k-i} appears).

For k in [i, i+n-1]:
- If k = 2i: (a*a)_k = c_k + x^2 (where c_k = (a*a)_k - x^2 evaluated at current x)
- If k ≠ 2i: (a*a)_k = c_k + 2x * a_{k-i}

So ph(x) = max(max_inactive, max_{k≠2i} (c_k + 2x*a_{k-i}), c_{2i} + x^2)

This is the max of linear and quadratic functions, which is convex in x (for x >= 0). The overall objective f(x) = 2n * ph(x) / (S')^2 where S' = (sum a) - a_i_old + x.

Since ph(x) is convex and (S')^2 is convex and increasing in x (for x > 0), the ratio is not necessarily convex. But it's a 1D problem, so we can just scan or use a numerical 1D optimizer.

For the coordinate descent polish:
- For each coordinate i (randomly permuted):
  - Compute the current values c_k for all k in [i, i+n-1]
  - Find the optimal x >= 0 that minimizes f(x) (using a 1D search)
  - Update a_i = x

"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive FISTA with backtracking, top-K subgradients,
coordinate descent polish, and multi-length exploration.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _get_smooth_weights(c, beta):
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _get_topk_weights(c, k, gamma):
    c_max = float(c.max())
    if gamma <= 0:
        active = np.where(c >= c_max - 1e-12)[0]
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        return w
    shifted = (c - c_max) / gamma
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    if k is not None and k < len(c):
        top_idx = np.argsort(c)[::-1][:k]
        w_masked = w.copy()
        w_masked[~np.isin(np.arange(len(c)), top_idx)] = 0.0
        s2 = float(w_masked.sum())
        if s2 > 1e-300:
            w = w_masked / s2
        # else keep original
    return w


def _fista_adaptive(a_init, beta, deadline, max_iter, use_backtrack=True):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = beta / (4.0 * n)
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    eval_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _get_smooth_weights(c_y, beta)
        grad = _compute_grad(y, w)

        a_next = _proj_simplex(y - step * grad)

        if use_backtrack and it < 500:
            # Simple backtracking: check if the smoothed objective decreased
            f_curr = float(c_y.max())
            c_next = _full_convolve(a_next, a_next)
            f_next = float(c_next.max())
            if f_next > f_curr * 1.01:
                step *= 0.7
                a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            eval_count += 1

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _subgradient_method(a_init, deadline, gamma_init, n_phases=12):
    """Exact subgradient method with decreasing gamma (top-K selection)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    gamma = gamma_init

    for phase in range(n_phases):
        phase_deadline = deadline - (n_phases - 1 - phase) * (deadline - time.monotonic()) / n_phases
        if time.monotonic() > phase_deadline:
            break
        it = 0
        max_iter_phase = 200000
        while time.monotonic() < phase_deadline and it < max_iter_phase:
            c = _full_convolve(a, a)
            w = _get_topk_weights(c, None, gamma)
            grad = _compute_grad(a, w)
            lr = 1.0 / (np.sqrt(n) * np.sqrt(it + 1) * 2.0)
            a_new = _proj_simplex(a - lr * grad)
            if it % 500 == 0:
                sc = _score_norm(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
            a = a_new
            it += 1
        gamma *= 0.5
        if gamma < 1e-15:
            gamma = 1e-15

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 20 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.93)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_polish(a_init, deadline, n_passes=10):
    """Coordinate descent polish: optimize one coordinate at a time."""
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    best_score = _score_norm(a)
    best_a = a.copy()
    c = _full_convolve(a, a)
    c_max = float(c.max())

    for pst in range(n_passes):
        if time.monotonic() > deadline:
            break
        perm = np.random.permutation(n)
        for i in perm:
            # Current value
            x_old = a[i]
            # Convolution terms depending on x: k in [i, i+n-1]
            # (a*a)_k for k = 2i: c_k + x^2
            # (a*a)_k for k != 2i: c_k + 2*x*a_{k-i}
            # We need to find x >= 0 minimizing 2n * max(...) / (sum)^2

            # Compute base convolution contributions (without a_i's effect)
            k_range = range(max(0, i), min(2 * n - 1, i + n))
            # For each k in range, compute the coefficient of x and x^2
            # (a*a)_k = base_k + linear_k * x + quad_k * x^2
            # quad_k = 1 if k == 2i, else 0
            # linear_k = 2*a_{k-i} if k != 2i, else 0
            # base_k = c_k - linear_k * x_old - quad_k * x_old^2

            sum_a = float(a.sum())
            best_x = x_old
            best_val = np.inf

            # Try a few values of x
            # The optimal x is likely near x_old, so try perturbations
            candidates = [x_old]
            if x_old > 0:
                for frac in [0.5, 0.75, 1.25, 1.5, 0.25, 2.0]:
                    candidates.append(x_old * frac)
            candidates.append(0.0)
            candidates.append(x_old + 0.01 / n)
            candidates.append(max(0, x_old - 0.01 / n))

            for x_trial in candidates:
                if x_trial < 0:
                    continue
                new_sum = sum_a - x_old + x_trial
                if new_sum < 1e-15:
                    continue
                # Compute new max convolution
                new_max = c_max
                for k in k_range:
                    if k == 2 * i:
                        val = (c[k] - x_old * x_old) + x_trial * x_trial
                    elif k - i < n:
                        val = (c[k] - 2.0 * x_old * a[k - i]) + 2.0 * x_trial * a[k - i]
                    else:
                        val = c[k]
                    if val > new_max:
                        new_max = val
                score_trial = 2.0 * n * new_max / (new_sum * new_sum)
                if score_trial < best_val:
                    best_val = score_trial
                    best_x = x_trial

            if best_x != x_old:
                a[i] = best_x
                # Update convolution
                c = _full_convolve(a, a)
                c_max = float(c.max())
                sc = 2.0 * n * c_max / (float(a.sum()) ** 2)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 40)), 'gtol': 1e-13, 'maxls': 15}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic_phase":
        t = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "banded":
        band_w = max(1, n // 8)
        a = np.zeros(n)
        for start in range(0, n, band_w * 2):
            end = min(start + band_w, n)
            t = np.linspace(0, 1, end - start)
            a[start:end] = 0.5 + 0.5 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _resize_sequence(a, new_n):
    """Resize sequence using interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_t = np.linspace(0, 1, old_n)
    new_t = np.linspace(0, 1, new_n)
    a_resized = np.interp(new_t, old_t, a)
    a_resized = np.maximum(a_resized, 0.0)
    s = a_resized.sum()
    if s > 1e-15:
        a_resized /= s
    else:
        a_resized = np.ones(new_n) / new_n
    return a_resized

            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)
        a_work = a_lb

    # === Phase 5: Coordinate descent polish ===
    print("Phase 5: Coordinate descent polish", flush=True)
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end - 10 and best_seq is not None:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        a_pol, _ = _coordinate_polish(a_pol, phase5_end, n_passes=5)
        sc_cd = _score(a_pol)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_pol.copy()
            a_work = a_pol
            print(f"  CoordDescent: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Try n=8192 ===
    print("Phase 6: Try n=8192", flush=True)
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end - 20 and best_seq is not None:
        a_8192 = _resize_sequence(best_seq, 8192)
        sc_8192 = _score(a_8192)
        print(f"  Resized to 8192: score={sc_8192:.10f}", flush=True)
        if sc_8192 < best_score:
            best_score = sc_8192
            best_seq = a_8192.copy()
        # Quick FISTA on the 8192 sequence
        if time.monotonic() < phase6_end - 15:
            c_8k = _full_convolve(a_8192, a_8192)
            b_8k = 0.005 * float(c_8k.max())
            a_8192, _ = _fista_adaptive(a_8192, b_8k, time.monotonic() + 40.0, 20000)
            a_8192 = a_8192 / a_8192.sum()
            sc_8k = _score(a_8192)
            if sc_8k < best_score:
                best_score = sc_8k
                best_seq = a_8192.copy()
                print(f"  8192 FISTA: score={sc_8k:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 7: Multi-restart exploration ===
    print("Phase 7: Multi-restart", flush=True)
    phase7_end = min(deadline, t_start + 1730.0)
    if time.monotonic() < phase7_end - 15 and best_seq is not None:
        n_restarts = 3
        per_r = (phase7_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            if i == 0:
                p = _perturb(best_seq, 0.01, rng)
            elif i == 1:
                p = _make_profile(n_main, "quadratic_phase")
                p = 0.7 * best_seq + 0.3 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            else:
                p = _perturb(best_seq, 0.02, rng)
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.0002 * max_cp
            p, _ = _fista_adaptive(p, b_r, time.monotonic() + per_r * 0.4, 30000)
            p, _ = _adam_opt(p, time.monotonic() + per_r * 0.6, b_r * 0.3, 0.004, 60000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 8: Final ultra-polish ===
    print("Phase 8: Final ultra-polish", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 8
        if per_t > 10:
            # Adam with tiny beta
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.5,
                                 0.000003 * max_cp, 0.001, 150000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
            # L-BFGS on remaining
            if time.monotonic() < deadline - 10:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000002 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 5.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}", flush=True)
            # Coord polish
            if time.monotonic() < deadline - 6:
                a_pol = best_seq.copy() / best_seq.sum()
                a_pol, _ = _coordinate_polish(a_pol, deadline - 3.0, n_passes=3)
                sc_cp = _score(a_pol)
                if sc_cp < best_score:
                    best_score = sc_cp
                    best_seq = a_pol.copy()
                    print(f"  Final CoordPolish: score={sc_cp:.10f}", flush=True)

    print(f"Done: best_score={best_score:.10f}", flush=True)
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"Eval error: {ex}", flush=True)
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
