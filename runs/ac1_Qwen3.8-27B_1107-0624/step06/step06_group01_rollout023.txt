"""
Active-set LP and gradient-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding on the active set of convolution peaks,
followed by fine gradient polishing and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_at_k(a, k):
    """Compute gradient of c_k = (a*a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    # g_j = 2*a_{k-j} if 0 <= k-j <= n-1
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        g[j_min:j_max+1] = 2.0 * a[k - j_max:k - j_min + 1][::-1]
    return g

def _lp_direction(a, c, active_indices):
    """
    Solve LP to find a direction d that reduces the active convolution values.
    min t s.t. c_k + grad_k . d <= t for k in active, d >= -a, sum(d) = 0
    Returns (direction, t_value) or (None, None) if infeasible.
    """
    n = len(a)
    m = len(active_indices)
    
    # Variables: d (n vars) + t (1 var) = n+1 vars
    # Objective: minimize t -> c_obj = [0,...,0,1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ineq @ x <= b_ineq
    # For each k in active: sum_j (2*a_{k-j}) d_j - t <= -c_k
    # Plus: -d_j <= a_j (i.e., d_j >= -a_j)
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for idx, k in enumerate(active_indices):
        # Row for constraint k
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            for j in range(j_min, j_max + 1):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
        # t coefficient: -1
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)
    
    # d_j >= -a_j  =>  -d_j <= a_j
    for j in range(n):
        rows.append(m + j)
        cols.append(j)
        vals.append(-1.0)
    
    A_ineq = sparse.csr_matrix((vals, (rows, cols)), shape=(m + n, n + 1))
    
    b_ineq = np.zeros(m + n)
    for idx, k in enumerate(active_indices):
        b_ineq[idx] = -c[k]
    b_ineq[m:] = a
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            d = res.x[:n]
            t_val = res.x[n]
            return d, t_val
        else:
            return None, None
    except Exception:
        return None, None

def _gradient_descent_step(a, c, active_fractions, step_scale):
    """
    Take a gradient descent step using weighted average gradient over active set.
    active_fractions: list of (k, weight) tuples
    """
    n = len(a)
    g = np.zeros(n)
    total_w = 0.0
    for k, w in active_fractions:
        g += w * _compute_grad_at_k(a, k)
        total_w += w
    if total_w > 0:
        g /= total_w
    return g

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adam_fine_polish(a_init, deadline, lr_init=0.005):
    """Fine Adam optimization in log-space with very small beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    while time.monotonic() < deadline:
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Use very sharp softmax (essentially exact max gradient averaged over top peaks)
        top_k = max(1, int(0.01 * (2 * n - 1)))
        sorted_idx = np.argsort(c)[::-1][:top_k]
        weights = np.zeros(2 * n - 1)
        weights[sorted_idx] = np.exp((c[sorted_idx] - cm) / (0.001 * cm))
        weights /= weights.sum()
        
        # Compute gradient as weighted average
        g = np.zeros(n)
        for idx in sorted_idx:
            if weights[idx] > 1e-10:
                g += weights[idx] * _compute_grad_at_k(a, int(idx))
        
        m_buf = b1 * m_buf + (1.0 - b1) * g
        v_buf = b2 * v_buf + (1.0 - b2) * (g * g)
        mh = m_buf / (1.0 - b1 ** (t + 1))
        vh = v_buf / (1.0 - b2 ** (t + 1))
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        z = z - update
        z = np.clip(z, -50, 50)

        t += 1

"""
Active-set LP and multi-phase gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding on convolution peaks, momentum-accelerated
subgradient descent, targeted coordinate descent, and pairwise mass transfer.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_at_k(a, k):
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        g[indices] = 2.0 * a[k - indices][::-1]
    return g


def _compute_weighted_grad(a, c, top_m, temperature=None):
    n = len(a)
    num_c = len(c)
    if temperature is None:
        temperature = 0.005 * c.max()
    cm = c.max()
    shifted = (c - cm) / temperature
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    # Focus on top entries
    g = np.zeros(n)
    top_idx = np.argsort(w)[::-1][:top_m]
    for idx in top_idx:
        if w[idx] > 1e-15:
            g += w[idx] * _compute_grad_at_k(a, int(idx))
    return g


def _lp_direction(a, c, active_indices, time_limit=5.0):
    n = len(a)
    m = len(active_indices)
    if m == 0:
        return None, None
    if m > 100:
        order = np.argsort(c[active_indices])[::-1][:100]
        active_indices = active_indices[order]
        m = 100

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    rows = []
    cols = []
    vals = []

    for idx, k in enumerate(active_indices):
        k = int(k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            indices = np.arange(j_min, j_max + 1)
            for pos, j in enumerate(indices):
                rows.append(idx)
                cols.append(j)
                vals.append(2.0 * a[k - j])
        rows.append(idx)
        cols.append(n)
        vals.append(-1.0)

    for j in range(n):
        rows.append(m + j)
        cols.append(j)
        vals.append(-1.0)

    A_ineq = sparse.csr_matrix((vals, (rows, cols)), shape=(m + n, n + 1))
    b_ineq = np.zeros(m + n)
    for idx, k in enumerate(active_indices):
        b_ineq[idx] = -c[int(k)]
    b_ineq[m:] = a

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    try:
        res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            d = res.x[:n]
            t_val = res.x[n]
            return d, t_val
    except Exception:
        pass
    return None, None


def _fista_active(a_init, deadline, top_m=50, temperature_frac=0.01):
    """FISTA with active-set gradient."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    momentum = 0.9
    lr = 0.5

    for it in range(100000):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        cm = float(c_y.max())
        temperature = temperature_frac * cm

        g = _compute_weighted_grad(y, c_y, top_m, temperature)

        # Backtracking line search
        step = lr
        a_next = None
        for _ in range(8):
            cand = y - step * g
            cand = _proj_simplex(cand)
            c_cand = _full_convolve(cand, cand)
            if c_cand.max() <= cm + 0.5 * step * step * np.dot(g, g):
                a_next = cand
                break
            step *= 0.5
        if a_next is None:
            a_next = _proj_simplex(y - step * g)

        sc = _score_norm(a_next)
        if sc < best_sc:
            best_sc = sc
            best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

        if it % 500 == 0:
            lr *= 0.999

    return best_a, best_sc


def _adam_fine(a_init, deadline, lr_init=0.002, top_frac=0.02):
    """Adam in log-space with very sharp softmax on the max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    top_m = max(5, int(top_frac * (2 * n - 1)))

    while time.monotonic() < deadline:
        a = _softmax_local(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        temperature = 0.002 * cm

        g = _compute_weighted_grad(a, c, top_m, temperature)
        if not np.all(np.isfinite(g)):
            break

        t += 1
        m_buf = b1 * m_buf + (1.0 - b1) * g
        v_buf = b2 * v_buf + (1.0 - b2) * (g * g)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 200 == 0:
            a_chk = _softmax_local(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()

        if t % 8000 == 0:
            current_lr = max(0.0001, current_lr * 0.9)

    a_final = _softmax_local(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _softmax_local(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _coordinate_polish(a, deadline, rng):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)

    c = _full_convolve(best_a, best_a)
    peak_k = int(np.argmax(c))

    # Rank coordinates by their contribution to the peak
    contrib = np.zeros(n)
    j_min = max(0, peak_k - (n - 1))
    j_max = min(n - 1, peak_k)
    if j_min <= j_max:
        indices = np.arange(j_min, j_max + 1)
        contrib[indices] = best_a[peak_k - indices]

    # Priority list: high contribution first, then random
    priority = np.argsort(contrib)[::-1]

    factors = [0.9, 0.95, 1.05, 1.1, 0.85, 1.15]

    it = 0
    while time.monotonic() < deadline:
        # Cycle through priority list
        i = int(priority[it % n])
        for f in factors:
            a_try = best_a.copy()
            a_try[i] *= f
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try /= s
                sc = _score_norm(a_try)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a_try.copy()
                    break
        it += 1
        # Occasionally try random coordinate
        if rng.random() < 0.05:
            i = int(rng.integers(0, n))
            for f in factors:
                a_try = best_a.copy()
                a_try[i] *= f
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try /= s
                    sc = _score_norm(a_try)
                    if sc < best_sc - 1e-15:
                        best_sc = sc
                        best_a = a_try.copy()
                        break

    return best_a, best_sc


def _pairwise_transfer(a, deadline, rng):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)

    c = _full_convolve(best_a, best_a)
    peak_k = int(np.argmax(c))

    # Important coordinates
    j_min = max(0, peak_k - (n - 1))
    j_max = min(n - 1, peak_k)
    important = list(range(j_min, j_max + 1))

    fracs = [0.05, 0.1, 0.15, 0.2, -0.05, -0.1, -0.15, -0.2]

    while time.monotonic() < deadline:
        if len(important) >= 2:
            i = important[int(rng.integers(0, len(important)))]
        else:
            i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        base = min(best_a[i], best_a[j])
        if base < 1e-15:
            continue
        for f in fracs:
            amt = f * base
            a_try = best_a.copy()
            a_try[i] -= amt
            a_try[j] += amt
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try /= s
                sc = _score_norm(a_try)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a_try.copy()
                    break

    return best_a, best_sc


def _block_descent(a, deadline, rng, block_size=32):
    """Optimize blocks of coordinates at a time."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)

    n_blocks = (n + block_size - 1) // block_size

    for _ in range(50):
        if time.monotonic() > deadline:
            break
        for b in range(n_blocks):
            if time.monotonic() > deadline:
                break
            start = b * block_size
            end = min(start + block_size, n)
            a_try = best_a.copy()
            # Optimize this block with a few Adam steps
            z_block = np.log(np.maximum(a_try[start:end], 1e-15))
            for _ in range(20):
                a_full = best_a.copy()
                a_full[start:end] = _softmax_local(z_block) * best_a[start:end].sum()
                a_full = np.maximum(a_full, 0)
                s = a_full.sum()
                if s > 1e-15:
                    a_full /= s
                    c = _full_convolve(a_full, a_full)
                    cm = c.max()
                    temp = 0.002 * cm
                    g_full = _compute_weighted_grad(a_full, c, 30, temp)
                    g_b = g_full[start:end]
                    z_block = z_block - 0.01 * g_b
                    z_block = np.clip(z_block, -30, 30)
            a_try[start:end] = _softmax_local(z_block) * best_a[start:end].sum()
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s > 1e-15:
                a_try /= s
                sc = _score_norm(a_try)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a_try.copy()

    return best_a, best_sc
