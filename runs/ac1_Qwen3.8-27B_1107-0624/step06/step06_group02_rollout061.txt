```python
def spectral_flatten(a, n_iter=1000):
    n = len(a)
    a = a.copy()
    A = np.fft.rfft(a)
    for i in range(n_iter):
        # Compute current power spectrum
        P = np.abs(A)**2
        # Flatten: reduce the non-DC components towards the average
        # Keep DC fixed
        P_non_dc = P[1:]
        avg_non_dc = P_non_dc.mean()
        # Blend: reduce peaks, boost valleys
        P_new = P.copy()
        P_new[1:] = 0.5 * P[1:] + 0.5 * avg_non_dc
        # Inverse: get new A
        phase = np.angle(A)
        A_new = np.sqrt(np.maximum(P_new, 0)) * np.exp(1j * phase)
        # IFFT
        a_new = np.fft.irfft(A_new, n)
        # Project: clip to non-negative
        a_new = np.maximum(a_new, 0)
        # Normalize
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        a = a_new
        A = np.fft.rfft(a)
    return a
```

```python
    """Nesterov-accelerated gradient on power-mean smoothed objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    # Compute initial gradient and Lipschitz estimate
    N_fft = _next_pow2(3 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y

        c = _convolve(y, y)
        c = np.maximum(c, 0.0)

            g = g * scale
```

```python
"""
Multi-scale optimizer for minimum peak autocorrelation non-negative sequences.
Uses Nesterov-accelerated proximal gradient with power-mean smoothing,
spectral domain flattening, coordinate descent with adaptive line search,
and multi-scale upsampling (4096 -> 8192 -> 16384).
Targets score < 1.5027.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_grad(a, p, N_fft):
    n = len(a)
    c = _convolve(a, a)
    c = np.maximum(c, 1e-300)
    w = np.power(c, p - 1)
    fw = np.fft.rfft(w, N_fft)
    fa = np.fft.rfft(a, N_fft)
    g_full = np.fft.irfft(np.conj(fw) * fa, N_fft)
    g = g_full[:n]
    c_power_sum = float(np.sum(np.power(c, p)))
    if c_power_sum > 1e-300:
        scale = c_power_sum ** (1.0 / p - 1.0)
        g = g * scale
    return g, c


def _nesterov_smooth(a_init, p_list, deadline, lr_init=0.05):
    n = len(a_init)
    N_fft = _next_pow2(3 * n)
    a_curr = a_init.copy()
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for p in p_list:
        if time.monotonic() > deadline:
            break
        a_prev = a_curr.copy()
        t_val = 1.0
        lr = lr_init

        it = 0
        while time.monotonic() < deadline:
            it += 1
            if it > 50000:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y

            g, c = _smoothed_grad(y, p, N_fft)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-300:
                break

            step = lr / g_norm
            a_cand = _proj_simplex(y - step * g)

            if it % 200 == 0:
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_cand.copy()

            a_prev = a_curr
            a_curr = a_cand
            t_val = t_next

                lr *= 0.85

def _spectral_flatten(a, n_iter=200, blend=0.3):
    n = len(a)
    a = a.copy()
    for i in range(n_iter):
        A = np.fft.rfft(a)
        P = np.abs(A) ** 2
        P_non_dc = P[1:]
        avg = float(P_non_dc.mean())
        P_new = P.copy()
        P_new[1:] = (1.0 - blend) * P[1:] + blend * avg
        P_new = np.maximum(P_new, 0.0)
        mag = np.sqrt(P_new)
        phase = np.angle(A)
        A_new = mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-30:
            a_new = a_new / s
        else:
            break
        a = a_new
    return a

    return p_new / p_new.sum()

            end_r = time.monotonic() + max(5.0, per_restart * 0.6)
            p_opt, sc_r = _nesterov_smooth(p_start, p_list_r, end_r, lr_init=0.03)
            p_opt = p_opt / p_opt.sum()

            # Quick CD
            if time.monotonic() < p5_end - 20:
                p_opt, _ = _coord_descent(p_opt, min(p5_end - 15, time.monotonic() + per_restart * 0.3),
                                          max_rounds=20, subset_size=min(128, len(p_opt) // 4))
                p_opt = p_opt / p_opt.sum()

            sc = _score(p_opt)
            if sc < best_score:
                best_score = sc
                best_seq = p_opt.copy()
                print(f"  Restart {i}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final polish ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        # Spectral
        a_pol = _spectral_flatten(a_pol, n_iter=30, blend=0.05)
        sc = _score(a_pol)
        if sc < best_score:
            best_score = sc
            best_seq = a_pol.copy()

        # Nesterov fine
        if time.monotonic() < deadline - 20:
            p_list_fin = [50, 100, 200]
            a_pol, _ = _nesterov_smooth(a_pol, p_list_fin, deadline - 15, lr_init=0.02)
            a_pol = a_pol / a_pol.sum()
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()

        # CD
        if time.monotonic() < deadline - 10:
            a_pol, _ = _coord_descent(a_pol, deadline - 5, max_rounds=30,
                                      subset_size=min(256, len(a_pol) // 4))
            a_pol = a_pol / a_pol.sum()
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()

        print(f"  Final polish: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
