from scipy.sparse import csr_matrix

def solve_primal_lp(a, M):
    n = len(a)
    m = 2 * n - 1  # number of constraints
    
    # Build C as sparse matrix
    # C[j, i] = a[i] if 0 <= j-i <= n-1
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for j in range(i, min(i + n, m)):
            rows.append(j)
            cols.append(i)
            vals.append(a[i])
    
    C = csr_matrix((vals, (rows, cols)), shape=(m, n))
    
    # max 1^T g s.t. C*g <= M*1, g >= 0
    # linprog minimizes, so: min -1^T g s.t. C*g <= M*1, g >= 0
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(m, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return None, 0
    
    g = res.x
    S = float(res.fun)  # this is -sum(g), so sum(g) = -res.fun
    return g, -float(res.fun)

```python
def solve_active_set_lp(a, M, K=100, max_iter=20):
    n = len(a)
    m = 2 * n - 1
    c = np.convolve(a, a)
    
    # Start with top K positions
    active = np.argsort(c)[-K:]
    
    for iteration in range(max_iter):
        # Solve LP with active constraints
        # max sum(g) s.t. (a*g)_j <= M for j in active, g >= 0
        # (a*g)_j = sum_{k} a_k * g_{j-k}
        
        # Build constraint matrix for active set
        A = np.zeros((len(active), n))
        for idx, j in enumerate(active):
            for k in range(n):
                if 0 <= j - k < m and 0 <= k < n:
                    A[idx, k] = a[k]
        
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(len(active), M), 
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        g = res.x
        S = -res.fun
        
        # Check if any constraint is violated
        conv_g = np.convolve(a, g)
        violated = np.where(conv_g > M + 1e-10)[0]
        
        if len(violated) == 0:
            return g, S
        else:
            # Add violated constraints to active set
            new_active = np.unique(np.concatenate([active, violated]))
            active = new_active[-min(len(new_active), 2*K):]  # Keep top 2K
    
    return g, S
```

LP-guided optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses active-set linear programming to find optimal descent directions at each step,
combined with Adam in log-space for initial refinement and coordinate descent for final polishing.
The LP approach directly optimizes the max-convolution constraint, providing exact
descent directions rather than the approximate gradients used in smoothed approaches.

from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a) = 1: 2n * max(a*a)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Full score: 2n * max(a*a) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _solve_active_set_lp(a, M, K=100, deadline=None):
    """Solve the active-set LP: max sum(g) s.t. (a*g)_j <= M for j in active set, g >= 0.
    Returns (g, S) where S = sum(g), or (None, 0) on failure.
    """
    n = len(a)
    m = 2 * n - 1
    c = _full_convolve(a, a)
    
    # Start with top K positions
    active = np.argsort(c)[-min(K, m):]
    
    best_g = None
    best_S = 0.0
    
    for iteration in range(30):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        ka = len(active)
        
        # Build constraint matrix: (a*g)_j for j in active
        # (a*g)_j = sum_{k=0}^{n-1} a_k * g_{j-k} (where 0 <= j-k < n)
        A = np.zeros((ka, n))
        for idx, j in enumerate(active):
            # g indices: j-k for k=0..n-1, valid when 0 <= j-k < n, i.e., max(0,j-n+1) <= k <= min(j, n-1)
            k_lo = max(0, j - n + 1)
            k_hi = min(j, n - 1)
            if k_lo <= k_hi:
                A[idx, k_lo:k_hi+1] = a[k_hi:k_lo-1:-1]  # a_{j-k} for k = k_lo..k_hi
        
        # Solve: max 1^T g s.t. A*g <= M*1, g >= 0
        # linprog: min -1^T g s.t. A*g <= M*1, g >= 0
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(ka, M),
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            if best_g is not None:
                return best_g, best_S
            return None, 0

        S = -float(res.fun)

            best_g = g.copy()
        
        # Check if all constraints are satisfied
        conv_g = _full_convolve(a, g)
        max_violation = float(conv_g.max())

        # Add violated constraints

            order = np.argsort(c[new_active])[::-1][:K]
            active = new_active[order]
        else:
            active = new_active
    
    return best_g, best_S

LP-guided optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: uses active-set linear programming to find exact descent directions that
directly optimize the max-convolution constraint, rather than relying on smoothed gradient
approximations. Combined with Adam in log-space for initialization and coordinate descent
for final polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_constraint_matrix(a, active_indices, n):
    """Build the constraint matrix A where A[idx, m] = a[j-m] for valid m."""
    ka = len(active_indices)
    A = np.zeros((ka, n))
    for idx, j in enumerate(active_indices):
        m_lo = max(0, j - n + 1)
        m_hi = min(j, n - 1)
        if m_lo <= m_hi:
            A[idx, m_lo:m_hi + 1] = a[j - np.arange(m_lo, m_hi + 1)]
    return A


def _solve_active_set_lp(a, M, K=100, deadline=None, max_active_set_grows=5):
    """Solve active-set LP: max sum(g) s.t. (a*g)_j <= M for active j, g >= 0.
    Returns (g, S) or (None, 0)."""
    n = len(a)
    m = 2 * n - 1
    c = _full_convolve(a, a)

    active = np.argsort(c)[-min(K, m):]
    best_g = None
    best_S = 0.0

    for iteration in range(max_active_set_grows):
        if deadline is not None and time.monotonic() > deadline:
            break

        ka = len(active)
        A = _build_constraint_matrix(a, active, n)

        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(ka, M),
                      bounds=[(0, None)] * n, method='highs')

        if not res.success:
            if best_g is not None:
                return best_g, best_S
            return None, 0

        g = res.x
        S = -float(res.fun)

        if S > best_S:
            best_S = S
            best_g = g.copy()

        conv_g = _full_convolve(a, g)
        max_viol = float(np.max(conv_g))

        if max_viol <= M * (1.0 + 1e-9):
            return g, S

        violated = np.where(conv_g > M * (1.0 + 1e-9))[0]
        new_active = np.unique(np.concatenate([active, violated]))
        if len(new_active) > 2 * K:
            order = np.argsort(c[new_active])[::-1][:K]
            active = new_active[order]
        else:
            active = new_active

    return best_g, best_S


def _lp_optimize(a_init, deadline, K=100, t_init=0.25, stall_limit=3000):
    """Main LP-based iterative optimizer. Returns (best_a, best_score)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = _score_norm(a)
    best_a = a.copy()
    t_step = t_init
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        M = float(_full_convolve(a, a).max())

        g, S = _solve_active_set_lp(a, M, K=K, deadline=deadline)
        if g is None or S <= 1.0 + 1e-8:
            stall += 1
            if stall > stall_limit:
                break
            if stall % 500 == 0:
                t_step = max(0.01, t_step * 0.8)
            continue

        stall = 0
        g = g / S  # normalize to sum=1

        # Try a few step sizes
        best_trial_score = np.inf
        best_trial_a = None
        for t in [t_step, t_step * 0.5, t_step * 1.5]:
            t = min(max(t, 0.01), 0.8)
            a_trial = (1.0 - t) * a + t * g
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            sc = _score_norm(a_trial)
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial

        if best_trial_a is not None and best_trial_score < best_score:
            a = best_trial_a
            best_score = best_trial_score
            best_a = a.copy()
            t_step = min(0.8, t_step * 1.05)
        else:
            t_step = max(0.01, t_step * 0.7)
            stall += 1
            if stall > stall_limit:
                break

        if it % 2000 == 0:
            print(f"    LP iter {it}: score={best_score:.10f}, S={S:.6f}, t={t_step:.4f}")

    return best_a, best_score


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, c, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, betas, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, c, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.98)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent polishing."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break

            c = _full_convolve(a, a)
            a_i_old = a[i]

            if a_i_old < 1e-15:
                a[i] = 1e-4
                a = a / a.sum()
                continue

            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old

            idx_top = np.argsort(c)[-top_k_w:]

            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5,
                           a_i_old * 0.3, a_i_old * 0.7, a_i_old * 1.2, a_i_old * 2.0]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 5.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)
                        t_candidates.append(t_star * 0.8)
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(5.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 10) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            a[i] = best_t
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc

    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n + 0.3 * freq)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "multi_freq":
        a = np.ones(n)
        rng = np.random.default_rng(123)
        for _ in range(8):
            freq = rng.uniform(0.5, 10.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.1, 0.4)
            a += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
        a = np.maximum(a, 0.005)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

                parent = parent_raw / s
    except Exception:

    if parent is not None:

        sc_parent = _score_full(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()

    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score_full(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try lengths: focus on 4096 (best from previous), also try 8192
    n_target = len(parent) if parent is not None else 4096
    lengths_to_try = [n_target]
    if n_target == 4096:
        lengths_to_try.append(8192)
    elif n_target == 8192:
        lengths_to_try.append(4096)

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic()
        print(f"\n=== n={n_try} (remaining: {remaining:.1f}s) ===")

        # Initialize
        if n_try == n_target and parent is not None:
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)
        a0 = np.maximum(a0, 1e-10)
        a0 = a0 / a0.sum()

        sc0 = _score_norm(a0)
        print(f"  Initial: score_norm={sc0:.10f}, full={sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # ---- Phase 1: Quick Adam refinement (15% of remaining time) ----
        p1_budget = min(remaining * 0.15, 200.0)
        p1_end = time.monotonic() + p1_budget
        print(f"  Phase 1 (Adam): {p1_budget:.1f}s budget")

        c_curr = _full_convolve(a0, a0)
        max_cc = float(c_curr.max())
        betas = []
        beta = 0.01 * max_cc
        while beta > 0.0001 * max_cc:

        a_adam, sc_adam = _adam_logspace(a0, p1_end, betas, lr_init=0.01,

        sc_adam = _score_norm(a_adam)
        print(f"  Phase 1 done: score={sc_adam:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()

        a_work = a_adam / a_adam.sum()

        # ---- Phase 2: LP-based optimization (60% of remaining time) ----
        p2_budget = min(remaining * 0.60, 1000.0)
        p2_end = time.monotonic() + p2_budget
        print(f"  Phase 2 (LP): {p2_budget:.1f}s budget")

        # Multiple LP rounds with different K values
        K_schedule = [50, 100, 200]
        per_K = p2_budget / len(K_schedule)

        for Ki, K in enumerate(K_schedule):

            sc_lp = _score_norm(a_work)
            print(f"  LP K={K} done: score={sc_lp:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_work.copy()

        p3_end = min(deadline - 60, time.monotonic() + (deadline - time.monotonic()) * 0.2)
        if time.monotonic() < p3_end:
            print(f"  Phase 3 (Fine LP): {p3_end - time.monotonic():.1f}s budget")
            a_fine, sc_fine = _lp_optimize(a_work, p3_end, K=100, t_init=0.1, stall_limit=10000)
            sc_fine = _score_norm(a_fine)
            print(f"  Phase 3 done: score={sc_fine:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_fine < best_score:
                best_score = sc_fine
                best_seq = a_fine.copy()
                a_work = a_fine

        # ---- Phase 4: Coordinate descent polishing ----
        if time.monotonic() < deadline - 30:
            p4_end = min(deadline - 20, time.monotonic() + 60)
            print(f"  Phase 4 (CoordDescend): {p4_end - time.monotonic():.1f}s budget")
            a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=2, top_k_w=50,
                                                deadline=p4_end)
            sc_cd = _score_norm(a_cd)
            print(f"  Phase 4 done: score={sc_cd:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                a_work = a_cd

        # ---- Phase 5: Restart with perturbation and LP ----
        if time.monotonic() < deadline - 30 and best_seq is not None:

            a_base = best_seq.copy()

        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    print(f"FALLBACK: score={_score_full(fallback):.10f}")
