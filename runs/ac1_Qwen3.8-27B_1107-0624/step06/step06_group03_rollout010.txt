import time

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * c.max() / (s * s)


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -100, 100))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, beta, k):
    n_c = len(c)
    k = min(k, n_c)
    if k == n_c:

        s = w.sum()

            w = np.zeros(n_c)
            w[c.argmax()] = 1.0
        else:
            w /= s
        return w
    # Top-k
    thresh = np.partition(c, -k)[-k]
    mask = c >= thresh
    w = np.zeros(n_c)
    c_masked = c[mask]
    c_max = c_masked.max()
    w_masked = np.exp((c_masked - c_max) / beta)
    s = w_masked.sum()
    if s < 1e-300:
        # Degenerate: put all weight on the argmax
        w[c.argmax()] = 1.0
    else:
        w[mask] = w_masked / s
    return w

    a = _softmax(z)
    n = len(a)
    c = _fft_conv(a, a)
    w = _topk_weights(c, beta, top_k)
