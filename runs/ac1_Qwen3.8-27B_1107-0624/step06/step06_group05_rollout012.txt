Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous version:
- Flat-convolution (Ramanujan) sequence as initial point
- Symmetry exploitation for initialization
- Exact subgradient method with Polyak averaging for fine-tuning
- Adaptive active-set weighting
- More aggressive beta annealing schedule

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    """Gradient of sum_k w_k * (conv(a,a))_k w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _active_set_weighting(c, tol_frac):
    """Compute weights for active set based on proximity to max."""
    c_max = float(c.max())
    if c_max < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if not np.any(active):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    # Weight by exponential proximity
    diffs = c_max - c[active]
    w_active = np.exp(-diffs / (tol_frac * c_max * 0.5))
    w_active = w_active / w_active.sum()
    w = np.zeros_like(c)
    w[active] = w_active
    return w


def _fista_opt(a_init, beta, deadline, max_iter):
    """FISTA on the simplex for smoothed max convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / max(beta, 1e-20)
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _exact_subgradient_opt(a_init, deadline, tol_frac=0.001, averaging=True):
    """Exact subgradient method with Polyak-Ruppert averaging."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    # Polyak averaging
    a_avg = a.copy()
    num_averaged = 0
    
    g_best = None  # best feasible value found
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Active set with weighting
        w = _active_set_weighting(c, tol_frac)
        grad = _compute_grad(a, w)
        
        # Step size: use diminishing or constant
        eta = 1.0 / (it + 1)
        # Scale: the gradient norm is roughly 2*max(a)*active_count, so
        # we want eta * ||grad|| to be O(1/sqrt(it))
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-20:
            eta = 1.0 / (np.sqrt(it + 1) * grad_norm)
            eta = min(eta, 0.5)
        
        a_new = _proj_simplex(a - eta * grad)
        
        # Track best
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Polyak-Ruppert averaging (start after burn-in)
        if it > 100:
            num_averaged += 1
            a_avg = a_avg + (a_new - a_avg) / num_averaged
        
        a = a_new
    
    # Check averaged solution
    if averaging and num_averaged > 100:
        sc_avg = _score_norm(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
    
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 20 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    """L-BFGS-B on log-space with fixed beta smoothing."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 0.5:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 50)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _flat_conv_sequence(n):
    """Generate the flat-convolution (Ramanujan) sequence truncated to length n."""
    # a_k = C(2k, k) / 4^k
    a = np.zeros(n)
    for k in range(n):
        # Use recurrence: a_{k+1} = a_k * (2k+1)/(2k+2)
        if k == 0:
            a[k] = 1.0
        else:
            a[k] = a[k-1] * (2*(k-1)+1) / (2*(k-1)+2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_conv":
        return _flat_conv_sequence(n)
    elif kind == "symmetric_taper":
        # Symmetric taper: sin^p profile
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = 1.0 - (2*t - 1)**4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _symmetrize(a):
    """Symmetrize: a_sym = (a + a[::-1]) / 2"""
    a_sym = 0.5 * (a + a[::-1])
    s = a_sym.sum()
    if s > 0:
        a_sym = a_sym / s
    return a_sym


def _local_search(a, n_iter=200, deadline=None):
    """Simple local search: try shifting mass between pairs of coordinates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        # Pick two random coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Try moving mass from i to j (and vice versa)
        best_delta = 0.0
        best_local = best_score
        # Bisection on delta in [-0.1, 0.1] (relative to a[i])
        max_delta = 0.05 * max(a[i], 1e-6)
        for trial in range(20):
            lo = -max_delta
            hi = max_delta
            mid = (lo + hi) / 2
            a_try = a.copy()
            a_try[i] = max(0, a[i] - mid)
            a_try[j] = a[j] + mid
            s = a_try.sum()
            if s > 0:
                a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_local:
                best_local = sc
                best_delta = mid
            # Determine direction: check midpoint
            if mid > 0:
                # Check if moving more in this direction helps
                a_test = a.copy()
                a_test[i] = max(0, a[i] - mid*1.5)
                a_test[j] = a[j] + mid*1.5
                st = a_test.sum()
                if st > 0:
                    a_test = a_test / st
                if _score_norm(a_test) < sc:
                    lo = mid
                else:
                    hi = mid
            else:
                a_test = a.copy()
                a_test[i] = max(0, a[i] - mid*1.5)
                a_test[j] = a[j] + mid*1.5
                st = a_test.sum()
                if st > 0:
                    a_test = a_test / st
                if _score_norm(a_test) < sc:
                    hi = mid
                else:
                    lo = mid
        
        if best_delta != 0:
            a[i] = max(0, a[i] - best_delta)
            a[j] = a[j] + best_delta
            s = a.sum()
            if s > 0:
                a = a / s
            if best_local < best_score:
                best_score = best_local
                best_a = a.copy()
    
    return best_a, best_score

                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # L-BFGS
            if time.monotonic() < deadline - 10:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000002 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 5.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
