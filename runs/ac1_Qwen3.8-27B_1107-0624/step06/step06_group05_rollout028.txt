import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

"""
Peak-chasing optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive active-set gradient descent with second-order corrections,
combined with L-BFGS polishing and multi-restart exploration.
Key improvements: focus on active convolution peaks, use Hessian-aware steps,
and try multiple sequence lengths.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * c.max() / n)


def _proj_simplex(v, target_sum=None):
    n = len(v)
    if n == 0:
        return v
    if target_sum is None:
        target_sum = 1.0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_conv_max(a, c, active_weights):
    """Compute gradient of max(conv(a,a)) using active set weights.
    a: sequence (length n), c: convolution (length 2n-1)
    active_weights: weights for each position in convolution (length 2n-1)
    Returns gradient w.r.t. a (length n).
    """
    n = len(a)
    # grad_k[i] = 2 * a[k-i] for valid k-i
    # We need: sum_k w[k] * grad_k = sum_k w[k] * 2 * a_rev_shifted
    # This is equivalent to 2 * conv(w, a[::-1])[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(active_weights, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _peak_chase_opt(a_init, deadline, max_iter=500000, tol=1e-12):
    """Active-set gradient descent with adaptive step size.
    Works with sum(a) = n normalization.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a * (n / s)
    
    best_a = a.copy()
    c_init = _full_convolve(a, a)
    best_max = float(c_init.max())
    best_score = 2.0 * best_max / n
    
    step_lr = 0.01
    active_tol = 0.02
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < best_max:
            best_max = c_max
            best_score = 2.0 * best_max / n
            best_a = a.copy()
        
        # Identify active set: positions within tol of max
        threshold = c_max * (1.0 - active_tol)
        active_mask = c >= threshold
        n_active = int(active_mask.sum())
        
        if n_active == 0:
            active_mask = np.zeros_like(c, dtype=bool)
            active_mask[int(np.argmax(c))] = True
            n_active = 1
        
        # Compute active weights (proportional to excess)
        excess = np.maximum(c - threshold, 0.0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            w = np.zeros_like(c)
            w[active_mask] = 1.0 / n_active
        else:
            w = excess / total_excess
        
        # Compute gradient
        grad = _grad_conv_max(a, c, w)
        
        # Gradient on the tangent space of {a >= 0, sum(a) = n}
        # Project gradient: subtract mean to stay on sum=n manifold
        grad_tangent = grad - grad.mean()
        
        # Adaptive step size: use curvature estimate
        grad_norm = np.linalg.norm(grad_tangent)
        if grad_norm < 1e-15:
            break
        
        # Try step with backtracking
        accepted = False
        for shrink in range(20):
            lr = step_lr / (2.0 ** shrink)
            a_trial = a - lr * grad_tangent
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial > 1e-15:
                a_trial = a_trial * (n / s_trial)
            c_trial = _full_convolve(a_trial, a_trial)
            max_trial = float(c_trial.max())
            if max_trial < c_max - 1e-10:
                a = a_trial
                accepted = True
                step_lr = min(step_lr * 1.5, 0.1)
                break
        
        if not accepted:
            step_lr *= 0.5
            if step_lr < 1e-10:
                # Try a random perturbation to escape
                rng = np.random.default_rng(it)
                pert = rng.standard_normal(n) * 0.01 * (a.mean() + 1e-6)
                a = a + pert
                a = np.maximum(a, 0.0)
                a = a * (n / a.sum())
                step_lr = 0.005
        
        # Periodically tighten the active set tolerance
        if it % 500 == 0 and it > 0:
            active_tol = max(0.001, active_tol * 0.9)
    
    return best_a, best_score


def _logsumexp_opt(a_init, deadline, max_iter=200000):
    """Optimize using log-sum-exp smoothing of the max.
    Works with sum(a) = n normalization.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a * (n / s)
    
    c_init = _full_convolve(a, a)
    beta = 0.01 * float(c_init.max())
    
    best_a = a.copy()
    best_score = 2.0 * float(c_init.max()) / n
    
    # Work in log space for non-negativity
    z = np.log(np.maximum(a, 1e-10))
    z_mean = z.mean()
    
    lr = 0.01
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_compute(z)
        a = a * n  # normalize to sum = n
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * c_max / n
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Log-sum-exp smoothing
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s_e = float(e.sum())
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e
        
        # Gradient of smoothed objective w.r.t. a
        grad_a = _grad_conv_max(a, c, w)
        
        # Convert to z-gradient via softmax Jacobian
        # d z / d a: grad_z = grad_a / a - sum(grad_a / a * a) / a ... 
        # Actually: a = softmax(z) * n, so grad_z[i] = a[i] * (grad_a[i] - sum(grad_a * a) / sum(a^2) * a[i])
        # Simpler: grad_z[i] = a[i] * (grad_a[i] - mean(grad_a * a / a))
        # Even simpler for softmax: if a_i = exp(z_i)/sum(exp(z)), then
        # dL/dz_i = a_i * (dL/da_i - sum_j a_j * dL/da_j / sum(a))
        # Since sum(a) = n: dL/dz_i = a_i * (dL/da_i - dot(a, grad_a)/n)
        correction = float(np.dot(a, grad_a)) / n
        grad_z = a * (grad_a - correction)
        
        # Update
        z = z - lr * grad_z
        z = np.clip(z, -50.0, 50.0)
        
        # Decay learning rate
        if it % 5000 == 0 and it > 0:
            lr *= 0.85
        
        # Anneal beta
        if it % 10000 == 0 and it > 0:
            beta *= 0.7
    
    return best_a, best_score


def _lbfgs_polish(a_init, deadline, beta=None):

    from scipy.optimize import minimize

    def obj_fn(z):
        a_raw = _softmax_compute(z)
        a = a_raw * n
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s_e = float(e.sum())
        if s_e < 1e-300:
            return 2.0 * cm / n
        return 2.0 * (cm + beta * np.log(s_e)) / n
    
    def grad_fn(z):
        a_raw = _softmax_compute(z)
        a = a_raw * n
        c = _full_convolve(a, a)
        cm = float(c.max())
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        s_e = float(e.sum())
        if s_e < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e
        grad_a = _grad_conv_max(a, c, w)
        correction = float(np.dot(a, grad_a)) / n
        grad_z = a * (grad_a - correction)
        return grad_z
    
    max_time = deadline - time.monotonic() - 2.0
    if max_time <= 1.0:
        a_out = _softmax_compute(best_z) * n
        return a_out, best_score
    
    try:
        maxiter = max(30, int(max_time * 50))
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': maxiter, 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt) * n
        sc_opt = _score(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    a_best = _softmax_compute(best_z) * n
    return a_best, best_score


def _conjugate_peak_opt(a_init, deadline, max_iter=10000):
    """Conjugate gradient on the active set for peak reduction.
    At each step, solve a small QP on the active peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a * (n / s)
    
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_max = float(c.max())
    best_score = 2.0 * best_max / n
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        if c_max < best_max:
            best_max = c_max
            best_score = 2.0 * c_max / n
            best_a = a.copy()
        
        # Top K peaks
        K = min(10, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:K]
        # Only keep peaks that are significantly above average
        avg_c = float(c.mean())
        top_k = [k for k in top_k if c[k] > avg_c * 1.1]
        if len(top_k) == 0:
            top_k = [int(np.argmax(c))]
        
        # For each active peak k, the gradient is 2*a_shifted
        # We want to find delta that reduces all active peaks
        # Using a simple weighted approach:
        weights = np.zeros(len(top_k))
        for idx, k in enumerate(top_k):
            weights[idx] = c[k] - avg_c
        
        total_w = weights.sum()
        if total_w < 1e-15:
            weights = np.ones(len(top_k)) / len(top_k)
        else:
            weights = weights / total_w
        
        # Combined direction
        direction = np.zeros(n)
        for idx, k in enumerate(top_k):
            # grad of conv[k] w.r.t. a[i] = 2*a[k-i] (if valid)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    direction[i] -= weights[idx] * a[j]
        
        # Project to tangent space (maintain sum)
        direction -= direction.mean()
        
        # Line search
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-15:
            break
        
        direction = direction / dir_norm
        
        found = False
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            a_trial = a + alpha * dir_norm * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_t = a_trial.sum()
            if s_t < 1e-15:
                continue
            a_trial = a_trial * (n / s_t)
            c_trial = _full_convolve(a_trial, a_trial)
            if float(c_trial.max()) < c_max * (1 - 1e-8):
                a = a_trial
                found = True
                break
        
        if not found:
            # Try negative direction
            for alpha in [0.005, 0.002, 0.001]:
                a_trial = a - alpha * dir_norm * direction
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * (n / s_t)
                c_trial = _full_convolve(a_trial, a_trial)
                if float(c_trial.max()) < c_max * (1 - 1e-8):
                    a = a_trial
                    found = True
                    break
        
        if not found:
            if it > 200:
                break
    
    return best_a, best_score


def _make_uniform(n):
    return np.ones(n)


def _perturb_log(a, sigma, seed):
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a, 1e-10))
    z = z + sigma * rng.standard_normal(len(a))
    return z

            best_seq = a_fin.copy()
        print(f"  Final: score={_score(best_seq):.10f}, time={time.monotonic()-t_start:.1f}s")
    
    print(f"Done: best_score={best_score:.10f}")
    
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    fallback = _make_uniform(4096)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
