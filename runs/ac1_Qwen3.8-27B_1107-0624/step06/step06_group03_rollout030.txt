from scipy.optimize import linprog, minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume a is on simplex (sum=1). Return 2n * max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Find t in [0,1] minimizing F((1-t)a + tg) where a, g on simplex."""
    # Golden section search
    phi = (1 - np.sqrt(5)) / 2  # ≈ -0.618
    a_val, b_val = 0.0, 1.0
    c_val = b_val - phi * (b_val - a_val)
    d_val = a_val + phi * (b_val - a_val)
    f_c = _score_norm((1-c_val)*a + c_val*g)
    f_d = _score_norm((1-d_val)*a + d_val*g)

            d_val = c_val

- We want to find g such that (f*g)_k < M for all k where (f*f)_k = M (the "active" constraints)
- The LP with (f*g)_k <= M for all k (or top-K k's) gives such a direction
- By mixing in this direction, the active constraints become inactive, reducing the max

This makes sense! The LP direction g is one where the cross-correlation with f stays below the current peak level. So when we move toward g, the peaks of f*f start to decrease.

Let me also consider using the full LP (all constraints) for smaller n, and the relaxed LP (top-K) for larger n.

For n=4096: 8191 constraints, 4096 variables → might be slow
For n=8192: 16383 constraints, 8192 variables → probably too slow

So for n=4096, I'll use the relaxed LP with K=100-200 constraints.

LP-direction optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding (from the "state of the art" LP approach),
golden section line search, coordinate descent polishing, and smooth gradient refinement.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _lp_direction(a, M, K=100):
    """Find g >= 0 maximizing sum(g) subject to top-K (a*g)_k <= M."""
    n = len(a)
    c = _full_convolve(a, a)
    # Get top-K indices
    if K >= len(c):
        top_k_idx = np.arange(len(c))
    else:
        top_k_idx = np.argsort(c)[-K:]
    top_k_idx = np.sort(top_k_idx)
    
    # Build constraint matrix
    A = np.zeros((len(top_k_idx), n))
    for idx, k in enumerate(top_k_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[idx, j_start:j_end+1] = a[j_start:j_end+1][::-1]
    
    # Solve: max 1^T g s.t. A g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_k_idx), float(M))
    
    # Use HiGHS for speed
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        s = g.sum()
        if s > 1e-15:
            return g / s
    # Fallback: return uniform
    return np.ones(n) / n

def _line_search(a, g, a_prev=None, f_prev=None):
    """Golden section search for best t mixing a and g."""
    if a_prev is not None and f_prev is not None:
        # We know F(0) = f_prev
        pass
    
    phi = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    lo, hi = 0.0, 1.0
    # Initial golden section points
    d = lo + (1 - phi) * (hi - lo)
    c = lo + phi * (hi - lo)
    f_d = _score_norm((1 - d) * a + d * g)
    f_c = _score_norm((1 - c) * a + c * g)
    
    for _ in range(50):
        if f_c < f_d:
            lo = d
            d = c
            f_d = f_c
            c = lo + phi * (hi - lo)
            f_c = _score_norm((1 - c) * a + c * g)
        else:
            hi = c
            c = d
            f_c = f_d
            d = lo + (1 - phi) * (hi - lo)
            f_d = _score_norm((1 - d) * a + d * g)
        if hi - lo < 1e-8:
            break
    
    t = (lo + hi) / 2.0
    f_t = _score_norm((1 - t) * a + t * g)
    return t, f_t

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t_step += 1
            
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad = 2.0 * n * a * (grad_a - mean_grad)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t_step % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
        
        m_buf *= 0.5
        v_buf *= 0.5
    
    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
LP-Direction + Adam hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding (solving a relaxed LP for descent direction),
golden-section line search, Adam in log-space for refinement, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lp_direction(a, M, K=80):
    """Find g >= 0 maximizing sum(g) subject to top-K (a*g)_k <= M.
    Uses dual formulation for speed when K is small relative to n."""
    n = len(a)
    c = _full_convolve(a, a)
    total_len = 2 * n - 1
    if K >= total_len:
        top_k_idx = np.arange(total_len)
    else:
        top_k_idx = np.argsort(c)[-K:]
    top_k_idx = np.sort(top_k_idx)
    Kn = len(top_k_idx)

    # Build A matrix: A[k_idx, j] = a[k_idx - j] if 0 <= k_idx - j < n
    A = np.zeros((Kn, n))
    for idx, k in enumerate(top_k_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A[idx, j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

    # Solve via dual: min M*sum(y) s.t. A^T y >= 1, y >= 0
    # Equivalent scipy form: min M*sum(y) s.t. -A^T y <= -1, y >= 0
    c_obj = float(M) * np.ones(Kn)
    A_dual = -A.T  # shape (n, Kn)
    b_dual = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_dual, b_ub=b_dual,
                      bounds=[(0, None)] * Kn, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            # Recover primal from dual via complementary slackness, or re-solve primal
            # Simpler: solve primal directly with reduced problem
            pass
    except Exception:
        pass

    # Fallback: solve primal directly
    try:
        c_prim = -np.ones(n)
        b_prim = np.full(Kn, float(M))
        res = linprog(c_prim, A_ub=A, b_ub=b_prim,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass

    return np.ones(n) / n


def _line_search(a, g, max_iter=40):
    """Golden section search for best t in [0,1] minimizing F((1-t)a + tg)."""
    phi = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    d = lo + (1.0 - phi) * (hi - lo)
    c_pt = lo + phi * (hi - lo)
    f_d = _score_norm((1.0 - d) * a + d * g)
    f_c = _score_norm((1.0 - c_pt) * a + c_pt * g)

    for _ in range(max_iter):
        if hi - lo < 1e-9:
            break
        if f_c < f_d:
            lo = d
            d = c_pt
            f_d = f_c
            c_pt = lo + phi * (hi - lo)
            f_c = _score_norm((1.0 - c_pt) * a + c_pt * g)
        else:
            hi = c_pt
            c_pt = d
            f_c = f_d
            d = lo + (1.0 - phi) * (hi - lo)
            f_d = _score_norm((1.0 - d) * a + d * g)

    t = (lo + hi) / 2.0
    f_t = _score_norm((1.0 - t) * a + t * g)
    return t, f_t

            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new

            sc_new = _score_norm(a_new)
            n_iter += 1

            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                patience = 0
                if n_iter % 20 == 1:
                    print(f"    LP iter {n_iter}: score={_score(a):.10f}, t={t_opt:.4f}")
            else:
                patience += 1
                # Try half step
                t_half = t_opt * 0.5
                a_half = (1.0 - t_half) * a + t_half * g
                a_half = np.maximum(a_half, 0.0)
                a_half = a_half / a_half.sum()
                sc_half = _score_norm(a_half)
                if sc_half < best_score:
                    best_score = sc_half
                    best_a = a_half.copy()
                    a = a_half
                    patience = 0
                else:
                    pass

        if patience > patience_limit:
            break

    return best_a, best_score


def _topk_softmax_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter_per_beta):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_beta:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t_step += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad = 2.0 * n * a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_step % 200 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0002, current_lr * 0.97)

        m_buf *= 0.3
        v_buf *= 0.3

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_fast(a, n_coords_per_pass, n_passes, deadline, rng):
    """Fast coordinate descent on random subsets of coordinates."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        idx = rng.choice(n, size=min(n_coords_per_pass, n), replace=False)
        for i in idx:
            if time.monotonic() > deadline:
                break
            a_i_old = a[i]
            if a_i_old < 1e-14:
                continue

            c = _full_convolve(a, a)

            # d[j] = 2*a[j-i] for valid j
            d_j = np.zeros(2 * n - 1)
            d_j[i:i + n] = 2.0 * a

            S0 = 1.0 - a_i_old

            # Find critical values
            top_idx = np.argsort(c)[-30:]
            candidates = [0.0, a_i_old * 0.3, a_i_old * 0.7, a_i_old, a_i_old * 1.3]
            for j in top_idx:
                if d_j[j] > 1e-14:
                    t_opt = 1.0 + a_i_old - 2.0 * c[j] / d_j[j]
                    if 0.0 <= t_opt <= 1.5 * a_i_old + 0.05:
                        candidates.append(t_opt)
                        candidates.append(max(0.0, t_opt * 0.7))

            candidates = list(set([max(0.0, min(1.0, round(tc, 12))) for tc in candidates]))

            best_t = a_i_old
            best_f = np.inf
            for tc in candidates:
                c_trial = c + (tc - a_i_old) * d_j
                max_c = float(c_trial.max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max_c / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            a[i] = best_t
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 7]:
            a += 0.15 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.6
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()

        if parent_raw.size > 0:

            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()

    else:

        parent = _structured_init(n_parent, "cos_sum")

        best_seq = parent.copy()
        best_score = sc_parent

    # ============ Phase 1: LP-Direction on n=4096 ============
    n1 = 4096

    a1 = _interp_to_length(parent, n1)
    sc1 = _score(a1)
    print(f"  Start: score={sc1:.10f}")
    if sc1 < best_score:
        best_score = sc1
        best_seq = a1.copy()

    phase1_deadline = t_start + min(500, (budget_s - 25) * 0.3)
    a1, sc1_norm = _lp_direction_loop(a1, phase1_deadline, K=60, patience_limit=25)

    print(f"  After LP: score={sc1:.10f}")
    if sc1 < best_score:

    # ============ Phase 2: Adam log-space on n=4096 ============
    print(f"\n=== Phase 2: Adam n={n1} (remaining: {deadline - time.monotonic():.0f}s) ===")
    a2 = a1.copy()
    c2 = _full_convolve(a2, a2)
    max_c2 = float(c2.max())
    betas2 = [0.004 * max_c2, 0.002 * max_c2, 0.001 * max_c2, 0.0005 * max_c2,
              0.0002 * max_c2, 0.0001 * max_c2, 0.00005 * max_c2]
    phase2_deadline = min(deadline - 300, time.monotonic() + min(500, (budget_s - 25) * 0.3))
    a2, sc2_norm = _adam_logspace(a2, phase2_deadline, betas2, lr_init=0.008,
                                  top_k=100, max_iter_per_beta=100000)
    sc2 = _score(a2)
    print(f"  After Adam: score={sc2:.10f}")
    if sc2 < best_score:
        best_score = sc2
        best_seq = a2.copy()

    # ============ Phase 3: Coord descent polish ============
    print(f"\n=== Phase 3: Coord Descend (remaining: {deadline - time.monotonic():.0f}s) ===")

    a3 = a3 / a3.sum()
    cd_deadline = min(deadline - 200, time.monotonic() + 150)

        K4 = 80 if n4 <= 4096 else 40

        a4, sc4_norm = _lp_direction_loop(a4, phase4_deadline, K=K4, patience_limit=20)
        sc4 = _score(a4)
        print(f"  After LP2: score={sc4:.10f}")

        phase5_deadline = min(deadline - 60, time.monotonic() + min(300, (budget_s - 25) * 0.15))
        a5, sc5_norm = _lp_direction_loop(a5, phase5_deadline, K=40, patience_limit=20)
        sc5 = _score(a5)
        print(f"  After LP: score={sc5:.10f}")

            best_score = sc5
            best_seq = a5.copy()

        # Quick Adam on n=8192
        if time.monotonic() < deadline - 40:
            c5 = _full_convolve(a5, a5)
            max_c5 = float(c5.max())
            betas5 = [0.003 * max_c5, 0.001 * max_c5, 0.0005 * max_c5]
            adam5_deadline = min(deadline - 40, time.monotonic() + 120)
            a5b, sc5b = _adam_logspace(a5, adam5_deadline, betas5, lr_init=0.005,
                                       top_k=60, max_iter_per_beta=50000)
            sc5b_full = _score(a5b)
            print(f"  After Adam: score={sc5b_full:.10f}")
            if sc5b_full < best_score:
                best_score = sc5b_full
                best_seq = a5b.copy()

    # ============ Phase 6: Diversity restarts ============
    if time.monotonic() < deadline - 40 and best_seq is not None:
        print(f"\n=== Phase 6: Restarts (remaining: {deadline - time.monotonic():.0f}s) ===")
        n_r = len(best_seq)
        a_base = best_seq.copy() / best_seq.sum()
        per_restart = max(20, (deadline - time.monotonic() - 20) / 3)

        for ri in range(3):
            if time.monotonic() > deadline - 20:
                break
            if ri == 0:
                p = a_base + 0.01 * rng.standard_normal(n_r)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_r // 2048))
                p = np.roll(a_base, shift)
            else:

                pert = 0.015 * np.cos(2 * np.pi * freq * np.arange(n_r) / n_r)

            p = p / p.sum()

            c_p = _full_convolve(p, p)

            betas_r = [0.003 * max_cp, 0.001 * max_cp, 0.0005 * max_cp]

                                       betas_r, lr_init=0.006, top_k=80,
                                       max_iter_per_beta=50000)
            sc_r = _score(p_best)
            print(f"  Restart {ri}: score={sc_r:.10f}")
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_best.copy()

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:

        sc_final = _score(final)
        print(f"FINAL VALIDATED: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) * 1.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
