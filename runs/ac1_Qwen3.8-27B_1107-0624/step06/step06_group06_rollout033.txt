        grad = 4.0 * n * a * (h - W)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 25 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.88
                if patience > 25:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.6

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_continuation(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with power-sum continuation, then log-sum-exp polish."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    # Phase A: Power-sum continuation (p from high to low)
    ps = [40, 20, 10, 5, 3, 2]
    for p in ps:
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break
        est_time = 0.003 * max(1, n / 2048.0)
        max_iter = min(4000, max(20, int(remaining / (est_time * 12))))

        def fun_ps(z_local):
            return _power_obj(z_local, float(p), n)

        def jac_ps(z_local):
            return _power_grad(z_local, float(p), n)

        try:
            result = minimize(
                fun_ps, z, jac=jac_ps, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    # Phase B: Log-sum-exp with adaptive beta
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005])

    for beta in betas:
        elapsed = time.monotonic()
        if elapsed + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - elapsed
        if remaining < 5:
            break
        est_time = 0.003 * max(1, n / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time * 12))))

        def fun_ls(z_local):
            return _smoothed_obj(z_local, beta, n)

        def jac_ls(z_local):
            return _smoothed_grad(z_local, beta, n)

        try:
            result = minimize(
                fun_ls, z, jac=jac_ls, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15,
                         'gtol': 1e-11, 'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage(a, deadline, fractions=None):
    """Zero out small elements and re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    if fractions is None:
        fractions = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30]

    a_sorted_idx = np.argsort(a)

    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            # Re-sort for next iteration
            a_sorted_idx = np.argsort(best_a)

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym":
        t = np.linspace(0, 1, n)
        a = np.exp(-2.0 * t) + 0.3 * np.exp(-2.0 * (1 - t))
        a = a + 0.1
        return a / a.sum()
    elif kind == "cos_asym":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t) + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_fourier(a, rng, n_modes=4, amp_range=(0.01, 0.08)):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_asym(a, rng):
    """Asymmetric perturbation: add a linear bias in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    bias = rng.uniform(-0.3, 0.3)
    t = np.linspace(0, 1, n)
    z = z + bias * t + rng.uniform(-0.05, 0.05)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_blocks=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(2, n // (2 * n_blocks + 2))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _multi_scale_optimize(profile_fn, n_target, deadline, min_time_remaining=60):
    """Start small, optimize, upsample, repeat."""
    n_steps = [256, 512, 1024, 2048, 4096, 8192]
    n_steps = [x for x in n_steps if x <= n_target]
    if not n_steps:
        n_steps = [n_target]
    if n_steps[-1] != n_target:
        n_steps.append(n_target)

    a_cur = profile_fn(n_steps[0])
    for i, n_try in enumerate(n_steps):
        if time.monotonic() + min_time_remaining > deadline:
            break
        if i == 0:
            a_cur = profile_fn(n_try)
        a_opt = _run_continuation(a_cur, deadline, min_time_remaining=min_time_remaining)
        a_cur = a_opt
        if i < len(n_steps) - 1:
            # Upsample
            x_old = np.linspace(0, 1, len(a_cur))
            x_new = np.linspace(0, 1, n_steps[i + 1])
            a_cur = np.interp(x_new, x_old, a_cur)
            a_cur = np.maximum(a_cur, 0)
            a_cur = a_cur / a_cur.sum()

    return a_cur


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining power-sum continuation, flat-spectrum
    projection, asymmetric perturbations, and aggressive support reduction.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: Flat-spectrum projection on parent ---
    print(f"\n[Phase 1] Flat-spectrum projection (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 400:
        a_fs = _flat_spectrum_proj(parent, n_iter=30, taper=0.0)
        sc_fs = _score(a_fs)
        update_best(a_fs, sc_fs, "flat_spec")
        # Continue optimizing from flat-spectrum point
        a_fs_opt = _run_continuation(a_fs, deadline, min_time_remaining=400)
        update_best(a_fs_opt, _score(a_fs_opt), "flat_spec_opt")

    # --- Phase 2: Main continuation from parent ---
    print(f"\n[Phase 2] Main continuation from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_continuation(parent, deadline, min_time_remaining=350)
    update_best(a_main, _score(a_main), "main_cont")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 100)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=2000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_1")

    # --- Phase 3: Shrinkage ---
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 250)
        update_best(a_shr, sc_shr, "shrink")
        if sc_shr < best_score and time.monotonic() < deadline - 150:
            a_shr2 = _run_continuation(a_shr, deadline, min_time_remaining=100)
            update_best(a_shr2, _score(a_shr2), "shrink_reopt")

    # --- Phase 4: Multi-n search with diverse profiles ---
    print(f"\n[Phase 4] Multi-n diverse profiles (t={time.monotonic() - t_start:.1f}s)")
    profiles = ["hann", "asym", "cos_asym", "tapered"]
    n_candidates = [2048, 4096]
    if n_parent not in n_candidates and n_parent <= 8192:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        for prof_name in profiles:
            if time.monotonic() > deadline - 80:
                break
            a_init = _make_profile(n_try, prof_name)
            # Apply flat-spectrum as pre-processing
            a_init = _flat_spectrum_proj(a_init, n_iter=10, taper=0.1)
            a_opt = _run_continuation(a_init, deadline, min_time_remaining=60)
            sc_opt = _score(a_opt)
            update_best(a_opt, sc_opt, f"n{n_try}_{prof_name}")

    # --- Phase 5: Diverse restarts from best ---
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        restart_fns = [
            (lambda: _perturb_fourier(best_seq, rng, n_modes=5, amp_range=(0.02, 0.1)), "fourier"),
            (lambda: _perturb_asym(best_seq, rng), "asym"),
            (lambda: _perturb_block_swap(best_seq, rng, n_blocks=4), "block"),
        ]
        for fn, label in restart_fns:
            if time.monotonic() > deadline - 40:
                break
            p_start = fn()
            a_r = _run_continuation(p_start, deadline, min_time_remaining=30)
            update_best(a_r, _score(a_r), f"restart_{label}")
            if _score(a_r) < best_score + 0.003 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _subgrad_polish(a_r, min(deadline - 10, time.monotonic() + 30),
                                              n_max_steps=500, lr_init=0.01)
                update_best(a_r2, sc_r2, f"restart_{label}_sg")

    # --- Phase 6: Asymmetric perturbation + re-optimize ---
    print(f"\n[Phase 6] Asymmetric exploration (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        for trial in range(3):
            if time.monotonic() > deadline - 20:
                break
            p_try = _perturb_asym(best_seq, rng)
            a_try = _run_continuation(p_try, deadline, min_time_remaining=15)
            update_best(a_try, _score(a_try), f"asym_{trial}")

    # --- Phase 7: Final fine polish ---
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very fine continuation
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            # Use very small beta
            beta_final = mb * 0.0005
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda z_local: _smoothed_obj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smoothed_grad(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 60}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient
        if best_seq is not None and time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 3,
                                             n_max_steps=500, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Final shrinkage attempt
        if best_seq is not None and time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _shrinkage(best_seq, deadline - 2,
                                         fractions=[0.01, 0.03, 0.05, 0.08])
            update_best(a_fin3, sc_fin3, "final_shrink")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
