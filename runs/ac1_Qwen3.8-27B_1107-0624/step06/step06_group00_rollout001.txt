        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cosine_anneal(T_start, T_end, frac):
    return T_end + 0.5 * (T_start - T_end) * (1.0 + np.cos(np.pi * frac))


def _compute_soft_gradient(a, c, T, active_k):
    n = len(a)
    c_shifted = (c - c.max()) / T
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    if active_k is not None and active_k < len(c):
        top_idx = np.argsort(c)[::-1][:active_k]
        p_masked = np.zeros_like(p)
        p_masked[top_idx] = p[top_idx]
        p_masked_sum = p_masked.sum()
        if p_masked_sum > 1e-300:
            p = p_masked / p_masked_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _compute_active_gradient(a, c, top_k):
    n = len(a)
    k_indices = np.argsort(c)[::-1][:top_k]
    p = np.zeros_like(c)
    p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=15):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return a
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a
        g = g / g.sum()
        best_t = 0.0
        best_sc = _score_norm(a)
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1.0 - t_val) * a + t_val * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
        return a
    except Exception:
        return a


def _adam_optimize(a_init, deadline, T_start, T_end, lr_start, lr_end,
                   max_iter=500000, active_k_start=20, active_k_end=3,
                   eval_interval=200, clip_norm=0.6, use_momentum=True):
    """Adam optimizer in z-space with cosine annealing and adaptive active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    g_momentum = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    current_lr_mult = 1.0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = _cosine_anneal(T_start, T_end, frac)
        lr = lr_start * (lr_end / lr_start) ** frac * current_lr_mult

        a = _softmax_stable(z)
        c = _fft_convolve(a, a)

        # Adaptive active set size
        active_k = max(active_k_end, int(active_k_start * (1.0 - frac) + active_k_end * frac))

        g = _compute_soft_gradient(a, c, T, active_k)

        # Gradient momentum (EMA)
        if use_momentum:
            g_momentum = 0.8 * g_momentum + 0.2 * g
            g = g_momentum

        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)

        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % eval_interval == 0:
            a_cur = _softmax_stable(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 4 and current_lr_mult > 0.1:
                current_lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                g_momentum = np.zeros(n)

    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, deadline, passes=3):
    """Coordinate descent: for each position, try small perturbations."""
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(123)

    for p in range(passes):
        if time.monotonic() > deadline:
            break
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.monotonic() > deadline:
                break
            orig = best[i]
            if orig < 1e-15:
                continue
            # Try multiplicative perturbations
            best_local = orig
            best_local_sc = best_sc
            for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]:
                trial = best.copy()
                trial[i] = orig * factor
                # Adjust sum to maintain normalization
                sc = _score_norm(trial)
                if sc < best_local_sc - 1e-12:
                    best_local_sc = sc
                    best_local = orig * factor
            if best_local != orig:
                best[i] = best_local
                best_sc = best_local_sc
        # Periodic full evaluation
        full_sc = _score(best)
        print(f"  CD pass {p+1}: norm_score={_score_norm(best):.10f}, full={full_sc:.10f}")

    return best


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _downsample(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    # Average pooling
    blocks = n0 // n_new
    if blocks < 1:
        return _interp_to_length(a, n_new)
    a_padded = a[:blocks * n_new]
    a_res = a_padded.reshape(n_new, blocks).mean(axis=1)
    a_res = np.maximum(a_res, 1e-15)
    return a_res / a_res.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. LP-based steps from parent
    2. Adam z-space with cosine annealing + adaptive active set
    3. n=8192 exploration
    4. Fine polish at best length
    5. Coordinate descent
    6. Geometric-distributed restarts
    7. Final ultra-fine polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Build LP indices
    rows, cols = _build_lp_indices(n)
    print(f"LP indices built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP iterations (0 - 100s)
    phase1_end = min(deadline, t_start + 100.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0
    while time.monotonic() < phase1_end:
        a = _lp_step(a, rows, cols, time_limit=12)
        sc = _score_norm(a)
        lp_count += 1
        if sc < prev_sc - 1e-12:
            stall = 0
        else:
            stall += 1
        full_sc = _score(a)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a.copy()
        if lp_count % 3 == 0:
            print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
        if stall > 12:
            break
        prev_sc = sc
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Adam z-space with cosine annealing, large-to-small active set (100s - 450s)
    phase2_end = min(deadline, t_start + 450.0)
    if time.monotonic() < phase2_end:
        c_best = _fft_convolve(best_a, best_a)
        max_c = float(c_best.max())
        a_adam, sc_adam = _adam_optimize(best_a, phase2_end - 3,
                                          T_start=0.01 * max_c, T_end=0.00005 * max_c,
                                          lr_start=0.05, lr_end=0.003,
                                          max_iter=600000,
                                          active_k_start=20, active_k_end=3,
                                          eval_interval=150, clip_norm=0.6)
        full_sc = _score(a_adam)
        print(f"Phase 2 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # Phase 3: Try n=8192 (450s - 850s)
    phase3_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase3_end:
        n_large = 8192
        a_large = _interp_to_length(best_a, n_large)
        c_large = _fft_convolve(a_large, a_large)
        max_cl = float(c_large.max())
        a_large_opt, sc_large = _adam_optimize(a_large, phase3_end - 5,
                                                T_start=0.005 * max_cl, T_end=0.00003 * max_cl,
                                                lr_start=0.04, lr_end=0.003,
                                                max_iter=300000,
                                                active_k_start=15, active_k_end=3,
                                                eval_interval=200, clip_norm=0.5)
        full_sc = _score(a_large_opt)
        print(f"Phase 3 n=8192 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_large_opT.copy() if False else a_large_opt.copy()
        # Try downsampling back
        if full_sc < 2.0:
            a_ds = _downsample(a_large_opt, min(n, 4096))
            c_ds = _fft_convolve(a_ds, a_ds)
            max_cd = float(c_ds.max())
            a_ds_opt, sc_ds = _adam_optimize(a_ds, phase3_end - 5,
                                              T_start=0.002 * max_cd, T_end=0.00002 * max_cd,
                                              lr_start=0.025, lr_end=0.002,
                                              max_iter=150000,
                                              active_k_start=10, active_k_end=3,
                                              eval_interval=150, clip_norm=0.4)
            full_sc_ds = _score(a_ds_opt)
            print(f"Phase 3 downsampled n={len(a_ds)}: score={full_sc_ds:.10f}")
            if np.isfinite(full_sc_ds) and full_sc_ds < best_score:
                best_score = full_sc_ds
                best_a = a_ds_opt.copy()

    # Phase 4: Fine polish at best length (850s - 1200s)
    phase4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase4_end:
        n_best = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_polish, sc_polish = _adam_optimize(best_a, phase4_end - 5,
                                              T_start=0.001 * max_cf, T_end=0.000005 * max_cf,
                                              lr_start=0.015, lr_end=0.001,
                                              max_iter=400000,
                                              active_k_start=8, active_k_end=3,
                                              eval_interval=100, clip_norm=0.3)
        full_sc = _score(a_polish)
        print(f"Phase 4 polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_polish.copy()

    # Phase 5: Coordinate descent (1200s - 1450s)
    phase5_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Coordinate descent, t={time.monotonic()-t_start:.0f}s")
        best_a = _coordinate_descent(best_a, phase5_end - 5, passes=4)
        full_sc = _score(best_a)
        print(f"Phase 5 CD result: score={full_sc:.10f}")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc

    # Phase 6: Geometric-distributed restarts (1450s - 1700s)
    phase6_end = min(deadline, t_start + 1700.0)
    n_best = len(best_a)
    if time.monotonic() < phase6_end:
        sigma_schedule = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002]
        for attempt, sigma in enumerate(sigma_schedule):
            if time.monotonic() > phase6_end - 30:
                break
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(n_best)
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            c_p = _fft_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            phase6_sub_end = min(phase6_end - 3, t_start + 150 + (phase6_end - t_start - 1500) * (attempt + 1) / len(sigma_schedule))
            a_pert_opt, sc_pert = _adam_optimize(a_pert, phase6_sub_end,
                                                  T_start=0.001 * max_cp, T_end=0.00001 * max_cp,
                                                  lr_start=0.02, lr_end=0.001,
                                                  max_iter=60000,
                                                  active_k_start=8, active_k_end=3,
                                                  eval_interval=100, clip_norm=0.3)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] sigma={sigma:.4f}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 7: Final ultra-fine polish (1700s - deadline)
    if time.monotonic() < deadline - 10:
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        a_final, sc_final = _adam_optimize(best_a, deadline - 8,
                                            T_start=0.0002 * max_cf, T_end=0.000001 * max_cf,
                                            lr_start=0.003, lr_end=0.0002,
                                            max_iter=200000,
                                            active_k_start=5, active_k_end=2,
                                            eval_interval=80, clip_norm=0.2)
        full_sc = _score(a_final)
        print(f"Phase 7 final: score={full_sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_final.copy()

        # Brief coordinate descent at the end
        if time.monotonic() < deadline - 5:
            best_a = _coordinate_descent(best_a, deadline - 3, passes=1)
            full_sc = _score(best_a)
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
