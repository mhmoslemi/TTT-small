def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_gradient(a, c, n, k_active=5):

def _lbfgs_smooth(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 80, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()

def _make_chirp(n, freq=1.0, eps=0.3):
    t = np.arange(n)
    a = 1.0 + eps * np.cos(2.0 * np.pi * freq * t * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_multiband(n, bands=None, eps=0.2):

    a = np.ones(n)

    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_random_positive(n, rng):

def _make_tapered(n, power=0.5):

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    if n_cur == target_n:
        return a.copy()
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-12)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _kkt_refine(a, deadline, k_active=4, max_iter=200):
    """Refine using KKT-inspired iterative method."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):

            break

        c_max = c.max()
        idx = np.argsort(c)[::-1][:k_active]

        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / len(idx)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        # Newton-like step: try to equalize active convolution values
        # by adjusting a in the direction that reduces the spread
        g = h - W  # This is the "autocorrelation gradient"
        g = g - g.mean()

        # Compute a step that reduces the active values
        # Use line search
        alpha = 1.0
        improved = False
        for _ in range(20):
            a_try = a - alpha * g / n
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try
                improved = True
                break
            alpha *= 0.7

        if not improved:

            c_try = _full_convolve(a, a)
            peaks = c_try[idx]
            spread = peaks.max() - peaks.min()
            if spread < 1e-15:
                break
            alpha *= 0.5
            if alpha < 1e-15:

def _kkt_solve(a_init, deadline, k_active=5):
    """
    Try to solve the KKT conditions using scipy fsolve.
    Variables: log(p) for indices in support (approximated as all n).
    We use the condition that the active convolution values are equal
    and the gradient of the "balanced" objective is zero on the simplex.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    # Identify active set
    c = _full_convolve(a, a)
    idx = np.argsort(c)[::-1][:k_active]

    # Build the system: we want to find z (log p) such that
    # the active convolution values are all equal, and sum(p) = 1.
    # The active convolution values as functions of z:
    # (p*p)[k] = sum_i p[i]*p[k-i] = sum_i exp(z[i]+z[k-i])

    def kkt_residuals(z):
        p = _softmax(z)
        c_val = _full_convolve(p, p)
        # We want c_val[idx] to all be equal
        # Return differences from the mean of active values
        active_vals = c_val[idx]
        diffs = active_vals - active_vals.mean()
        # Also add gradient conditions: the derivative of each active
        # value w.r.t. z should be orthogonal to the constraint gradient
        # For simplicity, just use the equalization conditions
        return diffs

    try:
        from scipy.optimize import fsolve
        z0 = np.log(np.maximum(a, 1e-18))
        z0 = z0 - z0.mean()

        sol, info, ier, msg = fsolve(kkt_residuals, z0, maxfev=2000,
                                     full_output=True)
        if ier == 1:
            p = _softmax(sol)
            sc = _score(p)
            if sc < best_sc:
                best_a = p.copy()
                best_sc = sc
    except Exception:
        pass

    return best_a, best_sc

        ("multi_12", _make_multiband(n_init, bands=[1, 2], eps=0.3)),
        ("multi_25", _make_multiband(n_init, bands=[2, 5], eps=0.2)),
        ("tapered05", _make_tapered(n_init, power=0.5)),
        ("tapered10", _make_tapered(n_init, power=1.0)),
    ]

    for name, a_init in structured_inits:
        candidates.append((name, a_init))
        sc = _score(a_init)
        print(f"  {name}: n={len(a_init)}, score={sc:.10f}")
        try_update(a_init, name)

    # Random initializations
    for r in range(3):
        a_r = _make_random_positive(n_init, rng)
        candidates.append((f"random_{r}", a_r))
        sc = _score(a_r)
        print(f"  random_{r}: score={sc:.10f}")
        try_update(a_r, f"random_{r}")

    # If parent was best, start from it. Otherwise use best candidate.
    if best_seq is None:
        best_seq = structured_inits[0][1].copy()
        best_score_val = _score(best_seq)
        best_n = len(best_seq)

    elapsed = time.monotonic() - t_start
    print(f"  Phase 1 done in {elapsed:.1f}s. Best: {best_score_val:.10f}")

    # === Phase 2: Main L-BFGS on best candidate ===
    print(f"\n=== Phase 2: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p2 = _adaptive_lbfgs(best_seq, deadline, reserve_time=600.0)
    try_update(a_p2, "main_lbfgs")

    # === Phase 3: Subgradient polishing ===
    print(f"\n=== Phase 3: Subgradient polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 400,
                                      k_active=5, max_steps=30000, lr_init=0.05)
        try_update(a_sg, "sg_z_k5")

    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg2, sc_sg2 = _subgrad_polish(best_seq, deadline - 300,
                                        k_active=3, max_steps=20000, lr_init=0.04)
        try_update(a_sg2, "sg_z_k3")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_d, sc_d = _direct_subgrad(best_seq, deadline - 200,
                                    k_active=3, max_steps=15000)
        try_update(a_d, "sg_direct")

    # === Phase 4: KKT refinement ===
    print(f"\n=== Phase 4: KKT refinement (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_kkt, sc_kkt = _kkt_refine(best_seq, deadline - 200, k_active=4)
        try_update(a_kkt, "kkt_refine")

    if best_seq is not None and time.monotonic() < deadline - 150:
        a_kkt2, sc_kkt2 = _kkt_solve(best_seq, deadline - 150, k_active=5)
        try_update(a_kkt2, "kkt_fsolve")

    # === Phase 5: More L-BFGS rounds ===
    print(f"\n=== Phase 5: Additional L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_p5 = _adaptive_lbfgs(best_seq, deadline, reserve_time=150.0)
        try_update(a_p5, "lbfgs_round2")

    # === Phase 6: Basin hopping ===
    print(f"\n=== Phase 6: Basin hopping (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_hop, sc_hop = _basin_hop(best_seq, deadline - 100, rng,
                                   n_hops=6, reserve=50.0)
        try_update(a_hop, "basin_hop")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        lengths_to_try = [8192, 2048, 10240, 16384]
        for target_n in lengths_to_try:
            if time.monotonic() > deadline - 60:
                break
            if target_n == best_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"    Interpolated score: {sc_interp:.10f}")

            # Quick optimization at new length
            if time.monotonic() < deadline - 50:
                a_opt = _adaptive_lbfgs(a_interp, deadline, reserve_time=40.0)
                try_update(a_opt, f"n_{target_n}")

    # === Phase 8: Final aggressive polishing ===
    print(f"\n=== Phase 8: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break

            # Micro L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            max_iter = min(5000, max(200, int((deadline - 5 - time.monotonic()) / 0.005)))
            a_f = _lbfgs_smooth(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_polish(best_seq, deadline - 5,
                                         k_active=3, max_steps=10000, lr_init=0.02)
                try_update(a_s, f"final_sg_{rnd}")

            # KKT
            if time.monotonic() < deadline - 5:
                a_k, _ = _kkt_refine(best_seq, deadline - 5, k_active=3)
                try_update(a_k, f"final_kkt_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
