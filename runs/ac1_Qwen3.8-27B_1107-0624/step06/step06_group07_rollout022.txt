"""
Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, Nesterov-accelerated active-set
subgradient with backtracking, coordinate descent refinement, and
multi-restart diversification. Key improvements: better beta schedule,
momentum-based subgradient, 1D coordinate search, and structured perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _exact_subgrad(a, n, active_fractile=0.05, rel_tol=1e-3):
    """Compute exact subgradient of max(conv(a,a)) with respect to a,
    using a mixture over the active set (top fractile or within rel_tol of max)."""
    c = _full_convolve(a, a)
    c_max = c.max()
    threshold = c_max - rel_tol * c_max
    active = c >= threshold
    if active.sum() < 1:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    # Also include top fractile
    n_active = max(1, int(active_fractile * len(c)))
    top_k = np.argsort(c)[-n_active:]
    active[top_k] = True
    
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # Gradient of max_c * 2n / S^2 (with S fixed, just 2n * grad max_c)
    grad = 2.0 * n * (h - W)
    return grad, c, c_max


def _lbfgs_run(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxcor': 30,
                                   'maxls': 30, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init.copy()


def _nesterov_subgrad(a_init, deadline, max_steps=8000, rel_tol=1e-3, active_frac=0.05):
    """Nesterov-accelerated subgradient with backtracking line search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_prev = z.copy()
    v = np.zeros_like(z)  # momentum
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.02
    momentum = 0.9
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        y = z + momentum * v
        a_y = _softmax(y)
        grad_a, c, c_max = _exact_subgrad(a_y, n, active_frac, rel_tol)
        
        # Convert gradient from a-space to z-space
        # da/dz = a * (I - a a^T) (for softmax)
        # grad_z = a * (grad_a - a dot grad_a)
        a_dot_grad = float(np.dot(a_y, grad_a))
        grad_z = a_y * (grad_a - a_dot_grad)
        
        # Backtracking line search
        step = lr
        a_cur = _softmax(z)
        sc_cur = _score(a_cur)
        
        for bt in range(20):
            z_try = z - step * grad_z
            z_try = np.clip(z_try, -80.0, 80.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)
            if sc_try < sc_cur - 1e-15:
                break
            step *= 0.5
        else:
            step = 0
        
        # Nesterov update
        v = momentum * v - step * grad_z
        v = np.clip(v, -5.0, 5.0)
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        z_prev = z.copy()
        
        # Track best
        if i % 100 == 0 or i < 10:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-13:
                best_sc = sc_check
                best_a = a_check.copy()
                stall = 0
                lr = min(lr * 1.05, 0.08)
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.7
                    momentum = min(momentum * 1.01, 0.95)
                if lr < 1e-5:
                    break
        
        # Adaptive momentum: increase if making progress
        if i % 500 == 0 and i > 0:
            momentum = 0.9 + 0.03 * np.sin(i / 2000.0 * np.pi)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a, best_sc


def _coord_descent(a_init, deadline, n_passes=3):
    """Coordinate descent: for each coordinate, try small adjustments."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try moving mass from j to a few other positions
            base = a.copy()
            base_sc = _score(base)
            
            # Try reducing a[j] and adding to random positions
            for delta_frac in [0.1, 0.05, 0.02]:
                mass = a[j] * delta_frac
                if mass < 1e-10:
                    continue
                # Add to a weighted random set of positions
                for trial in range(3):
                    a_try = base.copy()
                    a_try[j] -= mass
                    # Distribute to 3 random nearby positions
                    offsets = [1, 2, 3]
                    targets = [(j + offsets[trial]) % n,
                               (j - offsets[trial]) % n,
                               (j + n // 2) % n]
                    for t in targets:
                        a_try[t] += mass / 3.0
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-10:
                        continue
                    a_try = a_try / s
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-13:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        a = a_try.copy()
                        improved = True
                        break
            if not improved and p > 0:
                # Try increasing a[j] (taking from others)
                for delta_frac in [0.05, 0.02]:
                    mass = a.mean() * delta_frac
                    a_try = base.copy()
                    a_try[j] += mass
                    # Reduce others proportionally
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others_sum = a_try[mask].sum()
                    if others_sum > 1e-10:
                        a_try[mask] -= mass * a_try[mask] / others_sum
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-10:
                        continue
                    a_try = a_try / s
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-13:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        a = a_try.copy()
                        improved = True
                        break

    return best_a, best_sc


def _lbfgs_schedule(a_start, deadline, min_time_remaining=50.0):
    """Run L-BFGS with geometric beta decay."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start.copy()

    # Start with larger beta, decay faster
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006,
                              0.003, 0.0015, 0.0008, 0.0004,
                              0.0002, 0.0001, 0.00005, 0.00002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        # Scale iterations with remaining time and n
        est_time_per_iter = 0.004 * (n / 2048.0)
        max_iter = min(10000, max(200, int(remaining / (est_time_per_iter * 20))))
        
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _perturb_structured(a, rng, mode=0):

    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)

    if mode == 0:

    elif mode == 1:

            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2 * np.pi * freq * t / n + phase)
    elif mode == 2:
        # High-frequency noise
        z = z + 0.015 * rng.standard_normal(n)
    elif mode == 3:
        # Block perturbations
        n_blocks = 8
        block_sz = n // n_blocks
        for b in range(n_blocks):
            start = b * block_sz
            end = min(n, start + block_sz)
            z[start:end] += rng.uniform(-0.02, 0.02)
    elif mode == 4:
        # Taper modification
        taper = np.linspace(0, 1, n)
        z = z + 0.02 * rng.uniform(0.5, 2.0) * (taper - 0.5)
    elif mode == 5:

        # Positions in a that contribute to low-conv lags get boosted

def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

        try_update(a_cd, "phase4_coord")

    # === Phase 5: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 5: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    restart_modes = [0, 1, 2, 3, 4, 5, 0, 1]
    for idx, mode in enumerate(restart_modes):
        if time.monotonic() > deadline - 80:
            break
        p_start = _perturb_structured(best_seq, rng, mode=mode)
        sc_start = _score(p_start)
        
        # Run L-BFGS on perturbed
        remaining = deadline - 80 - time.monotonic()
        if remaining < 30:
            break
        a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=60.0)
        try_update(a_r, f"restart_{idx}_m{mode}")
        
        # Quick Nesterov polish
        if time.monotonic() < deadline - 50 and _score(a_r) < best_score_val + 0.01:
            a_r2, _ = _nesterov_subgrad(a_r, deadline - 40,
                                         max_steps=3000, rel_tol=1e-3,
                                         active_frac=0.05)
            try_update(a_r2, f"restart_{idx}_m{mode}_ng")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Different lengths (t={time.monotonic()-t_start:.1f}s) ===")
    current_n = len(best_seq)
    for target_n in [2048, 3072, 5120, 6144]:
        if time.monotonic() > deadline - 40:
            break
        if target_n == current_n:
            continue
        print(f"  Trying n={target_n}...")
        a_interp = _interpolate_to_n(best_seq, target_n)
        sc_interp = _score(a_interp)
        try_update(a_interp, f"interp_n{target_n}")
        if time.monotonic() < deadline - 25:
            a_opt = _lbfgs_schedule(a_interp, deadline, min_time_remaining=20.0)
            try_update(a_opt, f"opt_n{target_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    for rnd in range(3):
        if time.monotonic() > deadline - 5:
            break
        # Very tight active set refinement
        a_f, _ = _refine_active_set(best_seq, deadline - 5, max_iters=3000)
        try_update(a_f, f"final_as_{rnd}")
        
        # Quick L-BFGS with tiny beta
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        a_f2 = _lbfgs_run(best_seq, mb * 1e-7, n, max_iter=2000)
        try_update(a_f2, f"final_lbfgs_{rnd}")

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
