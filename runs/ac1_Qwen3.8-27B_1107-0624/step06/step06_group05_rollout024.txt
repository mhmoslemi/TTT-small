"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses aggressive FISTA continuation, exact Nesterov subgradient, Adam multi-beta,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_single(a, k_star):
    n = len(a)
    a_padded = np.zeros(2 * n - 1)
    a_padded[:n] = a
    grad = np.zeros(n)
    for i in range(n):
        ki = k_star - i
        if 0 <= ki < n:
            grad[i] = 2.0 * a[ki]
        else:
            grad[i] = 0.0
    return grad


def _fista_opt(a_init, beta, deadline, max_iter, line_search=True):

    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)

        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

            try_count = 0

                a_next = _proj_simplex(y - step * grad)

def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = float(_full_convolve(a_init, a_init).max())
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(100, int(max_time * 20)), 'gtol': 1e-14, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = float(_full_convolve(a_opt, a_opt).max())
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge_weighted":
        t = np.linspace(0, 1, n)
        a = 1.0 / (0.1 + t * (1.0 - t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.15) ** 2) + np.exp(-20.0 * (t - 0.85) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interpolate_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    beta = 0.05 * max_c0
    n_phases = 30

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        if remaining < 1:
            break
        per_time = min(remaining / max(1, n_phases - phase), 20.0)
        max_it = max(2000, int(per_time * 5000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = np.maximum(a_work, 0.0)
        s_w = a_work.sum()
        if s_w > 1e-30:
            a_work = a_work / s_w
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 5 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta *= 0.35
        if beta < 1e-12:
            beta = 1e-12

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Nesterov exact subgradient ===
    print("Phase 2: Nesterov exact subgradient")
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end - 10:
        a_ns, _ = _nestеров_subgrad(a_work, phase2_end, max_iter=200000)
        sc_ns = _score(a_ns)
        if sc_ns < best_score:
            best_score = sc_ns
            best_seq = a_ns.copy()
            print(f"  Nesterov subgrad: score={sc_ns:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_ns

    # === Phase 3: Adam multi-beta fine ===
    print("Phase 3: Adam multi-beta fine optimization")
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.001 * max_cc, 0.0002 * max_cc, 0.00005 * max_cc, 0.00001 * max_cc, 0.000002 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 10) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase3_end - 10:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.005, max_iter=300000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Try n=8192 ===
    print("Phase 4: Explore n=8192")
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 20:
        n_large = 8192
        a_large = _interpolate_to_length(best_seq, n_large)
        c_large = _full_convolve(a_large, a_large)
        beta_large = 0.02 * float(c_large.max())
        per_t = (phase4_end - time.monotonic() - 10)
        
        a_large, _ = _fista_opt(a_large, beta_large, time.monotonic() + per_t * 0.4, 200000)
        a_large = np.maximum(a_large, 0.0)
        a_large = a_large / a_large.sum()
        sc_l = _score(a_large)
        print(f"  n=8192 FISTA: score={sc_l:.10f}")
        if sc_l < best_score:
            best_score = sc_l
            best_seq = a_large.copy()
        
        if time.monotonic() < phase4_end - 10:
            c_l2 = _full_convolve(a_large, a_large)
            b_l2 = 0.0003 * float(c_l2.max())
            a_large, _ = _adam_opt(a_large, time.monotonic() + (phase4_end - time.monotonic() - 5) * 0.6,
                                   b_l2, 0.004, 200000)
            sc_l2 = _score(a_large)
            print(f"  n=8192 Adam: score={sc_l2:.10f}")
            if sc_l2 < best_score:
                best_score = sc_l2
                best_seq = a_large.copy()

    # === Phase 5: Multi-restart ===
    print("Phase 5: Multi-restart exploration")
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end - 20:
        n_restarts = 5
        per_r = (phase5_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            if i == 0:
                p = _perturb(best_seq, 0.01, rng)
            elif i == 1:
                p = _make_profile(len(best_seq), "edge_weighted")
                p = 0.7 * best_seq + 0.3 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                p = _make_profile(len(best_seq), "bimodal")
                p = 0.6 * best_seq + 0.4 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 3:
                shift = rng.integers(1, max(2, len(best_seq) // 1024))
                p = np.roll(best_seq, shift).copy()
                p = p / p.sum()
            else:
                p = _perturb(best_seq, 0.025, rng)
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.001 * max_cp
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.5, 50000)
            p = np.maximum(p, 0.0)
            p = p / p.sum()
            b_r2 = 0.0001 * float(_full_convolve(p, p).max())
            p, _ = _adam_opt(p, time.monotonic() + per_r * 0.5, b_r2, 0.003, 100000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-polish on best ===
    print("Phase 6: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 12:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.5,
                                 0.000003 * max_cp, 0.001, 200000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            if time.monotonic() < deadline - 8:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000001 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
