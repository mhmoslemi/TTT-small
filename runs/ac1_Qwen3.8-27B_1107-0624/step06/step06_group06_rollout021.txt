Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Multiplicative "flattening" updates (export method from radar waveform design)
2. Nesterov-accelerated subgradient with adaptive step size

import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk(a, c, n, topk=5):
    """Compute gradient using top-K peaks with equal weights."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    k = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k]
    # Weight: give more weight to higher peaks
    vals = c[top_idx]
    weights = vals / vals.sum()
    w = np.zeros_like(c)
    for i, idx in enumerate(top_idx):
        w[idx] = weights[i]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _flatten_mult(a, eta, topk=3, deadline=None):
    """Multiplicative flattening (export method). Returns new a and score."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return a, _score(a)
    # Get top-k peak lags
    idx = np.argsort(c)[::-1][:topk]
    # Build the "contribution" vector: for each i, sum of w_k * a[k-i]
    contrib = np.zeros(n)
    for k in idx:
        # contrib[i] += a[k-i] for valid i
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            contrib[i] += a[k - i]
    # Multiplicative update: a[i] *= exp(-eta * contrib[i])
    update = np.exp(-eta * contrib)
    a_new = a * update
    s = a_new.sum()
    if s < 1e-30:
        return a, _score(a)
    a_new = a_new / s
    return a_new, _score(a_new)


def _flatten_mult_fast(a, eta, topk=3):
    """Faster flattening using FFT for the contribution computation."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return a, _score(a)
    idx = np.argsort(c)[::-1][:topk]
    # contribution[i] = sum_{k in topk} a[k-i] (where valid)
    # This is: for each k, the vector [a[k-n+1], ..., a[k]] padded
    # Sum over k gives the contribution
    # Use the fact that contribution = sum_k P_k(a) where (P_k a)_i = a[k-i]
    # = cross-correlation of a with the "peak indicator"
    w = np.zeros(2 * n - 1)
    for k in idx:
        w[k] = 1.0
    # contribution[i] = sum_k w[k] * a[k-i] = (w * a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    contrib = wc[n - 1:2 * n - 1]
    # Ensure non-negative (numerical)
    contrib = np.maximum(contrib, 0)
    update = np.exp(-eta * contrib)
    a_new = a * update
    s = a_new.sum()
    if s < 1e-30:
        return a, _score(a)
    a_new = a_new / s
    return a_new, _score(a_new)


def _nesterov_subgrad(a_init, deadline, topk=5, lr_init=0.02, max_steps=50000):
    """Nesterov-accelerated subgradient with adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    y = z.copy()  # Nesterov lookahead
    x_prev = z.copy()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    mu = 0.5  # momentum
    patience = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        # Nesterov step
        z_curr = y + mu * (y - x_prev)
        a = _softmax(z_curr)
        c = _full_convolve(a, a)
        grad = _gradient_topk(a, c, n, topk=topk)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Adaptive step size (Polyak-like)
        c_max = float(c.max())
        lower_bound = 1.0 / (2 * n)  # score lower bound ~ 2n * (sum^2/(2n-1)) / sum^2 ≈ 1
        gap = max(c_max - lower_bound, 1e-10)
        step = lr * gap / max(grad_norm * grad_norm, 1e-20)
        step = min(step, 1.0)  # Cap step size

        y = z_curr - step * grad
        y = np.clip(y, -50, 50)
        y = y - y.mean()
        x_prev = z_curr

        # Update momentum
        mu = (1 + i + 1) / (mu + i + 1)

        # Periodic check
        if i % 100 == 0:
            a_check = _softmax(y)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr *= 1.02
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    # Reset to best
                    z_best = np.log(np.maximum(best_a, 1e-30))
                    z_best = z_best - z_best.mean()
                    y = z_best.copy()
                    x_prev = z_best.copy()
                    mu = 0.5
                    lr = lr_init
                    patience = 0

    a_final = _softmax(y)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(5000, max(20, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda zl: _smoobj(zl, beta, n),
                z,
                jac=lambda zl: _smograd(zl, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-11,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _flatten_iterations(a_init, deadline, max_iters=100000):
    """Run multiplicative flattening iterations with adaptive eta."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.1
    topk = 3
    patience = 0
    t_start = time.monotonic()

    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        a_new, sc_new = _flatten_mult_fast(a, eta, topk=topk)
        if sc_new < best_sc - 1e-12:
            improved = best_sc - sc_new
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
            patience = 0
            # Increase eta slightly on improvement
            eta = min(eta * 1.01, 0.5)
        else:
            patience += 1
            eta *= 0.95
            if patience > 50:
                # Reset eta and try different topk
                eta = max(0.01, eta * 0.5)
                topk = topk + 1 if topk < 8 else 3
                patience = 0
                if eta < 0.005:
                    break

        # Periodic re-optimization with L-BFGS
        if i > 0 and i % 5000 == 0 and time.monotonic() < deadline - 30:
            a_lbfgs = _run_lbfgs(a, deadline - 30, min_time_remaining=15)
            sc_lbfgs = _score(a_lbfgs)
            if sc_lbfgs < best_sc:
                best_sc = sc_lbfgs
                best_a = a_lbfgs.copy()
                a = a_lbfgs
                eta = 0.1

    return best_a, best_sc


def _alternating_lp(a_init, deadline, max_iters=50):
    """Alternating LP: minimize max_k (a*b)[k] s.t. sum(a)=1, a>=0, given b."""
    n = len(a_init)
    if n > 6000:
        # Too large for LP, fall back to subgradient
        return _nesterov_subgrad(a_init, deadline, topk=5, lr_init=0.015)

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        b = a.copy()
        # Build LP: min t s.t. (P_k b)^T a <= t, sum(a)=1, a>=0
        # A_ub has shape (2n-1, n+1): [P_k b | -1]
        # Use sparse for efficiency
        from scipy.sparse import lil_matrix
        A_lil = lil_matrix((2 * n - 1, n + 1), dtype=np.float64)
        for k in range(2 * n - 1):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                A_lil[k, i] = b[k - i]
            A_lil[k, n] = -1.0
        A_ub = A_lil.tocsr()
        b_ub = np.zeros(2 * n - 1)

        c_lin = np.zeros(n + 1)
        c_lin[n] = 1.0  # minimize t

        A_eq = np.ones((1, n + 1))
        A_eq[0, n] = 0.0
        b_eq = np.array([1.0])

        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(c_lin, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                a_new = res.x[:n]
                s = a_new.sum()
                if s > 1e-10:
                    a_new = a_new / s
                    sc_new = _score(a_new)
                    if sc_new < best_sc - 1e-10:
                        best_sc = sc_new
                        best_a = a_new.copy()
                        a = a_new
                    else:
                        # Try damped update
                        alpha = 0.5
                        a_damp = (1 - alpha) * a + alpha * a_new
                        a_damp = np.maximum(a_damp, 0)
                        sd = a_damp.sum()
                        if sd > 1e-10:
                            a_damp = a_damp / sd
                            sc_damp = _score(a_damp)
                            if sc_damp < best_sc - 1e-10:
                                best_sc = sc_damp
                                best_a = a_damp.copy()
                                a = a_damp
                        else:
                            break
                    if it % 5 == 0:
                        print(f"    ALP iter {it}: score={best_sc:.10f}")
        except Exception as e:
            print(f"    ALP failed at iter {it}: {e}")
            break

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.05, 0.10, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search combining L-BFGS, multiplicative flattening,
    alternating LP, Nesterov subgradient, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Phase 1: L-BFGS from parent ---
    print(f"\n[Phase 1] L-BFGS from parent (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=800.0)
    update_best(a_main, _score(a_main), "lbfgs_p1")

    # --- Phase 2: Multiplicative flattening (KEY NEW IDEA) ---
    print(f"\n[Phase 2] Multiplicative flattening (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 600:
        a_flat, sc_flat = _flatten_iterations(best_seq, deadline - 600, max_iters=200000)
        update_best(a_flat, sc_flat, "flatten")

    # --- Phase 3: L-BFGS polish after flattening ---
    print(f"\n[Phase 3] L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 500:
        a_pol = _run_lbfgs(best_seq, deadline, min_time_remaining=400.0)
        update_best(a_pol, _score(a_pol), "lbfgs_p3")

    # --- Phase 4: Nesterov subgradient ---
    print(f"\n[Phase 4] Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_nest, sc_nest = _nesterov_subgrad(best_seq, deadline - 400, topk=5, lr_init=0.02)
        update_best(a_nest, sc_nest, "nesterov")

    # --- Phase 5: More flattening from new best ---
    print(f"\n[Phase 5] Second flattening round (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        a_flat2, sc_flat2 = _flatten_iterations(best_seq, deadline - 350, max_iters=100000)
        update_best(a_flat2, sc_flat2, "flatten2")

    # --- Phase 6: Shrinkage ---
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 300)
        update_best(a_shr, sc_shr, "shrinkage")

    # --- Phase 7: Alternating LP (if n is manageable) ---
    print(f"\n[Phase 7] Alternating LP (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250 and len(best_seq) <= 6000:
        a_lp, sc_lp = _alternating_lp(best_seq, deadline - 250, max_iters=30)
        update_best(a_lp, sc_lp, "alternating_lp")

    # --- Phase 8: Multi-n exploration ---
    print(f"\n[Phase 8] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 3072, 4096, 5120]
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            s_init = a_init.sum()
            if s_init > 1e-10:
                a_init = a_init / s_init
            else:
                a_init = _make_profile(n_try, "tapered")
        else:
            a_init = best_seq.copy() if best_seq is not None else _make_profile(n_try, "tapered")

        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=60.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick flattening
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 40:
            a_fl, sc_fl = _flatten_iterations(a_opt, deadline - 40, max_iters=50000)
            update_best(a_fl, sc_fl, f"n_{n_try}_flat")

    # --- Phase 9: Diverse restarts ---
    print(f"\n[Phase 9] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        for i in range(4):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            elif i == 2:
                # Block swap
                n = len(best_seq)
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z - z.mean()
                bs = n // 8
                for _ in range(3):
                    ii = rng.integers(0, n - 2 * bs)
                    jj = rng.integers(0, n - 2 * bs)
                    if abs(ii - jj) > bs:
                        tmp = z[ii:ii + bs].copy()
                        z[ii:ii + bs] = z[jj:jj + bs]
                        z[jj:jj + bs] = tmp
                p_start = _softmax(z)
            else:
                # Antisymmetric perturbation
                n = len(best_seq)
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z - z.mean()
                anti = rng.standard_normal(n)
                anti = anti - anti[::-1]  # Make antisymmetric
                z = z + 0.01 * anti
                p_start = _softmax(z)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=15.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            # Quick flatten
            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 10:
                a_fl, sc_fl = _flatten_iterations(a_r, deadline - 10, max_iters=20000)
                update_best(a_fl, sc_fl, f"restart_{i}_flat")

    # --- Phase 10: Final combined polish ---
    print(f"\n[Phase 10] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # L-BFGS with very small beta
        n = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0005
        remaining = deadline - 5 - time.monotonic()
        if remaining > 10:
            max_iter = min(3000, max(50, int(remaining / 0.03)))
            try:
                result = minimize(
                    lambda zl: _smoobj(zl, beta_final, n),
                    z,
                    jac=lambda zl: _smograd(zl, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final flattening
        if time.monotonic() < deadline - 3:
            a_fl, sc_fl = _flatten_iterations(best_seq, deadline - 3, max_iters=30000)
            update_best(a_fl, sc_fl, "final_flat")

        # Final Nesterov
        if time.monotonic() < deadline - 2:
            a_nes, sc_nes = _nesterov_subgrad(best_seq, deadline - 2, topk=3, lr_init=0.01, max_steps=5000)
            update_best(a_nes, sc_nes, "final_nest")

    # --- Final validation and return ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
