    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_time_remaining=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.1, 0.05, 0.025, 0.0125, 0.00625,
                              0.003125, 0.0015625, 0.00078125,
                              0.000390625, 0.0001953125,
                              0.00009765625, 0.000048828125,
                              0.0000244140625, 0.00001220703125,
                              0.0000061035156])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 15:
            break

        est_iter_time = 0.002 * (n / 2048.0) ** 1.5
        max_iter = min(12000, max(200, int(remaining / (est_iter_time * 20))))

        a_new, success = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-16:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-5, max_steps=50000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + i * 0.0002)
        z = z - step * grad
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()

        if i % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.05
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 5:
                    lr *= 0.6
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _gs_flatten(a, deadline, max_iters=2000, alpha_start=1.0, alpha_min=0.3):
    """
    Gerchberg-Saxton inspired frequency domain flattening.
    Alternates between:
    - Time domain: enforce non-negativity and normalization
    - Frequency domain: flatten |A|^2 toward uniform
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    a_work = a.astype(np.float64).copy()
    best_a = a_work.copy()
    best_sc = _score(a_work)

    alpha = alpha_start
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Go to frequency domain
        A = np.fft.rfft(a_work, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flatten magnitude: push toward geometric mean
        log_mag = np.log(mag + 1e-30)
        log_mag_mean = log_mag.mean()
        # Blend toward flat
        log_mag_new = alpha * log_mag + (1.0 - alpha) * log_mag_mean
        mag_new = np.exp(log_mag_new)

        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        # Slowly reduce alpha (reduce flattening strength)
        alpha = max(alpha_min, alpha * 0.9995)

        # Occasionally try a "kick" - add small random phase perturbation
        if it % 100 == 50:
            rng = np.random.default_rng(it * 12345)
            phase_kick = rng.standard_normal(len(A)) * 0.01
            phase_kick[0] = 0
            A_kick = mag * np.exp(1j * (phase + phase_kick))
            a_kick = np.fft.irfft(A_kick, N)[:n]
            a_kick = np.maximum(a_kick, 0.0)
            sk = a_kick.sum()
            if sk > 1e-15:
                a_kick = a_kick / sk
                sc_kick = _score(a_kick)
                if sc_kick < best_sc:
                    best_sc = sc_kick
                    best_a = a_kick.copy()

        a_work = a_new

    return best_a, best_sc


def _pair_local_search(a, deadline, max_rounds=5000):
    """
    Local search that moves mass between pairs of coordinates.
    For each pair (i,j), try transferring a fraction of a[j] to a[i].
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(42)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Select a pair of coordinates
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n

        # Try several transfer amounts
        for frac in [0.05, 0.1, 0.2, 0.01, 0.02]:
            transfer = frac * min(best_a[i], best_a[j])
            if transfer < 1e-15:
                continue

            a_try = best_a.copy()
            # Move mass from larger to smaller (to flatten)
            if best_a[i] >= best_a[j]:
                a_try[i] -= transfer
                a_try[j] += transfer
            else:
                a_try[i] += transfer
                a_try[j] -= transfer

            if a_try[i] < 0 or a_try[j] < 0:
                continue

            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try.copy()
                best_sc = sc_try

        # Also try "boost" moves: increase one, decrease another proportionally
        if rnd % 3 == 0:
            i2 = int(rng.integers(0, n))
            j2 = int(rng.integers(0, n))
            if i2 == j2:
                j2 = (j2 + 1) % n
            for eps in [0.01, 0.005, 0.02]:
                a_try = best_a.copy()
                a_try[i2] *= (1.0 + eps)
                a_try[j2] *= (1.0 - eps)
                if a_try[i2] < 0 or a_try[j2] < 0:
                    continue
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_a = a_try.copy()
                    best_sc = sc_try

    return best_a, best_sc


def _directional_search(a, deadline, n_directions=50):
    """
    Search along random 2D directions in log-space.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(99)

    for d in range(n_directions):
        if time.monotonic() > deadline:
            break

        # Random direction in log-space
        v = rng.standard_normal(n)
        v = v / np.linalg.norm(v)

        # Line search in both directions
        for sign in [1.0, -1.0]:
            best_t = 0.0
            t = 0.01
            for _ in range(20):
                z = np.log(np.maximum(best_a, 1e-15))
                z = z + sign * t * v
                z = z - z.mean()
                a_try = _softmax(z)
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    best_t = t
                    t *= 1.5
                else:
                    t *= 0.5
                    if t < 1e-6:
                        break

    return best_a, best_sc


def _perturb_structured(a, rng, mode):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)

    if mode == "fourier_low":
        for k in [1, 2, 3, 5, 7, 11]:
            if k < n // 2:
                amp = 0.02 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2 * np.pi)
                z = z + amp * np.cos(2 * np.pi * k * t / n + phase)
    elif mode == "fourier_high":
        for _ in range(5):
            k = int(rng.integers(n // 4, max(n // 4 + 1, n // 2 - 1)))
            amp = 0.01 * rng.uniform(0.5, 2.0)
            phase = rng.uniform(0, 2 * np.pi)
            z = z + amp * np.cos(2 * np.pi * k * t / n + phase)
    elif mode == "gauss":
        z = z + 0.03 * rng.standard_normal(n)
    elif mode == "block":
        n_blocks = int(rng.integers(4, 16))
        block_size = n // n_blocks
        for _ in range(n_blocks):
            start = int(rng.integers(0, max(1, n - block_size)))
            end = min(n, start + block_size)
            z[start:end] += rng.uniform(-0.02, 0.02)
    elif mode == "mirror_break":
        # Break the mirror symmetry
        half = n // 2
        pert = rng.standard_normal(half) * 0.01
        z[:half] += pert
        z[half:] -= pert[::-1]

    return _softmax(z)


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with aggressive beta continuation,
    frequency-domain flattening (GS-inspired), pair-based local search,
    directional search, and structured multi-restart diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = np.ones(4096) / 4096.0
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Aggressive L-BFGS ===
    print(f"\n=== Phase 1: Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=300.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Frequency domain flattening ===
    print(f"\n=== Phase 2: GS Flattening (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_gs, sc_gs = _gs_flatten(best_seq, deadline - 200, max_iters=5000,
                                  alpha_start=1.0, alpha_min=0.5)
        try_update(a_gs, "phase2_gs")

    # === Phase 3: Subgradient polishing (tight) ===
    print(f"\n=== Phase 3: Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 200,
                                        tol_frac=1e-5, max_steps=50000, lr_init=0.025)
        try_update(a_sg, "phase3_sg")

    # === Phase 4: Pair local search ===
    print(f"\n=== Phase 4: Pair Local Search (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_ps, sc_ps = _pair_local_search(best_seq, deadline - 150, max_rounds=10000)
        try_update(a_ps, "phase4_pair")

    # === Phase 5: Directional search ===
    print(f"\n=== Phase 5: Directional Search (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_ds, sc_ds = _directional_search(best_seq, deadline - 100, n_directions=100)
        try_update(a_ds, "phase5_dir")

    # === Phase 6: Second round of optimization ===
    print(f"\n=== Phase 6: Second optimization round (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_p6 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=150.0)
        try_update(a_p6, "phase6_lbfgs")

        if best_seq is not None and time.monotonic() < deadline - 100:
            a_gs2, _ = _gs_flatten(best_seq, deadline - 100, max_iters=3000,
                                   alpha_start=0.9, alpha_min=0.6)
            try_update(a_gs2, "phase6_gs")

    # === Phase 7: Multi-restart with diverse perturbations ===
    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        pert_modes = ["fourier_low", "fourier_high", "gauss", "block", "mirror_break",
                      "fourier_low", "gauss", "block"]

        for i, mode in enumerate(pert_modes):
            if time.monotonic() > deadline - 60:
                break

            p_start = _perturb_structured(best_seq, rng, mode)
            sc_start = _score(p_start)
            print(f"  Restart [{mode}_{i}]: start={sc_start:.10f}")

            # Quick L-BFGS
            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=50.0)
            try_update(a_r, f"restart_{mode}_{i}")

            # GS flatten on result
            if time.monotonic() < deadline - 45 and sc_start < best_score_val + 0.01:
                a_r2, _ = _gs_flatten(a_r, deadline - 45, max_iters=1000,
                                      alpha_start=0.95, alpha_min=0.7)
                try_update(a_r2, f"restart_{mode}_{i}_gs")

            # Quick subgradient
            if time.monotonic() < deadline - 40 and _score(a_r) < best_score_val + 0.005:
                a_r3, _ = _subgrad_z_polish(a_r, deadline - 40,
                                            tol_frac=1e-5, max_steps=5000, lr_init=0.02)
                try_update(a_r3, f"restart_{mode}_{i}_sg")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_cur = len(best_seq)
        target_ns = sorted(set([n_cur // 2, n_cur * 3 // 2, n_cur * 2,
                                n_cur - n_cur // 16, n_cur + n_cur // 16]))
        target_ns = [tn for tn in target_ns if 512 <= tn <= 16384 and tn != n_cur]

        for target_n in target_ns[:3]:
            if time.monotonic() > deadline - 60:
                break
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=40.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final aggressive polish ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(6):
            if time.monotonic() > deadline - 3:
                break

            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 5e-7
            max_iter = min(8000, max(500, int((deadline - 3 - time.monotonic()) / 0.003)))
            a_f, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            if time.monotonic() < deadline - 3:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 3,
                                           tol_frac=1e-6, max_steps=10000, lr_init=0.01)
                try_update(a_s, f"final_sg_{rnd}")

            if time.monotonic() < deadline - 3:
                a_g, _ = _gs_flatten(best_seq, deadline - 3, max_iters=2000,
                                     alpha_start=0.99, alpha_min=0.9)
                try_update(a_g, f"final_gs_{rnd}")

            if time.monotonic() < deadline - 3:
                a_p, _ = _pair_local_search(best_seq, deadline - 3, max_rounds=3000)
                try_update(a_p, f"final_pair_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
