    n = len(a)
    c = 1.0 / np.maximum(a, 1e-30)  # c_i = 1/a[i]
    y = np.zeros(2*n - 1)

            y[i+n-1] += deficit
            running_sum += deficit

    return y, dual_obj

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _solve_dual_lp_greedy(a):
    """Solve the dual LP greedily. The dual is:
    min M*sum(y) s.t. sum_{j=0}^{n-1} y[i+j] >= 1/a[i] for all i, y >= 0.
    This is an interval covering problem solvable greedily in O(n^2).
    """
    n = len(a)
    c = 1.0 / np.maximum(a, 1e-30)
    y = np.zeros(2 * n - 1)
    for i in range(n):
        window_sum = y[i:i + n].sum()
        if window_sum < c[i]:
            deficit = c[i] - window_sum
            y[i + n - 1] += deficit
    return y


def _compute_conv_grad(a, w):
    """Compute gradient of sum_k w_k * conv(a,a)[k] with respect to a.
    conv(a,a)[k] = sum_i a[i]*a[k-i]
    d/d a[j] of sum_k w_k * conv(a,a)[k] = 2 * sum_k w_k * a[k-j]
    This is 2 * (w * a_rev)[j] where a_rev[a_rev[j] = a[n-1-j]]
    Actually: sum_k w_k * a[k-j] = (w convolved with reversed a)[n-1+j]... 
    Let me be more careful.
    
    sum_k w_k * a[k-j] where k-j must be in [0, n-1], i.e., k in [j, j+n-1]
    = sum_{k=j}^{j+n-1} w_k * a[k-j]
    = sum_{m=0}^{n-1} w[j+m] * a[m]
    = (w_shifted * a)[j] where w_shifted is w shifted...
    
    Actually this is the cross-correlation of w and a.
    Let's use FFT: result[j] = sum_m a[m] * w[j+m]
    This is the correlation of a with w (with w shifted).
    
    We can compute this as: if we define w_ext of length 2n-1, then
    result = ifft(fft(a, 2n-1) * conj(fft(w, 2n-1))) appropriately sliced.
    
    Actually, the simplest: result[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    This is the cross-correlation. We can compute via:
    result = _full_convolve(a, w)[n-1 : 2n-1]... no.
    
    Let me just use: _full_convolve(a, w) gives conv(a,w) of length n + (2n-1) - 1 = 3n-2.
    That's not right either since w has length 2n-1.
    
    Let me just do it directly with FFT:
    result[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    
    Define: flip w to get w_flip[t] = w[2n-2-t] (length 2n-1)
    Then (a * w_flip)[j] = sum_m a[m] * w_flip[j-m] = sum_m a[m] * w[2n-2-j+m]
    That's not quite what we want.
    
    Alternative: result = _full_convolve(w, a) gives length (2n-1) + n - 1 = 3n-2.
    result[k] = sum_i w[i] * a[k-i]
    We want: sum_m a[m] * w[j+m] = sum_m a[m] * w[j+m]
    Let k = j + 2(n-1), then... this is getting confusing.
    
    Let me just use the correlation directly:
    result[j] = <a, w[j:j+n]>
    
    I'll compute it using the fact that it's a cross-correlation:
    corr[j] = sum_{m=0}^{n-1} a[m] * w[j+m] for j = 0, ..., 2n-2-n = n-2... 
    
    Actually j goes from 0 to (2n-1)-n = n-1. So result has length n.
    result[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    
    This is the cross-correlation of a (length n) and w (length 2n-1) at lag j.
    
    Using FFT: 
    Let N >= n + (2n-1) - 1 = 3n-2
    FFT_a = fft(a, N), FFT_w = fft(w, N)
    corr_full = ifft(FFT_a * conj(FFT_w), N)
    Then corr[j] = corr_full[j] for j = 0, ..., n-1 (the "end" of the correlation)
    
    Wait, I need to be more careful with the definition.
    
    Cross-correlation: (a ⋆ w)[τ] = sum_t a[t+τ] * conj(w[t])
    
    What I want: result[j] = sum_{m=0}^{n-1} a[m] * w[j+m]
    
    Let me define b = a reversed: b[i] = a[n-1-i]
    Then result[j] = sum_{m=0}^{n-1} b[n-1-m] * w[j+m]
    Let p = n-1-m: result[j] = sum_{p=0}^{n-1} b[p] * w[j+n-1-p]
    = conv(b, w)[j+n-1]
    
    So result = conv(b, w)[n-1 : n-1+n] = _full_convolve(a[::-1], w)[n-1 : 2n-1]
    
    Let me verify: conv(b,w)[k] = sum_p b[p]*w[k-p]
    k = j+n-1: sum_p b[p]*w[j+n-1-p] = sum_p a[n-1-p]*w[j+n-1-p]
    Let q = n-1-p: sum_q a[q]*w[j+q] ✓
    
    Great! So result = _full_convolve(a[::-1], w)[n-1 : 2*n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    cw = _full_convolve(a_rev, w)
    return 2.0 * cw[n - 1:2 * n - 1]


def _fista_active_set(a_init, deadline, max_iter, top_k_frac=0.1, beta_init=None):
    """FISTA with active-set focused gradient.
    Only uses the top-k% largest convolution entries for the gradient.
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    if beta_init is None:
        beta = 0.01 * max_c
    else:
        beta = beta_init
    L = 4.0 * n * max_c / beta
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    top_k = max(1, int(n * top_k_frac))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())

        # Active set: top-k entries
        threshold = c_max * (1.0 - top_k_frac)
        active_mask = c_y >= threshold
        if active_mask.sum() < 1:
            active_mask[int(np.argmax(c_y))] = True

        # Softmax over active set with adaptive beta
        c_active = c_y[active_mask]
        c_shifted = (c_active - c_active.max()) / beta
        e = np.exp(c_shifted)
        z = float(e.sum())
        if z < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = np.zeros_like(c_y)
            w[active_mask] = e / z

        grad = _compute_conv_grad(y, w)

        # Adaptive Lipschitz
        L = 4.0 * n * c_max / beta
        step = 1.0 / L

        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            L = 4.0 * n * float(c_new.max()) / beta
            step = 1.0 / L
            # Adaptive beta: decrease over time
            beta *= 0.999

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _lp_blend_iteration(a, n_iter=5):
    """LP-based blend iteration.
    Solve the dual LP, use it to construct a direction, and blend.
    """
    n = len(a)
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    for it in range(n_iter):
        if time.monotonic() > 1800:
            break
        c = _full_convolve(a_curr, a_curr)
        M = float(c.max())
        S = float(a_curr.sum())

        # Solve dual LP
        y = _solve_dual_lp_greedy(a_curr)

        # Use complementary slackness to construct b
        # b_i > 0 where dual constraint i is tight
        # Dual constraint i: sum_{j=0}^{n-1} y[i+j] = 1/a[i]
        c_rhs = 1.0 / np.maximum(a_curr, 1e-30)
        dual_vals = np.array([y[i:i + n].sum() for i in range(n)])
        tol = 1e-6
        tight_mask = np.abs(dual_vals - c_rhs) < tol * c_rhs + 1e-10

        # For tight constraints, b_i > 0.
        # We need to find b such that conv(a,b)[k] = M where y_k > 0,
        # and b_i = 0 where dual constraint is not tight.
        # This is a linear system. Let's solve it.
        # Actually, let's use a simpler heuristic:
        # b_i = c_rhs[i] for tight i, 0 otherwise, then scale.
        b = np.zeros(n)
        if tight_mask.sum() > 0:
            b[tight_mask] = 1.0
            # Scale so that conv(a, b) <= M
            ab = _full_convolve(a_curr, b)
            ab_max = float(ab.max())
            if ab_max > M and ab_max > 0:
                b = b * (M / ab_max)
        
        if b.sum() < 1e-15:
            # Fallback: use y-weighted uniform
            b = np.ones(n)

        # Choose step size by bisection
        # We want max(conv(a + t*b, a + t*b)) <= M * (1 + small)
        # conv(a+t*b, a+t*b) = conv(a,a) + 2t*conv(a,b) + t^2*conv(b,b)
        # At t=0, max = M. We want to keep it bounded.
        # For small t, the max increases by about 2t*(max of conv(a,b) at the argmax of conv(a,a))
        
        # Try a few step sizes
        best_t = 0.0
        best_sc = best_score
        for t in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3]:
            a_try = a_curr + t * b
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc = _score_norm(a_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
                best_a = a_try.copy()

        if best_t == 0.0:
            break

        a_curr = best_a.copy()
        a_curr = a_curr / a_curr.sum()
        if best_sc < best_score:
            best_score = best_sc
            best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        phase_iter = max(5000, int((deadline - time.monotonic()) * 2000 / max(1, len(betas))))
        while it_in_phase < phase_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            # Softmax
            z_max = z.max()
            e = np.exp(z - z_max)
            s = e.sum()
            if s < 1e-300:
                a = np.ones(n) / n
            else:
                a = e / s

            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e_c = np.exp(c_shifted)
            s_c = float(e_c.sum())
            if s_c < 1e-300 or not np.isfinite(s_c):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e_c / s_c

            # Gradient in log-space
            # d/dz_j of score = (1/a_j) * d/da_j of score (chain rule for softmax)
            # d/da_j of (2n * max_c / ...) ≈ 2n * 2 * (w conv a_rev)[n-1+j]
            # So d/dz_j = (1/a_j) * 4n * (w conv a_rev)[n-1+j]
            # But we're optimizing score_norm = 2n * max_c (with sum=1 constraint)
            # The gradient of score_norm with respect to a is 4n * (w * a_rev) sliced
            # The gradient with respect to z (log-space) involves the softmax Jacobian
            
            # Actually for softmax parameterization a = softmax(z), sum(a) = 1 automatically.
            # d/da_j (sum_k w_k c_k) where c_k = conv(a,a)[k]
            # = sum_k w_k * d c_k / d a_j
            # = sum_k w_k * 2 * a[k-j] (as before)
            # = 2 * (w_shifted * a)[j]
            
            # For the log-space: d/dz_j = sum_i (da_i/dz_j) * (d score / da_i)
            # da_i/dz_j = a_i * (delta_{ij} - a_j)
            # So d/dz_j = a_j * (d score / da_j) - a_j * sum_i a_i * (d score / da_i)
                        # = a_j * (g_j - dot(a, g))
            # where g = gradient of score w.r.t. a
            
            g = _compute_conv_grad(a, w)
            mean_g = float(np.dot(a, g))
            grad_z = a * (g - mean_g)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = a.copy()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = a.copy()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _minimax_active_set(a_init, deadline, max_iter=5000):
    """Minimax approach: iteratively reduce the top few convolution peaks."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    lr = 0.01
    decay = 0.9995

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())

        # Find top-k peaks
        k = max(5, n // 100)
        # Get the k largest values and their positions
        top_vals = np.sort(c)[::-1][:k]
        threshold = top_vals[-1]
        active = c >= threshold * 0.99

        # Compute gradient of the sum of active convolution entries
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        g = _compute_conv_grad(a_curr, w)

        # Projected gradient step
        a_next = a_curr - lr * g
        a_next = np.maximum(a_next, 0.0)
        s = a_next.sum()
        if s > 1e-15:
            a_next = a_next / s
        else:
            a_next = a_curr.copy()

        sc = _score_norm(a_next)
        if sc < best_score:
            best_score = sc
            best_a = a_next.copy()
            a_curr = a_next.copy()
        else:
            # Reduce learning rate
            lr *= 0.95
            if lr < 1e-8:
                break

        lr *= decay

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20.0 * (t - 0.25) ** 2) + np.exp(-20.0 * (t - 0.75) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "endpoint_heavy":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        rng = np.random.default_rng(42)
        a = rng.random(n) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    z_shift = z - z.max()
    e = np.exp(z_shift)
    return e / e.sum()


def _spectral_init(n, rng):
    """Initialize with near-flat spectrum."""
    N = _next_pow2(3 * n)
    magnitude = np.ones(n)
    phase = rng.random(n) * 2 * np.pi
    spec = np.zeros(N // 2 + 1, dtype=complex)
    spec[:n] = magnitude * np.exp(1j * phase)
    # Make it conjugate symmetric
    for i in range(1, min(n, N // 2) + 1):
        if i < N // 2:
            spec[N // 2 + i] = np.conj(spec[N // 2 - i]) if N // 2 + i < len(spec) else 0
    a = np.fft.irfft(spec, N)[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

        a_opt, sc_opt = _adam_logspace(a_work, adam_dl, betas_fine, lr_init=0.01,
                                       max_iter_total=200000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f}")

    # === Phase 5: Multi-restart diversification ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n=== Phase 5: Multi-restart (score={best_score:.10f}) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        restarts = [
            ("perturb_small", _perturb_log, (0.02,)),
            ("perturb_med", _perturb_log, (0.05,)),
            ("endpoint_heavy_blend", None, None),
            ("spectral", None, None),
        ]

        per_restart = (deadline - time.monotonic() - 30.0) / len(restarts)

        for ri, (name, fn, args) in enumerate(restarts):
            if time.monotonic() > deadline - 25:
                break

            if fn is not None:
                sigma = args[0]
                p_start = fn(a_base, sigma, rng)
            elif name == "endpoint_heavy_blend":
                alt = _make_profile(n_best, "endpoint_heavy")
                p_start = 0.7 * a_base + 0.3 * alt
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif name == "spectral":
                p_start = _spectral_init(n_best, rng)
                p_start = 0.5 * a_base + 0.5 * p_start
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                continue

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())

            # Quick FISTA then Adam
            r_dl = time.monotonic() + max(5.0, per_restart * 0.4)
            p_best, _ = _fista_active_set(p_start, r_dl, max_iter=10000,
                                          top_k_frac=0.05, beta_init=0.005 * max_cs)
            p_best = p_best / p_best.sum()

            a_dl = time.monotonic() + max(3.0, per_restart * 0.5)
            bs = [0.001 * max_cs, 0.0003 * max_cs]
            p_best, _ = _adam_logspace(p_best, a_dl, bs, lr_init=0.008,
                                       max_iter_total=50000)
            sc_r = _score(p_best)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_best.copy()
                print(f"  Restart {ri} ({name}): score={sc_r:.10f}")
            else:
                print(f"  Restart {ri} ({name}): score={sc_r:.10f} (no improvement)")

    # === Phase 6: Try different lengths ===
    if best_seq is not None and time.monotonic() < deadline - 60:
        print("\n=== Phase 6: Length exploration ===")
        n_best = len(best_seq)
        other_lengths = []
        if n_best <= 4096:
            other_lengths.append(8192)
        if n_best >= 4096:
            other_lengths.append(2048)
        if n_best != 4096:
            other_lengths.append(4096)

        for n_try in other_lengths:
            if time.monotonic() > deadline - 40:
                break

            # Interpolate parent to new length
            x0 = np.linspace(0, 1, n_best)
            x1 = np.linspace(0, 1, n_try)
            a_interp = np.interp(x1, x0, best_seq)
            a_interp = np.maximum(a_interp, 1e-12)
            a_interp = a_interp / a_interp.sum()

            c_int = _full_convolve(a_interp, a_interp)
            max_ci = float(c_int.max())

            # Quick optimization
            len_dl = min(deadline - 40, time.monotonic() + 60.0)
            a_new, _ = _fista_active_set(a_interp, len_dl, max_iter=15000,
                                         top_k_frac=0.05, beta_init=0.01 * max_ci)
            a_new = a_new / a_new.sum()
            sc_new = _score(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                print(f"  Length {n_try}: score={sc_new:.10f}")
            else:
                print(f"  Length {n_try}: score={sc_new:.10f} (no improvement)")

    # === Phase 7: Ultra-fine final polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Phase 7: Final ultra-fine polish (score={best_score:.10f}) ===")
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00005 * max_cp, 0.00001 * max_cp, 0.000003 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 15:
            a_pol, sc_pol = _adam_logspace(a_pol, time.monotonic() + per_t,
                                           betas_ult, lr_init=0.002,
                                           max_iter_total=80000)
            a_pol = a_pol / a_pol.sum()
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f}")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
