Multi-phase optimizer with Newton active-set method, variance-aware objectives,
coarse-to-fine scaling, aggressive beta continuation, and diverse restarts

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta

    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _variance_obj(a):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    return float(np.mean((c - mean_c) ** 2))


def _variance_grad(a, n):
    c = _full_convolve(a, a)
    mean_c = np.mean(c)
    centered = c - mean_c
    dcd_a = _full_convolve(a[::-1], centered)
    g = 2.0 * dcd_a[n - 1:2 * n - 1]
    g = g - np.mean(g)
    return g


def _hessian_matvec(v, active, n):
    """Compute H*v where H = 2*sum_{k in active} P_k.
    (Hv)_i = 2 * sum_j v[j] * active[i+j] (for valid indices)
    This is a cross-correlation of v with the active indicator.
    """
    # active is defined on indices 0..2n-2
    N = _next_pow2(4 * n)
    fv = np.fft.rfft(v, N)
    fa = np.fft.rfft(active, N)
    # We want sum_j v[j] * active[i+j]
    # This is a "generalized correlation": corr(i) = sum_j v[j] * active[i+j]
    # = (v * flip(active))[i + len(active) - 1] ... let me use a different approach
    # corr(i) = sum_j v[j] * active[i+j]
    # Let's compute via FFT: it's the convolution of v with reversed active
    # Actually: sum_j v[j] * active[i+j] = (v convolved with active_shifted)[i]
    # where active_shifted[k] = active[2n-2-k] (reversed)
    # No wait. Let me think again.
    # sum_j v[j] * active[i+j] where i+j ranges over 0..2n-2
    # Let m = i+j, so j = m-i. sum_m v[m-i] * active[m] where 0<=m<=2n-2 and 0<=m-i<=n-1
    # This is (flip(v) * active)[i + n - 1] where flip(v)[k] = v[n-1-k]
    # Actually let's just do: result[i] = sum_j v[j] * active[i+j]
    # = cross-correlation. Use FFT:
    # corr[i] = sum_j v[j] * active[i+j]
    # = (v * active_rev)[i + (2n-2) - (n-1)] ... this is getting confusing.
    # Let me just use: result = ifft(fft(v) * conj(fft(active))) which gives
    # sum_j v[j] * conj(active)[i-j] ... not quite what we want.
    
    # Simplest: use the fact that for each i, (Hv)_i = 2 * sum_j v[j] * active[i+j]
    # This is a "generalized inner product" that can be computed as:
    # For i = 0: sum_j v[j] * active[j]  (j from 0 to n-1, active from 0 to 2n-2)
    # For i = 1: sum_j v[j] * active[1+j]
    # ...
    # This is like a "sliding dot product" = cross-correlation of v and active.
    # corr[i] = sum_j v[j] * active[i+j] = (v ⊛ active)[i]
    # where ⊛ denotes cross-correlation (not convolution).
    # Using FFT: corr = ifft(conj(fft(v, N)) * fft(active, N)) ... 
    # Actually numpy/scipy convention: 
    # sum_j v[j] * active[j+i] = ifft(fft(active) * conj(fft(v)))[i] ... 
    # Let me just verify with a small example mentally.
    # a = [1,2,3], b = [4,5,6,7,8] (len 3 and 5)
    # corr[0] = 1*4 + 2*5 + 3*6 = 4+10+18 = 32
    # corr[1] = 1*5 + 2*6 + 3*7 = 5+12+21 = 38
    # corr[2] = 1*6 + 2*7 + 3*8 = 6+14+24 = 44
    # This is (a_rev convolved with b)[n_a-1 + i] where a_rev = [3,2,1]
    # a_rev * b = [3*4, 3*5+2*4, 3*6+2*5+1*4, ...] = [12, 23, 38, ...]
    # Hmm that gives [12, 23, 38, ...] which doesn't match.
    # Let me try: corr[i] = sum_j a[j] * b[j+i]
    # = ifft(fft(a, N) * conj(fft(b, N)))[i] ... 
    # Actually the standard formula is: (a ⊛ b)[i] = sum_j a[j] * b[j+i]
    # In terms of convolution: a ⊛ b = flip(a) * b (where * is convolution)
    # So fft(a ⊛ b) = fft(flip(a)) * fft(b) = conj(fft(a)) * fft(b) (for real signals, approximately)
    # Actually for real signals: fft(flip(a))[k] = conj(fft(a)[k]) only if we flip around the center.
    # Let me just use scipy or a direct FFT approach.
    
    # Direct approach: pad both to length N, compute
    # result = ifft(conj(fft(padded_v)) * fft(padded_active))
    # This gives sum_j v[j] * active[j - i] (with circular wrap)
    # Which is NOT what we want. We want sum_j v[j] * active[j + i].
    
    # Let me use: result[i] = sum_j v[j] * active[i+j]
    # = (v_reversed * active)[n-1+i] where v_reversed[k] = v[n-1-k]
    # Because (v_rev * active)[m] = sum_k v_rev[k] * active[m-k] = sum_k v[n-1-k] * active[m-k]
    # Let j = n-1-k, k = n-1-j: = sum_j v[j] * active[m-(n-1-j)] = sum_j v[j] * active[m-n+1+j]
    # So if m = n-1+i: = sum_j v[j] * active[i+j]. Yes!
    
    v_rev = v[::-1]
    fa = np.fft.rfft(v_rev, N)
    fb = np.fft.rfft(active, N)
    r = np.fft.irfft(fa * fb, N)
    result = r[n - 1:2 * n - 1]
    return 2.0 * result


def _cg_solve(hess_vec, b, n, max_iter=200, tol=1e-10):
    """Conjugate gradient to solve Hx = b, where H is given by hess_vec."""
    x = np.zeros(n)
    r = b.copy()
    p = r.copy()
    rs = float(np.dot(r, r))
    b_norm = float(np.linalg.norm(b))
    if b_norm < 1e-30:
        return x
    for _ in range(max_iter):
        Hp = hess_vec(p)
        pH = float(np.dot(p, Hp))
        if abs(pH) < 1e-30:
            break
        alpha = rs / pH
        x = x + alpha * p
        r = r - alpha * Hp
        r_new = float(np.dot(r, r))
        if np.sqrt(r_new) < tol * b_norm:
            break
        beta_cg = r_new / rs if rs > 1e-30 else 0.0
        p = r + beta_cg * p
        rs = r_new
    return x


def _newton_active_set(a_init, deadline, max_iter=300):
    """Newton's method on the active set subproblem with CG linear solve."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Active set: indices where c is close to max
        tol = max(0.001 * c_max, 1e-12)
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Compute gradient of sum_{k in active} c_k
        # grad_k = 2 * a flipped, correlated with active
        # grad[i] = 2 * sum_j a[j] * ... 
        # Actually: d c_k / d a_i = 2 * a[k-i] (for valid k-i)
        # So grad[i] = sum_{k in active} 2 * a[k-i] = 2 * _hessian_matvec(a, active, n) / 2
        # Wait: _hessian_matvec(v, active, n) gives 2 * sum_j v[j] * active[i+j]
        # We want: grad[i] = sum_{k in active} 2 * a[k-i] (for valid k-i, i.e., 0<=k-i<n)
        # = 2 * sum_j a[j] * [i+j in active] ... no.
        # Let me redo: d c_k / d a_m = 2 * a[k-m] for 0 <= k-m < n
        # grad[m] = sum_{k in active} d c_k / d a_m = sum_{k in active} 2 * a[k-m] (for valid)
        # = 2 * sum_j a[j] * [k = j+m in active] = 2 * sum_j a[j] * active[j+m]
        # = _hessian_matvec(a, active, n)  [since that's exactly 2 * sum_j a[j] * active[i+j]]
        
        grad = _hessian_matvec(a, active, n)
        
        # Project to tangent space (sum = 1): remove mean
        grad_t = grad - np.mean(grad)
        gnorm = float(np.linalg.norm(grad_t))
        if gnorm < 1e-12:
            break

        # Newton step: solve (H + mu*I) dx = -grad_t + lambda*1
        # We need sum(dx) = 0, so we project.
        # Simplified: solve (H + mu*I) dx = -grad_t, then set dx = dx - mean(dx)
        mu = 1e-8
        def hess_vec(v):
            return _hessian_matvec(v, active, n) + mu * v

        # Use CG to solve (H + mu*I) dx = -grad_t
        # First, check if H + mu*I is positive definite by checking Pd > 0
        # Just try CG and see
        dx = _cg_solve(hess_vec, -grad_t, n, max_iter=200, tol=1e-10)
        dx = dx - np.mean(dx)  # Ensure sum(dx) = 0

        # Line search
        f_old = float(c[active].sum())
        alpha = 1.0
        a_new = a.copy()
        for ls in range(50):
            a_try = a + alpha * dx
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try[active].sum())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5
        else:
            # Failed line search, try smaller step or break
            if alpha < 1e-10:
                break
            a_new = a

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "truncated_gauss":
        t = np.linspace(0, 1, n)
        center = 0.5
        a = np.exp(-((t - center) ** 2) / (2 * 0.2 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with very aggressive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    # More aggressive schedule: more stages, lower final beta
    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                  0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 2:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(8000, max(50, int(remaining / (est_time_per_iter * 8))))

        def fun_local(z_local):
            return _smoobj(z_local, beta, n)

        def jac_local(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun_local, z, jac=jac_local, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxcor': 50, 'maxls': 100, 'maxfun': max_iter * 2}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _subgrad_polish(a_init, deadline, n_max_steps=3000, lr_init=0.02):
    """Subgradient descent on the max (active set gradient)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.001)
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.9
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _shrinkage(a, deadline):
    """Try zeroing out the smallest elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    for frac in [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _coarse_to_fine(a_coarse, n_target, deadline, min_time_remaining=60.0):
    """Upscale a coarse solution to target n and optimize."""
    x_old = np.linspace(0, 1, len(a_coarse))
    x_new = np.linspace(0, 1, n_target)
    a_target = np.interp(x_new, x_old, a_coarse)
    a_target = np.maximum(a_target, 0)
    a_target = a_target / a_target.sum()
    a_opt = _run_lbfgs_aggressive(a_target, deadline, min_time_remaining=min_time_remaining)
    return a_opt


def _perturb_log(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.03 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _variance_warm_start(a_init, deadline, n_steps=500):
    """Use variance minimization as a warm-start to flatten the autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.005

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        grad = _variance_grad(a, n)
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Check if variance improved
        v_new = _variance_obj(a_new)
        v_old = _variance_obj(a)
        if v_new < v_old:
            a = a_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.8
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS + Newton active-set + variance warm-start
    + coarse-to-fine + diverse restarts + aggressive beta continuation.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent loaded: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Variance warm-start ===
    print(f"\n[Phase 1] Variance warm-start (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 600:
        v_deadline = min(deadline - 600, time.monotonic() + 120)
        a_v, sc_v = _variance_warm_start(best_seq, v_deadline, n_steps=300)
        update_best(a_v, sc_v, "variance_ws")
        if sc_v < best_score:
            # L-BFGS from variance-improved point
            a_v2 = _run_lbfgs_aggressive(a_v, deadline, min_time_remaining=500)
            update_best(a_v2, _score(a_v2), "variance_ws_lbfgs")

    # === Phase 2: Main aggressive L-BFGS ===
    print(f"\n[Phase 2] Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 500:
        a_main = _run_lbfgs_aggressive(best_seq, deadline, min_time_remaining=500)
        update_best(a_main, _score(a_main), "lbfgs_main")

    # === Phase 3: Newton active-set ===
    print(f"\n[Phase 3] Newton active-set (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 400:
        n_deadline = min(deadline - 400, time.monotonic() + 300)
        a_newton, sc_newton = _newton_active_set(best_seq, n_deadline, max_iter=200)
        update_best(a_newton, sc_newton, "newton")

    # === Phase 4: Subgradient polish ===
    print(f"\n[Phase 4] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=2000, lr_init=0.015)
        update_best(a_sg, sc_sg, "sg_polish")

    # === Phase 5: Shrinkage ===
    print(f"\n[Phase 5] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 250)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score * 1.001:
            a_shr2 = _run_lbfgs_aggressive(a_shr, deadline, min_time_remaining=200)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")
            # Newton on shrunk
            if time.monotonic() < deadline - 150:
                a_shr3, sc_shr3 = _newton_active_set(a_shr2, min(deadline - 150, time.monotonic() + 120), max_iter=100)
                update_best(a_shr3, sc_shr3, "newton_post_shrink")

    # === Phase 6: Multi-n exploration ===
    print(f"\n[Phase 6] Multi-n exploration (t={time.monotonic() - t_start:.1f}s)")
    n_candidates = [2048, 4096, 8192]
    # Add parent n if different
    if n_parent not in n_candidates:
        n_candidates.append(n_parent)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if time.monotonic() > deadline - 120:
            break
        if best_seq is not None and len(best_seq) != n_try:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        elif best_seq is not None:
            a_init = best_seq.copy()
        else:
            a_init = _make_profile(n_try, "tapered")

        # L-BFGS
        a_opt = _run_lbfgs_aggressive(a_init, deadline, min_time_remaining=80)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}_lbfgs")

        # Newton
        if sc_opt < best_score * 1.01 and time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _newton_active_set(a_opt, min(deadline - 60, time.monotonic() + 80), max_iter=80)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_newton")

    # === Phase 7: Coarse-to-fine ===
    print(f"\n[Phase 7] Coarse-to-fine (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_best = len(best_seq)
        if n_best > 1024:
            # Downsample
            n_coarse = n_best // 4
            x_fine = np.linspace(0, 1, n_best)
            x_coarse = np.linspace(0, 1, n_coarse)
            a_coarse = np.interp(x_coarse, x_fine, best_seq)
            a_coarse = np.maximum(a_coarse, 0)
            a_coarse = a_coarse / a_coarse.sum()
            # Optimize coarse
            a_coarse_opt = _run_lbfgs_aggressive(a_coarse, deadline, min_time_remaining=60)
            # Upscale back
            a_up = np.interp(x_fine, x_coarse, a_coarse_opt)
            a_up = np.maximum(a_up, 0)
            a_up = a_up / a_up.sum()
            # Optimize fine
            if time.monotonic() < deadline - 50:
                a_cf = _run_lbfgs_aggressive(a_up, deadline, min_time_remaining=30)
                update_best(a_cf, _score(a_cf), "coarse_to_fine")

    # === Phase 8: Diverse restarts ===
    print(f"\n[Phase 8] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_restarts = 4
        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            if i == 0:
                p_start = _perturb_log(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            elif i == 2:
                p_start = _perturb_block(best_seq, rng, n_blocks=3)
            else:
                # Mix of perturbations
                p_start = _perturb_log(best_seq, 0.02, rng)
                p_start = _perturb_fourier(p_start, rng, n_modes=2)

            a_r = _run_lbfgs_aggressive(p_start, deadline, min_time_remaining=15)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score * 1.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _newton_active_set(a_r, min(deadline - 10, time.monotonic() + 30), max_iter=50)
                update_best(a_r2, sc_r2, f"restart_{i}_newton")

    # === Phase 9: Final combined polish ===
    print(f"\n[Phase 9] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very aggressive subgradient
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 5, n_max_steps=2000, lr_init=0.008)
        update_best(a_fin, sc_fin, "final_sg")

        # Newton on the result
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _newton_active_set(best_seq, max(deadline - 3, time.monotonic()), max_iter=50)
            update_best(a_fin2, sc_fin2, "final_newton")

        # Very low beta L-BFGS
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 0.00001
            remaining = deadline - 2 - time.monotonic()
            if remaining > 3:
                max_iter = min(3000, max(100, int(remaining / 0.01)))

                def fun_final(z_local):
                    return _smoobj(z_local, beta_final, n)

                def jac_final(z_local):
                    return _smograd(z_local, beta_final, n)

                try:
                    result = minimize(
                        fun_final, z, jac=jac_final, method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                 'maxcor': 50}
                    )
                    a_fin3 = _softmax(result.x)
                    update_best(a_fin3, _score(a_fin3), "final_ultra_lbfgs")
                except Exception:
                    pass

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        # Verify with evaluate_sequence
        sc_final = _score(final)
        try:
            sc_eval = evaluate_sequence([float(x) for x in final])
            if np.isfinite(sc_eval) and sc_eval < sc_final:
                sc_final = sc_eval
        except Exception:
            pass
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
