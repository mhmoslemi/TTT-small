Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: FISTA with top-K smoothing, Adam log-space, LP-based polishing,
and exact coordinate descent. Optionally exploits mirror symmetry.
"""
import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _topk_weights(c, beta, k):
    n_c = len(c)
    if k >= n_c:
        k = n_c
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _fista(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter_per_phase):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad_a = _compute_grad_conv(a, w)
            mean_grad = float(np.dot(grad_a, a))
            grad = 2.0 * n * a * (grad_a - mean_grad)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)
        
        m_buf *= 0.5
        v_buf *= 0.5

    sc_final = _score_norm(a_final)

        best_a = a_final.copy()

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_conv(a, w):
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _topk_weights(c, beta, k):
    n_c = len(c)
    k = min(k, n_c)
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w

def _greedy_lp_direction(a, c, M, n):
    """
    Greedy approach to find a feasible direction g >= 0 such that
    (a*g)[j] <= (M - c[j])/2 for all j.
    Returns the direction g (not normalized).
    """
    b = (M - c) / 2.0  # RHS of constraints
    b = np.maximum(b, 0.0)  # Ensure non-negative RHS
    g = np.zeros(n)
    
    # Process coordinates in order of a[i] (largest first for more impact)
    order = np.argsort(a)[::-1]
    
    for i in order:
        # For coordinate i, the constraint is:
        # sum_k a[k]*g[k] contributes a[j-i]*g[i] to (a*g)[j]
        # So we need: a[j-i]*g[i] <= b[j] - sum_{k != i} a[j-k]*g[k]
        # i.e., g[i] <= min_j (b[j] - (a*g)[j]_current) / a[j-i]
        # where the min is over j such that a[j-i] > 0 and 0 <= j-i < n
        
        # Compute current (a*g) contribution at all positions
        # Actually, let's compute the remaining budget at each position
        ag = _full_convolve(a, g)  # This is O(n log n) with FFT
        remaining = b - ag  # How much more each position can take
        
        # For coordinate i, incrementing g[i] by delta adds a[j-i]*delta to (a*g)[j]
        # So delta <= remaining[j] / a[j-i] for all valid j
        # Valid j: 0 <= j < 2n-1 and 0 <= j-i < n, i.e., i <= j <= i+n-1
        
        j_start = max(0, i)
        j_end = min(2 * n - 1, i + n - 1) + 1
        a_slice = a[j_start - i : j_end - i]  # a[0] to a[min(n-1, j_end-i-1)]
        rem_slice = remaining[j_start:j_end]
        
        # delta <= rem_slice[k] / a_slice[k] for all k where a_slice[k] > 0
        mask = a_slice > 1e-30
        if not np.any(mask):
            continue
        ratios = np.where(mask, rem_slice / a_slice, np.inf)
        max_delta = float(np.min(ratios))
        
        if max_delta > 1e-15:
            g[i] = max_delta
    
    return g

def _lp_polish(a, iterations, deadline, use_sparse=True):
    """
    LP-based polishing: iteratively find direction g that keeps conv <= M,
    then update a = (1-t)*a + t*g/sum(g).
    """

    a = a.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

        c = _full_convolve(a, a)
        M = float(c.max())
        
        # Solve LP: max sum(g) s.t. g >= 0, (a*g)[j] <= (M-c[j])/2
        b = (M - c) / 2.0
        
        # For the greedy approach (faster):
        g = _greedy_lp_direction(a, c, M, n)
        s_g = float(g.sum())
        
        if s_g < 1e-15:
            break
        
        # Update: a_new = (1-t)*a + t*g/s_g
        # Choose t to maintain sum = 1 (already satisfied since sum(a)=1 and sum(g/s_g)=1)
        # Use t = 1.0 for full step, or smaller for safety
        t = 0.5  # Conservative step
        a_new = (1.0 - t) * a + t * g / s_g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
