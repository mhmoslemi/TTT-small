"""
Multi-scale optimizer for minimizing 2n*max(a*a)/(sum(a))^2.
Uses log-sum-exp smoothed FISTA, Adam log-space, block coordinate descent,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(a*a)"""
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _logsumexp(z, beta):
    """Compute beta * log(sum(exp((z - z.max())/beta)))"""
    z_shifted = (z - z.max()) / beta
    np.clip(z_shifted, -500, 500, out=z_shifted)
    e = np.exp(z_shifted)
    s = float(e.sum())
    if s < 1e-300:
        return 0.0, None
    w = e / s
    return float(z.max() + beta * np.log(s)), w

def _grad_lse(y, w):
    """Gradient of the log-sum-exp smoothed max of y*y w.r.t. y"""
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

def _fista_lse(a_init, beta, deadline, max_iter, lr_scale=1.0):

    n = len(a_init)

    best_sc = _score_norm(a_curr)

    step = lr_scale / L
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0

        y = np.maximum(y, 0.0)
        s_y = y.sum()

        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:

        best_a = a_curr.copy()
    return best_a, best_sc

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
