def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_palindromic(a):
    n = len(a)
    return (a + a[::-1]) / 2.0


def _enforce_palindromic_in_z(z):
    z = (z + z[::-1]) / 2.0
    return z


def _compute_subgrad_topk(a, c, k, soft_tau=None):
    n = len(a)
    if soft_tau is not None and soft_tau > 0:
        c_shifted = (c - c.max()) / soft_tau
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            p = e / s
    else:
        k_indices = np.argsort(c)[::-1][:min(k, len(c))]
        p = np.zeros_like(c)
        p[k_indices] = 1.0 / len(k_indices)
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 2.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

def _build_lp_matrix(n):
    rows_list = []
    cols_list = []
    data_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        cnt = len(j_range)
        rows_list.append(np.full(cnt, k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
        data_list.append(np.ones(cnt))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    data = np.concatenate(data_list)
    return rows, cols, data

def _generate_gaussian_init(n, center_frac, width_frac):
    x = np.arange(n) / n
    center = center_frac
    width = width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _generate_cosine_init(n, mode):
    x = np.arange(n) / n
    if mode == 1:
        a = np.sin(np.pi * x)
    elif mode == 2:
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

                a_final_lp, _ = _lp_step(best_a, lp_mat[0], lp_mat[1], lp_mat[2], time_limit=10)
                update_best(a_final_lp)

        print(f"Phase 7 done: score={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Final validation and return
    if best_a is None:
        final = np.ones(256)
    else:
        final = np.clip(best_a, 0.0, 1000.0)

    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
        s = float(final.sum())

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
