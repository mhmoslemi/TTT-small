import numpy as np
import time


def _next_pow2(n):

def _softgrad_weighted(a, c, beta):
    """Compute smoothed gradient of max(c) w.r.t. a using softmax weights."""
    n = len(a)
    max_c = float(c.max())
    shifted = (c - max_c) / beta
    np.clip(shifted, -500, 0, out=shifted)
    w = np.exp(shifted)
    s = float(w.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s

def _adam_logspace(a_init, deadline, betas, lr_init, max_iter_total, gamma_switch=0.85):
    """
    Adam optimizer in log-space with beta continuation.
    Uses exact subgradient when max is concentrated (top w > gamma_switch),
    smoothed gradient otherwise.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    patience = 0

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_from_z(z)
            c = _convolve(a, a)
            cm = float(c.max())

            # Decide: exact subgradient or smoothed
            top1_weight = 0.0
            if beta > 0:
                shifted = (c - cm) / beta
                np.clip(shifted, -500, 0, out=shifted)
                w = np.exp(shifted)
                ws = float(w.sum())
                if ws > 1e-300:
                    top1_weight = float(w.max()) / ws

            if top1_weight > gamma_switch and beta < 0.01 * cm:
                grad_a = _exact_subgrad(a, c, frac=0.005)
            else:
                grad_a = _softgrad_weighted(a, c, beta)

            # Convert to z-gradient (tangent to simplex)
            grad_sum = float(np.dot(a, grad_a))
            grad_z = grad_a - a * grad_sum

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:

                    best_a = a_chk.copy()
                    patience = 0
                else:
                    patience += 1

            if it_in_phase % 8000 == 0:

        patience = 0

    a_final = _softmax_from_z(z)

    if np.isfinite(sc_final) and sc_final < best_score:

def _coord_descent(a_init, deadline, max_rounds=300):

    n = len(a_init)

            break
        if n > 2048:
            n_sample = min(256, n)
            indices = rng.choice(n, size=n_sample, replace=False)
        else:
            indices = np.arange(n)

        c = _convolve(a, a)

            old_val = a[j]

            # Try increasing and decreasing

                else:

            a[j] = old_val

def _fista_subgrad(a_init, deadline, max_iter, lr_init=0.01):

    FISTA-like accelerated method using exact subgradient.

        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)

    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a0, time.monotonic() + per_time,
            betas, lr_init=0.015, max_iter_total=5000000,
            gamma_switch=0.7
        )
        sc_opt_full = _score(a_opt)
        print(f"  Adam main: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 2: FISTA subgradient refinement ===
    print("Phase 2: FISTA subgradient")
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end - 30:
        per_time = phase2_end - time.monotonic() - 10
        a_fista, sc_fista = _fista_subgrad(
            a_work, time.monotonic() + per_time,
            max_iter=200000, lr_init=0.01
        )
        sc_fista_full = _score(a_fista)
        print(f"  FISTA: score={sc_fista_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_fista_full < best_score:
            best_score = sc_fista_full
            best_seq = a_fista.copy()

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 3: Second Adam pass with very fine betas ===
    print("Phase 3: Ultra-fine Adam")
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end - 30:
        c_w = _convolve(a_work, a_work)
        max_cw = float(c_w.max())
        betas_fine = []
        beta = 0.001 * max_cw
        while beta > 1e-12 * max_cw:
            betas_fine.append(beta)
            beta *= 0.3
        betas_fine.append(1e-12 * max_cw)
        if len(betas_fine) > 12:
            step = len(betas_fine) // 12
            betas_fine = betas_fine[::step] + [betas_fine[-1]]

        per_time = phase3_end - time.monotonic() - 10
        if per_time > 20:
            a_fine, sc_fine = _adam_logspace(
                a_work, time.monotonic() + per_time,
                betas_fine, lr_init=0.008, max_iter_total=3000000,
                gamma_switch=0.6
            )
            sc_fine_full = _score(a_fine)
            print(f"  Fine Adam: score={sc_fine_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_fine_full < best_score:
                best_score = sc_fine_full
                best_seq = a_fine.copy()

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 4: Multi-restart with diverse perturbations ===
    print("Phase 4: Multi-restart")
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 60:
        n_restarts = 10
        per_restart = (phase4_end - time.monotonic() - 30) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break

            # Different perturbation strategies
            strategy = i % 5
            if strategy == 0:
                p_start = _perturb(a_work, 0.01 + 0.003 * i, rng)
            elif strategy == 1:
                # Random shift
                shift = rng.integers(1, max(2, n_main // 1024))
                p_start = np.roll(a_work, int(shift)).copy()
                p_start = p_start / p_start.sum()
            elif strategy == 2:
                # Anti-phase perturbation
                z = np.log(np.maximum(a_work, 1e-15))
                noise = rng.standard_normal(len(z))
                # Add a low-frequency component
                freq = rng.integers(2, 10)
                noise += 0.5 * np.sin(2 * np.pi * freq * np.arange(len(z)) / len(z))
                z = z + 0.01 * noise
                z = z - z.mean()
                p_start = _softmax_from_z(z)
            elif strategy == 3:
                # Block swap
                block = max(2, n_main // 8)
                idx1 = rng.integers(0, n_main - block)
                idx2 = rng.integers(0, n_main - block)
                p_start = a_work.copy()
                p_start[idx1:idx1+block], p_start[idx2:idx2+block] = \
                    p_start[idx2:idx2+block].copy(), p_start[idx1:idx1+block].copy()
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            else:
                # Stretch/compress a random segment
                p_start = a_work.copy()
                seg_len = rng.integers(n_main // 16, n_main // 4)
                start = rng.integers(0, max(1, n_main - seg_len))
                seg = p_start[start:start+seg_len].copy()
                shift = rng.integers(-3, 4)
                p_start[start:start+seg_len] = np.roll(seg, int(shift))
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            sc_s = 2.0 * n_main * max_cs
            if sc_s < best_score:
                best_score = sc_s
                best_seq = p_start.copy()

            # Quick optimize
            betas_r = [0.001 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs,
                       0.00003 * max_cs, 0.00001 * max_cs]
            per_t = max(5.0, per_restart * 0.8)
            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + per_t,
                betas_r, lr_init=0.01, max_iter_total=100000,
                gamma_switch=0.65
            )
            sc_best_full = _score(p_best)
            if sc_best_full < best_score:
                best_score = sc_best_full
                best_seq = p_best.copy()
                print(f"  Restart {i} (strat {strategy}): score={sc_best_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # === Phase 5: Try n=8192 ===
    print("Phase 5: n=8192 exploration")
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end - 60:
        n_big = 8192
        a_big = _interp_profile(a_work, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n={n_big}: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase5_end - 30:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            betas_big = [0.01 * max_cb, 0.003 * max_cb, 0.001 * max_cb,
                         0.0003 * max_cb, 0.0001 * max_cb]
            per_t = phase5_end - time.monotonic() - 10
            if per_t > 20:
                a_big_opt, sc_bo = _adam_logspace(
                    a_big, time.monotonic() + per_t,
                    betas_big, lr_init=0.006, max_iter_total=500000,
                    gamma_switch=0.7
                )
                sc_bo_full = _score(a_big_opt)
                print(f"  Optimized n={n_big}: score={sc_bo_full:.10f}")
                if sc_bo_full < best_score:
                    best_score = sc_bo_full
                    best_seq = a_big_opt.copy()

    # === Phase 6: Coordinate descent polish ===
    print("Phase 6: Coordinate descent polish")
    if best_seq is not None and time.monotonic() < deadline - 80:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 60, time.monotonic() + 100)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=400)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 7: Final ultra-fine Adam ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.0001 * max_cp, 0.00003 * max_cp, 0.00001 * max_cp,
                     0.000003 * max_cp, 0.000001 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.003, max_iter_total=1000000,
                gamma_switch=0.5
            )
            sc_pol_full = _score(a_pol)
            print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.8f}")
    return [float(x) for x in fallback]
