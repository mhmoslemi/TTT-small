Uses p-norm smoothed optimization with L-BFGS, adaptive peak reduction,

import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    a = _softmax(z)
    c = _full_convolve(a, a)

    c = np.maximum(c, 0)
    return float(np.sum(c**p) ** (1.0/p))

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses p-norm smoothed objective with Adam optimizer, structure enforcement based on
first-order optimality conditions, multi-length exploration, and coordinate descent polish.

import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is normalized to sum=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _pnorm_obj_grad(z, p):
    """
    Compute p-norm of conv(softmax(z), softmax(z)) and its gradient w.r.t. z.
    """
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)
    
    # p-norm
    pnorm = float(np.sum(c ** p) ** (1.0 / p))
    
    # dc[dck] = c[k]^(p-1) / pnorm^(p-1)
    dc = c ** (p - 1) / (pnorm ** (p - 1))
    
    # grad_a[i] = 2 * sum_j dc[i+j] * a[j]  (for valid indices)
    # = 2 * corr(dc, a)[i]
    N = _next_pow2(2 * n)
    f_dc = np.fft.rfft(dc, N)
    f_a = np.fft.rfft(a, N)
    corr = np.fft.irfft(f_dc * np.conj(f_a), N)
    grad_a = 2.0 * corr[:n]
    
    # grad_z[i] = grad_a[i] * a[i] - a[i] * sum_j grad_a[j] * a[j]
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    
    return pnorm, grad_z


def _adam_pnorm(a_init, p_schedule, deadline, lr_init=0.01, max_iter=200000):
    """
    Optimize p-norm objective using Adam in log-space.
    p_schedule: list of p values to try (in order).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        
        phase_max_iter = max(1000, min(50000, max_iter // len(p_schedule)))
        
        for it in range(phase_max_iter):
            if time.monotonic() > deadline:
                break
            
            t += 1
            loss, grad = _pnorm_obj_grad(z, p)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            
            # Gradient clipping
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            # Periodic score check

                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    
            # Learning rate decay
            if it % 10000 == 0 and it > 0:
                current_lr = max(0.001, current_lr * 0.9)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _full_convolve(a, a)

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    c = np.maximum(c, 1e-30)

    log_pnorm = (1.0 / p) * float(np.sum(np.log(d) * p) + np.log(np.exp(np.log(np.sum(d**p)))))
    # Safer: compute logsumexp(p * log(d))
    pd = p * np.log(d)

    log_sum = pd_max + float(np.log(np.sum(np.exp(pd - pd_max))))
    log_pnorm = log_sum / p

    # dc[k] = d[k]^(p-1) / ||d||_p^(p-1) * (1/c_max) * c_max = d[k]^(p-1) / ||d||_p^(p-1)

    N = _next_pow2(2 * n)

    grad_a = 2.0 * corr[:n]

            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score_global = sc_parent
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        best_seq = a0.copy()
        best_score_global = _score(a0)
        print(f"No parent. n={n_main}, score={best_score_global:.10f}")
    n_parent = len(best_seq)
    # Determine lengths to try
    lengths = [n_parent]
    if n_parent <= 4096:
        lengths.append(8192)
    if n_parent <= 8192:
        lengths.append(16384)
    if n_parent > 8192:
        lengths.append(4096)
    lengths = list(dict.fromkeys(lengths))
    print(f"Lengths to try: {lengths}")
    # Allocate time
    total_opt_time = deadline - t_start - 60.0
    time_per_len = total_opt_time / len(lengths)
    for n_target in lengths:
        if time.monotonic() > deadline - 40:
            break
        print(f"\n=== n={n_target} (remaining={deadline - time.monotonic():.0f}s) ===")
        len_deadline = time.monotonic() + max(10.0, time_per_len - 20.0)
        # Build initializations
        inits = []
        if n_target == n_parent:
            inits.append(best_seq.copy())
        elif parent is not None:
            inits.append(_interp_profile(parent, n_target))
        else:
            inits.append(_make_profile(n_target, "uniform"))
        if n_target <= 16384:
            inits.append(_make_profile(n_target, "sinusoidal"))
            inits.append(_make_profile(n_target, "tapered"))
        # Try each initialization
        n_inits = len(inits)
        per_init_time = (len_deadline - time.monotonic()) / max(1, n_inits)
        for i_init, a_init in enumerate(inits):
            if time.monotonic() > len_deadline - 10:
                break
            a_init = np.maximum(a_init, 0.0)
            s_init = a_init.sum()
            if s_init < 1e-15:
                continue
            a_init = a_init / s_init
            init_deadline = time.monotonic() + max(5.0, per_init_time - 5.0)
            a_opt, sc_opt = _optimize_one(a_init, init_deadline, label=f"n={n_target},init={i_init}")
            if sc_opt < best_score_global:
                best_score_global = sc_opt
                best_seq = a_opt.copy()
                print(f"  NEW GLOBAL BEST: {sc_opt:.10f}")
    # Multi-restart diversification on best
    if best_seq is not None and time.monotonic() < deadline - 50:
        print(f"\n=== Multi-restart (best={best_score_global:.10f}) ===")
        n_best = len(best_seq)
        a_best = best_seq.copy()
        n_restarts = 3
        per_restart = (deadline - time.monotonic() - 30.0) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                a_start = _perturb(a_best, 0.015, rng)
            elif i == 1:
                a_start = _perturb(a_best, 0.04, rng)
            else:
                sz = max(1, n_best // 2048)
                shift = int(rng.integers(1, max(2, sz)))
                a_start = np.roll(a_best, shift).copy()
                a_start = a_start / a_start.sum()
            a_start = _structure_enforce(a_start, n_rounds=2)
            r_deadline = time.monotonic() + max(5.0, per_restart)
            a_r, sc_r = _optimize_one(a_start, r_deadline, label=f"restart={i}")
            if sc_r < best_score_global:
                best_score_global = sc_r
                best_seq = a_r.copy()
                a_best = a_r.copy()
                print(f"  Restart {i}: NEW BEST {sc_r:.10f}")
    # Final ultra-fine polish
    if best_seq is not None and time.monotonic() < deadline - 15:
        print(f"\n=== Final polish ===")
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        p_ultra = [128, 256, 512]
        pol_deadline = min(deadline, time.monotonic() + 30.0)
        a_pol2, sc_pol2 = _adam_pnorm(a_pol, p_ultra, pol_deadline, lr_init=0.003,
                                      max_iter_per_phase=15000)
        a_pol2 = a_pol2 / a_pol2.sum()
        sc_pol2_full = _score(a_pol2)
        if sc_pol2_full < best_score_global:
            best_score_global = sc_pol2_full
            best_seq = a_pol2.copy()
            print(f"  Ultra-fine: {sc_pol2_full:.10f}")
        # Final structure enforce + coord descent
        a_sf = _structure_enforce(best_seq, n_rounds=3)
        sc_sf = _score(a_sf)
        if sc_sf < best_score_global:
            best_score_global = sc_sf
            best_seq = a_sf.copy()
        cd_dead = min(deadline, time.monotonic() + 10.0)
        a_cd, sc_cd = _coord_desc_polish(best_seq, cd_dead, n_steps=100)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score_global:
            best_score_global = sc_cd_full
            best_seq = a_cd.copy()
    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_profile(256, "uniform")
    sc_final = _score(final)
    print(f"\n=== FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s ===")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
