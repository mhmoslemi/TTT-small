    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, w, n):
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

        m_buf *= 0.3
        v_buf *= 0.3
        current_lr = lr_init * 0.7

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descend_1d(a, i, c_current, deadline):
    """Optimize coordinate i using golden section search on the 1D objective."""
    n = len(a)
    a_i_old = a[i]
    S0 = 1.0 - a_i_old  # sum of all other coordinates

    # Compute d_j = a_{j-i} for valid j (where 0 <= j-i < n)
    # c_j_new(t) = c_j - a_i_old * d_j + t * d_j (for j where d_j != 0)
    # For j = 2i: c_j_new = c_j - a_i_old^2 + t^2 (special case)
    
    # Build the vector d of length 2n-1
    d = np.zeros(2 * n - 1)
    j_start = i
    j_end = min(i + n, 2 * n - 1)
    d[j_start:j_end] = a[0:j_end - j_start]
    
    # c_base[j] = c[j] - a_i_old * d[j]
    c_base = c_current - a_i_old * d
    
    # For j = 2i, we have a quadratic term. Let's handle it separately
    j_quad = 2 * i
    if j_quad < 2 * n - 1:
        c_base[j_quad] = c_current[j_quad] - a_i_old * a_i_old  # remove the quadratic part
    
    def _obj_1d(t):
        # c_j(t) = c_base[j] + t * d[j] for j != 2i
        # c_{2i}(t) = c_base[2i] + t * t (quadratic)
        # Actually for j=2i: the term is a_i * a_{j-i} = a_i * a_i = a_i^2
        # So c_{2i} = c_base[2i] + t^2 where c_base[2i] = c[2i] - a_i_old^2
        # For other j: c_j = c_base[j] + t * d[j]
        
        c_trial = c_base + t * d
        if j_quad < 2 * n - 1:
            c_trial[j_quad] = c_base[j_quad] + t * t
        max_c = float(c_trial.max())
        denom = (S0 + t) ** 2
        if denom < 1e-30:
            return np.inf
        return 2.0 * n * max_c / denom

    # Golden section search on [0, upper_bound]
    # Good range: [0, 3 * max(a_i_old, 1/n)]
    upper = max(3.0 * a_i_old, 3.0 / n, 0.1)
    
    ratio = (np.sqrt(5.0) - 1.0) / 2.0
    a_lo = 0.0
    a_hi = upper
    
    f_lo = _obj_1d(a_lo)
    f_hi = _obj_1d(a_hi)
    
    x1 = a_hi - ratio * (a_hi - a_lo)
    x2 = a_lo + ratio * (a_hi - a_lo)
    f_x1 = _obj_1d(x1)
    f_x2 = _obj_1d(x2)
    
    best_t = a_lo
    best_f = f_lo
    if f_x1 < best_f:
        best_t, best_f = x1, f_x1
    if f_x2 < best_f:
        best_t, best_f = x2, f_x2
    if f_hi < best_f:
        best_t, best_f = a_hi, f_hi
    
    # Iterative golden section
    for _ in range(40):
        if time.monotonic() > deadline:
            break
        if f_x1 < f_x2:
            a_hi = x2
            x2 = x1
            f_x2 = f_x1
            x1 = a_hi - ratio * (a_hi - a_lo)
            f_x1 = _obj_1d(x1)
        else:
            a_lo = x1
            x1 = x2
            f_x1 = f_x2
            x2 = a_lo + ratio * (a_hi - a_lo)
            f_x2 = _obj_1d(x2)
        
        if f_x1 < best_f:
            best_t, best_f = x1, f_x1
        if f_x2 < best_f:
            best_t, best_f = x2, f_x2
    
    return best_t, best_f


def _coord_descend_polish(a, n_passes, deadline, top_k_scan=30):
    """Cyclic coordinate descent with 1D golden section search."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        c_current = _full_convolve(a, a)
        improved = 0
        
        for idx in range(len(order)):
            if time.monotonic() > deadline:
                break
            i = order[idx]
            a_i_old = a[i]
            
            if a_i_old < 1e-10:
                # Skip zero coordinates (or try small value)
                continue
            
            best_t, best_f = _coord_descend_1d(a, i, c_current, deadline)
            
            if abs(best_t - a_i_old) > 1e-10:
                # Update a[i] and recompute affected part of convolution
                a[i] = best_t
                a = a / a.sum()
                c_current = _full_convolve(a, a)
                improved += 1
            
            if improved % 256 == 0 and improved > 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _lp_update(a, M, time_limit=15):
    """
    LP-based update: max sum(b) s.t. (a*b)_j <= M, b >= 0
    Returns b normalized to sum=1, or None.
    """
    n = len(a)
    t_start = time.monotonic()
    
    # Build sparse constraint matrix
    # A[j, i] = a[j-i] if 0 <= j-i < n, else 0
    # Rows: 2n-1, Cols: n
    # Non-zeros: n^2 total
    
    row_parts = []
    col_parts = []
    data_parts = []
    
    for j in range(2 * n - 1):
        i_start = max(0, j - n + 1)
        i_end = min(j, n - 1)
        count = i_end - i_start + 1
        if count > 0:
            row_parts.extend([j] * count)
            col_parts.extend(range(i_start, i_end + 1))
            data_parts.extend(a[i_start:i_end + 1])
    
    A = csr_matrix((data_parts, (row_parts, col_parts)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    
    if time.monotonic() - t_start > time_limit + 5:
        return None
    
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None


def _lp_alternate(a, deadline, n_lp_iters=5):
    """Alternate LP updates."""
    n = len(a)
    a = a.copy() / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for it in range(n_lp_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        
        b = _lp_update(a, M, time_limit=10)
        if b is None:
            break
        
        # Check if b has better score
        sc_b = _score_norm(b)
        
        # Try interpolation
        for t in [0.3, 0.5, 0.7, 1.0]:
            a_trial = (1 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="raised_cos"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.8
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 4]:
            a += 0.2 / freq * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "erfc_taper":
        t = np.linspace(0, 1, n)
        gamma = 3.0
        a = np.exp(-gamma * (2 * t - 1) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):

    for bi, beta in enumerate(betas_fista):

    if remaining > 60:

        for cyc in range(n_cycles):

                                           lr_init=0.005, top_k=40,

            if _score_norm(a_ult) < _score_norm(a_work):
                a_work = a_ult.copy()
            
            # Quick CD pass
            cd_end = min(deadline - 20, time.monotonic() + min(30, remaining / (n_cycles - cyc)))
            if cd_end - time.monotonic() > 10:
                a_cd2, sc_cd2 = _coord_descend_polish(a_work, n_passes=2, deadline=cd_end)
                sc_cd2_full = _score(a_cd2)

            if time.monotonic() > deadline - 10:
                break
            
            if ri == 0:
                p_start = a_base + 0.01 * rng.standard_normal(n_best)
            elif ri == 1:
                shift = rng.integers(1, max(2, n_best // 2048))
                p_start = np.roll(a_base, shift).copy()
            elif ri == 2:
                freq = rng.uniform(0.3, 5.0)
                pert = 0.02 * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            elif ri == 3:
                # Block scaling
                block = max(1, n_best // 16)
                p_start = a_base.copy()
                start = rng.integers(0, n_best - block)
                scale = rng.uniform(0.7, 1.4)
                p_start[start:start + block] *= scale
            else:
                # Mirror + blend
                p_start = 0.5 * a_base + 0.5 * a_base[::-1]
                p_start = p_start + 0.005 * rng.standard_normal(n_best)

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()
            
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.001 * max_cs, 0.0003 * max_cs, 0.0001 * max_cs]
            
            p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                             betas_r, lr_init=0.01, top_k=60,
                                             max_iter=50000)
            # Quick CD
            if time.monotonic() < deadline - 5:
                p_best, sc_best2 = _coord_descend_polish(p_best, n_passes=1,
                                                         deadline=deadline - 5)
                sc_best = min(sc_best, sc_best2)
            
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                a_work = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # Final validation
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL VALIDATED: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
