            # In that case, there's only ONE term: a_i * a_i = a_i^2.
            # Otherwise, there are TWO separate terms (for two different values of j), each equal
            # to a_i * a_{k-i}.
            
            # So the contribution of a_i to c_k is:
            #   0, if k-i < 0 or k-i >= n (i.e., k < i or k >= i+n)
            #   a_i * a_{k-i} (once), if k = 2i (and 0 <= k-i < n, which is 0 <= i < n, true)
            #   2 * a_i * a_{k-i}, if 0 <= k-i < n and k != 2i
            
            # OK so for the update:
            # When a_i changes from old to new:
            # For k in [i, i+n-1] ∩ [0, 2n-2]:
            #   If k == 2i: delta = new^2 - old^2
            #   Else: delta = 2*(new-old)*a_{k-i}
            
            # Great, now I can implement efficient coordinate descent.
            
            # But actually, for the coordinate descent, I want to find the optimal new value.
            # The objective is max_k c_k (to minimize). After changing a_i to t:
            # c_k(t) = c_k - delta_old_k + delta_new_k
            # where delta_old_k is the old contribution and delta_new_k is the new.
            
            # This is complicated because the max over k depends on t in a non-trivial way.
            
            # For simplicity, let me just try a few candidate values of t and pick the best.
            
            # Actually, let me just do a simpler version: for each coordinate, try
            # a few multiplicative factors and pick the best.
            
            if a_i_old < 1e-15:
                # Try a small positive value
                a[i] = 1e-6
                a = a / a.sum()
                continue
            
            candidates = [a_i_old * f for f in [0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5, 2.0, 0.1, 0.3]]
            best_t = a_i_old
            best_val = np.inf
            
            # Compute base c (without a_i's contribution)
            # c_base_k = c_k - contribution_of_ai_to_c_k
            c_base = c.copy()
            # Remove a_i's contribution
            lo = max(0, i)
            hi = min(2 * n - 1, i + n - 1)
            for k in range(lo, hi + 1):
                if k == 2 * i:
                    c_base[k] -= a_i_old * a_i_old
                else:
                    c_base[k] -= 2.0 * a_i_old * a[k - i]
            
            for t in candidates:
                # Add new contribution
                c_trial = c_base.copy()
                for k in range(lo, hi + 1):
                    if k == 2 * i:
                        c_trial[k] += t * t
                    else:
                        c_trial[k] += 2.0 * t * a[k - i]
                val = c_trial.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if best_t != a_i_old:
                a[i] = best_t
                a = np.maximum(a, 0)
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1
        
        if verbose and outer % 5 == 0:
            cur = _score_norm(a)
            if cur < best_score:
                best_score = cur
            print(f"  CD pass {outer}: improved={improved}, score={_score_norm(a):.10f}")
        
        if improved == 0:
            break
    
    final_score = _score_norm(a)
    if final_score < best_score:
        best_score = final_score
    return a, best_score
