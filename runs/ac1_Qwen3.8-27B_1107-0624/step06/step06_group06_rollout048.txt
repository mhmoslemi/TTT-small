def _next_pow2(n):

        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Recompute gradient with the smoothed weights
    L = _next_pow2(3 * n)
    a_pad = np.zeros(L)
    a_pad[:n] = a
    w_pad = np.zeros(L)
    w_pad[:2*n-1] = w
    
    A = np.fft.rfft(a_pad)
    W = np.fft.rfft(w_pad)
    g1 = np.fft.irfft(A * np.conj(W), L)[:n]
    g2 = np.fft.irfft(A * W, L)[:n]
    grad_smooth = g1 + g2
    
    # Adam update
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1 - beta1) * grad_smooth
    v = beta2 * v + (1 - beta2) * grad_smooth ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return z, m, v, _score(a)

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_active(a, c, n, tol_frac=0.002):
    """Exact subgradient using active set."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_single_peak(a, c, n, peak_idx=None):
    """Subgradient from a single random peak."""
    if peak_idx is None:
        c_max = float(c.max())
        active = np.where(c >= c_max - 1e-10)[0]
        peak_idx = int(np.random.choice(active))
    w = np.zeros_like(c)
    w[peak_idx] = 1.0
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(c[peak_idx])
    return 4.0 * n * a * (h - W)


def _subgrad_refined(a_init, deadline, n_steps=3000, tol_fracs=None):
    """
    Subgradient method with:
    - Alternating between active-set gradient and single-peak gradient
    - Adaptive step size (Polyak-style)
    - Periodic restart from best
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy() / a_init.sum()
    best_sc = _score(best_a)
    
    if tol_fracs is None:
        tol_fracs = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    # Polyak step size: need an upper bound on the optimum
    # score >= 1 (theoretical lower bound), so f* >= 1
    # f* <= best_sc
    # step = (f_current - f_lower_bound) / ||grad||^2
    f_lb = 1.0
    
    patience = 0
    max_patience = 100
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        sc = 2.0 * n * float(c.max())  # score (since sum=1)
        
        # Choose gradient type
        if i % 3 == 0:
            # Active set gradient
            tol = tol_fracs[min(i // 200, len(tol_fracs) - 1)]
            grad = _grad_active(a, c, n, tol_frac=tol)
        else:
            # Single peak gradient (more stochastic)
            c_max = float(c.max())
            active = np.where(c >= c_max * 0.999)[0]
            if len(active) == 0:
                active = np.array([int(np.argmax(c))])
            peak_idx = int(np.random.choice(active))
            grad = _grad_single_peak(a, c, n, peak_idx=peak_idx)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Polyak step size with cap
        gap = max(sc - f_lb, 1e-10)
        step = min(gap / (grad_norm ** 2), 0.1)
        step = max(step, 0.0001)
        
        # Damping
        step *= 0.5
        
        z = z - step * grad
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        # Check improvement
        if i % 50 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc - 1e-13:
                best_sc = sc_check
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    # Restart from best with noise
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    z = z + 0.01 * np.random.randn(n)
                    z = z - z.mean()
                    patience = 0
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _coord_descent(a_init, deadline, n_passes=5):
    """
    Coordinate descent: for each position, try shifting a small amount
    to a neighboring position (or adjusting the value).
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            # Try moving a small amount from position i to position j
            # Best j is where it reduces the peak most
            # grad[i] tells us the direction of increase for a[i]
            # We want to decrease the score, so move from high-grad to low-grad positions
            
            # Simple approach: try a few random transfer targets
            for _ in range(3):
                j = np.random.randint(n)
                if j == i:
                    continue
                delta = 0.01 * a[i]
                if a[i] < delta:
                    continue
                a_try = a.copy()
                a_try[i] -= delta
                a_try[j] += delta
                if a_try[i] < 0:
                    continue
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-13:
                    a = a_try
                    best_a = a_try.copy()
                    best_sc = sc_try
                    c = _full_convolve(a, a)
    
    return best_a, best_sc


def _lbfgs_tight(a_init, deadline, min_time_remaining=30.0):
    """
    L-BFGS with an aggressive beta schedule that goes very tight.
    """
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    # Very tight beta schedule
    betas = [max_c * 0.1, max_c * 0.05, max_c * 0.02, max_c * 0.01, 
             max_c * 0.005, max_c * 0.002, max_c * 0.001, max_c * 0.0005]
    
    for beta in betas:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Smaller maxiter for tighter betas (converges faster)
        max_iter = min(2000, max(50, int(remaining / 0.05)))
        
        try:
            result = minimize(
                lambda z_local, b=beta: _smoobj(z_local, b, n),
                z,
                jac=lambda z_local, b=beta: _smograd(z_local, b, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a, current_sc


def _freq_init(n, rng):
    """Create a diverse initialization using frequency-domain considerations."""
    # Start with a random phase pattern (gives flat power spectrum)
    A = np.ones(n // 2 + 1, dtype=complex)
    phases = rng.uniform(0, 2 * np.pi, size=n // 2 + 1)
    A *= np.exp(1j * phases)
    A[0] = np.abs(A[0])  # Keep DC real
    
    a = np.fft.irfft(A, n)
    a = np.maximum(a, 0)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a / a.sum()
    return a


def _make_profile(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t) + 0.1
        return a / a.sum()
    elif kind == "cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb_large(a, rng, scale=0.3):
    """Large perturbation to escape local minima."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_blocks(a, rng, n_blocks=4):
    """Perturb by permuting blocks."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = n // n_blocks
    # Random permutation of blocks
    order = rng.permutation(n_blocks)
    z_new = np.zeros_like(z)
    for old_i, new_i in enumerate(order):
        z_new[new_i * block_size:(new_i + 1) * block_size] = z[old_i * block_size:(old_i + 1) * block_size]
    return _softmax(z_new)


def _two_scale_optimize(a_coarse, n_fine, deadline):
    """
    Two-scale optimization: start from a coarse solution, 
    upsample to fine, then optimize.
    """
    n_c = len(a_coarse)
    x_old = np.linspace(0, 1, n_c)
    x_new = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_new, x_old, a_coarse)
    a_fine = np.maximum(a_fine, 1e-15)
    a_fine = a_fine / a_fine.sum()
    
    a_opt, sc_opt = _lbfgs_tight(a_fine, deadline, min_time_remaining=20)
    return a_opt, sc_opt
