"""

Uses adaptive L-BFGS with log-sum-exp, Nesterov-accelerated subgradient,
frequency-domain perturbations, and aggressive local search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + 0.0001 * i)
        z_candidate = z - step * grad

        # Nesterov update
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / (2.0 * t + 1.0)
        z_new = gamma * z_candidate + (1.0 - gamma) * z_prev

        z_new = np.clip(z_new, -90.0, 90.0)
        z_new = z_new - z_new.mean()

        z_prev = z
        z = z_new

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.02, 0.15)
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.7
                    momentum *= 0.95
                    if lr < 1e-8:
                        break
                else:
                    lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_a = a_final.copy()
        best_sc = sc_final

    return best_a, best_sc


def _direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 0.5 / np.sqrt(n)
    stall = 0
    momentum_vec = np.zeros(n)
    momentum_coeff = 0.5

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        g = _gradient_active(a, c, n, tol_frac)
        g = g - g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Momentum
        momentum_vec = momentum_coeff * momentum_vec + (1 - momentum_coeff) * g
        step_dir = momentum_vec

        a_try = a - step_size * step_dir
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        sc_cur = _score(a)
        if sc_try < sc_cur - 1e-16:
            a = a_try
            step_size = min(step_size * 1.01, 0.05)
        else:
            stall += 1
            step_size *= 0.8

        step_size = max(step_size, 1e-12)

        if stall > 300:
            break

    return best_a, best_sc


def _perturb_fourier_domain(a, rng, n_modes=5, amplitude=0.01):
    """Perturb in frequency domain for more global exploration."""
    n = len(a)
    F = np.fft.rfft(a)
    for _ in range(n_modes):
        k = int(rng.integers(1, len(F)))
        delta_real = amplitude * rng.uniform(-1, 1)
        delta_imag = amplitude * rng.uniform(-1, 1)
        F[k] += delta_real + 1j * delta_imag
        if k < len(F) - 1:
            F[len(F) - 1 - k] = np.conj(F[k])
    a_new = np.fft.irfft(F, n)
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()
    return a_new


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_structural(a, rng):
    """Perturb by modifying blocks with different patterns."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    n_blocks = int(rng.integers(3, 10))
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - 10)))
        length = int(rng.integers(5, max(6, n // 5)))
        end = min(n, start + length)
        # Random pattern: ramp, cosine bump, or constant shift
        pattern_type = rng.integers(0, 3)
        t = np.linspace(0, 1, end - start)
        if pattern_type == 0:
            pert = rng.uniform(-0.02, 0.02)
            z[start:end] += pert
        elif pattern_type == 1:
            z[start:end] += rng.uniform(-0.01, 0.01) * np.sin(np.pi * t)
        else:
            z[start:end] += rng.uniform(-0.015, 0.015) * (2 * t - 1)
    z = z - z.mean()
    return _softmax(z)


def _local_search_aggressive(a, deadline, max_rounds=2000):
    """Aggressive local search with multiple coordinate directions."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    rng_ls = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick 2 indices to modify
        idx1 = int(rng_ls.integers(0, n))
        idx2 = int(rng_ls.integers(0, n))
        if idx2 == idx1:
            idx2 = (idx2 + 1) % n

        improved = False
        # Try transferring mass between two indices
        for frac in [0.05, 0.1, 0.2, 0.05, -0.05, -0.1, -0.2]:
            transfer = frac * (best_a[idx1] + best_a[idx2]) * 0.5
            a_try = best_a.copy()
            a_try[idx1] = max(0.0, best_a[idx1] - transfer)
            a_try[idx2] = max(0.0, best_a[idx2] + transfer)
            s = a_try.sum()
            if s < 1e-30:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True
                break

        # Also try small additive perturbation
        for delta in [0.001, -0.001, 0.005, -0.005]:
            a_try = best_a.copy()
            a_try[idx1] = max(0.0, best_a[idx1] + delta)
            s = a_try.sum()
            if s < 1e-30:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_sc - 1e-14:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True
                break

        if not improved and rnd > 100:
            # Try more aggressive moves
            for _ in range(5):
                idx = int(rng_ls.integers(0, n))
                scale = rng.uniform(0.5, 1.5)
                a_try = best_a.copy()
                a_try[idx] *= scale
                s = a_try.sum()
                if s < 1e-30:
                    continue
                a_try = a_try / s
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-14:
                    best_a = a_try.copy()
                    best_sc = sc_try
                    improved = True

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def _final_polish_cycle(a_start, deadline, min_remaining=5.0):
    """Final aggressive polishing: alternating tiny-beta L-BFGS and subgradient."""
    n = len(a_start)
    current_a = a_start.copy()
    current_sc = _score(current_a)

    while time.monotonic() < deadline - min_remaining:
        c_b = _full_convolve(current_a, current_a)
        mb = float(c_b.max())
        beta = mb * 5e-7

        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 5:
            break
        max_iter = min(3000, max(100, int(remaining / 0.003)))
        a_new, _, _ = _lbfgs_run(current_a, beta, n, max_iter)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc_new

        if time.monotonic() < deadline - min_remaining:
            a_sg, _ = _nesterov_subgrad(current_a, deadline - min_remaining,
                                        tol_frac=1e-5, max_steps=3000, lr_init=0.01)
            sc_sg = _score(a_sg)
            if sc_sg < current_sc - 1e-15:
                current_a = a_sg
                current_sc = sc_sg

    return current_a, current_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive L-BFGS with log-sum-exp smoothing,
    Nesterov-accelerated subgradient, frequency-domain diversification,
    aggressive local search, and multi-resolution exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Adaptive L-BFGS (main optimization) ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 1: Adaptive L-BFGS (t={t_phase - t_start:.1f}s) ===")
    a_p1 = _lbfgs_adaptive(best_seq, deadline, min_time_remaining=600.0)
    try_update(a_p1, "phase1_adaptive_lbfgs")
    print(f"  Phase 1 done: score={_score(a_p1):.10f}, t={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Nesterov subgradient polishing ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 2: Nesterov subgradient (t={t_phase - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_sg, sc_sg = _nesterov_subgrad(best_seq, deadline - 400,
                                        tol_frac=1e-4, max_steps=15000, lr_init=0.03)
        try_update(a_sg, "phase2_nesterov")
        print(f"  Phase 2 done: score={sc_sg:.10f}, t={time.monotonic() - t_start:.1f}s")

    # === Phase 3: Direct-space momentum polish ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 3: Direct momentum polish (t={t_phase - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_dp, sc_dp = _direct_polish(best_seq, deadline - 250,
                                     tol_frac=1e-5, max_steps=50000)
        try_update(a_dp, "phase3_direct")
        print(f"  Phase 3 done: score={sc_dp:.10f}, t={time.monotonic() - t_start:.1f}s")

    # === Phase 4: Frequency-domain restarts ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 4: Freq-domain restarts (t={t_phase - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        restart_configs = [
            ("fourier_5", 5, 0.008),
            ("fourier_10", 10, 0.005),
            ("fourier_3", 3, 0.015),
            ("gauss_003", "gauss", 0.03),
            ("gauss_008", "gauss", 0.08),
            ("structural", "struct", 0.0),
        ]

        for name, arg1, arg2 in restart_configs:
            if time.monotonic() > deadline - 100:
                break

            if name.startswith("fourier"):
                p_start = _perturb_fourier_domain(best_seq, rng, n_modes=arg1, amplitude=arg2)
            elif name.startswith("gauss"):
                p_start = _perturb_gaussian(best_seq, arg1, rng)
            else:
                p_start = _perturb_structural(best_seq, rng)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            # Quick adaptive L-BFGS
            a_r = _lbfgs_adaptive(p_start, deadline, min_time_remaining=80.0)
            try_update(a_r, f"restart_{name}")

            # Quick Nesterov polish if promising
            if time.monotonic() < deadline - 60 and _score(a_r) < best_score_val + 0.01:
                a_r2, _ = _nesterov_subgrad(a_r, deadline - 60,
                                            tol_frac=1e-4, max_steps=2000, lr_init=0.02)
                try_update(a_r2, f"restart_{name}_sg")

    # === Phase 5: Aggressive local search ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 5: Local search (t={t_phase - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_ls, sc_ls = _local_search_aggressive(best_seq, deadline - 100, max_rounds=3000)
        try_update(a_ls, "local_search")
        print(f"  Phase 5 done: score={sc_ls:.10f}, t={time.monotonic() - t_start:.1f}s")

    # === Phase 6: Try different sequence lengths ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 6: Different lengths (t={t_phase - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n_cur = len(best_seq)
        for target_n in [2048, 3072, 5120, 8192]:
            if time.monotonic() > deadline - 60:
                break
            if abs(target_n - n_cur) < 256:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_adaptive(a_interp, deadline, min_time_remaining=40.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 7: Final aggressive polish cycle ===
    t_phase = time.monotonic()
    print(f"\n=== Phase 7: Final polish (t={t_phase - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_fp, sc_fp = _final_polish_cycle(best_seq, deadline, min_remaining=5.0)
        try_update(a_fp, "final_polish")

        # One more round of local search on the polished result
        if time.monotonic() < deadline - 5:
            a_ls2, _ = _local_search_aggressive(best_seq, deadline - 5, max_rounds=1000)
            try_update(a_ls2, "final_local")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
