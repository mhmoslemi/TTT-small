import numpy as np

import math

def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    a_rev = a[::-1]

    it = 0

        beta = beta_schedule[min(it // max(1, max_iter // num_phases), num_phases - 1)]

Actually, for continuous densities, if f is a density on [0,1], then the sum density g(x) = ∫_0^1 f(t)f(x-t)dt. We want to minimize max g(x). By the uncertainty principle type arguments, the minimum is achieved by a specific distribution.

For discrete sequences, the optimal sequence likely has some periodic structure or follows a specific formula.

Let me try a different practical approach: **Simulated annealing with structured moves**.

At each step:
1. Pick a random subset of indices
2. Try to redistribute the mass within that subset to reduce the max convolution
3. Accept with probability exp(-Δ/J) where Δ is the change in objective

This is like a Markov Chain Monte Carlo approach.

Actually, here's another idea: **Spectral method**.

In the Fourier domain, conv(a,a) = ifft(FFT(a)²). So we want to minimize max(ifft(FFT(a)²)). 

If we parameterize a in the Fourier domain (i.e., choose the FFT coefficients), the constraint is that a is real and non-negative. This is a complex constraint.

But here's a thought: if we could make FFT(a)² have a flat inverse FFT, that would be ideal. The inverse FFT of a constant is a delta function, which isn't what we want. We want ifft(FFT(a)²) to be as flat as possible.

"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
- FISTA with adaptive step size and geometric beta annealing
- Top-k active set gradient tracking
- Adam with aggressive LR schedule

import numpy as np
import time
import math

def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_smoothed(a, beta, k_active):
    """Compute smoothed gradient with top-k active set."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if k_active > 1 and c_max > 0:
        k = min(k_active, len(c))
        # Find top-k values
        top_vals = np.partition(c, len(c) - k)[len(c) - k:]
        threshold = top_vals.min()
        active = c >= threshold - 1e-15 * c_max
        w = active.astype(np.float64)
        w_sum = w.sum()
        if w_sum > 0:
            w = w / w_sum
        else:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
    else:
        # Smoothed weights
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    
    # Compute gradient: 2 * (w * a_rev)[n-1:2n-1]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista_adaptive(a_init, deadline, k_schedule):
    """
    FISTA with adaptive step size and scheduled active set size.
    k_schedule: list of (fraction_of_iterations, k_value) pairs
    """
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    beta = 0.05 * max_c
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    # Estimate Lipschitz constant
    L = 4.0 * n
    step = 1.0 / (L * max(beta, 1e-15))
    
    max_iter = 200000
    it = 0
    
    while time.monotonic() < deadline and it < max_iter:
        # Determine k from schedule
        frac = min(1.0, it / max(1, max_iter))
        k_val = n  # default: full smoothing
        for f, k in k_schedule:
            if frac >= f:
                k_val = k
        if k_val <= 1:
            k_val = 1
        
        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute gradient
        if k_val == 1:
            c_y = _full_convolve(y, y)
            c_max_y = float(c_y.max())
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
            a_rev = y[::-1]
            wc = _full_convolve(w, a_rev)
            grad = 2.0 * wc[n - 1:2 * n - 1]
        else:
            grad = _grad_smoothed(y, beta, k_val)
        
        # Adaptive step: cap the gradient norm
        grad_norm = float(np.linalg.norm(grad))
        if grad_norm > 1e-10:
            step = min(step, 0.5 / grad_norm)
        
        a_next = _proj_simplex(y - step * grad)
        
        # Periodic evaluation
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score and np.isfinite(sc):
                best_score = sc
                best_a = a_next.copy()
                # Reduce beta
                c_new = _full_convolve(a_next, a_next)
                max_c_new = float(c_new.max())
                beta = max(0.3 * max_c_new, 1e-12)
                step = 1.0 / (L * max(beta, 1e-15))
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
        it += 1
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_optimized(a_init, deadline, beta, lr_init=0.005, k_active=50):
    """Adam with top-k active set and aggressive LR decay."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    current_lr = lr_init
    
    max_iter = 300000
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        
        # Compute gradient
        grad = _grad_smoothed(a, beta, k_active)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        
        # Gradient norm capping
        update_norm = float(np.linalg.norm(update))
        max_norm = 0.5
        if update_norm > max_norm:
            update = update * (max_norm / update_norm)
        
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        
        # LR decay
        if it % 2000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _lbfgs_refine(a_init, deadline, beta):
    """L-BFGS-B refinement on log-space."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()
    
    k_active = max(10, n // 100)
    
    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Use top-k smoothing for speed
        k = min(k_active, len(c))
        top = np.partition(c, len(c) - k)[len(c) - k:]
        thresh = top.min()
        active = c >= thresh - 1e-15
        if active.sum() > 0:
            return 2.0 * n * float(c[active].max())
        return 2.0 * n * cm
    
    def grad_fn(z):
        a = _softmax_compute(z)
        grad = _grad_smoothed(a, beta, k_active)
        correction = float(np.dot(a, grad))
        return a * (grad - correction)
    
    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score
    
    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 20)), 'gtol': 1e-14, 'maxls': 10}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    
    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _make_sinusoidal_exp(n, freq=1.0, amp=1.0, phase=0.0):
    """Create exp(sin(freq * 2π * i/n + phase)) profile."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = np.exp(amp * np.sin(2.0 * np.pi * freq * t + phase))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_power_law(n, alpha=0.5):
    """Create a power-law taper profile."""
    t = np.linspace(0.5, 1.5, n)
    center = (n - 1) / 2.0
    dist = np.abs(np.arange(n) - center) / center
    a = (1.0 + dist) ** (-alpha)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _symmetrize(a):
    """Symmetrize: (a + reverse(a)) / 2"""
    return (a + a[::-1].copy()) / 2.0

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

import numpy as np

def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, w, n):
    """Gradient of sum(w_k * conv(a,a)_k) w.r.t. a = 2*(w * a_rev)[n-1:2n-1]"""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _get_weights(c, beta, k_active):
    """Get subgradient weights: top-k for k>1, softmax for k=1 with beta."""
    c_max = float(c.max())
    if c_max <= 0:
        n = len(c)
        w = np.zeros(n)
        w[0] = 1.0
        return w
    if k_active > 1:
        k = min(k_active, len(c))
        top_vals = np.partition(c, len(c) - k)[len(c) - k:]
        threshold = top_vals.min()
        tol = 1e-12 * c_max
        active = c >= threshold - tol
        w = active.astype(np.float64)
        ws = w.sum()
        if ws > 0:
            return w / ws
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    else:
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
            return w
        return e / s

def _lbfgs_refine(a_init, deadline, beta):

    from scipy.optimize import minimize
    n = len(a_init)

def _make_power_law(n, alpha=0.5):

        c_curr = _full_convolve(a_work, a_work)
        max_c = float(c_curr.max())
        beta_lb = 0.0001 * max_c
        a_lb, _ = _lbfgs_refine(a_work, phase3_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 4: Symmetrization + structured exploration ===
    print(f"Phase 4: Symmetry + exploration (t={time.monotonic()-t_start:.1f}s)")
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 10:
        # Try symmetrization
        a_sym = _symmetrize(a_work)
        a_sym = np.maximum(a_sym, 0)
        a_sym = a_sym / a_sym.sum()
        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_seq = a_sym.copy()
            print(f"  Symmetrized: score={sc_sym:.10f}")

        # Try anti-symmetric (abs of odd part)
        a_odd = a_work - _symmetrize(a_work)
        a_odd = np.abs(a_odd)
        if a_odd.sum() > 1e-10:
            a_odd = a_odd / a_odd.sum()
            sc_odd = _score(a_odd)
            if sc_odd < best_score:
                best_score = sc_odd
                best_seq = a_odd.copy()
                print(f"  Anti-sym: score={sc_odd:.10f}")

        # Try mixed: 0.9*work + 0.1*sym
        a_mix = 0.9 * a_work + 0.1 * _symmetrize(a_work)
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()
        sc_mix = _score(a_mix)
        if sc_mix < best_score:
            best_score = sc_mix
            best_seq = a_mix.copy()
            print(f"  Mixed: score={sc_mix:.10f}")

    # === Phase 5: Multi-restart ===
    print(f"Phase 5: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 15:
        n_restarts = 6
        per_r = (phase5_end - time.monotonic() - 10) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 5:
                break

            if i == 0:
                p = _perturb(a_work, 0.008, rng)
            elif i == 1:
                # Sinusoidal exponential blend
                p_sin = _make_sinusoidal_exp(n_main, freq=1.0, amp=0.2,
                                             phase=rng.uniform(0, 2 * np.pi))
                p = 0.8 * a_work + 0.2 * p_sin
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                # Power law blend
                p_pl = _make_power_law(n_main, alpha=rng.uniform(0.2, 0.6))
                p = 0.7 * a_work + 0.3 * p_pl
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 3:
                # Frequency domain perturbation
                N = _next_pow2(2 * n_main - 1)
                fa = np.fft.rfft(a_work, N)
                mag = np.abs(fa)
                phase_pert = rng.standard_normal(fa.shape[0]) * 0.02
                fa_new = fa * (1.0 + 0.01 * rng.standard_normal(fa.shape[0]))
                fa_new[0] = np.abs(fa_new[0])  # keep DC positive
                # Enforce real inverse
                p = np.fft.irfft(fa_new, N)[:n_main]
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 4:
                # Scale-free random perturbation
                z = np.log(np.maximum(a_work, 1e-15))
                pert = rng.standard_normal(n_main) * 0.02
                # Low-pass filter the perturbation
                Np = _next_pow2(n_main)
                fpert = np.fft.rfft(pert, Np)
                cutoff = n_main // 32
                if cutoff < fa.shape[0]:
                    fpert[cutoff:] *= 0.1
                pert = np.fft.irfft(fpert, Np)[:n_main]
                z = z + pert
                z = z - z.mean()
                p = _softmax_compute(z)
            else:
                # Larger perturbation
                p = _perturb(a_work, 0.02, rng)

            # Quick FISTA
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            k_sched = [(0.0, 1), (0.3, 5), (0.6, 30), (1.0, 150)]
            p, _ = _fista_bb(p, time.monotonic() + per_r * 0.5, 0.005 * max_cp, k_sched)
            p = p / p.sum()

            # Quick Adam
            c_p2 = _full_convolve(p, p)
            max_cp2 = float(c_p2.max())
            p, _ = _adam_refined(p, time.monotonic() + per_r * 0.5,
                                 0.0003 * max_cp2, 0.002, 30)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-polish ===
    print(f"Phase 6: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5

        if per_t > 15:
            # Ultra-fine Adam
            t_adam_end = time.monotonic() + per_t * 0.45
            a_pol, _ = _adam_refined(a_pol, t_adam_end,
                                     0.00001 * max_cp, 0.0008, 5)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")

            # Final L-BFGS
            if time.monotonic() < deadline - 8:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000005 * float(c_f.max())
                a_pol, _ = _lbfgs_refine(a_pol, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
