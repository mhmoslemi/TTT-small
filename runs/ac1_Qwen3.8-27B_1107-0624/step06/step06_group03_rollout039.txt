```python
def lp_direction(f, M):
    """Find g that maximizes sum(g) subject to g >= 0 and ||f*g||_inf <= M."""
    n = len(f)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)_i <= M for i = 0, ..., 2n-2
    # (f*g)_i = sum_j f_j * g_{i-j} (valid indices)
    # For each i, the constraint involves g_k where k = i-j, so j = i-k
    # (f*g)_i = sum_k f_{i-k} * g_k (where 0 <= i-k < n and 0 <= k < n)
    
    # Build constraint matrix A such that A @ g <= M * ones(2n-1)
    # A[i, k] = f[i-k] if 0 <= i-k < n, else 0
    
    # This is a Toeplitz-like matrix
    # A is (2n-1) x n
    # A[i, k] = f[i-k] if 0 <= i-k < n
    
    # Actually, let's think of it as: for each i (row) and k (column):
    # A[i,k] = f[i-k] if 0 <= i-k < n, else 0
    
    # This can be built as a sum of shifted copies of f
    A = np.zeros((2*n-1, n))
    for k in range(n):
        # Column k: A[i, k] = f[i-k] for valid i-k
        # i-k ranges from 0 to n-1, so i ranges from k to k+n-1
        valid_i = np.arange(k, min(k+n, 2*n-1))
        A[valid_i, k] = f[valid_i - k]
    
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if result.success:
        return result.x
    else:
        return None
```

This could be larger than `M` if `(f*g)_j` is much larger than `(f*f)_j`. But wait, `(f*g)_j <= M` by the LP constraint. So:
`(f_new*f_new)_j <= (f*f)_j + 2t*(M - (f*f)_j)`

For this to be `<= M`, we need:
`(f*f)_j + 2t*(M - (f*f)_j) <= M`
`2t*(M - (f*f)_j) <= M - (f*f)_j`
`2t <= 1` (when `(f*f)_j < M`)

So if `t <= 0.5`, all positions satisfy the bound to first order! This confirms the claim.

def lp_direction_cutting_plane(f, M, max_iters=20, initial_k=50):
    n = len(f)
    conv_ff = np.convolve(f, f)
    
    # Start with top-K constraints
    active = np.argsort(conv_ff)[-initial_k:]
    
    g = f.copy()  # Initial guess (feasible)
    
    for _ in range(max_iters):
        # Build LP with only active constraints
        A = np.zeros((len(active), n))
        for idx, i in enumerate(active):
            for k in range(max(0, i-n+1), min(n, i+1)):
                A[idx, k] = f[i-k]
        
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(len(active), M)
        bounds = [(0, None)] * n
        result = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not result.success:
            break
        
        g = result.x
        
        # Check for violated constraints
        conv_fg = np.convolve(f, g)
        violations = np.where(conv_fg > M + 1e-10)[0]
        
        if len(violations) == 0:
            break
        
        # Add the most violated constraints
        new_active = np.union1d(active, violations[:10])
        active = new_active
    
    return g
```

"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses cutting-plane LP direction finding, gradient-based fine-tuning, and coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _lp_direction(f, M, max_cp_iters=30, initial_k=100):
    """Find g maximizing sum(g) subject to g >= 0, (f*g)_i <= M for all i.
    Uses cutting-plane method."""
    n = len(f)
    
    # Start with top-K constraints
    conv_ff = _full_convolve(f, f)
    active = np.argsort(conv_ff)[-initial_k:]
    
    g = f.copy()  # Feasible initial guess
    
    for cp_iter in range(max_cp_iters):
        # Build constraint matrix for active constraints
        m = len(active)
        A = np.zeros((m, n))
        for idx, i in enumerate(active):
            lo = max(0, i - n + 1)
            hi = min(n, i + 1)
            k_range = np.arange(lo, hi)
            A[idx, k_range] = f[i - k_range]
        
        # Solve LP
        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'presolve': True})
        
        if not result.success:
            break
        
        g_candidate = result.x
        if g_candidate is None:
            break
        
        # Check all constraints
        conv_fg = _full_convolve(f, g_candidate)
        violated = np.where(conv_fg > M * (1 + 1e-8) + 1e-12)[0]
        
        if len(violated) == 0:
            g = g_candidate
            break
        
        # Add violated constraints (most violated first)
        viol_amounts = conv_fg[violated] - M
        worst = violated[np.argsort(-viol_amounts)][:20]
        active = np.unique(np.concatenate([active, worst]))
        # Keep at most 500 constraints
        if len(active) > 500:
            # Keep the most binding ones
            conv_vals = conv_fg[active]
            keep = active[np.argsort(-conv_vals)[:500]]
            active = keep
        
        g = g_candidate
    
    return g


def _adam_finetune(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.97)
        
        m_buf *= 0.5
        v_buf *= 0.5
    
    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, beta, k):
    n_c = len(c)
    k = min(k, n_c)
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, w, n):
    """Gradient in log-space for normalized problem (sum a = 1)."""
    # grad_a_i = 2 * sum_j w_j * a_{j-i} (valid indices)
    # This is 2 * cross-correlation of a with w
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    grad_a = 2.0 * h
    mean_grad = float(np.dot(grad_a, a))
    return n * a * (grad_a - mean_grad) * 2  # *2 for the 2n factor


def _coord_descend(a, n_passes, top_k_w, deadline):
    """Coordinate descent polishing."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i_old = a[i]
            if a_i_old < 1e-15:
                continue
            
            c = _full_convolve(a, a)
            
            # d_j = a_{j-i} for valid j
            d = np.zeros(2 * n - 1)
            valid_lo = max(0, i)
            valid_hi = min(2 * n - 1, i + n)
            d[valid_lo:valid_hi] = a[valid_lo - i : valid_hi - i]
            
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old
            
            # Candidate values
            t_candidates = [0.0, a_i_old * 0.3, a_i_old * 0.5, a_i_old * 0.7,
                          a_i_old, a_i_old * 1.3, a_i_old * 1.5, a_i_old * 2.0]
            
            # Add critical points from top constraints
            idx_top = np.argsort(c)[-top_k_w:]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if -1.0 <= t_star <= 5.0 * max(a_i_old, 0.1):
                        t_candidates.append(max(0, t_star))
            
            t_candidates = list(set([max(0.0, tc) for tc in t_candidates]))
            
            best_t = a_i_old
            best_f = _score_norm(a)
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max_c / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

```python
"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses cutting-plane LP direction finding (the "state of the art" method), Adam in log-space
for fine-tuning, and coordinate descent polishing. Multi-length exploration (4096, 8192).
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_lp_matrix(f, active, n):
    """Build the constraint matrix A where A[idx, k] = f[active[idx]-k] if valid."""
    m = len(active)
    A = np.zeros((m, n))
    for idx, i in enumerate(active):
        lo = max(0, i - n + 1)
        hi = min(n - 1, i)
        if lo <= hi:
            A[idx, lo:hi + 1] = f[i - hi:i - lo + 1][::-1]
    return A


def _lp_direction(f, M, max_cp_iters=25, initial_k=150, max_constraints=400):
    """Find g maximizing sum(g) s.t. g >= 0, (f*g)_i <= M for all i. Cutting-plane."""
    n = len(f)
    conv_ff = _full_convolve(f, f)
    active = np.argsort(conv_ff)[-initial_k:]
    
    for cp_iter in range(max_cp_iters):
        m = len(active)
        A = _build_lp_matrix(f, active, n)
        
        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n
        try:
            result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                            method='highs', options={'presolve': True, 'time_limit': 30})
        except Exception:
            break
        
        if not result.success or result.x is None:
            break
        
        g_candidate = result.x
        
        # Check all constraints
        conv_fg = _full_convolve(f, g_candidate)
        tol = M * 1e-10 + 1e-14
        violated = np.where(conv_fg > M + tol)[0]
        
        if len(violated) == 0:
            return g_candidate
        
        # Add most violated constraints
        viol_amounts = conv_fg[violated] - M
        worst = violated[np.argsort(-viol_amounts)][:30]
        active = np.unique(np.concatenate([active, worst]))
        
        if len(active) > max_constraints:
            conv_vals = conv_fg[active]
            keep = active[np.argsort(-conv_vals)[:max_constraints]]
            active = keep
    
    return None


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _topk_weights(c, beta, k):
    n_c = len(c)
    k = min(k, n_c)
    topk_thresh = np.partition(c, -k)[-k]
    topk_mask = c >= topk_thresh
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _compute_grad_a(a, w, n):
    """Compute grad_a_i = 2 * sum_j w_j * a_{j-i} (valid indices)."""
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _logspace_grad(a, w, n):
    """Gradient in log-space for F(a) = 2n*sum_j w_j*(conv(a,a))_j with sum(a)=1."""
    grad_a = _compute_grad_a(a, w, n)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _adam_finetune(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_weights(c, beta, top_k)
            grad = _logspace_grad(a, w, n)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

                current_lr = max(0.0003, current_lr * 0.97)

    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

def _coord_descend(a, n_passes, top_k_w, deadline):

        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i_old = a[i]
            if a_i_old < 1e-14:
                continue
            
            c = _full_convolve(a, a)
            
            d = np.zeros(2 * n - 1)
            valid_lo = max(0, i)
            valid_hi = min(2 * n - 1, i + n)
            d[valid_lo:valid_hi] = a[valid_lo - i:valid_hi - i]
            
            c_base = c - a_i_old * d
            S0 = 1.0 - a_i_old
            
            t_candidates = [0.0, a_i_old * 0.2, a_i_old * 0.4, a_i_old * 0.6,
                          a_i_old * 0.8, a_i_old, a_i_old * 1.2, a_i_old * 1.5,
                          a_i_old * 2.0, a_i_old * 3.0]
            
            idx_top = np.argsort(c)[-top_k_w:]
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if -0.5 <= t_star <= 5.0 * max(a_i_old, 0.05):
                        t_candidates.append(max(0.0, t_star))
                        t_candidates.append(max(0.0, t_star * 0.7))
                        t_candidates.append(max(0.0, t_star * 1.3))
            
            t_candidates = list(set([max(0.0, min(tc, 10.0 * max(a_i_old, 1.0))) for tc in t_candidates]))
            
            best_t = a_i_old
            best_f = _score_norm(a)
            for tc in t_candidates:
                if abs(tc - a_i_old) < 1e-15:
                    continue
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                denom = (S0 + tc) ** 2
                if denom < 1e-30:
                    continue
                f_val = 2.0 * n * max_c / denom
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5]:
            a += 0.25 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fista_smooth(a_init, deadline, beta, max_iter, top_k=100):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w = _topk_weights(c_y, beta, top_k)
        grad = _compute_grad_a(y, w, n)
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score
