"""
L-BFGS-B based optimizer with frequency-domain diversification for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses softmax parameterization, aggressive beta annealing, multi-length strategy,
and active-set polishing to improve upon previous results.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _full_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def _lse_smoothed_obj_grad(z, beta):
    """
    Compute the smoothed objective and gradient w.r.t. z.
    a = softmax(z), objective = 2*n*LSE(beta, conv(a,a))
    """
    n = len(z)
    a = _softmax(z)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

    # Softmax-smoothed max
    bc = beta * c
    bc_shift = bc - bc.max()
    p = np.exp(bc_shift)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= p_sum

    LSE_val = float(np.dot(p, c))
    obj = 2.0 * n * LSE_val

    # Gradient: d(obj)/da_j = 4*n * h[j] where h[j] = sum_m p[j+m]*a[m]
    # Compute h via FFT cross-correlation
    N2 = _next_pow2(3 * n - 2)
    fp = np.fft.fft(p, N2)
    fa2 = np.fft.fft(a, N2)
    h = np.fft.ifft(fp * np.conj(fa2)).real[:n]

    grad_a = 4.0 * n * h

    # Chain rule through softmax: d(obj)/dz_k = 4*n*a_k*(h_k - dot(h,a))
    dot_ha = float(np.dot(h, a))
    grad_z = 4.0 * n * a * (h - dot_ha)

    return obj, grad_z


def _lbfgs_smoothed(z_init, beta_list, deadline, max_iter_per_beta=200):
    """Run L-BFGS-B with beta annealing."""
    z_curr = z_init.copy()
    best_score = np.inf
    best_z = z_curr.copy()

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def fun(z):
            return _lse_smoothed_obj_grad(z, beta)[0]

        def jac(z):
            return _lse_smoothed_obj_grad(z, beta)[1]

        try:
            res = minimize(
                fun, z_curr, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-12, 'gtol': 1e-10}
            )
            z_curr = res.x
        except Exception:
            break

        a = _softmax(z_curr)
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_z = z_curr.copy()

    return best_z, best_score


def _adam_logspace_v2(a_init, deadline, beta_list, lr=0.01, max_iter=100000):
    """Adam optimizer in log-space with LSE smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        phase_iter = max(1000, max_iter // len(beta_list))

        for _ in range(phase_iter):
            if time.monotonic() > deadline:
                break
            t += 1

            a = _softmax(z)
            N = _next_pow2(2 * n - 1)
            fa = np.fft.rfft(a, N)
            c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

            bc = beta * c
            bc_shift = bc - bc.max()
            e = np.exp(bc_shift)
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            # Gradient in z-space
            N2 = _next_pow2(3 * n - 2)
            fw = np.fft.fft(w, N2)
            fa2 = np.fft.fft(a, N2)
            h = np.fft.ifft(fw * np.conj(fa2)).real[:n]
            dot_ha = float(np.dot(h, a))
            grad = 4.0 * n * a * (h - dot_ha)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 200 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        # Gradual LR decay
        lr = max(0.001, lr * 0.7)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _freq_flatten_restart(a, n_new, rng):
    """Create a new sequence by flattening the frequency magnitude of a."""
    n = len(a)
    a_norm = a / a.sum()
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a_norm, N)
    mag = np.abs(fa)
    phase = np.angle(fa)

    # Flatten magnitude with some randomness
    flat_mag = np.ones_like(mag)
    # Add some structure: boost or cut certain bands
    if rng.random() > 0.5:
        boost = np.exp(-((np.arange(len(flat_mag)) / len(flat_mag) - 0.5) ** 2) / 0.05)
        flat_mag = 0.5 + 0.5 * boost
    else:
        flat_mag = 1.0 + 0.3 * rng.random(len(flat_mag))

    new_spectrum = flat_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(new_spectrum, N)[:n_new]
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()


def _make_profile(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30.0 * (t - 0.3) ** 2) + np.exp(-30.0 * (t - 0.7) ** 2)
        return a / a.sum()
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "barker_tiled":
        # Use a Barker-inspired pattern
        barker = np.array([1, 1, 1, -1, -1, 1, -1])
        for i in range(n):
            pass
        a = np.array([1.0 if ((i * 7) % 13) in [0, 1, 2, 5, 7, 8, 12] else 0.1 for i in range(n)])
        return a / a.sum()
    else:
        return np.ones(n) / n


def _active_set_polish(a, deadline, n_top=5, n_iter=5000):
    """
    Focus on reducing the top convolution peaks.
    Uses a weighted sum of top peaks as the objective.
    """
    n = len(a)
    a_work = a.copy()
    best_score = _score_norm(a_work)
    best_a = a_work.copy()

    z = np.log(np.maximum(a_work, 1e-15))
    z = z - z.mean()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    lr = 0.005
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        N = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2 * n - 1]

        # Get top n_top peaks
        top_idx = np.argsort(c)[::-1][:n_top]
        top_c = c[top_idx]

        # Weighted objective: sum of top peaks (smoothed)
        beta = 50.0
        top_shifted = beta * top_c - (beta * top_c).max()
        w_top = np.exp(top_shifted)
        w_top /= w_top.sum()

        # Build full weight vector
        w = np.zeros_like(c)
        for i, idx in enumerate(top_idx):
            w[idx] = w_top[i]

        # Gradient
        N2 = _next_pow2(3 * n - 2)
        fw = np.fft.fft(w, N2)
        fa2 = np.fft.fft(a, N2)
        h = np.fft.ifft(fw * np.conj(fa2)).real[:n]
        dot_ha = float(np.dot(h, a))
        grad = 4.0 * n * a * (h - dot_ha)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if it % 100 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        # Decay learning rate
        if it % 500 == 0:
            lr *= 0.9

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        print(f"Eval error: {ex}")

    return [float(x) for x in final]
