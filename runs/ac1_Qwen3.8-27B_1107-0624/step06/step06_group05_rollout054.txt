"""
Multi-phase optimizer combining LP-based best response, Adam, and L-BFGS
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad(a, w, n):
    """Compute gradient of max(conv(a,a)) using weights w on peak positions."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 50 == 0:

            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def grad_fn(z):

            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]

        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    except Exception as e:

    return a_best, best_score


def _build_hankel(A_coeffs, n):
    """Build the (2n-1) x n Hankel matrix where H[k,j] = A_coeffs[k-j] if 0<=k-j<n else 0.
    We use the transposed version for efficiency: H^T is n x (2n-1) where H^T[j,k] = A_coeffs[k-j].
    """
    # H^T[j,k] = a[k-j] if 0 <= k-j < n, else 0
    # For row j, non-zero columns are k=j to k=j+n-1, values are a[0], a[1], ..., a[n-1]
    rows = []
    cols = []
    data = []
    for j in range(n):
        k_start = j
        k_end = j + n  # exclusive, max is 2n-1
        k_end = min(k_end, 2 * n - 1)
        idx = np.arange(k_start, k_end)
        rows.extend([j] * len(idx))
        cols.extend(idx.tolist())
        data.extend(A_coeffs.tolist())
    H_T = sp.csr_matrix((data, (rows, cols)), shape=(n, 2 * n - 1))
    return H_T


def _lp_best_response(a, M, max_iter_cp=50):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 using cutting plane.
    Returns the normalized solution g (sum=1)."""
    n = len(a)
    
    # Start with the constraint at the peak
    a_norm = a / a.sum()
    c_init = _full_convolve(a_norm, a_norm)
    peak_idx = int(np.argmax(c_init))
    
    # Track active constraints
    active_k = [peak_idx]
    
    g = None
    for cp_iter in range(max_iter_cp):
        m = len(active_k)
        # Build constraint matrix for active constraints
        # Constraint k: sum_j a[k-j] * g_j <= M (where 0 <= k-j < n)
        # This is a row vector times g
        rows_data = []
        row_indices = []
        col_indices = []
        for i, k in enumerate(active_k):
            # g_j appears when 0 <= k-j < n, i.e., max(0,k-n+1) <= j <= min(n-1,k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k) + 1
            for j in range(j_start, j_end):
                rows_data.append(a[k - j])
                row_indices.append(i)
                col_indices.append(j)
        
        if len(rows_data) == 0:
            g = np.ones(n) / n
            break
            
        A_cp = sp.csc_matrix((rows_data, (row_indices, col_indices)), shape=(m, n))
        b_cp = np.full(m, M)
        
        # Maximize sum(g) = - minimize -sum(g)
        c_obj = -np.ones(n)
        
        try:
            res = linprog(c_obj, A_ub=A_cp, b_ub=b_cp, bounds=[(0, None)] * n, method='highs')
            if not res.success:
                break
            g = res.x
        except Exception:
            break
        
        # Check if solution violates any other constraints
        if g is not None and g.sum() > 1e-10:
            conv_check = _full_convolve(a_norm, g)
            violated = np.where(conv_check > M * 1.001)[0]
            if len(violated) == 0:
                break
            # Add the most violated constraint
            worst = int(np.argmax(conv_check))
            if worst not in active_k:
                active_k.append(worst)
                # Limit active constraints to avoid LP blowup
                if len(active_k) > 200:
                    # Keep only the most binding ones
                    conv_vals = conv_check[active_k]
                    keep = np.argsort(conv_vals)[::-1][:100]
                    active_k = [active_k[i] for i in keep]
            else:
                # If the worst is already in active set, no progress
                break
    
    if g is None or g.sum() < 1e-10:
        g = np.ones(n) / n
    else:
        g = g / g.sum()
    return g


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    s_curr = a_curr.sum()
    L = 4.0 * n / (s_curr * s_curr) if s_curr > 0 else 4.0 * n
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w, n)
        y2 = y - step * grad
        y2 = np.maximum(y2, 0.0)
        s_y2 = y2.sum()
        if s_y2 > 1e-30:
            y2 = y2 / s_y2
        else:
            y2 = np.ones(n) / n
        a_prev = a_curr.copy()
        a_curr = y2
        t_val = t_next
        if it % 100 == 0:
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

"""
Multi-phase optimizer combining LP-based best response, FISTA, Adam, and L-BFGS
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: LP-based best response (solving max sum(g) s.t. conv(a,g)<=M, g>=0)
provides a stronger improvement direction than pure gradient methods.

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad(a, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 5000 == 0:
            current_lr = max(0.0002, current_lr * 0.92)
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init.copy(), best_score
    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 15)), 'gtol': 1e-10, 'maxls': 8}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _lp_solve_cutting_plane(a, M, max_cp=80, max_active=150):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0 via cutting plane."""
    n = len(a)
    active_k = []
    peak_idx = int(np.argmax(_full_convolve(a, a)))
    active_k.append(peak_idx)
    g = None
    for cp_iter in range(max_cp):
        m = len(active_k)
        rows_data = []
        row_idx = []
        col_idx = []
        for i, k in enumerate(active_k):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows_data.append(a[k - j])
                row_idx.append(i)
                col_idx.append(j)
        if len(rows_data) == 0:
            g = np.ones(n) / n
            break
        A_cp = sp.csc_matrix((rows_data, (row_idx, col_idx)), shape=(m, n))
        b_cp = np.full(m, M)
        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A_cp, b_ub=b_cp, bounds=bounds, method='highs')
            if not res.success or res.x is None:
                break
            g = np.maximum(res.x, 0.0)
        except Exception:
            break
        if g is not None and g.sum() > 1e-15:
            conv_check = _full_convolve(a, g)
            violated = np.where(conv_check > M * 1.0005 + 1e-15)[0]
            if len(violated) == 0:
                break
            worst = int(np.argmax(conv_check))
            if worst not in active_k:
                active_k.append(worst)
                if len(active_k) > max_active:
                    conv_vals = np.array([conv_check[k] for k in active_k])
                    keep = active_k[np.argsort(conv_vals)[::-1][:80]]
                    active_k = keep
            else:
                break
        elif g is None or g.sum() < 1e-15:
            g = np.ones(n) / n
            break
    if g is None or g.sum() < 1e-15:
        g = np.ones(n) / n
    else:
        g = g / g.sum()
    return g


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = 1.0 / (4.0 * n)
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w, n)
        y2 = y - step * grad
        y2 = np.maximum(y2, 0.0)
        s_y2 = y2.sum()
        if s_y2 > 1e-30:
            y2 = y2 / s_y2
        else:
            y2 = np.ones(n) / n
        a_prev = a_curr.copy()
        a_curr = y2
        t_val = t_next
        if it % 100 == 0:
            sc = _score_norm(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-6.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_bump":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2 + 0.3 * np.exp(-16.0 * (t - 0.3) ** 2) + 0.3 * np.exp(-16.0 * (t - 0.7) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

        sc = _score_norm(a_new)
        if sc < local_best_score:

            no_improve = 0
            if lp_iter % 5 == 0:
                print(f"    LP {lp_iter}: score={sc:.10f} M={M:.6e} ({t_lp1-t_lp0:.2f}s)")
        else:
            no_improve += 1
            if no_improve >= 8:
                print(f"    LP plateau at iter {lp_iter}, stopping")

        a_work = a_new

    # Phase B: FISTA continuation
    fista_end = time.monotonic() + 0.25 * (deadline - time.monotonic())

            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
