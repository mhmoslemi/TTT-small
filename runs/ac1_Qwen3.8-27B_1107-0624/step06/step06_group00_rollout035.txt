    # g[j] = sum_k p[k] * a[k-j]
    # If I define q = a (length n), then g[j] = sum_k p[k]*q[k-j]
    # This is (p * q_rev) reversed, where q_rev = q[::-1]
    # (p * q_rev)[m] = sum_j p[j]*q_rev[m-j] = sum_j p[j]*q[n-1-(m-j)] = sum_j p[j]*q[n-1-m+j]
    # We want g[j] = sum_k p[k]*a[k-j] = sum_k p[k]*q[k-j]
    # Let's set m-j = n-1-k, so k = n-1-m+j... this is getting circular.
    
    # Simplest: just compute it directly using the fact that
    # g[j] = sum_k p[k]*a[k-j] = (a * p)[j] in the "correlation" sense
    # = sum_k a[k]*p[k-j]... no that's different.

    # a = [a0, a1, a2], p = [p0, p1, p2, p3, p4] (length 5 = 2*3-1)

def _sproll_step(a, max_iter=100, tol=1e-10):
    """Sproll-type iterative refinement.
    repeatedly: find max of conv, reduce variables contributing to it.
    """
    n = len(a)
    a = a.copy()
    for _ in range(max_iter):
        c = _fft_convolve(a, a)
        M = c.max()
        # Find positions where c is close to max
        threshold = M * (1 - 1e-6)
        active = np.where(c >= threshold)[0]
        # For each active position k, the variables a[j] and a[k-j] contribute
        # Reduce them proportionally
        # Simple approach: for each active k, reduce a[j] for all j where a[j]*a[k-j] > 0
        # Better: reduce the "gradient" direction
        g = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                g[j] += 2.0 * a[k - j]
        # Scale g to be a step in the right direction
        step = M / (np.max(g) + 1e-15)
        a_new = a - step * g * 0.1  # damped
        a_new = np.maximum(a_new, 0)
        if a_new.sum() < 1e-10:
            break
        a_new = a_new / a_new.sum()
        # Check improvement
        c_new = _fft_convolve(a_new, a_new)
        if c_new.max() < M - tol:
            a = a_new
        else:
            # Try smaller step
            a_new = a - step * g * 0.01
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-10:
                break
            a_new = a_new / a_new.sum()
            c_new = _fft_convolve(a_new, a_new)
            if c_new.max() < M - tol:
                a = a_new
            else:
                break
    return a


def _subgrad_descend(a, deadline, active_k=10, max_iter=500000,
                      base_lr=0.01, min_lr=0.0001, patience=20,
                      eval_interval=100):
    """Subgradient descent with Polyak averaging and active-set gradient."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    patience_ctr = 0
    lr = base_lr
    avg_a = a.copy()
    avg_count = 1
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        M = c.max()
        # Get active set
        k_indices = np.argsort(c)[::-1][:active_k]
        g = _active_gradient(a, c, k_indices)
        # Center the gradient (project out the mean direction for simplex constraint)
        g_mean = g.mean()
        g_cent = g - g_mean
        # Polyak step size
        # f(a) - f* >= -M_base (we don't know f*, use initial as upper bound)
        # step = (f(a) - f*) / ||g||^2, but we use a simpler adaptive scheme
        step = lr / (np.linalg.norm(g_cent) + 1e-10)
        a_new = a - step * g_cent
        a_new = _project_simplex(a_new)
        # Evaluate
        avg_a = avg_a + a_new
        avg_count += 1
        a = a_new
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            sc_avg = _score_norm(avg_a / avg_count)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                patience_ctr = 0
            elif sc_avg < best_sc:
                best_sc = sc_avg
                best_a = (avg_a / avg_count).copy()
                patience_ctr = 0
            else:
                patience_ctr += 1
            if patience_ctr > patience and lr > min_lr:
                lr *= 0.7
                patience_ctr = 0
                a = best_a.copy()
    
    # Final check
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _adam_simplex(a, deadline, active_k=10, lr_start=0.05, lr_end=0.005,
                  max_iter=300000, eval_interval=100):
    """Adam optimizer directly in simplex space (not z-space)."""
    n = len(a)
    a = a.copy()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_sc = _score_norm(a)
    patience = 0
    lr_mult = 1.0
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1
        frac = min(1.0, t_count / max_iter)
        lr = (lr_start * (lr_end / lr_start) ** frac) * lr_mult
        
        c = _fft_convolve(a, a)
        k_indices = np.argsort(c)[::-1][:active_k]
        g = _active_gradient(a, c, k_indices)
        g_mean = g.mean()
        g_cent = g - g_mean
        
        m_buf = b1 * m_buf + (1.0 - b1) * g_cent
        v_buf = b2 * v_buf + (1.0 - b2) * (g_cent * g_cent)
        mh = m_buf / (1.0 - b1 ** t_count)
        vh = v_buf / (1.0 - b2 ** t_count)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        a_new = a - update
        a_new = _project_simplex(a_new)
        a = a_new
        
        if t % eval_interval == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            if patience > 15 and lr_mult > 0.1:
                lr_mult *= 0.7
                patience = 0
                a = best_a.copy()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc


def _line_search_grad_descend(a, deadline, active_k=10, max_iter=100000,
                               init_lr=0.01, min_lr=1e-6):
    """Gradient descent with backtracking line search."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    lr = init_lr
    patience = 0
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _fft_convolve(a, a)
        k_indices = np.argsort(c)[::-1][:active_k]
        g = _active_gradient(a, c, k_indices)
        g_mean = g.mean()
        g_cent = g - g_mean
        g_norm = np.linalg.norm(g_cent)
        if g_norm < 1e-12:
            break
        
        # Backtracking line search
        step = lr
        accepted = False
        for _ in range(20):
            a_new = _project_simplex(a - step * g_cent)
            sc_new = _score_norm(a_new)
            if sc_new < best_sc or sc_new < _score_norm(a):
                a = a_new
                accepted = True
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a.copy()
                    patience = 0
                break
            step *= 0.5
        if not accepted:
            lr *= 0.5
            patience += 1
            if patience > 20 and lr < min_lr:
                break
            a = best_a.copy()
        else:
            patience = 0
            lr = min(lr * 1.05, init_lr)
    
    return best_a, best_sc


def _multi_active_lp(a, deadline, time_per_lp=10, max_lp=200):
    """LP-based refinement: maximize sum(g) subject to conv(a,g) <= M, g >= 0.
    Then blend a with g."""
    from scipy.optimize import linprog
    from scipy import sparse as sp
    
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    lp_count = 0
    
    while time.monotonic() < deadline and lp_count < max_lp:
        c = _fft_convolve(a, a)
        M = c.max()
        # Build sparse matrix for conv(a, g) constraints
        # conv(a,g)[k] = sum_j a[j]*g[k-j] <= M
        # For each k, the constraint is: sum_j a[j]*g[k-j] <= M
        # which is linear in g with coefficients a[j] at appropriate positions
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
        
        if len(rows) == 0:
            break
            
        A = sp.csc_matrix((np.array(vals), (np.array(rows), np.array(cols))),
                          shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': time_per_lp})
            if res.status != 0:
                break
            g = np.maximum(res.x, 0.0)
            if g.sum() < 1e-15:
                break
            g = g / g.sum()
            # Blend
            best_t = 0
            for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_cand = (1 - t_val) * a + t_val * g
                sc = _score_norm(a_cand)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_cand.copy()
                    best_t = t_val
            if best_t > 0:
                a = best_a.copy()
            else:
                break
            lp_count += 1
        except Exception:
            break
    
    return best_a, best_sc, lp_count


def _cosine_init(n):
    """Initialize with a cosine-bell shape."""
    j = np.arange(n)
    a = 1.0 + np.cos(2 * np.pi * j / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _bimodal_init(n, separation=0.3):

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rflt(a, N) if False else np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gaussian_active_gradient(a, c, temperature):
    """Compute gradient using Gaussian-weighted active set.
    Weight on position k proportional to exp(-(M - c[k])/temperature).
    """
    n = len(a)
    M = c.max()
    weights = np.exp(-(M - c) / temperature)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p = weights / w_sum
    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g


def _active_gradient(a, c, k_indices):
    """Compute gradient using hard active set."""
    n = len(a)
    p = np.zeros(2 * n - 1)

    p_rev = p[::-1]
    conv_full = _fft_convolve(a, p_rev)
    g = 4.0 * n * conv_full[n - 1:2 * n - 1][::-1]
    return g

def _lp_step_reduced(a, active_fraction=0.5, time_limit=8):
    """LP step using only a fraction of the constraints (those near the max)."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = c.max()
    # Use only constraints where c[k] >= M * (1 - active_fraction)
    threshold = M * (1.0 - active_fraction)
    active_k = np.where(c >= threshold)[0]

    # Build sparse matrix for conv(a, g) constraints on active positions
    rows_list = []
    cols_list = []
    vals_list = []
    for k in active_k:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), len(rows_list) if not rows_list else sum(len(r) for r in rows_list) + len(j_range), dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
        vals_list.append(a[j_range])

    if not rows_list:
        return a, M

    # Rebuild properly
    rows_all = []
    cols_all = []
    vals_all = []
    row_offset = 0
    for k in active_k:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_all.extend([row_offset + i for i in range(len(j_range))])
        cols_all.extend(j_range)
        vals_all.extend(a[j_range])
        row_offset += len(j_range)

    if not rows_all:
        return a, M

    A = sp.csc_matrix((np.array(vals_all, dtype=np.float64),
                       (np.array(rows_all, dtype=np.int32),
                        np.array(cols_all, dtype=np.int32))),
                      shape=(row_offset, n))
    c_obj = -np.ones(n)
    b_ub = np.full(row_offset, M)
    bounds = [(0, None)] * n

    try:

                      method='highs', options={'time_limit': time_limit})
        if res.status != 0:
            return a, M
        g = np.maximum(res.x, 0.0)
        if g.sum() < 1e-15:
            return a, M
        g = g / g.sum()

        if g_norm < 1e-12:
            break

        current_sc = _score_norm(a)
        step = lr
        accepted = False
        for _ in range(15):
            a_new = _project_simplex(a - step * g_cent)
            sc_new = _score_norm(a_new)
            if sc_new < current_sc * 0.9999:
                a = a_new
                accepted = True
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a.copy()
                    patience = 0
                break
            step *= 0.5
        if not accepted:
            lr *= 0.5
            patience += 1
            if patience > 30 and lr < 1e-8:
                break
            if patience % 10 == 0:
                a = best_a.copy()
        else:
            patience = 0
            lr = min(lr * 1.02, init_lr * 2)

    return best_a, best_sc


def _sproll_refine(a, max_iter=100, damping=0.1):
    """Sproll-type iterative refinement."""
    n = len(a)
    a = a.copy()
    for it in range(max_iter):
        c = _fft_convolve(a, a)
        M = c.max()
        # Compute gradient at active positions
        k_indices = np.where(c >= M * 0.99)[0]
        if len(k_indices) == 0:
            k_indices = np.argsort(c)[::-1][:5]
        g = _active_gradient(a, c, k_indices)
        g_mean = g.mean()
        g_cent = g - g_mean
        step = M / (np.linalg.norm(g_cent) ** 2 + 1e-15) * damping
        a_new = a - step * g_cent
        a_new = _project_simplex(a_new)
        c_new = _fft_convolve(a_new, a_new)
        if c_new.max() < M:
            a = a_new
        else:
            step *= 0.1
            a_new = _project_simplex(a - step * g_cent)
            c_new = _fft_convolve(a_new, a_new)
            if c_new.max() < M:
                a = a_new
            else:
                break
    return a


def _interp_to_length(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: LP -> Nesterov z-space -> Adam z-space ->
    Line search -> Sproll -> multi-scale -> final polish.
    Uses Gaussian-weighted active sets for smoother gradients.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent or create initial
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.sum() > 0.01:
            a = parent / parent.sum()
            n = len(a)
        else:
            raise ValueError("sum too small")
    except Exception:
        n = 4096
        a = np.ones(n) / n

    best_a = a.copy()
    best_score = _score(a)
    print(f"Initial: n={n}, score={best_score:.10f}")

    # Phase 1: LP refinement with reduced constraints (0-250s)
    phase1_end = min(deadline, t_start + 250.0)
    if time.monotonic() < phase1_end:
        print("Phase 1: LP refinement...")
        a_lp, sc_lp, nlp = _lp_phase(a, phase1_end, active_fraction=0.3,
                                      time_limit=8, max_iter=60)
        full_sc = _score(a_lp)
        print(f"  LP: {nlp} iters, score={full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_lp.copy()
        a = best_a.copy()

    # Phase 2: Nesterov z-space with Gaussian weights (250-550s)
    phase2_end = min(deadline, t_start + 550.0)
    if time.monotonic() < phase2_end:
        print("Phase 2: Nesterov z-space...")
        c_cur = _fft_convolve(a, a)
        max_c = float(c_cur.max())
        a_nest, sc_nest = _nesterov_z_space(a, phase2_end - 5,
                                             temp_start=0.01 * max_c,
                                             temp_end=0.0005 * max_c,
                                             lr_start=0.03, lr_end=0.003,
                                             max_iter=300000,
                                             eval_interval=100,
                                             active_k=15,
                                             use_gaussian=True,
                                             gamma=0.85)
        full_sc = _score(a_nest)
        print(f"  Score: {full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_nest.copy()

    # Phase 3: Adam z-space, Gaussian weights, wider temperature (550-850s)
    phase3_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase3_end:
        print("Phase 3: Adam z-space (Gaussian)...")
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_adam, sc_adam = _adam_z_space(best_a, phase3_end - 5,
                                         temp_start=0.005 * max_c,
                                         temp_end=0.0002 * max_c,
                                         lr_start=0.04, lr_end=0.003,
                                         max_iter=300000,
                                         eval_interval=100,
                                         active_k=20,
                                         use_gaussian=True)
        full_sc = _score(a_adam)
        print(f"  Score: {full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_adam.copy()

    # Phase 4: Adam z-space, hard active set, low temperature (850-1100s)
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end:
        print("Phase 4: Adam z-space (hard active)...")
        c_cur = _fft_convolve(best_a, best_a)
        max_c = float(c_cur.max())
        a_hard, sc_hard = _adam_z_space(best_a, phase4_end - 5,
                                         temp_start=0.001 * max_c,
                                         temp_end=0.00005 * max_c,
                                         lr_start=0.02, lr_end=0.002,
                                         max_iter=250000,
                                         eval_interval=100,
                                         active_k=8,
                                         use_gaussian=False)
        full_sc = _score(a_hard)
        print(f"  Score: {full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_hard.copy()

    # Phase 5: Line search refinement (1100-1300s)
    phase5_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase5_end:
        print("Phase 5: Line search...")
        a_ls, sc_ls = _line_search_refine(best_a, phase5_end - 5,
                                           active_k=15, max_iter=20000,
                                           init_lr=0.003)
        full_sc = _score(a_ls)
        print(f"  Score: {full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_ls.copy()

    # Phase 6: Sproll + Adam follow-up (1300-1450s)
    phase6_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase6_end:
        print("Phase 6: Sproll...")
        a_spr = _sproll_refine(best_a, max_iter=50, damping=0.05)
        full_sc = _score(a_spr)
        print(f"  Sproll: {full_sc:.10f}")
        if full_sc < best_score:
            best_score = full_sc
            best_a = a_spr.copy()
        # Follow up with short Adam
        if time.monotonic() < phase6_end - 30:
            a_spr2, sc_spr2 = _adam_z_space(best_a, phase6_end - 5,
                                             temp_start=0.0005 * _fft_convolve(best_a, best_a).max(),
                                             temp_end=0.00002 * _fft_convolve(best_a, best_a).max(),
                                             lr_start=0.015, lr_end=0.001,
                                             max_iter=80000,
                                             eval_interval=100,
                                             active_k=10,
                                             use_gaussian=True)
            full_sc2 = _score(a_spr2)
            print(f"  Post-Sproll Adam: {full_sc2:.10f}")
            if full_sc2 < best_score:
                best_score = full_sc2
                best_a = a_spr2.copy()

    # Phase 7: Multi-scale exploration (1450-1650s)
    phase7_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase7_end:
        print("Phase 7: Multi-scale...")
        n_best = len(best_a)
        # Try n*2
        n_large = min(n_best * 2, 8192)
        if n_large != n_best and time.monotonic() < phase7_end - 80:
            a_large = _interp_to_length(best_a, n_large)
            max_cl = _fft_convolve(a_large, a_large).max()
            a_large_opt, _ = _adam_z_space(a_large, phase7_end - 40,
                                            temp_start=0.002 * max_cl,
                                            temp_end=0.00005 * max_cl,
                                            lr_start=0.03, lr_end=0.002,
                                            max_iter=100000,
                                            eval_interval=200,
                                            active_k=15,
                                            use_gaussian=True)
            full_sc = _score(a_large_opt)
            print(f"  n={n_large}: {full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_large_opt.copy()
            # Interp back
            if full_sc < 1.6 and time.monotonic() < phase7_end - 40:
                a_back = _interp_to_length(a_large_opt, n_best)
                max_cb = _fft_convolve(a_back, a_back).max()
                a_back_opt, _ = _adam_z_space(a_back, phase7_end - 10,
                                                temp_start=0.001 * max_cb,
                                                temp_end=0.00002 * max_cb,
                                                lr_start=0.02, lr_end=0.001,
                                                max_iter=60000,
                                                eval_interval=100,
                                                active_k=10,
                                                use_gaussian=True)
                full_sc_b = _score(a_back_opt)
                print(f"  interp back n={n_best}: {full_sc_b:.10f}")
                if full_sc_b < best_score:
                    best_score = full_sc_b
                    best_a = a_back_opt.copy()

        # Perturbation restarts
        for attempt in range(3):
            if time.monotonic() > phase7_end - 30:
                break
            sigma = 0.01 + 0.008 * attempt
            z = np.log(np.maximum(best_a, 1e-15))
            z = z + sigma * rng.standard_normal(len(best_a))
            z = z - z.mean()
            a_pert = _softmax_stable(z)
            max_cp = _fft_convolve(a_pert, a_pert).max()
            a_pert_opt, _ = _adam_z_space(a_pert, phase7_end - 10,
                                           temp_start=0.001 * max_cp,
                                           temp_end=0.00003 * max_cp,
                                           lr_start=0.02, lr_end=0.001,
                                           max_iter=50000,
                                           eval_interval=100,
                                           active_k=10,
                                           use_gaussian=True)
            full_sc = _score(a_pert_opt)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: {full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_pert_opt.copy()

    # Phase 8: Final ultra-fine polish (1650s-deadline)
    if time.monotonic() < deadline - 10:
        print("Phase 8: Final polish...")
        n_fin = len(best_a)
        c_fin = _fft_convolve(best_a, best_a)
        max_cf = float(c_fin.max())
        # Try both Gaussian and hard active set
        for (ug, ak) in [(True, 20), (False, 5), (True, 50)]:
            if time.monotonic() > deadline - 15:
                break
            a_pol, _ = _adam_z_space(best_a, deadline - 10,
                                      temp_start=0.0001 * max_cf,
                                      temp_end=0.000005 * max_cf,
                                      lr_start=0.003, lr_end=0.0002,
                                      max_iter=80000,
                                      eval_interval=80,
                                      active_k=ak,
                                      use_gaussian=ug)
            full_sc = _score(a_pol)
            print(f"  Polish (gauss={ug}, k={ak}): {full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_a = a_pol.copy()
            max_cf = _fft_convolve(best_a, best_a).max()

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
