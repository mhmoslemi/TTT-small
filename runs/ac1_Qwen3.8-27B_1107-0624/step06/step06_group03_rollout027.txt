def _next_pow2(n):
    k = 1

def _logsumexp_smoothed_max(c, tau):
    """Compute log(sum(exp(c/tau))) * tau ≈ max(c) + tau*log(len(c)).
    Returns (value, weights) where weights are the softmax probabilities."""
    c_scaled = c / tau
    c_max = float(c_scaled.max())
    exp_vals = np.exp(c_scaled - c_max)
    log_sum = c_max + np.log(float(exp_vals.sum()))
    value = tau * log_sum  # ≈ max(c) + tau * log(n_c)
    w = exp_vals / float(exp_vals.sum())
    return value, w


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    N = _next_pow2(2 * n)

    for tau in tau_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            _, w = _logsumexp_smoothed_max(c, tau)
            
            # Gradient in z-space: dF/dz_i = a_i * (dF/da_i - sum_j a_j dF/da_j)
            # dF/da_i ≈ 2 * cross_corr(w, a)[i]
            fa = np.fft.rfft(a, N)
            fw = np.fft.rfft(w, N)
            cross = np.fft.irfft(np.conj(fa) * fw, N)[:n]
            grad_a = 2.0 * cross
            
            mean_grad = float(np.dot(grad_a, a))
            grad_z = a * (grad_a - mean_grad)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score
