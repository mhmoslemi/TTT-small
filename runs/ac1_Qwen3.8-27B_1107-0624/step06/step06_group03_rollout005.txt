"""
Optimized step function search using:
- Power-mean (full softmax) smoothing with adaptive temperature
- Natural gradient (Fisher metric) in log-space
- Exact coordinate descent with critical point evaluation
- Multi-phase coarse-to-fine strategy
- Newton-style backtracking line search

import numpy as np
import time
from scipy.optimize import minimize_scalar

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_weights(c, beta):
    """Full softmax weights over all elements of c."""
    c_max = float(c.max())
    exp_vals = np.exp((c - c_max) / beta)
    s = float(exp_vals.sum())
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return exp_vals / s

def _power_mean_weights(c, p):
    """Power-mean weights: w_j = c_j^p / sum(c_k^p). Equivalent to softmax with beta=1/p."""
    # Use log-space for numerical stability
    c_pos = np.maximum(c, 1e-300)
    log_c = np.log(c_pos)
    p_log_c = p * log_c
    p_log_c -= p_log_c.max()
    exp_vals = np.exp(p_log_c)
    s = float(exp_vals.sum())
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return exp_vals / s

def _compute_grad_conv(a, w):
    """Gradient of <w, conv(a,a)> wrt a. = 2 * cross_corr(a, w)"""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h

def _logspace_grad(a, w, n):
    """Gradient in log-space for normalized problem.
    a is on simplex (sum=1).
    dF/dz_i = 2n * a_i * (grad_i - <grad, a>)
    """
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)

def _natural_logspace_grad(a, w, n):
    """Natural gradient in log-space: divide by a_i (Fisher metric).
    This gives: dF/dz_i / a_i = 2n * (grad_i - <grad, a>)
    """
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * (grad_a - mean_grad)

def _fista_power_mean(a_init, deadline, p_schedule, max_iter_per_phase):
    """FISTA with power-mean smoothing."""
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        a_curr, a_prev = a_curr.copy(), a_curr.copy()
        t_val = 1.0
        
        c_curr = _full_convolve(a_curr, a_curr)
        max_c = float(c_curr.max())
        # Lipschitz constant estimate
        L = 4.0 * n * p * max_c / n  # rough estimate
        step = 1.0 / L
        
        for it in range(max_iter_per_phase):
            if time.monotonic() > deadline:
                break
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            w = _power_mean_weights(c_y, p)
            grad = _compute_grad_conv(y, w)
            
            a_next = _proj_simplex(y - step * grad)
            
            if it % 500 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _softmax_normalize(z):

    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

            # Direction: d_j = a_{j-i} for valid j

            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            candidates = [0.0, a_i, a_i * 0.5, a_i * 1.5]
            
            for j in idx_top:
                if d[j] > 1e-15:

                    if t_star > 0 and t_star < 5.0 * max(a_i, 0.01):
                        candidates.append(t_star)

            best_t = a_i
            best_f = _score_norm(a)  # current score

                    best_t = tc

                a = a / a.sum()
                n_improved += 1

        if sc < best_score:

def _proj_simplex_fast(v):
    """Faster simplex projection using the sorting method."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho_max = np.where(cond)[0][-1]
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgradient_step(a, c, step_size):
    """Take a subgradient step: uniform weight on all maximizers."""
    n = len(a)
    max_c = float(c.max())
    # Find all indices within epsilon of max
    eps = 1e-10 * max_c + 1e-15
    mask = c >= max_c - eps
    n_active = mask.sum()
    
    if n_active == 0:
        mask[int(np.argmax(c))] = True
        n_active = 1
    
    w = np.zeros(2 * n - 1)
    w[mask] = 1.0 / n_active
    
    grad = _compute_grad_conv(a, w)
    a_new = _proj_simplex_fast(a - step_size * grad)
    return a_new

def _backtracking_fista(a_init, deadline, max_iter):
    """FISTA with exact subgradient and backtracking line search."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    step = 0.01  # initial step size
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        
        # Subgradient step with backtracking
        step_curr = step
        a_next = y.copy()
        for bt in range(30):
            a_next = _subgradient_step(y, c_y, step_curr)
            c_next = _full_convolve(a_next, a_next)
            # Check Armijo condition (approximate)
            # For simplicity, just accept if it doesn't increase too much
            if c_next.max() <= c_y.max() * (1 + 0.1):
                break
            step_curr *= 0.5
        
        step = min(step * 1.5, 0.1)  # grow step size
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _structured_init(n, kind="cos_sum"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 8]:
            a += 0.2 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2) + 0.05
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.5) ** 2) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

def _coord_descent_v2(a, n_passes, top_k, deadline):
    n = len(a)
    a = a.copy()
    c = _full_convolve(a, a)
    best_score = _score_norm(a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        for i in range(n):
            a_i = a[i]
            if a_i < 1e-12:
                continue
            
            # Direction
            d = np.zeros(2*n-1)
            end = min(i + n, 2*n - 1)
            d[i:end] = a[:end-i]  # a[0], a[1], ..., a[end-i-1]
            
            # Base conv (with a_i = 0)
            c_base = c - a_i * d
            S0 = 1.0 - a_i
            
            # Find top-K
            top_idx = np.argpartition(-c_base, top_k)[:top_k]
            top_c = c_base[top_idx]
            top_d = d[top_idx]
            
            # Critical points from top-K
            candidates = [0.0, a_i, a_i * 0.5, a_i * 1.5, a_i * 0.2, a_i * 2.0, a_i * 3.0]
            valid = top_d > 1e-15
            if np.any(valid):
                t_stars = S0 - 2.0 * top_c[valid] / top_d[valid]
                t_stars = t_stars[(t_stars > 0) & (t_stars < 5 * max(a_i, 0.01))]
                candidates.extend(t_stars.tolist())
            
            best_t = a_i
            best_f = np.inf
            
            for tc in candidates:
                tc = max(0.0, tc)
                # Evaluate max at candidate t
                vals = top_c + top_d * tc
                max_val = float(vals.max())
                # Also check the global max position
                gm = int(np.argmax(c))
                val_gm = c_base[gm] + d[gm] * tc
                max_val = max(max_val, val_gm)
                
                f_val = 2.0 * n * max_val / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if best_t != a_i:
                a[i] = best_t
                s = a.sum()
                a = a / s
                c = _full_convolve(a, a)
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    return a, best_score
