"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses ramp-based initialization, L-BFGS with adaptive smoothing,
Nesterov-accelerated subgradient, and coordinate descent polishing.
Key insight: non-symmetric sequences (front-loaded) can achieve scores ~1.33.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _grad_max_conv(a, n, tol_frac=0.001):
    """Compute subgradient of max(conv(a,a)) w.r.t. a (not log-param)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    # h[j] = sum_k w[k] * a[k-j] (for valid indices)
    # grad of max(conv) w.r.t. a[j] = 2 * h[j]
    # But we need to be careful: the gradient of sum_k w[k]*(a*a)[k] w.r.t. a[j]
    # = sum_k w[k] * 2*a[k-j] = 2*h[j]
    return 2.0 * h

def _project_simplex(a):
    """Project onto {x: sum(x)=1, x>=0}."""
    n = len(a)
    if a.sum() < 1e-15:
        return np.ones(n) / n
    # Simple projection: clip and renormalize
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def _nesterov_subgrad(a_init, deadline, n_max_steps=20000, momentum_init=0.5):
    """Nesterov-accelerated subgradient method for minimizing max(conv(a,a)) on simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    v = np.zeros(n)  # momentum
    momentum = momentum_init
    lr = 0.01
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov lookahead
        a_nest = a - momentum * v
        
        # Compute subgradient
        grad = _grad_max_conv(a_nest, n, tol_frac=0.002)
        
        # Update
        v = momentum * v - lr * grad
        a_new = a - v
        a_new = _project_simplex(a_new)
        
        # Check improvement
        if i % 100 == 0:
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_new.copy()
                momentum = min(momentum * 1.01, 0.9)
            else:
                momentum = max(momentum * 0.99, 0.1)
        
        a = a_new
        
        # Decay learning rate
        lr = 0.01 / (1.0 + 0.001 * i)
    
    a_final = _project_simplex(a)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _make_ramp(n):
    """Linear ramp, front-loaded: a[i] proportional to (n-1-i)."""
    a = np.arange(n - 1, -1, -1, dtype=np.float64)  # [n-1, n-2, ..., 0]
    a = a + 1  # [n, n-1, ..., 1] to avoid zeros
    return a / a.sum()

def _make_exponential(n, rate=2.0):
    """Exponential decay: a[i] proportional to exp(-rate*i/n)."""
    t = np.arange(n) / n
    a = np.exp(-rate * t)
    return a / a.sum()

def _make_bimodal(n):
    """High at front, low in middle, medium at back."""
    t = np.arange(n) / n
    a = np.exp(-4.0 * t) + 0.3 * np.exp(-4.0 * (1 - t))
    return a / a.sum()

def _make_quadratic(n):
    """Quadratic front-load: a[i] proportional to (n-1-i)^2."""
    a = np.arange(n - 1, -1, -1, dtype=np.float64)
    a = a ** 2 + 1
    return a / a.sum()

def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,
                                   0.003, 0.001, 0.0005, 0.0002, 0.0001])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(10000, max(50, int(remaining / (est_time_per_iter * 8))))

        try:
            result = minimize(
                lambda z_local: _smoobj(z_local, beta, n),
                z,
                jac=lambda z_local: _smograd(z_local, beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a

def _adam_subgrad(a_init, deadline, n_max_steps=50000):

    best_a = _softmax(z).copy()

    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.05
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        grad = _smograd(z, 0.001, n)  # use small beta for near-exact gradient
        
        t = i + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()
        
        if i % 200 == 0:
            a_check = _softmax(z)

"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: non-symmetric front-loaded (ramp) sequences achieve scores ~1.33,
far below the symmetric uniform bound of 2. Uses L-BFGS with adaptive log-sum-exp
smoothing, Adam-accelerated subgradient, Nesterov momentum, and diverse
initializations (ramp, exponential, quadratic). Multi-n exploration and
active-set refinement for final polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_active(a, n, tol_frac=0.002):
    """Subgradient of max(conv(a,a)) w.r.t. a using active set approximation."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _project_simplex(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _make_ramp(n):
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _make_exponential(n, rate=2.0):
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(-rate * t)
    return a / a.sum()


def _make_quadratic(n):
    a = np.arange(n, 0, -1, dtype=np.float64) ** 2
    return a / a.sum()


def _make_cubic(n):
    a = np.arange(n, 0, -1, dtype=np.float64) ** 3
    return a / a.sum()


def _make_power(n, p=1.5):
    a = np.arange(n, 0, -1, dtype=np.float64) ** p
    return a / a.sum()


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """L-BFGS with adaptive beta-continuation on log-softmax parameterization."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.003)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007,
                                  0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time = 0.002 * max(1.0, n / 2048.0)
        max_iter = min(10000, max(50, int(remaining / est_time)))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,

            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _adam_subgrad(a_init, deadline, n_max_steps=30000):
    """Adam-optimized subgradient in log-softmax space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = _softmax(z).copy()
    best_sc = _score(best_a)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    lr = 0.03
    beta_smo = 0.002

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break

        grad = _smograd(z, beta_smo, n)

        t = i + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - z.mean()

        if i % 300 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _nesterov_subgrad(a_init, deadline, n_max_steps=20000):
    """Nesterov-accelerated projected subgradient in primal space."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    v = np.zeros(n)
    momentum = 0.3
    lr = 0.005

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break

        a_nest = np.maximum(a - momentum * v, 0)
        a_nest = a_nest / a_nest.sum()

        grad = _grad_active(a_nest, n, tol_frac=0.003)

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = _project_simplex(a_new)

        if i % 200 == 0:
            sc = _score(a_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a_new.copy()
                momentum = min(momentum + 0.005, 0.6)
            else:
                momentum = max(momentum - 0.01, 0.1)

        a = a_new
        lr = 0.005 / (1.0 + 0.0005 * i)

    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc


def _shrinkage_refine(a_init, deadline):
    """Try zeroing smallest elements and re-normalizing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    sorted_idx = np.argsort(a_init)

    for frac in [0.01, 0.02, 0.05, 0.10, 0.15, 0.20, 0.30]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a_init.copy()
        a_try[sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer leveraging non-symmetric (front-loaded) initializations.
    Combines L-BFGS with log-sum-exp smoothing, Adam subgradient, Nesterov momentum,
    and shrinkage to find low-score sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
                print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
    except Exception:
        pass

    if parent is not None:
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = _make_ramp(n_parent)
        print(f"No parent, using ramp n={n_parent}")
        update_best(parent, _score(parent), "ramp_init")

    n_main = max(n_parent, 2048)

    # --- Phase 1: Rapid L-BFGS from ramp at several n values ---
    print(f"\n[Phase 1] Rapid ramp optimization (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [n_main, 2048, 4096]:
        if time.monotonic() > deadline - 800:
            break
        a_ramp = _make_ramp(n_try)
        sc_ramp = _score(a_ramp)
        update_best(a_ramp, sc_ramp, f"ramp_n{n_try}")
        a_opt = _run_lbfgs(a_ramp, deadline, min_time_remaining=500.0)
        update_best(a_opt, _score(a_opt), f"lbfgs_ramp_n{n_try}")

    # --- Phase 2: L-BFGS from other structured inits ---
    print(f"\n[Phase 2] Structured inits (t={time.monotonic() - t_start:.1f}s)")
    inits = [
        ("exp15", _make_exponential(n_main, 1.5)),
        ("exp3", _make_exponential(n_main, 3.0)),
        ("quad", _make_quadratic(n_main)),
        ("cubic", _make_cubic(n_main)),
        ("pow15", _make_power(n_main, 1.5)),
        ("pow3", _make_power(n_main, 3.0)),
    ]
    for name, a_init in inits:
        if time.monotonic() > deadline - 400:
            break
        update_best(a_init, _score(a_init), f"init_{name}")
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=300.0)
        update_best(a_opt, _score(a_opt), f"lbfgs_{name}")

    # --- Phase 3: L-BFGS from parent (if different from ramp) ---
    print(f"\n[Phase 3] From parent (t={time.monotonic() - t_start:.1f}s)")
    if parent is not None and time.monotonic() < deadline - 350:
        a_popt = _run_lbfgs(parent, deadline, min_time_remaining=250.0)
        update_best(a_popt, _score(a_popt), "lbfgs_parent")

    # --- Phase 4: Adam subgradient refinement on best ---
    print(f"\n[Phase 4] Adam subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_adam, sc_adam = _adam_subgrad(best_seq, min(deadline - 250, time.monotonic() + 250),
                                        n_max_steps=20000)
        update_best(a_adam, sc_adam, "adam")

    # --- Phase 5: Nesterov subgradient ---
    print(f"\n[Phase 5] Nesterov (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_nest, sc_nest = _nesterov_subgrad(best_seq, min(deadline - 150, time.monotonic() + 150),
                                            n_max_steps=15000)
        update_best(a_nest, sc_nest, "nesterov")

    # --- Phase 6: Shrinkage ---
    print(f"\n[Phase 6] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_shr, sc_shr = _shrinkage_refine(best_seq, deadline - 100)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=60.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # --- Phase 7: Multi-n with ramp ---
    print(f"\n[Phase 7] Multi-n ramp (t={time.monotonic() - t_start:.1f}s)")
    for n_try in [1024, 8192, 16384]:
        if time.monotonic() > deadline - 60:
            break
        a_try = _make_ramp(n_try)
        a_try_opt = _run_lbfgs(a_try, deadline, min_time_remaining=40.0)
        update_best(a_try_opt, _score(a_try_opt), f"ramp_n{n_try}_opt")

    # --- Phase 8: Final fine L-BFGS ---
    print(f"\n[Phase 8] Final refinement (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        beta_final = max(mb * 0.00005, 1e-10)
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(5000, max(100, int(remaining / 0.01)))

            def fun_final(z_local):
                return _smoobj(z_local, beta_final, n)

            def jac_final(z_local):
                return _smograd(z_local, beta_final, n)

            try:
                result = minimize(fun_final, z, jac=jac_final, method='L-BFGS-B',
                                  options={'maxiter': max_iter, 'ftol': 1e-16,
                                           'gtol': 1e-14, 'maxcor': 50})
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Adam polish
        if time.monotonic() < deadline - 3:
            a_pol, sc_pol = _adam_subgrad(best_seq, deadline - 3, n_max_steps=5000)
            update_best(a_pol, sc_pol, "adam_polish")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_ramp(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_ramp(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
