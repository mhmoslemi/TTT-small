"""
Multi-scale optimizer with asymmetric initialization for minimizing
2*n*max(conv(a,a))/sum(a)^2. Key insight: asymmetric (ramp-like) sequences
have J~4/3 vs J=2 for uniform, providing a much better starting point.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_n = a / s
    c = _full_convolve(a_n, a_n)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter, step_scale=1.0):
    n = len(a_init)
    a_curr = a_init.copy().astype(np.float64)
    s0 = a_curr.sum()
    if s0 > 0:
        a_curr = a_curr / s0
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = step_scale / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 100 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter, momentum_b1=0.9, momentum_b2=0.999):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = momentum_b1, momentum_b2, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 50 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 6000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 40)), 'gtol': 1e-12, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_ramp(n, power=1.0):
    """Create a monotonically increasing ramp profile."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** power
    return a / a.sum()


def _make_cosine_tilt(n):
    """Cosine-tilted profile: slightly asymmetric."""
    x = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(np.pi * x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_quadratic_ramp(n):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** 2
    return a / a.sum()


def _make_power_ramp(n, p):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def _upsample(a, new_n):
    """Upsample a sequence using linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 1e-15)
    return result / result.sum()


def _downsample(a, new_n):
    """Downsample a sequence."""
    old_n = len(a)
    if new_n >= old_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 1e-15)
    return result / result.sum()


def _optimize_at_scale(a, deadline, n_phases=8):
    """Run FISTA with decreasing beta at a given scale."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    c = _full_convolve(a, a)
    max_c = float(c.max())
    if max_c < 1e-15:
        max_c = 1.0 / n
    beta = 0.05 * max_c
    best_score = _score_norm(a)
    best_a = a.copy()

    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 1:
            break
        per_time = remaining / (n_phases - phase)
        max_it = max(500, int(per_time * 5000))
        a, _ = _fista_opt(a, beta, time.monotonic() + per_time * 0.8, max_it)
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        c = _full_convolve(a, a)
        max_c = float(c.max())
        beta = 0.3 * max_c
        if beta < 1e-12:
            beta = 1e-12

    return best_a, best_score


def _fine_optimize(a, deadline):
    """Fine optimization: Adam with decreasing beta, then L-BFGS."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    c = _full_convolve(a, a)
    max_c = float(c.max())
    best_score = _score_norm(a)
    best_a = a.copy()

    # Adam phase with multiple betas
    betas = [0.001 * max_c, 0.0003 * max_c, 0.0001 * max_c, 0.00003 * max_c, 0.00001 * max_c]
    per_beta = (deadline - time.monotonic() - 30) / (len(betas) + 1)
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 30:
            break
        per_time = min(per_beta, deadline - time.monotonic() - 25)
        if per_time < 3:
            break
        a, _ = _adam_opt(a, time.monotonic() + per_time, beta, lr_init=0.005, max_iter=500000)
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # L-BFGS phase
    if time.monotonic() < deadline - 20:
        c = _full_convolve(best_a, best_a)
        max_c = float(c.max())
        beta_lb = 0.00001 * max_c
        best_a, _ = _lbfgs_opt(best_a, deadline - 5, beta_lb)
        sc = _score_norm(best_a)
        if sc < best_score:
            best_score = sc

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    n_target = 4096

    # Try loading parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Loaded parent: n={len(parent)}, score={_score(parent):.10f}")

    # === Multi-scale optimization from asymmetric initializations ===
    # Key insight: ramp sequences give J~1.33 vs J=2 for uniform
    
    print("=== Multi-scale optimization ===")
    scales = [128, 512, 2048, n_target]
    
    # Initial profiles to try at coarsest scale
    init_profiles = {}
    init_profiles['ramp1'] = _make_ramp(scales[0], 1.0)
    init_profiles['ramp2'] = _make_ramp(scales[0], 2.0)
    init_profiles['ramp05'] = _make_ramp(scales[0], 0.5)
    init_profiles['ramp15'] = _make_ramp(scales[0], 1.5)
    init_profiles['cosine'] = _make_cosine_tilt(scales[0])
    
    # Also try downsampled parent if available
    if parent is not None:
        init_profiles['parent'] = _downsample(parent, scales[0])
    
    # Optimize each profile through all scales
    best_coarse = None
    best_coarse_name = ''
    
    time_per_profile = min(45.0, (deadline - t_start - 600) / len(init_profiles))
    
    for name, a0 in init_profiles.items():
        if time.monotonic() > deadline - 400:
            break
        a_work = a0.copy()
        
        # Multi-scale: optimize at each scale
        for si, n_scale in enumerate(scales):
            target_t = t_start + time_per_profile * (si + 1)
            if time.monotonic() > target_t:
                a_work = _upsample(a_work, n_scale)
                continue
            if n_scale > len(a_work):
                a_work = _upsample(a_work, n_scale)
            
            # Optimize at this scale
            a_work, sc = _optimize_at_scale(a_work, min(target_t, deadline - 300), n_phases=4)
        
        sc_final = _score_norm(a_work)
        print(f"  {name}: coarse score={sc_final:.10f} (n={len(a_work)})")
        
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_work.copy()
            best_coarse = a_work.copy()
            best_coarse_name = name

    print(f"Best coarse: {best_coarse_name} score={best_score:.10f}")
    
    # === Fine optimization at target scale ===
    if best_coarse is not None:
        print("=== Fine optimization ===")
        a_fine = _upsample(best_coarse, n_target)
        
        # Phase 1: FISTA with aggressive beta reduction
        phase1_end = min(deadline - 300, t_start + 400)
        if time.monotonic() < phase1_end - 5:
            a_fine, sc = _optimize_at_scale(a_fine, phase1_end, n_phases=10)
            sc2 = _score_norm(a_fine)
            if sc2 < best_score:
                best_score = sc2
                best_seq = a_fine.copy()
                print(f"  FISTA phase1: score={sc2:.10f} t={time.monotonic()-t_start:.1f}s")
        
        # Phase 2: Adam fine-tuning
        phase2_end = min(deadline - 200, t_start + 900)
        if time.monotonic() < phase2_end - 10:
            c = _full_convolve(a_fine, a_fine)
            max_c = float(c.max())
            betas_fine = [0.0005 * max_c, 0.00015 * max_c, 0.00005 * max_c, 0.00002 * max_c, 0.000005 * max_c]
            per_beta = (phase2_end - time.monotonic() - 10) / len(betas_fine)
            for beta in betas_fine:
                if time.monotonic() > phase2_end - 10:
                    break
                per_time = min(per_beta, phase2_end - time.monotonic() - 8)
                if per_time < 5:
                    break
                a_fine, _ = _adam_opt(a_fine, time.monotonic() + per_time, beta, lr_init=0.004, max_iter=300000)
                a_fine = a_fine / a_fine.sum()
                sc = _score_norm(a_fine)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_fine.copy()
                    print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        
        # Phase 3: L-BFGS
        if time.monotonic() < deadline - 50:
            c = _full_convolve(a_fine, a_fine)
            max_c = float(c.max())
            beta_lb = 0.00001 * max_c
            a_fine, _ = _lbfgs_opt(a_fine, min(deadline - 30, t_start + 1300), beta_lb)
            sc = _score_norm(a_fine)
            if sc < best_score:
                best_score = sc
                best_seq = a_fine.copy()
                print(f"  L-BFGS: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Multi-restart exploration ===
    print("=== Multi-restart exploration ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n_restarts = 5
        per_r = (deadline - time.monotonic() - 80) / n_restarts
        restart_profiles = []
        
        # Diverse restarts
        restart_profiles.append(_make_ramp(n_target, 1.2))
        restart_profiles.append(_make_ramp(n_target, 0.8))
        restart_profiles.append(_make_ramp(n_target, 1.5))
        restart_profiles.append(_make_power_ramp(n_target, 0.7))
        restart_profiles.append(_make_power_ramp(n_target, 2.5))
        
        # Also try perturbed best
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z + 0.01 * rng.standard_normal(n_target)
        z = z - z.mean()
        restart_profiles.append(_softmax_compute(z))
        
        for i, p in enumerate(restart_profiles):
            if time.monotonic() > deadline - 80:
                break
            per_time = min(per_r, deadline - time.monotonic() - 60)
            if per_time < 20:
                break
            
            # Quick FISTA + Adam
            if i == len(restart_profiles) - 1:
                # Perturbed best: just fine-tune
                p, sc = _optimize_at_scale(p, time.monotonic() + per_time * 0.5, n_phases=4)
                c = _full_convolve(p, p)
                beta_a = 0.0001 * float(c.max())
                p, _ = _adam_opt(p, time.monotonic() + per_time * 0.5, beta_a, 0.003, 100000)
            else:
                p, sc = _optimize_at_scale(p, time.monotonic() + per_time * 0.6, n_phases=5)
                c = _full_convolve(p, p)
                beta_a = 0.0002 * float(c.max())
                p, _ = _adam_opt(p, time.monotonic() + per_time * 0.4, beta_a, 0.004, 100000)
            
            sc_final = _score_norm(p)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = p.copy()
                print(f"  Restart {i}: score={sc_final:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final polish ===
    print("=== Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c = _full_convolve(a_pol, a_pol)
        max_c = float(c.max())
        
        # Ultra-fine Adam
        per_t = deadline - time.monotonic() - 10
        if per_t > 15:
            beta_final = 0.000003 * max_c
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.5, beta_final, 0.001, 100000)
            sc = _score_norm(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc:.10f}")
            
            # L-BFGS on remainder
            if time.monotonic() < deadline - 12:
                c2 = _full_convolve(a_pol, a_pol)
                beta_f = 0.000002 * float(c2.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 5, beta_f)
                sc = _score_norm(a_pol)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_ramp(256, 1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:

            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:

    # Fallback: ramp sequence
    fallback = _make_ramp(4096, 1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
