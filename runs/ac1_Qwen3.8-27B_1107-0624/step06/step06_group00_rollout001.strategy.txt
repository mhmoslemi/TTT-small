        normalizer = initial_vector.sum()
        if normalizer > 0.00001:
            initial_vector_opt = initial_vector / normalizer
            score_opt, seq_opt = optimize_step_function(initial_vector_opt.tolist())
            if score_opt < best_score:
                best_score = score_opt
                best_sequence = np.array(seq_opt)
        
        time_spent += 5.0
        print(f"Time spent: {time_spent} seconds")
        if time_spent >= total_budget:
            break
            
    # Final evaluation to be sure
    try:
        final_score, final_seq = evaluate_sequence(best_sequence)
    except:
        final_score = np.inf
        
    if final_score < best_score:
        best_score = final_score
        best_sequence = final_seq
        
    return best_sequence.tolist()
