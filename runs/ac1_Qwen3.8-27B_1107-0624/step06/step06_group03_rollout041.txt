"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines water-filling multiplicative updates, full-softmax Adam in log-space,
and exact 1D coordinate descent polishing.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _cross_corr(a, w):
    """Compute F[i] = sum_j a[j] * w[i+j] for i=0..len(a)-1.
    a has length n, w has length 2n-1 (or any length >= n)."""
    n = len(a)
    # F[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is cross-correlation of a and w at lag i
    # Using FFT: cross_corr(a, w)[i] = ifft(conj(fft(a,N)) * fft(w,N))[i]
    N = _next_pow2(n + len(w) - 1)
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w, N)
    result = np.fft.ifft(np.conj(fa) * fw).real[:n]
    return result

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _softmax_full(c, beta):
    """Full softmax over the entire convolution vector."""
    c_shifted = c - c.max()
    e = np.exp(beta * c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return e / s

def _water_filling(a, n_iter, alpha_start, alpha_end, deadline):
    """Water-filling multiplicative update."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(1, n_iter))
        
        c = _full_convolve(a, a)
        S = 1.0 / (2 * n - 1)  # target level when sum(a)=1
        excess = np.maximum(0, c - S)
        excess_sum = excess.sum()
        
        if excess_sum < 1e-15:
            break
        
        w = excess / excess_sum
        F = _cross_corr(a, w)
        F_max = F.max()
        
        if F_max < 1e-15:
            break
        
        a = a * (1.0 - alpha * F / F_max)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, beta, lr_init, max_iter):
    """ Adam optimizer in log-space with full softmax. """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Softmax normalize
        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()
        
        c = _full_convolve(a, a)
        w = _softmax_full(c, beta)
        F = _cross_corr(a, w)  # F[i] = sum_k w[k] * a[k-i]... 
        
        # Wait, I need to double-check the gradient formula
        # f(a) = 2n * sum_k w_k * c_k (smoothed max approximation)
        # df/da_i = 2n * sum_k w_k * a[k-i] = 2n * F[i]
        # where F[i] = sum_k w_k * a[k-i]
        
        # But _cross_corr(a, w) gives F[i] = sum_j a[j] * w[i+j]
        # And I want G[i] = sum_k w[k] * a[k-i]
        # Let's check: G[i] = sum_k w[k] * a[k-i], let j = k-i, k = i+j
        # G[i] = sum_j w[i+j] * a[j] (for valid j and i+j)
        # This IS _cross_corr(a, w)[i] = sum_j a[j] * w[i+j] ✓
        
        grad_a = 2.0 * n * F  # gradient of smoothed f wrt a_i
        
        # Log-space gradient
        mean_grad = float(np.dot(a, grad_a))
        ln_grad = a * (grad_a - mean_grad)
        
        if not np.all(np.isfinite(ln_grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * ln_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (ln_grad * ln_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update / update_norm
        
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % 100 == 0:
            z_shift = z - z.max()
            e = np.exp(z_shift)
            a_chk = e / e.sum()
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                print(f"    Adam it={t}: score={sc:.10f}")
        
        if it % 5000 == 0 and it > 0:
            current_lr *= 0.95
    
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a_final = e / e.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _coord_descend(a, n_passes, deadline, top_k=50):
    """Coordinate descent with exact 1D optimization."""
    n = len(a)
    a = a.copy() / a.sum()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i_old = a[i]
            
            # d[k] = a[k-i] for valid k (0 <= k-i < n)
            # c[k] = (a*a)[k]
            # If we change a[i] to t (and renormalize), the new convolution is:
            # c_new[k] = c[k] - a_i_old * d[k] + t * d[k] = c[k] + (t - a_i_old) * d[k]
            # sum_new = 1 - a_i_old + t
            # score(t) = 2n * max_k(c[k] + (t-a_i)*d[k]) / (1 - a_i + t)^2
            
            c = _full_convolve(a, a)
            
            # d vector (length 2n-1)
            d = np.zeros(2 * n - 1)
            # d[k] = a[k-i] if 0 <= k-i < n
            valid_start = i
            valid_end = min(i + n, 2 * n - 1)
            d[valid_start:valid_end] = a[0:valid_end - i]
            
            S0 = 1.0 - a_i_old
            
            # Find candidate t values
            t_candidates = [0.0, a_i_old, a_i_old * 0.5, a_i_old * 1.5]
            
            # Critical points from top-k active constraints
            # For active k: t* = S0 + 2*c[k]/d[k] - a_i_old (from d/dt = 0)
            # Wait: score(t) = 2n*(c[k] + (t-a_i)*d[k]) / (S0+t)^2
            # d/dt = 2n*[d[k]*(S0+t) - 2*(c[k]+(t-a_i)*d[k])] / (S0+t)^3
            # = 2n*[d[k]*S0 + d[k]*t - 2*c[k] - 2*t*d[k] + 2*a_i*d[k]] / (S0+t)^3
            # = 2n*[d[k]*(S0 + a_i) - d[k]*t - 2*c[k]] / (S0+t)^3
            # = 2n*[d[k] - d[k]*t - 2*c[k]] / (S0+t)^3  (since S0 + a_i = 1)
            # Setting numerator to 0: d[k]*(1-t) = 2*c[k] => t = 1 - 2*c[k]/d[k]
            
            if a_i_old > 1e-15:
                topk_idx = np.argsort(c)[-top_k:]
                for k in topk_idx:
                    if d[k] > 1e-15:
                        t_star = 1.0 - 2.0 * c[k] / d[k]
                        if -0.1 <= t_star <= 10.0:
                            t_candidates.append(max(0.0, t_star))
                            t_candidates.append(max(0.0, t_star * 0.9))
                            t_candidates.append(max(0.0, t_star * 1.1))
            
            t_candidates = list(set([max(0.0, min(5.0, tc)) for tc in t_candidates]))
            
            best_t = a_i_old
            best_f = 2.0 * n * c.max()
            
            for tc in t_candidates:
                c_trial = c + (tc - a_i_old) * d
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if abs(best_t - a_i_old) > 1e-12:
                a[i] = best_t
                s = a.sum()
                if s > 1e-15:
                    a = a / s
        
        # Evaluate after each pass
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
    
    return a, best_score

def _structured_init(n, kind="sin_taper"):

        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_window":
        t = np.linspace(0, 1, n)

    elif kind == "raised_cos":

        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-8 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy() / p.sum()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines water-filling multiplicative updates, full-softmax Adam in log-space,
and exact 1D coordinate descent polishing. Key innovation: water-filling algorithm
that iteratively reduces contribution to convolution peaks via multiplicative updates.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _cross_corr(a, w):
    """Compute F[i] = sum_j a[j] * w[i+j] for i=0..len(a)-1.
    a has length n, w has length >= n. Uses real FFT for speed."""
    n = len(a)
    N = _next_pow2(n + len(w) - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    result = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return result


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_full(c, beta):
    """Full softmax over the entire convolution vector."""
    c_shifted = c - c.max()
    e = np.exp(beta * c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
        return w
    return e / s


def _water_filling(a, n_iter, alpha_start, alpha_end, momentum, deadline):
    """Water-filling multiplicative update with momentum.
    Iteratively reduces the contribution of each coordinate to convolution peaks."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    v_mom = np.zeros(n)  # momentum buffer
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        # Anneal alpha
        frac = it / max(1, n_iter - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        c = _full_convolve(a, a)
        S = 1.0 / (2 * n - 1)  # target level when sum(a)=1
        excess = np.maximum(0, c - S)
        excess_sum = excess.sum()
        
        if excess_sum < 1e-15:
            break
        
        w = excess / excess_sum
        F = _cross_corr(a, w)
        F_max = F.max()
        
        if F_max < 1e-15:
            break
        
        # Multiplicative update with momentum
        update = a * (1.0 - alpha * F / F_max)
        v_mom = momentum * v_mom + (1.0 - momentum) * (update - a)
        a = a + v_mom
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _water_filling_v2(a, n_iter, alpha, deadline):
    """Alternative water-filling using frequency-domain shaping.
    Multiplies the spectrum by a function designed to flatten the autocorrelation."""
    n = len(a)
    a = a.copy() / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    
    N = _next_pow2(2 * n)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        S = 1.0 / (2 * n - 1)
        excess = np.maximum(0, c - S)
        if excess.sum() < 1e-15:
            break
        
        # Compute the "error" in frequency domain
        # The autocorrelation c has FFT |A(f)|^2
        # We want to reduce the components that contribute to the excess
        A = np.fft.rfft(a, N)
        power = np.abs(A) ** 2
        power_sum = power.sum()
        if power_sum < 1e-30:
            break
        
        # Target: make power spectrum as uniform as possible (flat autocorrelation)
        # Weight the reduction by how much each frequency contributes to peaks
        # Simple approach: reduce frequencies where power is above target
        target_power = power_sum / len(power)
        excess_power = np.maximum(0, power - target_power)
        
        if excess_power.sum() < 1e-15:
            break
        
        # Scale down the excess power components
        scale = 1.0 - alpha * excess_power / excess_power.max()
        A_new = A * np.sqrt(np.maximum(scale, 0.1))
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        if it % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta, lr_init, max_iter):
    """Adam optimizer in log-space with full softmax gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Softmax normalize
        z_shift = z - z.max()
        e = np.exp(z_shift)
        a = e / e.sum()
        
        c = _full_convolve(a, a)
        w = _softmax_full(c, beta)
        F = _cross_corr(a, w)
        
        # Gradient in a-space: grad_a = 2n * F
        # Gradient in z-space: a * (grad_a - <a, grad_a>)
        grad_a = 2.0 * n * F
        mean_grad = float(np.dot(a, grad_a))
        ln_grad = a * (grad_a - mean_grad)
        
        if not np.all(np.isfinite(ln_grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * ln_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (ln_grad * ln_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 2.0:
            update = update * (2.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % 200 == 0:
            z_shift = z - z.max()
            e = np.exp(z_shift)
            a_chk = e / e.sum()
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                if best_score < 1.51:
                    print(f"    Adam it={t}: score={sc:.10f}")
        
        if it % 10000 == 0 and it > 0:
            current_lr *= 0.95
    
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a_final = e / e.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _coord_descend(a, n_passes, deadline, top_k=80):
    """Coordinate descent with exact 1D optimization.
    Computes full convolution once per pass for efficiency."""
    n = len(a)
    a = a.copy() / a.sum()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)  # compute once per pass
        c_max = float(c.max())
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            a_i_old = a[i]
            
            # d[k] = a[k-i] for valid k (0 <= k-i < n)
            d = np.zeros(2 * n - 1)
            valid_start = i
            valid_end = min(i + n, 2 * n - 1)
            d[valid_start:valid_end] = a[0:valid_end - i]
            
            S0 = 1.0 - a_i_old
            
            # Candidate t values
            t_candidates = [0.0, a_i_old]
            
            if a_i_old > 1e-10:
                # Critical points from top-k active constraints
                # t* = 1 - 2*c[k]/d[k] for active k
                topk_idx = np.argsort(c)[-top_k:]
                for k in topk_idx:
                    dk = d[k]
                    if dk > 1e-12:
                        t_star = 1.0 - 2.0 * c[k] / dk
                        if -0.01 <= t_star <= 5.0:
                            t_candidates.append(max(0.0, t_star))
                            t_candidates.append(max(0.0, t_star * 0.85))
                            t_candidates.append(max(0.0, t_star * 1.15))
                
                # Geometric candidates
                t_candidates.append(a_i_old * 0.5)
                t_candidates.append(a_i_old * 0.75)
                t_candidates.append(a_i_old * 1.25)
                t_candidates.append(a_i_old * 1.5)
                t_candidates.append(a_i_old * 0.25)
            
            else:
                # Try small positive values
                t_candidates.extend([1e-4, 1e-3, 5e-3, 1e-2])
            
            # Deduplicate and clip
            t_candidates = list(set([max(0.0, min(5.0, tc)) for tc in t_candidates]))
            
            best_t = a_i_old
            best_f = 2.0 * n * c_max
            
            for tc in t_candidates:
                c_trial = c + (tc - a_i_old) * d
                max_c = float(c_trial.max())
                denom = S0 + tc
                if denom < 1e-15:
                    continue
                f_val = 2.0 * n * max_c / (denom * denom)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc
            
            if abs(best_t - a_i_old) > 1e-12:
                a[i] = best_t
                s = a.sum()
                if s > 1e-15:
                    a = a / s
        
        # Check score after each pass
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
    
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
    
    return a, best_score


def _structured_init(n, kind="sin_taper"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos_window":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5)) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-6 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "optimal_shape":
        # Shape designed to approximate flat convolution
        # The ideal c[k] is flat, which corresponds to a specific a shape
        t = np.linspace(0, 1, n)
        # Use a sinc-like shape (inverse of flat spectrum)
        a = np.sinc(2 * t - 1) ** 2
        a = np.maximum(a, 0.01)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy() / p.sum()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fista_simplex(a_init, deadline, beta, max_iter, top_k=100):
    """FISTA on simplex with top-K softened max (for quick initial optimization)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        
        # Top-K softmax weights
        if top_k < len(c_y):
            topk_thresh = np.partition(c_y, -top_k)[-top_k]
            topk_mask = c_y >= topk_thresh
            topk_indices = np.where(topk_mask)[0]
        else:
            topk_indices = np.arange(len(c_y))
        
        c_topk = c_y[topk_indices]
        c_max_tk = float(c_topk.max())
        exp_vals = np.exp((c_topk - c_max_tk) / beta)
        w_sum = float(exp_vals.sum())
        
        w = np.zeros(len(c_y))
        if w_sum < 1e-300:
            w[topk_indices[0]] = 1.0
        else:
            w[topk_indices] = exp_vals / w_sum
        
        grad = 2.0 * n * _cross_corr(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

            a_opt, _ = _adam_logspace(a_wfx, t_ad, beta_mix, 0.006, max_iter=30000)
            
            sc_opt = _score(a_opt)
            print(f"  {init_name}: score={sc_opt:.10f}, time={time.monotonic()-t_start:.1f}s")
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
        
        # Shift perturbation of best
        if time.monotonic() < deadline - 30:
            shift = rng.integers(1, max(2, n // 2048))
            a_shift = np.roll(best_seq, int(shift)).copy()
            a_shift = a_shift / a_shift.sum()
            t_s = time.monotonic() + min(20, (deadline - time.monotonic()) * 0.5)
            a_shift, _ = _water_filling(a_shift, n_iter=2000, alpha_start=0.2, alpha_end=0.02,
                                         momentum=0.3, deadline=t_s)
            sc_shift = _score(a_shift)
            print(f"  Shift({shift}): score={sc_shift:.10f}")
            if sc_shift < best_score:
                best_score = sc_shift
                best_seq = a_shift.copy()
    
    # Final validation and output
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq)} ===")
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) * (0.01 / 256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    return [float(x) for x in final]
