def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _lp_update(f, M):
    """Solve the LP: max sum(b) s.t. b >= 0, conv(f, b) <= M."""
    n = len(f)
    # Variables: b_0, ..., b_{n-1}
    # Objective: max sum(b) => min -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: conv(f, b)[k] <= M for k = 0, ..., 2n-2
    # conv(f, b)[k] = sum_i f[i] * b[k-i] (valid indices)
    # This is a linear function of b with coefficients from f.
    # A_ub has 2n-1 rows, n columns.

    rows = []
    cols = []
    vals = []

        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based Frank-Wolfe (new), L-BFGS with adaptive log-sum-exp,
tight active-set subgradient, shrinkage, and multi-n search.
Key innovations: exact LP subproblem for Frank-Wolfe, tighter active set,
backtracking subgradient with tighter tolerance.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.sparse import diags as _sparse_diags


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(z, a, c, n, tol_frac=0.001):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03, tol_frac=0.0005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(z, a, c, n, tol_frac)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
                lr = min(lr * 1.03, lr_init)
            else:
                patience += 1
                lr *= 0.8
                if patience > 40:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
                    tol_frac = max(0.0001, tol_frac * 0.5)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _build_conv_matrix(n, f):
    """Build sparse matrix A where A @ b = conv(f, b). Shape: (2n-1, n)."""
    diags_vals = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))
    return _sparse_diags(diags_vals, offsets, shape=(2 * n - 1, n), format='csr')


def _frank_wolfe_lp(f_init, deadline, max_iter=300, tol=1e-10):
    """LP-based Frank-Wolfe method for min max(conv(a,a)) s.t. sum(a)=1, a>=0."""
    from scipy.optimize import linprog
    
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return f_init
    f = f / s
    best_f = f.copy()
    best_sc = _score(f)
    t_global = 1.0  # step size multiplier
    
    print(f"  [FW-LP] Starting: n={n}, score={best_sc:.10f}")
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(f, f)
        M = float(c.max())
        
        # Build constraint matrix
        A = _build_conv_matrix(n, f)
        b_ub = np.full(2 * n - 1, M)
        c_obj = -np.ones(n)
        
        t_lp_start = time.monotonic()
        try:
            result = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs',
                            options={'time_limit': 120})
        except Exception as e:
            print(f"  [FW-LP] LP error at iter {it}: {e}")
            break
        
        t_lp_end = time.monotonic()
        
        if result.status != 0:
            if it < 5:
                print(f"  [FW-LP] LP status={result.status} at iter {it}, breaking")
            break
        
        b = np.maximum(result.x, 0)
        s_b = b.sum()
        if s_b < 1e-15:
            break
        b = b / s_b
        
        # Check if b is essentially the same as f (convergence)
        if np.linalg.norm(b - f) < tol:
            print(f"  [FW-LP] Converged at iter {it} (norm diff < {tol})")
            break
        
        # Line search
        improved = False
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            t_actual = t * t_global
            f_try = (1 - t_actual) * f + t_actual * b
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-15:
                continue
            f_try = f_try / s_try
            sc_try = _score(f_try)
            if sc_try < best_sc - 1e-12:
                f = f_try
                best_f = f_try.copy()
                best_sc = sc_try
                improved = True
                t_global = min(1.0, t_global * 1.1)
                break
            elif sc_try < best_sc:
                # Accept even without strict improvement (for progress)
                f = f_try
                best_f = f_try.copy()
                best_sc = sc_try
                improved = True
                break
        
        if not improved:
            t_global *= 0.5
            if t_global < 0.01:
                print(f"  [FW-LP] Step size too small at iter {it}, breaking")
                break
        
        if it % 20 == 0:
            print(f"  [FW-LP] iter {it}: score={best_sc:.10f}, "
                  f"lp_time={t_lp_end-t_lp_start:.2f}s, t_global={t_global:.3f}")
    
    print(f"  [FW-LP] Final: score={best_sc:.10f} after {it+1} iters")
    return best_f


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0, beta_schedule=None):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted) - 1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.005)

    if beta_schedule is None:
        betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                                       0.007, 0.003, 0.001, 0.0005, 0.0002, 0.0001])
    else:
        betas = beta_schedule

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 8))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-13,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a
