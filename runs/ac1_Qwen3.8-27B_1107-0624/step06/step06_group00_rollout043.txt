import numpy as np

def _next_pow2(n):
    k=1
    while k < n:
        k <<= 1
    return max(2,k)

def _softmax(z):
    if not np.all(np.isfinite(z)):
       z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
       return np.ones_like(z) / float(len(z))
    return e / s

def _normalize(a):
    a = np.maximum(a,0.0)
    s = float(a.sum())
    if s <= 1e-15:
       return np.ones_like(a) / float(len(a))
    return a / s

def _interp_to_length(a, n_new):
    n0=len(a)
    if n0 == n_new:
       return _normalize(a.copy())
    x0=np.linspace(0.0,1.0,n0)
    x1=np.linspace(0.0,1.0,n_new)
    b=np.interp(x1,x0,a)
    b=np.maximum(b,1e-15)
    return _normalize(b)

def _self_conv(a,N):

    if N < L: N = _next_pow2(N) # but if N not pow? use next of max

    c=np.fft.irfft(fa*fa,N)
    c=c[:L]
    if c.size:
       np.maximum(c,0.0,out=c)

```python
"""
Search for nonnegative coefficient sequences minimizing
2*n*max(np.convolve(a,a))/(sum(a))^2.

Strategy:
- Start from the provided parent sequence when available.
- Multi-scale optimization: polish small lengths, then interpolate up to
  8192, 16384, and 32768.
- Main optimizer: Adam in log-softmax (z) space using carefully scaled
  subgradients from smoothed / active-top convolution coefficients.
- Final polishing: active-constraint LP step plus a low-temperature Adam pass.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if len(a) == 0:
        return a
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones_like(a, dtype=np.float64) / float(len(a))
    return a / s


def _softmax(z):
    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=50.0, neginf=-50.0)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / float(len(z))
    return e / s


def _interp_to_length(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = len(a)
    if n0 == n_new:
        return _normalize(a)
    x0 = np.linspace(0.0, 1.0, n0, dtype=np.float64)
    x1 = np.linspace(0.0, 1.0, n_new, dtype=np.float64)
    b = np.interp(x1, x0, a)
    b = np.maximum(b, 1e-15)
    return _normalize(b)


def _self_conv(a, N):
    n = len(a)
    L = 2 * n - 1
    minN = _next_pow2(L)
    if N < minN:
        N = minN
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:L]
    if c.size:
        np.maximum(c, 0.0, out=c)
    return c, fa


def _conv_max(a, N=None):
    n = len(a)
    L = 2 * n - 1
    if N is None or N < L:
        N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return float(np.max(c[:L]))


def _score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a = a / s
    return 2.0 * n * _conv_max(a, _next_pow2(2 * n - 1))


def _grad_weights(a, fa, w, N):
    n = len(a)
    fw = np.fft.rfft(w[::-1], N)
    hfull = np.fft.irfft(fa * fw, N)
    return hfull[n - 1:2 * n - 1][::-1]


def _grad_direct(a, idx, weights):
    n = len(a)
    h = np.zeros(n, dtype=np.float64)
    for k, wg in zip(idx, weights):
        if wg == 0.0:
            continue
        k = int(k)
        start = max(0, k - n + 1)
        end = min(k, n - 1) + 1
        if start >= end:
            continue
        j_start = k - (end - 1)
        j_end = k - start + 1
        h[start:end] += wg * a[j_start:j_end][::-1]
    return h


def _adam_z(a_init, deadline, rng=None,
            lr_start=0.045, lr_end=0.004,
            alpha_start=1.0, alpha_end=2.5,
            k_end=16, clip_norm=1.0, patience=800, name=""):
    if deadline <= time.monotonic():
        a0 = _normalize(a_init)
        return a0, _score(a0)

    if rng is None:
        rng = np.random.default_rng(12345)

    a_init = _normalize(a_init)
    n = len(a_init)
    L = 2 * n - 1
    Nself = _next_pow2(L)
    Ngrad = _next_pow2(3 * n - 2)

    best_score = 2.0 * n * _conv_max(a_init, Nself)
    best_a = a_init.copy()

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z = np.clip(z, -60.0, 60.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    corr1 = 0.0
    corr2 = 0.0

    w = np.empty(L, dtype=np.float64)

    t = 0
    lr_mult = 1.0
    since_best = 0
    t0 = time.monotonic()
    duration = max(1.0, deadline - t0)

    K_start = int(min(4096, max(256, L // 32)))
    direct_th = max(8, min(32, k_end))
    last_print = 0.0

    while time.monotonic() < deadline:
        t += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, max(0.0, elapsed / duration))

        if frac < 0.30:
            mode = "dense"
            K = L
        else:
            ts = (frac - 0.30) / 0.45
            if ts < 0.0:
                ts = 0.0
            elif ts > 1.0:
                ts = 1.0
            K = int(np.ceil(K_start + (k_end - K_start) * ts))
            if K < k_end:
                K = k_end
            if K > L:
                K = L

            if K > max(512, L // 8):
                mode = "dense"
            elif frac > 0.75 and K <= direct_th:
                mode = "direct"
            else:
                mode = "sparse"

        a = _softmax(z)

        if mode == "direct":
            c, _ = _self_conv(a, Nself)
        else:
            c, fa = _self_conv(a, Ngrad)

        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break

        sc = 2.0 * n * M
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            since_best = 0
        else:
            since_best += 1
            if since_best > patience and lr_mult > 0.03:
                lr_mult *= 0.7
                since_best = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z = np.clip(z, -60.0, 60.0)
                z = z + 0.01 * rng.standard_normal(n)
                z = z - z.mean()
                z = np.clip(z, -60.0, 60.0)
                m.fill(0.0)
                v.fill(0.0)

        if name and (time.monotonic() - last_print > 30.0):
            print(f"  {name}: frac={frac:.2f} score={sc:.10f} best={best_score:.10f} mult={lr_mult:.2f} mode={mode}")
            last_print = time.monotonic()

        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, frac / 0.9)
        one_alpha = abs(alpha - 1.0) < 1e-92 if False else abs(alpha - 1.0) < 1e-9

        if mode == "dense":
            if one_alpha:
                w[:] = c
            else:
                np.power(c, alpha, out=w)
            sw = float(w.sum())
            if sw <= 0.0 or not np.isfinite(sw):
                w.fill(1.0 / float(L))
            else:
                w *= 1.0 / sw
            h = _grad_weights(a, fa, w, Ngrad)

        elif mode == "sparse":
            if K >= L:
                idx = np.arange(L)
                vals = c
            else:
                kth = L - K
                idx = np.argpartition(c, kth)[-K:]
                vals = c[idx]

            if one_alpha:
                wt = vals.copy()
            else:
                wt = np.empty(K, dtype=np.float64)
                np.power(vals, alpha, out=wt)

            sw = float(wt.sum())
            if sw <= 0.0 or not np.isfinite(sw):
                wt.fill(1.0 / float(K))
            else:
                wt *= 1.0 / sw

            w.fill(0.0)
            w[idx] = wt
            h = _grad_weights(a, fa, w, Ngrad)

        else:
            K = k_end
            if K >= L:
                idx = np.arange(L)
                vals = c
            else:
                kth = L - K
                idx = np.argpartition(c, kth)[-K:]
                vals = c[idx]

            if one_alpha:
                wt = vals.copy()
            else:
                wt = np.empty(K, dtype=np.float64)
                np.power(vals, alpha, out=wt)

            sw = float(wt.sum())
            if sw <= 0.0 or not np.isfinite(sw):
                wt.fill(1.0 / float(K))
            else:
                wt *= 1.0 / sw

            h = _grad_direct(a, idx, wt)

        if not np.all(np.isfinite(h)):
            h = np.zeros(n, dtype=np.float64)

        hs = h * float(n)
        dot = float(np.dot(a, hs))
        gz = a * (hs - dot)

        if not np.all(np.isfinite(gz)):
            gz = np.zeros(n, dtype=np.float64)

        m = b1 * m + (1.0 - b1) * gz
        v = b2 * v + (1.0 - b2) * (gz * gz)
        corr1 = b1 * corr1 + (1.0 - b1)
        corr2 = b2 * corr2 + (1.0 - b2)

        mh = m / corr1
        vh = v / corr2

        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        update = lr * mh / (np.sqrt(vh) + eps)

        un = float(np.linalg.norm(update))
        if un > clip_norm and un > 0.0:
            update *= clip_norm / un

        z = z - update

        if not np.all(np.isfinite(z)):
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
            z = np.clip(z, -60.0, 60.0)
        else:
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

    a = _softmax(z)
    sc = 2.0 * n * _conv_max(a, Nself)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    if name:
        print(f"  {name}: final best={best_score:.10f} n={n} time={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def _lp_polish(a, deadline, rng=None, max_active=None, max_iters=30, name=""):
    if deadline <= time.monotonic():
        return _normalize(a), _score(a)

    if linprog is None:
        return _normalize(a), _score(a)

    a = _normalize(a)
    n = len(a)
    Nself = _next_pow2(2 * n - 1)
    best_score = 2.0 * n * _conv_max(a, Nself)
    best_a = a.copy()

    if max_active is None:
        max_active = 32 if n >= 16384 else 64

    K = max(16, min(max_active, n // 256))
    it = 0

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        c, _ = _self_conv(a, Nself)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0.0:
            break

        L = 2 * n - 1
        if K > L:
            K = L

        if K >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(c, L - K)[-K:]

        A = np.zeros((K, n), dtype=np.float64)
        for r, k in enumerate(idx):
            k = int(k)
            start = max(0, k - n + 1)
            end = min(k, n - 1) + 1
            if start >= end:
                continue
            j_start = k - (end - 1)
            j_end = k - start + 1
            A[r, start:end] = a[j_start:j_end][::-1]

        rem = deadline - time.monotonic()
        lp_time = max(1.0, min(8.0, rem / 3.0))

        try:
            res = linprog(
                -a,
                A_ub=A,
                b_ub=np.full(K, M * (1.0 + 1e-7), dtype=np.float64),
                bounds=(0.0, 100.0),
                method="highs",
                options={
                    "time_limit": lp_time,
                    "disp": False,
                    "primal_feasibility_tol": 1e-8,
                    "dual_feasibility_tol": 1e-8,
                },
            )
        except Exception:
            res = None

        g = None
        if res is not None and hasattr(res, "x") and res.x is not None:
            g = np.asarray(res.x, dtype=np.float64)

        if g is None or not np.all(np.isfinite(g)):
            break

        g = np.maximum(g, 0.0)
        gs = float(g.sum())
        if gs <= 1e-12:
            break
        g /= gs

        found = False
        for tt in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02):
            cand = (1.0 - tt) * a + tt * g
            cand = np.maximum(cand, 0.0)
            cs = float(cand.sum())
            if cs <= 0.0:
                continue
            cand /= cs
            sc = 2.0 * n * _conv_max(cand, Nself)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = cand.copy()
                a = cand.copy()
                found = True
                break

        if found:
            if name:
                print(f"  {name}: LP n={n} K={K} score={best_score:.10f} it={it}")
        else:
            if K < max_active:
                K = min(max_active, K + max(8, K // 2))
            else:
                break

    return best_a, best_score


def _load_base_sequence():
    try:
        seq = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        return None

    if len(seq) == 0:
        return None

    seq = np.clip(seq, 0.0, 1000.0)
    s = float(seq.sum())
    if s < 1e-12:
        return None

    return seq / s


def _default_base(n=4096):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.power(x + 0.02, 0.75)
    return _normalize(a)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    total_budget = max(120.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    base = _load_base_sequence()
    if base is None:
        base = _default_base(4096)

    base = _normalize(base)

    max_n = 32768
    scales = []
    n = len(base)
    while n < max_n:
        scales.append(n)
        if n * 2 > max_n:
            break
        n *= 2

    scales = [x for x in scales if x >= 1024]
    if not scales:
        scales = [max(1024, len(base))]
    if len(scales) > 5:
        scales = scales[-5:]

    m = len(scales)
    if m == 1:
        weights = [1.0]
    elif m == 2:
        weights = [0.35, 0.65]
    elif m == 3:
        weights = [0.18, 0.37, 0.45]
    elif m == 4:
        weights = [0.12, 0.22, 0.34, 0.32]
    elif m == 5:
        weights = [0.09, 0.14, 0.27, 0.28, 0.22]
    else:
        weights = [0.06, 0.09, 0.15, 0.21, 0.25, 0.24]

    total_scale_time = max(60.0, (deadline - t_start))
    reserve_final = max(100.0, min(350.0, total_scale_time * 0.18))
    scale_budget_total = max(120.0, total_scale_time - reserve_final)
    sw = float(sum(weights[:m]))
    budgets = [w / sw * scale_budget_total for w in weights[:m]]

    global_best_a = base.copy()
    global_best_score = _score(global_best_a)

    source_a = base.copy()
    source_score = global_best_score

    for i, n in enumerate(scales):
        if time.monotonic() > deadline - 35.0:
            break

        budget = min(budgets[i], max(0.0, deadline - time.monotonic() - 30.0))
        a_clean = _interp_to_length(source_a, n)
        clean_score = _score(a_clean)

        if clean_score < global_best_score:
            global_best_score = clean_score
            global_best_a = a_clean.copy()

        if budget <= 20.0:
            a_opt = a_clean
            sc_opt = clean_score
            if sc_opt < source_score:
                source_a = a_opt.copy()
                source_score = sc_opt
            continue

        if n <= 8192:
            sigma = 0.008
        else:
            sigma = 0.004

        a_init = a_clean * (1.0 + sigma * rng.standard_normal(n))
        a_init = _normalize(a_init)

        local_deadline = min(deadline - 25.0, time.monotonic() + budget)
        a_opt, sc_opt = _adam_z(
            a_init,
            local_deadline,
            rng=rng,
            lr_start=0.046 if n <= 8192 else 0.040,
            lr_end=0.0035 if n <= 8192 else 0.0030,
            alpha_start=1.0,
            alpha_end=2.4 if n <= 16384 else 2.2,
            k_end=12 if n >= 16384 else 16,
            patience=900 if n <= 8192 else 1200,
            name=f"scale n={n}",
        )
        a_opt = _normalize(a_opt)
        sc_opt = _score(a_opt)

        if clean_score < sc_opt:
            a_opt = a_clean
            sc_opt = clean_score

        if np.isfinite(sc_opt) and sc_opt < global_best_score:
            global_best_score = sc_opt
            global_best_a = a_opt.copy()

        if np.isfinite(sc_opt) and (sc_opt < source_score or (n > len(source_a) and sc_opt < source_score + 2e-4)):
            source_a = a_opt.copy()
            source_score = sc_opt

    best_a = global_best_a.copy()
    best_score = global_best_score

    if not np.isfinite(best_score):
        best_a = base.copy()
        best_score = _score(best_a)

    if time.monotonic() < deadline - 120.0:
        lp_budget = min(120.0, max(30.0, (deadline - time.monotonic()) * 0.35))
        a_lp, sc_lp = _lp_polish(
            best_a,
            min(deadline - 90.0, time.monotonic() + lp_budget),
            rng=rng,
            name="LP polish",
        )
        a_lp = _normalize(a_lp)
        sc_lp = _score(a_lp)
        if np.isfinite(sc_lp) and sc_lp < best_score:
            best_score = sc_lp
            best_a = a_lp.copy()

    if time.monotonic() < deadline - 220.0 and best_score > 1.5:
        restart_budget = min(130.0, max(40.0, (deadline - time.monotonic()) * 0.35))
        a_restart = best_a * (1.0 + 0.03 * rng.standard_normal(len(best_a)))
        a_restart = _normalize(a_restart)
        a_rec, sc_rec = _adam_z(
            a_restart,
            min(deadline - 150.0, time.monotonic() + restart_budget),
            rng=rng,
            lr_start=0.032,
            lr_end=0.0020,
            alpha_start=1.0,
            alpha_end=2.0,
            k_end=12,
            patience=700,
            name="restart",
        )
        a_rec = _normalize(a_rec)
        sc_rec = _score(a_rec)
        if np.isfinite(sc_rec) and sc_rec < best_score:
            best_score = sc_rec
            best_a = a_rec.copy()

    if time.monotonic() < deadline - 12.0:
        final_deadline = min(deadline - 5.0, time.monotonic() + max(30.0, deadline - time.monotonic() - 20.0))
        a_final, sc_final = _adam_z(
            best_a,
            final_deadline,
            rng=rng,
            lr_start=0.012,
            lr_end=0.0008,
            alpha_start=1.0,
            alpha_end=1.8,
            k_end=8,
            patience=500,
            name="final Adam",
        )
        a_final = _normalize(a_final)
        sc_final = _score(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()

    final = np.asarray(best_a, dtype=np.float64).ravel()
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(1024, dtype=np.float64)

    print(f"FINAL internal score={_score(final):.12f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    return [float(x) for x in final]
```
