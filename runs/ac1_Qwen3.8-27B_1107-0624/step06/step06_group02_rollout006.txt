def _sobz_func_val(z, n, beta):
    """Objective value for L-BFGS: a = softmax(z), return 2n * smoothed_max(a*a)."""
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    a = e / e.sum()
    c = _convolve(a, a)
    cm = c.max()
    vals = (c - cm) / beta
    np.clip(vals, -50, 50, out=vals)
    lse = cm + beta * np.log(np.exp(vals).sum())
    return 2.0 * n * lse


def _sobz_func_grad(z, n, beta):
    """Gradient of the L-BFGS objective w.r.t. z."""
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    a = e / e.sum()
    c = _convolve(a, a)
    cm = c.max()
    vals = (c - cm) / beta
    np.clip(vals, -50, 50, out=vals)
    w = np.exp(vals)
    w = w / w.sum()
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    grad_a = 2.0 * wc[n - 1:2 * n - 1]
    dot = np.dot(grad_a, a)
    grad_z = a * (grad_a - dot)
    return 2.0 * n * grad_z

def _next_pow2(n):
