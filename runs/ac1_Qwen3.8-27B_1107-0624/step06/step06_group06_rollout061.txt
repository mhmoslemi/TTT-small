```python
    n = len(a)
    c = full_convolve(a, a)
    c_max = c.max()
    
    # Find top peaks
    top_idx = np.argsort(c)[::-1][:n_top_peaks]
    active = np.zeros_like(c, dtype=bool)
    active[top_idx] = True
    
    # Compute subgradient
    g = np.zeros(n)
    for k in top_idx:
        # grad of b_k: g[j] = 2 * a[k-j] if 0 <= k-j < n
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                g[j] += 2 * a[kj]
    g /= len(top_idx)
    
    # Project gradient onto simplex tangent space
    g = g - np.mean(g)
    
    # Take step
    a_new = a - alpha * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new
```

This is related to the design of "low autocorrelation" sequences, but for the LINEAR (not circular) autocorrelation, and with the non-negativity constraint.

OK so with this understanding, let me think about what optimization approaches work best.

The objective is: min_a max_k (a*a)_k / (sum a)^2 * 2n

Since it's scale-invariant: min_a max_k (a*a)_k subject to sum a = 1, a >= 0.

The peak will typically be near k = n-1 (the center). The constraint that matters is:
sum_i a_i a_{n-1-i} <= T (for the target T)

And we need all other c[k] <= T as well.

This is a system of quadratic inequality constraints. The feasible set is non-convex.

**Optimization approach: Bisection on T**

For a given T, check if there exists a >= 0, sum a = 1, max_k (a*a)_k <= T.
This is a feasibility problem with quadratic constraints. We could try:
- SDP relaxation (replace the quadratic constraints with their SDP relaxations)
- Or: use penalty/barrier methods
- Or: use iterative methods

Actually, for a given T, the problem is:
min 0
s.t. sum a = 1
     a >= 0
     (a*a)_k <= T for all k

The constraint (a*a)_k <= T can be written as a^T M_k a <= T where M_k is a symmetric matrix with (M_k)_{ij} = 1/2 if i+j=k (within range).

Note that M_k is PSD (it's a sum of rank-1 matrices v_i v_i^T where v_i are standard basis vectors... actually no).

Let me think: (a*a)_k = sum_i a_i a_{k-i}. The matrix M_k has entries (M_k)_{i,j} = 1 if i+j=k, 0 otherwise (for the off-diagonal), and (M_k)_{i,i} = 1 if 2i=k. So a^T M_k a = sum_{i+j=k} a_i a_j = (a*a)_k.

M_k is a symmetric matrix. Is it PSD? M_k = (1/2) * (v v^T) where v is... hmm, not quite.

Actually, M_k = sum_{i: 0<=i<n, 0<=k-i<n} e_i e_{k-i}^T symmetrized.
If k is even and k/2 is in range, M_k has a 1 on the diagonal at position k/2, and 1/2 at positions (i, k-i) for i ≠ k/2.
If k is odd, M_k has 1/2 at positions (i, k-i) for all valid i.

Let's check if M_k is PSD. Consider k=0: M_0 = e_0 e_0^T, which is PSD.
k=1: M_1 has (M_1)_{0,1} = (M_1)_{1,0} = 1/2. This is a 2x2 matrix [[0, 1/2], [1/2, 0]] embedded in n x n. Its eigenvalues are ±1/2. So it's NOT PSD!

So the constraints (a*a)_k <= T are NON-CONVEX (they define non-convex sets). This is why the problem is hard.

However, we can use the following approach:
- The constraint (a*a)_k <= T is equivalent to a^T M_k a <= T
- We can use the fact that for any PSD matrix S >= M_k (in Loewner order), a^T S a >= a^T M_k a, so a^T M_k a <= T implies a^T S a <= T + (a^T (S-M_k) a)
- This gives an outer approximation but not directly useful.

Alternatively, we can use the Lagrangian dual or penalty methods.

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.02, n_peaks=10, tol_frac=0.005):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    patience = 0
    lr_min = 1e-5
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        g, c, c_max = _subgrad_step(a, n_peaks=n_peaks, tol_frac=tol_frac)
        # Project onto tangent space of simplex
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Adaptive step size
        alpha = lr / gnorm  # Normalize step size
        
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            lr *= 0.9
            continue
        a_new = a_new / s_new
        
        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            a = a_new
            best_a = a_new.copy()
            best_sc = sc_new
            patience = 0
            lr = min(lr * 1.02, lr_init)
        else:
            patience += 1
            lr *= 0.95
            if lr < lr_min:
                # Try a random restart from best
                a = best_a.copy()
                lr = lr_init * 0.5
                if patience > 50:
                    break
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_a, best_sc

def _peak_shaving(a, deadline, n_iters=1000):
    """Coordinate-level peak shaving: try to reduce the max peak by adjusting individual elements."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        peak_pos = int(np.argmax(c))
        
        # Find the top contributing pairs to the peak
        # c[peak_pos] = sum_i a[i] * a[peak_pos - i]
        valid = np.arange(max(0, peak_pos - n + 1), min(n, peak_pos + 1))
        products = a[valid] * a[peak_pos - valid]
        
        # Try reducing the element with the largest product
        # and increasing an element that doesn't contribute to the peak
        if len(valid) < 2:
            break
        
        # Sort by product (descending)
        order = np.argsort(products)[::-1]
        
        improved = False
        for idx in order[:3]:
            i = valid[idx]
            j = peak_pos - i
            if i == j:
                continue
            # Try reducing a[i] and increasing a[j'] where j' doesn't contribute to peak
            # Actually, let's try a simple swap: move mass from i to some other position
            eps = a[i] * 0.01  # 1% of the element
            if eps < 1e-15:
                continue
            
            # Try moving mass from i to position m (randomly chosen from low-contribution positions)
            # For simplicity, try a few random targets
            for _ in range(5):
                m = np.random.randint(n)
                if m == i or m == j:
                    continue
                # Check if m contributes to the peak
                if abs(m - peak_pos) < n and 0 <= peak_pos - m < n:
                    # m DOES contribute to the peak (as a[peak_pos - m])
                    # Skip if it would increase the peak
                    continue
                
                a_try = a.copy()
                a_try[i] -= eps
                a_try[m] += eps
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try < best_sc - 1e-12:
                    a = a_try
                    best_a = a_try.copy()
                    best_sc = sc_try
                    improved = True
                    break
            if improved:
                break
        
        if not improved and it > 100:
            # Try a global perturbation
            z = np.log(np.maximum(a, 1e-30))
            z += 0.01 * np.random.randn(n)
            z -= z.mean()
            a = _softmax(z)
    
    return best_a, best_sc
```

```python
Step function optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
refined L-BFGS with log-sum-exp smoothing, subgradient polish with
Armijo backtracking, and diverse restarts. Key improvements over baseline:
much finer beta schedule, Armijo step size, active set cycling, and

import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgrad_of_max(a, tol_frac=0.003):
    """Compute subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if not active.any():
        active[int(np.argmax(c))] = True
    n_act = int(active.sum())
    w = active.astype(np.float64) / n_act
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    # Subgradient of max ≈ (2/n_act) * sum_{k in active} a[k-j]
    # h[j] = sum_k w[k] * a[k-j] = (1/n_act) * sum_{k in active} a[k-j]
    # So subgrad = 2 * h
    g = 2.0 * h
    return g, c, c_max

def _subgrad_polish(a_init, deadline, lr_init=0.015, tol_frac=0.003, max_steps=10000):
    """Subgradient method with Armijo backtracking."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score(a_init)
    a = a / s
    best_a = a.copy()
    best_cmax = float(_full_convolve(a, a).max())
    best_sc = 2.0 * n * best_cmax / s**2  # s=1 now
    best_sc = _score(a)
    lr = lr_init
    patience = 0
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        g, c, c_max = _subgrad_of_max(a, tol_frac=tol_frac)
        # Project onto tangent space
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        # Armijo backtracking
        alpha = lr
        f_cur = c_max
        improved = False
        for _ in range(30):
            a_try = a - alpha * g / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_cur - 1e-12 * alpha * gnorm:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
                best_cmax = f_try
                patience = 0
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.97
        else:
            patience += 1
            lr *= 0.9
            if patience > 100:
                a = best_a.copy()
                lr = lr_init * 0.5
                patience = 0
                if lr < 1e-6:
                    break
    
    return best_a, best_sc

    n = len(a_init)

        return a_init.copy()

def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = rng.uniform(0.01, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

def _perturb_block(a, rng, n_blocks=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks * 2))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i+block_size].copy()
        z[i:i+block_size] = z[j:j+block_size]
        z[j:j+block_size] = tmp
    return _softmax(z)

def _zero_small(a, frac):
    """Zero out the smallest frac fraction of elements and renormalize."""
    n = len(a)
    n_zero = max(1, int(n * frac))
    idx = np.argsort(a)[:n_zero]
    a_try = a.copy()
    a_try[idx] = 0.0
    s = a_try.sum()
    if s < 1e-15:
        return a
    a_try = a_try / s
    return a_try

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":

        a = 1.0 + 0.5 * (2*t - 1)**2
```

```python
"""
Step function optimizer: minimizes 2*n*max(conv(a,a))/sum(a)^2 using
refined L-BFGS with very fine log-sum-exp beta schedule, subgradient
polish with Armijo backtracking, diverse restarts, shrinkage, and
multi-n exploration. Key differences from baseline: much finer beta grid,
Armijo line search in subgradient, active set cycling, and structured
spectral perturbations.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_fn(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_of_max(a, tol_frac=0.003):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if not active.any():
        active[int(np.argmax(c))] = True
    n_act = int(active.sum())
    w = active.astype(np.float64) / n_act
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    g = 2.0 * h
    return g, c, c_max


def _subgrad_polish(a_init, deadline, lr_init=0.015, tol_frac=0.003, max_steps=10000):
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init.copy(), _score(a_init)
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    patience = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        g, c, c_max = _subgrad_of_max(a, tol_frac=tol_frac)
        g = g - np.mean(g)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        alpha = lr
        f_cur = c_max
        improved = False
        for _ in range(30):
            a_try = a - alpha * g / gnorm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_cur - 1e-12 * alpha * gnorm:
                a = a_try
                improved = True
                break
            alpha *= 0.5
        if improved:
            sc = _score(a)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_a = a.copy()
                patience = 0
                lr = min(lr * 1.01, lr_init)
            else:
                patience += 1
                lr *= 0.97
        else:
            patience += 1
            lr *= 0.92
            if patience > 80:
                a = best_a.copy()
                lr = max(lr_init * 0.4, 1e-5)
                patience = 0
                if lr < 1e-6:
                    break
    return best_a, best_sc


def _run_lbfgs(a_init, deadline, min_time_remaining=60.0):
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init.copy()
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)] if len(c_sorted) > 20 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.02)
    betas = []
    b = base_beta
    for _ in range(80):
        betas.append(b)
        b *= 0.48
        if b < max_c * 5e-11:
            break
    while len(betas) < 20:
        betas.append(betas[-1] * 0.5)
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_iter_time = 0.005 * max(1.0, n / 4096.0)
        max_iter = min(4000, max(20, int(remaining / (est_iter_time * 12))))
        try:
            result = minimize(
                _smoobj_fn, z, args=(beta, n), jac=_smograd_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                         'maxcor': 50, 'maxls': 100}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
            if i % 15 == 0:
                print(f"    beta={beta:.3e}, score={current_sc:.10f}, phase_idx={i}")
        except Exception:
            pass
    return current_a


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = rng.uniform(0.01, 0.06)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks * 3))
    for _ in range(n_blocks):
        i = int(rng.integers(0, max(1, n - 2 * block_size)))
        j = int(rng.integers(0, max(1, n - 2 * block_size)))
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _zero_small(a, frac):
    n = len(a)
    n_zero = max(1, int(n * frac))
    idx = np.argsort(a)[:n_zero]
    a_try = a.copy()
    a_try[idx] = 0.0
    s = a_try.sum()
    if s < 1e-15:
        return a
    return a_try / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _try_update(state, p, label=""):
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    sc = _score(p)
    if np.isfinite(sc) and sc < state['score']:
        prev = state['score']
        state['seq'] = p.copy()
        state['score'] = float(sc)
        print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] "
              f"improve={prev - sc:.2e} t={state['t'] - state['t0']:.1f}s")

    state = {'seq': None, 'score': np.inf, 't0': t_start, 't': t_start}

    # Load parent

    try:

    except Exception:
        pass

    if parent is not None:

        print(f"Parent: n={n_parent}, score={sc_p:.10f}")
        state['t'] = time.monotonic()

    else:
        n_parent = 4096

        state['t'] = time.monotonic()
        _try_update(state, parent, "init")

    # Phase 1: Main L-BFGS
```
