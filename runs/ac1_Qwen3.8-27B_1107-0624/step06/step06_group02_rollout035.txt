Multi-phase optimizer using Lp-norm and log-sum-exp smoothed objectives
with Adam in log-space. Targets sub-1.5027 score for non-negative step functions.

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_Lp_logspace(z, p):
    """Gradient of sum(c_k^p) w.r.t. z, where a = softmax(z), c = a*a."""
    n = len(z)
    a = _softmax(z)
    c = _convolve(a, a)
    if p == 1:
        w = np.ones_like(c)
    else:
        w = p * np.power(np.maximum(c, 1e-300), p - 1)
    a_rev = a[::-1]
    conv = _convolve(a_rev, w)
    grad_a = 2.0 * conv[n - 1:2 * n - 1]
    inner = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - inner)
    return grad_z


def _grad_lse_logspace(z, beta):
    """Gradient of log(sum(exp(c_k/beta))) w.r.t. z, where a = softmax(z), c = a*a."""
    n = len(z)
    a = _softmax(z)
    c = _convolve(a, a)
    c_shifted = (c - c.max()) / beta
    np.clip(c_shifted, -100, 100, out=c_shifted)
    e = np.exp(c_shifted)
    w = e / e.sum() / beta  # w_k = (1/beta) * exp(c_k/beta) / sum(exp(c_l/beta))
    a_rev = a[::-1]
    conv = _convolve(a_rev, w)
    grad_a = 2.0 * conv[n - 1:2 * n - 1]
    inner = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - inner)
    return grad_z


def _adam_optimize_Lp(z_init, p_schedule, deadline, lr=0.01, max_iter_per_p=5000):
    """Adam optimization in log-space with Lp objective."""
    n = len(z_init)
    z = z_init.copy()
    a = _softmax(z)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        lr_current = lr
        for it in range(max_iter_per_p):
            if time.monotonic() > deadline:
                break
            t += 1
            grad = _grad_Lp_logspace(z, p)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1**t)
            vh = v_buf / (1 - b2**t)
            
            update = lr_current * mh / (np.sqrt(vh) + eps)
            # Gradient clipping in log-space
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update *= 0.5 / update_norm
            
            z = z - update
            z = np.clip(z, -50, 50)
            
            if t % 200 == 0:
                a = _softmax(z)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        # Decay learning rate between p values
        lr = max(0.001, lr * 0.5)
    
    a = _softmax(z)
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score, z

def _subgradient_step(a, step_size):
    """Direct subgradient step on the non-smooth max objective."""
    n = len(a)
    c = _convolve(a, a)
    c_max = c.max()
    # Active set: positions within small epsilon of max
    active = np.zeros_like(c)
    active[np.abs(c - c_max) < 0.01 * c_max] = 1.0
    # Or just use the argmax
    # active[int(np.argmax(c))] = 1.0
    
    a_rev = a[::-1]
    conv = _convolve(a_rev, active)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    
    a_new = a - step_size * grad
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = a
    return a_new


def _coord_descent(a_init, deadline, max_rounds=200):
    """Coordinate descent polish."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(42)
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        
        improved = False
        for j in indices:
            if time.monotonic() > deadline - 1:
                break
            old_val = a[j]
            if old_val < 1e-15:
                # Try setting it to a small value
                a[j] = a.mean() * 0.1
                sc = _score_norm(a / a.sum())
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = 0.0
                continue
            for factor in [0.95, 0.98, 1.02, 1.05]:
                a[j] = old_val * factor
                sc = _score_norm(a / a.sum())
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
                a[j] = old_val
        
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 30:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):

    p_new = np.interp(x1, x0, p * np.sqrt(n0))
