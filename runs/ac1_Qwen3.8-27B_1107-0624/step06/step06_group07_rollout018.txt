Multi-strategy optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, LP-guided steepest descent,

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_only(a):
    c = _full_convolve(a, a)
    return c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gauss_center":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-12):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 40, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_time_remaining=100.0, aggressive=False):

def _lp_direction(a, c, n, c_max, active_set, epsilon=0.01):
    """Solve LP to find steepest descent direction for max(c).
    Returns direction d (sum=0) and expected reduction."""
    num_active = len(active_set)
    if num_active == 0:
        return None, 0.0

    # Build constraint matrix for active constraints only
    # Constraint k: 2 * sum_i a[k-i] * d_i - t <= -c_k
    # Variables: d_0,...,d_{n-1}, t (total n+1 variables)
    # Plus: sum(d) = 0
    # Plus: -eps <= d_i <= eps

    A_rows = []
    b_vals = []

    for k in active_set:
        row = np.zeros(n + 1)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[i] = 2.0 * a[j]
        row[n] = -1.0  # coefficient of t
        A_rows.append(row)
        b_vals.append(-c[k])

    # Box constraints: d_i <= eps
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_rows.append(row)
        b_vals.append(epsilon)

    # Box constraints: -d_i <= eps
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_vals.append(epsilon)

    A = np.array(A_rows)
    b = np.array(b_vals)

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=[(None, None)] * n + [(c_max * 0.5, c_max * 1.5)],
                      method='highs')
        if res.success:
            d = res.x[:n]
            t_val = res.x[n]
            reduction = c_max - t_val
            # Normalize d to have sum 0 (should already, but enforce)
            d = d - d.mean()
            return d, reduction
    except Exception:
        pass

    return None, 0.0

def _subgrad_direct(a_init, deadline, tol_frac=1e-5, max_steps=50000):

    a = a_init.copy()
    best_a = a.copy()

        c = _full_convolve(a, a)

        a_try = a_try / s

def _coordinate_descent(a_init, deadline, max_rounds=2000):
    """Pairwise coordinate descent: redistribute mass between pairs."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    rng_stepper = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Pick a random pair
        i = int(rng_stepper.integers(0, n))
        j = int(rng_stepper.integers(0, n))
        if i == j:
            j = (j + 1) % n

        s_pair = best_a[i] + best_a[j]
        if s_pair < 1e-15:
            continue

        current_frac = best_a[i] / s_pair

        # Try several redistributions
        candidates = [current_frac + 0.1, current_frac - 0.1,
                      current_frac + 0.3, current_frac - 0.3,
                      current_frac + 0.5, current_frac - 0.5]
        candidates = [max(0.0, min(1.0, c)) for c in candidates]

        best_local_sc = best_sc
        best_local_frac = current_frac

        for frac in candidates:
            if abs(frac - current_frac) < 1e-8:
                continue
            a_try = best_a.copy()
            a_try[i] = s_pair * frac
            a_try[j] = s_pair * (1.0 - frac)
            sc_try = _score(a_try)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_local_frac = frac

        if best_local_frac != current_frac:
            best_a[i] = s_pair * best_local_frac
            best_a[j] = s_pair * (1.0 - best_local_frac)
            best_a = best_a / best_a.sum()
            best_sc = best_local_sc

    return best_a, best_sc

def _interpolate_to_n(a, target_n):

def _lp_polish(a_init, deadline, max_iter=20):
    """Use LP directions to polish the solution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.005 * c_max
        active_set = list(np.where(c >= (c_max - tol))[0])

        if len(active_set) > 200:
            # Subsample if too many active constraints
            active_set = list(np.random.choice(active_set, 200, replace=False))

        d, reduction = _lp_direction(a, c, n, c_max, active_set, epsilon=1.0/n)

        if d is None:
            break

        if reduction < 1e-10:
            break

        # Take step with line search
        best_step_sc = best_sc
        best_step_a = a.copy()
        for alpha_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = alpha_mult * (1.0 / np.linalg.norm(d))
            a_try = a + alpha * d
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            sc_try = _score(a_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                best_step_a = a_try.copy()

        if best_step_sc < best_sc:
            best_sc = best_step_sc
            best_a = best_step_a.copy()
            a = best_step_a.copy()
        else:
            break

    return best_a, best_sc

    print(f"\n=== Phase 7: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        restart_configs = [
            ("gauss_002", 0.02),
            ("gauss_005", 0.05),
            ("gauss_010", 0.10),
            ("gauss_015", 0.15),
        ]

        for name, sigma in restart_configs:
            if time.monotonic() > deadline - 60:
                break

            p_start = _perturb(best_seq, sigma, rng)
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            if sc_start < best_score_val + 0.01:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=40.0,
                                       aggressive=True)
                try_update(a_r, f"restart_{name}")

                if time.monotonic() < deadline - 35:
                    a_r2, _ = _subgrad_momentum(a_r, deadline - 35,
                                                tol_frac=1e-4, max_steps=5000,
                                                lr_init=0.02, momentum=0.85)
                    try_update(a_r2, f"restart_{name}_sg")

    # === Phase 8: Try different lengths ===
    print(f"\n=== Phase 8: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        current_n = len(best_seq)
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 50:
                break
            if target_n == current_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"  Interpolated score: {sc_interp:.10f}")
            if sc_interp < 2.0:
                a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=30.0,
                                         aggressive=True)
                try_update(a_try, f"n_{target_n}")

    # === Phase 9: Final aggressive LP + subgradient cycle ===
    print(f"\n=== Phase 9: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        for rnd in range(5):
            if time.monotonic() > deadline - 10:
                break

            # LP step
            if time.monotonic() < deadline - 15:
                a_lp, _ = _lp_polish(best_seq, deadline - 15, max_iter=10)
                try_update(a_lp, f"final_lp_{rnd}")

            # Tiny beta L-BFGS
            if time.monotonic() < deadline - 10:
                n = len(best_seq)
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = max(mb * 1e-7, 1e-15)
                a_f, _ = _lbfgs_run(best_seq, beta, n, 3000)
                try_update(a_f, f"final_lbfgs_{rnd}")

            # Momentum subgradient
            if time.monotonic() < deadline - 10:
                a_s, _ = _subgrad_momentum(best_seq, deadline - 10,
                                           tol_frac=1e-5, max_steps=3000,
                                           lr_init=0.01, momentum=0.8)
                try_update(a_s, f"final_sg_{rnd}")

    # === Phase 10: Fourier perturb + quick optimize ===
    print(f"\n=== Phase 10: Fourier perturb (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for nmodes in [2, 5, 8]:
            if time.monotonic() > deadline - 15:
                break
            p_start = _perturb_fourier(best_seq, rng, n_modes=nmodes)
            sc_start = _score(p_start)
            if sc_start < best_score_val + 0.005:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=10.0,
                                       aggressive=True)
                try_update(a_r, f"fourier_{nmodes}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}, n={len(fallback)}")
    return [float(x) for x in fallback]
