def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)

    s = e.sum()

        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _perturb_gaussian(a, sigma, rng):

def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.004 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(4, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
        z[start:end] += rng.uniform(-0.015, 0.015)
    z = z - z.mean()
    return _softmax(z)


def _local_search(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.005

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        idx = int((rnd * 997) % n)
        jdx = int((rnd * 997 + 7) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        for delta_frac in [0.3, 0.5, 0.7]:
            s_pair = best_a[idx] + best_a[jdx]
            new_val_i = s_pair * (1.0 - delta_frac + step)
            new_val_j = s_pair * (delta_frac - step)
            if new_val_i < 0 or new_val_j < 0:
                continue

            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True

        if not improved and rnd > 20:
            step *= 0.7
            if step < 1e-6:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp

def simple_convolve(a):
    """Compute full linear self-convolution"""
    n = len(a)
    out_n = 2 * n - 1
    fa = np.fft.rfft(a, 2 * n)
    fb = np.fft.rfft(a, 2 * n)
    fc = np.fft.irfft(fa * fb)
    return fc[:out_n] if len(fc) > out_n else fc[:out_n+1]
