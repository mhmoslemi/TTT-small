For the linear (non-cyclic) convolution, it's more complex, but for sequences that are "spread out" enough, the cyclic and linear convolutions are similar (the wrap-around terms are small).

If I use the cyclic autocorrelation as a proxy, the problem becomes:
`minimize max_k IDFT(|A|^2)[k]` where `A = DFT(a)`

This is: `minimize max_k sum_j |A[j]|^2 * exp(2*pi*i*j*k/n)`

By Parseval's theorem, the mean of `autocorr` is `sum(|A|^2)/n = sum(a^2)`. To minimize the max, we want `autocorr` to be as flat as possible, which means `|A|^2` should be concentrated at `j=0` (the DC component). But `|A[0]|^2 = (sum a)^2`, so if `sum(a) = 1`, then `|A[0]|^2 = 1` and the remaining `|A[j]|^2` for `j > 0` should be as small as possible while keeping `a >= 0`.

This is a different angle but might not be directly helpful for the linear convolution case.

import numpy as np
import time
import sys


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _convolve_self(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2*n*max(a*a)"""
    n = len(a)
    c = _convolve_self(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_weights(c, a, w):
    """
    Compute gradient of smoothed max using softmax weights w over convolution values.
    grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    This equals 2 * cross_correlation(w, a) at the appropriate lags.
    """
    n = len(a)
    # grad[j] = 2 * sum_k w[k] * a[k-j] for 0<=k-j<n
    # Let a_rev[i] = a[n-1-i], then this is a convolution
    a_rev = a[::-1]
    # (w * a_rev)[j + n - 1] = sum_k w[k] * a_rev[j+n-1-k] = sum_k w[k] * a[n-1-(j+n-1-k)] = sum_k w[k] * a[k-j]
    wc = _convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad


def _fista_smoothed(a_init, beta, deadline, max_iter, verbose=False):

    n = len(a_init)

    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    # Pre-allocate
    c_buf = np.empty(2 * n - 1)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve_self(y)
        c_max = float(c_y.max())

        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_from_weights(w, y, w)

            sc = _score_norm(a_next)
            if sc < best_sc:

                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _fista_exact_subgrad(a_init, deadline, max_iter, top_k_frac=0.01):
    """FISTA-like method with exact subgradient (uniform over top-K peaks)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    # Initial step size
    step = 0.3  # Will adapt
    eval_count = 0
    best_ever = best_sc

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve_self(y)
        c_max = float(c_y.max())

        # Find top-K active set
        k_active = max(1, int(top_k_frac * (2 * n - 1)))
        c_sorted_idx = np.argpartition(c_y, -k_active)[-k_active:]
        active_mask = np.zeros(2 * n - 1)
        active_mask[c_sorted_idx] = 1.0
        # Uniform weights on active set
        w = active_mask / k_active

        grad = _grad_from_weights(c_y, y, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        a_next = _proj_simplex(y - step * grad)

        sc = _score_norm(a_next)
        if sc < best_sc:
            best_sc = sc
            best_a = a_next.copy()

        # Adaptive step: decrease if increased, increase if decreased significantly
        if sc > c_max_n := float(c_y.max()) * 2 * n:
            step *= 0.8
        else:
            step = min(step * 1.05, 0.5)

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

        if it % 500 == 0 and verbose:
            print(f"    subgrad it={it}, score_norm={sc:.10f}, step={step:.4f}")

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _subgradient_polish(a_init, deadline, max_iter=50000, top_k=50):
    """Polyak-style subgradient method with exact top-k subgradient."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_f = _score_norm(a)
    # Upper bound for Polyak step
    f_star_est = best_f * 0.999  # running estimate of optimal value

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _convolve_self(a)
        c_max = float(c.max())
        current_f = 2.0 * n * c_max

        if current_f < best_f:
            best_f = current_f
            best_a = a.copy()
            f_star_est = current_f

        # Top-k subgradient
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(2 * n - 1)
        w[idx] = 1.0 / top_k

        grad = _grad_from_weights(c, a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Polyak step
        gap = current_f - f_star_est
        if gap < 1e-12:
            break
        step = gap / (grad_norm * grad_norm)
        step = min(step, 0.1)  # cap for stability

        a_new = _proj_simplex(a - step * grad)
        a = a_new

    return best_a, best_f


def _adam_logspace(a_init, deadline, lr_init=0.005, beta_init=None, max_iter=200000):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    current_beta = beta_init if beta_init is not None else 0.01

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _convolve_self(a)
        cm = float(c.max())
        c_shifted = (c - cm) / current_beta
        np.clip(c_shifted, -500, 500, out=c_shifted)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 100 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        # Adapt beta and lr

            current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()
    return best_a, best_score


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

        sizes_to_try.insert(0, n_main)

    print(f"Sizes to try: {sizes_to_try}")

    # Allocate time: 50% to parent size, remaining split
    time_for_parent = 0.45 * (deadline - time.monotonic())
    time_per_other = (deadline - time.monotonic() - time_for_parent) / max(1, len(sizes_to_try) - 1)

    for size_idx, sz in enumerate(sizes_to_try):
        if time.monotonic() > deadline - 30:
            break

        if sz == n_main:
            a_sz = a0.copy()
            alloc_time = time_for_parent
        else:
            a_sz = _interp_profile(best_seq, sz)
            alloc_time = time_per_other

        sc_sz = _score(a_sz)
        print(f"  Size {sz}: initial score={sc_sz:.10f}")

        if sc_sz < best_score:
            best_score = sc_sz
            best_seq = a_sz.copy()

        # Optimize
        sz_deadline = time.monotonic() + alloc_time
        a_opt = _optimize_sequence(a_sz.copy(), sz_deadline, verbose=(sz == n_main))
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  Size {sz} optimized: score={sc_opt:.10f}")

    # === Multi-restart with perturbations ===
    print(f"Before restarts: best_score={best_score:.10f}")
    if time.monotonic() < deadline - 60 and best_seq is not None:
        n_restarts = 6
        per_restart = (deadline - time.monotonic() - 30) / n_restarts

        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()

        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break

            # Diverse perturbations
            if i % 3 == 0:
                p_start = _perturb(a_work, 0.01 + 0.005 * i, rng)
            elif i % 3 == 1:
                # Circular shift
                shift = rng.integers(1, len(a_work) // 4)
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                # Block permutation
                block = len(a_work) // 3
                p_start = a_work.copy()
                p_start[:block] = a_work[block:2*block]
                p_start[block:2*block] = a_work[2*block:3*block]
                p_start[2*block:3*block] = a_work[:block]
                p_start = p_start / p_start.sum()

            p_deadline = time.monotonic() + per_restart * 0.8
            p_opt = _optimize_sequence(p_start, p_deadline, verbose=False)
            sc_p = _score(p_opt)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = p_opt.copy()
                print(f"  Restart {i}: score={sc_p:.10f} (n={len(p_opt)})")
                a_work = p_opt.copy()

    # === Final ultra-fine polish on best ===
    print(f"Before final polish: best_score={best_score:.10f}")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        # Very fine FISTA
        c_pol = _convolve_self(a_pol)
        max_cp = float(c_pol.max())
        fine_betas = [0.00001 * max_cp, 0.000005 * max_cp, 0.000001 * max_cp]
        per_beta = (deadline - time.monotonic() - 5) / (len(fine_betas) + 1)

        for beta in fine_betas:
            if time.monotonic() > deadline - 5:
                break
            a_pol, sc = _fista_smoothed(a_pol, beta, time.monotonic() + per_beta, 100000)
            a_pol = a_pol / a_pol.sum()
            sc_full = _score(a_pol)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_pol.copy()

        # Final subgradient
        if time.monotonic() < deadline - 5:
            a_pol, sc = _subgradient_polish(a_pol, deadline - 3, max_iter=30000,
                                             top_k=max(20, len(a_pol) // 50))
            a_pol = a_pol / a_pol.sum()
            sc_full = _score(a_pol)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_pol.copy()

    # === Final validation ===
    print(f"Final: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}, "
          f"time={time.monotonic()-t_start:.1f}s")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"Validated: score={sc_final:.10f}, n={len(final)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
