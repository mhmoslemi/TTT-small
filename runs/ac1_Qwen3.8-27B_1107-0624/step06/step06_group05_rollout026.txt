"""
Multi-phase optimizer with backtracking FISTA, level method, and active-set refinement
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations vs previous: backtracking line search, level method (Fletcher),
active-set oscillation, and adaptive subgradient with Polyak step.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes a is on simplex (sum=1). Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_from_weights(w, a):
    """Compute 2*(W*a) where W = sum_k w_k Q_k, using FFT."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _subgradient(a):
    """Compute a subgradient of max(conv(a,a)) at point a."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    tol = 1e-9 * c_max + 1e-15
    active = np.where(c >= c_max - tol)[0]
    w = np.zeros(2 * n - 1)
    w[active] = 1.0 / len(active)
    return _compute_grad_from_weights(w, a)


def _smoothed_grad(a, beta):
    """Compute gradient of log-sum-exp smoothing of max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    w = np.exp((c - c_max) / beta)
    w_sum = float(w.sum())
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    return _compute_grad_from_weights(w, a), c


def _fista_backtrack(a_init, beta, deadline, max_iter, lr_init=0.01):
    """FISTA with backtracking line search on log-sum-exp smoothed objective."""
    n = len(a_init)

        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:

        else:
            y = np.ones(n) / n

        obj_y = _lse_obj(y, beta)

        a_next = y
        alpha = step

            f_cand = _lse_obj(a_cand, beta)

def _lse_obj(a, beta):
    """Log-sum-exp approximation of max(conv(a,a))."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if beta < 1e-15:
        return c_max
    val = c_max + beta * np.log(np.sum(np.exp((c - c_max) / beta)))
    return float(val)


def _level_method(a_init, deadline, max_iter):

    n = len(a_init)

    m = f_curr  # level

    delta = 1e-8

        g_norm_sq = float(g @ g)
        if g_norm_sq < 1e-30:
            break

            # Steepest descent to get back to level
            # Solve: min ||y - a||^2 s.t. f_curr + g^T(y-a) <= m

        a = _softmax_compute(z)
        grad_a, c = _smoothed_grad(a, beta)

        # Convert gradient from a-space to z-space
        # da/dz = a * (I - a^T 1) ... Jacobian of softmax
        # grad_z = (I - a*a^T)^T * grad_a / a  (element-wise)
        # Simplified: grad_z_i = grad_a_i - sum_j a_j * grad_a_j (for softmax param)
        correction = float(a @ grad_a)
        grad_z = grad_a - correction
        # In log-space: grad_z_i = a_i * (grad_a_i - correction) ... actually
        # If a = softmax(z), then d/dz [f(a(z))]_i = sum_j (da_j/dz_i) * grad_a_j
        # da_j/dz_i = a_j * (delta_ji - a_i)
        # So grad_z_i = sum_j a_j * (delta_ji - a_i) * grad_a_j = a_i * grad_a_i - a_i * sum_j a_j * grad_a_j
        # = a_i * (grad_a_i - correction)
        grad_z = a * grad_z  # Wait, this double-counts. Let me redo.
        # Actually: grad_z_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
        # But since sum_j a_j * grad_a_j is the same for all i, and
        # grad_z should be: a * (grad_a - meanweighted)
        # Let me just use: grad_z = grad_a / a (element-wise, with clipping)
        # No, that's wrong too. Let me use the correct formula.
        # grad_z = a * (grad_a - 1^T a^T grad_a) ... but 1^T a^T grad_a doesn't make sense
        # Correct: grad_z = a * (grad_a - a @ grad_a) where the subtraction is scalar
        # Hmm, let me just use the chain rule properly.
        # z -> a = softmax(z): a_i = exp(z_i) / sum exp(z_j)
        # d a_i / d z_j = a_i (delta_ij - a_j)
        # grad_z_j = sum_i (d a_i / d z_j) * grad_a_i = sum_i a_i (delta_ij - a_j) grad_a_i
        # = a_j grad_a_j - a_j sum_i a_i grad_a_i = a_j (grad_a_j - a @ grad_a)
        a_at_ga = float(a @ grad_a)
        grad_z = a * (grad_a - a_at_ga)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.2:
            update = update * (0.2 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.95)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_final(a_init, deadline, beta):
    """L-BFGS-B on log-space for final refinement."""
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        grad_a, c = _smoothed_grad(a, beta)
        a_at_ga = float(a @ grad_a)
        grad_z = a * (grad_a - a_at_ga)
        return grad_z

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(30, int(max_time * 20)), 'gtol': 1e-13, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: FISTA with backtracking (aggressive) ===
    print("Phase 1: FISTA backtracking + aggressive beta annealing")
    phase1_end = min(deadline, t_start + 400.0)
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    beta = 0.02 * max_c0
    n_phases = 30

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        if remaining < 1:
            break
        per_time = remaining / max(1, n_phases - phase)
        per_time = min(per_time, 20.0)
        max_it = max(2000, int(per_time * 5000))
        a_work, _ = _fista_backtrack(a_work, beta, time.monotonic() + per_time, max_it,
                                     lr_init=0.02)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 5 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.3 * float(c_curr.max())
        if beta < 1e-12:
            beta = 1e-12

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Level method ===
    print("Phase 2: Fletcher level method")
    phase2_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase2_end - 10:
        a_lm, sc_lm = _level_method(a_work, phase2_end, max_iter=200000)
        if sc_lm < best_score:
            best_score = sc_lm
            best_seq = a_lm.copy()
            print(f"  Level method: score={sc_lm:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lm

    # === Phase 3: Active-set refinement ===
    print("Phase 3: Active-set refinement")
    phase3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < phase3_end - 15:
        # Try different K values
        for K in [10, 20, 40]:
            if time.monotonic() > phase3_end - 30:
                break
            a_as, sc_as = _active_set_opt(a_work, time.monotonic() + min(60, phase3_end - time.monotonic() - 10), K=K)
            if sc_as < best_score:
                best_score = sc_as
                best_seq = a_as.copy()
                print(f"  Active set K={K}: score={sc_as:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = a_as

    # === Phase 4: Polyak subgradient ===
    print("Phase 4: Polyak subgradient")
    phase4_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase4_end - 15:
        a_poly, sc_poly = _polyak_subgradient(a_work, phase4_end, max_iter=100000,
                                              f_star_est=best_score * 0.99)
        if sc_poly < best_score:
            best_score = sc_poly
            best_seq = a_poly.copy()
            print(f"  Polyak: score={sc_poly:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_poly

    # === Phase 5: Adam fine-tuning on log-space ===
    print("Phase 5: Adam fine-tuning")
    phase5_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase5_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0005 * max_cc, 0.0001 * max_cc, 0.00002 * max_cc]
        per_beta = (phase5_end - time.monotonic() - 10) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase5_end - 10:
                break
            a_work, sc = _adam_logspace(a_work, time.monotonic() + max(5.0, per_beta),
                                        beta, lr_init=0.004, max_iter=300000)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: L-BFGS refinement ===
    print("Phase 6: L-BFGS refinement")
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00003 * max_cc
        a_lb, sc_lb = _lbfgs_final(a_work, phase6_end, beta_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 7: Multi-restart exploration ===
    print("Phase 7: Multi-restart")
    phase7_end = min(deadline, t_start + 1770.0)
    if time.monotonic() < phase7_end - 20:
        n_restarts = 3
        per_r = (phase7_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            if i == 0:
                p = _perturb(a_work, 0.01, rng)
            elif i == 1:
                # Structured perturbation: random block scaling
                p = a_work.copy()
                block_size = n_main // 64
                for b in range(64):
                    start = b * block_size
                    end = min(start + block_size, n_main)
                    scale = rng.uniform(0.5, 1.5)
                    p[start:end] *= scale
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            else:
                p = _perturb(a_work, 0.025, rng)

            # Quick FISTA + Adam
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.0002 * max_cp
            p, _ = _fista_backtrack(p, b_r, time.monotonic() + per_r * 0.4, 20000, lr_init=0.01)
            p = p / p.sum()
            p, _ = _adam_logspace(p, time.monotonic() + per_r * 0.5, b_r * 0.3, 0.003, 50000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 8: Final ultra-polish ===
    print("Phase 8: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            # Final Adam with tiny beta
            a_pol, sc_p = _adam_logspace(a_pol, time.monotonic() + per_t * 0.5,
                                         0.000003 * max_cp, 0.001, 100000)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # Final L-BFGS
            if time.monotonic() < deadline - 6:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000002 * float(c_f.max())
                a_pol, sc_f = _lbfgs_final(a_pol, deadline - 3.0, b_f)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
