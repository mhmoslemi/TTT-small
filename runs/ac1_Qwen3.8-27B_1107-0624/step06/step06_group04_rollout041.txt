    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # === Phase 1: Nesterov-subgradient on parent (strong K focus) ===
    print("Phase 1: Nesterov-subgradient K=3")
    phase1_end = min(deadline, t_start + 250.0)
    a_work = a0.copy()
    a_work = a_work / a_work.sum()
    a_opt, sc_opt = _nesterov_opt(a_work, phase1_end, max_iter=100000, K=3, top_power=30)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Nesterov K=3: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Nesterov with K=1 (pure max chase) ===
    print("Phase 2: Nesterov-subgradient K=1")
    phase2_end = min(deadline, t_start + 450.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    a_opt, sc_opt = _nesterov_opt(a_work, phase2_end, max_iter=100000, K=1, top_power=1)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Nesterov K=1: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam fine-tuning ===
    print("Phase 3: Adam fine-tuning")
    phase3_end = min(deadline, t_start + 700.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    a_opt, sc_opt = _adam_finetune(a_work, phase3_end, lr_init=0.008, max_iter=200000)
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Try n=8192 ===
    print("Phase 4: Interpolate to n=8192")
    phase4_end = min(deadline, t_start + 1050.0)
    n_8192 = 8192
    a_8192 = _interp_profile(best_seq, n_8192)
    sc_8192 = _score(a_8192)
    print(f"  Interpolated n=8192: score={sc_8192:.10f}")

    if sc_8192 < best_score + 0.05:  # Only worthwhile if competitive
        a_opt8, sc_opt8 = _nesterov_opt(a_8192, min(phase4_end - 60, t_start + 900.0),
                                         max_iter=50000, K=3, top_power=20)
        sc_opt8_full = _score(a_opt8)
        if sc_opt8_full < best_score:
            best_score = sc_opt8_full
            best_seq = a_opt8.copy()
            print(f"  8192 Nesterov: score={sc_opt8_full:.10f} t={time.monotonic()-t_start:.1f}s")

        if time.monotonic() < phase4_end - 30:
            a_opt8, sc_opt8 = _adam_finetune(a_opt8, phase4_end - 10, lr_init=0.005, max_iter=100000)
            sc_opt8_full = _score(a_opt8)
            if sc_opt8_full < best_score:
                best_score = sc_opt8_full
                best_seq = a_opt8.copy()
                print(f"  8192 Adam: score={sc_opt8_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-restart diversification ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1400.0)
    n_best = len(best_seq)
    a_best = best_seq.copy()
    a_best = a_best / a_best.sum()

    n_restarts = 6
    per_restart = (phase5_end - time.monotonic() - 20) / max(1, n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > phase5_end - 10:
            break

        r = rng.random()
        if r < 0.3:
            p_start = _perturb_log(a_best, 0.01 + 0.005 * i, rng)
        elif r < 0.6:
            p_start = _perturb_log(a_best, 0.03 + 0.01 * i, rng)
        else:
            # Mix with shifted version
            shift = int(rng.integers(1, max(2, n_best // 1024)))
            a_shift = np.roll(a_best, shift).copy()
            a_shift = a_shift / a_shift.sum()
            mix = 0.2 + 0.1 * i
            p_start = mix * a_shift + (1 - mix) * a_best
            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

        # Quick Nesterov then Adam
        t_sub = time.monotonic() + max(5.0, per_restart * 0.4)
        p_opt, _ = _nesterov_opt(p_start, t_sub, max_iter=20000, K=2, top_power=15)
        p_opt = p_opt / p_opt.sum()

        if time.monotonic() < phase5_end - 15:
            t_sub2 = phase5_end - time.monotonic() - 10
            p_opt, _ = _adam_finetune(p_opt, t_sub2, lr_init=0.005, max_iter=50000)
            sc_r = _score(p_opt)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_opt.copy()
                n_best = len(best_seq)
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Coordinate descent polish ===
    print("Phase 6: Coordinate descent")
    phase6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase6_end - 20:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        a_cd, sc_cd = _coord_descent_fast(a_work, phase6_end - 10, n_sweeps=2)
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 7: Final ultra-fine Adam ===
    print("Phase 7: Final polish")
    phase7_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase7_end - 15:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        a_pol, sc_pol = _adam_finetune(a_work, phase7_end - 5, lr_init=0.002, max_iter=150000)
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Final Adam: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # One more Nesterov pass
        if time.monotonic() < phase7_end - 10:
            a_work = best_seq.copy()
            a_work = a_work / a_work.sum()
            a_pol2, sc_pol2 = _nesterov_opt(a_work, phase7_end - 5, max_iter=30000, K=2, top_power=50)
            sc_pol2_full = _score(a_pol2)
            if sc_pol2_full < best_score:
                best_score = sc_pol2_full
                best_seq = a_pol2.copy()
                print(f"  Final Nesterov: score={sc_pol2_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256, dtype=np.float64) / 256
            s = final.sum()

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(512, dtype=np.float64) / 512
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
