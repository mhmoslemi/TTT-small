"""
Peak-shaving optimizer using mirror descent (multiplicative weights) with momentum,
adaptive temperature schedule, FFT-based gradient computation, and coordinate descent polish.
Targets sub-1.5027 score for non-negative step functions of length 4096/8192.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _gradient_from_weights(a, w):
    """Compute grad[k] = 2 * sum_j w[j] * a[j-k] via FFT."""
    n = len(a)
    m = len(w)
    N = _next_pow2(n + m - 1)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, N)
    fa_rev = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(fw * fa_rev, N)
    result = conv[n - 1:2 * n - 1]
    return 2.0 * result


def _compute_weights(c, K, T):
    """Compute softmax weights over top-K peaks of c."""
    n_c = len(c)
    if K >= n_c:
        shift = (c - c.max()) / T
        np.clip(shift, -500, 500, out=shift)
        w = np.exp(shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(n_c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        return w
    else:
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        shift = (top_vals - top_vals.max()) / T
        np.clip(shift, -500, 500, out=shift)
        w_top = np.exp(shift)
        w_sum = w_top.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(n_c)
            w[int(np.argmax(c))] = 1.0
            return w
        w_top = w_top / w_sum
        w = np.zeros(n_c)
        w[top_idx] = w_top
        return w


def _mirror_descent(a, deadline, T_start, T_end, K, eta_start, momentum,
                    n_steps_per_T, min_improvement=1e-12):
    """Mirror descent with multiplicative weights and momentum."""
    n = len(a)
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    vel = np.zeros(n)
    lr = eta_start
    total_steps = max_steps_per_T = n_steps_per_T

    n_log = 20
    T_schedule = np.geomspace(T_start, T_end, n_log) if T_start > T_end else np.linspace(T_start, T_end, n_log)

    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        for step in range(total_steps):
            if time.monotonic() > deadline:
                break
            c = _convolve(a_curr, a_curr)
            w = _compute_weights(c, K, T)
            grad = _gradient_from_weights(a_curr, w)
            g_max = float(np.max(np.abs(grad)))
            if g_max < 1e-30:
                break
            grad_norm = grad / g_max

            vel = momentum * vel + (1.0 - momentum) * grad_norm
            effective_lr = lr / (1.0 + np.abs(vel).mean())
            np.clip(-effective_lr * vel, -100, 100, out=vel)
            a_new = a_curr * np.exp(-effective_lr * vel)
            a_sum = a_new.sum()
            if a_sum < 1e-300 or not np.isfinite(a_sum):
                a_new = a_curr + effective_lr * 0.01 * (np.random.rand(n) - 0.5)
                a_new = np.maximum(a_new, 0)
                a_sum = a_new.sum()
                if a_sum < 1e-300:
                    break
                a_new = a_new / a_sum
            else:
                a_new = a_new / a_sum

            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_sc - min_improvement:
                best_sc = sc
                best_a = a_new.copy()
            a_curr = a_new
            if step % 500 == 0:
                lr *= 0.97

    return best_a, best_sc


def _fista_smoothed(a_init, deadline, T, K, max_iter):
    """FISTA with smoothed max objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    L_est = 4.0 / T * 2.0 * n
    step = 1.0 / L_est

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        w = _compute_weights(c_y, K, T)
        grad = _gradient_from_weights(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0:
            sc = _score_norm(a_next)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _coord_descent_1d(a, deadline, max_rounds=100, n_try=4096):
    """Coordinate descent trying small multiplicative perturbations per coordinate."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(999)

    factors = [0.95, 0.98, 1.02, 1.05, 0.90, 1.10]

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > n_try:
            indices = rng.choice(n, size=n_try, replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old = a[j]
            if old < 1e-15:
                continue
            for f in factors:
                a[j] = old * f
                s = a.sum()
                if s < 1e-30:
                    a[j] = old
                    continue
                sc = _score_norm(a / s)
                if np.isfinite(sc) and sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = (a / s).copy()
                    improved = True
                    old = a[j]
                    break
                a[j] = old

        if not improved and r > 10:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    from math import exp
    e = np.exp(z - z.max())
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):

    # === Phase 3: Fine mirror descent (very small T, top few peaks) ===
    print("Phase 3: Fine mirror descent")
    phase3_end = min(deadline, t_start + 1100.0)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()
    c_w = _convolve(a_work, a_work)
    max_cw = float(c_w.max())

    T3_end = 0.00001 * max_cw
