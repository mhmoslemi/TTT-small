"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with fine
beta-continuation, symmetrization, active-set refinement, and coordinate descent.
Key improvements: finer beta schedule, symmetry exploitation, targeted active-set
optimization, and multi-n exploration with resampling.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active_set(a, c, n, tol_frac):
    """Gradient using active set (lags near max)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _symmetrize(a):
    """Symmetrize: average with reverse."""
    return 0.5 * (a + a[::-1])


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "fejer":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = a * a
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.25
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif kind == "parabolic":
        t = np.linspace(0, 1, n)
        a = 1 - (2 * t - 1) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _run_lbfgs_fine(a_init, deadline, min_time_remaining, beta_multiplier=1.0, max_beta_steps=25):
    """L-BFGS with fine-grained beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta: start based on gap structure
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(20, len(c_sorted) - 1)]
    if gap < 1e-15:
        gap = max_c * 0.05
    base_beta = max(gap * 0.5, max_c * 0.02) * beta_multiplier

    # Fine-grained geometric schedule
    ratio = 0.75
    betas = []
    b = base_beta
    for _ in range(max_beta_steps):
        betas.append(b)
        b *= ratio
        if b < max_c * 1e-6:
            break

    current_a = a_init.copy()
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale max_iter based on remaining time and beta index
        # Earlier betas get more iterations (coarse shape), later get fewer (fine polish)
        frac = 1.0 / (1.0 + 0.05 * i)
        est_time = 0.003 * max(1.0, n / 2048.0)
        max_iter = int(min(3000, max(20, remaining * frac / (est_time * 15))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            sc_cur = _score(current_a)
            if sc_new < sc_cur:
                current_a = a_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _active_set_refine(a_init, deadline, max_iter=2000):
    """Refine using active-set gradient descent with line search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.0005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True

        # Gradient of the active-set objective
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = a * (h - W)  # gradient on simplex (before normalization)

        # Project to tangent space
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break

        # Backtracking line search
        alpha = 0.1 / (1.0 + 0.01 * it)
        improved = False
        for ls in range(20):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            f_old = c_max
            if f_try < f_old - 1e-13 * alpha * gnorm:
                a = a_try
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try smaller step
            alpha = 1e-4 / (1.0 + 0.01 * it)
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 1e-15:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < c_max:
                    a = a_try

        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=100):
    """Coordinate descent: try adjusting pairs of elements to reduce max convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        c_max = float(c[k_max])

        # Find which elements contribute most to c[k_max]
        # c[k_max] = sum_i a_i * a[k_max - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        # Elements with highest contribution to the peak
        top_idx = np.argsort(contrib)[::-1][:min(20, n)]

        # Try moving mass from high-contributing elements
        for idx in top_idx:
            if time.monotonic() > deadline:
                break
            # Try reducing a[idx] slightly and adding to a random other element
            for target in np.random.choice(n, size=min(5, n), replace=False):
                if target == idx:
                    continue
                delta = 0.01 * a[idx]
                a_try = a.copy()
                a_try[idx] -= delta
                a_try[target] += delta
                if a_try[idx] < 0:
                    continue
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                if c_try.max() < c_max - 1e-12:
                    a = a_try
                    c_max = float(c_try.max())
                    c = c_try
                    break

        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _resample(a, n_new):
    """Resample sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return np.ones(n_new) / n_new
    return a_new / s

        parent = _make_profile(n_parent, "raised_cos")
        print(f"  No parent, using raised_cos n={n_parent}")
        update_best(parent, _score(parent), "init")

    n_main = len(parent)

    # --- Phase 1: Main L-BFGS from parent (most time) ---
    print(f"\n[Phase 1] Main L-BFGS (n={n_main}) t={time.monotonic() - t_start:.1f}s")
    time_remaining_for_phase1 = deadline - 600.0 - time.monotonic()
    if time_remaining_for_phase1 > 60:
        a_main = _run_lbfgs_fine(parent, deadline, min_time_remaining=600.0,
                                  beta_multiplier=1.0, max_beta_steps=30)
        update_best(a_main, _score(a_main), "lbfgs_main")

        # Symmetrize result and check
        a_main_sym = _symmetrize(a_main)
        a_main_sym = a_main_sym / a_main_sym.sum()
        sc_sym = _score(a_main_sym)
        sc_main = _score(a_main)
        if sc_sym < sc_main:
            update_best(a_main_sym, sc_sym, "lbfgs_main_sym")
            a_main = a_main_sym

    # --- Phase 2: Symmetric-constrained optimization ---
    print(f"\n[Phase 2] Symmetric L-BFGS t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_sym = _run_symmetric_lbfgs(best_seq, deadline, min_time_remaining=400.0)
        update_best(a_sym, _score(a_sym), "sym_lbfgs")

    # --- Phase 3: Active-set refinement ---
    print(f"\n[Phase 3] Active-set refinement t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_ar, sc_ar = _active_set_refine(best_seq, min(deadline - 200, time.monotonic() + 150),
                                          max_iter=1000)
        update_best(a_ar, sc_ar, "active_set")

        # Re-run L-BFGS from improved point
        if sc_ar < best_score and time.monotonic() < deadline - 150:
            a_ar2 = _run_lbfgs_fine(a_ar, deadline, min_time_remaining=150.0,
                                     beta_multiplier=0.3, max_beta_steps=15)
            update_best(a_ar2, _score(a_ar2), "lbfgs_post_active")

    # --- Phase 4: Multi-n exploration ---
    print(f"\n[Phase 4] Multi-n exploration t={time.monotonic() - t_start:.1f}s")
    n_candidates = [n_main]
    # Try nearby power-of-2 and other sizes
    for n_try in [3072, 4096, 5120, 6000, 2048]:
        if n_try != n_main and n_try not in n_candidates:
            n_candidates.append(n_try)
    n_candidates.sort(abs(lambda x: x - n_main))
    n_candidates = n_candidates[:4]  # Limit to 4 n values

    for n_try in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if n_try == n_main:
            continue  # Already explored

        if best_seq is not None:
            a_init = _resample(best_seq, n_try)
        else:
            a_init = _make_profile(n_try, "raised_cos")

        sc_init = _score(a_init)
        a_opt = _run_lbfgs_fine(a_init, deadline, min_time_remaining=80.0,
                                 beta_multiplier=0.5, max_beta_steps=15)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick active-set
        if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
            a_opt2, sc_opt2 = _active_set_refine(a_opt, min(deadline - 30, time.monotonic() + 40),
                                                  max_iter=300)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_active")

    # --- Phase 5: Diverse restarts ---
    print(f"\n[Phase 5] Diverse restarts t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for i in range(3):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=6)
            else:
                # Block perturbation
                z = np.log(np.maximum(best_seq, 1e-30))
                z = z - z.mean()
                n = len(z)
                bs = n // 8
                idx = rng.integers(0, n - bs)
                z[idx:idx + bs] += 0.05 * rng.standard_normal(bs)
                p_start = _softmax(z)

            a_r = _run_lbfgs_fine(p_start, deadline, min_time_remaining=20.0,
                                   beta_multiplier=0.3, max_beta_steps=10)
            update_best(a_r, _score(a_r), f"restart_{i}")

    # --- Phase 6: Final coordinate descent + polish ---
    print(f"\n[Phase 6] Final polish t={time.monotonic() - t_start:.1f}s")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Coordinate descent
        a_cd, sc_cd = _coordinate_descent(best_seq, min(deadline - 5, time.monotonic() + 30),
                                           max_rounds=50)
        update_best(a_cd, sc_cd, "coord_descent")

        # Final fine L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_best = _full_convolve(best_seq, best_seq)
            mb = float(c_best.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_final = mb * 5e-5
            remaining = deadline - 3 - time.monotonic()
            if remaining > 3:
                max_iter = min(1000, max(30, int(remaining / 0.05)))
                try:
                    result = minimize(
                        lambda z_local: _smoobj(z_local, beta_final, n),
                        z,
                        jac=lambda z_local: _smograd(z_local, beta_final, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                    )
                    a_fin = _softmax(result.x)
                    update_best(a_fin, _score(a_fin), "final_lbfgs")
                    # Symmetrize
                    a_fin_s = _symmetrize(a_fin)
                    a_fin_s = a_fin_s / a_fin_s.sum()
                    update_best(a_fin_s, _score(a_fin_s), "final_lbfgs_sym")
                except Exception:
                    pass

        # Very final active set
        if best_seq is not None and time.monotonic() < deadline - 2:
            a_f, sc_f = _active_set_refine(best_seq, deadline - 2, max_iter=200)
            update_best(a_f, sc_f, "final_active")

    # --- Final output ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = float(final.sum())
        sc_final = _score(final)
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            if np.isfinite(eval_score):
                print(f"FINAL: score={sc_final:.10f}, eval={eval_score:.10f}, "
                      f"n={len(final)}, time={time.monotonic() - t_start:.1f}s")
            else:
                print(f"WARNING: evaluate_sequence returned inf! Using fallback.")
                final = _make_profile(256, "uniform")
                sc_final = _score(final)
        except Exception:
            pass
        print(f"Returns n={len(final)}, score={sc_final:.10f}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
