    # OK I think the cleanest way is:
    # grad[i] = sum_k w[k] * (a[k-i] + a[2i-k])  [a[j]=0 outside [0,n-1]]
    # 
    # Term 1: T1[i] = sum_k w[k]*a[k-i] = conv(w_pad, a)[i] where w_pad is w placed at position 0 in a length (2n-1)+n-1 array... 
    # 
    # You know what, let me just use the formula that the previous code uses and verify it works:
    # a_rev = a[::-1]
    # wc = _full_convolve(w, a_rev)  # length (2n-1) + n - 1 = 3n-2
    # h = wc[n-1:2n-1]  # length n
    # 
    # Let me verify for a=[a0,a1], w=[w0,w1,w2]:
    # a_rev = [a1, a0]
    # conv(w, a_rev) = [w0*a1, w0*a0+w1*a1, w1*a0+w2*a1, w2*a0]
    # h = wc[1:3] = [w0*a0+w1*a1, w1*a0+w2*a1]
    # 
    # Expected: grad[0] = 2*(w0*a0 + w1*a1), grad[1] = 2*(w1*a0 + w2*a1)
    # So h should be half the gradient: h[0] = w0*a0+w1*a1 = grad[0]/2 ✓, h[1] = w1*a0+w2*a1 = grad[1]/2 ✓
    # 
    # Great, so the formula is: grad = 2 * h where h = conv(w, a_rev)[n-1:2n-1]
    # And the full objective gradient (for 2*n*max_c / sum_a^2) is more complex due to the denominator.
    #
    # But for the z-space (log-space) optimization, we use the chain rule:
    # d/dz_j [2*n*f(a)] where a = softmax(z)
    # = 2*n * sum_i [df/da_i] * [da_i/dz_j]
    # da_i/dz_j = a_i*(delta_ij - a_j)
    # So grad_z[j] = 2*n * a_j * (sum_i df/da_i * a_i - df/da_j)
    # = 2*n * a_j * (a^T * grad_f - grad_f[j])
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    grad_f = 2.0 * h  # gradient of max_c w.r.t. a (approximately, using soft weights)
    # Actually grad of sum_k w[k]*c[k] w.r.t. a is 2*h as verified above
    # But we want the gradient of the smoothed max, which is sum_k w[k]*c[k] where w = softmax((c-c_max)/beta)
    # The gradient w.r.t. a is 2*h (as computed)
    # Then the gradient w.r.t. z is:
    # grad_z[j] = 2*n * a[j] * (sum_i (2*h[i])*a[i] / sum_a^2 - 2*(2*n*h[j]*sum_a - 2*n*c_soft*2*a[j])/sum_a^2)
    # This is getting complicated. Let me just use the simpler approach:
    # Since a is already normalized (softmax gives sum=1), sum_a = 1
    # And the objective is 2*n*soft_max_c (since sum_a=1)
    # dobjective/dz_j = 2*n * sum_i (2*h[i]) * a[j]*(delta_ij - a[i])
    # = 2*n * 2*a[j] * (h[j] - sum_i h[i]*a[i])
    W = float(np.dot(h, a))
    return 4.0 * n * a * (h - W)


def _fista_smoothed(z0, beta, n, max_iter, tol=1e-12):
    """FISTA (accelerated projected gradient) for the smoothed problem in z-space."""
    z = z0.copy()
    z_prev = z0.copy()
    t = 1.0
    best_f = _log_softmax_obj(z, beta, n)
    best_z = z.copy()
    
    # Compute Lipschitz constant (rough estimate)
    a0 = _softmax(z)
    c0 = _full_convolve(a0, a0)
    # L ~ 2*n*max(c)/beta * something... use backtracking
    L = 2.0 * n * c0.max() / (beta * a0.min() + 1e-30)
    L = max(L, 1.0)
    
    for it in range(max_iter):
        f = _log_softmax_obj(z, beta, n)
        if f < best_f:
            best_f = f
            best_z = z.copy()
        
        g = _log_softmax_grad(z, beta, n)
        
        # Gradient step
        step = 1.0 / L
        z_new = z - step * g
        z_new = z_new - z_new.mean()  # recenter for gauge invariance
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = z_new + ((t - 1.0) / t_new) * (z_new - z_prev)
        z = z - z.mean()
        z_prev = z_new
        t = t_new
        
        # Backtracking (simple: every 10 steps, re-estimate L)
        if (it + 1) % 10 == 0:
            a_check = _softmax(z)
            c_check = _full_convolve(a_check, a_check)
            L = max(1.0, 2.0 * n * c_check.max() / (beta * a_check.min() + 1e-30))
    
    return best_z, best_f


def _power_sum_smooth(z, p, n):
    """Power-mean smoothing: (sum c^p / (2n-1))^(1/p) approximates max as p->inf."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return 2.0 * n * c_max
    # Use power sum: (1/(2n-1) * sum (c/c_max)^p)^(1/p) * c_max
    # Simpler: just use c_max + (1/p) * sum((c/c_max - 1)^p) for c < c_max... 
    # Actually let's just use the log-sum-exp which is equivalent
    # For variety, use a different parameterization
    return _log_softmax_obj(z, 1.0/p, n)  # beta = 1/p, larger p = smaller beta = sharper


def _convolution_surgery(a, deadline, n_top=5, max_rounds=200):
    """
    Directly reduce the top convolution peaks by adjusting the sequence.
    For each peak lag k, find the pairs (i, k-i) that contribute most,
    and reduce the smaller of the pair.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = best_a.copy()
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Get top-N lags
        top_indices = np.argsort(c)[-n_top:][::-1]
        
        improved = False
        for k in top_indices:
            if time.monotonic() > deadline:
                break
            
            # Contribution of each pair (i, k-i) to c[k]
            # c[k] = sum_i a[i]*a[k-i] for valid i
            i_vals = np.arange(max(0, k - n + 1), min(n, k + 1))
            j_vals = k - i_vals
            contribs = a[i_vals] * a[j_vals]
            total = contribs.sum()
            
            if total < 1e-30:
                continue
            
            # Reduce the largest contributor
            idx_max = np.argmax(contribs)
            i_idx = int(i_vals[idx_max])
            j_idx = int(j_vals[idx_max])
            
            # Reduce the smaller of the two (or split the reduction)
            if a[i_idx] <= a[j_idx]:
                target = i_idx
                other = j_idx
            else:
                target = j_idx
                other = i_idx
            
            # Reduce target proportionally
            for reduce_frac in [0.1, 0.05, 0.02, 0.01]:
                a_try = best_a.copy()
                a_try[target] *= (1.0 - reduce_frac)
                a_try[other] *= (1.0 - reduce_frac * 0.5)  # reduce both a bit
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                sc = _score(a_try)
                if sc < best_sc - 1e-14:
                    best_a = a_try.copy()
                    best_sc = sc
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_sc


def _structured_init(n, kind="random", rng=None):
    """Create structured initial sequences."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "random":
        a = rng.uniform(0, 1, n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_bumps":
        t = np.linspace(0, 1, n)
        a = np.exp(-40 * (t - 0.3)**2) + np.exp(-40 * (t - 0.7)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bandlimited":
        # Low-pass filtered random
        a = rng.uniform(0, 1, n)
        A = np.fft.rfft(a)
        # Zero out high frequencies
        cutoff = n // 4
        A[cutoff:] = 0
        a = np.fft.irfft(A, n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_window":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = (1 - t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _iterative_flattening(a, deadline, max_rounds=500, n_active=3):
    """
    Iteratively identify the top convolution lags and use a local QP-like
    update to reduce them. This is a projected subgradient method with
    momentum.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    momentum = np.zeros(n)
    beta_mom = 0.9
    step_size = 0.1 / np.sqrt(n)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = best_a
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Active set: lags within 1% of max
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        
        # Compute gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_f = 2.0 * h  # gradient of weighted sum of c[k] w.r.t. a
        
        # Project gradient to account for sum constraint
        grad = grad_f - grad_f.mean()
        
        # Momentum update
        momentum = beta_mom * momentum + (1 - beta_mom) * grad
        direction = momentum - momentum.mean()
        
        # Adaptive step size (barrier backtracking)
        for ls in range(30):
            s = step_size * (0.8 ** ls)
            a_try = a - s * direction
            a_try = np.maximum(a_try, 0.0)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try = a_try / ss
            sc_try = _score(a_try)
            if sc_try < _score(a) - 1e-15 or ls == 29:
                break
        
        if sc_try < best_sc - 1e-14:
            best_a = a_try.copy()
            best_sc = sc_try
            step_size = min(step_size * 1.01, 0.5)
        else:
            step_size *= 0.85
        
        if step_size < 1e-12:
            break
    
    return best_a, best_sc


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _log_softmax_obj(z, beta, n)
    
    def jac(z):
        return _log_softmax_grad(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _multi_beta_lbfgs(a_start, deadline, min_time_remaining=60.0):
    """L-BFGS with decreasing beta (continuation method)."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.02, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002, 0.00001,
                              0.000005, 0.000002])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate iterations possible
        est_iter_time = 0.002 * (n / 2048.0) ** 1.3
        max_iter = min(10000, max(50, int(remaining / (est_iter_time * 10))))
        
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
    
    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: L-BFGS continuation + FISTA + convolution surgery
    + iterative flattening + multi-length exploration + multi-restart.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score_val = np.inf
    
    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _structured_init(4096, "random", rng)
        print(f"No parent. Random n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_random")
    
    # === Phase 1: L-BFGS continuation on parent ===
    print(f"\n=== Phase 1: L-BFGS continuation (t={time.monotonic()-t_start:.1f}s) ===")
    a_p1 = _multi_beta_lbfgs(best_seq, deadline, min_time_remaining=400.0)
    try_update(a_p1, "phase1_lbfgs")
    
    # === Phase 2: Iterative flattening with momentum ===
    print(f"\n=== Phase 2: Iterative flattening (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_sg, sc_sg = _iterative_flattening(best_seq, deadline - 300,
                                            max_rounds=2000, n_active=3)
        try_update(a_sg, "phase2_flatten")
    
    # === Phase 3: Convolution surgery ===
    print(f"\n=== Phase 3: Convolution surgery (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_surg, sc_surg = _convolution_surgery(best_seq, deadline - 200,
                                               n_top=5, max_rounds=500)
        try_update(a_surg, "phase3_surgery")
    
    # === Phase 4: Fine L-BFGS ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7])
        
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(5000, max(100, int(remaining / 0.003)))
            a_new = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")
    
    # === Phase 5: Multi-restart with diverse initializations ===
    print(f"\n=== Phase 5: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        n = len(best_seq)
        restart_configs = [
            # (name, perturbation_type, params)
            ("gauss_001", "gauss", 0.01),
            ("gauss_003", "gauss", 0.03),
            ("gauss_008", "gauss", 0.08),
            ("gauss_015", "gauss", 0.15),
            ("fourier_low", "fourier_low", 3),
            ("fourier_high", "fourier_high", 10),
            ("block", "block", 6),
            ("random_structured", "structured", 0),
        ]
        
        for name, ptype, param in restart_configs:
            if time.monotonic() > deadline - 80:
                break
            
            if ptype == "gauss":
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + param * rng.standard_normal(n)
                z = z - z.mean()
                p_start = _softmax(z)
            elif ptype == "fourier_low":
                z = np.log(np.maximum(best_seq, 1e-15))
                t = np.arange(n)
                for _ in range(param):
                    freq = int(rng.integers(1, max(2, n // 16)))
                    amp = 0.005 * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2.0 * np.pi)
                    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
                z = z - z.mean()
                p_start = _softmax(z)
            elif ptype == "fourier_high":
                z = np.log(np.maximum(best_seq, 1e-15))
                t = np.arange(n)
                for _ in range(param):
                    freq = int(rng.integers(n // 16, max(n // 16 + 2, n // 2)))
                    amp = 0.003 * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2.0 * np.pi)
                    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
                z = z - z.mean()
                p_start = _softmax(z)
            elif ptype == "block":
                z = np.log(np.maximum(best_seq, 1e-15))
                block_size = max(4, n // param)
                for _ in range(param):
                    start = int(rng.integers(0, max(1, n - block_size)))
                    end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
                    z[start:end] += rng.uniform(-0.02, 0.02)
                z = z - z.mean()
                p_start = _softmax(z)
            else:  # structured
                p_start = _structured_init(n, "random", rng)
            
            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")
            
            # Quick L-BFGS
            a_r = _multi_beta_lbfgs(p_start, deadline, min_time_remaining=40.0)
            try_update(a_r, f"restart_{name}")
            
            # Quick flattening
            if time.monotonic() < deadline - 35 and _score(a_r) < best_score_val + 0.01:
                a_r2, _ = _iterative_flattening(a_r, deadline - 35,
                                                 max_rounds=500, n_active=3)
                try_update(a_r2, f"restart_{name}_flat")
    
    # === Phase 6: Try larger lengths ===
    print(f"\n=== Phase 6: Larger lengths (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        current_n = len(best_seq)
        for target_n in [current_n * 2, current_n + 1024, 8192, 16384]:
            if time.monotonic() > deadline - 60:
                break
            if target_n <= current_n and target_n != 8192 and target_n != 16384:
                continue
            if target_n == current_n:
                continue
            print(f"  Trying n={target_n}...")
            # Interpolate
            t_old = np.linspace(0, 1, current_n)
            t_new = np.linspace(0, 1, target_n)
            a_interp = np.interp(t_new, t_old, best_seq)
            a_interp = np.maximum(a_interp, 1e-10)
            a_interp = a_interp / a_interp.sum()
            
            a_try = _multi_beta_lbfgs(a_interp, deadline, min_time_remaining=30.0)
            try_update(a_try, f"n_{target_n}")
    
    # === Phase 7: Final aggressive polish ===
    print(f"\n=== Phase 7: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(5):
            if time.monotonic() > deadline - 5:
                break
            
            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-7
            a_f = _lbfgs_run(best_seq, beta, n, 3000)
            try_update(a_f, f"final_lbfgs_{rnd}")
            
            # Flattening
            if time.monotonic() < deadline - 5:
                a_s, _ = _iterative_flattening(best_seq, deadline - 5,
                                               max_rounds=2000, n_active=5)
                try_update(a_s, f"final_flat_{rnd}")
            
            # Surgery
            if time.monotonic() < deadline - 5:
                a_surg, _ = _convolution_surgery(best_seq, deadline - 5,
                                                 n_top=8, max_rounds=200)
                try_update(a_surg, f"final_surg_{rnd}")
    
    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp continuation, momentum-based iterative flattening,
convolution surgery, active-set simulated annealing, multi-length exploration,
and diverse restarts to escape local minima.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(h, a))
    return 4.0 * n * a * (h - W)


def _active_grad(a, c, n, tol_frac=1e-4, n_top=None):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    if n_top is not None and n_top < len(c):
        idx = np.argsort(c)[-n_top:]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    else:
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _lse_obj(z, beta, n)

    def jac(z):
        return _lse_grad(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _multi_beta_lbfgs(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005,
                              0.002, 0.001, 0.0005, 0.0002,
                              0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
                              1e-6, 5e-7, 2e-7, 1e-7])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:

        sc = _score(a_new)

def _momentum_flatten(a_init, deadline, max_rounds=5000, n_top=3,
                      beta_mom=0.9, step_init=0.1):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    momentum = np.zeros(n)
    step_size = step_init / np.sqrt(n)
    stall = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a = best_a
        c = _full_convolve(a, a)
        grad = _active_grad(a, c, n, n_top=n_top)
        grad = grad - grad.mean()

        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
        direction = momentum - momentum.mean()

        for ls in range(40):
            s = step_size * (0.75 ** ls)
            a_try = a - s * direction
            a_try = np.maximum(a_try, 0.0)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try = a_try / ss
            sc_try = _score(a_try)
            if sc_try < _score(a) - 1e-16 or ls == 39:
                break

        if sc_try < best_sc - 1e-13:
            best_a = a_try.copy()
            best_sc = sc_try
            step_size = min(step_size * 1.008, 1.0)
            stall = 0
        else:
            stall += 1
            step_size *= 0.82

        if stall > 300:
            break
        if step_size < 1e-13:
            break

    return best_a, best_sc


def _conv_surgery(a_init, deadline, n_top=5, max_rounds=1000):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        c_max = c.max()
        top_idx = np.argsort(c)[-n_top:][::-1]

        improved = False
        for k in top_idx:
            if time.monotonic() > deadline:
                break

            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            i_vals = np.arange(i_lo, i_hi + 1)
            j_vals = k - i_vals
            contribs = best_a[i_vals] * best_a[j_vals]
            total_c = contribs.sum()
            if total_c < 1e-30:
                continue

            top_pairs = np.argsort(contribs)[-3:][::-1]
            for pi in top_pairs:
                ii = int(i_vals[pi])
                jj = int(j_vals[pi])
                for frac in [0.15, 0.08, 0.04, 0.02, 0.01]:
                    a_try = best_a.copy()
                    if best_a[ii] <= best_a[jj]:
                        a_try[ii] *= (1.0 - frac)
                        a_try[jj] *= (1.0 - frac * 0.4)
                    else:
                        a_try[jj] *= (1.0 - frac)
                        a_try[ii] *= (1.0 - frac * 0.4)
                    a_try = np.maximum(a_try, 0.0)
                    ss = a_try.sum()
                    if ss < 1e-10:
                        continue
                    a_try = a_try / ss
                    sc = _score(a_try)
                    if sc < best_sc - 1e-13:
                        best_a = a_try.copy()
                        best_sc = sc
                        improved = True
                        break

        if not improved:
            break

    return best_a, best_sc


def _anneal_active(a_init, deadline, max_rounds=3000, n_candidates=3):
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    step_size = 0.05 / np.sqrt(n)
    momentum = np.zeros(n)
    stall = 0
    rng_local = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a = best_a
        c = _full_convolve(a, a)
        c_max = c.max()

        # Randomly select which peaks to target (simulated annealing style)
        temperature = max(0.01, 0.5 * (1.0 - rnd / max_rounds))
        probs = c / c.sum()
        # Mix between "target highest" and "random"
        if temperature > 0.1:
            q = (1 - temperature) * probs + temperature / len(c)
        else:
            q = probs
        selected = rng_local.choice(len(c), size=n_candidates, replace=False, p=q)

        active = np.zeros(len(c), dtype=bool)
        active[selected] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / n_candidates

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        grad = grad - grad.mean()

        momentum = 0.85 * momentum + 0.15 * grad
        direction = momentum - momentum.mean()

        sc_a = _score(a)
        for ls in range(30):
            s = step_size * (0.8 ** ls)
            a_try = a - s * direction
            a_try = np.maximum(a_try, 0.0)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try = a_try / ss
            sc_try = _score(a_try)
            if sc_try < sc_a - 1e-16 or ls == 29:
                break

        if sc_try < best_sc - 1e-13:
            best_a = a_try.copy()
            best_sc = sc_try
            step_size = min(step_size * 1.01, 0.5)
            stall = 0
        else:
            stall += 1
            step_size *= 0.88

        if stall > 400:
            break
        if step_size < 1e-13:
            break

    return best_a, best_sc

        return np.ones(n) / n

        a = rng.uniform(0, 1, n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
