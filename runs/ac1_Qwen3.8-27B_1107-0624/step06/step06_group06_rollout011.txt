            # For a given x = a[j], the new convolution is:
            # c_new[k] = c_old[k] + contribution from changing a[j]
            # This is complex, so let's just try a few values
            
            # Try reducing a[j] (since reducing large elements tends to help)
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try: set a[j] to 0, to half, to 0.1
            best_local = a[j]
            best_local_sc = c_max
            
            for frac in [0.0, 0.3, 0.5, 0.7]:
                a[j] = old_val * frac
                c_try = _full_convolve(a, a)
                sc_try = float(c_try.max())
                if sc_try < best_local_sc:
                    best_local_sc = sc_try
                    best_local = a[j]
            a[j] = best_local
            
            if best_local != old_val:
                improved = True
        
        if not improved:
            break
    
    return a

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    """Subgradient method with adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active_set(z, a, c, n, tol_frac=0.003)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        step = lr / (1.0 + 0.001 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.9
                if patience > 50:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.5
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

def _run_lbfgs(a_init, deadline, min_time_remaining=120.0):
    """L-BFGS with adaptive beta-continuation (more stages, finer at end)."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Design beta schedule: geometric with more points at the low end
    c_sorted = np.sort(c_cur)[::-1]
    # Use the gap between top values to set initial beta
    gap = c_sorted[0] - c_sorted[min(5, len(c_sorted)-1)] if len(c_sorted) > 5 else c_sorted[0]
    base_beta = max(gap * 0.5, max_c * 0.05)
    
    # More aggressive schedule: start at 5x base, end at 0.0001x base
    # Use log-spaced betas for finer control
    n_betas = 16
    betas = base_beta * (10 ** np.linspace(0.6, -4.0, n_betas))
    betas = np.maximum(betas, max_c * 1e-6)
    
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        
        # Estimate time per iteration
        est_time_per_iter = 0.003 * max(1, n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 15))))
        
        def fun(z_local):
            return _smoobj(z_local, beta, n)
        
        def jac(z_local):
            return _smograd(z_local, beta, n)
        
        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass
    
    return current_a

def _shrinkage_reoptimize(a, deadline):
    """Zero out small elements, re-optimize."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    a_sorted_idx = np.argsort(a)
    
    # Try progressively larger fractions to zero out
    for frac in [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.20]:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
    
    return best_a, best_sc

def _perturb(a, sigma, rng):

    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

                z = z - z.mean()
        except Exception:
            pass

    return current_a


def _shrinkage_reoptimize(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = best_a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-11:
            best_sc = sc_try
            best_a = a_try.copy()

    return best_a, best_sc


def _coord_descent_polish(a, deadline, max_sweeps):
    n = len(a)
    a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        idx = rng.permutation(n)
        for j in idx:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            c = _full_convolve(a, a)
            c_max = float(c.max())

            best_local = old_val
            best_local_max = c_max
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8]:
                a[j] = old_val * frac
                c_try = _full_convolve(a, a)
                m_try = float(c_try.max())
                if m_try < best_local_max - 1e-15:
                    best_local_max = m_try
                    best_local = a[j]
            a[j] = best_local
            if best_local != old_val:
                improved = True

        if not improved:
            break

    sc = _score(a)
    if sc < best_sc:
        best_sc = sc
    return a


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t_arr = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_blocks(a, rng, n_blocks):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, max(1, n - 2 * block_size))
        j = rng.integers(0, max(1, n - 2 * block_size))
        if abs(int(i) - int(j)) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _update_best(state, p, sc, label):
    if p is None:
        return
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return
    if sc is None:
        sc = _score(p)
    if np.isfinite(sc) and sc < state[1]:
        prev = state[1]
        state[0] = p.copy()
        state[1] = float(sc)
        print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] "
              f"improve={prev - sc:.2e} t={state[2] - state[3]:.1f}s")


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    state = [None, np.inf, t_start, t_start]
    # state[0] = best_seq, state[1] = best_score, state[2] = t_start_ref, state[3] = t_start

    # Load parent
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        _update_best(state, parent, _score(parent), "parent")
    else:
        n_parent = 4096
        parent = np.ones(n_parent) / n_parent
        print(f"No parent, using uniform n={n_parent}")
        _update_best(state, parent, _score(parent), "init")

    # Phase 1: Main L-BFGS with extended beta schedule
    print(f"\n[Phase 1] Main L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=450.0)
    sc_main = _score(a_main)
    _update_best(state, a_main, sc_main, "lbfgs_main")

    # Phase 2: Subgradient polish
    print(f"\n[Phase 2] Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if state[0] is not None and time.monotonic() < deadline - 350:
        sg_deadline = min(deadline - 350, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_polish(state[0], sg_deadline, 5000, 0.02, 0.002)
        _update_best(state, a_sg, sc_sg, "sg_polish")

    # Phase 3: Shrinkage + re-optimize
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if state[0] is not None and time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage_reoptimize(state[0], deadline - 250)
        _update_best(state, a_shr, sc_shr, "shrinkage")
        if sc_shr < state[1]:
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=180.0)
            _update_best(state, a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # Phase 4: Coordinate descent
    print(f"\n[Phase 4] Coordinate descent (t={time.monotonic() - t_start:.1f}s)")
    if state[0] is not None and time.monotonic() < deadline - 180:
        cd_deadline = min(deadline - 180, time.monotonic() + 80)
        a_cd = _coord_descent_polish(state[0], cd_deadline, 4)
        sc_cd = _score(a_cd)
        _update_best(state, a_cd, sc_cd, "coord_descent")

    # Phase 5: Diverse restarts
    print(f"\n[Phase 5] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if state[0] is not None and time.monotonic() < deadline - 80:
        for i in range(5):
            if time.monotonic() > deadline - 40:
                break
            if i == 0:
                p_start = _perturb_gaussian(state[0], 0.02, rng)
            elif i == 1:
                p_start = _perturb_gaussian(state[0], 0.08, rng)
            elif i == 2:
                p_start = _perturb_fourier(state[0], rng, 4)
            elif i == 3:
                p_start = _perturb_blocks(state[0], rng, 4)
            else:
                # Mix of perturbations
                p1 = _perturb_gaussian(state[0], 0.04, rng)
                p_start = _perturb_fourier(p1, rng, 2)

            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=40.0)
            sc_r = _score(a_r)
            _update_best(state, a_r, sc_r, f"restart_{i}")

            if sc_r < state[1] + 0.003 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _subgrad_polish(a_r, min(deadline - 10, time.monotonic() + 30),
                                              800, 0.01, 0.002)
                _update_best(state, a_r2, sc_r2, f"restart_{i}_sg")

    # Phase 6: Final fine L-BFGS + subgradient
    print(f"\n[Phase 6] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if state[0] is not None and time.monotonic() < deadline - 5:
        n = len(state[0])
        c_best = _full_convolve(state[0], state[0])
        mb = float(c_best.max())
        z = np.log(np.maximum(state[0], 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.00003
        remaining = deadline - 5 - time.monotonic()
        if remaining > 5:
            max_iter = min(4000, max(100, int(remaining / 0.02)))
            try:
                result = minimize(
                    _smoobj, z, args=(beta_final, n),
                    jac=_smograd,
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13, 'maxcor': 50}
                )
                a_fin = _softmax(result.x)
                _update_best(state, a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(state[0], deadline - 3, 600, 0.004, 0.001)
            _update_best(state, a_fin2, sc_fin2, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if state[0] is not None:
        final = np.clip(state[0], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
