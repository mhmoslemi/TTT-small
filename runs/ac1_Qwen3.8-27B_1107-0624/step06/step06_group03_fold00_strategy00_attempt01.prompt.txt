def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):

    c = _full_convolve(a, a)

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Compute gradient of <w, conv(a,a)> wrt a.
    grad_i = 2 * sum_{k=0}^{n-1} w_{i+k} * a_k
    This is 2 * cross_correlation(a, w)[i]
    """
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # cross_corr(a,w)[i] = sum_k a[k]*w[k+i] = ifft(conj(fft(a)) * fft(w))
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_softmax_weights(c, beta, k):
    """Compute softmax weights over only the top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    # Find top-k indices
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, c, w, n):
    """Compute gradient in log-space (z-space) for the normalized problem.
    a is on the simplex (sum=1).
    dF/dz_i = 2n * a_i * (grad_i - <grad, a>)
    where grad_i is the gradient of the smoothed max of conv(a,a) wrt a_i.
    """
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):

    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:

                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, c, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0005, current_lr * 0.97)

        # Reset momentum at phase boundaries
        m_buf *= 0.5
        v_buf *= 0.5

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:

def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent polishing.
    For each coordinate i, optimize a_i >= 0 (renormalizing the rest)
    using a 1D search that considers top constraints.
    """
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for i in order:
            if time.monotonic() > deadline:
                break
            # Current state
            c = _full_convolve(a, a)
            # For coordinate i, (a*a)_j = c_j - a_i*a_{j-i} + t*a_{j-i} (if 0<=j-i<n)
            # = c_j + (t - a_i) * a_{j-i}
            # So d_j = a_{j-i} for valid j, c_j_new = c_j - a_i * a_{j-i}
            # S0 = sum(a) - a_i = 1 - a_i (assuming normalized)
            
            a_i_old = a[i]
            if a_i_old < 1e-15:
                # Try setting to small value
                a[i] = 1e-4
                a = a / a.sum()
                continue

            # Compute d_j = a_{j-i} for valid j
            # d_j = a[(j-i) % n] if 0 <= j-i < n else 0
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            # c_j = (a*a)_j - a_i * d_j
            c_base = c - a_i_old * d

            # S0 = 1 - a_i_old (since sum a = 1)
            S0 = 1.0 - a_i_old

            # Find top-K j by c_base_j + d_j * t for t near a_i_old
            # Approximate: use c_base + d * a_i_old = c (current conv)
            # Find top-K positions
            idx_top = np.argsort(c)[-top_k_w:]
            
            # Evaluate F(t) = 2n * max_j (c_base[j] + d[j]*t) / (S0+t)^2 for candidate t values
            # Candidate t values: 0, a_i_old/2, a_i_old, 2*a_i_old, and values from 1D optimization
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5, a_i_old * 2.0,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2]
            
            # Also add critical points: for each active j, t* = S0 - 2*c_base[j]/d[j] (if d[j]!=0)
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 10.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)
                        t_candidates.append(t_star * 0.8)
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(10.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 10) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1

            if improved % 100 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    """Create structured initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        # Sum of a few cosines with different frequencies
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n
