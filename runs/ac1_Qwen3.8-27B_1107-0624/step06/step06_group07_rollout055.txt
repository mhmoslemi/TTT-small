    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _subgradient(a, c, K=10):
    """Compute subgradient of max(c) w.r.t. a using top K lags."""
    n = len(a)

    top_k = np.argsort(c)[-K:][::-1]

    h = wc[n-1:2*n-1]

    return 2.0 * (h - W)

    theta = (cssv[rho] - target_sum) / (rho + 1.0)

        ("chirp_3", _make_chirp(n, 3.0)),
        ("taper_0.7", _make_tapered(n, 0.7)),
        ("taper_1.5", _make_tapered(n, 1.5)),
        ("taper_3.0", _make_tapered(n, 3.0)),
        ("cos_2_03", _make_cosine(n, 2, 0.3)),
        ("cos_5_02", _make_cosine(n, 5, 0.2)),
    ]
    for name, a_init in inits:
        if time.monotonic() > deadline - 60:
            break
        sc_init = _score(a_init)
        if sc_init < state['best_score'] + 0.05:
            sub_dl4 = min(deadline - 50, time.monotonic() + 40)
            a_opt, _ = _nesterov_subgrad(a_init, sub_dl4, K_start=20, K_end=2,
                                          max_iter=15000, lr_init=0.04)
            _update_best(state, a_opt, f"restart_{name}")

    # === Phase 6: Final tight polishing cycles ===
    print(f"\n=== Phase 6: Final polish (t={time.monotonic()-t_start:.1f}s) ===")
    for cycle in range(5):
        if time.monotonic() > deadline - 10:
            break
        # Tight Nesterov
        sub_dl5 = min(deadline - 10, time.monotonic() + 60)
        a_t, _ = _nesterov_subgrad(state['best_seq'], sub_dl5, K_start=5, K_end=1,
                                    max_iter=20000, lr_init=0.02)
        _update_best(state, a_t, f"polish_ns{cycle}")
        # Tight z-space
        if time.monotonic() < deadline - 10:
            sub_dl6 = min(deadline - 10, time.monotonic() + 40)
            a_tz, _ = _nesterov_zspace(state['best_seq'], sub_dl6, K=2,
                                        max_iter=15000, lr_init=0.02)
            _update_best(state, a_tz, f"polish_z{cycle}")
        # Pair swap
        if time.monotonic() < deadline - 10:
            sub_dl7 = min(deadline - 10, time.monotonic() + 40)
            a_tp, _ = _pair_swap(state['best_seq'], sub_dl7, max_rounds=30000)
            _update_best(state, a_tp, f"polish_ps{cycle}")

    # Final validation
    print(f"\n{'=' * 60}")
    if state['best_seq'] is not None:
        final = np.clip(state['best_seq'], 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
