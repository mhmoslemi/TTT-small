import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(y, w):
    n = len(y)
    y_rev = y[::-1]
    wc = _convolve(w, y_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return grad

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        phase_iter = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            phase_iter += 1
            t += 1

            a = _softmax(z)
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if phase_iter % 3000 == 0:
                current_lr = max(0.0003, current_lr * 0.93)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=300):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(99999)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if n > 4096:
            indices = rng.choice(n, size=min(768, n), replace=False)
        elif n > 1024:
            indices = rng.choice(n, size=min(512, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = 0
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            s = a.sum()
            for factor in [0.95, 0.98, 1.02, 1.05]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                sc = _score_norm(a / a.sum())
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved += 1
                else:
                    a[j] = old_val
        a = a / a.sum()
        if improved == 0 and r > 30:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _blend_two(a, b, t, rng):
    return (1.0 - t) * a + t * b


def _gaussian_envelope(n, width):
    x = np.linspace(-3.0, 3.0, n)
    env = np.exp(-x**2 / (2 * width**2))
    return env / env.sum()


def _triangular_envelope(n):
    x = np.arange(n)
    env = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0)
    env = np.maximum(env, 0.0)
    return env / env.sum()


def _power_envelope(n, alpha):
    x = np.arange(n, dtype=np.float64)
    env = (x + 1.0) ** (-alpha)
    return env / env.sum()


def _peaking_reduction(a, n_iters=200, deadline=None):
    """Directly reduce peaks by transferring mass from high-convolution-contributing positions."""
    n = len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        cm = float(c.max())
        # Find top active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:20]

        # Compute pressure: for each position j, how much does it contribute to active peaks?
        pressure = np.zeros(n)
        for k in active:
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    pressure[j] += a[kj]

        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-30:
            break
        pressure = pressure / p_max

        # Transfer: move mass from high-pressure to low-pressure positions
        # Use a soft version: a_new[j] proportional to a[j] * (1 - eta * pressure[j])
        eta = 0.3 + 0.4 * (it / float(n_iters))
        a_new = a * (1.0 - eta * pressure)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s

        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller eta
            a_new2 = a * (1.0 - 0.1 * pressure)
            a_new2 = np.maximum(a_new2, 0.0)
            s2 = a_new2.sum()
            if s2 > 1e-30:
                a_new2 = a_new2 / s2
                sc2 = _score_norm(a_new2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = a_new2.copy()
                    a = a_new2
            else:
                break

    return best_a, best_sc
