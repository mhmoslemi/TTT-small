import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_normalized(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)

    Returns the subgradient vector (not normalized)."""

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""

def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(ind[cond])
    theta = cssv[idx - 1] / idx
    return np.maximum(v - theta, 0.0)


def _subgradient_max_conv(a):
    """Exact subgradient of max(conv(a,a)) w.r.t. a (assuming a is on simplex)."""
    n = len(a)

    k_star = int(np.argmax(c))

    k = k_star

def _adam_logspace(a_init, deadline, T_start, T_end, lr_start, lr_end, max_iter=200000):
    """Adam optimizer in log-space with soft gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    patience = 0
    lr_mult = 1.0
    clip_norm = 0.5
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        frac = min(1.0, t / max_iter)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac * lr_mult
        
        a = np.exp(z) 
        a = a / a.sum()
        c = _fft_convolve(a, a)
        max_c = c.max()
        T_eff = max(T, max_c * 1e-8)
        g = _soft_gradient(a, T_eff)
        
        # Project gradient to tangent space of simplex (remove mean component weighted by a)
        W = float(np.dot(a, g))
        z_grad = a * (g - W)
        
        if not np.all(np.isfinite(z_grad)):
            z_grad = np.zeros(n)
        
        m_buf = b1 * m_buf + (1.0 - b1) * z_grad
        v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)
        
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % 200 == 0:
            a_cur = np.exp(z)
            a_cur = a_cur / a_cur.sum()
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 10 and lr_mult > 0.05:
                lr_mult *= 0.7
                patience = 0
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
    
    a_final = np.exp(z)
    a_final = a_final / a_final.sum()

            break
        
        c = _fft_convolve(a, a)
        max_c = c.max()
        
        # Adaptive tolerance: tighten over time
        frac = min(1.0, t / max_iter)
        rel_tol_cur = 0.1 * (0.001) ** frac  # from 0.1 to 0.0001
        
        # Get active subgradient
        active = np.where(c >= max_c * (1 - rel_tol_cur))[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += a[k-lo1 : k-hi1-1 : -1]
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                grad[lo2:hi2+1] += a[lo2-k : hi2-k+1]
        grad /= len(active)
        
        # Project gradient to simplex tangent
        grad_mean = grad.mean()
        grad_tangent = grad - grad_mean
        
        # Step
        a_new = _project_simplex(a - lr * grad_tangent)
        
        # Line search (backtracking)
        val_new = _score_norm(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            patience = 0
            lr = min(lr * 1.1, 0.1 / n)
        else:
            patience += 1
            lr *= 0.95
        
        a = a_new
        
        if patience > 50:
            # Try a different direction: random perturbation
            if lr < 0.001 / n:
                # Reset
                a = best_a.copy()
                lr = 0.01 / n
                patience = 0
            else:
                break
    
    return best_a, best_val
