def _fista(a_init, beta, deadline, max_iter, L=None):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    if L is None:
        L = 4.0 / beta  # Lipschitz constant estimate for smoothed max
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc

def _adam_logspace(a_init, deadline, betas, lr_init=0.01, max_iter_total=200000, lr_decay=0.97, decay_every=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % decay_every == 0:
                current_lr = max(0.0003, current_lr * lr_decay)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _coord_descent_peaks(a_init, deadline, max_rounds=200, n_peaks=16):
    """Coordinate descent targeting the peaks in the convolution."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(42)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a, a)
        # Find peak positions
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        # For each peak, try to reduce it by adjusting contributing a[j] values
        for pk in peak_indices:
            if time.monotonic() > deadline:
                break
            # b[pk] = sum_j a[j]*a[pk-j]
            # Try reducing the largest product terms
            # Find j values where a[j]*a[pk-j] is largest
            valid_j = range(max(0, pk - n + 1), min(n, pk + 1))
            products = np.array([a[j] * a[pk - j] for j in valid_j])
            top_positions = np.argsort(products)[::-1][:min(8, len(products))]
            
            for pos in top_positions:
                j = valid_j[pos]
                old_val = a[j]
                if old_val < 1e-15:
                    continue
                # Try reducing a[j]
                for factor in [0.9, 0.95, 0.99]:
                    a[j] = old_val * factor
                    # Redistribute to other positions
                    deficit = old_val - a[j]
                    if deficit > 0:
                        # Add to random positions
                        targets = rng.choice(n, size=min(4, n), replace=False)
                        a[targets] += deficit / len(targets)
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                    sc = _score_norm(a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        break
                    a[j] = old_val
                    a = a / a.sum()
        
        # Also try increasing some a[j] values
        for j in rng.choice(n, size=min(32, n), replace=False):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for factor in [1.05, 1.1, 1.2]:
                a[j] = old_val * factor
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc = _score_norm(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
                a[j] = old_val
                a = a / a.sum()
        
        if not improved and r > 10:
            break
        a = best_a.copy()
    
    return best_a, best_sc

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    p_new = p_new / p_new.sum()
    return p_new

def _interp_cubic(p, n_new):
    """Cubic interpolation for smoother upscaling."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    from scipy.interpolate import interp1d
    f = interp1d(x0, p * n0, kind='cubic', fill_value='extrapolate')
    p_new = f(x1)
    p_new = np.maximum(p_new, 1e-15)
    p_new = p_new / p_new.sum()
    return p_new

"""

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _fista(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    it = 0
    while it < max_iter:
        if time.monotonic() > deadline:
            break
        it += 1
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        c_max = float(c_y.max())
        exponents = (c_y - c_max) / beta
        np.clip(exponents, -500, 500, out=exponents)
        w = np.exp(exponents)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_smoothed(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 300 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, betas, lr_init=0.01, max_iter_total=300000,
                   lr_decay=0.95, decay_every=6000, momentum_warmup=500):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            np.clip(c_shifted, -500, 500, out=c_shifted)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            lr_eff = current_lr * min(1.0, t / momentum_warmup)
            update = lr_eff * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_step = 0.4
            if update_norm > max_step:
                update = update * (max_step / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t % 80 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % decay_every == 0:
                current_lr = max(0.0002, current_lr * lr_decay)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _mass_redistribution(a_init, deadline, max_rounds=50, n_top_peaks=8):
    """Redistribute mass from peak-contributing to non-peak-contributing positions."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(777)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _convolve(a, a)
        # Get top peak positions
        top_peaks = np.argsort(c)[::-1][:n_top_peaks]
        
        # Compute contribution vector: for each position i, how much does it contribute to peaks?
        contrib = np.zeros(n)
        for pk in top_peaks:
            for j in range(max(0, pk - n + 1), min(n, pk + 1)):
                contrib[j] += a[pk - j]
        
        # Normalize contribution
        c_max_val = contrib.max()
        if c_max_val < 1e-30:
            break
        contrib_norm = contrib / c_max_val
        
        # Create a "target" that is inverse to contribution (spread mass where it doesn't help peaks)
        target = 1.0 / (1.0 + contrib_norm * 10.0)
        target = target / target.sum()
        
        # Blend: move some mass toward the target
        alpha = 0.05 + 0.02 * r  # increasing blend
        a_new = (1.0 - alpha) * a + alpha * target
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        sc_new = _score_norm(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller alpha
            for small_alpha in [0.01, 0.005, 0.002]:
                a_try = (1.0 - small_alpha) * a + small_alpha * target
                a_try = np.maximum(a_try, 0.0)
                a_try = a_try / a_try.sum()
                sc_try = _score_norm(a_try)
                if sc_try < best_sc:
                    best_sc = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    break

        # Random perturbations to escape stalls
        if r % 10 == 5:
            z = np.log(np.maximum(a, 1e-15))
            z += 0.01 * rng.standard_normal(n)
            z = z - z.mean()
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc_new = _score_norm(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new

    return best_a, best_sc

def _interp_upscale(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1.0, n0)
    x1 = np.linspace(0, 1.0, n_new)
    p_new = np.interp(x1, x0, p * n0)
    p_new = np.maximum(p_new, 1e-15)
    p_new = p_new / p_new.sum()
    return p_new


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    result = np.exp(z - z.max())
    return result / result.sum()

                a_8k, sc_8k_a = _adam_logspace(
                    a_8k, time.monotonic() + per_t,
                    bs_8k, lr_init=0.004, max_iter_total=200000,
                    lr_decay=0.93, decay_every=5000
                )
                a_8k = np.maximum(a_8k, 0.0)
                a_8k = a_8k / a_8k.sum()
                sc_8k_a_full = _score(a_8k)
                print("  Adam n=8192: score={0:.10f} t={1:.1f}s".format(
                    sc_8k_a_full, time.monotonic() - t_start))
                if sc_8k_a_full < best_score:
                    best_score = sc_8k_a_full
                    best_seq = a_8k.copy()

    # === Phase 5: Expand to n=16384 ===
    print("=== Phase 5: Expand to n=16384 ===")
    n_16k = 16384
    phase5_end = min(deadline, t_start + 1600.0)
    a_16k = _interp_upscale(best_seq, n_16k)
    sc_16k = _score(a_16k)
    print("  Interpolated n=16384: score={0:.10f}".format(sc_16k))
    if sc_16k < best_score:
        best_score = sc_16k
        best_seq = a_16k.copy()

    if time.monotonic() < phase5_end - 30:
        # FISTA on 16384
        c_16k = _convolve(a_16k, a_16k)
        max_c16k = float(c_16k.max())
        beta_16k = 0.02 * max_c16k
        per_fista = min(100.0, phase5_end - time.monotonic() - 80)
        if per_fista > 20:
            a_16k, sc_16k_f = _fista(a_16k, beta_16k, time.monotonic() + per_fista, 20000)
            a_16k = np.maximum(a_16k, 0.0)
            a_16k = a_16k / a_16k.sum()
            sc_16k_f_full = _score(a_16k)
            print("  FISTA n=16384: score={0:.10f} t={1:.1f}s".format(
                sc_16k_f_full, time.monotonic() - t_start))
            if sc_16k_f_full < best_score:
                best_score = sc_16k_f_full
                best_seq = a_16k.copy()

        # Fine FISTA
        if time.monotonic() < phase5_end - 40:
            c_16k2 = _convolve(a_16k, a_16k)
            max_c16k2 = float(c_16k2.max())
            beta_16k2 = 0.002 * max_c16k2
            per_fista2 = min(50.0, phase5_end - time.monotonic() - 40)
            if per_fista2 > 15:
                a_16k, sc_16k_f2 = _fista(a_16k, beta_16k2, time.monotonic() + per_fista2, 15000)
                a_16k = np.maximum(a_16k, 0.0)
                a_16k = a_16k / a_16k.sum()
                sc_16k_f2_full = _score(a_16k)
                if sc_16k_f2_full < best_score:
                    best_score = sc_16k_f2_full
                    best_seq = a_16k.copy()
                    print("  FISTA2 n=16384: score={0:.10f} t={1:.1f}s".format(
                        sc_16k_f2_full, time.monotonic() - t_start))

        # Adam on 16384
        if time.monotonic() < phase5_end - 20:
            c_16o = _convolve(a_16k, a_16k)
            max_c16o = float(c_16o.max())
            bs_16k = [0.001 * max_c16o, 0.0005 * max_c16o, 0.0002 * max_c16o,
                      0.0001 * max_c16o, 0.00005 * max_c16o]
            per_t = phase5_end - time.monotonic() - 10
            if per_t > 20:
                a_16k, sc_16k_a = _adam_logspace(
                    a_16k, time.monotonic() + per_t,
                    bs_16k, lr_init=0.003, max_iter_total=80000,
                    lr_decay=0.93, decay_every=4000
                )
                a_16k = np.maximum(a_16k, 0.0)
                a_16k = a_16k / a_16k.sum()
                sc_16k_a_full = _score(a_16k)
                print("  Adam n=16384: score={0:.10f} t={1:.1f}s".format(
                    sc_16k_a_full, time.monotonic() - t_start))
                if sc_16k_a_full < best_score:
                    best_score = sc_16k_a_full
                    best_seq = a_16k.copy()

    # === Phase 6: Mass redistribution + coordinate descent on best ===
    print("=== Phase 6: Polish ===")
    phase6_end = min(deadline, t_start + 1750.0)
    if best_seq is not None and time.monotonic() < phase6_end - 20:
        n_best = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        # Mass redistribution (only for manageable sizes)
        if n_best <= 8192:
            per_mr = min(40.0, phase6_end - time.monotonic() - 20)
            if per_mr > 10:
                a_mr, sc_mr = _mass_redistribution(
                    a_pol, time.monotonic() + per_mr, max_rounds=40, n_top_peaks=12
                )
                sc_mr_full = _score(a_mr)
                print("  MassRedist: score={0:.10f} t={1:.1f}s".format(
                    sc_mr_full, time.monotonic() - t_start))
                if sc_mr_full < best_score:
                    best_score = sc_mr_full
                    best_seq = a_mr.copy()
                    a_pol = a_mr.copy()

        # Coordinate descent
        if n_best <= 8192:
            per_cd = min(40.0, phase6_end - time.monotonic() - 10)
            if per_cd > 10:
                a_cd, sc_cd = _coord_descent(a_pol, time.monotonic() + per_cd, max_rounds=80)
                sc_cd_full = _score(a_cd)
                print("  CoordDescent: score={0:.10f} t={1:.1f}s".format(
                    sc_cd_full, time.monotonic() - t_start))
                if sc_cd_full < best_score:
                    best_score = sc_cd_full
                    best_seq = a_cd.copy()

    # === Phase 7: Final ultra-fine Adam ===
    print("=== Phase 7: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp, 0.000001 * max_cp]
        per_t = deadline - time.monotonic() - 15
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.001, max_iter_total=150000,
                lr_decay=0.92, decay_every=4000, momentum_warmup=200
            )
            a_pol = np.maximum(a_pol, 0.0)
            a_pol = a_pol / a_pol.sum()
            sc_pol_full = _score(a_pol)
            print("  Final polish: score={0:.10f} t={1:.1f}s".format(
                sc_pol_full, time.monotonic() - t_start))
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

    # === Final validation ===
    print("All phases done: best_score={0:.10f}".format(best_score))
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print("FINAL: score={0:.10f}, n={1}, time={2:.1f}s".format(
            sc_final, len(final), time.monotonic() - t_start))
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print("Evaluator score: {0:.10f}".format(sc_eval))
        except Exception as ex:
            print("Eval error: {0}".format(ex))
        return [float(x) for x in final]

    fallback = np.ones(256)
    print("FALLBACK: score={0:.10f}".format(_score(fallback)))
    return [float(x) for x in fallback]
