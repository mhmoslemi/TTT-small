def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    # Clean up numerical noise
    r[np.abs(r) < 1e-15] = 0.0
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)

    w = np.zeros(len(c))

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-18, gtol=1e-15):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 100,
                                   'maxls': 100, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _spectral_flatten(a_init, deadline, max_iter=200):
    """
    Spectral domain approach: iteratively adjust phases of DFT(a)
    to flatten the convolution, then project back to non-negative.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    N = _next_pow2(2 * n - 1)
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute DFT of a (zero-padded to N)
        A = rfft(a, N)
        
        # Target: |A(k)|^2 should make conv flat
        # Current conv in frequency domain: |A(k)|^2
        # We want the IFFT of |A(k)|^2 to be as flat as possible
        # Adjust: reduce magnitude at frequencies that contribute to peaks
        
        # Compute current convolution
        b = irfft(A * A, N)[:2*n-1]
        b_max = b.max()
        
        # Find lag of max
        lag_max = int(np.argmax(b))
        
        # The peak at lag_max is caused by constructive interference
        # in the frequency domain. To reduce it, we can add a phase
        # rotation to A to "spread out" the energy.
        
        # Simple approach: multiply A by a random phase in a subset
        # of frequencies, then project
        # Better: use the "flat spectrum" target
        
        # Target spectrum: want |A(k)|^2 to be proportional to |DFT(flat_{2n-1})(k)|
        # DFT of uniform sequence of length (2n-1) padded to N:
        target = np.zeros(N // 2 + 1)
        for k in range(N // 2 + 1):
            # DFT of ones(2n-1) at frequency k
            target[k] = np.abs(np.sum(np.exp(-2j * np.pi * k * np.arange(2*n-1) / N)))
        
        # Current |A|
        mag = np.abs(A)
        
        # Blend towards target
        alpha = 0.05  # blending factor
        target_mag = np.sqrt(target) + 1e-10
        new_mag = (1 - alpha) * mag + alpha * target_mag
        # Keep phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse DFT
        a_new = irfft(A_new, N)[:n]
        
        # Project to non-negative, sum=1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new = a_new / s
        
        sc_new = _score(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Revert and try smaller alpha
            continue

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, max_rounds=5000):
    """
    Coordinate descent: at each step, find the pair of coordinates
    whose adjustment most reduces the max convolution value.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    rng = np.random.default_rng(777)

    # Pre-compute convolution for initial
    c = _full_convolve(a, a)
    c_max = c.max()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        # Select a random pair of coordinates
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            j = (j + 1) % n

        # Try several transfer amounts
        s_pair = a[i] + a[j]
        best_local_sc = _score(a)
        best_frac = s_pair / (2.0 * n)  # current split

        for frac in [best_frac * 1.05, best_frac * 1.1, best_frac * 1.2,
                     best_frac * 1.5, best_frac / 1.05, best_frac / 1.1,
                     best_frac / 1.2, best_frac / 1.5]:
            if frac < 0 or frac > s_pair:
                continue
            a_try = a.copy()
            a_try[i] = frac
            a_try[j] = s_pair - frac
            sc_try = _score(a_try)
            if sc_try < best_local_sc:

                best_frac = frac

        if best_local_sc < best_sc - 1e-14:

def _lp_direction(a, c, n, tol_frac=1e-4):
    """
    Solve a small LP to find a direction that reduces the active peaks.
    Returns the direction vector (to be added to log-space z).
    """

        active[int(np.argmax(c))] = True

    # - a + delta >= 0
    # - sum(delta) = 0
    # - 2 * conv(delta, a)_k <= -eps for k in active
    # - ||delta||_1 is minimized (for stability)
    #
    # conv(delta, a)_k = sum_i delta_i * a_{k-i} (linear in delta)
    # So the constraint is: sum_i delta_i * a_{k-i} <= -eps/2 for k in active
    
    # This is an LP with n variables and (2n-1 + 2) constraints
    # For large n, this might be too expensive. Use a reduced version.
    
    # Alternative: just use the subgradient direction but with
    # adaptive step size based on the LP's dual variables.
    
    # For now, return the normalized subgradient
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
