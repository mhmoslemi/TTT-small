        else:
            patience += 1
            step_size = max(0.001, step_size * 0.6)
            if patience >= max_patience:
                break

    return best_a, best_score


def _adaptive_coord_polish(a, budget_s=60.0, rng=None):
    """Adaptive coordinate descent with line search."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    t_start = time.monotonic()

    # Precompute convolution for efficiency
    while time.monotonic() - t_start < budget_s:
        # Pick a random coordinate
        i = int(rng.integers(n))
        a_i = a[i]
        if a_i < 1e-12:
            a[i] = 1e-6
            a /= a.sum()
            continue

        # Compute current convolution
        c = _full_convolve(a, a)

        # For changing a[i] to t (with renormalization), the new conv is:
        # c_new[j] = c[j] - a_i * a[j-i] + (t / (1 - a_i + a_i)) * ... 
        # Actually it's easier to just try a few values directly.
        
        # Generate candidate values
        candidates = [0.0, a_i * 0.3, a_i * 0.6, a_i * 1.4, a_i * 2.0]
        # Also try values based on the gradient
        grad_i = c[i] if i < len(c) else 0  # rough

        best_t = a_i
        best_local_sc = best_score

        for tc in candidates:
            tc = max(0.0, min(tc, 1.0))
            if abs(tc - a_i) < 1e-10:
                continue
            a[i] = tc
            s = a.sum()
            if s < 1e-15:
                a[i] = a_i
                continue
            a_norm = a / s
            sc = _score_norm(a_norm)
            if np.isfinite(sc) and sc < best_local_sc:
                best_local_sc = sc
                best_t = tc

        a[i] = best_t
        s = a.sum()
        if s > 1e-15:
            a = a / s

        if _score_norm(a) < best_score:
            best_score = _score_norm(a)
            best_a = a.copy()

    return best_a, best_score


def _interp_to_length(a, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_init(n, kind="varying"):
    """Create structured initializations."""
    t = np.linspace(0, 1, n)
    if kind == "uniform":
        a = np.ones(n)
    elif kind == "cos_multi":
        a = np.ones(n)
        for freq in [1, 2, 3, 5, 7]:
            a += 0.2 / freq * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
    elif kind == "tapered":
        a = np.sin(np.pi * t) ** 0.6
        a = np.maximum(a, 1e-6)
    elif kind == "bimodal":
        a = np.exp(-30 * (t - 0.3) ** 2) + np.exp(-30 * (t - 0.7) ** 2) + 0.05
    elif kind == "plateau":
        a = np.ones(n) * 0.1
        mask = (t > 0.2) & (t < 0.8)
        a[mask] = 1.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _topk_weights(c, k, beta):
    """Compute top-K softmax weights on convolution."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    topk_thresh = np.partition(c, n_c - k)[n_c - k]
    topk_mask = c >= topk_thresh
    topk_idx = np.where(topk_mask)[0]
    if len(topk_idx) == 0:
        topk_idx = np.array([int(np.argmax(c))])
    c_topk = c[topk_idx]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    w = np.zeros(n_c)
    if w_sum < 1e-300:
        w[topk_idx[0]] = 1.0
    else:
        w[topk_idx] = exp_vals / w_sum
    return w


def _adam_fista_hybrid(a_init, deadline, n):
    """Hybrid Adam+FISTA in log-space with top-K smoothing, larger K than before."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.02

    # Phase 1: larger K (coarse)
    for phase in range(4):
        if time.monotonic() > deadline:
            break
        k_val = max(200, min(n, 500 * (phase + 1)))
        for beta_ratio in [0.01, 0.005, 0.002, 0.001]:
            if time.monotonic() > deadline:
                break
            for _ in range(2000):
                if time.monotonic() > deadline:
                    break
                t += 1
                a = _softmax(z)
                c = _full_convolve(a, a)
                max_c = float(c.max())
                beta = beta_ratio * max_c
                w = _topk_weights(c, k_val, beta)
                grad_a = _grad_conv(a, w)
                mean_g = float(np.dot(grad_a, a))
                grad_z = 2.0 * n * a * (grad_a - mean_g)

                if not np.all(np.isfinite(grad_z)):
                    break

                m_buf = b1 * m_buf + (1 - b1) * grad_z
                v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
                mh = m_buf / (1 - b1 ** t)
                vh = v_buf / (1 - b2 ** t)

                update = lr * mh / (np.sqrt(vh) + eps)
                update_norm = np.linalg.norm(update)
                if update_norm > 1.0:
                    update = update / update_norm

                z = z - update
                z = np.clip(z, -50.0, 50.0)
                z = z - z.mean()

                if t % 200 == 0:
                    a_chk = _softmax(z)
                    sc = _score_norm(a_chk)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_chk.copy()

            lr = max(0.002, lr * 0.7)
            m_buf *= 0.3
            v_buf *= 0.3

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search: multi-phase optimization with power-smoothed L-BFGS-B,
    subgradient polishing, and multi-start exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
        n_parent = len(parent)
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_multi")
        sc_parent = _score(parent)
        print(f"No parent. Init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple lengths
    if n_parent == 4096:
        lengths = [4096, 8192]
    elif n_parent == 8192:
        lengths = [8192, 4096]
    else:
        lengths = [4096, 8192, 2048]

    for n_try in lengths:
        if time.monotonic() > deadline - 120:
            break

        remaining = deadline - time.monotonic()
        print(f"\n=== n={n_try}, remaining={remaining:.0f}s ===")

        if n_try == n_parent:
            a0 = parent.copy()
        else:
            a0 = _interp_to_length(parent, n_try)
        a0 = a0 / a0.sum()
        sc0 = _score_norm(a0)
        print(f"  Initial: {sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # --- Phase 1: L-BFGS-B with power smoothing ---
        p_list = [4, 8, 16, 32, 64, 128, 256]
        phase1_deadline = time.monotonic() + remaining * 0.4
        if phase1_deadline > deadline - 60:
            phase1_deadline = deadline - 60

        z0 = np.log(np.maximum(a0, 1e-15))
        z0 = z0 - z0.mean()
        max_iter_p = max(500, int((phase1_deadline - time.monotonic()) * 500))

        z_best, a_lbfgs, sc_lbfgs = _lbfgsb_phase(z0, n_try, p_list, phase1_deadline, max_iter_p)
        if sc_lbfgs < best_score:
            best_score = sc_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"  L-BFGS-B: {sc_lbfgs:.10f} (t={time.monotonic()-t_start:.0f}s)")

        a_work = a_lbfgs if sc_lbfgs < sc0 else a0

        # --- Phase 2: Adam+FISTA hybrid with large K ---
        phase2_deadline = time.monotonic() + min(remaining * 0.3, 300.0)
        if phase2_deadline > deadline - 60:
            phase2_deadline = deadline - 60

        a_adam, sc_adam = _adam_fista_hybrid(a_work, phase2_deadline, n_try)
        if sc_adam < best_score:
            best_score = sc_adam
            best_seq = a_adam.copy()
            print(f"  Adam hybrid: {sc_adam:.10f} (t={time.monotonic()-t_start:.0f}s)")

        a_work = a_adam if sc_adam < _score_norm(a_lbfgs) else a_lbfgs

        # --- Phase 3: Subgradient polish ---
        phase3_deadline = time.monotonic() + min(remaining * 0.15, 90.0)
        if phase3_deadline > deadline - 30:
            phase3_deadline = deadline - 30

        a_sub, sc_sub = _subgrad_polish(a_work, max(5, phase3_deadline - time.monotonic()))
        if sc_sub < best_score:
            best_score = sc_sub
            best_seq = a_sub.copy()
            print(f"  Subgrad polish: {sc_sub:.10f} (t={time.monotonic()-t_start:.0f}s)")
            a_work = a_sub

        # --- Phase 4: Coordinate descent polish ---
        phase4_deadline = min(deadline - 30, time.monotonic() + 45.0)
        a_cd, sc_cd = _adaptive_coord_polish(a_work, max(5, phase4_deadline - time.monotonic()), rng)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd.copy()
            print(f"  Coord polish: {sc_cd:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # --- Multi-start exploration ---
    if time.monotonic() < deadline - 40 and best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        print(f"\n=== Multi-start: remaining={remaining:.0f}s ===")
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_base = len(a_base)

        n_starts = max(2, int(remaining / 80))
        per_start = remaining / n_starts

        for si in range(n_starts):
            if time.monotonic() > deadline - 20:
                break

            # Different perturbation strategies
            if si == 0:
                pert = 0.01 * rng.standard_normal(n_base)
                p_start = a_base + pert
            elif si == 1:
                # Log-space perturbation
                z = np.log(np.maximum(a_base, 1e-15))
                z += 0.05 * rng.standard_normal(n_base)
                p_start = _softmax(z)
            elif si == 2:
                # Frequency perturbation
                freq = rng.uniform(1.0, 5.0)
                p_start = a_base + 0.02 * np.sin(2 * np.pi * freq * np.arange(n_base) / n_base)
            elif si == 3:
                # Anti-correlation perturbation
                c = _full_convolve(a_base, a_base)
                w = c / c.sum()
                p_start = a_base - 0.01 * _grad_conv(a_base, w)
            else:
                # Shift perturbation
                shift = int(rng.integers(1, max(2, n_base // 512)))
                p_start = np.roll(a_base, shift)

            p_start = np.maximum(p_start, 1e-12)
            p_start = p_start / p_start.sum()

            # Quick optimization: L-BFGS-B with p=32,64 then subgrad
            p_deadline = time.monotonic() + per_start
            z_p = np.log(np.maximum(p_start, 1e-15))
            z_p = z_p - z_p.mean()
            _, a_p, sc_p = _lbfgsb_phase(z_p, n_base, [16, 32, 64], p_deadline - 15, 1500)

            if p_deadline - time.monotonic() > 10:
                a_p, sc_p2 = _subgrad_polish(a_p, max(5, p_deadline - time.monotonic() - 5))
                sc_p = min(sc_p, sc_p2)

            sc_full = _score(a_p)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_p.copy()
                print(f"  Start {si}: {sc_full:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # Final validation
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(np.asarray(best_seq, dtype=np.float64), 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * 0.01 / 256
        sc_final = _score(final)
        print(f"RETURN: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) * 1.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
