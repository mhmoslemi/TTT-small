def _adam_optimize(a_init, deadline, beta_schedule, lr_init=0.01, max_iter=500000):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = float(_convolve(a_init, a_init).max())
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        per_beta_deadline = min(deadline, time.monotonic() + max(10, (deadline - time.monotonic()) / max(1, len(beta_schedule))))
        
        while time.monotonic() < per_beta_deadline:
            t += 1
            a = np.exp(z - z.max())
            a = a / a.sum()
            
            c = _convolve(a, a)
            cm = float(c.max())
            
            # Softmax weights
            shifted = (c - cm) / beta
            np.clip(shifted, -500, 500, out=shifted)
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[cm.argmax() if hasattr(cm, 'argmax') else int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Gradient: 2 * cross_correlation(w, a)
            N = _next_pow2(3 * n)
            W = np.fft.rfft(w, N)
            A = np.fft.rfft(a, N)
            cc = np.fft.irfft(W * np.conj(A), N)[:n]
            grad = 2.0 * cc
            
            # Scale gradient for log-space
            # d/dz = d/da * da/dz = grad * a (in log-space, da_i/dz_i = a_i, da_i/dz_j = 0 for i!=j)
            # Actually for softmax: a_i = exp(z_i) / sum(exp(z)), da_i/dz_j = a_i(δ_ij - a_j)
            # So grad_z_i = sum_j (da_j/dz_i) * grad_j = sum_j a_j(δ_ji - a_i) * grad_j
            # = a_i * grad_i - a_i * sum_j a_j * grad_j
            # = a_i * (grad_i - dot(a, grad))
            lag = float(np.dot(a, grad))
            grad_z = a * (grad - lag)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = float(_convolve(a_chk, a_chk).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if t % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.9)
    
    # Final
    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = float(_convolve(a_final, a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _peak_shave(a, deadline, max_rounds=100):
    """
    Peak shaving: find peaks in convolution and reduce contributing elements.
    """
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_sc = float(_convolve(a, a).max())
    rng = np.random.default_rng(42)
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        top_k = 50
        # Get indices of top values
        # Sort in descending order, take top k
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_indices = top_indices[np.argsort(c[top_indices])[::-1]]
        
        improved = False
        for k in top_indices[:20]:  # Process top 20 peaks
            if time.monotonic() > deadline:
                break
            
            # Find the dominant contributing pair for this peak
            # b[k] = sum_j a[j] a[k-j]
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            if lo > hi:
                continue
            
            indices = np.arange(lo, hi + 1)
            products = a[indices] * a[k - indices]
            
            # Find the index with maximum product
            j_star = indices[np.argmax(products)]
            
            # Try reducing a[j_star] or a[k-j_star]
            for target in [j_star, k - j_star]:
                old = a[target]
                # Reduce by a small amount
                for factor in [0.99, 0.98, 0.95]:
                    new = old * factor
                    a[target] = new
                    s = a.sum()
                    a = a / s  # Renormalize
                    sc = float(_convolve(a, a).max())
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[target] = old
                        a = a / s  # Renormalize back (s might have changed)
                        # Actually this is getting messy with renormalization
                        # Let me just break
                        break
        
        if not improved and r > 20:
            break
    
    return best_a, best_sc


def _interp(a, n_new):
    """Interpolate sequence to new length."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def _coord_descent(a_init, deadline, max_rounds=500):
    """Coordinate descent polish."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = float(_convolve(a, a).max())
    rng = np.random.default_rng(999)
    
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        if n > 4096:
            indices = rng.choice(n, size=min(1024, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)
        
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            for delta_frac in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                new_val = old_val * (1 + delta_frac)
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                # Renormalize
                a_norm = a / s
                sc = float(_convolve(a_norm, a_norm).max())
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    old_val = a[j]
                    improved = True
                else:
                    a[j] = old_val
        
        if not improved and r > 30:
            break
    
    return best_a, best_sc


def _perturb(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.max()
    return np.exp(z) / np.exp(z).sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-resolution optimizer combining FISTA, Adam, peak shaving, and coordinate descent.
    Strategy: 
    1. Start from parent (n=4096) or uniform (n=4096)
    2. Optimize at n=4096 with FISTA + Adam
    3. Upscale to n=8192 and optimize
    4. Upscale to n=16384 if time permits
    5. Peak shaving + coordinate descent polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

                s = float(parent_raw.sum())
                if s > 0.01:
                    parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_main = len(parent)

        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        # Start with a raised cosine shape (better than uniform)

        a0 = 0.5 * (1 + np.cos(2 * np.pi * x))  # Raised cosine (zero at edges)
        a0 = np.maximum(a0, 0)
        a0 = a0 / a0.sum()
        sc0 = _score(a0)

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}, n={n_main}")
    
    # === Phase 1: FISTA at current n ===
    print(f"\nPhase 1: FISTA at n={n_main}")
    phase1_end = min(deadline, t_start + 300.0)
    
    c_curr = _convolve(a0, a0)
    max_cc = float(c_curr.max())
    beta = 0.02 * max_cc
    
    while time.monotonic() < phase1_end - 5 and beta > 1e-15 * max_c0:

        if per_time < 5:
            break
        a0, sc_norm = _fista_adaptive(a0, time.monotonic() + per_time, beta, max_iter=50000)
        a0 = a0 / a0.sum()
        sc_full = _score(a0)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a0.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        
        c_curr = _convolve(a0, a0)
        max_cc = float(c_curr.max())
        beta *= 0.4
    
    print(f"End Phase 1: score={best_score:.10f}")
    
    # === Phase 2: Adam fine-tuning at current n ===
    print(f"\nPhase 2: Adam at n={n_main}")
    phase2_end = min(deadline, t_start + 600.0)
    
    c_curr = _convolve(a0, a0)
    max_cc = float(c_curr.max())
    betas_fine = [max_cc * f for f in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001]]
    betas_fine = [b for b in betas_fine if b > 1e-15]
    
    if time.monotonic() < phase2_end - 10:
