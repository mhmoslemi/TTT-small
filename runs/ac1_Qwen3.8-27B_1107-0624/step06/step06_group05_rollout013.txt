"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: tilted (asymmetric) profiles like linear ramps give scores ~4/3 < 2,
far below symmetric profiles. We use ramp initializations with FISTA/Adam/L-BFGS
continuation to find near-optimal asymmetric sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 40 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_opt(a_init, deadline, beta, lr_init, max_iter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        if t % 30 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 3000 == 0:
            current_lr = max(0.0002, current_lr * 0.9)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lbfgs_opt(a_init, deadline, beta):
    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

    def obj_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

    def grad_fn(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))
        return a * (grad_a - correction)

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 50)), 'gtol': 1e-13, 'maxls': 20}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception:
        pass

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_ramp(n, power=1.0):
    """Create a tilted profile: a[i] = (i+1)^power, normalized."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** power
    return a / a.sum()


def _make_exp_ramp(n, alpha=2.0):
    """Exponential ramp: a[i] = exp(alpha * i / n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * i / n)
    return a / a.sum()


def _make_quadratic_ramp(n):
    """Quadratic ramp: a[i] = (i+1)^2."""
    return _make_ramp(n, power=2.0)


def _make_cubic_ramp(n):
    """Cubic ramp."""
    return _make_ramp(n, power=3.0)


def _make_half_box(n, frac=0.6):
    """Box that's high on the first frac portion, low on the rest."""
    k = int(n * frac)
    a = np.zeros(n)
    a[:k] = 1.0
    if n - k > 0:
        a[k:] = 0.3
    return a / a.sum()


def _make_two_bump(n, ratio=0.7, gap_frac=0.0):
    """Two bumps: one large, one small, separated."""
    k1 = int(n * 0.4)
    k2 = int(n * 0.3)
    a = np.zeros(n)
    center1 = k1 // 2
    center2 = n - k2 // 2
    for i in range(k1):
        t = abs(i - center1) / (k1 / 2 + 0.5)
        a[i] = ratio * (1.0 + np.cos(np.pi * min(t, 1.0))) / 2.0
    for i in range(k2):
        idx = n - k2 + i
        t = abs(i - k2 // 2) / (k2 / 2 + 0.5)
        a[idx] = (1.0 - ratio) * (1.0 + np.cos(np.pi * min(t, 1.0))) / 2.0
    return a / a.sum()


def _coordinate_descent(a, deadline, max_sweeps=50):
    """Cyclic coordinate descent: for each coordinate, find the optimal value
    given all others fixed, to minimize max(conv(a,a))."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_norm = _score_norm(a)
    best_a = a.copy()
    step_sizes = [0.01, 0.005, 0.002, 0.001]

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        improved = False
        for coord in range(0, n, max(1, n // 200)):
            if time.monotonic() > deadline:
                break
            orig_val = a[coord]
            for step in step_sizes:
                for direction in [1, -1]:
                    trial = a.copy()
                    new_val = max(0.0, orig_val + direction * step)
                    if new_val == orig_val:
                        continue
                    delta = new_val - orig_val
                    trial[coord] = new_val
                    # sum stays same (we're on simplex)
                    # Actually for CD on simplex, we adjust: increase one, decrease another
                    # Simpler: just try perturbation in log space
                    pass
            # Log-space perturbation
            z = np.log(np.maximum(a, 1e-15))
            z_pert = z.copy()
            z_pert[coord] += 0.1
            z_pert = z_pert - z_pert.mean()
            a_pert = _softmax_compute(z_pert)
            c_pert = _full_convolve(a_pert, a_pert)
            if c_pert.max() < c_max - 1e-12:
                a = a_pert
                c = c_pert
                c_max = float(c.max())
                improved = True
        if _score_norm(a) < best_norm:
            best_norm = _score_norm(a)
            best_a = a.copy()
        if not improved and sweep > 5:
            break

    return best_a, best_norm


def _multi_beta_fista(a_init, deadline, num_phases=30):
    """FISTA with aggressive beta continuation."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    c_curr = _full_convolve(a_curr, a_curr)
    beta = 0.05 * float(c_curr.max())
    phase_time = (deadline - time.monotonic()) / num_phases

    for phase in range(num_phases):
        if time.monotonic() > deadline:
            break
        t_end = time.monotonic() + min(phase_time, 20.0)
        a_curr, _ = _fista_opt(a_curr, beta, t_end, max_iter=50000)
        a_curr = a_curr / a_curr.sum()
        sc = _score_norm(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
        c_curr = _full_convolve(a_curr, a_curr)
        beta = 0.4 * float(c_curr.max())
        if beta < 1e-12:
            beta = 1e-12

    return best_a, best_score


def _refine_sequence(a, deadline):
    """Full refinement pipeline: FISTA -> Adam -> L-BFGS."""
    n = len(a)
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()

    # FISTA continuation
    fista_end = time.monotonic() + (deadline - time.monotonic()) * 0.4
    a, _ = _multi_beta_fista(a, fista_end, num_phases=20)
    sc = _score(a)
    if sc < best_score / _score_norm(a) * _score(a):
        pass
    sn = _score_norm(a)
    if sn < best_score:
        best_score = sn
        best_a = a.copy()

    # Adam fine-tuning
    c_curr = _full_convolve(a, a)
    max_c = float(c_curr.max())
    adam_betas = [0.001 * max_c, 0.0003 * max_c, 0.0001 * max_c]
    adam_end = time.monotonic() + (deadline - time.monotonic()) * 0.4
    per_beta = (adam_end - time.monotonic()) / len(adam_betas)
    for beta in adam_betas:
        if time.monotonic() > adam_end:
            break
        a, _ = _adam_opt(a, time.monotonic() + per_beta, beta, lr_init=0.004, max_iter=200000)
        sn = _score_norm(a)
        if sn < best_score:
            best_score = sn
            best_a = a.copy()

    # L-BFGS
    if time.monotonic() < deadline - 10:
        c_curr = _full_convolve(a, a)
        lb_beta = 0.00005 * float(c_curr.max())
        a, _ = _lbfgs_opt(a, deadline, lb_beta)
        sn = _score_norm(a)
        if sn < best_score:
            best_score = sn
            best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Try multiple n values
    n_candidates = [4096, 8192]

    # Also try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_candidates.insert(0, len(parent))
    except Exception:
        pass

    print(f"Starting optimization, deadline in {budget_s - 15:.0f}s")

    # === Phase 0: Generate and evaluate initial profiles ===
    print("Phase 0: Evaluating initial profiles")
    phase0_end = min(deadline, t_start + 30.0)

    initial_profiles = {}

    n_main = n_candidates[0]

    profiles = {
        "ramp1": _make_ramp(n_main, 1.0),
        "ramp0.5": _make_ramp(n_main, 0.5),
        "ramp1.5": _make_ramp(n_main, 1.5),
        "ramp2": _make_ramp(n_main, 2.0),
        "exp2": _make_exp_ramp(n_main, 2.0),
        "exp3": _make_exp_ramp(n_main, 3.0),
        "exp4": _make_exp_ramp(n_main, 4.0),
        "half_box": _make_half_box(n_main, 0.5),
        "two_bump": _make_two_bump(n_main),
    }

    if parent is not None:
        profiles["parent"] = parent.copy()
        print(f"  Parent loaded: n={len(parent)}, score={_score(parent):.8f}")

    for name, prof in profiles.items():
        sc = _score(prof)
        initial_profiles[name] = (prof, sc)
        print(f"  {name}: n={len(prof)}, score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_seq = prof.copy()
        if time.monotonic() > phase0_end:
            break

    # Also try larger n
    for n_large in n_candidates[1:]:
        if time.monotonic() > phase0_end:
            break
        for name_f, func in [("ramp", _make_ramp), ("exp3", _make_exp_ramp)]:
            if name_f == "ramp":
                prof = func(n_large, 1.0)
            else:
                prof = func(n_large, 3.0)
            sc = _score(prof)
            print(f"  large_{name_f}_n{n_large}: score={sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_seq = prof.copy()
            initial_profiles[f"large_{name_f}_n{n_large}"] = (prof, sc)

    # Pick the best 3 initial profiles for optimization
    sorted_profiles = sorted(initial_profiles.items(), key=lambda x: x[1][1])
    top_profiles = sorted_profiles[:3]
    print(f"Top 3 profiles: {[k for k, _ in top_profiles]}")

    # === Phase 1: Optimize top profiles ===
    print("Phase 1: Optimizing top profiles")
    phase1_end = min(deadline, t_start + 600.0)
    per_profile = (phase1_end - time.monotonic()) / len(top_profiles)

    for idx, (name, (prof, sc)) in enumerate(top_profiles):
        if time.monotonic() > phase1_end:
            break
        t_end = time.monotonic() + per_profile
        print(f"  Optimizing '{name}' (init score={sc:.8f}), time budget={per_profile:.0f}s")
        a_opt, sn = _refine_sequence(prof, t_end)
        sc_opt = _score(a_opt)
        print(f"  '{name}' -> score={sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()

    # === Phase 2: Fine-tuning the best ===
    print(f"Phase 2: Fine-tuning (current best={best_score:.8f})")
    phase2_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        n = len(a_work)

        # Multi-restart from best
        c_curr = _full_convolve(a_work, a_work)
        max_c = float(c_curr.max())

        betas = [0.0005 * max_c, 0.0002 * max_c, 0.00005 * max_c, 0.00001 * max_c]
        per_beta = (phase2_end - time.monotonic()) / (len(betas) + 1)

        for beta in betas:
            if time.monotonic() > phase2_end - 30:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + per_beta, beta,
                                  lr_init=0.003, max_iter=300000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.8f} t={time.monotonic()-t_start:.0f}s")

        # L-BFGS
        if time.monotonic() < phase2_end - 15:
            c_curr = _full_convolve(a_work, a_work)
            lb_beta = 0.000005 * float(c_curr.max())
            a_work, _ = _lbfgs_opt(a_work, phase2_end, lb_beta)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  L-BFGS: score={sc:.8f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: Novel restarts with different structures ===
    print(f"Phase 3: Novel restarts (current best={best_score:.8f})")
    phase3_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase3_end - 20:
        n_main = len(best_seq) if best_seq is not None else 4096

        restart_profiles = []
        # Perturbed versions of best
        if best_seq is not None:
            z = np.log(np.maximum(best_seq, 1e-15))
            for sigma in [0.05, 0.1, 0.2]:
                z_p = z + sigma * rng.standard_normal(len(z))
                z_p = z_p - z_p.mean()
                restart_profiles.append(_softmax_compute(z_p))
            # Shifted
            for shift in [1, 3, 10]:
                shifted = np.roll(best_seq, shift)
                # Make it non-negative and renormalize
                shifted = np.maximum(shifted, 0)
                s = shifted.sum()
                if s > 0.01:
                    restart_profiles.append(shifted / s)

        # New structural profiles
        restart_profiles.append(_make_ramp(n_main, 0.7))
        restart_profiles.append(_make_ramp(n_main, 1.3))
        restart_profiles.append(_make_exp_ramp(n_main, 1.5))
        restart_profiles.append(_make_exp_ramp(n_main, 5.0))
        restart_profiles.append(_make_half_box(n_main, 0.4))
        restart_profiles.append(_make_half_box(n_main, 0.7))

        # Sort by score, take best 4
        scored = []
        for p in restart_profiles:
            sc = _score(p)
            scored.append((sc, p))
        scored.sort(key=lambda x: x[0])

        per_restart = (phase3_end - time.monotonic() - 10) / min(4, len(scored))
        for i in range(min(4, len(scored))):
            if time.monotonic() > phase3_end - 10:
                break
            sc_r, p_r = scored[i]
            t_end = time.monotonic() + per_restart
            a_opt, _ = _refine_sequence(p_r, t_end)
            sc_final = _score(a_opt)
            if sc_final < best_score:
                best_score = sc_final
                best_seq = a_opt.copy()
                print(f"  Restart {i} (init={sc_r:.6f}): score={sc_final:.8f} t={time.monotonic()-t_start:.0f}s")

    # === Phase 4: Final ultra-polish ===
    print(f"Phase 4: Final polish (best={best_score:.8f})")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        n = len(a_pol)
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        remaining = deadline - time.monotonic() - 8

        # Very fine Adam
        if remaining > 20:
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + remaining * 0.5,
                                 0.000003 * max_cp, 0.001, 500000)
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc:.10f}")

        # Final L-BFGS with tiny beta
        if time.monotonic() < deadline - 10:
            c_f = _full_convolve(a_pol, a_pol)
            b_f = 0.000001 * float(c_f.max())
            a_pol, _ = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
            sc = _score(a_pol)
            if sc < best_score:
                best_score = sc
                best_seq = a_pol.copy()
                print(f"  Final L-BFGS: score={sc:.10f}")

    # === Final verification ===
    print(f"\n=== FINAL RESULT ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_ramp(4096, 1.0)
        sc_final = _score(final)
        print(f"Score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_ramp(4096, 1.0)
    print(f"FALLBACK: score={_score(fallback):.8f}")
    return [float(x) for x in fallback]
