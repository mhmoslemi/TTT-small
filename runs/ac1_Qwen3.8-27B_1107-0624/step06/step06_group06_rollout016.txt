def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):

def _dual_direction(a, n, n_iter=50):

    c = _full_convolve(a, a)
    c_max = float(c.max())

    # and sum(g) is maximized.
    # Dual: min c_max * sum(lambda) s.t. sum_k lambda_k * a[k-j] >= 1 for all j, lambda >= 0
    # (where a[k-j] means a indexed appropriately)
    
    # Use a simple iterative approach: start with uniform weight on active set
    # and iteratively weight more on constraints that are "tightest"
    
    # Actually, let's just use the smoothed gradient as a proxy direction
    # and "invert" it to get a direction that would flatten the convolution
    
    z = np.zeros(n)
    for it in range(n_iter):
        g = _softmax(z)
        g = g / g.sum()
        ag = _full_convolve(a, g)
        # We want ag <= c_max everywhere. If some components exceed, 
        # reduce g at positions that cause those peaks.
        excess = np.maximum(ag - c_max, 0)
        # Gradient of excess w.r.t. g is: for each position in excess,
        # the derivative w.r.t. g_j is a[k-j] (the contribution of g_j to ag_k)
        # So we want to decrease g_j where it contributes to excess
        a_rev = a[::-1]
        # grad_j = sum_{k: excess_k > 0} a[k-j] (if k-j in range)

        grad = ce[n-1:2*n-1]
        # Projected gradient step on the log domain
        lr = 0.1
        z = z - lr * np.maximum(grad, 0) + lr * (excess.sum() / n)  # bias to keep sum
        z = np.clip(z, -20, 20)
        z = z - z.mean()
    
    g = _softmax(z)
    return g / g.sum()

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive L-BFGS with extended beta continuation, active-set conjugate
gradient polish, dual-direction perturbation, and diverse structured restarts.
Key improvements: more beta stages, CG on active-set QP, structured init profiles,
better time allocation, and tighter final tolerances.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k *= 2
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_polish(a_init, deadline, n_max_steps=8000, lr_init=0.025):

    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, tol_frac=0.0015)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:

        step = lr / (1.0 + 0.0002 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)

                best_a = a_check.copy()

                lr *= 0.92

                    z = z - z.mean()
                    patience = 0
                    lr = lr_init * 0.4

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()
    return best_a, best_sc


def _active_cg(a_init, deadline, n_max_iter=5000, tol_frac=0.001):
    """Conjugate gradient on the active-set QP: min max(b_k) s.t. sum(a)=1, a>=0."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

        if time.monotonic() > deadline:

        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        n_active = int(active.sum())
        if n_active == 0:
            active[int(np.argmax(c))] = True
            n_active = 1

        # Gradient of (1/n_active) * sum_{k in active} b_k w.r.t. a
        # = (1/n_active) * sum_{k in active} 2 * (P_k a) (symmetric part)
        # (P_k a)_j = a[k-j] if 0<=k-j<=n-1
        # Sum over active k: sum_j a[j] * active[j+i] for each i
        # = correlation of a with active
        w = np.zeros_like(c)
        w[active] = 1.0 / n_active
        a_flip = a[::-1]
        wc = _full_convolve(w, a_flip)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        # Project onto tangent space of sum(a)=1
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-14:
            break

        # Line search
        alpha = 1.0
        f_old = float(c.max())
        a_new = a.copy()
        for ls in range(40):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-12:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try <= f_old + 1e-15:
                a_new = a_try
                break
            alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
        elif it > 100 and sc > best_sc + 1e-10:
            break

    return best_a, best_sc


def _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=60.0):
    """L-BFGS with extended adaptive beta continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init, _score(a_init)

    c_sorted = np.sort(c_cur)[::-1]
    if len(c_sorted) > 30:
        gap = c_sorted[0] - c_sorted[29]
    else:
        gap = c_sorted[0]
    base_beta = max(gap * 0.3, max_c * 0.02)

    # Extended beta schedule: more stages, wider range
    mults = [1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.035,
             0.022, 0.014, 0.009, 0.0055, 0.0035, 0.002, 0.0012, 0.0007,
             0.0004, 0.0002]
    betas = base_beta * np.array(mults)

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_time = 0.003 * max(1.0, n / 2048.0)
        # More iterations for smaller beta
        scale = 1.0 + 0.5 * i
        max_iter = min(10000, max(50, int(remaining / (est_time * 12 * scale))))

        try:
            result = minimize(
                _smoobj_factory(beta, n), z,
                jac=_smograd_factory(beta, n),
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-16,
                    'gtol': 1e-12,
                    'maxcor': 50,
                    'maxls': 100,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a, current_sc


def _smoobj_factory(beta, n):
    """Create a closure-free objective wrapper using a class."""
    class Obj:
        def __call__(self, z_local):
            return _smoobj(z_local, beta, n)
    return Obj()


def _smograd_factory(beta, n):
    """Create a closure-free gradient wrapper using a class."""
    class Grad:
        def __call__(self, z_local):
            return _smograd(z_local, beta, n)
    return Grad()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2) + 0.01
        return a / a.sum()
    elif kind == "periodic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 6.0 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * (t - 0.5) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.025 * rng.uniform(0.3, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = max(1, n // (2 * n_blocks))
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i + block_size].copy()
        z[i:i + block_size] = z[j:j + block_size]
        z[j:j + block_size] = tmp
    return _softmax(z)


def _dual_direction(a, n, n_iter=60):
    """Find g >= 0, sum(g)=1 that flattens conv(a, g)."""
    a = a / a.sum()
    z = np.zeros(n)

    for it in range(n_iter):
        g = _softmax(z)
        g = g / g.sum()
        ag = _full_convolve(a, g)
        ag_max = float(ag.max())
        tol = 0.01 * ag_max
        active = ag >= (ag_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(ag))] = True
        w = np.zeros_like(ag)
        w[active] = 1.0 / active.sum()

        # Gradient of smoothed max(ag) w.r.t. g
        # d ag_k / d g_j = a[k-j] (if in range)
        # So grad_g[j] = sum_k w_k * a[k-j]
        # = cross-correlation of w and a
        a_flip = a[::-1]
        wa = _full_convolve(w, a_flip)
        grad_g = wa[n - 1:2 * n - 1]

        lr = 0.08
        z = z - lr * grad_g
        z = z - z.mean()
        z = np.clip(z, -30.0, 30.0)

    g = _softmax(z)
    return g / g.sum()


def _shrinkage(a, deadline):
    """Progressive shrinkage: zero out smallest elements and check improvement."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    a_sorted_idx = np.argsort(a)

    fractions = [0.003, 0.005, 0.01, 0.015, 0.02, 0.03, 0.04, 0.06, 0.08, 0.10, 0.12, 0.15]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-12:
            best_sc = sc_try
            best_a = a_try.copy()
            a = a_try

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive L-BFGS + active-set CG + subgradient
    + dual direction + shrinkage + diverse restarts + structured inits.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"imp={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        parent_sc = _score(parent)
        print(f"Parent: n={n_parent}, score={parent_sc:.10f}")
        update_best(parent, parent_sc, "parent")
    else:
        n_parent = 4096
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # === Phase 1: Main L-BFGS from parent ===
    print(f"\n[Phase 1] L-BFGS main (t={time.monotonic() - t_start:.1f}s)")
    a_main, sc_main = _run_lbfgs_adaptive(parent, deadline, min_time_remaining=600.0)
    update_best(a_main, sc_main, "lbfgs_main")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 450:
        sg_dl = min(deadline - 450, time.monotonic() + 200)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_dl, n_max_steps=6000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_1")

    # === Phase 2: Active-set CG ===
    print(f"\n[Phase 2] Active-set CG (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 350:
        cg_dl = min(deadline - 350, time.monotonic() + 120)
        a_cg, sc_cg = _active_cg(best_seq, cg_dl, n_max_iter=3000, tol_frac=0.0008)
        update_best(a_cg, sc_cg, "cg_1")

    # === Phase 3: Shrinkage + re-optimize ===
    print(f"\n[Phase 3] Shrinkage (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 250)
        update_best(a_shr, sc_shr, "shrink")
        if sc_shr <= best_score + 1e-11:
            a_shr2, sc_shr2 = _run_lbfgs_adaptive(a_shr, deadline, min_time_remaining=150.0)
            update_best(a_shr2, sc_shr2, "lbfgs_post_shrink")
            if best_seq is not None and time.monotonic() < deadline - 180:
                a_cg2, sc_cg2 = _active_cg(best_seq, min(deadline - 180, time.monotonic() + 60),
                                           n_max_iter=2000, tol_frac=0.0005)
                update_best(a_cg2, sc_cg2, "cg_post_shrink")

    # === Phase 4: Dual direction ===
    print(f"\n[Phase 4] Dual direction (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 180:
        n = len(best_seq)
        g_dual = _dual_direction(best_seq, n, n_iter=50)
        for t_blend in [0.1, 0.2, 0.35, 0.5]:
            a_blend = (1.0 - t_blend) * best_seq + t_blend * g_dual
            a_blend = np.maximum(a_blend, 0)
            s = a_blend.sum()
            if s > 0.01:
                a_blend = a_blend / s
                sc_b = _score(a_blend)
                update_best(a_blend, sc_b, f"dual_t{t_blend}")
        # Optimize the best dual blend
        a_d_best = (1.0 - 0.2) * best_seq + 0.2 * g_dual
        a_d_best = np.maximum(a_d_best, 0)
        a_d_best = a_d_best / a_d_best.sum()
        if time.monotonic() < deadline - 120:
            a_do, sc_do = _run_lbfgs_adaptive(a_d_best, deadline, min_time_remaining=80.0)
            update_best(a_do, sc_do, "dual_opt")

    # === Phase 5: Structured initializations (different basins) ===
    print(f"\n[Phase 5] Structured inits (t={time.monotonic() - t_start:.1f}s)")
    profiles_to_try = ["tapered2", "periodic", "quadratic", "bimodal"]
    for pk in profiles_to_try:
        if time.monotonic() > deadline - 80:
            break
        n_try = n_parent if n_parent in [1024, 2048, 3072, 4096] else 4096
        if pk == "periodic":
            n_try = 4096
        a_init = _make_profile(n_try, pk)
        # Blend with parent to stay in a good region
        if parent is not None and len(parent) == n_try:
            a_init = 0.7 * a_init + 0.3 * parent
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        a_opt, sc_opt = _run_lbfgs_adaptive(a_init, deadline, min_time_remaining=50.0)
        update_best(a_opt, sc_opt, f"profile_{pk}")

    # === Phase 6: Diverse restarts from best ===
    print(f"\n[Phase 6] Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        restarts = [
            ("g002", _perturb(best_seq, 0.02, rng)),
            ("g004", _perturb(best_seq, 0.04, rng)),
            ("f5", _perturb_fourier(best_seq, rng, n_modes=5)),
            ("f8", _perturb_fourier(best_seq, rng, n_modes=8)),
            ("blk", _perturb_block(best_seq, rng, n_blocks=5)),
        ]
        for label, p_start in restarts:
            if time.monotonic() > deadline - 15:
                break
            a_r, sc_r = _run_lbfgs_adaptive(p_start, deadline, min_time_remaining=20.0)
            update_best(a_r, sc_r, f"rst_{label}")

    # === Phase 7: Final intensive polish ===
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Subgradient with very smalllr
        if time.monotonic() < deadline - 8:
            a_fp1, sc_fp1 = _subgrad_polish(best_seq, deadline - 8, n_max_steps=4000, lr_init=0.008)
            update_best(a_fp1, sc_fp1, "final_sg")

        # Active-set CG with very tight tolerance
        if time.monotonic() < deadline - 4:
            a_fp2, sc_fp2 = _active_cg(best_seq, deadline - 4, n_max_iter=3000, tol_frac=0.0003)
            update_best(a_fp2, sc_fp2, "final_cg")

        # Ultra-fine L-BFGS
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.mean()
            beta_fin = mb * 0.00005
            remaining = deadline - 2 - time.monotonic()
            if remaining > 2:
                max_iter = min(4000, max(200, int(remaining / 0.02)))
                try:
                    result = minimize(
                        _smoobj_factory(beta_fin, n), z,
                        jac=_smograd_factory(beta_fin, n),
                        method='L-BFGS-B',
                        options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13,
                                 'maxcor': 50, 'maxls': 100}
                    )
                    a_ff = _softmax(result.x)
                    update_best(a_ff, _score(a_ff), "final_lbfgs_ultra")
                except Exception:
                    pass

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        # Cross-check with evaluator
        try:
            eval_sc = evaluate_sequence([float(x) for x in final])
            print(f"EVALUATE: {eval_sc:.10f}")
        except Exception as e:
            print(f"Eval check error: {e}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
