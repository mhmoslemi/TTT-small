"""
Multi-phase optimizer for non-negative step functions minimizing 2n*max(a*a)/(sum(a))^2.
Key improvements: proper 1D coordinate minimization, multiplicative weight updates,
n=16384 exploration, and better time allocation for fine-tuning.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_conv(a, w):
    n = len(a)
    w_rev = w[::-1]
    wc = _convolve(w_rev, a)
    return 2.0 * wc[n - 1:2 * n - 1]


def _softmax_weights(c, beta):
    c_max = float(c.max())
    exponents = (c - c_max) / beta
    np.clip(exponents, -500, 500, out=exponents)
    w = np.exp(exponents)
    w_sum = float(w.sum())
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    return w


def _fista(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 / beta
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _convolve(y, y)
        w = _softmax_weights(c_y, beta)
        grad = _grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 500 == 0:
            sc = _score_norm(a_next)
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    return best_a, best_sc


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = a_init * np.exp(z - z.max())
            a = a / a.sum()
            c = _convolve(a, a)
            w = _softmax_weights(c, beta)

            a_rev = a[::-1]
            wc = _convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.2:
                update = update * (0.2 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if it_in_phase % 100 == 0:
                a_chk = a_init * np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.95)

    a_final = a_init * np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_min_1d(a, c, i):
    """
    Find optimal a[i] given all other coordinates fixed.
    Uses ternary search on the 1D function.
    a is normalized (sum=1), c = convolve(a,a).
    Returns the optimal value for a[i].
    """
    n = len(a)
    old_ai = a[i]
    S_other = 1.0 - old_ai  # since sum = 1

    # Precompute affected convolution values
    # Affected k: k in [i, i+n-1] where k-i in [0, n-1]
    k_aff = np.arange(i, i + n)  # these are the affected positions
    # For each k in k_aff, a_{k-i} is the relevant coefficient
    a_ki = a[0:n]  # a_{k-i} for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]

    # c_affected[k_idx] = c[k_aff[k_idx]]
    c_aff = c[k_aff]

    # Unaffected max
    # All k not in k_aff
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[k_aff] = False
    c_unaff = c[mask]
    C_unaff = float(c_unaff.max()) if len(c_unaff) > 0 else 0.0

    # The function f(x) = max(max_aff(x), C_unaff) / (S_other + x)^2
    # where max_aff(x) = max(max(c_aff + 2*(x - old_ai)*a_ki), c[2i] + x^2 - old_ai^2) if 2i in k_aff

    # Check if 2i is in k_aff
    two_i = 2 * i
    in_aff = (two_i >= i) and (two_i < i + n)
    if in_aff:
        two_i_idx = two_i - i  # index in k_aff array
        c_2i = c_aff[two_i_idx]

    def f_val(x):
        if x < 0:
            x = 0.0
        denom = (S_other + x) ** 2
        if denom < 1e-30:
            return 1e30
        # Linear part: c_aff + 2*(x - old_ai) * a_ki
        linear = c_aff + 2.0 * (x - old_ai) * a_ki
        max_lin = float(linear.max())
        # Quadratic part
        if in_aff:
            quad = c_2i + x * x - old_ai * old_ai
            max_aff = max(max_lin, quad)
        else:
            max_aff = max_lin
        total_max = max(max_aff, C_unaff)
        return total_max / denom

    # Ternary search
    lo = 0.0
    hi = max(old_ai * 3.0, 0.05)
    # Extend hi if needed
    f_hi = f_val(hi)
    f_lo = f_val(lo)
    f_mid = f_val(hi / 2)
    if f_hi < f_lo and f_hi < f_mid:
        hi = hi * 2
        for _ in range(20):
            if f_val(hi) > f_val(hi / 2):
                break
            hi *= 2
            if hi > 10.0:
                break

    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = f_val(m1)
        f2 = f_val(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    x_opt = (lo + hi) / 2.0
    # Also check boundaries
    candidates = [0.0, x_opt, old_ai]
    best_x = candidates[0]
    best_f = f_val(candidates[0])
    for cx in candidates[1:]:
        cv = f_val(cx)
        if cv < best_f:
            best_f = cv
            best_x = cx
    return best_x


def _coord_descent_proper(a_init, deadline, max_sweeps=500):
    """Proper 1D coordinate descent with ternary search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _convolve(a, a)

    rng = np.random.default_rng(777)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Process coordinates in random order
        order = np.arange(n)
        rng.shuffle(order)

        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            new_val = _coord_min_1d(a, c, j)
            if new_val != old_val:
                # Update a[j] and renormalize
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                a = a / s
                c = _convolve(a, a)
                sc = _score_norm(a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                else:
                    # Revert
                    a[j] = old_val
                    s = a.sum()
                    a = a / s
                    c = _convolve(a, a)

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return best_a, best_sc


def _mul_weight_update(a_init, deadline, max_iter, beta_init):
    """Cautious multiplicative weight update in log-space."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    beta = beta_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _convolve(a, a)
        w = _softmax_weights(c, beta)
        grad = _grad_conv(a, w)

        # Multiplicative update: a_i <- a_i * exp(-lr * (grad_i - mean_grad))
        lr = 0.5
        mean_g = float(grad.mean())
        update = lr * (grad - mean_g)
        np.clip(update, -2.0, 2.0, out=update)
        a = a * np.exp(-update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Adapt beta
        if it % 1000 == 0 and it > 0:
            beta *= 0.7

        if it % 200 == 0:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    m = z.max()
    e = np.exp(z - m)
    return e / e.sum()

            mc_r2b = float(c_r2b.max())
            bs = [0.0001 * mc_r2b, 0.00005 * mc_r2b, 0.00001 * mc_r2b,
                  0.000005 * mc_r2b, 0.000001 * mc_r2b]
            a_r2, _ = _adam_logspace(
                best_seq, time.monotonic() + per_a,
                bs, lr_init=0.005, max_iter_total=1000000
            )
            sc_r2 = _score(a_r2)
            if sc_r2 < best_score:
                best_score = sc_r2
                best_seq = a_r2.copy()
                print(f"  Phase 5: score={sc_r2:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: n=8192 exploration (1450-1650s) ===
    print("Phase 6: n=8192")
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end - 30 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interp n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        c_big = _convolve(a_big, a_big)
        mc_big = float(c_big.max())

        # FISTA on big
        per_f = min(30.0, (phase6_end - time.monotonic()) / 2 - 5)
        if per_f > 10:
            a_big, _ = _fista(a_big, 0.005 * mc_big, time.monotonic() + per_f, 100000)
            a_big = a_big / a_big.sum()
            sc_bf = _score(a_big)
            if sc_bf < best_score:
                best_score = sc_bf
                best_seq = a_big.copy()
                print(f"  n=8192 FISTA: score={sc_bf:.10f}")

        # Adam on big
        per_a = phase6_end - time.monotonic() - 5
        if per_a > 10:
            c_bo = _convolve(a_big, a_big)
            mc_bo = float(c_bo.max())
            bs = [0.001 * mc_bo, 0.0005 * mc_bo, 0.0001 * mc_bo,
                  0.00005 * mc_bo, 0.00001 * mc_bo]
            a_big, _ = _adam_logspace(
                a_big, time.monotonic() + per_a,
                bs, lr_init=0.008, max_iter_total=500000
            )
            sc_ba = _score(a_big)
            if sc_ba < best_score:
                best_score = sc_ba
                best_seq = a_big.copy()
                print(f"  n=8192 Adam: score={sc_ba:.10f}")

    # === Phase 7: Final coordinate descent on best found (1650-1780s) ===
    print("Phase 7: Final coordinate descent")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_fin = best_seq.copy()
        a_fin = a_fin / a_fin.sum()
        n_fin = len(a_fin)
        # For large n, limit the coordinate descent
        max_sweeps_fin = 500 if n_fin <= 8192 else 100
        a_fin, sc_fin = _coord_descent_proper(
            a_fin, deadline - 10, max_sweeps=max_sweeps_fin
        )
        sc_fin_full = _score(a_fin)
        print(f"  Final CD: score={sc_fin_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_fin_full < best_score:
            best_score = sc_fin_full
            best_seq = a_fin.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
