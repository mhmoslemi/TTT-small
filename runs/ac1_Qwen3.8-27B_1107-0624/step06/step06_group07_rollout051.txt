Multi-strategy optimizer using FISTA acceleration, momentum-based
projected subgradient, and structured Fourier perturbations to minimize
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """FISTA (Nesterov accelerated gradient) on the smoothed objective."""
    x = z0.copy()
    y = z0.copy()

    g0 = _smograd(y, beta, n)
    g0_norm = np.linalg.norm(g0)
    if g0_norm > 1e-15:
        L = 2.0 * n * n / max(beta, 1e-10)  # rough upper bound
    else:
        L = 1.0
    
    # Better: estimate L from first few steps
    L = 4.0 * n * n  # conservative

        if i % 100 == 0:

Actually, one more important idea: **use the structure of the active set**. At the optimum, the set of lags where `conv(a,a)_k` achieves its maximum is the "active set". If we can characterize this set, we can set up a system of equations.

At the optimum of `min max_k conv(a,a)_k` subject to `sum(a)=1, a>=0`:
- By KKT conditions, there exist multipliers `lambda_k >= 0` (sum to 1) for the active lags, and multipliers `mu_i >= 0` for the non-negativity constraints.
- The stationarity condition: `sum_k lambda_k * ∇conv(a,a)_k + mu - nu * 1 = 0` (where nu is the multiplier for sum(a)=1).

This gives: `sum_k lambda_k * 2 * C_k a + mu = nu * 1`.

In components: `2 * sum_k lambda_k * (C_k a)_i + mu_i = nu` for all i.

For i where `a_i > 0`: `mu_i = 0`, so `2 * sum_k lambda_k * (C_k a)_i = nu`.
For i where `a_i = 0`: `2 * sum_k lambda_k * (C_k a)_i + mu_i = nu` with `mu_i >= 0`.

This system of equations is what we're trying to solve implicitly via optimization.

A "KKT polishing" algorithm would:
1. Identify the active set (lags where conv ≈ max)
2. Identify the support (indices where a > 0)
3. Solve the KKT equations for the current active set and support
4. Check if the solution is feasible (a >= 0, correct active set)
5. If not, update the active set or support

This is essentially what the LP-based approach in the problem description does! Let me implement a version of this.

**KKT-based polishing**:
1. Given current `a`, identify active lags `K* = {k : conv(a,a)_k >= max - eps}` and support `S = {i : a_i > eps_a}`.
2. On the support S, we want `sum_{k in K*} lambda_k * (C_k a)_i = nu` for all `i in S` (equal gradient condition).
3. This means: `sum_{k in K*} lambda_k * (C_k a)` should be constant on S.
4. We can find the direction that most reduces the maximum by solving:
   - Min: `a_new^T (sum_k lambda_k C_k) a_new`
   - s.t.: `a_new >= 0`, `sum(a_new) = 1`, `a_new` supported on S (or expanded S)
   
   This is a QP! The matrix `sum_k lambda_k C_k` is symmetric. If it's PSD (which it might not be), this is convex.

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_at(a, n, tol_frac=1e-4):
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter=5000, ftol=1e-18, gtol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 60,
                                   'maxls': 40, 'disp': False})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _adaptive_lbfgs(a_start, deadline, min_time_remaining=100.0):
    """Adaptive geometric beta annealing with L-BFGS."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    current_a = a_start.copy()
    current_sc = _score(current_a)

    # Geometric beta schedule
    beta = max_c * 0.2
    min_beta = max_c * 5e-7
    ratio = 0.5

    while beta > min_beta:
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 20:
            break

        # Estimate iterations based on n and available time
        est_per_iter = 0.002 * (n / 2048.0) ** 1.2
        max_iter = min(20000, max(200, int(remaining * 0.6 / est_per_iter)))

        t0 = time.monotonic()
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        el = time.monotonic() - t0
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
            beta *= ratio
        else:
            # No improvement, try smaller perturbation or break
            beta *= 0.7
            if beta < min_beta:
                break

    return current_a


def _nesterov_subgrad(a_init, deadline, tol_frac=1e-4, max_iter=80000):
    """Nesterov-accelerated projected subgradient on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_sc = _score(a)

    # Estimate Lipschitz-like constant for step sizing
    g0 = _subgrad_at(y, n, tol_frac)
    g0_norm = np.linalg.norm(g0)
    base_step = 1.0 / max(g0_norm, 1e-10)
    base_step = min(base_step, 0.5)

    for i in range(max_iter):
        if time.monotonic() > deadline:
            break

        g = _subgrad_at(y, n, tol_frac)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        # Polyak step size with decay
        step = base_step / np.sqrt(i + 1)
        step = min(step, 0.3)

        # Gradient step
        x = y - step * g
        x = _project_simplex(x)

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x + (t_mom - 1.0) / t_new * (x - y)
        y = _project_simplex(y)
        t_mom = t_new

        # Check periodically
        if (i + 1) % 2000 == 0:
            sc = _score(y)
            if sc < best_sc:
                best_sc = sc
                best_a = y.copy()
            if (i + 1) % 10000 == 0:
                print(f"    Nesterov iter {i+1}: sc={sc:.10f}, best={best_sc:.10f}")

    sc = _score(y)
    if sc < best_sc:
        best_sc = sc
        best_a = y.copy()

    return best_a, best_sc


def _peak_target_perturb(a, rng, n_peaks=5):
    """Perturb the sequence to target the worst autocorrelation peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()

    # Find the top peaks
    peak_indices = np.argsort(c)[::-1][:n_peaks]

    # For each peak at lag k, the gradient of conv(a,a)_k is 2*(C_k a)
    # which corresponds to a reversed/shifted copy of a.
    # We want to move in the direction that reduces these peaks.
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for k in peak_indices:
        # Gradient of conv(a,a)_k w.r.t. a
        a_rev = a[::-1]
        # (C_k a)_i = a_rev[k-i] (with bounds checking)
        grad_k = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad_k[i] = a[j]
        # Normalize and subtract
        grad_k = grad_k - grad_k.mean()
        gnorm = np.linalg.norm(grad_k)
        if gnorm > 1e-15:
            grad_k /= gnorm
        z = z - 0.01 * grad_k

    z = np.clip(z, -50, 50)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=5, amp_scale=0.01):
    """Perturb using random Fourier modes."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 6)))
        amp = amp_scale * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _gauss_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=50000):
    """Direct-space projected subgradient with adaptive step and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    # Use momentum
    v = np.zeros(n)
    momentum = 0.9
    step = 0.05
    stall = 0
    no_improve = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 2.0 * (h - W)
        grad = grad - grad.mean()  # Remove component in null direction

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

        # Momentum update
        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            step *= 0.5
            v *= 0.5
            continue
        a_new = a_new / s
        sc_new = _score(a_new)

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if sc_new <= _score(a) + 1e-12:
            a = a_new
            step = min(step * 1.01, 0.1)
            stall = 0
        else:
            stall += 1
            step *= 0.8
            v *= 0.5

        step *= (1.0 - 5e-6 * i)
        step = max(step, 1e-12)

        if stall > 100:
            break
        if no_improve > 50000:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: adaptive L-BFGS, Nesterov subgradient,
    peak-targeted perturbations, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(4096, "uniform")
        print(f"No parent. Uniform n=4096, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Adaptive L-BFGS (main) ===
    print(f"\n=== Phase 1: Adaptive L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _adaptive_lbfgs(best_seq, deadline, min_time_remaining=600.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Nesterov-accelerated subgradient ===
    print(f"\n=== Phase 2: Nesterov subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_ns, sc_ns = _nesterov_subgrad(best_seq, deadline - 400,
                                        tol_frac=1e-4, max_iter=60000)
        try_update(a_ns, "phase2_nesterov")

    # === Phase 3: Direct polish with momentum ===
    print(f"\n=== Phase 3: Direct polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_dp, sc_dp = _direct_polish(best_seq, deadline - 300,
                                     tol_frac=1e-5, max_steps=40000)
        try_update(a_dp, "phase3_direct")

    # === Phase 4: Fine L-BFGS at very small beta ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - 150 - time.monotonic()
            if remaining < 15:
                break
            max_iter = min(10000, max(500, int(remaining / 0.003)))
            a_new = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")
            # Re-derive best_seq for next iteration
            if best_seq is not None:
                pass  # best_seq already updated by try_update

    # === Phase 5: Peak-targeted restarts ===
    print(f"\n=== Phase 5: Peak-targeted restarts (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        perturbations = [
            ("peak_target", lambda: _peak_target_perturb(best_seq, rng, 3)),
            ("peak_target5", lambda: _peak_target_perturb(best_seq, rng, 5)),
            ("fourier_3", lambda: _fourier_perturb(best_seq, rng, 3, 0.015)),
            ("fourier_8", lambda: _fourier_perturb(best_seq, rng, 8, 0.01)),
            ("gauss_003", lambda: _gauss_perturb(best_seq, 0.03, rng)),
            ("gauss_008", lambda: _gauss_perturb(best_seq, 0.08, rng)),
        ]

        for name, perturb_fn in perturbations:
            if time.monotonic() > deadline - 120:
                break
            time_per_restart = 120.0
            if time.monotonic() + time_per_restart > deadline - 80:
                time_per_restart = max(30, deadline - 80 - time.monotonic())

            p_start = perturb_fn()
            sc_start = _score(p_start)
            if sc_start < best_score_val:
                try_update(p_start, f"perturb_{name}")
                continue

            # Quick L-BFGS from perturbed start
            n = len(p_start)
            c_s = _full_convolve(p_start, p_start)
            mb_s = float(c_s.max())
            beta = mb_s * 0.05

            remaining = deadline - 80 - time.monotonic()
            if remaining < 20:
                break
            max_iter = min(8000, max(500, int(remaining * 0.7 / 0.003)))
            a_r = _lbfgs_run(p_start, beta, n, max_iter)

            # Second pass with smaller beta
            if time.monotonic() < deadline - 60:
                c_r = _full_convolve(a_r, a_r)
                beta2 = float(c_r.max()) * 0.01
                max_iter2 = min(5000, max(200, int((deadline - 60 - time.monotonic()) / 0.003)))
                a_r2 = _lbfgs_run(a_r, beta2, n, max_iter2)
                try_update(a_r2, f"{name}_fine")

            try_update(a_r, f"restart_{name}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        cur_n = len(best_seq)
        for target_n in [2048, 8192]:
            if time.monotonic() > deadline - 60:
                break
            if target_n == cur_n:
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _adaptive_lbfgs(a_interp, deadline, min_time_remaining=40.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 7: Extended final polishing cycle ===
    print(f"\n=== Phase 7: Final polishing (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        for rnd in range(6):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * (5e-6 * (0.5 ** rnd))
            remaining = deadline - 5 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(5000, max(300, int(remaining * 0.4 / 0.0025)))
            a_f = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Direct polish
            if time.monotonic() < deadline - 5:
                a_s, _ = _direct_polish(best_seq, deadline - 5,
                                        tol_frac=1e-6, max_steps=15000)
                try_update(a_s, f"final_direct_{rnd}")

            # Nesterov subgradient
            if time.monotonic() < deadline - 5:
                a_ns, _ = _nesterov_subgrad(best_seq, deadline - 5,
                                            tol_frac=1e-5, max_iter=20000)
                try_update(a_ns, f"final_nesterov_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
