"""
Multi-phase optimizer using exponentiated gradient, FISTA, and Adam in log-space
to minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements: adaptive beta curriculum, peak-targeted reweighting,
multiplicative updates with Nesterov momentum, and multi-restart diversification.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_weights(c, beta):
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _grad_conv(a, w, n):
    """Gradient of sum_k w[k]*(conv(a,a))[k] wrt a.
    Result: grad[m] = 2 * sum_k w[k] * a[k-m] (with a[j]=0 for out of bounds)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _grad_norm(a, w, n):
    """Normalized gradient for the objective 2n*max(c) with softmax smoothing."""
    g = _grad_conv(a, w, n)
    W = float(w.sum())
    if W < 1e-30:
        W = 1.0
    return 2.0 * n * g / W


def _topk_weights(c, k_frac, beta=None):
    """Create weights concentrated on the top k_frac of convolution values."""
    n_c = len(c)
    k = max(1, int(k_frac * n_c))
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(k, n_c - 1)]
    if beta is None:
        beta = 0.01 * (c.max() - threshold + 1e-30)
    shifted = (c - c.max()) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _fista_step(a_init, beta, deadline, max_iter=5000):
    """FISTA on the simplex with softmax-smoothed max convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())
    L = 4.0 * n * max_c / (beta * len(c_curr))
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _softmax_weights(c_y, beta)
        grad = _grad_conv(y, w, n)
        W = float(w.sum())
        full_grad = 2.0 * n * grad / W

        a_next = _proj_simplex(y - step * full_grad)

        if it % 500 == 0 or it < 10:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
            c_new = _full_convolve(a_next, a_next)
            max_c_new = float(c_new.max())
            L = 4.0 * n * max_c_new / (beta * len(c_new))
            step = 1.0 / L

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _exp_gradient(a_init, beta, deadline, lr_init=0.1, momentum=0.9):
    """Exponentiated gradient (natural gradient) with Nesterov momentum in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)  # momentum buffer
    t = 0
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    current_lr = lr_init

    while time.monotonic() < deadline:
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _full_convolve(a, a)
        w = _softmax_weights(c, beta)
        grad_z = _grad_conv(a, w, n)
        W = float(w.sum())
        grad_z = 2.0 * n * grad_z / W

        # Natural gradient in log-space
        nat_grad = a * grad_z
        nat_grad = nat_grad - np.mean(nat_grad)  # center for scale invariance

        # Nesterov momentum
        v = momentum * v + nat_grad
        update = current_lr * v

        # Clip update for stability
        update_norm = np.linalg.norm(update)
        if update_norm > 2.0:
            update = update * (2.0 / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % 200 == 0:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        # Decay learning rate
        if t % 2000 == 0:
            current_lr *= 0.95
            current_lr = max(current_lr, 0.001)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_list, lr_init=0.01, max_iter_total=500000):
    """Adam optimizer in log-space with decreasing beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        phase_iter = max(1000, int((deadline - time.monotonic()) * 500))
        while it_in_phase < phase_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()

            c = _full_convolve(a, a)
            w = _softmax_weights(c, beta)
            grad_z = _grad_conv(a, w, n)
            W = float(w.sum())
            grad_z = 2.0 * n * grad_z / W

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            # Center the update
            update = update - np.mean(update)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.5:
                update = update * (1.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 100 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 5000 == 0:
                current_lr = max(0.0003, current_lr * 0.9)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _peak_reducing_step(a, n_peaks=5, n_iters=100):
    """Directly target the top peaks of the convolution and reduce them."""
    n = len(a)
    a_work = a.copy()
    a_work = a_work / a_work.sum()
    best_a = a_work.copy()
    best_score = _score_norm(a_work)

    for _ in range(n_iters):
        c = _full_convolve(a_work, a_work)
        max_c = float(c.max())
        # Get top peak positions
        top_idx = np.argsort(c)[::-1][:n_peaks]
        # Use top-k weighted gradient
        k = max(n_peaks, len(c) // 50)
        w = _topk_weights(c, k / len(c), beta=0.005 * max_c)

        # Compute which coordinates contribute most to peaks
        a_rev = a_work[::-1]
        wc = _full_convolve(w, a_rev)

        # Reduce high-contribution coordinates

        sc = _score_norm(a_try)
        if sc < best_score:

def _coordinate_polish(a, deadline, n_trials=5000):
    """Random coordinate perturbation polish."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        # Pick 1-3 random coordinates
        n_mod = rng.integers(1, 4)
        idxs = rng.integers(0, n, size=n_mod)
        a_try = best_a.copy()
        for idx in idxs:
            a_try[idx] *= factors[rng.integers(0, len(factors))]

            sc = _score_norm(a_try)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()

    a_work, sc = _adam_logspace(a_work, adam_deadline, betas_fine, lr_init=0.012)
    a_work = a_work / a_work.sum()
    sc_full = _score(a_work)
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_work.copy()
        print(f"  Adam: score={sc_full:.10f}")

    # ========== PHASE 4: Peak-targeted reduction ==========
    print(f"\nPhase 4: Peak reduction (score={best_score:.10f})")
    if time.monotonic() < deadline - 60:
        a_work, sc = _peak_reducing_step(a_work, n_peaks=10, n_iters=200)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  Peak reduction: score={sc_full:.10f}")

    # ========== PHASE 5: Multi-restart diversification ==========
    print(f"\nPhase 5: Multi-restart (score={best_score:.10f})")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()
        n_base = len(a_base)

        c_base = _full_convolve(a_base, a_base)
        max_cb = float(c_base.max())

        restarts = [
            ("small_perturb", lambda: _perturb(a_base, 0.01, rng)),
            ("medium_perturb", lambda: _perturb(a_base, 0.03, rng)),
            ("shift", lambda: np.roll(a_base, rng.integers(1, 64)) / np.roll(a_base, rng.integers(1, 64)).sum()),
            ("blend_sine", lambda: (0.8 * a_base + 0.2 * _make_profile(n_base, "sinusoidal"))),
            ("blend_taper", lambda: (0.8 * a_base + 0.2 * _make_profile(n_base, "tapered"))),
        ]
        # Remove lambda closures - use explicit functions
        def _restart_small():
            return _perturb(a_base, 0.01, rng)
        def _restart_medium():
            return _perturb(a_base, 0.03, rng)
        def _restart_shift():
            sh = rng.integers(1, 64)
            r = np.roll(a_base, sh)
            return r / r.sum()
        def _restart_sine():
            return (0.8 * a_base + 0.2 * _make_profile(n_base, "sinusoidal"))
        def _restart_taper():
            return (0.8 * a_base + 0.2 * _make_profile(n_base, "tapered"))

        restart_fns = [
            ("small_perturb", _restart_small),
            ("medium_perturb", _restart_medium),
            ("shift", _restart_shift),
            ("blend_sine", _restart_sine),
            ("blend_taper", _restart_taper),
        ]

        n_restarts = len(restart_fns)
        per_restart_time = (deadline - time.monotonic() - 60.0) / max(1, n_restarts)

        for name, fn in restart_fns:
            if time.monotonic() > deadline - 60:
                break
            a_try = fn()
            if a_try is None:
                continue
            a_try = np.maximum(a_try, 1e-10)
            a_try = a_try / a_try.sum()

            c_try = _full_convolve(a_try, a_try)
            max_ct = float(c_try.max())
            betas_r = [0.0005 * max_ct, 0.0002 * max_ct, 0.0001 * max_ct,
                       0.00005 * max_ct, 0.00002 * max_ct]
            r_deadline = time.monotonic() + max(10.0, per_restart_time)
            a_opt, sc_opt = _adam_logspace(a_try, r_deadline, betas_r, lr_init=0.008)
            a_opt = a_opt / a_opt.sum()
            sc_full = _score(a_opt)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_opt.copy()
                print(f"  Restart '{name}': score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s")

    # ========== PHASE 6: Try n=8192 ==========
    print(f"\nPhase 6: Try n=8192 (score={best_score:.10f})")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        a_big = a_big / a_big.sum()
        c_big = _full_convolve(a_big, a_big)
        max_cbig = float(c_big.max())

        # Quick optimization on n=8192
        betas_big = [0.002 * max_cbig, 0.001 * max_cbig, 0.0005 * max_cbig,
                     0.0002 * max_cbig, 0.0001 * max_cbig]
        big_deadline = min(deadline - 60, time.monotonic() + 200.0)
        a_opt_big, sc_big = _adam_logspace(a_big, big_deadline, betas_big, lr_init=0.01)
        a_opt_big = a_opt_big / a_opt_big.sum()
        sc_big_full = _score(a_opt_big)
        if sc_big_full < best_score:
            best_score = sc_big_full
            best_seq = a_opt_big.copy()
            print(f"  n=8192: score={sc_big_full:.10f}")
        else:
            print(f"  n=8192: score={sc_big_full:.10f} (not better)")

    # ========== PHASE 7: Ultra-fine Adam polish ==========
    print(f"\nPhase 7: Ultra-fine polish (score={best_score:.10f})")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp,
                     0.000001 * max_cp, 0.0000005 * max_cp]
        pol_deadline = min(deadline - 10, time.monotonic() + 120.0)
        a_pol, sc_pol = _adam_logspace(a_pol, pol_deadline, betas_ult, lr_init=0.003)
        a_pol = a_pol / a_pol.sum()
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Ultra-fine: score={sc_pol_full:.10f}")

    # ========== PHASE 8: Coordinate descent polish ==========
    print(f"\nPhase 8: Coordinate polish (score={best_score:.10f})")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_cd, sc_cd = _coordinate_polish(best_seq, time.monotonic() + min(30.0, deadline - time.monotonic() - 5))
        sc_cd_full = _score(a_cd)
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()
            print(f"  Coord descent: score={sc_cd_full:.10f}")

    # ========== FINAL ==========
    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
            s = float(final.sum())
        sc_final = _score(final)
        print(f"Final score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
