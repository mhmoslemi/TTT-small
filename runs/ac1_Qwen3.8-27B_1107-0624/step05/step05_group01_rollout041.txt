One idea that might help: **alternating between reducing individual peaks and balancing them**.

Algorithm:
1. Find the current peak: k* = argmax(c).
2. Reduce c[k*] by adjusting a: find the direction that most reduces c[k*].
3. After reducing c[k*], check if other peaks have increased. If so, reduce them.
4. Repeat.

This is similar to the "cyclic coordinate descent" approach but at the "peak level" instead of the "coordinate level".

The Lagrangian is:
L(a, t, λ, μ, ν) = t + λ(sum(a) - 1) + sum_k μ_k((a*a)[k] - t) - sum_i ν_i a_i

For the inf over t: the coefficient of t is `1 - sum_k μ_k`. For the inf to be finite, we need `sum_k μ_k = 1`. Then the t terms cancel.

The inf over a is finite only if the quadratic form is PSD and the linear term is non-positive. This is getting complicated.

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp Adam, subgradient method, and coordinate descent.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _corr_full(p, a):
    """Compute np.correlate(p, a, 'full') using FFT."""
    out_n = len(p) + len(a) - 1
    N = _next_pow2(out_n)
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(P * np.conj(A), N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score with sum(a) assumed to be 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def _binomial_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return _normalize(a)


def _exp_init(n, rate=3.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * (t - 0.5))
    return _normalize(a)


def _u_shape_init(n):
    t = np.linspace(0, 1, n)
    a = 1.0 + 4.0 * (t - 0.5)**2
    return _normalize(a)


def _random_smooth_init(n, seed):
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n))
    # Smooth
    k = max(3, n // 100)
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    return _normalize(a)


def _interp_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return _normalize(a)
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    return _normalize(a_new)


def _lse_obj_grad(a, beta, n):
    """
    Compute the log-sum-exp objective and its gradient w.r.t. a.
    Objective: 2n * LSE(c / beta) where c = conv(a, a).
    Returns (obj, grad) where grad is d(obj)/d(a).
    """
    c = _full_convolve(a, a)
    c_shift = c / beta
    c_max = float(c_shift.max())
    e = np.exp(c_shift - c_max)
    s = e.sum()
    p = e / s  # softmax weights, length 2n-1
    obj = float(2.0 * n * (c_max + np.log(s)))
    
    # Gradient: (2n/beta) * corr(p, a)[:n]
    h = _corr_full(p, a)[:n]
    grad = (2.0 * n / beta) * h
    
    # Project to sum=0 direction (since we'll renormalize after each step)
    grad = grad - np.sum(grad) / n
    
    return obj, grad


def _opt_lse_adam(a, deadline, lr=0.1, beta_start=0.1, beta_end=0.001, n_steps_anneal=500):
    """
    Optimize using log-sum-exp with Adam optimizer and beta annealing.
    """
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_norm(best_a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Initial beta
    c0 = _full_convolve(a, a)
    M0 = float(c0.max())
    beta = beta_start * M0
    
    no_improve = 0
    check_every = 20
    
    while time.monotonic() < deadline:
        t += 1
        
        # Anneal beta
        progress = min(1.0, t / max(1, n_steps_anneal))
        beta = M0 * beta_start * (beta_end / beta_start) ** progress
        beta = max(beta, 1e-12 * M0)
        
        obj, grad = _lse_obj_grad(a, beta, n)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        # Adaptive learning rate
        lr_t = lr * 0.5 ** (no_improve // 10)
        
        a = a - lr_t * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0.0)
        a = _normalize(a)
        
        # Check improvement
        if t % check_every == 0:
            sc = _score_norm(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                # Restart with perturbation
                a = best_a + 0.01 * np.random.randn(n)
                a = _normalize(a)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
    
    return best_a, _score_norm(best_a)


def _opt_subgrad(a, deadline, lr=0.01, momentum=0.9):
    """
    Optimize using exact max subgradient with momentum.
    """
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score_norm(best_a)
    
    vel = np.zeros(n)
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        M = float(c.max())
        sc = 2.0 * n * M
        
        # Find active set (peaks within 99% of max)
        threshold = 0.99 * M
        active = np.where(c >= threshold)[0]
        
        # Compute average subgradient over active set
        # grad[j] = mean over active k of a[k-j] (if valid)
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        grad /= len(active)
        
        # Project to sum=0
        grad = grad - np.sum(grad) / n
        
        # Momentum update
        vel = momentum * vel - lr * grad
        a = a + vel
        a = np.maximum(a, 0.0)
        a = _normalize(a)
        
        # Decay learning rate
        lr *= 0.9999
        
        # Check
        sc_new = _score_norm(a)
        if sc_new < best_sc - 1e-10:
            best_sc = sc_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                a = best_a + 0.005 * np.random.randn(n)
                a = _normalize(a)
                vel = np.zeros(n)
                lr = 0.01
                no_improve = 0
    
    return best_a, best_sc

The optimal δ minimizes `max(c[k] + δ * a[k-j] for k in [j, j+n-1])` subject to `a[j] + δ >= 0`.

2. The "most constraining" line is the one that achieves M_aff: `k* = argmax(c[j:j+n])`.
3. If we decrease a[j] (δ < 0), the affected values decrease (since a[k-j] >= 0). So the max in the affected region will decrease.
4. But the unconstrained values don't change.
5. So the optimal δ is the one that makes `max(c[j:j+n] + δ*a)` equal to `max(c outside [j,j+n])` (or the boundary δ = -a[j]).

Actually, for the 1D problem `min_δ max_k (g[k] + δ * s[k])`:
- This is equivalent to finding the lowest horizontal line that intersects all the lines `g[k] + δ * s[k]` for some δ.
- The answer is: δ* is the value where the "upper envelope" of the lines achieves its minimum.
- This can be found by: sort the lines by slope, then find the "median crossing point".

For a quick approximation:
- If the max of g is at index k1 and the max of s is at index k2:
  - If k1 == k2: δ* = -g[k1]/s[k1] (if s[k1] > 0), which makes g[k1] + δ*s[k1] = 0.
  - If k1 != k2: δ* = (g[k2] - g[k1]) / (s[k1] - s[k2]) (the crossing point of the top two lines).

At each step:

2. Find the max: M = max(c), and the set of indices S where c >= (1-eps)*M.
3. Compute the "peak gradient": g = (1/|S|) * sum_{k in S} g_k, where g_k[j] = a[k-j] (if valid).
4. Project g to the sum=0 hyperplane: g = g - mean(g).
5. Take a step: a = a - step * g.
6. Clip and renormalize.

The step size is crucial. If too large, we might oscillate. If too small, we make slow progress.

def _coord_descent(a, deadline):

    c = _conv(a, a).copy()

    sweeps = 0

        any_improve = False
        for j in range(n):

            if a_orig < 1e-12:

            else:

                unaff_max = 0.0
                if j > 0:
                    unaff_max = max(unaff_max, float(c[:j].max()))
                end = j + n
                if end < 2 * n - 1:
                    unaff_max = max(unaff_max, float(c[end:].max()))
                gm = max(aff_max, unaff_max)
                if gm < best_cand_max:
                    best_cand_max = gm

        print(f"FINAL: score={sc:.10f} n={len(final)}")
        if len(final) <= 10000:
            try:
                ev = float(evaluate_sequence([float(x) for x in final]))
                if np.isfinite(ev):
                    print(f"Evaluator: {ev:.10f}")
                else:
                    print("Evaluator inf, using fallback")
                    return [float(x) for x in _init_binomial(512)]
            except Exception:
                pass
        return [float(x) for x in final]

    fb = _init_binomial(512)
    return [float(x) for x in fb]
