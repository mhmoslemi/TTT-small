Multi-strategy optimization for minimum peak autocorrelation sequences.
Uses SDP relaxation for warm start, exponentiated gradient with annealing,
L-BFGS refinement, and Adam polish.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(p):
    n = len(p)
    s = p.sum()
    if s < 1e-15:
        return np.inf
    p = p / s
    c = _fft_conv(p, p)
    return 2.0 * n * c.max()

def _norm_score(p):
    """Score assuming p is already normalized (sum=1)."""
    n = len(p)
    c = _fft_conv(p, p)
    return 2.0 * n * c.max()

    m = c.max()
