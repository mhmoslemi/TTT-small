"""
Optimized step function using binomial initialization and spectral-domain L-BFGS.
Key improvements over previous approach:
1. Binomial coefficient initialization C(2k,k)/4^k - theoretically optimal infinite sequence
2. FFT-based gradient computation O(n log n) per evaluation
3. Aggressive logsumexp continuation with adaptive beta
4. Active-set equalization (Remez-like) for final polishing
5. Multi-scale coarse-to-fine optimization
6. Peak-targeted local search with analytical moves

import numpy as np
from scipy.optimize import minimize
from scipy.special import comb
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve_fft(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve_fft(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve_fft(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_binomial_profile(n, tail_power=2.0):
    k = np.arange(n, dtype=np.float64)
    # C(2k, k) / 4^k using log for stability
    log_a = np.zeros(n)
    for i in range(1, n):
        log_a[i] = log_a[i-1] + np.log(2*i) + np.log(2*i-1) - 2*np.log(4*i)
    # More stable: use recurrence a[k] = a[k-1] * (2k-1)/(2k)
    a = np.ones(n)
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    # Apply tail window to taper the end
    t = k / max(1, n - 1)
    w = (1.0 - t) ** tail_power
    w = np.maximum(w, 1e-15)
    a = a * w
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_binomial_profile_v2(n, taper_start=0.7, taper_power=2.0):
    """Binomial profile with only tail tapering (keep first part flat)."""
    a = np.ones(n)
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)
    t = np.arange(n) / max(1, n - 1)
    w = np.ones(n)
    mask = t > taper_start
    t_rel = (t[mask] - taper_start) / (1.0 - taper_start)
    w[mask] = (1.0 - t_rel) ** taper_power
    a = a * w
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_one_sided_power(n, p=0.3):
    """One-sided profile: high at start, decays as (1-k/n)^p."""
    t = np.arange(n) / max(1, n - 1)
    a = (1.0 - t) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_hann_one_sided(n):
    """Hann window applied one-sided."""
    t = np.arange(n) / max(1, n)
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _compute_logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve_fft(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm
    log_s = float(np.log(s))
    return 2.0 * n * (cm + beta * log_s)


def _compute_logsumexp_grad(z, beta, n, cache=None):
    a = _softmax_fn(z)
    c = _full_convolve_fft(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # grad_conv[i] = 2 * sum_j w[i+j] * a[j] (cross-correlation)
    # = convolve(w, a_reversed)[i + n - 1]
    a_rev = a[::-1]
    N = _next_pow2(len(w) + len(a_rev) - 1)
    W_fft = np.fft.rfft(w, N)
    A_fft = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(W_fft * A_fft, N)
    grad_conv = conv_result[n-1:2*n-1]  # length n
    grad_conv *= 2.0

    # Full gradient of J = 2n * logsumexp
    grad_J = 2.0 * n * grad_conv

    # Chain rule for softmax: dJ/dz[i] = a[i]*(dJ/da[i] - sum_j a[j]*dJ/da[j])
    mean_grad = float(np.dot(a, grad_J))
    grad_z = a * (grad_J - mean_grad)

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros_like(z)

    return grad_z, (a, c, cm, w)


def _run_lbfgs_logsumexp(a_init, deadline, beta_scale=1.0, min_beta=0.00002, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve_fft(a_init, a_init)
    max_c0 = float(c0.max())

    # Beta schedule: geometric progression
    n_betas = 15
    betas = max_c0 * beta_scale * (min_beta / (max_c0 * beta_scale)) ** (np.arange(n_betas) / (n_betas - 1))
    betas = np.sort(betas)[::-1]  # descending

    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / n_betas)
    else:
        per_level = 30.0

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj_fn(z_):
            return _compute_logsumexp_obj(z_, beta, n)

        def grad_fn(z_):
            g, _ = _compute_logsumexp_grad(z_, beta, n)
            return g

        z_work = z.copy()
        try:
            result = minimize(
                obj_fn, z_work, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_adaptive_continuation(a_init, deadline, rng):

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve_fft(a_init, a_init)
    max_c0 = float(c0.max())

    # Start with moderate beta, decrease adaptively
    beta = max_c0 * 0.05
    min_beta = max_c0 * 0.00001

    while beta > min_beta and time.monotonic() < deadline - 3.0:
        # Run L-BFGS at current beta
        level_deadline = min(deadline, time.monotonic() + max(5.0, 60.0))

        def obj_fn(z_):
            return _compute_logsumexp_obj(z_, beta, n)

        def grad_fn(z_):
            g, _ = _compute_logsumexp_grad(z_, beta, n)
            return g

        z_work = z.copy()
        prev_obj = _compute_logsumexp_obj(z_work, beta, n)

        try:
            result = minimize(
                obj_fn, z_work, jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-9,
                         'maxls': 40, 'maxfun': 8000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

        # Check convergence: if objective didn't change much, reduce beta
        if time.monotonic() > level_deadline:
            beta *= 0.5
        else:
            # Converged relatively quickly, reduce beta more aggressively
            beta *= 0.4

    return best_a, best_score


def _active_set_equalization(a_init, deadline, n_active=5, max_iters=200):
    """Remez-like: iteratively equalize the top K convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve_fft(a, a)
        # Find top K peaks
        K = min(n_active, len(c))
        top_idx = np.argsort(c)[-K:]
        top_vals = c[top_idx]
        target = float(top_vals.mean())

        # Compute gradient combining all active peaks
        # grad[i] = sum_{m in active} w_m * 2*a[m-i]
        w_active = np.ones(K) / K
        grad = np.zeros(n)
        for wi, mi in zip(w_active, top_idx):
            mi = int(mi)
            for i in range(n):
                j = mi - i
                if 0 <= j < n:
                    grad[i] += wi * 2.0 * a[j]

        # Center the gradient to preserve sum
        grad_centered = grad - grad.mean()

        # Adaptive step size: try to make all active peaks equal
        # Reduction in c[m] for peak m: delta_c[m] = -step * sum_j grad[m-j]*a[j]... 
        # Simpler: use line search
        best_step = 0.0
        best_obj = best_score
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = a - frac * grad_centered
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_obj:
                best_obj = sc
                best_step = frac

        if best_step == 0.0:
            break

        a = a - best_step * grad_centered
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a /= s

        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _peak_local_search(a_init, deadline, n_trials=50000, rng=None):
    """Fast local search targeting peaks."""
    if rng is None:
        rng = np.random.default_rng(42)
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve_fft(a, a)
        max_lag = int(np.argmax(c))

        if it % 3 == 0:
            # Reduce the element contributing most to the peak
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                idx = int(pairs[np.argmax(products)])

                a_trial = a.copy()
                frac = float(rng.uniform(0.3, 0.7))
                delta = a_trial[idx] * frac
                a_trial[idx] = max(0.0, a_trial[idx] - delta)

                # Add to safe position (low contribution to top peaks)
                top_lags = np.argsort(c)[-3:]
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo = max(0, kl - (n - 1))
                    hi = min(n - 1, kl)
                    contrib[lo:hi+1] += a[kl-hi:kl-lo+1][::-1]

                safe = np.argsort(contrib)[:max(1, n // 4)]
                add_idx = int(safe[rng.integers(len(safe))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm_fn(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()

        elif it % 3 == 1:
            # Block perturbation
            block_size = int(rng.integers(3, min(32, n // 16)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.75, 1.25, size=block_size)
            a_trial[start:start+block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        else:
            # Paired exchange
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = float(rng.uniform(0.1, 0.35))
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1 + amt2)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2 + amt1)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a, best_sc


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _coarse_to_fine(a_coarse, n_fine, deadline, rng):
    """Optimize at coarse scale, then refine."""
    a_fine = _interp_profile(a_coarse, n_fine)
    sc = _score_norm_fn(a_fine)

    dl = min(deadline - 10, time.monotonic() + max(30.0, (deadline - time.monotonic()) * 0.5))
    a_best, sc_best = _run_lbfgs_logsumexp(a_fine, dl, beta_scale=0.5, max_wall_time=120.0)
    return a_best, sc_best


def _gradient_descent_topK(a_init, deadline, K=5, lr_init=0.01, rng=None):
    """Gradient descent with top-K peak targeting, Adam updates."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    max_iters = 30000
    while t < max_iters:
        if time.monotonic() > deadline - 1.0:
            break
        t += 1

        a_curr = _softmax_fn(z)
        c = _full_convolve_fft(a_curr, a_curr)
        k_actual = min(K, len(c))
        top_idx = np.argsort(c)[-k_actual:]

        # Gradient at top peaks
        grad_conv = np.zeros(n)
        for mi in top_idx:
            mi = int(mi)
            for i in range(n):
                j = mi - i
                if 0 <= j < n:
                    grad_conv[i] += 2.0 * a_curr[j]
        grad_conv /= k_actual * 2.0
        grad_J = 4.0 * n * grad_conv

        # Softmax gradient
        mean_g = float(np.dot(a_curr, grad_J))
        grad_z = a_curr * (grad_J - mean_g)

        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        # Decay LR
        if t % 1000 == 0:
            lr *= 0.95

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer using binomial initialization, spectral L-BFGS,
    active-set equalization, and peak-targeted local search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # --- Choose working n ---
    n_work = 8192
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
        n_work = n_parent
    else:
        n_work = 8192
        parent = _make_binomial_profile(n_work, tail_power=2.0)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, binomial n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    print(f"Working with n={n_work}")

    # --- Phase 0: Diverse initialization (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    profiles = [
        (_make_binomial_profile(n_work, 1.5), "binom_p1.5"),
        (_make_binomial_profile(n_work, 2.0), "binom_p2.0"),
        (_make_binomial_profile(n_work, 3.0), "binom_p3.0"),
        (_make_binomial_profile(n_work, 4.0), "binom_p4.0"),
        (_make_binomial_profile_v2(n_work, 0.6, 2.0), "binom_v2_0.6"),
        (_make_binomial_profile_v2(n_work, 0.7, 2.5), "binom_v2_0.7"),
        (_make_binomial_profile_v2(n_work, 0.8, 3.0), "binom_v2_0.8"),
        (_make_one_sided_power(n_work, 0.5), "onesided_0.5"),
        (_make_one_sided_power(n_work, 0.7), "onesided_0.7"),
        (_make_one_sided_power(n_work, 1.0), "onesided_1.0"),
        (_make_hann_one_sided(n_work), "hann_onesided"),
        (_make_raised_cosine(n_work), "raised_cos"),
        (_make_triangular_weight(n_work), "triangular"),
    ]

    candidates = []
    for p, label in profiles:
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), label))
        update_best(p, sc, label)

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f} [{candidates[0][2] if candidates else 'none'}]")

    # --- Phase 1: L-BFGS continuation on top initializations (500s) ---
    phase1_end = min(deadline, t_start + 540.0)
    n_top = min(3, len(candidates))
    per_cand = max(30.0, (phase1_end - time.monotonic()) / max(1, n_top))

    for ci in range(n_top):
        if time.monotonic() > phase1_end - 30:
            break
        a_start = candidates[ci][1].copy()
        dl = min(phase1_end - 10, time.monotonic() + per_cand)
        print(f"Phase 1: L-BFGS on {candidates[ci][2]}, start={candidates[ci][0]:.10f}, time={per_cand:.0f}s")
        a_best, sc_best = _run_lbfgs_logsumexp(a_start, dl, beta_scale=1.0, max_wall_time=per_cand)
        update_best(a_best, sc_best, f"lbfgs_{candidates[ci][2]}")

    # Also run adaptive continuation on the overall best
    if time.monotonic() < phase1_end - 30 and best_seq is not None:
        dl = min(phase1_end - 10, time.monotonic() + max(30.0, (phase1_end - time.monotonic()) * 0.5))
        a_best, sc_best = _run_adaptive_continuation(best_seq, dl, rng)
        update_best(a_best, sc_best, "adaptive_cont")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Active-set equalization (200s) ---
    phase2_end = min(deadline, t_start + 740.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        dl = phase2_end - 5.0
        a_best, sc_best = _active_set_equalization(best_seq, dl, n_active=5, max_iters=300)
        update_best(a_best, sc_best, "active_set_k5")

        if time.monotonic() < phase2_end - 30:
            dl2 = phase2_end - 5.0
            a_best2, sc_best2 = _active_set_equalization(best_seq, dl2, n_active=3, max_iters=200)
            update_best(a_best2, sc_best2, "active_set_k3")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Top-K gradient descent (200s) ---
    phase3_end = min(deadline, t_start + 940.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        dl = phase3_end - 5.0
        a_best, sc_best = _gradient_descent_topK(best_seq, dl, K=5, lr_init=0.008, rng=rng)
        update_best(a_best, sc_best, "topK_k5")

        if time.monotonic() < phase3_end - 30:
            dl2 = phase3_end - 5.0
            a_best2, sc_best2 = _gradient_descent_topK(best_seq, dl2, K=8, lr_init=0.004, rng=rng)
            update_best(a_best2, sc_best2, "topK_k8")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Peak local search (200s) ---
    phase4_end = min(deadline, t_start + 1140.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl = phase4_end - 5.0
        a_best, sc_best = _peak_local_search(best_seq, dl, n_trials=40000, rng=rng)
        update_best(a_best, sc_best, "peak_search")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (200s) ---
    phase5_end = min(deadline, t_start + 1340.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 3
        per_restart = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise = 0.015 + 0.01 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _run_lbfgs_logsumexp(p_start, dl, beta_scale=0.5, max_wall_time=per_restart)
            update_best(a_best, sc_best, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different n (150s) ---
    phase6_end = min(deadline, t_start + 1490.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)

        # Try larger n
        for target_n in [n_curr * 2]:
            if target_n > 16384 or target_n <= n_curr:
                continue
            if time.monotonic() > phase6_end - 60:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp to n={target_n}: score={sc_interp:.10f}")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

            if time.monotonic() < phase6_end - 30:
                dl = min(phase6_end - 20, time.monotonic() + 80.0)
                a_best, sc_best = _run_lbfgs_logsumexp(p_interp, dl, beta_scale=0.4, max_wall_time=80.0)
                update_best(a_best, sc_best, f"lbfgs_n{target_n}")

        # Try binomial at different n
        for target_n in [4096, 16384]:
            if target_n == n_curr or time.monotonic() > phase6_end - 40:
                continue
            p_binom = _make_binomial_profile(target_n, 2.0)
            sc_b = _score_norm_fn(p_binom)
            if sc_b < best_score:
                update_best(p_binom, sc_b, f"binom_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final intensive polishing ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")

        # 7a: Fine L-BFGS
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _run_lbfgs_logsumexp(best_seq, dl, beta_scale=0.3, min_beta=0.00001, max_wall_time=remaining * 0.35)
            update_best(a_best, sc_best, "final_lbfgs")

        # 7b: Active set
        if time.monotonic() < deadline - 40:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _active_set_equalization(best_seq, dl, n_active=7, max_iters=300)
            update_best(a_best, sc_best, "final_active")

        # 7c: More local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _peak_local_search(best_seq, dl, n_trials=20000, rng=rng)
            update_best(a_best, sc_best, "final_local")

        # 7d: One more L-BFGS pass
        if time.monotonic() < deadline - 20 and remaining > 30:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.15)
            a_best, sc_best = _run_lbfgs_logsumexp(best_seq, dl, beta_scale=0.1, min_beta=0.000005, max_wall_time=remaining * 0.15)
            update_best(a_best, sc_best, "ultra_final")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_binomial_profile(8192, 2.0)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_binomial_profile(8192, 2.0)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
