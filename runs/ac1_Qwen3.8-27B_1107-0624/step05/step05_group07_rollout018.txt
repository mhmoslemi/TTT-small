import numpy as np

from scipy.sparse import csr_matrix
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None

def _power_sum_obj(z, p, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    sp = np.sum(c_pos ** p)
    if sp < 1e-300:
        return 0.0
    lse_equiv = sp ** (1.0 / p)
    return 2.0 * n * lse_equiv


def _power_sum_grad(z, p, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    cp = c_pos ** p
    sp = np.sum(cp)
    if sp < 1e-300:
        return np.zeros(n)
    S_p = sp ** (1.0 / p)
    w = cp / sp
    w_unnorm = c_pos ** (p - 1)
    a_rev = a[::-1].copy()
    h = _full_convolve(w_unnorm, a_rev)
    h = h[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad_a = 2.0 * h / (S_p ** (p - 1)) * (S_p ** (p - 1)) / S_p
    # Simplify: grad of S_p w.r.t. a = (1/S_p) * sum_k w_k * dc_k/da
    # dc_k/da_m = a_{k-m} + a_{m-k} (approximately 2*h_m for symmetric w)
    # Actually let's use: grad_Sp_da = (1/S_p) * 2 * conv(w, a_rev)[n-1:2n-1]
    h_w = _full_convolve(w, a_rev)
    h_w = h_w[n - 1:2 * n - 1]
    grad_a = (2.0 / S_p) * h_w
    # Now convert to grad w.r.t. z: grad_z = grad_a * a - a * (a dot grad_a)
    grad_z = grad_a * a - a * float(np.dot(a, grad_a))
    # Multiply by 2n for the full objective
    grad_z = 2.0 * n * grad_z
    return grad_z

def _mirror_descent(a_init, deadline, eta_init=0.1, max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()

    avg_a = np.zeros(n)
    eta = eta_init
    count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        # Subgradient of max_k c_k w.r.t. a:
        # dc_k/da_m = a_{k-m} (if valid) + a_{m-k} (if valid)
        subgrad = np.zeros(n)
        for m in range(n):
            idx1 = k_star - m
            idx2 = m - k_star
            val = 0.0
            if 0 <= idx1 < n:
                val += a[idx1]
            if 0 <= idx2 < n:
                val += a[idx2]
            subgrad[m] = val
        # Mirror descent update: a_new ∝ a * exp(-eta * subgrad)
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - eta * subgrad
        log_a_new = log_a_new - np.max(log_a_new)
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        # Shrinkage toward uniform to avoid collapse
        a_new = 0.98 * a_new + 0.02 / n
        a_new = a_new / a_new.sum()
        a = a_new
        avg_a = avg_a + a
        count += 1
        # Periodically check the average
        if count % 100 == 0:
            avg_check = avg_a / count
            sc = _score_norm(avg_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = avg_check.copy()
        # Decay eta
        if it % 500 == 0 and it > 0:
            eta *= 0.95
        # Also check current iterate
        if count % 500 == 0:
            sc_cur = _score_norm(a)
            if sc_cur < best_score:
                best_score = sc_cur
                best_a = a.copy()

    # Final check of average
    if count > 0:
        avg_final = avg_a / count
        sc = _score_norm(avg_final)

def _hotspot_reduction(a_init, deadline, n_iters=10000):
    """Target the maximum convolution coefficient directly."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        max_c = float(c.max())
        k_star = int(np.argmax(c))
        # Find which a_i contribute most to c[k_star]
        # c[k_star] = sum_i a_i * a[k_star - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]
        # Reduce the largest contributors
        top_k = max(1, n // 20)
        top_idx = np.argsort(contribs)[-top_k:]
        # Reduce them and redistribute
        total_reduce = 0.0
        for idx in top_idx:
            frac = contribs[idx] / (max_c + 1e-30)
            reduce_amt = 0.01 * frac * a[idx]
            a[idx] -= reduce_amt
            total_reduce += reduce_amt
        # Redistribute to bottom contributors
        bottom_k = max(1, n // 20)
        bottom_idx = np.argsort(contribs)[:bottom_k]
        for idx in bottom_idx:
            a[idx] += total_reduce / len(bottom_idx)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_a, best_score


def _make_profiles(n):

    u = 2 * t - 1
    profiles = []

        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:

        profiles.append(a / a.sum())
    for pwr in [0.5, 1.0, 2.0, 3.0]:

    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_lse(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001]
        beta_list = [b * max_c0 for b in fracs]

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = beta_list[idx:]
        time_per = (deadline - time.monotonic()) / max(1, len(remaining))
        time_per = min(time_per, 80.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(100, int(time_per * 25))
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-16,
                         'maxls': 40, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = np.ones(n_init) / n_init
        update_best(parent, _score_norm(parent), "uniform")

    n_parent = len(parent)
    print(f"Starting best: {best_score:.10f}, n={n_parent}")

    # Phase 1: LP iterations (300s)
    phase1_end = min(deadline, t_start + 280.0)
    print(f"\nPhase 1: LP iterations")
    n_lp = min(n_parent, 4096)
    if n_parent > n_lp:
        a_lp = _interp_profile(parent, n_lp)
    else:
        a_lp = parent.copy()

    for lp_round in range(8):
        if time.monotonic() > phase1_end:
            break
        step = 0.4 / (lp_round + 1)
        a_lp, sc_lp = _lp_iterate(a_lp, n_iters=20, deadline=phase1_end, step_size=step)
        update_best(a_lp, sc_lp, f"lp_r{lp_round}_n{n_lp}")

    if n_parent > n_lp:
        a_rec = _interp_profile(a_lp, n_parent)
        sc_rec = _score_norm(a_rec)
        update_best(a_rec, sc_rec, f"lp_interp_n{n_parent}")
    else:
        a_rec = a_lp.copy()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Power-sum L-BFGS (400s)
    phase2_end = min(deadline, t_start + 680.0)
    print(f"\nPhase 2: Power-sum L-BFGS")

    n_for_ls = len(best_seq)
    a_start = best_seq.copy()
    p_list = [16, 32, 64, 128, 256, 512]
    a_ps, sc_ps = _run_power_sum_lbfgs(a_start, phase2_end, p_list=p_list)
    update_best(a_ps, sc_ps, "power_sum")

    # Phase 3: Log-sum-exp L-BFGS (300s)
    phase3_end = min(deadline, t_start + 980.0)
    print(f"\nPhase 3: LSE L-BFGS")

    a_lse, sc_lse = _run_lbfgs_lse(a_start, phase3_end)
    update_best(a_lse, sc_lse, "lse_lbfgs")

    # Phase 4: Mirror descent (250s)
    phase4_end = min(deadline, t_start + 1230.0)
    print(f"\nPhase 4: Mirror descent")

    a_md, sc_md = _mirror_descent(a_start, phase4_end, eta_init=0.05, max_iter=30000)
    update_best(a_md, sc_md, "mirror_descent")

    # Phase 5: Subgradient polish (150s)
    phase5_end = min(deadline, t_start + 1380.0)
    print(f"\nPhase 5: Subgradient polish")

    a_sg, sc_sg = _subgradient_polish(a_start, phase5_end, n_iters=15000, eta_init=0.03)
    update_best(a_sg, sc_sg, "subgrad_polish")

    # Phase 6: Coordinate descent (150s)
    phase6_end = min(deadline, t_start + 1530.0)
    print(f"\nPhase 6: Coordinate descent")

    a_cd, sc_cd = _coord_descent_logspace(a_start, phase6_end, max_rounds=20)
    update_best(a_cd, sc_cd, "coord_descent")

    # Phase 7: Try larger n (150s)
    phase7_end = min(deadline, t_start + 1680.0)
    print(f"\nPhase 7: Larger n exploration")

    for n_try in [8192, 16384]:
        if time.monotonic() > phase7_end:
            break
        if n_try <= len(best_seq):
            continue
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")
        # Brief L-BFGS on the interpolated sequence
        if time.monotonic() < phase7_end - 30:
            a_big, sc_big = _run_lbfgs_lse(p_interp, phase7_end - 10,
                                           beta_list=[0.01, 0.005, 0.002, 0.001])
            update_best(a_big, sc_big, f"lbfgs_n{n_try}")

    # Phase 8: Final fine-tuning (remaining time)
    print(f"\nPhase 8: Final fine-tuning")
    a_final, sc_final = _run_lbfgs_lse(
        best_seq, deadline - 5,
        beta_list=[0.001 * _full_convolve(best_seq, best_seq).max(),
                   0.0005 * _full_convolve(best_seq, best_seq).max(),
                   0.0002 * _full_convolve(best_seq, best_seq).max()]
    )
    update_best(a_final, sc_final, "final_lbfgs")

    # Very fine coordinate descent
    if time.monotonic() < deadline - 20:
        a_cd2, sc_cd2 = _coord_descent_logspace(best_seq, deadline - 5, max_rounds=10)
        update_best(a_cd2, sc_cd2, "final_cd")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
