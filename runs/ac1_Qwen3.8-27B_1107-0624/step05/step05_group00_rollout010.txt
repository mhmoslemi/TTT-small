def _next_pow2(n):
    k = 1

def _full_convolve(a, b):

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

def _proj_simplex(v):

    if not np.any(cond):

    rho = np.where(cond)[0][-1]

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"Uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # Phase 1: FISTA + Adam at native n (0 - 350s)
    print("Phase 1: Optimize at native n")
    p1_end = min(deadline, t_start + 350.0)
    a_work = a0.copy()

    a_f, mc_f = _optimize_fista(a_work, min(p1_end, t_start + 150.0), n_phases=14, top_k_base=12)
    sc_f = _score(a_f)
    print(f"  FISTA: score={sc_f:.10f}")
    if sc_f < best_score:
        best_score = sc_f
        best_seq = a_f.copy()

    if time.monotonic() < p1_end - 10:
        a_a, mc_a = _optimize_adam(a_f, p1_end, n_phases=6)
        sc_a = _score(a_a)
        print(f"  Adam: score={sc_a:.10f}")
        if sc_a < best_score:
            best_score = sc_a
            best_seq = a_a.copy()

    # Phase 2: n=8192 (350 - 950s)
    print("Phase 2: n=8192")
    p2_end = min(deadline, t_start + 950.0)
    if time.monotonic() < p2_end - 80:
        n2 = 8192
        a2 = _interp_to_n(best_seq, n2)
        sc2i = _score(a2)
        print(f"  Interp: score={sc2i:.10f}")
        if sc2i < best_score:
            best_score = sc2i
            best_seq = a2.copy()

        t_sub = min(p2_end - 60.0, time.monotonic() + 250.0)
        a2, _ = _optimize_fista(a2, t_sub, n_phases=12, top_k_base=10)
        sc2f = _score(a2)
        print(f"  FISTA n=8192: score={sc2f:.10f}")
        if sc2f < best_score:
            best_score = sc2f
            best_seq = a2.copy()

        if time.monotonic() < p2_end - 20:
            a2, _ = _optimize_adam(a2, p2_end, n_phases=5)
            sc2a = _score(a2)
            print(f"  Adam n=8192: score={sc2a:.10f}")
            if sc2a < best_score:
                best_score = sc2a
                best_seq = a2.copy()

    # Phase 3: n=16384 (950 - 1550s)
    print("Phase 3: n=16384")
    p3_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p3_end - 80:
        n3 = 16384
        a3 = _interp_to_n(best_seq, n3)
        sc3i = _score(a3)
        print(f"  Interp: score={sc3i:.10f}")
        if sc3i < best_score:
            best_score = sc3i
            best_seq = a3.copy()

        t_sub = min(p3_end - 60.0, time.monotonic() + 250.0)
        a3, _ = _optimize_fista(a3, t_sub, n_phases=10, top_k_base=8)
        sc3f = _score(a3)
        print(f"  FISTA n=16384: score={sc3f:.10f}")
        if sc3f < best_score:
            best_score = sc3f
            best_seq = a3.copy()

        if time.monotonic() < p3_end - 20:
            a3, _ = _optimize_adam(a3, p3_end, n_phases=4)
            sc3a = _score(a3)
            print(f"  Adam n=16384: score={sc3a:.10f}")
            if sc3a < best_score:
                best_score = sc3a
                best_seq = a3.copy()

    # Phase 4: Diverse restarts on best n (1550 - 1760s)
    print("Phase 4: Restarts")
    p4_end = min(deadline, t_start + 1760.0)
    if time.monotonic() < p4_end - 20 and best_seq is not None:
        n_br = len(best_seq)
        n_restarts = 4
        per_restart = (p4_end - time.monotonic() - 10) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > p4_end - 10:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 2:
                shift = rng.integers(1, max(2, n_br // 2048))
                p_start = np.roll(best_seq, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb(best_seq, 0.08, rng)

            t_r = time.monotonic() + max(5.0, per_restart * 0.6)
            p_best, _ = _optimize_fista(p_start, t_r, n_phases=6, top_k_base=6)
            if time.monotonic() < p4_end - 15:
                p_best, _ = _optimize_adam(p_best, min(p4_end - 10.0, time.monotonic() + per_restart * 0.3), n_phases=3)
            sc_r = _score(p_best)
            print(f"  Restart {i}: score={sc_r:.10f}")
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p_best.copy()

    # Phase 5: Final ultra-fine polish
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = np.clip(best_seq, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        a_pol, _ = _optimize_adam(a_pol, deadline, n_phases=3)
        sc_pol = _score(a_pol)
        print(f"  Polish: score={sc_pol:.10f}")
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol.copy()

    # Final
    print(f"Best: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    return [float(x) for x in fallback]
