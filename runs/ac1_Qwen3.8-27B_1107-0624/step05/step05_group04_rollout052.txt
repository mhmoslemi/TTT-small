Multi-objective optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: (1) spectral equalization (Gerchberg-Saxton style) for landscape exploration,
(2) L2 variance minimization for overall flatness, (3) log-sum-exp peak reduction,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _spectral_equalize(a, alpha, N=None):
    """One step of spectral equalization: flatten magnitude, keep phase."""
    n = len(a)
    if N is None:
        N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = A / np.maximum(mag, 1e-300)
    target = float(np.sqrt(np.sum(a ** 2)))
    mag_new = np.power(mag, alpha) * np.power(target, 1.0 - alpha)
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(n) / n
    return a_new


def _spectral_equalize_loop(a, deadline, rng, n_steps=200, alpha_init=0.05, alpha_end=0.5):
    """Run multiple spectral equalization steps with increasing alpha."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = _score_norm(a)
    alphas = np.linspace(alpha_init, alpha_end, n_steps)
    for i, alpha in enumerate(alphas):
        if time.monotonic() > deadline:
            break
        a_new = _spectral_equalize(a, alpha, N)
        sc = _score_norm(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_sc


def _l2_grad(a):
    """Gradient of sum(c^2) w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    c = _conv(a, a)
    # d(sum c^2)/da[j] = 4 * sum_i c[j+i] * a[i]
    # This is 4 * (cross-correlation of c and a)[j]
    a_rev = a[::-1]
    cc = _conv(a_rev, c)
    h = cc[n - 1:2 * n - 1]
    return 4.0 * h


def _adam_l2(a_init, deadline, rng, lr=0.01, max_iter=50000):
    """Adam optimization minimizing L2 norm of convolution."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        grad = _l2_grad(a)
        # Project gradient to simplex tangent
        offset = float(np.dot(a, grad))
        grad_z = a * (grad - offset)
        if not np.all(np.isfinite(grad_z)):
            break
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.8)
    return best_a, best_sc


def _lse_obj(z, beta):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _lse_grad(z, beta):
    a = _softmax(z)
    c = _conv(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(len(a))
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    n = len(a)
    a_rev = a[::-1]
    w_conv = _conv(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adam_lse(a_init, deadline, rng, lr=0.05, beta=0.01, max_iter=100000):
    """Adam optimization for log-sum-exp objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        grad_z = _lse_grad(z, beta)
        if not np.all(np.isfinite(grad_z)):
            break
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if t % 10000 == 0:
            lr = max(0.005, lr * 0.7)
    return best_a, best_sc


def _lbfgs_lse(a_init, deadline, rng, beta, maxiter=5000):
    """L-BFGS on log-sum-exp objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    try:
        result = minimize(
            _lse_obj, z, args=(beta,),
            method='L-BFGS-B', jac=_lse_grad,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxls': 50}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            z_new = np.clip(result.x, -80, 80)
            z_new = z_new - z_new.mean()
            a_new = _softmax(z_new)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_sc


def _peak_surgery(a_init, deadline, rng, max_rounds=500):
    """Surgically reduce peaks by adjusting individual elements."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax(z)
        c = _conv(a_cur, a_cur)
        cm = float(c.max())
        peak_idx = int(np.argmax(c))
        valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        if len(valid_i) == 0:
            break
        prod = a_cur[valid_i] * a_cur[peak_idx - valid_i]
        order = np.argsort(prod)[::-1]
        improved = False
        for idx_pos in order[:6]:
            idx = valid_i[idx_pos]
            for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try increasing small elements to spread energy
            small = np.argsort(a_cur)[:4]
            for idx in small:
                for delta in [0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_sc - 1e-13:
                        best_sc = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        if not improved and rnd > 50:
            break
    return best_a, best_sc


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _continuation_lbfgs(a_init, deadline, n_betas=15):
    """L-BFGS with decreasing beta continuation."""
    n = len(a_init)
    c0 = _conv(a_init, a_init)
    max_c0 = float(c0.max())
    if max_c0 < 1e-15:
        return a_init, _score_norm(a_init)

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    beta_start = 0.3 * max_c0
    beta_end = max(1e-8, 1e-6 * max_c0)
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        dl = time.monotonic() + time_budget
        maxiter = 3000 if beta < 0.01 * max_c0 else 1000

        a_opt, sc_opt = _lbfgs_lse(best_a, dl, None, beta, maxiter)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(256, 1.5)
        update_best(a0, _score_norm(a0), "ramp_init")
        best_seq = a0

    # === Phase 1: Spectral equalization to find better landscape (15% time) ===
    p1_end = t_start + (budget_s - 30.0) * 0.15
    print(f"\nPhase 1: Spectral equalization (deadline {p1_end - t_start:.0f}s)")

    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    # Run spectral equalization
    a_se, sc_se = _spectral_equalize_loop(a_cur, p1_end - 10, rng, n_steps=300,
                                          alpha_init=0.03, alpha_end=0.4)
    update_best(a_se, sc_se, "spectral_eq")

    # Also try from a ramp at the same size
    a_ramp = _make_ramp(n_cur, 2.0)
    a_se2, sc_se2 = _spectral_equalize_loop(a_ramp, p1_end - 5, rng, n_steps=150,
                                            alpha_init=0.05, alpha_end=0.3)
    update_best(a_se2, sc_se2, "spectral_eq_ramp")

    a_cur = best_seq.copy()
    print(f"  Phase 1 done: {best_score:.10f}")

    # === Phase 2: L2 variance minimization (10% time) ===
    p2_end = t_start + (budget_s - 30.0) * 0.25
    print(f"\nPhase 2: L2 minimization (deadline {p2_end - t_start:.0f}s)")

    if time.monotonic() < p2_end - 10:
        a_l2, sc_l2 = _adam_l2(a_cur, p2_end - 10, rng, lr=0.02, max_iter=80000)
        update_best(a_l2, sc_l2, "l2_adam")
        a_cur = best_seq.copy()

    print(f"  Phase 2 done: {best_score:.10f}")

    # === Phase 3: Multi-scale continuation for higher n (30% time) ===
    p3_end = t_start + (budget_s - 30.0) * 0.55
    print(f"\nPhase 3: Multi-scale (deadline {p3_end - t_start:.0f}s)")

    if len(best_seq) < 4096:
        # Upsample and optimize at higher scales
        target_scales = [s for s in [2048, 4096] if s > len(best_seq)]
        a_work = best_seq.copy()
        for ts in target_scales:
            if time.monotonic() > p3_end - 30:
                break
            a_up = _upsample(a_work, ts)
            sc_up = _score_norm(a_up)
            print(f"  Upsampled to n={ts}: score={sc_up:.10f}")

            # Spectral equalization at new scale
            time_left = p3_end - time.monotonic() - 60
            if time_left < 30:
                break
            a_se3, sc_se3 = _spectral_equalize_loop(a_up, time.monotonic() + min(60, time_left * 0.2),
                                                    rng, n_steps=100, alpha_init=0.03, alpha_end=0.2)
            update_best(a_se3, sc_se3, f"spec_eq_n{ts}")

            # L-BFGS continuation
            if time.monotonic() < p3_end - 30:
                a_c, sc_c = _continuation_lbfgs(a_se3, p3_end - 30, n_betas=10)
                update_best(a_c, sc_c, f"lbfgs_n{ts}")
                a_work = best_seq.copy()

    print(f"  Phase 3 done: {best_score:.10f}")

    # === Phase 4: L-BFGS continuation (25% time) ===
    p4_end = t_start + (budget_s - 30.0) * 0.80
    print(f"\nPhase 4: L-BFGS continuation (deadline {p4_end - t_start:.0f}s)")

    if time.monotonic() < p4_end - 20:
        a_c4, sc_c4 = _continuation_lbfgs(best_seq, p4_end - 20, n_betas=20)
        update_best(a_c4, sc_c4, "lbfgs_phase4")

    # Perturbed restart
    if time.monotonic() < p4_end - 30:
        n_b = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        noise = 0.05 * rng.normal(size=n_b)
        z = z + noise
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_pert = _softmax(z)
        a_c4b, sc_c4b = _continuation_lbfgs(a_pert, p4_end - 20, n_betas=10)
        update_best(a_c4b, sc_c4b, "lbfgs_perturb")

    print(f"  Phase 4 done: {best_score:.10f}")

    # === Phase 5: Adam LSE + Peak surgery (20% time) ===
    p5_end = deadline
    print(f"\nPhase 5: Adam LSE + Peak surgery (deadline {p5_end - t_start:.0f}s}")

    if time.monotonic() < p5_end - 30:
        # Adaptive beta Adam
        c_cur = _conv(best_seq, best_seq)
        cm = float(c_cur.max())
        beta_ad = 0.1 * cm

        a_ad, sc_ad = _adam_lse(best_seq, min(p5_end - 60, time.monotonic() + 300),
                                rng, lr=0.03, beta=beta_ad, max_iter=200000)
        update_best(a_ad, sc_ad, "adam_lse")

        # Peak surgery
        if time.monotonic() < p5_end - 20:
            a_ps, sc_ps = _peak_surgery(best_seq, p5_end - 20, rng, max_rounds=1000)
            update_best(a_ps, sc_ps, "peak_surgery")

        # Fine Adam
        if time.monotonic() < p5_end - 10:
            a_f, sc_f = _adam_lse(best_seq, p5_end - 10, rng, lr=0.005,
                                  beta=0.01 * cm, max_iter=100000)
            update_best(a_f, sc_f, "adam_fine")

    print(f"  Phase 5 done: {best_score:.10f}")

    # === Final: Tiny-beta L-BFGS polish ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        c_final = _conv(best_seq, best_seq)
        cm_final = float(c_final.max())
        beta_tiny = 1e-5 * cm_final
        a_f, sc_f = _lbfgs_lse(best_seq, deadline - 5, None, beta_tiny, maxiter=20000)
        update_best(a_f, sc_f, "tiny_beta_lbfgs")

    # === Validation and return ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s > 0.01:
            final = final / s
        else:
            final = _make_ramp(256, 1.5)
        sc_final = _score_full(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}")
        return [float(x) for x in final]

    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
