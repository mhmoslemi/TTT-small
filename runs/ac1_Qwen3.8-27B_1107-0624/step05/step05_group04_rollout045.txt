import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _downsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _adam_optimize(a_init, deadline, rng, lr_init=0.1, beta_init=None, top_k=5,
                   kick_interval=5000, kick_size=0.02, lr_decay_period=2000):
    """Adam optimizer on log-space with top-k peak targeting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())
    if beta_init is None:
        beta = 0.1 * max_c0
    else:
        beta = beta_init

    steps_per_sec_estimate = max(1, int(50.0 / (n / 4096.0)))
    last_time = time.monotonic()
    step_count_window = 0
    window_start = last_time

    for it in range(5000000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break

        # Adaptive beta: decrease over time, also adapt based on peak spread
        if it % 200 == 0:
            sorted_c = np.sort(c)[::-1]
            k_check = min(top_k + 5, len(sorted_c) - 1)
            gap = sorted_c[0] - sorted_c[k_check]
            if gap < 0.02 * cm:
                beta = max(1e-8, 0.02 * cm)
            else:
                beta = max(1e-8, min(beta * 0.999, 0.3 * gap))

        # Top-k weighted gradient
        if top_k >= len(c):
            top_k = len(c)
        peak_indices = np.argsort(c)[::-1][:top_k]
        peak_vals = c[peak_indices]
        # Weight peaks by their value (exponential weighting)
        weights = np.exp(peak_vals / (beta * 0.5))
        weights = weights / weights.sum()

        # h[j] = sum_k w[k] * a[k-j] for valid k-j
        # Use all peaks with their weights
        w_full = np.zeros(2 * n - 1)
        w_full[peak_indices] = weights
        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w_full)
        h = w_conv[n - 1:2 * n - 1]

        # Also add uniform component for stability
        uniform_w = np.ones(2 * n - 1) / (2 * n - 1)
        w_conv_u = _full_convolve(a_rev, uniform_w)
        h_u = w_conv_u[n - 1:2 * n - 1]
        h = 0.7 * h + 0.3 * h_u

        # Gradient wrt a (with sum=1 constraint)
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Learning rate with cosine decay
        progress = min(1.0, t / 50000.0)
        lr = lr_init * (0.5 + 0.5 * np.cos(np.pi * progress))
        lr = max(lr, 0.001)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Periodic kick
        if t % kick_interval == 0:
            z = z + kick_size * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z = z - z.mean()

        # Timing check
        if t % 1000 == 0:
            elapsed = time.monotonic() - last_time
            if elapsed > 5.0:
                steps_per_sec = int(1000 / elapsed)
                last_time = time.monotonic()

    return best_a, best_score


def _topk_adam_optimize(a_init, deadline, rng, lr_init=0.05, top_k=3,
                        beta_start=None, kick_interval=3000, kick_size=0.015):
    """Sharper Adam with fewer top peaks for targeted reduction."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())
    if beta_start is None:
        beta = 0.05 * max_c0
    else:
        beta = beta_start

    for it in range(5000000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break

        # Very focused on top peaks
        if top_k >= len(c):
            top_k = len(c)
        peak_indices = np.argsort(c)[::-1][:top_k]
        peak_vals = c[peak_indices]

        # Sharp weighting: only truly dominant peaks get weight
        threshold = peak_vals[0] - beta
        active_peaks = peak_indices[peak_vals > threshold]
        if len(active_peaks) == 0:
            active_peaks = peak_indices[:1]
        weights = np.ones(len(active_peaks)) / len(active_peaks)

        w_full = np.zeros(2 * n - 1)
        w_full[active_peaks] = weights
        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w_full)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Decaying LR
        lr = lr_init * max(0.1, 1.0 - t / 100000.0)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

            z = np.clip(z, -80, 80)
            z = z - z.mean()

        # Slowly decrease beta
        if t % 1000 == 0:
            beta = max(1e-8, beta * 0.95)

    return best_a, best_score


def _peak_coordination(a_init, deadline, rng, max_rounds=2000):
    """Directly target top peaks by reducing their contributors."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top 5 peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices:
            # Contributors to this peak
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            # Try reducing top contributors
            order = np.argsort(prod)[::-1][:6]
            for idx_pos in order:
                idx = valid_i[idx_pos]
                for delta in [0.002, 0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    # Compensate by increasing a low-contribution element
                    # Find element with least impact on peaks
                    small_idx = np.argsort(a_cur)[:3]
                    for comp_idx in small_idx:
                        z_trial2 = z_trial.copy()
                        z_trial2[comp_idx] += delta * 0.5
                        a_trial = _softmax_fn(z_trial2)
                        sc = _score_norm(a_trial)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial2.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements (spreading energy)
            small_idx = np.argsort(a_cur)[:8]
            for idx in small_idx:
                for delta in [0.002, 0.005, 0.01]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try a small random perturbation
            z = z + 0.005 * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z = z - z.mean()

    return best_a, best_score


def _polynomial_smooth(a, degree=15):
    """Fit a polynomial to the sequence and return the smoothed version."""
    n = len(a)
    x = np.linspace(0, 1, n)
    coeffs = np.polyfit(x, a, degree)
    smooth = np.polyval(coeffs, x)
    smooth = np.maximum(smooth, 1e-15)
    return smooth / smooth.sum()


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

            update_best(a_8k, sc_8k, "n8192_adam")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        final_score = _score(final_seq)
        print(f"\nFINAL: score={final_score:.10f}, score_norm={best_score:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
