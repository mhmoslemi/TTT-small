```python
"""
Nesterov-accelerated logsumexp optimization with peak-targeted local search
for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_b_spline(n, order=6):
    t = np.linspace(0.0, 1.0, n)
    u = t * 0.95 + 0.025
    u = np.clip(u, 0.005, 0.995)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
```

```python
def _nesterov_optimize(z_init, deadline, beta_schedule, nesterov_alpha=0.9):
    """
    Nesterov-accelerated optimization on the logsumexp-smoothed objective.
    """
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    
    best_z = z.copy()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)
    
    momentum = np.zeros(n)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline - 2.0:
            break
        
        # Number of iterations for this beta level
        # Smaller beta -> more iterations needed
        n_iters = max(100, int(5000 * (1.0 / max(beta, 0.001))))
        n_iters = min(n_iters, 20000)
        
        step = 1.0  # Will be adapted
        
        for it in range(n_iters):
            if time.monotonic() > deadline - 2.0:
                break
            
            # Nesterov lookahead
            y = z + nesterov_alpha * momentum  # Simplified momentum form
            a_y = _softmax_fn(y)
            
            # Compute gradient at y
            c = _full_convolve(a_y, a_y)
            L = float(c.max())
            shifted = (c - L) / beta
            e = np.exp(shifted)
            S = e.sum()
            if S < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / S
            
            W = float(np.dot(w, c))
            
            # h[j] = sum_k w[k] * a_y[k-j]
            # This is the convolution of w with reversed a_y, taking the relevant part
            a_rev = a_y[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            
            grad = (2.0 / beta) * a_y * (h - W)
            
            # Gradient step with momentum
            momentum = nesterov_alpha * momentum - step * grad
            z_new = z + momentum
            
            # Check if objective improved
            if it % 10 == 0:
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if sc < best_score:
                    best_score = sc
                    best_z = z_new.copy()
                    best_a = a_new.copy()
            
            # Adaptive step size
            if it % 50 == 0 and it > 0:
                a_check = _softmax_fn(z_new)
                sc_check = _score_norm_fn(a_check)
                # If stalled, reduce step
                if not np.isfinite(sc_check):
                    step *= 0.5
                    momentum *= 0.5
            
            z = z_new
        
        # Update momentum at beta transition
        momentum *= 0.5  # Damp momentum between beta levels
    
    return best_a, best_score
```

```python
def _linear_perturbation_lp(b, deadline, rng):
    """
    Find optimal perturbation c such that a = b + c minimizes max(conv(a,a)).
    Linearized: minimize t s.t. conv(b,b)[k] + 2*conv(b,c)[k] <= t
    """
    import cvxpy as cp
    from scipy.sparse import csr_matrix
    
    n = len(b)
    c_base = _full_convolve(b, b)
    
    # conv(b, c)[k] = sum_j b[j] * c[k-j]
    # So conv(b,c) is a matrix times c: (B @ c)[k] where B is convolution matrix
    
    # B is (2n-1) x n matrix where B[k,j] = b[k-j] if 0<=k-j<n, else 0
    
    # For the LP: minimize t
    # subject to: c_base[k] + 2*(B@c)[k] <= t for k=0..2n-2
    #              -b[j]*margin <= c[j] (ensuring a[j] = b[j]+c[j] >= 0)
    #              sum(c) = 0 (maintain normalization)
    
    # Build sparse matrix B
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(b[k - j])
    
    B = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        c_base + 2 * (B @ c_var) <= t_var,
        c_var >= -b * 0.9,  # a[j] = b[j] + c[j] >= 0.1 * b[j]
        cp.sum(c_var) == 0
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.PDLP, max_iter=10000)
        if prob.status in ['optimal', 'optimal_inaccurate'] and c_var.value is not None:
            c_opt = c_var.value
            a_opt = np.maximum(b + c_opt, 0)
            a_opt = a_opt / a_opt.sum()
            return a_opt, _score_norm_fn(a_opt)
    except Exception:
        pass
    
    return b, _score_norm_fn(b)
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(x):
    """Project x onto {y: y >= 0, sum(y) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = u * arange + 1.0 > cv
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)

                                                 momentum_coeff=0.94,
                                                 base_step=0.3 / np.sqrt(n_work),
                                                 max_iter_per_beta=10000)
            _update_best(a_best_f, sc_f, "final_momentum")

        # 6b: Another round of iterative linearization
        if time.monotonic() < deadline - 60:
            a_best_il, sc_il = _iterative_linearization(best_seq, min(deadline - 60, time.monotonic() + 100.0),
                                                        n_outer=4, n_inner=150)
            _update_best(a_best_il, sc_il, "final_il")

        # 6c: Final local search
        if time.monotonic() < deadline - 30:
            a_best_ls, sc_ls = _peak_local_search(best_seq, min(deadline - 25, time.monotonic() + 80.0),
                                                  n_trials=20000)
            _update_best(a_best_ls, sc_ls, "final_local")

    # === Try interpolation to slightly different n ===
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_curr = len(best_seq)
        for target_n in [n_curr // 2, n_curr * 2]:
            if target_n < 128 or target_n > 16384:
                continue
            if time.monotonic() > deadline - 40:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            _update_best(p_interp, sc_interp, f"interp_n{target_n}")

    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_b_spline(8192, 6)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator if size permits
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirms: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_b_spline(8192, 6)
    print(f"FALLBACK: n=8192, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
```
