"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses power-mean objective (differentiable approximation of max), Adam in log-space,
and L-BFGS quasi-Newton polishing. Explores n=4096 and n=8192.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _power_mean(a, p):
    """(sum c^p / (2n-1))^(1/p) where c = conv(a,a)"""
    n = len(a)
    c = _full_convolve(a, a)
    cp = np.sum(c ** p)
    return float((cp / (2 * n - 1)) ** (1.0 / p))


def _power_mean_grad_a(a, p):
    """Gradient of (sum c^p)^(1/p) w.r.t. a. Returns (grad, c)."""
    n = len(a)
    c = _full_convolve(a, a)
    cmax = float(c.max())
    if cmax < 1e-300:
        return np.zeros(n), c
    # Normalize to avoid overflow: d_k = (c_k/cmax)^(p-1)
    cn = c / cmax
    d = cn ** (p - 1)
    # h_j = sum_m d[j+m] * a[m], j=0..n-1
    # Compute via FFT: conv(d, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    N = _next_pow2(3 * n - 1)
    d_fft = np.fft.rfft(d, N)
    ar_fft = np.fft.rfft(a_rev, N)
    h_full = np.fft.irfft(d_fft * ar_fft, N)
    h = h_full[n - 1:2 * n - 1]
    # grad = 2 * S^(1/p - 1) * h * (normalization factor from cn)
    # S = sum(c^p) = cmax^p * sum(cn^p)
    S = cmax ** p * np.sum(cn ** p)
    if S < 1e-300:
        return 2.0 * h, c
    scale = S ** (1.0 / p - 1.0) / (cmax ** (p - 1))
    grad = 2.0 * h * scale
    return grad, c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _power_mean_grad_z(z, p):
    """Gradient of power_mean w.r.t. z where a = softmax(z)."""
    a = _softmax(z)
    n = len(a)
    grad_a, c = _power_mean_grad_a(a, p)
    # dz = a * (grad_a - dot(a, grad_a))
    centered = grad_a - float(np.dot(a, grad_a))
    z_grad = a * centered
    return z_grad, _power_mean(a, p)


def _adam_optimize_z(z_init, deadline, p_schedule, lr_init, max_iter):
    """Adam optimizer in z-space with power-mean objective."""
    z = z_init.copy()
    n = len(z)
    best_z = z.copy()
    best_val = np.inf
    best_a = _softmax(z).copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        lr = lr_init
        stagnant = 0
        local_best = best_val

        while t < max_iter and time.monotonic() < deadline:
            t += 1
            z_grad, val = _power_mean_grad_z(z, p)

            if not np.all(np.isfinite(z_grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * z_grad
            v_buf = b2 * v_buf + (1.0 - b2) * (z_grad * z_grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 1.0:
                update = update * (1.0 / un)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:

                sc = _score_norm(a_chk)

                    best_z = z.copy()
                    best_a = a_chk.copy()
                    stagnant = 0
                else:
                    stagnant += 1

        # Decay learning rate between phases
        lr_init *= 0.5

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_val:
        best_val = sc_final
        best_z = z.copy()
        best_a = a_final.copy()

    return best_a, best_val, best_z


def _lbfgs_polish(a_init, deadline, p=256, max_iter=500):
    """L-BFGS polish in z-space."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    n = len(z0)

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    def objective(z):
        a = _softmax(z)
        return _power_mean(a, p)

    def gradient(z):
        zg, _ = _power_mean_grad_z(z, p)
        return zg

    t_start = time.monotonic()
    try:
        result = scipy_minimize(
            objective, z0, method='L-BFGS-B',
            jac=gradient,
            options={'maxiter': max_iter, 'maxfun': 2000,
                     'ftol': 1e-15, 'gtol': 1e-10, 'iprint': 0}
        )
        if time.monotonic() - t_start > deadline:
            pass  # Just note it took long
        if result.fun < np.inf:
            a_res = _softmax(result.x)
            sc = _score_norm(a_res)
            if sc < best_score:
                best_score = sc
                best_a = a_res.copy()
    except Exception:
        pass

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()

        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
