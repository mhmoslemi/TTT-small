"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive LSE-continuation L-BFGS with peak-targeted local search,
multi-threshold peak identification, and simulated annealing restarts.
Key innovations: adaptive beta based on convolution spread, multi-peak

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 2
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:2 * n - 1]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _conv(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_sum1(a):
    n = len(a)
    c = _conv(a)
    return float(2.0 * n * c.max())


def _lse_obj_and_grad(a, beta):
    n = len(a)
    c = _conv(a)
    c_max = float(c.max())
    beta_c = beta * (c - c_max)
    np.clip(beta_c, -60, 60, out=beta_c)
    exp_bc = np.exp(beta_c)
    s = exp_bc.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse = c_max
    else:
        w = exp_bc / s
        lse = c_max + float(np.log(s)) / beta
    N = _next_pow2(2 * n)
    fw = rfft(w, N)
    fa = rfft(a, N)
    cross = irfft(fw * np.conj(fa), N)[:n]
    grad = 2.0 * cross
    return lse, grad, c

    z = z - z.max()

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()

    spread = float(c.max() - c.mean())
    if spread < 1e-15:

    return gamma / (spread * len(c))

def _peak_targeted_search(a_init, deadline, rng, n_trials=50000):

            products = a[pairs] * a[top_lag - pairs]
            idx = pairs[int(np.argmax(products))]

            delta = a_trial[idx] * frac

            a_trial[add_idx] += delta

                a_trial /= s

def _coordinate_descent(a_init, deadline, rng, max_rounds=200):
    """Coordinate descent: for each position, try small local adjustments."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_sum1(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 2:
            break

        c = _conv(a)
        c_max = float(c.max())
        peak_lag = int(np.argmax(c))

        # Randomly select a subset of coordinates to try
        n_try = min(n, max(100, n // 4))
        idxs = rng.integers(0, n, size=n_try)

        for idx in idxs:
            if time.monotonic() > deadline - 2:
                break
            # Try reducing this position and adding to a safe spot
            if a[idx] < 1e-10:
                continue
            # Check if this position contributes to the peak
            partner = peak_lag - idx
            if 0 <= partner < n and a[idx] * a[partner] > 0.01 * c_max:
                a_trial = a.copy()
                frac = rng.uniform(0.05, 0.2)
                delta = a_trial[idx] * frac
                a_trial[idx] -= delta
                # Add to a position far from contributing to peak
                safe_candidates = rng.integers(0, n, size=20)
                best_add = -1
                best_add_sc = np.inf
                for cand in safe_candidates:
                    a_t = a.copy()
                    a_t[idx] -= delta
                    a_t[int(cand)] += delta
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-10:
                        continue
                    a_t /= st
                    sc = _score_sum1(a_t)
                    if sc < best_add_sc:
                        best_add_sc = sc
                        best_add = int(cand)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_t.copy()
                        a = a_t.copy()
                        break
                if best_add >= 0 and best_add_sc < best_sc * 1.001:
                    a_trial = a.copy()
                    a_trial[idx] -= delta
                    a_trial[best_add] += delta
                    a_trial = np.maximum(a_trial, 0.0)
                    st = a_trial.sum()
                    if st > 1e-10:
                        a_trial /= st
                        sc = _score_sum1(a_trial)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()

    return best_a, best_sc


def _spectral_shaping(a_init, deadline, rng):
    """Use spectral analysis to guide adjustments."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_sum1(a)

    # Compute power spectrum
    N = _next_pow2(2 * n)
    spectrum = np.abs(rfft(a, N)) ** 2

    for it in range(100):
        if time.monotonic() > deadline - 2:
            break

        c = _conv(a)
        c_max = float(c.max())
        peak_lag = int(np.argmax(c))

        # Identify which frequency components cause the peak
        # The peak at lag k corresponds to constructive interference
        # Try to reduce amplitude at frequencies that align with the peak lag
        freqs = np.arange(N // 2 + 1)
        phase_factor = np.exp(-2j * np.pi * freqs * peak_lag / N)
        alignment = np.abs(spectrum * phase_factor)

        # Perturb a in the direction that reduces aligned spectral power
        # Convert to time domain perturbation
        pert_freq = np.zeros(N // 2 + 1, dtype=np.complex128)
        top_freq_count = max(1, N // 20)
        top_freq_idx = np.argsort(alignment)[-top_freq_count:]
        for fi in top_freq_idx:
            pert_freq[int(fi)] = -0.01 * alignment[int(fi)] / (np.max(alignment) + 1e-15)
        # Add small noise
        pert_freq += 0.001 * rng.standard_normal(N // 2 + 1)

        pert = np.zeros(N)
        pert[:N // 2 + 1] = pert_freq
        # Make it correspond to a real signal
        pert = np.real(irfft(pert, N) * N)
        pert_time = pert[:n]

        a_trial = a + 0.01 * pert_time
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        sc = _score_sum1(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining adaptive LSE L-BFGS, peak-targeted
    local search, coordinate descent, and spectral shaping.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_sum1(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_work = 4096
        parent = _triangular(n_work)
        sc_parent = _score_sum1(parent)
        print(f"No parent, using triangular n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    print(f"Working with n={n_work}")

    # --- Phase 1: Diverse initializations (30s) ---
    phase1_end = min(deadline, t_start + 30.0)
    candidates = []

    profiles = [
        _triangular(n_work),
        _parabolic(n_work),
        _w_parabolic(n_work, 2, 0.2),
        _w_parabolic(n_work, 3, 0.15),
        _w_parabolic(n_work, 4, 0.1),
        _w_parabolic(n_work, 1, 0.3),
        _w_parabolic(n_work, 5, 0.08),
        _dip_profile(n_work, 0.2, 0.1),
        _dip_profile(n_work, 0.3, 0.15),
        _dip_profile(n_work, 0.15, 0.08),
        _edge_sharp(n_work, 1.5),
        _edge_sharp(n_work, 2.0),
        _edge_sharp(n_work, 3.0),
        _edge_sharp(n_work, 0.7),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_sum1(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}")

    # Also try parent if it's better
    sc_p = _score_sum1(parent)
    best_start = parent.copy()
    if candidates and candidates[0][0] < sc_p:
        best_start = candidates[0][1].copy()

    # --- Phase 2: Adaptive LSE L-BFGS (400s) ---
    phase2_end = min(deadline, t_start + 430.0)
    print(f"Phase 2: Adaptive LSE L-BFGS, start score={_score_sum1(best_start):.10f}")
    dl = phase2_end - 5.0
    a_best, sc_best = _run_lbfgs_adaptive(best_start, dl, max_wall=phase2_end - t_start - 30.0)
    update_best(a_best, sc_best, "lbfgs_main")

    # Second pass from parent if different
    if sc_p < _score_sum1(best_start) - 0.01 and time.monotonic() < phase2_end - 60:
        dl2 = phase2_end - 5.0
        a_best2, sc_best2 = _run_lbfgs_adaptive(parent.copy(), dl2, max_wall=60.0)
        update_best(a_best2, sc_best2, "lbfgs_parent")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Peak-targeted local search (300s) ---
    phase3_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Peak-targeted local search")
        dl = phase3_end - 5.0
        a_best, sc_best = _peak_targeted_search(best_seq, dl, rng, n_trials=40000)
        update_best(a_best, sc_best, "peak_search")

        # Second pass
        if time.monotonic() < phase3_end - 60:
            dl2 = phase3_end - 5.0
            a_best2, sc_best2 = _peak_targeted_search(best_seq, dl2, rng, n_trials=20000)
            update_best(a_best2, sc_best2, "peak_search2")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: More L-BFGS polishing (200s) ---
    phase4_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: L-BFGS polishing")
        dl = phase4_end - 5.0
        a_best, sc_best = _run_lbfgs_adaptive(best_seq, dl, max_wall=phase4_end - t_start - 60.0)
        update_best(a_best, sc_best, "lbfgs_polish")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Spectral shaping (100s) ---
    phase5_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Spectral shaping")
        dl = phase5_end - 5.0
        a_best, sc_best = _spectral_shaping(best_seq, dl, rng)
        update_best(a_best, sc_best, "spectral")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart with noise (200s) ---
    phase6_end = min(deadline, t_start + 1230.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Multi-restart")
        n_restarts = 5
        per_restart = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise_level = 0.01 + 0.01 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _to_simplex(z)

            dl = min(phase6_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _run_lbfgs_adaptive(p_start, dl, max_wall=per_restart)
            update_best(a_best, sc_best, f"restart{i}")

            # Quick local search after each restart
            if time.monotonic() < phase6_end - 10:
                a_best2, sc_best2 = _peak_targeted_search(
                    best_seq if best_seq is not None else p_start,
                    min(phase6_end - 5, time.monotonic() + 15.0),
                    rng, n_trials=5000)
                update_best(a_best2, sc_best2, f"restart{i}_local")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Try different n (150s) ---
    phase7_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        print(f"Phase 7: Trying different n, current n={n_curr}")

        for target_n in [n_curr // 2, n_curr * 2]:
            if target_n < 512 or target_n > 16384:
                continue
            if time.monotonic() > phase7_end - 60:
                break
            p_interp = _interp(best_seq, target_n)
            sc_interp = _score_sum1(p_interp)
            # Adjust score for different n: score = 2*n*max(c), so we need to compare properly
            sc_adjusted = _score(p_interp)
            print(f"  n={target_n}: score_sum1={sc_interp:.10f}, actual_score={sc_adjusted:.10f}")
            if sc_adjusted < best_score:
                update_best(p_interp, sc_adjusted, f"interp_n{target_n}")

            if time.monotonic() < phase7_end - 30 and sc_adjusted < best_score + 0.05:
                dl = min(phase7_end - 20, time.monotonic() + 60.0)
                a_best, sc_best = _run_lbfgs_adaptive(p_interp, dl, max_wall=60.0)
                sc_actual = _score(a_best)
                update_best(a_best, sc_actual, f"opt_n{target_n}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final intensive polishing (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 8: Final polishing, remaining={remaining:.0f}s")

        # 8a: L-BFGS with very fine settings
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _run_lbfgs_adaptive(best_seq, dl, max_wall=remaining * 0.35)
            update_best(a_best, sc_best, "final_lbfgs")

        # 8b: More local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _peak_targeted_search(best_seq, dl, rng, n_trials=20000)
            update_best(a_best, sc_best, "final_local")

        # 8c: Coordinate descent
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _coordinate_descent(best_seq, dl, rng, max_rounds=50)
            update_best(a_best, sc_best, "final_coord")

        # 8d: One more L-BFGS pass
        if time.monotonic() < deadline - 15:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.15)
            a_best, sc_best = _run_lbfgs_adaptive(best_seq, dl, max_wall=remaining * 0.15)
            update_best(a_best, sc_best, "final_lbfgs2")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _triangular(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum_initial={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _triangular(4096)
    sc_fb = _score(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
