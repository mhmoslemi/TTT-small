"""
Hybrid optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses min-max LP steps, multiplicative weight descent with adaptive temperature,
structured coordinate descent, and soft-max gradient polishing.
Starts from parent sequence and exploits precise 1D subproblem solutions
for coordinate updates.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp_g(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _solve_lp_minmax(f, rows, cols, time_limit=45):
    """Solve: minimize M s.t. (f*g)[k] <= M for all k, sum(g)=1, g>=0."""
    n = len(f)
    num_g = 2 * n - 1
    num_vars = n + 1
    # Variables: g[0..n-1], M
    # Objective: minimize M = var[n]
    c_obj = np.zeros(num_vars)
    c_obj[n] = 1.0
    # Inequality: (f*g)[k] - M <= 0 for all k
    # (f*g)[k] = sum_j f[rows_k - cols_k] * g[cols_k]
    # So: sum_j A[k,j]*g[j] - M <= 0
    ii = list(rows)
    jj = list(cols)
    vv = list(f[rows - cols])
    # Add -M terms: for each k, add (k, n, -1)
    ii.extend(range(num_g))
    jj.extend([n] * num_g)
    vv.extend([-1.0] * num_g)
    A_ub = sp.csc_matrix((vv, (ii, jj)), shape=(num_g, num_vars))
    b_ub = np.zeros(num_g)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, num_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None, None
        return np.maximum(res.x[:n], 0.0), float(res.x[n])
    except Exception:
        return None, None


def _lp_step_minmax(f, rows, cols, time_limit=45):
    g, M_new = _solve_lp_minmax(f, rows, cols, time_limit)
    if g is None or g.sum() < 1e-15:
        return f
    g = g / g.sum()
    # Line search for best blending
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.1, 0.2, 0.4]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f


def _coord_descent(a, c, M, n, max_passes=500, deadline=None):
    """Structured coordinate descent. Returns improved a and c."""
    a = a.copy()
    c = c.copy()
    M = float(M)
    count = 0
    for _pass in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Process coordinates in random-ish order (strided)
        for i in range(n):
            if a[i] < 1e-15:
                continue
            # Affected range: k in [i, i+n-1] intersect [0, 2n-2]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            if k_lo > k_hi:
                continue
            d = a[k_lo - i: k_hi - i + 1]  # a[0], a[1], ..., a[n-1] (or truncated)
            c_aff = c[k_lo: k_hi + 1]
            # Unaffected max
            if k_lo > 0:
                M_unaff = max(c[:k_lo].max() if k_lo > 0 else 0.0,
                              c[k_hi+1:].max() if k_hi + 1 < 2*n-1 else 0.0)
            elif k_hi + 1 < 2 * n - 1:
                M_unaff = c[k_hi+1:].max()
            else:
                M_unaff = 0.0
            # Find optimal t (amount to decrease a[i])
            # Need: c_aff[k] - t * d[k] <= M_unaff for all k
            # t >= (c_aff[k] - M_unaff) / d[k] for c_aff[k] > M_unaff and d[k] > 0
            excess = c_aff - M_unaff
            mask = (excess > 1e-12) & (d > 1e-15)
            if np.any(mask):
                t_needed = np.max(excess[mask] / d[mask])
            else:
                t_needed = 0.0
            t_star = min(a[i], max(0.0, t_needed))
            if t_star < a[i] * 0.01:
                continue  # negligible change
            # Apply update
            a[i] -= t_star
            c_aff -= t_star * d
            c[k_lo: k_hi + 1] = c_aff
            if c_aff.max() < M:
                M = float(c_aff.max())
                if M_unaff > M:
                    M = M_unaff
            improved = True
            count += 1
        if not improved:
            break
    return a, c, M


def _mwu_step(a, c, c_max, beta, eta):
    """One multiplicative weight update step."""
    n = len(a)
    w = np.exp((c - c_max) / max(beta, 1e-15))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / s
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    # Gradient of the "soft max" objective
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    z -= z.mean()
    a_new = _softmax(z)
    return a_new


def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=500000, lr=0.01):
    """Adam optimizer with log-sum-exp smoothing for fine polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max(max_iter, 1))
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        if t % 5000 == 0:
            if best_score < prev_best * 0.9995:
                current_lr = min(0.05, current_lr * 1.05)
            else:
                current_lr = max(0.001, current_lr * 0.97)
            prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    # Build LP indices
    rows, cols = _build_lp_indices(n)
    print(f"LP indices ready, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: Aggressive min-max LP (0 - 350s)
    phase1_end = min(deadline, t_start + 350.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0

    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a_new = _lp_step_minmax(a, rows, cols, time_limit=30)
        dt = time.monotonic() - t0
        lp_count += 1
        sc = _score_norm(a_new)
        if sc < prev_sc - 1e-12:
            a = a_new
            stall = 0
        else:
            stall += 1

        if lp_count % 3 == 0 or stall > 10:
            full_sc = _score(a)
            print(f"  LP[{lp_count}] score={full_sc:.10f} dt={dt:.1f}s stall={stall} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()

        if stall > 25:
            print(f"  LP stalled at iter {lp_count}")
            break
        prev_sc = _score_norm(a)

    full_sc = _score(a)
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: MWU with adaptive beta (350 - 800s)
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end:
        a = best_seq.copy()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        betas = [0.01 * c_max, 0.005 * c_max, 0.002 * c_max, 0.001 * c_max,
                 0.0005 * c_max, 0.0002 * c_max, 0.0001 * c_max]
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase2_end:
                break
            eta = 0.08 if bi < 3 else 0.04
            for it in range(200):
                if time.monotonic() > phase2_end:
                    break
                a = _mwu_step(a, _full_convolve(a, a), float(_full_convolve(a, a).max()), beta, eta)
            sc = _score(a)
            print(f"  MWU beta={beta:.6e}: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: Coordinate descent refinement (800 - 1200s)
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end:
        a = best_seq.copy()
        c = _full_convolve(a, a)
        M = float(c.max())
        a, c, M = _coord_descent(a, c, M, n, max_passes=2000, deadline=phase3_end)
        sc = _score(a)
        print(f"Phase 3 coord: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()

    # Phase 4: More LP from improved point (1200 - 1400s)
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp4 = 0
        while time.monotonic() < phase4_end:
            a = _lp_step_minmax(a, rows, cols, time_limit=25)
            sc = _score_norm(a)
            lp4 += 1
            if sc < prev_sc - 1e-12:
                stall = 0
            else:
                stall += 1
            if lp4 % 3 == 0:
                full_sc = _score(a)
                print(f"  LP2[{lp4}] score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
            if stall > 20:
                break
            prev_sc = sc
        print(f"Phase 4 done: {lp4} iters, best={best_score:.10f}")

    # Phase 5: Adam fine polishing (1400 - 1650s)
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        a_polished, sc_polished = _adam_smooth(
            best_seq, phase5_end - 5,
            beta_start=0.001 * max_cb, beta_end=0.00002 * max_cb,
            max_iter=300000, lr=0.006
        )
        full_sc = _score(a_polished)
        print(f"Phase 5 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_polished.copy()

    # Phase 6: Perturbed restarts + quick refinement (1650 - deadline)
    if time.monotonic() < deadline - 20:
        for attempt in range(4):
            if time.monotonic() > deadline - 15:
                break
            pert = _perturb(best_seq, 0.02 + 0.015 * attempt, rng)
            # Quick LP
            pert_sc = _score_norm(pert)
            for _ in range(8):
                if time.monotonic() > deadline - 10:
                    break
                pert = _lp_step_minmax(pert, rows, cols, time_limit=20)
            # Quick coord descent
            c_p = _full_convolve(pert, pert)
            pert, c_p, _ = _coord_descent(pert, c_p, float(c_p.max()), n, max_passes=200,
                                          deadline=deadline - 5)
            full_sc = _score(pert)
            print(f"  Restart[{attempt}]: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = pert.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
