"""
Optimized step function search using ADAM in log-space with log-sum-exp smoothing,
beta-continuation, exact active-set subgradient polishing, and multi-restart strategy.
Improvements over previous: ADAM optimizer, better beta schedule, active-set FISTA polishing,
and diverse initialization strategies.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _exact_grad(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    tol = 1e-4 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_optimize(z0, beta, n, deadline, max_iter=5000, lr=0.01):
    """ADAM optimizer in z-space for the smoothed objective."""
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_sc = _score(_softmax(z))
    best_z = z.copy()
    beta1_t = 1.0
    beta2_t = 1.0

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        beta1_t = beta1_t * beta1 + (1 - beta1)
        beta2_t = beta2_t * beta2 + (1 - beta2)

        grad = _smograd(z, beta, n)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1_t)
        v_hat = v / (1 - beta2_t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 200 == 0 or t == max_iter - 1:
            a = _softmax(z)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_z = z.copy()

    return best_z, best_sc

def _subgrad_polish(a_init, deadline, max_steps=10000, lr_init=0.02):

    z = np.log(np.maximum(a_init, 1e-15))

def _lbfgs_phase(a_start, beta, n, deadline, max_iter=3000):
    """Single L-BFGS-B phase."""
    from scipy.optimize import minimize

    z0 = np.log(np.maximum(a_start, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxcor': 40, 'maxls': 200, 'disp': False}
        )
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_start


def _make_profile(n, kind="uniform"):

    elif kind == "cosine":

    elif kind == "gaussian":

    elif kind == "bimodal":

    elif kind == "edge_boost":

    else:

def _beta_continuation_adam(z0, n, deadline, min_time=5.0):

    max_c = float(c.max())

        return z0

    num_betas = 12

    current_sc = _score(_softmax(z_cur))

        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:

        # Scale iterations by available time
        est_time_per_iter = 0.004 * (n / 2048.0)

        z_new, sc_new = _adam_optimize(z_cur, beta, n, deadline - min_time,

        a_new = _softmax(z_new)
        sc = _score(a_new)

        if sc < current_sc - 1e-12:
            current_sc = sc
            z_cur = z_new

    return z_cur


def _full_schedule(a_start, n, deadline, min_time=5.0, use_adam=True):
    """Full optimization schedule: ADAM/L-BFGS continuation + polishing."""
    z0 = np.log(np.maximum(a_start, 1e-15))
    z0 = z0 - z0.mean()

    if use_adam:
        z_out = _beta_continuation_adam(z0, n, deadline, min_time)
    else:
        # L-BFGS schedule
        a_cur = a_start.copy()
        c_cur = _full_convolve(a_cur, a_cur)
        mc = float(c_cur.max())
        betas = mc * 0.3 * (0.5 ** np.arange(10))
        for beta in betas:
            if time.monotonic() + min_time > deadline:
                break
            remaining = deadline - min_time - time.monotonic()
            if remaining < 3:
                break
            max_iter = min(2000, max(50, int(remaining / 0.02)))
            a_cur = _lbfgs_phase(a_cur, beta, n, deadline - min_time, max_iter)
        z_out = np.log(np.maximum(a_cur, 1e-15))
        z_out = z_out - z_out.mean()

    return _softmax(z_out), _score(_softmax(z_out))


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: combines ADAM in log-space, L-BFGS, FISTA active-set
    polishing, and subgradient methods with diverse initializations and restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, using uniform n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "init_uniform")

    # --- Phase 1: Main ADAM optimization from parent ---
    print(f"\nPhase 1: Main ADAM (t={time.monotonic() - t_start:.1f}s)")
    a_main, sc_main = _full_schedule(a0, n_main, deadline, min_time=300.0, use_adam=True)
    update_best(a_main, sc_main, "main_adam")

    # --- Phase 2: FISTA active-set polish ---
    print(f"\nPhase 2: FISTA polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        fista_deadline = min(deadline - 200.0, time.monotonic() + 180.0)
        a_fista, sc_fista = _fista_active_polish(best_seq, fista_deadline, max_iter=20000)
        update_best(a_fista, sc_fista, "fista")

    # --- Phase 3: Subgradient polish ---
    print(f"\nPhase 3: Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        sg_deadline = min(deadline - 100.0, time.monotonic() + 120.0)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, max_steps=5000, lr_init=0.015)
        update_best(a_sg, sc_sg, "subgrad")

    # --- Phase 4: L-BFGS fine-tuning with very small beta ---
    print(f"\nPhase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.0002, 0.0001, 0.00005, 0.00002])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 5:
                break
            a_new = _lbfgs_phase(best_seq, beta, n, deadline - 50, max_iter=1500)
            sc = _score(a_new)
            update_best(a_new, sc, f"fine_lbfgs_{i}")

    # --- Phase 5: Multi-restart with diverse initializations ---
    print(f"\nPhase 5: Multi-restart (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        n = len(best_seq)
        remaining = deadline - 30 - time.monotonic()
        n_restarts = min(5, max(1, int(remaining / 120)))

        strategies = [
            ("noise_small", 0.02),
            ("noise_med", 0.05),
            ("cosine_perturb", None),
            ("gaussian_perturb", None),
            ("fourier_perturb", None),
        ]

        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break

            label, sigma = strategies[i % len(strategies)]

            z_best = np.log(np.maximum(best_seq, 1e-15))
            z_best = z_best - z_best.mean()

            if label == "noise_small" or label == "noise_med":
                z_p = z_best + sigma * rng.standard_normal(n)
            elif label == "cosine_perturb":
                t = np.arange(n)
                freq = rng.integers(1, max(2, n // 8))
                amp = 0.03 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2 * np.pi)
                z_p = z_best + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
            elif label == "gaussian_perturb":
                t = np.linspace(-3, 3, n)
                center = rng.uniform(0.3, 0.7) * n
                width = rng.uniform(n // 10, n // 3)
                z_p = z_best + 0.03 * np.exp(-((t - center / n * 6) ** 2) / (2 * (width / n * 6) ** 2))
            else:
                z_p = z_best.copy()
                for _ in range(3):
                    freq = rng.integers(1, max(2, n // 4))
                    amp = 0.02 * rng.uniform(0.5, 2.0)
                    phase = rng.uniform(0, 2 * np.pi)
                    t = np.arange(n)
                    z_p = z_p + amp * np.cos(2.0 * np.pi * freq * t / n + phase)

            z_p = np.clip(z_p, -40.0, 40.0)
            z_p = z_p - z_p.mean()
            a_p = _softmax(z_p)

            sc_start = _score(a_p)
            print(f"  Restart {i} ({label}): start_score={sc_start:.10f}")

            # Run ADAM continuation
            a_r, sc_r = _full_schedule(a_p, n, deadline, min_time=30.0, use_adam=True)
            update_best(a_r, sc_r, f"restart_{i}_{label}")

            # Quick FISTA polish
            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _fista_active_polish(a_r, deadline - 10, max_iter=5000)
                update_best(a_r2, sc_r2, f"restart_{i}_fista")

    # --- Phase 6: Try different n values ---
    print(f"\nPhase 6: Different n (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        for n_try in [1024, 4096]:
            if time.monotonic() > deadline - 40:
                break
            # Initialize from parent by resampling or using a profile
            if parent is not None and len(parent) != n_try:
                # Interpolate
                x_old = np.linspace(0, 1, len(parent))
                x_new = np.linspace(0, 1, n_try)
                a_try = np.interp(x_new, x_old, parent)
                a_try = a_try / a_try.sum()
            else:
                a_try = _make_profile(n_try, "uniform")

            a_r, sc_r = _full_schedule(a_try, n_try, deadline, min_time=30.0, use_adam=True)
            update_best(a_r, sc_r, f"n_{n_try}")

            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _fista_active_polish(a_r, deadline - 15, max_iter=3000)
                update_best(a_r2, sc_r2, f"n_{n_try}_fista")

    # --- Phase 7: Final intensive polishing ---
    print(f"\nPhase 7: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # FISTA with small lr
        a_fin, sc_fin = _fista_active_polish(best_seq, deadline - 3, max_iter=20000,
                                             lr_init=0.005)
        update_best(a_fin, sc_fin, "final_fista")

        # Subgradient with very small lr
        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, max_steps=5000, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    sc_fb = _score(fallback)
    print(f"FALLBACK: score={sc_fb:.10f}")
    return [float(x) for x in fallback]
