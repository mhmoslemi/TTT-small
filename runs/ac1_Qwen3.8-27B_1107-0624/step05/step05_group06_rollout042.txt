import numpy as np

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _spectral_equalize(a, n_iters=100):
    """Alternate between flattening spectrum and enforcing non-negativity."""
    a = a.copy()
    for _ in range(n_iters):
        # Frequency domain: flatten magnitude
        A = np.fft.rfft(a)
        mag = np.abs(A)
        mean_mag = mag.mean()
        # Smooth the magnitude
        new_mag = mean_mag + 0.5 * (mag - mean_mag)  # partial flattening
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_real = np.fft.irfft(A_new, len(a))
        # Time domain: enforce non-negativity
        a_real = np.maximum(a_real, 0)
        s = a_real.sum()
        if s > 1e-15:
            a = a_real / s
        else:
            break
    return a

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()

        w = np.zeros_like(c)

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_optimize(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

        return _smoobj(z, beta, n)

        return _smograd(z, beta, n)

                'maxls': 50,

Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: spectral equalization pre-processing, L-BFGS with log-sum-exp
smoothing and aggressive beta-continuation, active-set subgradient polishing,
top-K lags, adaptive restarts, and multi-length search.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_topk(a, c, n, k):
    """Exact gradient using top-K lags with equal weight."""
    idx = np.argsort(c)[-k:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=0.002):
    """Exact gradient using all lags within tol_frac of max."""
    c_max = c.max()
    active = c >= (c_max - tol_frac * c_max)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_optimize(a_init, beta, n, max_iter=3000):
    """Run L-BFGS-B on the log-sum-exp smoothed objective."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxcor': 40,
                'maxls': 80,
                'disp': False,
            }
        )
        return _softmax(result.x), result.success
    except Exception:
        return a_init, False


def _spectral_equalize(a, n_iters=100, strength=0.3):
    """
    Alternate between flattening |FFT(a)| in frequency domain
    and enforcing non-negativity + normalization in time domain.
    """
    a = a.copy().astype(np.float64)
    for _ in range(n_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        mean_mag = mag.mean()
        # Partial flattening: pull magnitudes toward the mean
        new_mag = mean_mag + (1.0 - strength) * (mag - mean_mag)
        new_mag = np.maximum(new_mag, 1e-10)
        A_new = new_mag * np.exp(1j * np.angle(A))
        a_real = np.fft.irfft(A_new, len(a))
        a_real = np.maximum(a_real, 0.0)
        s = a_real.sum()
        if s > 1e-15:
            a = a_real / s
        else:
            break
    return a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.04, topk=1, tol_frac=0.002):
    """
    Subgradient polishing in z-space with adaptive step size.
    Uses active-set or top-K gradient.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)

        if topk > 0:
            grad = _gradient_topk(a, c, n, topk)
        else:
            grad = _gradient_active(a, c, n, tol_frac)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + 0.0003 * i)
        z = z - step * grad
        z = np.clip(z, -55.0, 55.0)
        z = z - z.mean()

        if i % 80 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                lr *= 0.92
                if patience > 25:
                    break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    """
    Run a full L-BFGS beta-continuation schedule.
    Uses a more gradual beta decrease for better convergence.
    """
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    # Gradual beta schedule (more phases, smoother decrease)
    ratios = [0.5, 0.3, 0.18, 0.1, 0.06, 0.035, 0.02, 0.012, 0.007,
              0.004, 0.0025, 0.0015, 0.0008, 0.0005, 0.0003, 0.00015]
    betas = max_c * np.array(ratios)
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        # Estimate time per iteration
        est_iter_time = 0.003 * (n / 2048.0) ** 0.7
        max_iter = min(4000, max(80, int(remaining / (est_iter_time * 12))))

        a_new, success = _lbfgs_optimize(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _local_search(a_init, deadline, n_trials=50000, delta=0.01):
    """
    Random local search: move small amounts of mass between pairs of indices.
    """
    n = len(a_init)
    a = a_init.copy()
    best_sc = _score(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for i in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Pick two random indices
        idx = rng.integers(0, 4)
        if idx == 0:
            # Move mass from i to j
            p, q = rng.integers(0, n), rng.integers(0, n)
            if p == q:
                continue
            amount = delta * min(a[p], a[q])
            a_new = a.copy()
            a_new[p] -= amount
            a_new[q] += amount
            if a_new[p] < 0:
                continue
        elif idx == 1:
            # Blend two adjacent elements
            p = rng.integers(0, n - 1)
            q = p + 1
            w = rng.uniform(0.1, 0.9)
            a_new = a.copy()
            old_p, old_q = a_new[p], a_new[q]
            a_new[p] = w * old_p + (1 - w) * old_q
            a_new[q] = (1 - w) * old_p + w * old_q
        else:
            # Squeeze two elements toward their mean
            p, q = rng.integers(0, n), rng.integers(0, n)
            if p == q:
                continue
            a_new = a.copy()
            mean_val = (a_new[p] + a_new[q]) / 2
            a_new[p] = mean_val
            a_new[q] = mean_val

        sc = _score(a_new)
        if sc < best_sc - 1e-15:
            a = a_new
            best_sc = sc
            best_a = a.copy()
        # Accept with small probability (simulated annealing-like)
        elif rng.random() < 0.001:
            a = a_new

    return best_a, best_sc


def _interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    if len(a) < 2:
        return np.ones(new_n) / new_n
    f = interp1d(x_old, a, kind='cubic', fill_value='extrapolate')
    a_new = f(x_new)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        rng = np.random.default_rng(99)
        # Random smooth signal via random Fourier coefficients
        coeffs = rng.standard_normal(n) + 1j * rng.standard_normal(n)
        # Low-pass filter
        for k in range(n // 3, n):
            coeffs[k] = 0
        a = np.abs(np.fft.irfft(coeffs * n, n))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

        sc_se2 = _score(a_se2)
        print(f"  Spectral eq (str=0.5): {sc_se2:.10f}")
        update_best(a_se2, sc_se2, "spec_eq_05")

            print(f"  L-BFGS: {sc_lb:.10f}")

                if time.monotonic() < deadline - 50:
                    a_sg2, sc_sg2 = _subgrad_polish(a_sg, deadline - 50,
