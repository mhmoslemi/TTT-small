def _nesterov_smoothed(a_init, deadline, beta_init, n_steps_max=50000):
    """Nesterov-accelerated gradient descent on the smoothed objective (simplex parameterization)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Nesterov momentum
    z_prev = z.copy()
    t = 1.0
    lr = beta_init  # initial learning rate (will adapt)
    
    for step in range(n_steps_max):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        t_new = 1.0 / (1.0 - np.sqrt(1.0 - 1.0 / (t + 1)))  # or use t_new = t + 1
        # Simpler: use the standard Nesterov scheme
        z_new = z + (t / (t + 3.0)) * (z - z_prev)
        z_prev = z.copy()
        t += 1.0
        
        # Compute gradient at z_new
        a_new = _softmax_fn(z_new)
        c = _full_convolve(a_new, a_new)
        cm = float(c.max())
        beta = max(1e-8, 0.02 * cm)
        vals = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a_new[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a_new * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Gradient step with decreasing lr
        lr = 2.0 / (t + 5.0)  # Nesterov step size
        z_new = z_new - lr * grad
        z_new = np.clip(z_new, -100, 100)
        z_new = z_new - z_new.mean()
        
        z = z_new
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if step % 5000 == 0:
            print(f"    Nesterov step {step}: score={sc:.10f}, best={best_score:.10f}, beta={beta:.6f}")
    
    return best_a, best_score


def _run_lbfgs(a_init, deadline, beta_list=None):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
        beta_list = [b * max_c0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        time_per = min(60.0, (deadline - time.monotonic()) / max(1, len(beta_list)))
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 1000, 'ftol': 1e-25, 'gtol': 1e-16, 'maxls': 50}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _coord_descent_subgrad(a_init, deadline, max_rounds=100):
    """Coordinate descent where each coordinate is optimized by 1D search."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = best_a.copy()
        improved = False
        # Randomize coordinate order
        order = np.random.default_rng(rnd * 12345 + 7).permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            # Try several perturbations at coordinate j
            best_local = best_score
            best_val = a[j]
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                a_trial = a.copy()
                a_trial[j] = a[j] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_val = a_trial[j]
            if best_val != a[j]:
                a[j] = best_val
                s = a.sum()
                a = a / s
                if _score_norm_fn(a) < best_score:
                    best_score = _score_norm_fn(a)
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    
    return best_a, best_score


def _structured_restart(a, rng, n_perturb=10):
    """Generate perturbed versions of a using structured perturbations."""
    n = len(a)
    x = np.linspace(0, 1, n)
    candidates = [a.copy()]
    
    for i in range(n_perturb):
        a_pert = a.copy()
        pert_type = rng.integers(0, 5)
        eps = rng.uniform(0.05, 0.3)
        
        if pert_type == 0:
            # Low-frequency sinusoidal
            k = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            a_pert = a_pert * (1.0 + eps * np.sin(2 * np.pi * k * x + phase))
        elif pert_type == 1:
            # Polynomial perturbation
            deg = rng.integers(1, 6)
            coeffs = rng.normal(0, eps / deg, deg + 1)
            poly = np.zeros(n)
            for d in range(deg + 1):
                poly += coeffs[d] * x**d
            a_pert = a_pert * (1.0 + poly)
        elif pert_type == 2:
            # Local bump
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.3)
            a_pert = a_pert * (1.0 + eps * np.exp(-((x - center) / width) ** 2))
        elif pert_type == 3:
            # Edge emphasis
            edge = rng.choice([0, 1])
            decay = rng.uniform(2, 8)
            if edge == 0:
                a_pert = a_pert * (1.0 + eps * np.exp(-decay * x))
            else:
                a_pert = a_pert * (1.0 + eps * np.exp(-decay * (1 - x)))
        else:
            # Multi-scale noise
            k1 = rng.integers(1, 10)
            k2 = rng.integers(10, 50)
            p1 = rng.uniform(0, 2 * np.pi)
            p2 = rng.uniform(0, 2 * np.pi)
            a_pert = a_pert * (1.0 + eps * 0.5 * np.sin(2 * np.pi * k1 * x + p1) + 
                              eps * 0.1 * np.sin(2 * np.pi * k2 * x + p2))
        
        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()
        candidates.append(a_pert)
    
    return candidates


def _lp_minimax(f, deadline):
    """Solve the LP: max sum(g) s.t. f*g <= M, sum(g)=1, g>=0"""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    
    # Build the constraint matrix A where (A@g)_k = (f*g)_k
    # A is (2n-1) x n, A[k,j] = f[k-j] if 0<=k-j<n else 0
    row_idx = []
    col_idx = []
    data = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            row_idx.append(k)
            col_idx.append(j)
            data.append(f[k - j])
    
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    g = cp.Variable(n)
    constraints = [A @ g <= M, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 0.01:
                g_val = g_val / s
                return g_val
    except Exception:
        pass
    return None


def _lp_iterate(a, n_iters, deadline, step_size=0.3):
    """Iterate: find LP direction, blend, repeat."""
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        g = _lp_minimax(a, deadline)
        if g is None:
            break
        a_new = (1 - step_size) * a + step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}")
    
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    profiles.append(np.ones(n) / n)
    for slope in [0.5, 1.0, -1.0, 2.0]:
        a = 1.0 + slope * u
        profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    for rate in [1.0, -1.0, 2.0]:
        a = np.exp(rate * u)
        profiles.append(a / a.sum())
    for p in [1.0, 2.0, 3.0]:
        a = np.maximum(1 - u**2, 1e-12) ** p
        profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _spline_param_optimize(a, n_ctrl, deadline):
    """Optimize a B-spline parameterization of a."""
    from scipy.interpolate import BSpline
    n = len(a)
    x = np.linspace(0, 1, n)
    # Create knot vector for n_ctrl control points
    # Using clamped cubic B-spline
    n_knots = n_ctrl + 4  # for cubic
    t_knots = np.zeros(n_knots)
    t_knots[3:-3] = np.linspace(0, 1, n_ctrl - 2)
    t_knots[0] = -1
    t_knots[-1] = 2
    t_knots[1] = 0
    t_knots[2] = 0
    t_knots[-2] = 1
    t_knots[-3] = 1
    # Actually let me use a simpler approach: just use a polynomial or Fourier basis
    
    # Use trigonometric basis: a(x) = sum_k c_k * cos(2*pi*k*x) + d_k * sin(2*pi*k*x)
    # This is equivalent to a low-pass filtered version
    n_freq = n_ctrl // 2
    x_grid = np.linspace(0, 1, n)
    
    def spline_from_coeffs(coeffs):
        """Reconstruct a from Fourier coefficients."""
        s = np.ones(n) * (a.sum() / n)  # DC term
        for k in range(1, n_freq + 1):
            idx = 2 * (k - 1)
            c_k = coeffs[idx]
            d_k = coeffs[idx + 1]
            s += c_k * np.cos(2 * np.pi * k * x_grid) + d_k * np.sin(2 * np.pi * k * x_grid)
        s = np.maximum(s, 1e-12)
        return s / s.sum()
    
    def spline_score(coeffs):
        s = spline_from_coeffs(coeffs)
        return _score_norm_fn(s)
    
    # Initial coefficients from FFT of a
    coeffs_init = np.zeros(2 * n_freq)
    a_norm = a - a.mean()
    for k in range(1, n_freq + 1):
        idx = 2 * (k - 1)
        coeffs_init[idx] = 2 * np.sum(a_norm * np.cos(2 * np.pi * k * x_grid)) / n
        coeffs_init[idx + 1] = 2 * np.sum(a_norm * np.sin(2 * np.pi * k * x_grid)) / n
    
    # Optimize
    result = minimize(spline_score, coeffs_init, method='L-BFGS-B',
                      options={'maxiter': 2000, 'ftol': 1e-20})
    
    if result.fun < _score_norm_fn(a):
        a_opt = spline_from_coeffs(result.x)
        return a_opt, float(result.fun)
    return a, _score_norm_fn(a)

    print(f"Starting: n={n_curr}, score={best_score:.10f}")

    # Phase 1: LP iterations (200s)
    phase1_end = min(deadline, t_start + 200.0)
    print(f"\nPhase 1: LP iterations")
    
    # Use n=2048 for LP (faster)
    n_lp = min(n_curr, 2048)
    a_lp = _interp_profile(best_seq, n_lp) if n_curr > n_lp else best_seq.copy()
    
    for round_num in range(4):
        if time.monotonic() > phase1_end:
            break
        step = 0.6 / (round_num + 1)
        a_lp, sc_lp = _lp_iterate(a_lp, n_iters=20, deadline=phase1_end, step_size=step)
        if n_curr > n_lp:
            a_rec = _interp_profile(a_lp, n_curr)
            sc_rec = _score_norm_fn(a_rec)
            update_best(a_rec, sc_rec, f"lp_r{round_num}_interp")
        else:
            update_best(a_lp, sc_lp, f"lp_r{round_num}")

    # Phase 2: Nesterov accelerated smoothed GD (300s)
    phase2_end = min(deadline, t_start + 500.0)
    print(f"\nPhase 2: Nesterov smoothed GD (n={n_curr})")
    
    beta_init = 0.05 * float(_full_convolve(best_seq, best_seq).max())
    a_nest, sc_nest = _nesterov_smoothed(best_seq, phase2_end, beta_init)
    update_best(a_nest, sc_nest, "nesterov")

    # Phase 3: L-BFGS continuation (400s)
    phase3_end = min(deadline, t_start + 900.0)
    print(f"\nPhase 3: L-BFGS continuation")
    
    for restart in range(3):
        if time.monotonic() > phase3_end - 30:
            break
        if restart == 0:
            a_cur = best_seq.copy()
        else:
            # Perturb
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_scale = 0.03 * (restart + 1)
            z = z + noise_scale * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
        
        per_dl = min(phase3_end - 10, time.monotonic() + 120.0)
        a_lb, sc_lb = _run_lbfgs(a_cur, per_dl)
        update_best(a_lb, sc_lb, f"lbfgs_r{restart}")

    # Phase 4: Structured restarts + re-optimization (350s)
    phase4_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 4: Structured restarts")
    
    n_restarts = 8
    per_restart_time = (phase4_end - time.monotonic()) / max(1, n_restarts)
    
    for r in range(n_restarts):
        if time.monotonic() > phase4_end - 10:
            break
        
        candidates = _structured_restart(best_seq, rng, n_perturb=3)
        for c in candidates:
            if time.monotonic() > phase4_end - 10:
                break
            sc_c = _score_norm_fn(c)
            if sc_c < best_score:
                update_best(c, sc_c, f"restart_r{r}")
                # Quick L-BFGS on the improved candidate
                per_dl = min(phase4_end - 5, time.monotonic() + 30.0)
                a_occ, sc_occ = _run_lbfgs(c, per_dl, beta_list=[0.001 * float(_full_convolve(c,c).max())])
                update_best(a_occ, sc_occ, f"restart_r{r}_refine")

    # Phase 5: Spline/Fourier shape optimization (150s)
    phase5_end = min(deadline, t_start + 1400.0)
    print(f"\nPhase 5: Fourier shape optimization")
    
    if time.monotonic() < phase5_end:
        per_dl = min(phase5_end - 5, time.monotonic() + 100.0)
        a_spline, sc_spline = _spline_param_optimize(best_seq, 20, per_dl)
        update_best(a_spline, sc_spline, "spline")

    # Phase 6: Final polishing (remaining time)
    print(f"\nPhase 6: Final polishing")
    
    if time.monotonic() < deadline - 10 and best_seq is not None:
        # Coordinate descent
        per_dl = min(deadline - 30, time.monotonic() + 100.0)
        a_cd, sc_cd = _coord_descent_subgrad(best_seq, per_dl, max_rounds=50)
        update_best(a_cd, sc_cd, "coord_descent")
        
        # Final ultra-fine L-BFGS
        if time.monotonic() < deadline - 10:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_ultra = 1e-5 * max_c_final
            per_dl2 = min(deadline - 5, time.monotonic() + 50.0)
            a_ult, sc_ult = _run_lbfgs(best_seq, per_dl2, beta_list=[beta_ultra, beta_ultra * 0.5])
            update_best(a_ult, sc_ult, "ultra_lbfgs")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            pass  # Keep as-is, score is scale-invariant
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[0]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
