"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines L-BFGS with log-sum-exp smoothing, active-set projected gradient with Nesterov
momentum, power-smoothed optimization, and structured local search across multiple
sequence lengths.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _power_smoobj(a, p, n):
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-300:
        return 0.0
    shifted = c / c_max
    sp = np.sum(shifted ** p)
    return 2.0 * n * c_max * (sp / len(c)) ** (1.0 / p)


def _power_smograd(a, p, n):
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-300:
        return np.zeros_like(a)
    shifted = c / c_max
    sp = np.sum(shifted ** p)
    # d/d a of c_max * (sp/(2n-1))^(1/p)
    # = c_max * (1/p) * (sp/(2n-1))^(1/p - 1) * (1/(2n-1)) * p * shifted^(p-1) * (1/c_max) * dc/da
    # = (sp/(2n-1))^(1/p - 1) * (1/(2n-1)) * shifted^(p-1) * dc/da
    L = len(c)
    coeff = (sp / L) ** (1.0 / p - 1.0) / L
    w = shifted ** (p - 1)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * n * coeff * h


def _project_simplex(x, target_sum=1.0):
    n = len(x)
    u = np.sort(x)[::-1]
    cv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.where(cond)[0][-1]
    theta = (cv[rho] - target_sum) / (rho + 1)
    return np.maximum(x - theta, 0.0)


def _active_set_grad(a, n, top_k=None, rel_tol=0.05):
    """Compute gradient of average of top-k (or within rel_tol of max) convolution values."""
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-300:
        return np.zeros_like(a), c_max
    
    if top_k is not None:
        idx = np.argsort(c)[::-1][:top_k]
        active = np.zeros_like(c)
        active[idx] = 1.0
        active /= active.sum()
    else:
        threshold = c_max * (1.0 - rel_tol)
        active = (c >= threshold).astype(float)
        if active.sum() == 0:
            active[int(np.argmax(c))] = 1.0
        else:
            active /= active.sum()
    
    a_rev = a[::-1]
    wc = _full_convolve(active, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h, c_max


def _lbfgs_phase(a_init, deadline, n, min_time=5.0):
    """Run L-BFGS beta continuation from a_init."""
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init
    
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break
        
        est_iter_time = 0.004 * (n / 2048.0)
        max_iter = min(2000, max(30, int(remaining / (est_iter_time * 10))))
        
        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()
        
        def fun(z):
            return _smoobj(z, beta, n)
        
        def jac(z):
            return _smograd(z, beta, n)
        
        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'maxcor': 20, 'maxls': 50,
                                       'disp': False})
            a_new = _softmax(result.x)
            sc = _score(a_new)
            if sc < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc
        except Exception:
            break
    
    return current_a


def _gradient_descent_with_momentum(a_init, deadline, top_k=10, rel_tol=0.03,
                                     lr_init=0.01, momentum_init=0.9, max_steps=5000):
    """Projected gradient descent with Nesterov momentum on the active set objective."""
    n = len(a_init)
    a = a_init / a_init.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    momentum = momentum_init
    patience = 500
    best_sc_run = best_sc
    
    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov look-ahead
        a_lookup = np.maximum(a + momentum * v, 0)
        a_lookup = a_lookup / a_lookup.sum()
        
        # Compute gradient at look-ahead point
        grad, c_max = _active_set_grad(a_lookup, n, top_k=top_k, rel_tol=rel_tol)
        
        # Update velocity
        v = momentum * v - lr * grad
        a_new = a + v
        
        # Project to simplex
        a_new = _project_simplex(a_new, 1.0)
        
        # Evaluate
        sc = _score(a_new)
        
        if sc < best_sc_run:
            best_sc_run = sc
            patience = 500
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        else:
            patience -= 1
        
        a = a_new
        
        # Adaptive learning rate
        if patience % 100 == 0 and patience < 400:
            lr *= 0.9
            momentum = min(0.99, momentum * 1.001)
        
        if patience <= 0:
            break
        
        # Periodically switch to exact top-1 gradient
        if step % 500 == 499:
            grad_exact, _ = _active_set_grad(a, n, top_k=1, rel_tol=0.0)
            v = 0.5 * v - lr * grad_exact
            a = _project_simplex(a + v, 1.0)
    
    return best_a, best_sc


def _power_smo_lbfgs(a_init, deadline, n):
    """L-BFGS with power smoothing (p-continuation)."""
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    p_values = [4, 8, 16, 32, 64, 128, 256, 512]
    
    for p in p_values:

            break
        remaining = deadline - 5 - time.monotonic()

            break
        
        max_iter = min(500, max(20, int(remaining / 0.05)))
        
        def fun(a_vec):
            a_p = _project_simplex(a_vec, 1.0)
            return _power_smoobj(a_p, p, n)
        
        def jac(a_vec):
            a_p = _project_simplex(a_vec, 1.0)
            return _power_smograd(a_p, p, n)
        
        try:
            result = minimize(fun, current_a, jac=jac, method='L-BFGS-B',

            a_new = _project_simplex(result.x, 1.0)
            sc = _score(a_new)
            if sc < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc
        except Exception:
            break
    
    return current_a

            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 2048
        parent = _make_profile(n_parent, "uniform")
        sc_parent = _score(parent)
        print(f"No parent, uniform n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    # --- Strategy: Allocate time across different approaches and lengths ---
    # Primary: optimize parent at its native length
    # Secondary: try other lengths with different initializations
    # Tertiary: restarts and local search on best found
    
    n_main = len(parent)
    
    # Phase A: Main optimization on parent (50% of time)
    print(f"\n[A] Main optimization n={n_main} (t={time.monotonic()-t_start:.1f}s)")
    deadline_a = t_start + (deadline - t_start) * 0.50
    a_main = _optimize_sequence(parent, deadline_a, n_main, min_time_remaining=30.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "main_opt")
    
    # Phase B: Try different sequence lengths (25% of time)
    print(f"\n[B] Multi-length exploration (t={time.monotonic()-t_start:.1f}s)")
    deadline_b = t_start + (deadline - t_start) * 0.75
    alt_lengths = [1024, 4096]
    # Remove the main length if it's already in the list
    alt_lengths = [l for l in alt_lengths if l != n_main]
    
    for n_alt in alt_lengths:
        if time.monotonic() > deadline_b - 60:
            break
        remaining_b = deadline_b - 60 - time.monotonic()
        if remaining_b < 30:
            break
        
        # Create initialization by resampling or using profile
        if best_seq is not None:
            # Resample best_seq to new length
            old_n = len(best_seq)
            t_old = np.linspace(0, 1, old_n)
            t_new = np.linspace(0, 1, n_alt)
            a_init = np.interp(t_new, t_old, best_seq)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        else:
            a_init = _make_profile(n_alt, "cosine")
        
        sc_init = _score(a_init)
        print(f"  n={n_alt}: init_score={sc_init:.10f}")
        
        t_sub = min(time.monotonic() + remaining_b * 0.45, deadline_b - 60)
        a_alt = _optimize_sequence(a_init, t_sub, n_alt, min_time_remaining=20.0)
        sc_alt = _score(a_alt)
        update_best(a_alt, sc_alt, f"len_{n_alt}")
    
    # Phase C: Restart and refinement on best (remaining time)
    print(f"\n[C] Refinement and restarts (t={time.monotonic()-t_start:.1f}s)")
    deadline_c = deadline
    
    if best_seq is not None and time.monotonic() < deadline_c - 30:
        # Refine current best with aggressive gradient
        n_best = len(best_seq)
        t_refine = min(time.monotonic() + (deadline_c - time.monotonic()) * 0.3, deadline_c - 30)
        a_ref, sc_ref = _gradient_descent_with_momentum(
            best_seq, t_refine, top_k=5, rel_tol=0.02,
            lr_init=0.003, momentum_init=0.92, max_steps=3000)
        update_best(a_ref, sc_ref, "refine_grad")
        
        # Local search
        if time.monotonic() < deadline_c - 15:
            a_ls, sc_ls = _local_search(best_seq, deadline_c - 15, n_trials=300)
            update_best(a_ls, sc_ls, "local_search")
        
        # Restart with perturbation
        if time.monotonic() < deadline_c - 20:
            a_pert = _perturb(best_seq, 0.02, rng)
            t_restart = min(time.monotonic() + (deadline_c - time.monotonic()) * 0.4, deadline_c - 10)
            a_rs = _optimize_sequence(a_pert, t_restart, n_best, min_time_remaining=10.0)
            update_best(a_rs, _score(a_rs), "restart_pert")
        
        # Final micro-optimization
        if time.monotonic() < deadline_c - 5:
            a_fin, sc_fin = _gradient_descent_with_momentum(
                best_seq, deadline_c - 5, top_k=2, rel_tol=0.01,
                lr_init=0.001, momentum_init=0.95, max_steps=1000)
            update_best(a_fin, sc_fin, "final_micro")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
