from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    """
    Exact gradient of 2n * beta * logsumexp(c/beta) w.r.t. log-params z.

    a_rev = a[::-1]
    h1 = _full_convolve(w, a_rev)[n - 1:2 * n - 1]
    h2 = _full_convolve(w, a)[:n]
    G = 2.0 * n * (h1 + h2)
    G_dot_a = float(np.dot(G, a))
    return a * (G - G_dot_a)
