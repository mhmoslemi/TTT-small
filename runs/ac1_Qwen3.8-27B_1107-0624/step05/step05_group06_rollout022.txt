"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over
non-negative sequences. Combines L-BFGS with log-sum-exp smoothing,
exact coordinate descent, and Nesterov-accelerated subgradient methods.
"""
import numpy as np
import time
from scipy.optimize import minimize

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def smoobj(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def smograd(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def gradient_active(z, a, c, n, rel_tol=0.005):
    c_max = c.max()
    active = c >= (c_max - rel_tol * c_max)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def lbfgs_optimize(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return smoobj(z, beta, n)
    
    def jac(z):
        return smograd(z, beta, n)
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxcor': 40,
                'maxls': 200,
                'disp': False,
            }
        )
        a_out = softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a_init.copy(), False, 0

def subgrad_polish(a_init, deadline, n_max_steps=10000, lr_init=0.05, use_nesterov=True):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = score(a_init)
    lr = lr_init
    
    # Nesterov momentum
    z_mom = z.copy()
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        if use_nesterov:
            z_eval = z_mom + (i / (i + 1 + 200)) * (z_mom - z)
        else:
            z_eval = z
        
        a = softmax(z_eval)
        c = full_convolve(a, a)
        grad = gradient_active(z_eval, a, c, n, rel_tol=0.002)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        step = lr / (1.0 + 0.01 * i)
        z_mom = z_mom - step * grad
        z_mom = np.clip(z_mom, -50.0, 50.0)
        z_mom = z_mom - z_mom.mean()
        
        if not use_nesterov:
            z = z_mom.copy()
        
        if i % 100 == 0:
            a_check = softmax(z if not use_nesterov else z_mom)
            sc = score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            else:
                lr *= 0.95
    
    a_final = softmax(z if not use_nesterov else z_mom)
    sc_final = score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def coordinate_descent(a_init, deadline, n_sweeps=200, step_tol=1e-12):
    """
    Exact coordinate descent: for each coordinate, find the optimal value
    via golden section search on the 1D subproblem.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()  # normalize
    
    R = full_convolve(a, a)
    max_R = R.max()
    S_total = a.sum()  # = 1 after normalization
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        max_improvement = 0.0
        old_max_R = max_R
        
        for i in range(n):
            if time.monotonic() > deadline:
                break
            
            x_old = a[i]
            S = S_total - x_old  # sum of all except a[i]
            
            # Affected k range
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            
            if k_lo > k_hi:
                continue
            
            # Max over unaffected k
            if k_lo > 0 or k_hi < 2 * n - 2:
                # Build mask
                unaffected = np.ones(2 * n - 1, dtype=bool)
                unaffected[k_lo:k_hi + 1] = False
                C = R[unaffected].max()
            else:
                C = 0.0
            
            # For affected k, compute the linear coefficients and constants
            k_range = np.arange(k_lo, k_hi + 1)
            # a[k-i] for k in k_range: this is a[0 : k_hi - k_lo + 1]
            a_ki = a[0:k_hi - k_lo + 1].copy()
            
            # Contribution of a[i] to R[k]:
            # For k != 2i: 2 * a[i] * a[k-i]
            # For k == 2i: a[i]^2
            contrib = 2.0 * x_old * a_ki
            if 2 * i >= k_lo and 2 * i <= k_hi:
                idx_2i = 2 * i - k_lo
                contrib[idx_2i] = x_old * x_old
            else:
                idx_2i = -1
            
            c_aff = R[k_lo:k_hi + 1] - contrib
            
            # Now define f(x) = 2*n * max(C, 2*x*a_ki + c_aff, [x^2 + c_aff[idx_2i]]) / (x + S)^2
            # Minimize over x in [0, x_hi]
            
            # Upper bound for x: the current value times some factor, or a reasonable bound
            x_hi = max(3.0 * x_old + 1.0 / n, 5.0 / n)
            
            def f_1d(x):
                # Compute R[k](x) for affected k
                r_aff = 2.0 * x * a_ki + c_aff
                if idx_2i >= 0:
                    r_aff[idx_2i] = x * x + c_aff[idx_2i]
                r_max = max(C, r_aff.max())
                denom = (x + S) ** 2
                if denom < 1e-30:
                    return 1e30
                return 2.0 * n * r_max / denom
            
            # Golden section search
            a_lo = 0.0
            a_hi = x_hi
            
            # Evaluate at initial points
            x1 = a_lo + gr * (a_hi - a_lo)
            x2 = a_hi - gr * (a_hi - a_lo)
            f1 = f_1d(x1)
            f2 = f_1d(x2)
            
            best_x = x_old
            best_f = f_1d(x_old)
            
            for _ in range(50):
                if f1 < f2:
                    a_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = a_lo + gr * (a_hi - a_lo)
                    f1 = f_1d(x1)
                else:
                    a_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = a_hi - gr * (a_hi - a_lo)
                    f2 = f_1d(x2)
                
                # Track best
                mid = (a_lo + a_hi) / 2
                f_mid = f_1d(mid)
                if f_mid < best_f:
                    best_f = f_mid
                    best_x = mid
            
            # Also check x=0
            f_0 = f_1d(0.0)
            if f_0 < best_f:
                best_f = f_0
                best_x = 0.0
            
            # Update
            if abs(best_x - x_old) > step_tol:
                a[i] = best_x
                S_total = a.sum()
                a = a / S_total
                S_total = 1.0
                R = full_convolve(a, a)
                max_R = R.max()
                max_improvement = max(max_improvement, abs(best_x - x_old))
            
            if time.monotonic() > deadline:
                break
        
        # Check convergence
        if max_improvement < step_tol * 10:
            if sweep > 5:
                break
    
    return a

def perturb_small(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return softmax(z)

"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative
sequences. Combines L-BFGS with log-sum-exp smoothing, exact coordinate descent
with golden section 1D search, and Nesterov-accelerated subgradient polishing.

import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def obj_fun(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def obj_jac(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def grad_active(a, c, n, rel_tol=0.003):
    c_max = c.max()
    active = c >= (c_max - rel_tol * c_max)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def eval_1d(x, a_ki, c_aff, C_const, idx_2i, n_val, S_val):
    r_aff = 2.0 * x * a_ki + c_aff
    if idx_2i >= 0:
        r_aff[idx_2i] = x * x + c_aff[idx_2i]
    r_max = max(C_const, float(r_aff.max()))
    denom = (x + S_val) ** 2
    if denom < 1e-30:
        return 1e30
    return 2.0 * n_val * r_max / denom


def coordinate_descent(a_init, deadline, n_sweeps=200):
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()

    R = full_convolve(a, a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        max_change = 0.0

        for i in range(n):
            if time.monotonic() > deadline:
                break

            x_old = a[i]
            S = 1.0 - x_old

            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            if k_lo > k_hi:
                continue

            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                C_const = float(R[mask].max())
            else:
                C_const = 0.0

            length = k_hi - k_lo + 1
            a_ki = a[0:length].copy()

            contrib = 2.0 * x_old * a_ki
            idx_2i = -1
            if 2 * i >= k_lo and 2 * i <= k_hi:
                idx_2i = 2 * i - k_lo
                contrib[idx_2i] = x_old * x_old

            c_aff = R[k_lo:k_hi + 1] - contrib

            x_hi = max(4.0 * x_old + 2.0 / n, 5.0 / n)
            a_lo = 0.0
            a_hi = x_hi

            x1 = a_lo + gr * (a_hi - a_lo)
            x2 = a_hi - gr * (a_hi - a_lo)
            f1 = eval_1d(x1, a_ki, c_aff, C_const, idx_2i, n, S)
            f2 = eval_1d(x2, a_ki, c_aff, C_const, idx_2i, n, S)

            best_x = x_old
            best_f = eval_1d(x_old, a_ki, c_aff, C_const, idx_2i, n, S)

            for _ in range(35):
                if f1 < f2:
                    a_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = a_lo + gr * (a_hi - a_lo)
                    f1 = eval_1d(x1, a_ki, c_aff, C_const, idx_2i, n, S)
                else:
                    a_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = a_hi - gr * (a_hi - a_lo)
                    f2 = eval_1d(x2, a_ki, c_aff, C_const, idx_2i, n, S)

                mid = (a_lo + a_hi) / 2.0
                f_mid = eval_1d(mid, a_ki, c_aff, C_const, idx_2i, n, S)
                if f_mid < best_f:
                    best_f = f_mid
                    best_x = mid

            f_0 = eval_1d(0.0, a_ki, c_aff, C_const, idx_2i, n, S)
            if f_0 < best_f:
                best_f = f_0
                best_x = 0.0

            if abs(best_x - x_old) > 1e-10:
                a[i] = best_x
                s_new = a.sum()
                a = a / s_new
                R = full_convolve(a, a)
                max_change = max(max_change, abs(best_x - x_old))

        if time.monotonic() > deadline:
            break
        if sweep > 3 and max_change < 1e-12:
            break

    return a
