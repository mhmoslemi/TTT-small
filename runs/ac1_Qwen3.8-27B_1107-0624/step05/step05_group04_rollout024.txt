```python
"""
Multi-scale optimization with Nesterov acceleration and LP-based peak reduction.
Strategy:
1. Multi-resolution: optimize at n=128 through n=1024 with L-BFGS + log-sum-exp.
2. Upsample to n=4096, then use Nesterov-accelerated gradient descent with adaptive beta.
3. Novel LP-based peak reduction: solve successive linear programs to target top peaks.
4. Final polish with very fine-grained gradient steps and peak surgery.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    new_a = new_a / new_a.sum()
    return new_a


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.05 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.3 * gap)


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_make_ramp(n, s))
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Triangle (symmetric)
    tri = 1.0 - 2.0 * np.abs(t - 0.5)
    tri = np.maximum(tri, 1e-12)
    profiles.append(tri / tri.sum())
    # Cosine bump
    cos_b = 0.5 + 0.5 * np.cos(2 * np.pi * (t - 0.5))
    cos_b = np.maximum(cos_b, 1e-12)
    profiles.append(cos_b / cos_b.sum())
    # Power-law decay from center
    for p in [0.5, 1.0, 2.0]:
        dist = np.abs(t - 0.5)
        pl = (1.0 - dist) ** p + 1e-8
        profiles.append(pl / pl.sum())
    return profiles


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.3 * max_c0
    beta_end = 1e-6 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget

        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 3000)
        maxfun = maxiter * 5

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_nesterov(a_init, deadline, rng, lr_init=0.5, momentum_init=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    z_prev = z.copy()
    momentum = momentum_init
    lr = lr_init
    t = 0

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        if it % 200 == 0:
            beta = _adaptive_beta(c)

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        # Nesterov step
        z = z + momentum * (z - z_prev)
        z_prev = z.copy()

        # Evaluate gradient at y (Nesterov look-ahead)
        a_y = _softmax_fn(z)
        c_y = _full_convolve(a_y, a_y)
        cm_y = float(c_y.max())
        shifted_y = (c_y - cm_y) / beta
        e_y = np.exp(shifted_y)
        w_y = e_y / e_y.sum()
        a_rev_y = a_y[::-1]
        w_conv_y = _full_convolve(a_rev_y, w_y)
        h_y = w_conv_y[n - 1:2 * n - 1]
        grad_a_y = 2.0 * h_y
        offset_y = float(np.dot(a_y, grad_a_y))
        grad_z_y = a_y * (grad_a_y - offset_y)

        if not np.all(np.isfinite(grad_z_y)):
            z = z_prev.copy()
            break

        z = z - lr * grad_z_y
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive momentum and lr
        if t % 5000 == 0:
            momentum = max(0.5, momentum * 0.999)
            lr = max(0.01, lr * 0.95)

        # Periodic kick
        if t % 20000 == 0:
            noise = 0.02 * rng.normal(size=n)
            z = z + noise
            z = z - z.mean()
            z_prev = z.copy()

    return best_a, best_score


def _run_lp_refinement(a_init, deadline, rng, max_iters=80):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(best_a, best_a)
    current_max = float(c.max())

    cutting_planes = []  # List of (coeff_vector, rhs)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a_cur = best_a.copy()
        c_cur = _full_convolve(a_cur, a_cur)
        max_cur = float(c_cur.max())

        # Add cutting planes for top peaks
        peak_indices = np.argsort(c_cur)[::-1]
        n_peaks = min(30, len(peak_indices))

        for pk in peak_indices[:n_peaks]:
            # Compute gradient of f_pk at a_cur
            # grad_i = a_cur[pk-i] + a_cur[i] (with a_cur[j]=0 outside [0,n-1])
            grad = np.zeros(n)
            for i in range(n):
                g = 0.0
                if 0 <= pk - i < n:
                    g += a_cur[pk - i]
                g += a_cur[i]
                grad[i] = g

            # f_pk(a_cur) = conv_cur[pk]
            rhs = float(c_cur[pk])
            cutting_planes.append((grad, rhs))

        # Limit cutting planes
        if len(cutting_planes) > 100:
            # Keep only the most "active" ones
            # Active = where the constraint is tight (grad @ a ≈ rhs)
            scores = []
            for idx, (g, r) in enumerate(cutting_planes):
                val = float(np.dot(g, a_cur)) - r
                scores.append((val, idx))
            scores.sort(reverse=True)
            keep = [s[1] for s in scores[:80]]
            cutting_planes = [cutting_planes[k] for k in keep]

        # Build LP
        # Variables: a[0],...,a[n-1], t
        # Minimize t
        # Subject to: sum(a) = 1, a >= 0, t >= 0
        #             grad_k @ a - t <= rhs_k for each cutting plane k

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t

        # Inequality constraints: grad_k @ a - t <= rhs_k
        A_ub = []
        b_ub = []
        for (g, r) in cutting_planes:
            row = np.concatenate([g, [-1.0]])
            A_ub.append(row)
            b_ub.append(r)

        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)

        # Equality: sum(a) = 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Bounds: a[i] >= 0, t >= 0
        bounds = [(0, None)] * n + [(0, None)]

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
            if result.success:
                a_new = result.x[:n]
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-10:
                    a_new = a_new / s_new
                    sc_new = _score_norm_fn(a_new)
                    c_new = _full_convolve(a_new, a_new)
                    max_new = float(c_new.max())

                    if np.isfinite(sc_new) and sc_new < best_score:
                        # Check if true max improved
                        if max_new < current_max * 1.01:
                            best_score = sc_new
                            best_a = a_new.copy()
                            current_max = max_new
                        else:
                            # Blend
                            for alpha in [0.3, 0.5, 0.7]:
                                a_blend = (1 - alpha) * best_a + alpha * a_new
                                a_blend = a_blend / a_blend.sum()
                                sc_blend = _score_norm_fn(a_blend)
                                if np.isfinite(sc_blend) and sc_blend < best_score:
                                    best_score = sc_blend
                                    best_a = a_blend.copy()
                                    c_check = _full_convolve(best_a, best_a)
                                    current_max = float(c_check.max())
                                    break
                    elif max_new < current_max:
                        # Even if score didn't improve numerically, the max reduced
                        # Try blending
                        for alpha in [0.5, 0.7, 0.9]:
                            a_blend = (1 - alpha) * best_a + alpha * a_new
                            a_blend = a_blend / a_blend.sum()
                            sc_blend = _score_norm_fn(a_blend)
                            if np.isfinite(sc_blend) and sc_blend < best_score:
                                best_score = sc_blend
                                best_a = a_blend.copy()
                                c_check = _full_convolve(best_a, best_a)
                                current_max = float(c_check.max())
                                break
        except Exception:
            pass

        # Convergence check
        if it > 10:
            c_check = _full_convolve(best_a, best_a)
            if float(c_check.max()) > max_cur * 0.999:
                # No improvement in last few iterations
                if it > 20:
                    break

    return best_a, best_score


def _peak_surgery(a_init, deadline, rng, max_rounds=300):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top peaks
        top_k = min(8, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices[:4]:
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:6]
            for idx_pos in order[:3]:
                idx = valid_i[idx_pos]
                for delta in [0.01, 0.02, 0.05, 0.1]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            small_idx = np.argsort(a_cur)[:5]
            for idx in small_idx:
                for delta in [0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rnd > 50:
            break

    return best_a, best_score


def _run_adam_fine(a_init, deadline, rng, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        if t % 500 == 0:
            beta = _adaptive_beta(c)

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 5000 == 0:
            lr = max(0.005, lr * 0.9)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        _update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(256, 1.5)
        _update_best(a0, _score_norm_fn(a0), "default_ramp")
        best_seq = a0.copy()
        best_score = _score_norm_fn(a0)

    # Phase 1: Multi-scale from best profile (40% of time)
    phase1_frac = 0.35
    phase1_end = t_start + (budget_s - 30.0) * phase1_frac
    print(f"\nPhase 1: Multi-scale optimization (budget {phase1_frac*100:.0f}%)")

    # Try multiple profiles at n=128
    n_start = 128
    profiles = _make_profiles(n_start)
    a_cur = None
    sc_cur = np.inf
    for prof in profiles:
        sc = _score_norm_fn(prof)
        if sc < sc_cur:
            sc_cur = sc
            a_cur = prof.copy()

    # Also try parent upsampled to 128
    if parent is not None:
        p_128 = _upsample(parent, n_start)
        sc_p128 = _score_norm_fn(p_128)
        if sc_p128 < sc_cur:
            sc_cur = sc_p128
            a_cur = p_128.copy()

    print(f"  Best initial profile at n={n_start}: score_norm={sc_cur:.10f}")

    # L-BFGS at n=128
    quick_dl = min(phase1_end, t_start + 15.0)
    a_cur, sc_cur = _run_lbfgs_continuation(a_cur, quick_dl, rng, n_betas=8)
    print(f"  After n=128: {sc_cur:.10f}")

    # Multi-scale up
    scales = [256, 512, 1024]
    time_alloc = {256: 30, 512: 50, 1024: 80}

    for target_n in scales:
        if time.monotonic() > phase1_end - 15:
            break

        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)
        print(f"  Upsampled to n={target_n}: {sc_target:.10f}")

        alloc = time_alloc.get(target_n, 60)
        opt_dl = min(phase1_end - 5, time.monotonic() + alloc)
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_dl, rng, n_betas=12)
        print(f"  After n={target_n}: {sc_opt:.10f}")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    _update_best(a_cur, sc_cur, "multi_scale_1024")

    # Also try from parent directly if parent is large
    if parent is not None and len(parent) >= 1024:
        print(f"  Parent has n={len(parent)}, using directly")
        _update_best(parent, _score_norm_fn(parent), "parent_direct")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Upsample to 4096, Nesterov GD (40% of time)
    phase2_frac = 0.35
    phase2_end = t_start + (budget_s - 30.0) * (phase1_frac + phase2_frac)
    print(f"\nPhase 2: Nesterov GD at n=4096 (budget {phase2_frac*100:.0f}%)")

    # Use best_seq for upsampling
    a_4096 = _upsample(best_seq, 4096)
    sc_4096 = _score_norm_fn(a_4096)
    print(f"  Upsampled to 4096: {sc_4096:.10f}")
    _update_best(a_4096, sc_4096, "upsample_4096")

    nest_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.7)
