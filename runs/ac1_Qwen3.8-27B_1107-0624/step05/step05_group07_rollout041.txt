def _fft_convolve(a, b):
    """Compute full convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

"""
Optimized search for minimal 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) L-BFGS with log-sum-exp continuation, (2) projected subgradient
on exact max, (3) multi-restart with diverse perturbations, (4) fine polishing.
Key innovation: exact subgradient method with FFT-based gradient computation
and adaptive backtracking step sizes.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-30:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(0.0, v - theta)


def _grad_of_weighted_conv(w, a):
    """Compute 2 * sum_k w_k * a_{k-j} for j = 0,...,n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    w_rev = w[::-1].copy()
    conv_result = _fft_convolve(a, w_rev)
    # grad_j = 2 * conv_result[2n-2-j] for j = 0,...,n-1
    # = 2 * conv_result[n-1 : 2n-1][::-1]
    return 2.0 * conv_result[n - 1:2 * n - 1][::-1]


def _lse_objective(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_gradient(z, beta, n):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    h = _grad_of_weighted_conv(w, a) / 2.0  # sum_k w_k * a_{k-j}
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgradient_step(a, tau=0.01, eta=0.05):
    """One projected subgradient step on the exact max.
    
    tau: tolerance for "near-active" set
    eta: step size
    """
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    # Near-active set
    active = c >= (M - tau * M)
    if active.sum() == 0:
        active = np.zeros(len(c), dtype=bool)
        active[np.argmax(c)] = True
    
    w = active.astype(np.float64) / active.sum()
    grad = _grad_of_weighted_conv(w, a)
    
    # Projected step
    v = a - eta * grad / (grad.max() + 1e-30)  # normalize gradient
    a_new = _project_simplex(v)
    return a_new


def _run_subgradient(a_init, deadline, eta_init=0.1, tau=0.01, max_iter=50000):
    """Run projected subgradient method on exact max."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(a)
    eta = eta_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        active = c >= (M - tau * M)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
        
        w = active.astype(np.float64) / active.sum()
        grad = _grad_of_weighted_conv(w, a)
        
        # Try multiple step sizes
        best_sc = _score(a)
        best_a_new = a.copy()

        sc = best_sc
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    c0 = _fft_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.005, 0.003,
                 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001]
        beta_list = [b * max_c0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        remaining = len(beta_list) - beta_list.index(beta)
        time_per = min(60.0, (deadline - time.monotonic()) / max(1, remaining))
        maxiter = max(100, int(time_per * 20))
        
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_gradient,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):

                s_new = _score(a_new)
                if np.isfinite(s_new) and s_new < best_score:

                    best_a = a_new.copy()
                    z = result.x.copy()

def _run_pegasus(a_init, deadline, rng, max_iter=20000):
    """Pegasos-style algorithm: averaged projected subgradient with Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    a = _softmax(z)
    best_a = a.copy()
    best_score = _score(a)
    
    # Nesterov momentum
    a_prev = a.copy()
    tau_mom = 1.0
    eta = 0.1
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t = it + 1
        tau_mom = (1 + np.sqrt(1 + 4 * tau_mom**2)) / 2
        gamma = (tau_mom - 1) / (tau_mom + 1)
        
        # Nesterov extrapolation
        z_trial = z + gamma * (z - z - z)  # This is just z for the first iteration
        # Actually, let me use the simpler form:
        a_trial = a + gamma * (a - a_prev)
        a_trial = _project_simplex(a_trial)
        
        # Compute subgradient at trial point
        c = _fft_convolve(a_trial, a_trial)
        M = float(c.max())
        active = c >= (M - 0.005 * M)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
        w = active.astype(np.float64) / active.sum()
        grad = _grad_of_weighted_conv(w, a_trial)
        
        # Adaptive step size
        eta = 2.0 / (t ** 0.7 + 10)
        
        # Update
        a_prev = a.copy()
        a = _project_simplex(a_trial - eta * grad)
        
        # Update log-parameters
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if it % 5000 == 0:
            print(f"    Pegasos iter {it}: score={sc:.10f}, best={best_score:.10f}, eta={eta:.6f}")
    
    return best_a, best_score


def _make_init(n, rng):
    """Create diverse initializations."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    
    inits = []
    # Uniform
    inits.append(np.ones(n))
    # Raised cosine
    inits.append(np.maximum(1.0 + np.cos(2 * np.pi * t), 0.01))
    # Half cosine (concentrated at edges)
    inits.append(np.maximum(np.cos(np.pi * (t - 0.5)), 0.01))
    # Gaussian centered
    inits.append(np.exp(-4 * (t - 0.5) ** 2))
    # Two Gaussians (concentrated at 1/3 and 2/3)
    inits.append(np.exp(-20 * (t - 1/3)**2) + np.exp(-20 * (t - 2/3)**2))
    # Power law
    inits.append(np.maximum(1.0 - np.abs(u), 0.01) ** 2)
    # Random smooth
    freqs = np.array([1, 2, 3, 5, 7])
    random_profile = np.ones(n)
    for f in freqs:
        random_profile += rng.random() * np.cos(2 * np.pi * f * t + rng.random() * 2 * np.pi)
    inits.append(np.maximum(random_profile, 0.01))
    
    # Normalize
    result = []
    for a in inits:
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            result.append(a / s)
    return result


def _adaptive_tau_subgradient(a_init, deadline, rng, max_iter=30000):

        best_eta = eta
        for mult in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
            eta_trial = eta * mult
            v = a - eta_trial * grad
            a_trial = _project_simplex(v)
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a_new = a_trial
                best_eta = eta_trial
        
        a = best_a_new
        eta = best_eta
        
        sc = best_sc
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            # Increase tau slightly on improvement (expand active set)
            tau = min(0.05, tau * 1.1)
        else:
            # No improvement: reduce step, shrink tau
            eta *= 0.7
            tau = max(0.001, tau * 0.9)
            
            # Occasionally add noise to escape local minima
            if it % 2000 == 0:
                noise = 0.01 * rng.normal(size=n)
                a = _project_simplex(a + noise)
        
        if it % 5000 == 0:
            print(f"    ADT-SG iter {it}: score={sc:.10f}, best={best_score:.10f}, "
                  f"eta={eta:.6f}, tau={tau:.4f}, |active|={active.sum()}")
    
    return best_a, best_score


def _equalize_active(a_init, deadline, max_iter=10000):
    """Try to equalize the active lags using a Lagrangian approach."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a, a)
        M = float(c.max())
        # Active set: lags within 1% of max
        active = c >= (M * 0.99)
        n_active = active.sum()
        if n_active < 2:
            break
        
        # Compute gradient as average of active gradients
        w = active.astype(np.float64) / n_active
        grad = _grad_of_weighted_conv(w, a)
        
        # Projected gradient descent with small step
        eta = 0.01 / (1 + it * 0.001)
        a_new = _project_simplex(a - eta * grad)
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            eta *= 0.5
            if eta < 1e-8:
                break
    
    return best_a, best_score
