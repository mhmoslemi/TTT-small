Nesterov-accelerated gradient optimization with adaptive log-sum-exp smoothing
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines fast gradient iterations, multi-restart exploration, and multi-n search.

import numpy as np
import time

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad(a, beta, c=None):
    """Compute gradient of log-sum-exp smoothed objective in log-space."""
    n = len(a)
    if c is None:
        c = _fft_convolve(a, a)
    M = c.max()
    shifted = (c - M) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    t = _fft_convolve(w, a_rev)[n-1:2*n-1]
    W = np.dot(w, c)
    grad = 4.0 * n * a * (t - W)
    return grad, w, c

def _nesterov_optimize(a_init, deadline, lr=0.05, momentum=0.9, 
                        beta_start=0.02, beta_end=1.0, check_interval=50):
    """Nesterov-accelerated gradient descent in log-space."""
    n = len(a_init)
    z = np.log(a_init + 1e-30)
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _fft_convolve(a_init, a_init).max()
    
    v = np.zeros(n)
    prev_z = z.copy()
    z = z.copy()
    
    total_steps = 0
    step = 0
    current_beta = beta_start * (best_score)  # scale beta to the problem
    
    while time.monotonic() < deadline:
        step += 1
        total_steps += 1
        
        a = _softmax(z)
        c = _fft_convolve(a, a)
        M = c.max()
        
        # Adaptive beta: increase over time
        progress = min(1.0, step / 50000)
        current_beta = (beta_start + (beta_end - beta_start) * progress) * M
        
        # Compute gradient
        grad, w, c = _compute_grad(a, current_beta, c)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Nesterov update
        z_look = z - momentum * v
        a_look = _softmax(z_look)
        grad_look, _, _ = _compute_grad(a_look, current_beta)
        
        v = momentum * v + lr * grad_look
        z = z - v
        
        # Normalize z (removes the constant offset, doesn't change a)
        if step % 1000 == 0:
            z = z - z.mean()
        
        # Periodic evaluation
        if step % check_interval == 0:
            a_check = _softmax(z)
            sc = _fft_convolve(a_check, a_check).max()
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        
        # Decay learning rate

            lr *= 0.9
            momentum = min(0.999, momentum + 0.005)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()

"""
Nesterov-accelerated gradient optimization with adaptive log-sum-exp smoothing,
mini-LP rescue, multi-restart, and multi-n search for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_convolve(a, b):
    """FFT-based full convolution."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_logspace(a, beta, c=None):
    """Compute gradient of beta-logsumexp(conv(a,a)) in log-space (z where a=softmax(z))."""
    n = len(a)
    if c is None:
        c = _fft_convolve(a, a)
    M = c.max()
    shifted = (c - M) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    t = _fft_convolve(w, a_rev)[n - 1:2 * n - 1]
    W = np.dot(w, c)
    grad = 4.0 * n * a * (t - W)
    return grad


def _nesterov_optimize(a_init, deadline, lr=0.05, momentum=0.9,
                       beta_start=0.02, beta_end=2.0, check_interval=30):
    """Nesterov-accelerated gradient descent in log-space with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _fft_convolve(a_init, a_init)
    best_score = float(c_init.max())

    v = np.zeros(n)
    step = 0
    stagnant_steps = 0
    prev_best = best_score

    while time.monotonic() < deadline:
        step += 1

        # Adaptive beta schedule
        progress = min(1.0, step / 80000.0)
        current_beta = (beta_start + (beta_end - beta_start) * progress) * best_score
        current_beta = max(current_beta, 1e-10)

        # Nesterov lookahead
        a = _softmax_fn(z)
        grad = _compute_grad_logspace(a, current_beta)

        if not np.all(np.isfinite(grad)):
            break

        # Momentum update
        v = momentum * v + lr * grad
        z = z - v

        # Periodic cleanup
        if step % 500 == 0:
            z = np.clip(z, -80, 80)
            z = z - z.mean()

        # Periodic evaluation
        if step % check_interval == 0:
            a_check = _softmax_fn(z)
            sc = float(_fft_convolve(a_check, a_check).max())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_check.copy()
                stagnant_steps = 0
            else:
                stagnant_steps += check_interval
            prev_best = best_score

            # Decay learning rate periodically

def _mini_lp_rescue(a, top_k=15, time_limit=15):
    """Solve a mini-LP using only the top K convolution constraints for a rescue direction."""
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())

    # Get indices of top K entries
    top_idx = np.argsort(c)[-top_k:][::-1]

    # Build constraint matrix: A[p, j] = a[top_idx[p] - j] if valid
    k_vals = top_idx.astype(np.int64)
    j_vals = np.arange(n, dtype=np.int64)
    diff = k_vals[:, None] - j_vals[None, :]
    mask = (diff >= 0) & (diff < n)
    A = np.zeros((top_k, n))
    A[mask] = a[diff[mask]]

    # Solve: max sum(g) s.t. A @ g <= M * ones(K), g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(top_k, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _interp_profile(p, n_new):
    """Interpolate a sequence to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-10)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    """Perturb a sequence in log-space."""
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_fn(z)

    x = np.arange(n) / n
    for freq, amp in [(1, 0.3), (2, 0.5), (3, 0.4), (5, 0.3)]:
        p = 1 + amp * np.cos(2 * np.pi * freq * x)
        p = np.maximum(p, 0.01)
        profiles.append(p / p.sum())
    # Triangular
    t = np.abs(x - 0.5) * 2
    p = 1 - t
    p = np.maximum(p, 0.01)
    profiles.append(p / p.sum())
    # Bimodal
    p = np.exp(-5 * (x - 0.25) ** 2) + np.exp(-5 * (x - 0.75) ** 2)
    profiles.append(p / p.sum())
    # Random
    p = np.abs(rng.standard_normal(n)) + 0.1
    profiles.append(p / p.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function combining Nesterov optimization, mini-LP rescue,
    multi-restart, Adam fine-tuning, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        print(f"Parent loaded: n={n_main}, score={_score(parent):.10f}")
        update_best(parent)
    else:
        n_main = 8192
        parent = np.ones(n_main) / n_main
        print(f"No parent, uniform n={n_main}, score={_score(parent):.10f}")
        update_best(parent)

    # === Phase 1: Main Nesterov optimization (0 - 650s) ===
    phase1_end = min(deadline, t_start + 650.0)
    print(f"\n=== Phase 1: Nesterov optimization (n={n_main}) ===")

    a_work = parent.copy()
    a_work_norm = a_work / a_work.sum()

    # First Nesterov run
    a1, s1 = _nesterov_optimize(a_work_norm, phase1_end, lr=0.04, momentum=0.9,
                                 beta_start=0.02, beta_end=1.5)
    update_best(a1)

    # If time remains, second run with different params
    if time.monotonic() < phase1_end - 90:
        a2, s2 = _nesterov_optimize(a1, phase1_end, lr=0.02, momentum=0.95,
                                     beta_start=0.08, beta_end=3.0, check_interval=20)
        update_best(a2)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Mini-LP rescue + Nesterov (650s - 850s) ===
    phase2_end = min(deadline, t_start + 850.0)
    print(f"\n=== Phase 2: Mini-LP rescue + refine ===")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_cur = best_seq / best_seq.sum()
        # Try mini-LP rescue
        g = _mini_lp_rescue(a_cur, top_k=20, time_limit=20)
        if g is not None:
            g_norm = g / g.sum()
            # Blend and optimize
            for t_val in [0.3, 0.5, 0.7]:
                a_blend = (1 - t_val) * a_cur + t_val * g_norm
                a_blend = np.maximum(a_blend, 0)
                a_blend = a_blend / a_blend.sum()
                update_best(a_blend)
                # Quick Nesterov from this point
                if time.monotonic() < phase2_end - 40:
                    a_r, _ = _nesterov_optimize(a_blend, time.monotonic() + 40,
                                                 lr=0.02, momentum=0.9,
                                                 beta_start=0.05, beta_end=1.0,
                                                 check_interval=15)
                    update_best(a_r)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam fine-tuning (850s - 1100s) ===
    phase3_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 3: Adam fine-tuning ===")

    if best_seq is not None:
        a_best_norm = best_seq / best_seq.sum()
        remaining = phase3_end - time.monotonic() - 20
        if remaining > 60:
            a3, s3 = _adam_optimize(a_best_norm, time.monotonic() + remaining, lr=0.006)
            update_best(a3)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse profiles (1100s - 1400s) ===
    phase4_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 4: Multi-restart ===")

    if best_seq is not None:
        a_best_norm = best_seq / best_seq.sum()
        n = len(a_best_norm)

        # Generate diverse starting points
        starts = []
        # Perturbations of best
        for sigma in [0.03, 0.1, 0.25]:
            starts.append(_perturb(a_best_norm, sigma, rng))
        # Some structured profiles interpolated to current n
        profiles = _make_profiles(n, rng)
        for p in profiles[:3]:
            starts.append(p)

        per_start = max(30.0, (phase4_end - time.monotonic() - 20) / max(1, len(starts)))

        for i, s_start in enumerate(starts):
            if time.monotonic() > phase4_end - 25:
                break
            dl_i = time.monotonic() + min(per_start, phase4_end - time.monotonic() - 15)
            if dl_i - time.monotonic() > 25:
                a_r, sc_r = _nesterov_optimize(s_start, dl_i, lr=0.03, momentum=0.9,
                                                beta_start=0.03, beta_end=1.0,
                                                check_interval=20)
                update_best(a_r)
                # Quick Adam from result
                if time.monotonic() < phase4_end - 30:
                    a_ad, _ = _adam_optimize(a_r, time.monotonic() + 30, lr=0.004)
                    update_best(a_ad)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Larger n exploration (1400s - 1600s) ===
    phase5_end = min(deadline, t_start + 1600.0)
    print(f"\n=== Phase 5: Larger n exploration ===")

    if best_seq is not None and time.monotonic() < phase5_end - 60:
        current_n = len(best_seq)
        target_ns = [16384] if current_n < 16384 else [4096]
        for target_n in target_ns:
            if time.monotonic() > phase5_end - 80:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)
            remaining = phase5_end - time.monotonic() - 40
            if remaining > 100:
                a_r, _ = _nesterov_optimize(p_interp, time.monotonic() + remaining,
                                             lr=0.03, momentum=0.9,
                                             beta_start=0.03, beta_end=1.0,
                                             check_interval=25)
                update_best(a_r)
                if time.monotonic() < phase5_end - 50:
                    a_ad, _ = _adam_optimize(a_r, time.monotonic() + 50, lr=0.004)
                    update_best(a_ad)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final Adam polish (1600s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 15
        if remaining > 40:
            a_final, _ = _adam_optimize(a_best_norm, time.monotonic() + remaining,
                                         lr=0.002,
                                         beta_scales=[0.01, 0.005, 0.002, 0.001, 0.0005])
            update_best(a_final)

    print(f"Phase 6 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check failed: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
