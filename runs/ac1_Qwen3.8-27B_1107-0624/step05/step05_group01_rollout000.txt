"""
Optimized step function search minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LP peak equalization, top-K log-sum-exp continuation,
mirror descent polish, and adaptive multi-phase strategy.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _hann_init(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _raised_cosine_init(n, beta_param=2.0):
    t = np.linspace(0, 1, n)
    a = np.cos(np.pi * t * beta_param / 2.0) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _triangular_init(n):
    t = np.linspace(0, 1, n)
    a = np.abs(1 - 2 * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _asym_power_init(n, p=1.5):
    t = np.linspace(0, 1, n)
    a = (0.01 + 0.99 * t) ** p
    return a / a.sum()


def _topk_softmin_obj(z, beta, n, K):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    peak_idx = np.argsort(c)[::-1][:K]
    peaks = c[peak_idx]
    pm = float(peaks.max())
    if pm < 1e-300:
        return 0.0
    sh = (peaks - pm) / beta
    lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
    return float(2.0 * n * lse)


def _topk_softmin_grad(z, beta, n, K):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    peak_idx = np.argsort(c)[::-1][:K]
    peaks = c[peak_idx]
    pm = float(peaks.max())
    if pm < 1e-300:
        return np.zeros_like(z)
    sh = (peaks - pm) / beta
    e = np.exp(sh - sh.max())
    w = e / e.sum()

    grad_a = np.zeros(n)
    for idx_k in range(len(peak_idx)):
        k = peak_idx[idx_k]
        wi = w[idx_k]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            seg = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
            grad_a[i_lo:i_hi + 1] += wi * 2.0 * seg

    Wa = float(np.dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _full_softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _full_softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs_full(a_init, deadline, beta_list=None, maxiter_per_beta=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002,
                 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _full_softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_full_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxls': 80, 'maxfun': 15000,
                         'max_correction': 10}

            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)

                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score

def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat

        if len(active) > 300:
            active = active[:300]

        A_mat = _build_lp_matrix(best_a, active, n)
        b_vec = np.full(len(active), M)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20, 'presolve': True})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [0.3, 0.5, 0.7, 1.0, 0.2, 0.4, 0.6, 0.8]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-14:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                staggered = stagnant
                break

        if not improved:
            stagnant += 1
            if stagnant > 10:
                break
        else:
            stagnant = 0

    return best_a, best_score


def _mirror_descent_polish(a_init, deadline, lr_init=0.5, max_iter=100000, K=5):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = lr_init
    no_improve = 0
    check_every = 20

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        peak_idx = np.argsort(c)[::-1][:K]
        peaks = c[peak_idx]
        pm = float(peaks.max())

        grad_a = np.zeros(n)
        for k in peak_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_a[i_lo:i_hi + 1] += 2.0 * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        # Center the gradient (remove component along all-ones to preserve sum)
        grad_a = grad_a - grad_a.mean()

        # Multiplicative update (mirror descent with entropy)
        a_new = a * np.exp(-lr * grad_a / (pm + 1e-30))
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        a = a_new
        lr *= 0.9995

        if it % check_every == 0:
            sc = _score_norm_fn(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 50:
                    # Restart with perturbation
                    a = best_a.copy()
                    noise = np.random.default_rng(it).normal(0, 0.01, n)
                    a = np.maximum(a + noise, 1e-15)
                    a = a / a.sum()
                    lr = lr_init * 0.5
                    no_improve = 0

    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_iter=50000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    lr = 0.002
    no_improve = 0
    check_every = 30

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        pm = float(c.max())
        peak_mask = c >= pm - 1e-12 * pm
        peak_idx = np.where(peak_mask)[0]

        grad_a = np.zeros(n)
        for k in peak_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_a[i_lo:i_hi + 1] += 2.0 * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        grad_a = grad_a / len(peak_idx)
        grad_a = grad_a - grad_a.mean()

        a = np.maximum(a - lr * grad_a, 1e-15)
        a = a / a.sum()
        lr *= 0.999

        if it % check_every == 0:
            sc = _score_norm_fn(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 100:
                    break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS continuation with full + top-K smoothing,
    aggressive LP peak equalization, mirror descent and subgradient polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: L-BFGS full continuation (450s) ===
    print("Phase 1: L-BFGS full continuation")
    p1_end = min(deadline, t_start + 450.0)

    a_s = best_seq.copy()
    a_b1, sc1 = _run_lbfgs_full(a_s, p1_end, maxiter_per_beta=4000)
    update_best(a_b1, sc1, "lbfgs_full_main")

    # Try from greedy init
    if time.monotonic() < p1_end - 60:
        a_g = _greedy_init(n)
        sc_g = _score_norm_fn(a_g)
        update_best(a_g, sc_g, "greedy")
        a_gb, sc_gb = _run_lbfgs_full(a_g, min(p1_end - 10, time.monotonic() + 120))
        update_best(a_gb, sc_gb, "lbfgs_greedy")

    # Try other inits
    if time.monotonic() < p1_end - 30:
        for name, init_fn in [("hann", _hann_init), ("cos2", lambda x: _raised_cosine_init(x, 2.0)),
                               ("asp1.5", lambda x: _asym_power_init(x, 1.5))]:
            if time.monotonic() > p1_end - 30:
                break
            a_p = init_fn(n)
            sc_p = _score_norm_fn(a_p)
            if sc_p < best_score * 1.05:
                a_pb, sc_pb = _run_lbfgs_full(a_p, min(p1_end - 10, time.monotonic() + 60))
                update_best(a_pb, sc_pb, f"lbfgs_{name}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Top-K L-BFGS refinement (350s) ===
    print("Phase 2: Top-K L-BFGS")
    p2_end = min(deadline, t_start + 800.0)

    if time.monotonic() < p2_end and best_seq is not None:
        for K in [3, 5, 7]:
            if time.monotonic() > p2_end - 40:
                break
            a_tk, sc_tk = _run_lbfgs_topk(best_seq, min(p2_end - 20, time.monotonic() + 100),
                                           K=K, maxiter_per_beta=5000)
            update_best(a_tk, sc_tk, f"topK_{K}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Aggressive LP peak equalization (450s) ===
    print("Phase 3: LP peak equalization")
    p3_end = min(deadline, t_start + 1250.0)

    if time.monotonic() < p3_end and best_seq is not None:
        a_lp, sc_lp = _lp_aggressive(best_seq, min(p3_end - 60, time.monotonic() + 200),
                                      K=50, max_rounds=40)
        update_best(a_lp, sc_lp, "lp_K50")

        # Second round with different K
        if time.monotonic() < p3_end - 40:
            a_lp2, sc_lp2 = _lp_aggressive(best_seq, p3_end, K=80, max_rounds=30)
            update_best(a_lp2, sc_lp2, "lp_K80")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Fine L-BFGS after LP (200s) ===
    print("Phase 4: Fine L-BFGS after LP")
    p4_end = min(deadline, t_start + 1450.0)

    if time.monotonic() < p4_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                                         0.00002, 0.00001, 0.000005]]
        a_f, sc_f = _run_lbfgs_full(best_seq, p4_end, beta_list=tiny_betas,
                                     maxiter_per_beta=8000)
        update_best(a_f, sc_f, "final_tiny_beta")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Mirror descent polish (150s) ===
    print("Phase 5: Mirror descent polish")
    p5_end = min(deadline, t_start + 1600.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_md, sc_md = _mirror_descent_polish(best_seq, p5_end, lr_init=0.3, K=5)
        update_best(a_md, sc_md, "mirror_K5")

        if time.monotonic() < p5_end - 30:
            a_md2, sc_md2 = _mirror_descent_polish(best_seq, p5_end, lr_init=0.1, K=7)
            update_best(a_md2, sc_md2, "mirror_K7_slow")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Subgradient + final LP (100s) ===
    print("Phase 6: Subgradient + final LP")
    p6_end = min(deadline, t_start + 1700.0)

    if time.monotonic() < p6_end and best_seq is not None:
        a_sg, sc_sg = _subgradient_polish(best_seq, min(p6_end - 20, time.monotonic() + 50))
        update_best(a_sg, sc_sg, "subgrad")

        # Final LP round
        if time.monotonic() < p6_end - 20:
            a_lp3, sc_lp3 = _lp_aggressive(best_seq, p6_end, K=60, max_rounds=20)
            update_best(a_lp3, sc_lp3, "final_lp")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Try different n if time allows ===
    if time.monotonic() < deadline - 80 and best_seq is not None:
        n_cur = len(best_seq)
        for n_try in [n_cur * 2, n_cur // 2]:
            if n_try < 1024 or n_try > 16384:
                continue
            if time.monotonic() > deadline - 60:
                break
            print(f"Phase 7: Trying n={n_try}")
            a_interp = _interp_profile(best_seq, n_try)
            sc_in = _score_norm_fn(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_try}")
            if time.monotonic() < deadline - 40:
                a_in_b, sc_in_b = _run_lbfgs_full(a_interp, min(deadline - 20, time.monotonic() + 40),
                                                    maxiter_per_beta=2000)
                update_best(a_in_b, sc_in_b, f"lbfgs_n{n_try}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, norm_score={_score_norm_fn(final_seq):.10f}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
