"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses FFT-based convolution, L-BFGS log-sum-exp continuation, active-set peak balancing,
Nesterov Adam polish, and multi-resolution (n-doubling) strategy.
Key improvements over prior: finer beta schedule, Nesterov momentum, peak-specific
gradient focusing, and adaptive n selection.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_fn(vals, beta):
    """Log-sum-exp of vals with temperature beta."""
    vals = np.asarray(vals, dtype=np.float64)
    if len(vals) == 0:
        return 0.0
    vmax = float(vals.max())
    if vmax < 1e-300:
        return 0.0
    shifted = (vals - vmax) / beta
    e = np.exp(shifted - shifted.max())
    return vmax + beta * (float(shifted.max()) + float(np.log(e.sum())))


def _nesterov_adam_polish(a_init, deadline, lr=0.015, max_iter=500000):
    """Nesterov-accelerated Adam with adaptive beta log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.005 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30
    momentum = 0.9
    z_mom = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Adaptive beta: decay over time, floor at tiny value
        beta = max(1e-14 * cm, cm0 * 0.005 * (0.96 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        # Nesterov look-ahead
        z_mom = momentum * z_mom + (1 - momentum) * grad
        z_look = z - lr * z_mom
        a_look = _softmax_fn(z_look)
        c_look = _full_convolve(a_look, a_look)
        cm_look = float(c_look.max())
        beta_look = max(1e-14 * cm_look, cm0 * 0.005 * (0.96 ** (t // 30)))
        shifted_l = (c_look - cm_look) / beta_look
        max_sl = float(shifted_l.max())
        e_l = np.exp(shifted_l - max_sl)
        w_l = e_l / e_l.sum()
        W_l = float(np.dot(w_l, c_look))
        a_rev_l = a_look[::-1]
        w_conv_l = _full_convolve(w_l, a_rev_l)
        h_l = w_conv_l[n - 1:2 * n - 1]
        grad_look = 4.0 * n * a_look * (h_l - W_l)

        if not np.all(np.isfinite(grad_look)):
            grad_look = grad

        m = b1 * m + (1 - b1) * grad_look
        v = b2 * v + (1 - b2) * (grad_look * grad_look)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                # Perturbation with adaptive scale
                scale = 0.05 * cm0 / max(cm, 1e-15)
                z = z + 0.05 * np.random.randn(n) * min(scale, 1.0)

                no_improve = 0

def _run_lbfgs_refined(a_init, deadline, beta_list=None, maxiter_per_beta=6000):
    """L-BFGS with log-sum-exp continuation, refined gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015,
                 0.01, 0.007, 0.005, 0.003, 0.002, 0.001, 0.0007,
                 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def obj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)

        def jac(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return np.zeros_like(z_in)
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            e = np.exp(shifted - max_s)
            w = e / e.sum()
            W = float(np.dot(w, c_in))
            a_rev = a_in[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_in * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_in)
            return g

        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=jac,
                              options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 80, 'maxfun': 30000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _active_set_refine(a_init, deadline, K=5, max_rounds=50):
    """Direct active-set: minimize max of top-K peaks via projected gradient on softmax params."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for outer in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]
        peaks = c[peak_indices]
        pm = float(peaks.max())

        # Gradient of max of top-K peaks (smoothed with tiny beta)
        beta = 1e-8 * pm
        sh = (peaks - pm) / beta
        e = np.exp(sh - sh.max())
        w = e / e.sum()

        # Compute gradient w.r.t. a
        grad_a = np.zeros(n)
        for idx_k in range(len(peak_indices)):
            k = int(peak_indices[idx_k])
            wi = float(w[idx_k])
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_cur[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        Wa = float(np.dot(grad_a, a_cur))
        grad_z = 2.0 * n * a_cur * (grad_a - Wa)

        if not np.all(np.isfinite(grad_z)):
            break

        # Line search
        best_step_score = best_score
        best_step_z = z.copy()
        for step_mult in [1.0, 0.5, 0.3, 0.15, 0.07, 0.03]:
            lr = step_mult * pm / (np.max(np.abs(grad_z)) + 1e-30)
            lr = min(lr, 0.5)
            z_trial = z - lr * grad_z
            z_trial = np.clip(z_trial, -80, 80)
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax_fn(z_trial)
            sc_trial = _score_norm_fn(a_trial)
            if np.isfinite(sc_trial) and sc_trial < best_step_score:
                best_step_score = sc_trial
                best_step_z = z_trial.copy()

        if best_step_score < best_score - 1e-14:
            best_score = best_step_score
            best_a = _softmax_fn(best_step_z).copy()
            z = best_step_z.copy()
        else:
            # Try small random perturbation
            z = z + 0.02 * np.random.randn(n)
            z = z - z.mean()
            a_trial = _softmax_fn(z)
            sc_trial = _score_norm_fn(a_trial)
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()
            else:
                z = best_step_z.copy()
                if outer > 20:
                    break

    return best_a, best_score


def _peak_reduction(a_init, deadline, max_rounds=20000):
    """Greedy peak reduction: identify max peak, reduce contributing elements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    rng = np.random.default_rng(123)

    for it in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        k_max = int(np.argmax(c))
        current_score = 2.0 * n * cm

        # Find contributors to peak at k_max
        i_lo = max(0, k_max - n + 1)
        i_hi = min(n - 1, k_max)
        if i_lo > i_hi:
            break

        # Products at the peak
        j_range = range(i_lo, i_hi + 1)
        products = np.array([a[j] * a[k_max - j] for j in j_range])
        j_indices = np.array(list(j_range))

        # Find the largest contributor
        max_pidx = int(np.argmax(products))
        j_star = int(j_indices[max_pidx])
        partner = k_max - j_star

        # Try reducing a[j_star] or a[partner] and redistributing
        # Reduce the larger of the two
        if a[j_star] >= a[partner]:
            reduce_idx = j_star
        else:
            reduce_idx = partner

        if reduce_idx < 0 or reduce_idx >= n:
            continue

        # Try different reduction amounts
        improved = False
        for frac in [0.01, 0.005, 0.002, 0.001]:
            delta = frac * a[reduce_idx]
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[reduce_idx] -= delta
            # Redistribute to a random position that doesn't contribute to the peak
            # Choose position far from k_max
            candidates = np.arange(n)
            # Exclude positions that contribute to top few peaks
            top_peaks = np.argsort(c)[::-1][:10]
            excluded = set()
            for kp in top_peaks:
                for jj in range(max(0, int(kp) - n + 1), min(n - 1, int(kp)) + 1):
                    excluded.add(jj)
            valid = [i for i in candidates if i not in excluded and a[i] < a[reduce_idx]]
            if not valid:
                valid = [i for i in candidates if i != reduce_idx]
            if not valid:
                continue
            # Add to the position with smallest value among valid
            target = int(valid[np.argmin([a[i] for i in valid])])
            a_trial[target] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = 2.0 * n * float(c_trial.max())
            if sc_trial < current_score - 1e-15:
                a = a_trial.copy()
                current_score = sc_trial
                improved = True
                break

        if improved:
            sc = _score_norm_fn(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
        else:
            # Stagnation: try random restart in local neighborhood
            if it % 100 == 0:
                a_pert = a + 0.001 * rng.standard_normal(n)
                a_pert = np.maximum(a_pert, 0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert /= s
                    sc_p = _score_norm_fn(a_pert)
                    if sc_p < current_score:
                        a = a_pert.copy()
                        current_score = sc_p
                    if current_score < best_score:
                        best_score = current_score
                        best_a = a.copy()

    return best_a, best_score


def _greedy_init(n):
    """Binomial-based initialization (central binomial coefficients reversed)."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _powerlaw_init(n, alpha=0.5):
    """Power-law initialization: a[j] ~ (j+1)^alpha."""
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** alpha
    a = a / a.sum()
    return a


def _exp_init(n, rate=1.0):
    """Exponential initialization: a[j] ~ exp(rate * j / n)."""
    t = np.arange(n, dtype=np.float64) / n
    a = np.exp(rate * t)
    a = a / a.sum()
    return a


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: try multiple initializations at n=8192,
    L-BFGS continuation, active-set refinement, Nesterov Adam polish,
    peak reduction, and n-doubling. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Try alternative initializations (60s) ===
    print("Phase 1: Alternative initializations")
    p1_end = min(deadline, t_start + 60.0)

    inits = {}
    if n <= 16384:
        inits['greedy'] = _greedy_init(n)
        inits['powerlaw_0.3'] = _powerlaw_init(n, 0.3)
        inits['powerlaw_0.5'] = _powerlaw_init(n, 0.5)
        inits['powerlaw_0.7'] = _powerlaw_init(n, 0.7)
        inits['exp_1'] = _exp_init(n, 1.0)
        inits['exp_2'] = _exp_init(n, 2.0)
        inits['exp_3'] = _exp_init(n, 3.0)

    for name, init in inits.items():
        sc = _score_norm_fn(init)
        if sc < best_score * 1.01:
            update_best(init, sc, f"init_{name}")

    # Quick L-BFGS on best non-parent init
    if time.monotonic() < p1_end - 5 and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        quick_betas = [m_b * b for b in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]]
        a_q, sc_q = _run_lbfgs_refined(best_seq, p1_end, beta_list=quick_betas, maxiter_per_beta=3000)
        update_best(a_q, sc_q, "quick_lbfgs_p1")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Main L-BFGS continuation (700s) ===
    print("Phase 2: Main L-BFGS continuation")
    p2_end = min(deadline, t_start + 760.0)

    a_main, sc_main = _run_lbfgs_refined(best_seq, p2_end, maxiter_per_beta=8000)
    update_best(a_main, sc_main, "lbfgs_main")

    # Second pass with even finer betas
    if time.monotonic() < p2_end - 30:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        fine_betas = [m_b * b for b in [0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00001, 0.000005]]
        a_fine, sc_fine = _run_lbfgs_refined(best_seq, p2_end, beta_list=fine_betas, maxiter_per_beta=10000)
        update_best(a_fine, sc_fine, "lbfgs_fine")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Active-set refinement (200s) ===
    print("Phase 3: Active-set refinement")
    p3_end = min(deadline, t_start + 960.0)

    for K in [3, 5, 7]:
        if time.monotonic() > p3_end - 20:
            break
        a_act, sc_act = _active_set_refine(best_seq, min(p3_end, time.monotonic() + 60), K=K, max_rounds=100)
        update_best(a_act, sc_act, f"active_K{K}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Nesterov Adam polish (300s) ===
    print("Phase 4: Nesterov Adam polish")
    p4_end = min(deadline, t_start + 1260.0)

    a_ad1, sc_ad1 = _nesterov_adam_polish(best_seq, min(p4_end, time.monotonic() + 150), lr=0.015)
    update_best(a_ad1, sc_ad1, "nesterov_1")

    if time.monotonic() < p4_end - 10:
        a_ad2, sc_ad2 = _nesterov_adam_polish(best_seq, p4_end, lr=0.008)
        update_best(a_ad2, sc_ad2, "nesterov_2")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Peak reduction (150s) ===
    print("Phase 5: Peak reduction")
    p5_end = min(deadline, t_start + 1410.0)

    a_pr, sc_pr = _peak_reduction(best_seq, p5_end, max_rounds=50000)
    update_best(a_pr, sc_pr, "peak_reduction")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final L-BFGS + Adam combo (200s) ===
    print("Phase 6: Final polish")
    p6_end = min(deadline, t_start + 1610.0)

    if time.monotonic() < p6_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        ultra_betas = [m_b * b for b in [0.00002, 0.00001, 0.000005, 0.000003, 0.000001]]
        a_ultra, sc_ultra = _run_lbfgs_refined(best_seq, min(p6_end, time.monotonic() + 100),
                                               beta_list=ultra_betas, maxiter_per_beta=12000)
        update_best(a_ultra, sc_ultra, "ultra_lbfgs")

    if time.monotonic() < p6_end - 10:
        a_final_ad, sc_final_ad = _nesterov_adam_polish(best_seq, p6_end, lr=0.005)
        update_best(a_final_ad, sc_final_ad, "final_adam")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Try n*2 if time allows ===
    if time.monotonic() < deadline - 120 and best_seq is not None:
        n_old = len(best_seq)
        n_new = min(2 * n_old, 16384)
        if n_new > n_old:
            print(f"Phase 7: Interpolating to n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm_fn(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")

            if time.monotonic() < deadline - 30:
                c_in = _full_convolve(a_interp, a_interp)
                m_in = float(c_in.max())
                interp_betas = [m_in * b for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]
                a_in_b, sc_in_b = _run_lbfgs_refined(a_interp, deadline - 10,
                                                     beta_list=interp_betas, maxiter_per_beta=5000)
                update_best(a_in_b, sc_in_b, f"lbfgs_n{n_new}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
