**LP-based direction finding:**
Given current a with max convolution M:
- We want to find a direction d that, when added to a (with small step t), decreases the score
- The LP: maximize sum(d) subject to:
  - d >= -a (so a + t*d >= 0 for small t > 0)
  - For each active peak k: (a + t*d) convolved with (a + t*d) at position k <= M + epsilon
  - For small t, this is approximately: sum_j d[j]*a[k-j] + a[j]*d[k-j] <= epsilon (linearization)

Actually, the exact LP from the description is:
- Maximize sum(b) subject to b >= 0 and ||f * b||_inf <= M

where f is the current sequence and M is its max convolution.

This finds a new "target" sequence b that has total mass as large as possible while keeping the cross-convolution with f bounded by M. Then we interpolate: a_new = (1-t)*a + t*b_normalized.

The intuition: if b has a large sum but its cross-convolution with f is bounded, then moving toward b should reduce the max of f*f (because the "new" parts of the sequence don't create large peaks when convolved with the "old" parts).

def _lp_direction(a, M, max_constraints=100):
    """Solve LP to find direction: max sum(b) s.t. b>=0, (f*b)[k] <= M for all k."""
    n = len(a)
    conv_len = 2 * n - 1
    
    # Variables: b[0], ..., b[n-1] (n variables)
    # Objective: maximize sum(b) => minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints:
    # 1. b >= 0 (handled by bounds)
    # 2. (a * b)[k] <= M for k = 0, ..., 2n-2
    #    (a * b)[k] = sum_j a[j] * b[k-j]
    #    This is a linear function of b.
    
    # Build constraint matrix A such that A @ b <= M * ones(conv_len)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[idx]
    
    # For large n, this matrix is huge. We need to select the most binding constraints.
    # The binding constraints are those where (a*a)[k] is close to M.
    c_aa = _full_convolve(a, a)
    # Select top constraints
    # Actually, we only need constraints that could be binding.
    # (a*b)[k] <= (a*a)[k] * (sum(b)/sum(a)) by some inequality? Not exactly.
    # Let's just use all constraints if n is small enough, or select the top ones.
    
    if n <= 512:
        # Use all constraints
        A_ub = A
        b_ub = np.full(conv_len, M)
    else:
        # Select top constraints (where a*a is largest)
        top_k = min(conv_len, 200)

    if result.success:
        b = result.x
        return b
    else:
        return None

"""
Hybrid optimizer: L-BFGS + LP refinement + projected subgradient with momentum.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations vs previous: (1) LP-based direction finding, (2) momentum subgradient
in a-space, (3) diverse initializations with structural priors.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _cross_conv(a, b):
    """Compute conv(a, b) where a has length n and b has length m. Returns n+m-1."""
    return _full_convolve(a, b)

def _subgrad_score(a, tol_frac=0.01):
    """Compute a subgradient of the score function.
    Returns (gradient, score, c_max, active_mask)."""
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)
    
    # Active set
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    
    # Compute d c_max / d a[i] (subgradient of max conv)
    # g[i] = (1/|A|) * sum_{k in A} (a[k-i] + a[i-k]) [with out-of-range = 0]
    n_a = int(active.sum())
    g = np.zeros(n)
    active_indices = np.where(active)[0]
    
    for k in active_indices:
        # First term: a[k-i] for valid i
        i_lo1 = max(0, k - n + 1)
        i_hi1 = min(n - 1, k)
        if i_lo1 <= i_hi1:
            # a[k-i] for i in [i_lo1, i_hi1]
            # k-i goes from k-i_lo1 down to k-i_hi1
            g[i_lo1:i_hi1+1] += a[k-i_hi1:k-i_lo1+1][::-1]
        
        # Second term: a[i-k] for valid i
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            # a[i-k] for i in [i_lo2, i_hi2]
            g[i_lo2:i_hi2+1] += a[i_lo2-k:i_hi2-k+1]
    
    g /= n_a
    
    # Gradient of score = 2n/S^3 * (S*g - 2*c_max)
    grad = 2.0 * n / (S**3) * (S * g - 2.0 * c_max)
    
    return grad, score, c_max, active

def _lp_refine(a, deadline, n_lp_solves=10):
    """LP-based refinement: find directions via LP and interpolate."""
    n = len(a)
    a = a.copy()
    best_a = a.copy()
    best_score = _score(a)
    
    for it in range(n_lp_solves):
        if time.monotonic() > deadline:
            break
        
        c_aa = _full_convolve(a, a)
        M = float(c_aa.max())
        S = float(a.sum())
        
        # Select top constraints
        conv_len = 2 * n - 1
        K = min(conv_len, 100)
        top_idx = np.argsort(-c_aa)[:K]
        
        # Build constraint matrix for selected constraints
        # A_ub[k, j] = f[k-j] if 0 <= k-j < n
        A_ub = np.zeros((K, n))
        for i, k in enumerate(top_idx):
            # (f*g)[k] = sum_j f[j]*g[k-j]
            # For j in [0, n-1], k-j is in [k-n+1, k]
            # Valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                A_ub[i, j_lo:j_hi+1] = a[k-j_hi:k-j_lo+1][::-1]
        
        # LP: maximize sum(g) subject to A_ub @ g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0, None)] * n
        
        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            if result.success:
                g = result.x
                S_g = float(g.sum())
                if S_g > 1e-10:
                    # Rescale to match sum
                    g = g / S_g * S
                    # Interpolate
                    t = 0.5  # large step
                    a_new = (1 - t) * a + t * g
                    a_new = np.maximum(a_new, 0)

"""
Hybrid optimizer combining L-BFGS, LP-based direction finding, and momentum
projected subgradient for minimizing 2*n*max(conv(a,a))/sum(a)^2 over
non-negative sequences. Key innovations: (1) LP-based refinement using the
state-of-the-art approach, (2) momentum subgradient in a-space (allowing
exact zeros), (3) diverse structural initializations, (4) multi-length search.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgrad_score(a, tol_frac=0.005):
    """Compute a subgradient of the score function 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    c_max = float(c.max())
    score = 2.0 * n * c_max / (S * S)

    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    n_act = int(active.sum())
    if n_act == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_act = 1

    g = np.zeros(n)
    active_indices = np.where(active)[0]
    for k in active_indices:
        i_lo1 = max(0, k - n + 1)
        i_hi1 = min(n - 1, k)
        if i_lo1 <= i_hi1:
            g[i_lo1:i_hi1 + 1] += a[k - i_hi1:k - i_lo1 + 1][::-1]
        i_lo2 = max(0, k)
        i_hi2 = min(n - 1, k + n - 1)
        if i_lo2 <= i_hi2:
            g[i_lo2:i_hi2 + 1] += a[i_lo2 - k:i_hi2 - k + 1]
    g /= n_act

    grad = 2.0 * n / (S ** 3) * (S * g - 2.0 * c_max)
    return grad, score, c_max, active


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_phase(a_init, deadline, label=""):
    """Run L-BFGS with beta continuation. Returns optimized a."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15 or time.monotonic() > deadline - 30:
        return a_init

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008])
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 15:
            break
        remaining = deadline - 15 - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.004 * (n / 2048.0)
        max_iter = min(4000, max(50, int(remaining / (est_time * 10))))

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        try:
            result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                              options={'maxiter': max_iter, 'ftol': 1e-14,
                                       'gtol': 1e-10, 'maxcor': 40, 'maxls': 80})
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-12:
                current_a = a_new
                current_sc = sc_new
                if label and i % 3 == 0:
                    print(f"    {label} beta_phase {i}: score={sc_new:.10f}")
        except Exception:
            pass

    return current_a


def _lp_refine(a, deadline, max_iters=20):
    """LP-based refinement: solve LP to find good directions, interpolate."""
    n = len(a)
    a_work = a.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c_aa = _full_convolve(a_work, a_work)
        M = float(c_aa.max())
        S = float(a_work.sum())
        conv_len = 2 * n - 1
        K = min(conv_len, 150)
        top_idx = np.argsort(-c_aa)[:K]

        A_ub = np.zeros((K, n))
        for i in range(K):
            k = top_idx[i]
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                A_ub[i, j_lo:j_hi + 1] = a_work[k - j_hi:k - j_lo + 1][::-1]

        c_obj = -np.ones(n)
        b_ub = np.full(K, M)
        bounds = [(0, None)] * n

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                             method='highs', options={'presolve': True})
            if result.success and result.x is not None:
                g = result.x
                S_g = float(g.sum())
                if S_g > 1e-10:
                    g_norm = g / S_g * S
                    for t_val in [0.7, 0.4, 0.2, 0.1]:
                        a_try = (1.0 - t_val) * a_work + t_val * g_norm
                        a_try = np.maximum(a_try, 0.0)
                        sc_try = _score(a_try)
                        if sc_try < best_sc:
                            best_sc = sc_try
                            best_a = a_try.copy()
                            a_work = a_try.copy()
                            break
            else:
                break
        except Exception:
            break

    return best_a, best_sc


def _momentum_subgrad(a_init, deadline, n_steps=3000, lr_init=0.02, momentum=0.85):
    """Projected subgradient with heavy-ball momentum in a-space."""
    n = len(a_init)
    a = a_init.copy()
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        grad, sc_cur, c_max, active = _subgrad_score(a)
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)

        if i % 20 == 0:
            sc_new = _score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                a = a_new
            else:
                lr *= 0.95
                v *= 0.5
        else:
            a = a_new

        if i % 200 == 0:
            sc_check = _score(a)
            if i % 400 == 0:
                print(f"    Momentum iter {i}: score={sc_check:.10f}, lr={lr:.6f}")
            if np.linalg.norm(v) < 1e-12:

    a_final = np.maximum(a, 0.0)
    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()

    return best_a, best_sc


def _make_tilted(n, alpha=1.0, kind="linear"):
    t = np.linspace(0, 1, n)
    if kind == "linear":
        a = 1.0 + alpha * (1.0 - 2.0 * t)
    elif kind == "exponential":
        a = np.exp(alpha * (1.0 - 2.0 * t))
    elif kind == "cosine_tilt":
        a = (1.0 + np.cos(np.pi * t)) * (1.0 + alpha * (t - 0.5))
    elif kind == "power":
        a = (1.0 - t) ** alpha
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_fourier_tilt(n, rng, n_modes=3, alpha=0.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.3 * (1.0 - 2.0 * t)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = alpha * rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2.0 * np.pi * freq * t + phase)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _perturb_a_space(a, sigma, rng):
    n = len(a)
    noise = sigma * rng.standard_normal(n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s
