"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based peak minimization (minimize max formulation), L-BFGS continuation,
peak carving, and Adam polish with aggressive annealing.
Key innovation: LP formulation directly minimizes the peak rather than maximizing sum.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _u_shape_init(n):
    t = np.linspace(0, 1, n)
    a = 0.3 + 0.7 * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _parametric_init(n, etype):
    t = np.linspace(0, 1, n)
    if etype == 'u_quad':
        a = 0.2 + 0.8 * (2.0 * t - 1.0) ** 2
    elif etype == 'u_cubic':
        a = 0.1 + 0.9 * (2.0 * t - 1.0) ** 3
    elif etype == 'cos':
        a = 0.5 + 0.5 * np.cos(np.pi * t)
    elif etype == 'exp':
        a = np.exp(4.0 * (t - 0.5) ** 2 * 4.0)
    elif etype == 'asym':
        a = np.exp(2.0 * (t - 0.3))
    elif etype == 'bimodal':
        a = np.exp(-8.0 * (t - 0.2) ** 2) + np.exp(-8.0 * (t - 0.8) ** 2) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002,
                 0.001, 0.0005, 0.0003, 0.0001, 0.00005, 0.00002]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _build_conv_matrix_row(a, k, n):
    """Get the row of the convolution matrix for peak index k."""
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    row = np.zeros(n)
    if i_lo <= i_hi:
        row[i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return row


def _lp_minimize_peak(a_init, deadline, K=50, max_rounds=30, tightness=1.0):
    """LP that directly minimizes the max peak: minimize t s.t. (A*b)[k] <= (t + conv_star[k])/2"""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    S = float(best_a.sum())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.90 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        active = active[active < 2 * n - 1]
        if len(active) > 300:
            active = active[:300]

        K_actual = len(active)
        conv_star = c[active]

        # Build LP: minimize t
        # Variables: [b_0, ..., b_{n-1}, t] (n+1 variables)
        # Constraints:
        #   For each active k: (A*b)[k] <= (t + conv_star[k]) / 2
        #     => 2*(A*b)[k] - t <= conv_star[k]
        #     => -2*(A*b)[k] + t >= -conv_star[k]
        #     => 2*(A*b)[k] - t <= conv_star[k]
        #   sum(b) = S  (equality)
        #   b >= 0, t >= 0

        # For linprog: minimize c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
        # x = [b, t], objective = [0, 0, ..., 0, 1]

        A_ub = np.zeros((K_actual, n + 1))
        b_ub = np.zeros(K_actual)
        for row_idx, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_ub[row_idx, i_lo:i_hi + 1] = 2.0 * a_init[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
            A_ub[row_idx, n] = -1.0
            b_ub[row_idx] = conv_star[row_idx] * tightness

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])

        bounds = [(0, None)] * n + [(0, None)]

        try:
            res = linprog(np.zeros(n + 1)[:-1] + np.array([1.0]),
                          A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,

                          options={'time_limit': 20})
            if res.status != 0:
                # Try with slack
                res = linprog(np.zeros(n + 1)[:-1] + np.array([1.0]),
                              A_ub=A_ub, b_ub=b_ub,
                              A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs',
                              options={'time_limit': 20, 'presolve': True})
                if res.status != 0:
                    break

            b_new = res.x[:n]
            t_val = res.x[n]
            b_new = np.maximum(b_new, 0.0)
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        # Try blending with binary search
        improved = False
        current_score = 2.0 * n * M / (S * S)
        best_t_blend = 0.0
        best_trial_score = current_score

        lo, hi = 0.0, 1.0
        for _ in range(12):
            mid = (lo + hi) / 2.0
            a_trial = (1.0 - mid) * best_a + mid * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                break
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < best_trial_score:
                best_trial_score = sc_trial
                best_t_blend = mid
                lo = mid
            else:
                hi = mid
            if hi - lo < 0.01:
                break

        if best_t_blend > 0 and best_trial_score < current_score - 1e-12:
            a_trial = (1.0 - best_t_blend) * best_a + best_t_blend * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            best_a = a_trial.copy()
            best_score = _score_norm_fn(best_a)
            improved = True
        else:
            # Try direct use
            a_trial = b_scaled
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (S * S)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(best_a)
                improved = True
            else:
                # Try small blend
                for t in [0.05, 0.1, 0.15, 0.2]:
                    a_trial = (1.0 - t) * best_a + t * b_scaled
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:

def _peak_carving(a_init, deadline, top_k=10, max_rounds=50):
    """Directly reduce the highest peaks by adjusting contributing coefficients."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    S = float(best_a.sum())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        top_peaks = np.argsort(c)[::-1][:top_k]

        # For each top peak, find the largest contributor and reduce it
        improved = False
        a_trial = best_a.copy()

        for k in top_peaks:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            # Contributors are a[i] for i in [i_lo, i_hi]
            # conv[k] = sum_{i=i_lo}^{i_hi} a[i] * a[k-i]
            # To reduce conv[k], reduce the largest a[i] * a[k-i] product
            contribs = np.zeros(i_hi - i_lo + 1)
            for idx, i in enumerate(range(i_lo, i_hi + 1)):
                contribs[idx] = a_trial[i] * a_trial[k - i]
            max_idx = np.argmax(contribs)
            target_i = i_lo + max_idx

            # Reduce a[target_i] slightly
            factor = 0.95
            old_val = a_trial[target_i]
            a_trial[target_i] = old_val * factor

            # Renormalize to maintain sum
            s_trial = a_trial.sum()
            if s_trial > 0:
                a_trial = a_trial * (S / s_trial)

        # Check if this helped
        c_trial = _full_convolve(a_trial, a_trial)
        M_trial = float(c_trial.max())
        S_trial = float(a_trial.sum())
        sc_trial = 2.0 * n * M_trial / (S_trial * S_trial)
        current_score = 2.0 * n * M / (S * S)

        if sc_trial < current_score - 1e-10:
            best_a = a_trial.copy()
            best_score = _score_norm_fn(best_a)
            improved = True
        else:
            # Try reducing multiple peaks simultaneously with smaller factor
            a_trial = best_a.copy()
            for k in top_peaks[:5]:
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo > i_hi:
                    continue
                contribs = np.zeros(i_hi - i_lo + 1)
                for idx, i in enumerate(range(i_lo, i_hi + 1)):
                    contribs[idx] = a_trial[i] * a_trial[k - i]
                max_idx = np.argmax(contribs)
                target_i = i_lo + max_idx
                a_trial[target_i] *= 0.98

            s_trial = a_trial.sum()
            if s_trial > 0:
                a_trial = a_trial * (S / s_trial)

            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            S_trial = float(a_trial.sum())
            sc_trial = 2.0 * n * M_trial / (S_trial * S_trial)
            if sc_trial < current_score - 1e-10:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(best_a)
                improved = True
            else:
                break

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.03, max_iter=500000):
    """Adam optimizer with adaptive beta log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.96 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _active_set_phase(a_init, deadline, K=5, max_rounds=50):
    """Minimize max of top K peaks directly via L-BFGS."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for outer in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]

        def ao(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            beta = 1e-8 * pm
            sh = (peaks - pm) / beta
            lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)

        def ag(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            beta = 1e-8 * pm
            sh = (peaks - pm) / beta
            e = np.exp(sh - sh.max())
            w = e / e.sum()

            grad_a = np.zeros(n)
            for idx_k in range(len(peak_indices)):
                k = peak_indices[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            result = minimize(ao, z, method='L-BFGS-B', jac=ag,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 50, 'maxfun': 15000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score - 1e-14:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _multi_start_search(a_init, deadline, n_starts=5):
    """Try multiple perturbations and optimize each."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    rng = np.random.default_rng(12345)

    time_per_start = (deadline - time.monotonic()) / max(1, n_starts)

    for s in range(n_starts):
        if time.monotonic() > deadline - 10:
            break
        sub_deadline = min(deadline - 10, time.monotonic() + time_per_start)

        # Different perturbation strategies
        if s == 0:
            a_try = a_init + 0.1 * rng.standard_normal(n)
        elif s == 1:
            a_try = a_init * np.exp(0.2 * rng.standard_normal(n))
        elif s == 2:
            # Swap random segments
            a_try = a_init.copy()
            for _ in range(5):
                i = rng.integers(0, n - 100)
                j = rng.integers(0, n - 100)
                length = 50
                a_try[i:i+length], a_try[j:j+length] = a_try[j:j+length].copy(), a_try[i:i+length].copy()
        elif s == 3:
            # Smooth perturbation
            kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
            a_try = a_init.copy()
            for _ in range(10):
                i = rng.integers(20, n - 20)
                a_try[i-20:i+21] += 0.05 * kernel * rng.standard_normal()
        else:
            a_try = a_init * (1.0 + 0.3 * rng.random(n))

        a_try = np.maximum(a_try, 1e-12)
        a_try = a_try / a_try.sum()
        sc_try = _score_norm_fn(a_try)

        if sc_try < best_score:
            best_a = a_try.copy()
            best_score = sc_try

        # Quick L-BFGS polish
        a_pol, sc_pol = _run_lbfgs(a_try, sub_deadline, maxiter_per_beta=2000)
        if sc_pol < best_score:
            best_a = a_pol.copy()
            best_score = sc_pol

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization with LP peak minimization, L-BFGS continuation,
    peak carving, active-set refinement, and Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        actual_parent = _score_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={actual_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Quick L-BFGS (200s) ===
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 200.0)

    a_b1, sc1 = _run_lbfgs(best_seq, p1_end)
    update_best(a_b1, sc1, "lbfgs_main")

    # Try U-shaped init
    if time.monotonic() < p1_end - 30:
        a_u = _u_shape_init(n)
        sc_u = _score_norm_fn(a_u)
        print(f"  U-shape init: {sc_u:.10f}")
        if sc_u < best_score * 1.05:
            a_ub, sc_ub = _run_lbfgs(a_u, min(p1_end - 5, time.monotonic() + 50))
            update_best(a_ub, sc_ub, "lbfgs_u_shape")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: LP minimize peak (400s) ===
    print("Phase 2: LP minimize peak")
    p2_end = min(deadline, t_start + 600.0)

    if time.monotonic() < p2_end and best_seq is not None:
        a_lp1, sc_lp1 = _lp_minimize_peak(best_seq, p2_end, K=30, max_rounds=20, tightness=1.0)
        update_best(a_lp1, sc_lp1, "lp_min_peak_K30")

        if time.monotonic() < p2_end - 60:
            a_lp2, sc_lp2 = _lp_minimize_peak(best_seq, p2_end, K=60, max_rounds=20, tightness=0.99)
            update_best(a_lp2, sc_lp2, "lp_min_peak_K60_tight")

        if time.monotonic() < p2_end - 30:
            a_lp3, sc_lp3 = _lp_maximize_sum(best_seq, p2_end, K=40, max_rounds=10)
            update_best(a_lp3, sc_lp3, "lp_max_sum")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak carving (200s) ===
    print("Phase 3: Peak carving")
    p3_end = min(deadline, t_start + 800.0)

    if time.monotonic() < p3_end and best_seq is not None:
        a_pc, sc_pc = _peak_carving(best_seq, p3_end, top_k=10, max_rounds=100)
        update_best(a_pc, sc_pc, "peak_carving")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Active set (200s) ===
    print("Phase 4: Active set refinement")
    p4_end = min(deadline, t_start + 1000.0)

    if time.monotonic() < p4_end and best_seq is not None:
        for K in [3, 5, 7]:
            if time.monotonic() > p4_end - 30:
                break
            a_act, sc_act = _active_set_phase(best_seq, p4_end, K=K, max_rounds=30)
            update_best(a_act, sc_act, f"active_K{K}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Second LP round with more peaks (250s) ===
    print("Phase 5: LP round 2 (more peaks)")
    p5_end = min(deadline, t_start + 1250.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_lp4, sc_lp4 = _lp_minimize_peak(best_seq, p5_end, K=80, max_rounds=25, tightness=0.995)
        update_best(a_lp4, sc_lp4, "lp_round2_K80")

        if time.monotonic() < p5_end - 30:
            a_lp5, sc_lp5 = _lp_minimize_peak(best_seq, p5_end, K=100, max_rounds=15, tightness=0.99)
            update_best(a_lp5, sc_lp5, "lp_round2_K100")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Adam polish (300s) ===
    print("Phase 6: Adam polish")
    p6_end = min(deadline, t_start + 1550.0)

    if time.monotonic() < p6_end and best_seq is not None:
        a_ad, sc_ad = _adam_polish(best_seq, p6_end, lr=0.02)
        update_best(a_ad, sc_ad, "adam_lr02")

        if time.monotonic() < deadline - 100:
            a_ad2, sc_ad2 = _adam_polish(best_seq, deadline - 50, lr=0.008)
            update_best(a_ad2, sc_ad2, "adam_slow")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final L-BFGS with tiny beta (100s) ===
    print("Phase 7: Final tiny-beta L-BFGS")
    p7_end = min(deadline, t_start + 1650.0)

    if time.monotonic() < p7_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
        a_f, sc_f = _run_lbfgs(best_seq, p7_end, beta_list=tiny_betas, maxiter_per_beta=8000)
        update_best(a_f, sc_f, "final_tiny_beta")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Multi-start if time allows (100s) ===
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print("Phase 8: Multi-start")
        a_ms, sc_ms = _multi_start_search(best_seq, deadline - 30, n_starts=3)
        update_best(a_ms, sc_ms, "multi_start")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(256)
    print(f"FALLBACK: n=256, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
