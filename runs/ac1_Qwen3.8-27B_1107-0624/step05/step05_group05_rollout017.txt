```python
import numpy as np

import scipy.sparse as sp

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(2, N)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())
```

```python
def _slp_step(a, max_iter_lp=1):
    """One SLP iteration: linearize at a, solve LP, return new point."""
    n = len(a)
    a = a / a.sum()  # normalize
    conv = _full_convolve(a, a)
    M = float(conv.max())
    
    # Build LP matrix
    # Variables: a_0, ..., a_{n-1}, t (n+1 vars)
    # Constraints: for each k, 2*sum_j a[j]*a[k-j] - t <= conv[k]
    # Plus: sum a[j] = 1, a[j] >= 0
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(2.0 * a[k - j])
        # t column
        rows.append(k)
        cols.append(n)  # t is variable n
        vals.append(-1.0)
    
    A_ub = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = conv  # rhs
    
    # Equality: sum a[j] = 1
    A_eq = sp.csc_matrix((np.ones(n), (np.arange(n), np.arange(n))), shape=(1, n + 1))
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    bounds = [(0, None)] * (n + 1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': 30})
    
    if res.status != 0:
        return a, M
    
    a_new = np.maximum(res.x[:n], 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, M
    a_new = a_new / s
    
    return a_new, M
```
