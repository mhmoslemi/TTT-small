import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find rho
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - S))[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _solve_lp(f, M, solver='CLARABEL'):
    """Solve LP: max sum(b) s.t. conv(f, b)_k <= M for all k, b >= 0."""
    n = len(f)
    b = cp.Variable(n)
    
    # Build constraint matrix: A[k, j] = f[k-j] (with zero padding)
    # (f*b)_k = sum_j f[k-j] * b[j] = ess_j f[max(0, k-j) if 0<=k-j<n else 0] * b[j]
    # Actually: (f*b)_k = sum_j b[j] * f[k-j] where f[i] = 0 for i not in [0, n-1]

    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    
    # Build using scipy's toefli or manually
    # A[k,j] = f[k-j] for k=0..2n-2, j=0..n-1
    # This is a convolution matrix
    
    # Actually, let's just build it directly
    # For n=2048, the matrix is 4095 x 2048 = ~8.4M entries, which is fine
    
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = f[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        A[j:j+n, j] = f
    
    constraints = [A @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if b.value is not None:
            return np.maximum(b.value, 0)
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    while k < n:

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    """Build the convolution matrix A where (A @ b)[k] = (f*b)[k]."""
    n = len(f)
    # A has shape (2n-1, n)
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # For fixed j, A[:, j] is f shifted by j positions (padded)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def _solve_lp(f, M, solver=None):
    """Solve: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    b = cp.Variable(n)
    A = _build_conv_matrix(f)
    
    constraints = [A @ b <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        if solver:
            prob.solve(solver=solver)
        else:
            prob.solve()
        if b.value is not None:
            return np.maximum(b.value, 0)
    except Exception as e:
        print(f"  LP solve error: {e}")
    return None

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.minimum(shifted, 50))  # clip to avoid overflow
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(np.minimum(shifted, 50))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgrad_step(a, n, lr):
    """One subgradient step with projection onto simplex."""
    c = _full_convolve(a, a)
    c_max = c.max()
    # Active set: indices where c is within 1% of max
    tol = 0.01 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    
    # Gradient: g_m = 2 * sum_i a_i * w_{i+m}
    # Using the convolution formula from before
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    g = 2.0 * wc[n-1:2*n-1]
    
    # Subgradient step
    a_new = a - lr * g
    # Project onto simplex {x >= 0, sum(x) = sum(a)}
    S = float(np.sum(a))
    a_new = _project_simplex(a_new, S)
    return a_new

def _project_simplex(v, S):
    """Project v onto {x >= 0, sum(x) = S}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - S)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    rho = rho[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    solvers_to_try = [cp.GLOP, cp.CBC, cp.ECOS, cp.CVXOPT]
    active_solver = None
    no_improve_count = 0
    prev_sc = _score(f)

    for it in range(50000):
        if time.monotonic() > lp_deadline:
            break

        c = _full_convolve(f, f)
        M = float(c.max())

        b_val = None
        if active_solver is None:
            for sol_name in solvers_to_try:
                b_val = _solve_lp(f, M, sol_name)
                if b_val is not None and float(b_val.sum()) > 1e-10:
                    active_solver = sol_name
                    break
        else:
            b_val = _solve_lp(f, M, active_solver)
            if b_val is None or float(b_val.sum()) < 1e-10:
                for sol_name in solvers_to_try:
                    if sol_name == active_solver:
                        continue
                    b_val = _solve_lp(f, M, sol_name)
                    if b_val is not None and float(b_val.sum()) > 1e-10:
                        active_solver = sol_name
                        break

        if b_val is None or float(b_val.sum()) < 1e-10:
            print(f"  LP failed at iter {it}, switching to next phase")
            break

        b_sum = float(b_val.sum())
        g = (S_val / b_sum) * b_val

        # Line search
        best_t = 0.0
        current_sc = _score(f)
        for t_try in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            f_trial = (1.0 - t_try) * f + t_try * g
            sc_trial = _score(f_trial)
            if sc_trial < current_sc - 1e-14:
                best_t = t_try
                current_sc = sc_trial
                break

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            sc_new = _score(f)
            update_best(f, sc_new, f"lp_{it}")
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 20:
                print(f"  LP stuck after {no_improve_count} no-improve iters at iter {it}")
                break

        if it % 50 == 0:
            print(f"  LP iter {it}: score={_score(f):.10f}, M={M:.4f}, t={best_t:.3f}")

    print(f"Phase 1 done: score={_score(f):.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: L-BFGS with beta continuation ===
    print(f"\n=== Phase 2: L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    lbfgs_deadline = deadline - (0.25 * (deadline - t_start))
    a_curr = f / f.sum()

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    for i, beta in enumerate(betas):
        if time.monotonic() > lbfgs_deadline:
            break
        remaining = lbfgs_deadline - time.monotonic()
        if remaining < 10:
            break

        max_iter = min(3000, max(100, int(remaining / 0.05)))
        print(f"  L-BFGS beta={beta:.2e}, max_iter={max_iter}, time_left={remaining:.1f}s")

        a_new = _lbfgs_phase(a_curr, beta, n, lbfgs_deadline, max_iter)
        a_new = a_new * S_val / a_new.sum()
        sc_new = _score(a_new)

        if sc_new < _score(a_curr):
            a_curr = a_new
            update_best(a_curr, sc_new, f"lbfgs_{i}")
        else:
            # Try with tiny perturbation
            z_p = np.log(np.maximum(a_curr, 1e-30))
            z_p = z_p + 0.01 * rng.standard_normal(n)
            z_p = z_p - z_p.mean()
            a_pert = _softmax(z_p) * S_val
            a_new2 = _lbfgs_phase(a_pert, beta, n, lbfgs_deadline, max(100, max_iter // 3))
            a_new2 = a_new2 * S_val / a_new2.sum()
            sc_new2 = _score(a_new2)
            if sc_new2 < _score(a_curr):
                a_curr = a_new2
                update_best(a_curr, sc_new2, f"lbfgs_{i}_pert")

    print(f"Phase 2 done: score={_score(a_curr * S_val):.10f}, t={time.monotonic()-t_start:.1f}s")

    # Update best if a_curr is better
    a_curr_scaled = a_curr * S_val
    update_best(a_curr_scaled, _score(a_curr_scaled), "post_lbfgs")

    # === Phase 3: Subgradient polishing ===
    print(f"\n=== Phase 3: Subgradient Polish (t={time.monotonic()-t_start:.1f}s) ===")
    sg_deadline = deadline - (0.12 * (deadline - t_start))
    if best_seq is not None:
        a_sg_start = best_seq.copy()
        a_sg, sc_sg = _subgrad_polish(a_sg_start, sg_deadline, n_steps=20000, lr_init=0.01)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 4: Multi-restart with diverse inits ===
    print(f"\n=== Phase 4: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    restart_deadline = deadline - 30.0
    if time.monotonic() < restart_deadline:
        remaining = restart_deadline - time.monotonic()
        kinds = ["tapered", "cosine", "bimodal", "flat_center", "uniform"]
        max_restarts = min(5, max(1, int(remaining / 120)))

        for r in range(max_restarts):
            if time.monotonic() > restart_deadline:
                break

            kind = kinds[r % len(kinds)]
            a_r = _make_initial(n, kind) * S_val
            sc_r_init = _score(a_r)
            print(f"  Restart {r} ({kind}): init_score={sc_r_init:.10f}")

            # Quick L-BFGS
            c_r = _full_convolve(a_r / a_r.sum(), a_r / a_r.sum())
            max_cr = float(c_r.max())
            betas_r = max_cr * np.array([0.1, 0.05, 0.01, 0.003])
            a_r_curr = a_r / a_r.sum()

            for beta_r in betas_r:
                if time.monotonic() > restart_deadline - 10:
                    break
                a_r_curr = _lbfgs_phase(a_r_curr, beta_r, n, restart_deadline - 5, max_iter=500)
                a_r_curr = a_r_curr * S_val / a_r_curr.sum()
                sc_r = _score(a_r_curr)
                if sc_r < best_score + 0.01:
                    update_best(a_r_curr, sc_r, f"restart_{r}_b{beta_r:.2e}")

            # Quick subgradient
            if time.monotonic() < restart_deadline - 10:
                a_r2, sc_r2 = _subgrad_polish(a_r_curr, restart_deadline - 5, n_steps=2000, lr_init=0.01)
                update_best(a_r2, sc_r2, f"restart_{r}_sg")

    # === Phase 5: Final refinement on best ===
    print(f"\n=== Phase 5: Final Refinement (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = best_seq.copy()
        # Very fine L-BFGS
        c_fin = _full_convolve(a_fin / a_fin.sum(), a_fin / a_fin.sum())
        max_cf = float(c_fin.max())
        beta_fin = max_cf * 0.0001
        a_fin_norm = a_fin / a_fin.sum()
        a_fin2 = _lbfgs_phase(a_fin_norm, beta_fin, n, deadline - 3, max_iter=2000)
        a_fin2 = a_fin2 * a_fin.sum() / a_fin2.sum()
        update_best(a_fin2, _score(a_fin2), "final_lbfgs")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 1, n_steps=5000, lr_init=0.005)
            update_best(a_fin3, sc_fin3, "final_sg")

        # LP step from best
        if time.monotonic() < deadline - 2:
            c_best = _full_convolve(a_fin / a_fin.sum(), a_fin / a_fin.sum())
            M_best = float(c_best.max())
            b_final = _solve_lp(a_fin / a_fin.sum(), M_best, active_solver if active_solver else cp.GLOP)
            if b_final is not None and float(b_final.sum()) > 1e-10:
                g_final = (a_fin.sum() / float(b_final.sum())) * b_final
                for t_try in [0.5, 0.3, 0.1, 0.05]:
                    f_try = (1 - t_try) * a_fin + t_try * g_final
                    sc_try = _score(f_try)
                    if sc_try < best_score:
                        update_best(f_try, sc_try, "final_lp")
                        break
