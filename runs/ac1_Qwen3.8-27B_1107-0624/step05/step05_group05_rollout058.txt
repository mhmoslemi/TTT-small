Actually, let me reconsider. The LP "minimize max(a*g) subject to g>=0, sum(g)=1" is equivalent to:
- Minimize `t`
- Subject to `sum_i a[k-i] * g[i] <= t` for all `k`
- `sum_i g[i] = 1`
- `g[i] >= 0`

The constraint matrix `A` has `A[k,i] = a[k-i]` if `0 <= k-i < n`, else 0. This is a banded (actually, Toeplitz-like) matrix.

"""
Hybrid LP + Mirror Descent + Coordinate Descent optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses a direct LP formulation (minimize max(a*g)) for principled updates,
exponential family mirror descent for intermediate refinement, and
coordinate descent for final polishing. Explores multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is normalized (sum=1). Returns 2n*max(a*a)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(a):
    """Build sparse matrix A where A[k,i] = a[k-i] if 0<=k-i<n."""
    n = len(a)
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
        # A[k, j] = a[k - j]
        vals_list.append(a[k - j_range])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_min_max_dir(a):
    """Solve: min t s.t. (a*g)[k] <= t for all k, g>=0, sum(g)=1.
    Returns (g, t) or (None, None) on failure."""
    n = len(a)
    A = _build_conv_matrix(a)
    # Variables: g[0..n-1], t
    # Objective: minimize t (index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    # Inequality: A*g - t <= 0, i.e., A_aug * [g; t] <= 0
    # A_aug = [A, -1] (shape (2n-1, n+1))
    neg_ones = -np.ones(2 * n - 1)
    A_aug = sp.hstack([A, sp.csc_matrix((neg_ones, (np.arange(2*n-1), np.full(2*n-1, n))), shape=(2*n-1, 1))], format='csc')
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 120})
        if res.status != 0:
            return None, None
        g = np.maximum(res.x[:n], 0.0)
        t = float(res.x[n])
        g_sum = g.sum()
        if g_sum < 1e-15:
            return None, None
        g /= g_sum  # Re-normalize for numerical safety
        return g, t
    except Exception as e:
        print(f"  LP error: {e}")
        return None, None

def _lp_step_v2(a, t_line_search=True):
    """LP-based update: find g minimizing max(a*g), then interpolate."""
    n = len(a)

    for t in candidates:

        a_cand = np.maximum(a_cand, 0.0)
        s = a_cand.sum()
        if s < 1e-15:

        a_cand /= s

def _mirror_descent_step(a, eta, k_max=None):
    """One step of mirror descent (exponential family) on the simplex."""
    n = len(a)
    c = _full_convolve(a, a)
    if k_max is None:
        k_max = int(np.argmax(c))
    # Gradient of b[k_max] w.r.t. a[i] = 2*a[k_max - i] (if valid)
    grad = np.zeros(n)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # Mirror descent: z = log(a) - eta * grad
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    return _softmax(z)

def _mirror_descent_multi(a, eta, n_max_indices=3):
    """Mirror descent using a weighted average of gradients from top indices."""
    n = len(a)
    c = _full_convolve(a, a)
    # Get top indices
    top_idx = np.argsort(c)[-n_max_indices:]
    top_vals = c[top_idx]
    top_max = top_vals.max()
    # Weights: softmax of top values
    weights = np.exp((top_vals - top_max) / max(top_max * 0.01, 1e-10))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += w * 2.0 * a[j]
    
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    return _softmax(z)

    n = len(a)
    best_a = a.copy()

        if deadline is not None and time.monotonic() > deadline:

def _upsample(a, new_n):
    """Upsample sequence to new length using smooth interpolation."""
    n0 = len(a)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_n)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new

"""
Hybrid LP + Mirror Descent + Adam optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a direct LP formulation (minimize max(a*g) over simplex) for principled direction-finding,
exponential-family mirror descent for intermediate refinement, and Adam in log-space for polishing.
Explores both n=2048 and n=4096 sequence lengths with multi-restart strategy.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_pattern(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _make_conv_matrix(a, rows, cols):

    vals = a[rows - cols]
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_min_max_dir(a, rows, cols):
    n = len(a)
    A = _make_conv_matrix(a, rows, cols)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    nr = 2 * n - 1
    aug_cols = np.concatenate([rows, np.full(nr, n, dtype=np.int32)])
    aug_rows = np.concatenate([cols, np.arange(nr, dtype=np.int32)])
    aug_vals = np.concatenate([a[rows - cols], -np.ones(nr)])
    A_aug = sp.csc_matrix((aug_vals, (aug_rows, aug_cols)), shape=(nr, n + 1))
    b_ub = np.zeros(nr)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * (n + 1)
    try:
        res = linprog(c_obj, A_ub=A_aug, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 90, 'primal_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None
        g = np.maximum(res.x[:n], 0.0)
        t = float(res.x[n])
        g_sum = g.sum()
        if g_sum < 1e-15:
            return None, None
        g /= g_sum
        return g, t
    except Exception:

def _mirror_step(a, eta, n_top=5):
    n = len(a)
    c = _full_convolve(a, a)
    top_idx = np.argsort(c)[-n_top:]
    top_vals = c[top_idx]
    top_max = top_vals.max()
    weights = np.exp((top_vals - top_max) / max(top_max * 0.01, 1e-10))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        neg = np.maximum(0, -k)
        pos = np.minimum(n, n - (k - (n - 1)))
        lo = max(0, k - (n - 1))
        hi = min(n, k + 1)
        idx = np.arange(lo, hi)
        j_idx = k - idx
        grad[idx] += w * 2.0 * a[j_idx]
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    return _softmax(z)

    v_buf = np.zeros(n)
    t = 0

    current_lr = lr_start
    no_improve = 0

    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)

        j_idx = k_max - idx
        grad_a = np.zeros(n)
        grad_a[idx] = 2.0 * a[j_idx]

        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z

def _upsample(a, new_n):
    n0 = len(a)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, new_n)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-15)
    a_new /= a_new.sum()
    return a_new

            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a_p.copy()

    # === Phase 6: Final Adam (1650 - deadline) ===
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 6: Final Adam ===")
        a_fin, sc_fin = _adam_polish(best_seq, deadline - 5, lr_start=0.002, lr_min=0.0002, patience=200)
        full_sc = _score(a_fin)
        print(f"  Final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_fin.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
