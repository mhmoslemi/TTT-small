"""
Optimized step function coefficient search using Nesterov-accelerated
first-order methods with adaptive log-sum-exp smoothing, periodic
L-BFGS refinement, and LP-based peak balancing.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0, a, c, cm
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    lse = cm + beta * (max_s + float(np.log(np.sum(e))))
    return float(2.0 * n * lse), a, c, cm


def _lse_grad(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z), a, c, cm
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z), a, c, cm
    return grad, a, c, cm

                        max_iter=500000, restart_interval=5000):

    z = z_init.copy()
    z_prev = z.copy()
    y = z.copy()  # Nesterov lookahead point
    
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)
    
    c0 = _full_convolve(best_a, best_a)
    cm0 = float(c0.max())
    if beta_init is None:
        beta = 0.1 * cm0
    
    it = 0
    no_improve = 0
    check_interval = 100
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        # Adaptive beta: smaller when peaks are flat
        a_y = _softmax_fn(y)
        c_y = _full_convolve(a_y, a_y)
        sorted_c = np.sort(c_y)[::-1]
        c_max = sorted_c[0]
        c_2nd = sorted_c[1] if len(sorted_c) > 1 else 0
        gap = (c_max - c_2nd) / max(c_max, 1e-300)
        # Target beta: proportional to the gap
        target_beta = max(1e-8 * c_max, min(0.5 * c_max, gap * c_max * 0.5))
        # Smoothly adjust beta
        beta = 0.95 * beta + 0.05 * target_beta
        
        # Compute gradient at y (Nesterov lookahead)
        g, a_y, c_y, cm_y = _lse_grad_from(y, beta, n, a_y, c_y)
        
        # Nesterov update
        y = z + momentum * (z - z_prev)
        g, _, _, _ = _lse_grad_from(y, beta, n, None, None)
        z_prev = z.copy()
        z = y - lr * g
        
        # Periodic restart (reset momentum)
        if it % restart_interval == 0:
            z_prev = z.copy()
            y = z.copy()
        
        # Check progress
        if it % check_interval == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_check.copy()

                z = z - z.mean()
                z_prev = z.copy()
                y = z.copy()
                no_improve = 0
    
    return best_a, best_score, z

"""
Optimized coefficient search using Nesterov-accelerated first-order methods with
adaptive log-sum-exp smoothing, periodic L-BFGS refinement, LP peak balancing,
and multi-resolution exploration. Targets score < 1.5027.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_value_grad(z, beta, n, a_cache=None, c_cache=None):
    """Compute log-sum-exp smoothed objective and its gradient w.r.t. z."""
    if a_cache is not None and c_cache is not None:
        a, c = a_cache, c_cache
    else:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0, np.zeros_like(z), a, c, cm
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    e_sum = float(np.sum(e))
    lse = cm + beta * (max_s + np.log(e_sum))
    obj = float(2.0 * n * lse)
    w = e / e_sum
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return obj, grad, a, c, cm


def _adapt_beta(c, beta, floor_frac=1e-7, cap_frac=0.5, rate=0.08):
    """Adaptively adjust beta based on peak gap."""
    cm = float(c.max())
    if cm < 1e-300:
        return beta
    sorted_c = np.sort(c)[::-1]
    c_second = float(sorted_c[1]) if len(sorted_c) > 1 else 0.0
    gap = (cm - c_second) / cm
    target_beta = max(floor_frac * cm, min(cap_frac * cm, 0.4 * gap * cm))
    new_beta = (1.0 - rate) * beta + rate * target_beta
    return new_beta


def _nesterov_phase(z_init, deadline, beta_init=None, lr=0.004, momentum=0.9,
                    max_iter=300000, restart_every=15000, check_every=200):
    """Nesterov-accelerated gradient descent with adaptive beta and periodic restarts."""
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    z_prev = z.copy()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)

    c0 = _full_convolve(best_a, best_a)
    cm0 = float(c0.max())
    if beta_init is None:
        beta = 0.1 * cm0

    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Nesterov lookahead
        y = z + momentum * (z - z_prev)

        # Compute gradient at y
        a_y = _softmax_fn(y)
        c_y = _full_convolve(a_y, a_y)
        obj, g, a_y, c_y, cm_y = _lse_value_grad(y, beta, n, a_y, c_y)

        # Adaptive beta
        if it % 50 == 0:
            beta = _adapt_beta(c_y, beta)

        # Gradient step
        z_new = y - lr * g

        # Simple line search (check if objective decreased)
        if it % 10 == 0:
            a_check = _softmax_fn(z_new)
            c_check = _full_convolve(a_check, a_check)
            obj_check, _, _, _, _ = _lse_value_grad(z_new, beta, n, a_check, c_check)
            if obj_check > obj * 1.01 + 1e-12:
                # Halve the step
                z_new = y - lr * 0.5 * g

        z_prev = z
        z = z_new

        # Periodic restart (reset momentum to avoid oscillation)
        if it % restart_every == 0:
            z_prev = z.copy()

        # Check progress
        if it % check_every == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:

                z = z - z.mean()
                z_prev = z.copy()
                no_improve = 0
                # Also try to reduce beta if stuck

    return best_a, best_score, z


def _lbfgs_run(z_init, deadline, beta_list, maxiter=5000, n=None):
    """Run L-BFGS with a list of beta values (continuation method)."""
    if n is None:
        n = len(z_init)
    z = z_init.copy()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)

    def obj_fn(z_in):
        val, _, _, _, _ = _lse_value_grad(z_in, beta, n)
        return val

    def grad_fn(z_in):
        _, g, _, _, _ = _lse_value_grad(z_in, beta, n)
        return g

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(obj_fn, z, method='L-BFGS-B', jac=grad_fn,
                           options={'maxiter': maxiter, 'ftol': 1e-25,
                                    'gtol': 1e-15, 'maxls': 50, 'maxfun': 20000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass

    return best_a, best_score, z


def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat

def _greedy_init(n):
    """Binomial-based initialization."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _gaussian_init(n, alpha=3.0):
    """Gaussian-like initialization."""
    t = np.linspace(0, 1, n)
    a = np.exp(-alpha * (t - 0.5) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _parametric_init(n, etype='exp3'):
    t = np.linspace(0, 1, n)
    if etype == 'exp3':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'linear':
        a = 0.1 + 0.9 * t
    elif etype == 'cubic':
        a = 0.05 + 0.5 * t + 0.4 * t ** 3
    elif etype == 'bimodal':
        a = np.exp(-8.0 * (t - 0.3) ** 2) + 0.5 * np.exp(-8.0 * (t - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

    print("\n=== Phase 5: Polish and larger-n exploration ===")

    # Nesterov polish with smaller lr
    if time.monotonic() < deadline - 80 and best_seq is not None:
        z_pol = np.log(np.maximum(best_seq, 1e-15))
        z_pol = z_pol - z_pol.mean()
        c_pol = _full_convolve(best_seq, best_seq)
        m_pol = float(c_pol.max())
        a_pol, sc_pol, _ = _nesterov_phase(z_pol, deadline - 80,
                                           beta_init=0.005 * m_pol,
                                           lr=0.002, momentum=0.95,
                                           max_iter=200000, restart_every=20000)
        update_best(a_pol, sc_pol, "nesterov_polish")

    # Try larger n
    if time.monotonic() < deadline - 60 and best_seq is not None:
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"  Trying n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm_fn(a_interp)
            print(f"  Interpolated score: {sc_in:.12f}")
            update_best(a_interp, sc_in, f"interp_n{n_new}")

            if time.monotonic() < deadline - 30 and sc_in < 2.0:
                z_in = np.log(np.maximum(a_interp, 1e-15))
                z_in = z_in - z_in.mean()
                c_in = _full_convolve(a_interp, a_interp)
                m_in = float(c_in.max())
                a_in2, sc_in2, _ = _nesterov_phase(z_in, deadline - 20,
                                                   beta_init=0.05 * m_in,
                                                   lr=0.003, momentum=0.9,
                                                   max_iter=100000)
                update_best(a_in2, sc_in2, f"nesterov_n{n_new}")

    print(f"  Phase 5 done: best={best_score:.12f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.12f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.12f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.12f}")
    return [float(x) for x in fb]
