"""
Multi-scale Adam optimization with LP corrections for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses progressive sequence length increase and adaptive temperature scheduling.

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def _lp_step(f, M, n, time_limit=10):
    """Single LP step: find g maximizing sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds_lp = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_lp,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    z = z + sigma * rng.standard_normal(len(a))

"""
Multi-scale gradient optimization with adaptive temperature and LP corrections
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses progressive sequence length increase (1024->4096->8192->16384),
Adam with adaptive softmax temperature, periodic LP corrections,
and structured multi-restart for escaping local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _adam_optimize(a_init, deadline, lr=0.01, beta_start=0.01, beta_end=0.00005, rng=None):
    n = len(a_init)
    if rng is None:
        rng = np.random.default_rng(0)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c = max(best_score, 1e-10)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    iteration = 0
    patience = 0
    total_iters = 0
    last_check = time.monotonic()

    # Temperature schedule: geometric decay over total budget
    total_time = max(1.0, deadline - time.monotonic())
    
    while time.monotonic() < deadline:
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adaptive temperature based on elapsed time
        elapsed = time.monotonic() - (deadline - total_time)
        progress = min(1.0, elapsed / total_time)
        beta = beta_start * max_c * (beta_end / beta_start) ** progress

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** (t_step + 1))
        vh = v_buf / (1.0 - b2 ** (t_step + 1))
        t_step += 1

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        iteration += 1
        total_iters += 1

        # Learning rate decay

            current_lr = max(0.0003, current_lr * 0.93)

        # Periodic score check

            last_check = time.monotonic()
            sc = float(_full_convolve(a, a).max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

        # Stuck detection: perturb and reset momentum
        if patience > 3000:
            z = z + 0.02 * rng.standard_normal(n)
            z = z - z.mean()
            m_buf *= 0.5
            v_buf *= 0.5
            patience = 0

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lp_optimize(f, deadline, max_steps=50):
    """LP-based Frank-Wolfe optimization. Returns optimized f."""
    n = len(f)
    f = f / f.sum()
    steps_done = 0
    stuck = 0
    current_t = 0.3

    while time.monotonic() < deadline - 5 and steps_done < max_steps:
        c = _full_convolve(f, f)
        M = float(c.max())

        A = _build_lp_matrix(f, n)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds_lp = [(0, None)] * n
        g = None
        try:
            tl = min(20, deadline - time.monotonic() - 5)
            if tl < 2:
                break
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'time_limit': tl, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
                if g.sum() < 1e-15:
                    g = None
        except Exception:
            g = None

        if g is None:
            stuck += 1
            if stuck > 10:
                break
        else:
            stuck = 0
            g_sum = float(g.sum())
            g_norm = g / g_sum

            improved = False
            t_try = current_t
            for _ in range(5):
                f_new = (1 - t_try) * f + t_try * g_norm
                f_new = np.maximum(f_new, 0)
                f_new_sum = float(f_new.sum())
                if f_new_sum < 1e-15:
                    t_try *= 0.5
                    continue
                f_new = f_new / f_new_sum
                c_new = _full_convolve(f_new, f_new)
                M_new = float(c_new.max())
                if M_new < M * (1 - 1e-10):
                    f = f_new
                    current_t = min(1.0, t_try * 1.3)
                    improved = True
                    steps_done += 1
                    break
                else:
                    t_try *= 0.5

            if not improved:
                current_t *= 0.5
                stuck += 1

    return f


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_structured(a, rng, n_modes=5):
    """Structured perturbation using low-frequency sinusoids."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        z += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    z = z - z.mean()
    return _softmax(z)
