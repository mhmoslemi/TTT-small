            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
                # Increase beta to fix numerical issues
                beta *= 2
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** max(1, t_step))
            vh = v_buf / (1.0 - b2 ** max(1, t_step))

            update = lr_phase * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 1.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)

            z = z - update
            z = np.clip(z, -35.0, 35.0)
            z = z - z.mean()

            if t_step % 400 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    phase_best = sc
                    if name and idx < 4:
                        print(f"    {name}: M={sc:.10f} beta={beta_scale:.5f} "
                              f"t={time.monotonic():.1f}s")

        # End of phase: check if we should continue
        if phase_best < best_score * (1 - 1e-10):
            best_score = phase_best

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _peak_shave(a, n_shaves=50, deadline=None):
    """Iteratively shave the peak of the convolution."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_m = float(c.max())

    for i in range(n_shaves):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        k_star = int(np.argmax(c))

        # Descent direction for c[k_star]: -2*a[k_star - i] for valid i
        desc = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                desc[i] = -a[j]

        # Project to sum-zero direction (maintain normalization)
        desc = desc - desc.mean()

        # Step size
        step = 0.01
        for _ in range(20):
            a_try = a + step * desc
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                step *= 0.5
                continue
            a_try = a_try / s
            c_try = _full_convolve(a_try, a_try)
            M_try = float(c_try.max())
            if M_try < M * (1 - 1e-10):
                a = a_try
                M = M_try
                if M < best_m:
                    best_m = M
                    best_a = a.copy()
                break
            step *= 0.5

    return best_a, best_m


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Main Adam optimization with cosine annealing (0-900s)
    Phase 2: Peak shaving + re-optimization (900-1200s)
    Phase 3: Multi-restart with diverse init (1200-1500s)
    Phase 4: Larger n exploration (1500-1700s)
    Phase 5: Final polish (1700-1780s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: score={best_score:.10f} (n={len(p)}) "
                  f"t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent loaded: n={n_main}, score={sc0:.10f}")
        update_best(a0)
    else:
        n_main = 8192
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")
        update_best(a0)

    # === Phase 1: Main Adam optimization (0 - 900s) ===
    phase1_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 1: Main Adam (n={n_main}, until {phase1_end - t_start:.0f}s) ===")

    if time.monotonic() < phase1_end - 30:
        # Run Adam v2 (more granular temperature schedule)
        a_best, sc_best = _adam_optimize_v2(
            a0, phase1_end - 10, lr=0.04, name="P1-main"
        )
        update_best(a_best)

        # Second run from the result with different LR
        if time.monotonic() < phase1_end - 60:
            a2, sc2 = _adam_optimize(
                a_best, phase1_end - 10, lr=0.02, name="P1-refine"
            )
            update_best(a2)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Peak shaving + re-optimization (900s - 1200s) ===
    phase2_end = min(deadline, t_start + 1200.0)
    print(f"\n=== Phase 2: Peak shaving + re-opt ===")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_best_norm = best_seq / best_seq.sum()

        # Peak shave
        a_shaved, m_shaved = _peak_shave(a_best_norm, n_shaves=100,
                                         deadline=phase2_end - 90)
        update_best(a_shaved)

        # Re-optimize from shaved result
        if time.monotonic() < phase2_end - 20:
            a_reopt, _ = _adam_optimize_v2(
                a_shaved, phase2_end - 10, lr=0.01, name="P2-reopt"
            )
            update_best(a_reopt)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart with diverse initialization (1200s - 1500s) ===
    phase3_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 3: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 5
        per_restart = (phase3_end - time.monotonic() - 20) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break
            deadline_i = min(time.monotonic() + per_restart, phase3_end - 10)
            if deadline_i - time.monotonic() < 15:
                break

            # Diverse initializations
            if i == 0:
                p_start = _perturb(a_best_norm, 0.03, rng)
                tag = f"P3-perturb-small"
            elif i == 1:
                p_start = _perturb(a_best_norm, 0.08, rng)
                tag = f"P3-perturb-med"
            elif i == 2:
                p_start = _spectral_init(n_main, rng)
                tag = f"P3-spectral-1"
            elif i == 3:
                p_start = _spectral_init(n_main, rng)
                tag = f"P3-spectral-2"
            else:
                p_start = _perturb(a_best_norm, 0.15, rng)
                tag = f"P3-perturb-large"

            a_r, _ = _adam_optimize_v2(p_start, deadline_i, lr=0.03, name=tag)
            update_best(a_r)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Larger n exploration (1500s - 1700s) ===
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 4: Larger n exploration ===")

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        for target_n in [16384, 32768]:
            if time.monotonic() > phase4_end - 80:
                break
            remaining = phase4_end - time.monotonic() - 30
            if remaining < 60:
                break

            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            if remaining > 80:
                a_r, _ = _adam_optimize_v2(
                    p_interp, time.monotonic() + remaining - 10,
                    lr=0.02, name=f"P4-n{target_n}"
                )
                update_best(a_r)
            elif remaining > 30:
                # Quick peak shave at larger n
                a_shaved, _ = _peak_shave(p_interp, n_shaves=30,
                                          deadline=time.monotonic() + remaining - 5)
                update_best(a_shaved)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish (1700s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 40:
            # Very fine Adam
            a_final, _ = _adam_optimize_v2(
                a_best_norm, time.monotonic() + remaining,
                lr=0.005, name="P5-polish"
            )
            update_best(a_final)
        elif remaining > 15:
            # Quick peak shave
            a_shaved, _ = _peak_shave(a_best_norm, n_shaves=20,
                                      deadline=time.monotonic() + remaining - 5)
            update_best(a_shaved)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
              f"t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")
        return [float(x) for x in final]

    # Fallback
    if parent is not None:
        final = np.clip(parent, 0.0, 1000.0)
    else:
        final = np.ones(1024) / 1024
    print(f"FALLBACK: score={_score(final):.10f}")
    return [float(x) for x in final]
