def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)

        grad = 2.0 * n * (h - W)  # grad of 2n*max(c) w.r.t. a (factor of 2 from self-conv)
        
        # Wait, let me recompute. The gradient of f(a) = 2n * max_k c_k
        # dc_k/da_i = 2*a_{k-i} (approximately)
        # So grad_i = 2n * sum_k w_k * 2*a_{k-i} = 4n * h_i
        # But we also need to subtract the mean for the simplex constraint
        
        # Actually for mirror descent on the simplex, the update is:
        # a_new_i = a_i * exp(-lr * grad_i) / Z
        # where grad is the full subgradient
        
        grad_full = 4.0 * n * h  # This is df/da without the W subtraction
        
        # For mirror descent, we use the raw gradient (no projection needed)
        if use_nesterov:
            beta_nd = 2.0 / (it + 2.0)
            y = (1 + beta_nd) * a - beta_nd * v
            y = np.maximum(y, 1e-300)
            y = y / y.sum()
            # Recompute grad at y
            c_y = _full_convolve(y, y)
            sh_y = (c_y - c_y.max()) / max(0.001 * c_y.max(), 1e-15)
            e_y = np.exp(sh_y)
            s_y = e_y.sum()
            if s_y < 1e-300:
                w_y = np.zeros_like(c_y)
                w_y[int(np.argmax(c_y))] = 1.0
            else:
                w_y = e_y / s_y
            h_y = _full_convolve(w_y, y[::-1])[n-1:2*n-1]
            grad_y = 4.0 * n * h_y
        else:
            y = a
            grad_y = grad_full
        
        # Mirror descent step
        exponent = -lr * grad_y
        exponent = np.clip(exponent, -30, 30)
        a_new = y * np.exp(exponent)
        sn = a_new.sum()
        if sn < 1e-300:
            a_new = y.copy()
        else:
            a_new = a_new / sn
        
        if use_nesterov:
            v = a_new - a
            a = a_new
        else:
            a = a_new
        
        # Learning rate decay
        lr = lr_init / (1.0 + 0.0005 * it)
        
        # Check score periodically
        if it % 50 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def _subgradient_avg(a_init, deadline, n_iters=50000):
    """Subgradient method with Polyak step and iterate averaging."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    # Lower bound estimate
    lb = 2.0 * n / (2.0 * n - 1.0)  # Theoretical lower bound
    
    # Running average of iterates
    avg_a = np.zeros(n)
    count = 0
    
    c0 = _full_convolve(a, a)
    max_c0 = c0.max()
    gap = max_c0 - 1.0/(2*n-1)  # Initial gap
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Compute subgradient
        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        grad = np.zeros(n)
        lo = max(0, k_max - (n-1))
        hi = min(n-1, k_max)
        for j in range(lo, hi+1):
            grad[j] = 2.0 * a[k_max - j]
            if 2*j == k_max:
                grad[j] = a[k_max - j]
        grad = 2.0 * n * grad  # Scale by 2n
        
        # Polyak step size
        # f(a) - f* ≈ 2n * (max_c - 1/(2n-1))
        max_c = float(c.max())
        gap = max_c - 1.0/(2*n-1)
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq > 1e-30:
            eta = 2.0 * gap / grad_norm_sq
        else:
            eta = 1e-6
        
        # Projected subgradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        a = a_new
        
        # Accumulate average
        avg_a += a
        count += 1
        
        # Check both current and average
        if it % 100 == 0:
            sc_avg = _score_norm_fn(avg_a / count)
            sc_curr = _score_norm_fn(a)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = (avg_a / count).copy()
            elif sc_curr < best_score:
                best_score = sc_curr
                best_a = a.copy()
    
    return best_a, best_score

def _peak_local_search(a_init, deadline, n_trials=50000):
    """Peak-targeted local search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    rng = np.random.default_rng(12345)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        max_lag = int(np.argmax(c))
        
        # Find the top contributors to the peak
        lo = max(0, max_lag - (n-1))
        hi = min(n-1, max_lag)
        pairs = np.arange(lo, hi+1)
        products = a[pairs] * a[max_lag - pairs]
        top_contrib = pairs[int(np.argmax(products))]
        
        strategy = it % 4
        a_trial = a.copy()
        
        if strategy == 0:
            # Reduce top contributor, add to random low-contributor
            delta = a[top_contrib] * rng.uniform(0.1, 0.3)
            a_trial[top_contrib] -= delta
            a_trial[top_contrib] = max(0.0, a_trial[top_contrib])
            # Add to a position far from the peak
            safe_positions = np.arange(n)
            # Positions where a[pos] * a[max_lag - pos] is small
            contribs = np.zeros(n)
            for pos in range(lo, hi+1):
                contribs[pos] = a[pos] * a[max_lag - pos]
            safe = np.argsort(contribs)[:max(3, n//10)]
            add_idx = int(safe[rng.integers(len(safe))])
            a_trial[add_idx] += delta
        elif strategy == 1:
            # Block perturbation
            bs = rng.integers(2, min(16, n//8)+1)
            start = int(rng.integers(0, n-bs+1))
            mult = rng.uniform(0.8, 1.2, size=bs)
            a_trial[start:start+bs] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
        elif strategy == 2:
            # Pair swap
            if lo < hi:
                i1 = lo + int(rng.integers(hi - lo + 1))
                i2 = max_lag - i1 if 0 <= max_lag - i1 < n else int(rng.integers(n))
                if i1 != i2:
                    swap_amt = min(a[i1], a[i2]) * rng.uniform(0.2, 0.5)
                    a_trial[i1] -= swap_amt
                    a_trial[i2] += swap_amt
                    a_trial = np.maximum(a_trial, 1e-15)
        else:
            # Gaussian perturbation
            noise = rng.normal(0, 0.02, size=n)
            a_trial = a * (1.0 + noise)
            a_trial = np.maximum(a_trial, 1e-15)
        
        s = a_trial.sum()
        if s > 1e-10:
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc
