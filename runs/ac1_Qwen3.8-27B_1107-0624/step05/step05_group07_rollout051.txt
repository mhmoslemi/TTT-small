- `conv(a,a)[n-1] = sum_{i=0}^{n-1} a_i * a_{n-1-i}`
- ...
- `conv(a,a)[2n-2] = a_{n-1}^2`

So the term at index `n-1` is `sum a_i * a_{n-1-i}`. If the sequence is symmetric, this equals `sum a_i^2`.

For a non-symmetric sequence, this can be different.

The maximum of the convolution is likely near the center (index n-1) for most sequences, but could be elsewhere.

"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses aggressive L-BFGS beta continuation, FISTA with restarts, subgradient polish on the true max,
and adaptive coordinate descent. Starts from the parent sequence (n=4096) and explores
multiple n values with diverse perturbations to escape local minima.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Returns 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Returns 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _grad_true_max(z, n):
    """Subgradient of 2*n*max(conv(softmax(z), softmax(z)))."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    j_star = int(np.argmax(c))
    # dc_j/dz_k = 2*a_k*(a_{j-k} - c_j) for valid indices
    # a_{j-k} is a[j-k] if 0 <= j-k < n, else 0
    lo = max(0, j_star - (n - 1))
    hi = min(n - 1, j_star)
    h = np.zeros(n)
    idx = np.arange(lo, hi + 1)
    h[idx] = a[j_star - idx]
    grad = 4.0 * n * a * (h - c[j_star])
    return grad


def _fista_optimize(a_init, beta, deadline, max_iter=20000, restart_interval=2000):
    """FISTA (Nesterov accelerated gradient) on the log-sum-exp objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    t_val = 1.0
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    best_obj = _logsumexp_obj(z, beta, n)

    # Compute Lipschitz constant estimate
    obj0 = _logsumexp_obj(z, beta, n)
    grad0 = _logsumexp_grad(z, beta, n)
    gnorm = np.linalg.norm(grad0)
    if gnorm < 1e-15:
        gnorm = 1.0
    # Use a conservative step size
    L = gnorm * n * 2.0
    step = 1.0 / max(L, 1e-10)
    step = min(step, 0.5)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        z_new = y + (t_val - 1) / (t_val + 1) * (y - z)
        grad = _logsumexp_grad(z_new, beta, n)
        if not np.all(np.isfinite(grad)):
            break
        z_new = z_new - step * grad
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()

        obj_new = _logsumexp_obj(z_new, beta, n)
        if not np.isfinite(obj_new) or obj_new > best_obj + 1.0:
            step *= 0.5
            if step < 1e-10:
                break
            continue
        # Successful step
        if obj_new > best_obj:
            step *= 0.8
        elif obj_new < best_obj * (1 - 1e-6):
            if step < 0.5:
                step *= 1.1

        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_val * t_val))
        z = z_new
        y = z.copy()
        t_val = t_new

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (it + 1) % restart_interval == 0:
            t_val = 1.0
            y = z.copy()

    return best_a, best_score
