    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)

        w_rev = w[::-1].copy()

        w_rev_pad[:len(w_rev)] = w_rev
        a_pad = np.zeros(N)
        a_pad[:n] = a

        fft_w = np.fft.rfft(w_pad)
        fft_w_rev = np.fft.rfft(w_rev_pad)
        
        # F1[m] = (w_rev conv a)[2n-2+m] for m=0,...,n-1
        conv1 = np.fft.irfft(fft_w_rev * fft_a, N)
        F1 = conv1[2*n-2 : 3*n-2]
        
        # F2[m] = (w conv a)[m] for m=0,...,n-1
        conv2 = np.fft.irfft(fft_w * fft_a, N)
        F2 = conv2[:n]
        
        H = F1 + F2
        offset = float(np.dot(a, H))
        grad = a * (H - offset)
        return val, grad

        lr = lr_init / (1.0 + 0.1 * p)
        
        # Run some iterations at this p
        iters_at_p = min(5000, max(100, 20000 // max(1, int(np.log2(p)))))

            t += 1
            val, grad = _lp_obj_and_grad(z, p, n)
            
            if not np.all(np.isfinite(grad)):

            # Adam update
            m_state = b1 * m_state + (1 - b1) * grad
            v_state = b2 * v_state + (1 - b2) * (grad * grad)
            mh = m_state / (1 - b1**t)

                best_val = val
                best_z = z.copy()
            
            # Learning rate decay

                lr *= 0.9
