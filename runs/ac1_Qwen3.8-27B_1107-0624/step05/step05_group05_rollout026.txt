```python
def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    N = max(N, 2)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]
```

```python
"""
Hybrid LP + FFT-gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP for large-step direction finding, FFT-based gradient descent in log-space
for efficient polishing, and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_sparse(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols, time_limit=60):
    """
    Solve LP: maximize sum(b) s.t. (a*b)[k] <= M, b >= 0.
    Return normalized direction q and the LP value M/sum(b_opt).
    """
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None, M
        b = np.maximum(res.x, 0.0)
        s = float(b.sum())
        if s < 1e-15:
            return None, M
        q = b / s
        return q, M
    except Exception:
        return None, M


def _line_search_quad(pp, pq, qq, max_tau=1.0, n_grid=100):
    """
    Find tau minimizing max_k [(1-tau)^2*pp[k] + 2*tau*(1-tau)*pq[k] + tau^2*qq[k]]
    """
    def f(tau):
        val = (1.0 - tau) ** 2 * pp + 2.0 * tau * (1.0 - tau) * pq + tau ** 2 * qq
        return float(val.max())
    
    best_tau = 0.0
    best_val = f(0.0)
    
    taus = np.linspace(0, max_tau, n_grid)
    for tau in taus:
        val = f(float(tau))
        if val < best_val:
            best_val = val
            best_tau = float(tau)
    
    # Refine
    if best_tau > 0:
        lo = max(0.0, best_tau - max_tau / n_grid)
        hi = min(max_tau, best_tau + max_tau / n_grid)
        for _ in range(30):
            mid = (lo + hi) / 2
            if f(mid) < f((lo + mid) / 2):
                lo = (lo + mid) / 2
            else:
                hi = mid
            if hi - lo < 1e-10:
                break
        best_tau = (lo + hi) / 2
        best_val = f(best_tau)
    
    return best_tau, best_val


def _fft_grad_simplex(p, beta):
    """
    Compute the projected gradient of (1/beta)*log(sum exp(beta*(p*p))) on the simplex.
    Returns grad_z for the softmax parameterization: grad_z[i] = p[i]*(grad_p[i] - <p, grad_p>)
    """
    n = len(p)
    pp = _fft_convolve(p, p)
    pp_shifted = pp - pp.max()
    exp_pp = np.exp(beta * pp_shifted)
    total = exp_pp.sum()
    if total < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(pp))] = 1.0
    else:
        w = exp_pp / total
    
    # T1[i] = sum_j p[j] * w[i+j] = (p * w_rev)[2n-2-i]
    # T2[i] = sum_j p[j] * w[i-j] = (p * w)[i]
    # T3[i] = p[i] * w[2i]
    
    N = _next_pow2(3 * n - 1)
    P = np.fft.rfft(p, N)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w[::-1], N)
    
    conv_pw = np.fft.irfft(P * W, N)
    conv_pwr = np.fft.irfft(P * Wr, N)
    
    T1 = conv_pwr[n - 1:2 * n - 1][::-1]
    T2 = conv_pw[0:n]
    T3 = p * w[0:2 * n - 1:2]
    
    grad_p = T1 + T2 - T3
    
    # Project to simplex tangent
    W_dot = float(np.dot(p, grad_p))
    grad_proj = grad_p - W_dot
    
    # Natural gradient for softmax parameterization
    grad_z = p * grad_proj
    return grad_z


def _adam_logspace(p_init, deadline, beta_seq, lr=0.01, max_iter=500000):
    """
    Adam optimizer in log-space (softmax parameterization).
    p = softmax(z), minimize max(p*p) via log-sum-exp smoothing.
    """
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-300))
    z = z - z.mean()
    best_p = p_init.copy()
    best_score = _score_norm(p_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_seq:
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1
            p = _softmax(z)
            grad_z = _fft_grad_simplex(p, beta)
            
            if not np.all(np.isfinite(grad_z)):
                t = max_iter
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = lr * mh / (np.sqrt(vh) + eps)
            
            # Clip update
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -100.0, 100.0)
            z = z - z.mean()
            
            if t % 500 == 0:
                p_check = _softmax(z)
                sc = _score_norm(p_check)
                if sc < best_score:
                    best_score = sc
                    best_p = p_check.copy()
    
    p_final = _softmax(z)
    sc_final = _score_norm(p_final)
    if sc_final < best_score:
        best_score = sc_final
        best_p = p_final.copy()
    
    return best_p, best_score


def _lp_optimize(p, rows, cols, deadline, max_iter=1000):
    """
    Iterate LP steps with line search until deadline or convergence.
    """
    n = len(p)
    best_p = p.copy()
    best_score = _score_norm(p)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        q, M = _lp_step(p, rows, cols, time_limit=30)
        if q is None:
            break
        
        # Line search
        pp = _fft_convolve(p, p)
        pq = _fft_convolve(p, q)
        qq = _fft_convolve(q, q)
        tau, _ = _line_search_quad(pp, pq, qq, max_tau=1.0, n_grid=50)
        
        if tau < 1e-6:
            # Try smaller steps
            for t_try in [0.01, 0.05, 0.1]:
                p_try = (1 - t_try) * p + t_try * q
                sc = _score_norm(p_try)
                if sc < best_score:
                    tau = t_try
                    break
        
        if tau < 1e-6:
            break
        
        p = (1.0 - tau) * p + tau * q
        p = np.maximum(p, 0.0)
        p = p / p.sum()
        
        sc = _score_norm(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()
        
        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, tau={tau:.4f}, M={M:.8f}")
    
    return best_p, best_score
```

```python
"""
Hybrid LP + FFT-gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP for principled large-step direction finding, FFT-based natural gradient
descent in log-space for efficient polishing, multi-length exploration, and
restart strategies to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_sparse(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_direction(a, rows, cols, time_limit=30):
    n = len(a)
    c = _fft_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None, M
        b = np.maximum(res.x, 0.0)
        s = float(b.sum())
        if s < 1e-15:
            return None, M
        q = b / s
        return q, M
    except Exception:
        return None, M


def _line_search_tau(p, q, n_grid=60):
    pp = _fft_convolve(p, p)
    pq = _fft_convolve(p, q)
    qq = _fft_convolve(q, q)

    def f_of_tau(tau):
        val = (1.0 - tau) ** 2 * pp + 2.0 * tau * (1.0 - tau) * pq + tau ** 2 * qq
        return float(val.max())

    best_tau = 0.0
    best_val = f_of_tau(0.0)
    taus = np.linspace(0.0, 1.0, n_grid)
    for tau in taus:
        tau_f = float(tau)
        val = f_of_tau(tau_f)
        if val < best_val:
            best_val = val
            best_tau = tau_f

    if best_tau > 0:
        step = 1.0 / n_grid
        for _ in range(40):
            improved = False
            for dt in [step, -step]:
                t_try = best_tau + dt
                if 0.0 <= t_try <= 1.0:
                    val = f_of_tau(t_try)
                    if val < best_val:
                        best_val = val
                        best_tau = t_try
                        improved = True
            if not improved:
                break
            step *= 0.5

    return best_tau, best_val


def _fft_grad_z(p, beta):
    """Compute natural gradient (w.r.t. log-space z) of smoothed max(p*p)."""
    n = len(p)
    pp = _fft_convolve(p, p)
    pp_shifted = pp - pp.max()
    exp_vals = np.exp(beta * pp_shifted)
    total = exp_vals.sum()
    if total < 1e-300 or not np.isfinite(total):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(pp))] = 1.0
    else:
        w = exp_vals / total

    N = _next_pow2(3 * n - 1)
    P = np.fft.rfft(p, N)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w[::-1], N)

    conv_pw = np.fft.irfft(P * W, N)
    conv_pwr = np.fft.irfft(P * Wr, N)

    T1 = conv_pwr[n - 1:2 * n - 1][::-1]
    T2 = conv_pw[0:n]
    T3 = p * w[0:2 * n - 1:2]

    grad_p = T1 + T2 - T3

    if not np.all(np.isfinite(grad_p)):
        return np.zeros(n)

    dot = float(np.dot(p, grad_p))
    grad_proj = grad_p - dot
    grad_z = p * grad_proj
    return grad_z


def _adam_gradient(p_init, deadline, beta_values, lr=0.05, max_iter=500000):
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-300))
    z = z - z.mean()
    best_p = p_init.copy()
    best_score = _score_norm(p_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    for bi, beta in enumerate(beta_values):
        iter_in_beta = 0
        while t < max_iter and iter_in_beta < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1
            iter_in_beta += 1
            p = _softmax(z)
            grad_z = _fft_grad_z(p, beta)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)

            update_norm = np.linalg.norm(update)
            if update_norm > 2.0:
                update = update * (2.0 / update_norm)

            z = z - update
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()

            if iter_in_beta % 1000 == 0:
                p_check = _softmax(z)
                sc = _score_norm(p_check)
                if sc < best_score:
                    best_score = sc
                    best_p = p_check.copy()
            # Adaptive LR
            if iter_in_beta % 5000 == 0 and iter_in_beta > 5000:
                sc_now = _score_norm(_softmax(z))
                if sc_now < best_score * 0.999:
                    current_lr = min(0.2, current_lr * 1.05)
                elif sc_now > best_score * 1.001:
                    current_lr = max(0.005, current_lr * 0.9)

    p_final = _softmax(z)
    sc_final = _score_norm(p_final)
    if sc_final < best_score:
        best_score = sc_final
        best_p = p_final.copy()

    return best_p, best_score


def _lp_run(p, rows, cols, deadline, max_iter=500, tag=""):
    n = len(p)
    best_p = p.copy()
    best_score = _score_norm(p)
    stall = 0
    prev_sc = best_score

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        q, M = _lp_direction(p, rows, cols, time_limit=20)
        if q is None:
            break

        tau, tau_val = _line_search_tau(p, q, n_grid=50)

        if tau < 1e-8:
            stall += 1
        else:
            p_new = (1.0 - tau) * p + tau * q
            p_new = np.maximum(p_new, 0.0)
            p_new = p_new / p_new.sum()
            sc = _score_norm(p_new)
            if sc < prev_sc - 1e-12:
                stall = 0
                p = p_new
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
            else:
                stall += 1

        if it % 20 == 0:
            print(f"    {tag} iter {it}: score={_score_norm(p):.10f} tau={tau:.4f} stall={stall} t={time.monotonic()-time.time()+time.time():.0f}s")
            prev_sc = _score_norm(p)

        if stall > 50:
            print(f"    {tag} stalled at iter {it}")
            break
        prev_sc = _score_norm(p)

    return best_p, best_score


def _perturb_log(p, sigma, rng):
    z = np.log(np.maximum(p, 1e-300))
    z = z + sigma * rng.standard_normal(len(p))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
    else:
        parent = np.ones(2048) / 2048
        print("No parent, using uniform n=2048")

    n_parent = len(parent)

    # Phase 0: Quick multi-length evaluation
    phase0_end = min(deadline, t_start + 90.0)
    candidate_ns = [1024, 2048, 4096, 8192]
    length_results = {}

    for n_try in candidate_ns:
        if time.monotonic() > phase0_end - (30 if n_try <= 4096 else 60):
            break
        x0 = np.linspace(0, 1, n_parent)
        x1 = np.linspace(0, 1, n_try)
        p_try = np.interp(x1, x0, parent)
        p_try = np.maximum(p_try, 1e-15)
        p_try = p_try / p_try.sum()

        sc0 = _score_norm(p_try)
        print(f"  Phase0 n={n_try}: init score={sc0:.10f}")

        # Quick LP: 5 iterations
        if n_try <= 4096:
            rows_t, cols_t = _build_lp_sparse(n_try)
            for _ in range(5):
                if time.monotonic() > phase0_end - 5:
                    break
                q, _ = _lp_direction(p_try, rows_t, cols_t, time_limit=8)
                if q is None:
                    break
                tau, _ = _line_search_tau(p_try, q, n_grid=30)
                if tau < 1e-8:
                    break
                p_try = np.maximum((1.0 - tau) * p_try + tau * q, 0.0)
                p_try = p_try / p_try.sum()

        sc1 = _score_norm(p_try)
        length_results[n_try] = (sc1, p_try.copy())
        print(f"  Phase0 n={n_try}: after LP score={sc1:.10f}")

        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_seq = p_try.copy()

    # Choose best length
    if length_results:
        best_n = min(length_results, key=lambda k: length_results[k][0])
    else:
        best_n = 2048
    print(f"Best length from Phase 0: n={best_n}, score={length_results.get(best_n, (best_score,))[0]:.10f}")
    print(f"Phase 0 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Set up main optimization on best_n
    n = best_n
    if best_n in length_results:
        p = length_results[best_n][1].copy()
    else:
        x0 = np.linspace(0, 1, n_parent)
        x1 = np.linspace(0, 1, n)
        p = np.interp(x1, x0, parent)
        p = np.maximum(p, 1e-15)
        p = p / p.sum()

    rows, cols = _build_lp_sparse(n)
    print(f"Main phase: n={n}, initial score={_score_norm(p):.10f}")

    # Phase 1: Extended LP optimization
    phase1_end = min(deadline, t_start + 700.0)
    p, sc1 = _lp_run(p, rows, cols, phase1_end, max_iter=400, tag="LP1")
    if sc1 < best_score:
        best_score = sc1
        best_seq = p.copy()
    print(f"Phase 1 done: score={sc1:.10f}, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: FFT gradient descent in log-space
    phase2_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase2_end:
        p_start = best_seq.copy()
        c_best = _fft_convolve(p_start, p_start)
        max_c = float(c_best.max())

        beta_values = [
            0.05 * max_c, 0.03 * max_c, 0.015 * max_c, 0.008 * max_c,
            0.004 * max_c, 0.002 * max_c, 0.001 * max_c,
            0.0005 * max_c, 0.0002 * max_c, 0.0001 * max_c,
            0.00005 * max_c,
        ]

        p_grad, sc_grad = _adam_gradient(
            p_start, phase2_end, beta_values, lr=0.04, max_iter=800000
        )
        full_sc = _score(p_grad)
        print(f"Phase 2 gradient: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = p_grad.copy()

    # Phase 3: More LP from gradient-improved point
    phase3_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase3_end and len(best_seq) == n:
        p, sc3 = _lp_run(best_seq.copy(), rows, cols, phase3_end, max_iter=300, tag="LP2")
        if sc3 < best_score:
            best_score = sc3
            best_seq = p.copy()
        print(f"Phase 3 LP: score={sc3:.10f}, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Restart + LP from perturbed points
    phase4_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase4_end:
        for attempt in range(4):
            if time.monotonic() > phase4_end - 20:
                break
            sigma = 0.02 + 0.015 * attempt
            p_pert = _perturb_log(best_seq, sigma, rng)
            p_res, sc_res = _lp_run(p_pert, rows, cols, min(phase4_end - 5, time.monotonic() + 30),
                                     max_iter=30, tag=f"R{attempt}")
            full_sc = _score(p_res)
            print(f"  Restart {attempt} (sigma={sigma:.3f}): score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_res.copy()

    # Phase 5: Final fine gradient
    if time.monotonic() < deadline - 10:
        p_start = best_seq.copy()
        c_best = _fft_convolve(p_start, p_start)
        max_c = float(c_best.max())
        beta_fine = [
            0.00005 * max_c, 0.00002 * max_c, 0.00001 * max_c,
            0.000005 * max_c, 0.000002 * max_c,
        ]
        p_final, sc_final = _adam_gradient(
            p_start, deadline - 5, beta_fine, lr=0.015, max_iter=200000
        )
        full_sc = _score(p_final)
        print(f"Phase 5 fine: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = p_final.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
