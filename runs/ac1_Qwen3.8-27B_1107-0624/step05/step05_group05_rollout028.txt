    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=120):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, M_scale=1.0, time_limit=120):
    c = _full_convolve(f, f)
    M = float(c.max()) * M_scale
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, float(c.max())
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.02, 0.05, 0.08]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, float(c.max())


def _adam_optimize(a_init, deadline, beta_start, beta_end, lr_init, max_iter,
                   mom_reset_interval=5000, clip_range=50.0, score_interval=500):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    prev_best = best_score
    no_improve_count = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        if c_max < 1e-300:
            break

        beta = beta_start * (max(beta_end, 1e-300) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-300)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -clip_range, clip_range)
        z = z - z.mean()

        if t % mom_reset_interval == 0:
            m_buf = np.zeros(n)

        if t % score_interval == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += score_interval

            if no_improve_count > 20000:
                if best_score < prev_best * 0.9999:
                    current_lr = min(0.1, current_lr * 1.05)
                else:
                    current_lr = max(0.001, current_lr * 0.9)
                prev_best = best_score
                no_improve_count = 0
                m_buf = np.zeros(n)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_multistage(a_init, deadline, stages):
    """
    Multi-stage Adam with different beta ranges per stage.
    stages: list of (beta_start, beta_end, lr, max_iter, time_frac)
    """
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    total_time = deadline - time.monotonic()

    for beta_s, beta_e, lr, max_it, tf in stages:
        if time.monotonic() > deadline:
            break
        stage_deadline = min(deadline, time.monotonic() + total_time * tf)
        a, sc = _adam_optimize(best_a, stage_deadline, beta_s, beta_e, lr, max_it)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"  Stage beta=[{beta_s:.2e},{beta_e:.2e}] lr={lr}: score={sc:.10f} t={time.monotonic()-t_start_global:.0f}s")

    return best_a, best_score


def _upsample_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb_structured(a, rng, strength=0.05):
    """Perturb by moving mass from peak contributors to random positions."""
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    # Find top contributors to the peak
    a_rev = a[::-1]
    shift = k_star - (n - 1)
    if shift >= 0:
        contrib = a[:n - shift] * a_rev[shift:]
    else:
        contrib = a[-shift:] * a_rev[:n + shift]
    top_idx = np.argsort(contrib)[-max(1, n // 50):]
    # Random target indices
    n_targets = len(top_idx)
    target_idx = rng.integers(0, n, size=n_targets)
    delta = strength * a[top_idx]
    a_new = a.copy()
    a_new[top_idx] -= delta
    a_new[target_idx] += delta
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

def propose_candidate(seed=42, budget_s=1800):
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        best_n = n
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        best_n = n
        print(f"Uniform init: n={n}, score={init_sc:.10f}")

    # Phase 1: LP optimization (0 - 350s)
    phase1_end = min(deadline, t_start_global + 350.0)
    print(f"Phase 1: LP optimization, t={time.monotonic()-t_start_global:.0f}s")
    rows, cols = _build_lp_indices(n)
    print(f"  LP indices: {len(rows)} nonzeros")

    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0
    m_scale = 1.0

    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a, M = _lp_step(a, rows, cols, M_scale=m_scale, time_limit=90)
        sc = _score_norm(a)
        lp_count += 1
        dt_lp = time.monotonic() - t0

        if sc < prev_sc - 1e-12:
            stall = 0
        else:
            stall += 1

        if stall > 20 and m_scale > 0.85:
            m_scale = max(0.85, m_scale - 0.05)
            stall = 0
            print(f"  Aggressing LP: M_scale={m_scale:.2f}")

        if lp_count % 3 == 0 or stall > 30:
            full_sc = _score(a)
            print(f"  LP[{lp_count}] score={full_sc:.10f} M={M:.8f} dt={dt_lp:.1f}s stall={stall} t={time.monotonic()-t_start_global:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
                best_n = n

        if stall > 50:
            print(f"  LP stalled at iter {lp_count}")
            break
        prev_sc = sc

    full_sc = _score(a)
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
        best_n = n
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start_global:.0f}s")

    # Phase 2: Adam optimization on n=2048 (350 - 900s)
    phase2_end = min(deadline, t_start_global + 900.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Adam on n={n}, t={time.monotonic()-t_start_global:.0f}s")
        a_work = best_seq.copy()
        c_ref = _full_convolve(a_work, a_work)
        max_c = float(c_ref.max())

        stages = [
            (0.01 * max_c, 0.001 * max_c, 0.05, 500000, 0.3),
            (0.001 * max_c, 0.0001 * max_c, 0.03, 500000, 0.3),
            (0.0001 * max_c, 0.00001 * max_c, 0.015, 500000, 0.25),
            (0.00001 * max_c, 0.000001 * max_c, 0.008, 500000, 0.15),
        ]
        a_work, sc = _adam_multistage(a_work, phase2_end, stages)
        full_sc = _score(a_work)
        print(f"Phase 2 result: score={full_sc:.10f}, t={time.monotonic()-t_start_global:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_work.copy()
            best_n = n

    # Phase 3: Try n=4096 (900 - 1400s)
    phase3_end = min(deadline, t_start_global + 1400.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Try n=4096, t={time.monotonic()-t_start_global:.0f}s")
        n_big = 4096
        a_big = _upsample_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Upsampled score: {sc_big:.10f}")

        if sc_big < best_score + 0.5:  # Only worth trying if not too bad
            c_big = _full_convolve(a_big, a_big)
            max_cb = float(c_big.max())
            stages_big = [
                (0.01 * max_cb, 0.0005 * max_cb, 0.04, 800000, 0.4),
                (0.0005 * max_cb, 0.00005 * max_cb, 0.02, 800000, 0.35),
                (0.00005 * max_cb, 0.000005 * max_cb, 0.01, 800000, 0.25),
            ]
            a_big, sc_big = _adam_multistage(a_big, phase3_end - 5, stages_big)
            full_sc_big = _score(a_big)
            print(f"  n=4096 result: score={full_sc_big:.10f}, t={time.monotonic()-t_start_global:.0f}s")
            if np.isfinite(full_sc_big) and full_sc_big < best_score:
                best_score = full_sc_big
                best_seq = a_big.copy()
                best_n = n_big

    # Phase 4: Structured perturbation + quick Adam (1400 - 1650s)
    phase4_end = min(deadline, t_start_global + 1650.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Perturbation search, t={time.monotonic()-t_start_global:.0f}s")
        n_cur = len(best_seq)
        for attempt in range(5):
            if time.monotonic() > phase4_end - 30:
                break
            # Different perturbation strategies
            if attempt == 0:
                a_pert = _perturb_structured(best_seq, rng, 0.03)
            elif attempt == 1:
                a_pert = _perturb_structured(best_seq, rng, 0.08)
            elif attempt == 2:
                z = np.log(np.maximum(best_seq, 1e-300)) + 0.05 * rng.standard_normal(n_cur)
                a_pert = _softmax(z - np.mean(z))
            elif attempt == 3:
                a_pert = _perturb_structured(best_seq, rng, 0.01)
            else:
                # Bipartite perturbation: swap front/back
                half = n_cur // 2
                a_pert = best_seq.copy()
                swap_amt = 0.02 * a_pert[:half].sum()
                a_pert[:half] -= swap_amt / half
                a_pert[half:] += swap_amt / half
                a_pert = np.maximum(a_pert, 0)
                a_pert = a_pert / a_pert.sum()

            # Quick Adam on perturbed
            c_p = _full_convolve(a_pert, a_pert)
            max_cp = float(c_p.max())
            stages_p = [
                (0.005 * max_cp, 0.0001 * max_cp, 0.03, 200000, 0.6),
                (0.0001 * max_cp, 0.00001 * max_cp, 0.01, 200000, 0.4),
            ]
            a_pert, sc_p = _adam_multistage(a_pert, min(phase4_end - 10, time.monotonic() + 40), stages_p)
            full_sc_p = _score(a_pert)
            print(f"  Pert[{attempt}]: score={full_sc_p:.10f}, t={time.monotonic()-t_start_global:.0f}s")
            if np.isfinite(full_sc_p) and full_sc_p < best_score:
                best_score = full_sc_p
                best_seq = a_pert.copy()
                best_n = len(a_pert)

    # Phase 5: Final ultra-fine Adam (1650 - deadline)
    if time.monotonic() < deadline - 15:
        print(f"Phase 5: Final polish, t={time.monotonic()-t_start_global:.0f}s")
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        max_cf = float(c_fin.max())
        deadline5 = deadline - 10
        a_fin, sc_fin = _adam_optimize(
            best_seq, deadline5,
            beta_start=0.00001 * max_cf,
            beta_end=0.0000001 * max_cf,
            lr_init=0.005,
            max_iter=5000000,
            mom_reset_interval=10000,
            clip_range=40.0,
            score_interval=1000
        )
        full_sc_fin = _score(a_fin)
        print(f"  Final Adam: score={full_sc_fin:.10f}, t={time.monotonic()-t_start_global:.0f}s")
        if np.isfinite(full_sc_fin) and full_sc_fin < best_score:
            best_score = full_sc_fin
            best_seq = a_fin.copy()
            best_n = n_fin

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
