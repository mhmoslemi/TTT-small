Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based direction finding, L-BFGS with log-sum-exp smoothing,
and active-set subgradient polishing.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, dia_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(f, n):
    """Build matrix A where (A*b)_k = conv(f,b)_k for k=0..2n-2."""
    m = 2 * n - 1
    # A[k,j] = f[j] if 0 <= j < n and 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A

"""
Hybrid LP + L-BFGS optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based active-set direction finding, log-sum-exp smoothed L-BFGS with
beta-continuation, subgradient polishing, and multi-restart diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0.0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f, n):
    """Build dense matrix A where (A*b)[k] = conv(f,b)[k] for k=0..2n-2."""
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        end = min(j + n, m)
        A[j:end, j] = f[j]
    return A


def _lp_direction(f, M, deadline):
    """Solve LP: max sum(b) s.t. conv(f,b)_k <= M for all k, b >= 0."""
    n = len(f)
    m = 2 * n - 1
    A = _build_conv_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(m, M)
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0.0, None)] * n,
                         method='highs',
                         options={'time_limit': max(1.0, deadline - time.monotonic())})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, rel_tol=0.005):
    """Gradient using exact active set with relative tolerance."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    threshold = c_max * (1.0 - rel_tol)
    active = c >= threshold
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter=2000):
    """Run L-BFGS-B optimization for one beta phase."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-15,
                                   'gtol': 1e-11, 'maxcor': 40, 'maxls': 200,
                                   'disp': False})
        return _softmax(result.x), result.success
    except Exception:
        return a_init.copy(), False


def _subgrad_polish(a_init, deadline, max_steps=5000, lr_init=0.04, momentum=0.9):
    """Subgradient method with momentum in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(a, c, n, rel_tol=0.005)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step = lr / (1.0 + 0.002 * i)
        v = momentum * v - step * grad
        z = z + v
        z = np.clip(z, -70.0, 70.0)
        z = z - z.mean()

            a_chk = _softmax(z)
            sc = _score(a_chk)
            if sc < best_sc:

                best_a = a_chk.copy()
                lr *= 1.01
            else:

                v *= 0.5

    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_schedule(a_start, deadline, min_remaining=150.0):
    """Full L-BFGS beta-continuation schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001,

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:

    return _softmax(z)

                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            else:
                p_start = _perturb_z(best_seq, 0.05, rng)

            sc_s = _score(p_start)
            print(f"  Restart {i}: start={sc_s:.10f}")

            a_r = _lbfgs_schedule(p_start, deadline, min_remaining=30.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if time.monotonic() < deadline - 20 and sc_r < best_score + 0.005:
                a_r2, sc_r2 = _subgrad_polish(a_r, deadline - 20,
                                              max_steps=300, lr_init=0.015, momentum=0.85)
                update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # === Phase 7: Final fine-tuning ===
    print(f"\nPhase 7: Final tuning (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 1e-06
        a_fin, _ = _lbfgs_run(best_seq, beta_final, n, max_iter=1500)
        update_best(a_fin, _score(a_fin), "final_lbfgs")

        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2,
                                              max_steps=300, lr_init=0.008, momentum=0.9)
            update_best(a_fin2, sc_fin2, "final_sg")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
