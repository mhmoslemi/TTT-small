"""
Nesterov-accelerated gradient descent with logsumexp continuation for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Key innovations:
1. FFT-based O(n log n) gradient computation via cross-correlation
2. Automatic palindromic symmetry preservation through softmax parameterization
3. Nesterov momentum (faster than L-BFGS for this high-dimensional smooth problem)
4. Multi-stage aggressive beta continuation
5. Peak-specific surgical moves for final polish
6. Multi-initialization strategy with diverse profile families
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _compute_grad_and_info(a, beta):
    """Compute gradient of smoothed objective w.r.t. a.
    F_smooth = 2n * beta * logsumexp(c / beta)
    Returns: (g_a, max_c, smooth_value)"""
    n = len(a)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    c_shifted = (c - max_c) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        smooth_val = max_c
    else:
        w = e / s
        smooth_val = max_c + beta * float(np.log(s))

    W = float(np.dot(w, c))

    # h_i = sum_k w_k * a_{k-i} via FFT cross-correlation
    N = _next_pow2(3 * n)
    W_fft = np.fft.fft(w, N)
    A_fft = np.fft.fft(a, N)
    h_full = np.fft.ifft(W_fft * np.conj(A_fft))[:n].real

    g_a = 4.0 * n * a * (h_full - W)
    return g_a, max_c, smooth_val


def _softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_bspline_profile(n, order):
    t = np.linspace(0.0, 1.0, n)
    u = np.clip(t * 0.95 + 0.025, 0.01, 0.99)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_like(n, sigma=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-(t - 0.5) ** 2 / (2.0 * sigma ** 2))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_bimodal(n, sep=0.3, width=0.12):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2
    c2 = 0.5 + sep / 2
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _nesterov_optimize(a_init, deadline, beta_start_frac=0.1, n_beta_levels=20,
                        lr_start=0.1, momentum=0.9, iters_per_level=5000,
                        momentum_anneal=False):
    """Nesterov accelerated gradient descent with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    v = np.zeros(n)

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    # Compute initial max_c for scaling beta
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Beta schedule: geometric decay
    beta_end_frac = 1e-6
    ratio = (beta_end_frac / beta_start_frac) ** (1.0 / max(1, n_beta_levels - 1))
    betas = [beta_start_frac * max_c0 * (ratio ** i) for i in range(n_beta_levels)]

    total_iters = 0
    for level in range(n_beta_levels):
        if time.monotonic() > deadline:
            break
        beta = betas[level]

        # Adaptive learning rate: decrease as beta decreases
        progress = level / max(1, n_beta_levels - 1)
        lr = lr_start * (1.0 - progress) ** 1.5
        lr = max(lr, lr_start * 0.01)

        # Adaptive momentum: increase slightly over time
        mu = momentum + 0.05 * progress if momentum_anneal else momentum

        for it in range(iters_per_level):
            if time.monotonic() > deadline:
                break

            a = _softmax_from_z(z)
            g_a, max_c, smooth_val = _compute_grad_and_info(a, beta)
            scalar = float(np.dot(g_a, a))
            g_z = a * (g_a - scalar)

            # Nesterov update (heavy ball variant)
            v = mu * v + lr * g_z
            z = z - v
            z = z - z.mean()
            total_iters += 1

            # Periodic evaluation
            if (it + 1) % 200 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        # Update z for next level
        pass  # z is already updated

    return best_a, best_score, total_iters


def _nesterov_adaptive(a_init, deadline, beta_start_frac=0.05):
    """Nesterov with adaptive learning rate and stopping criteria."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    v = np.zeros(n)

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Multi-stage beta schedule
    beta_fracs = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                  0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
                  0.000005, 0.000002, 0.000001, 5e-7, 2e-7, 1e-7,
                  5e-8, 2e-8, 1e-8, 5e-9, 2e-9, 1e-9, 5e-10,
                  2e-10, 1e-10, 5e-11, 2e-11, 1e-11]

    lr_base = 0.15
    momentum = 0.91
    best_smooth = np.inf
    stuck_count = 0

    for bi, bfrac in enumerate(beta_fracs):
        if time.monotonic() > deadline:
            break
        beta = bfrac * max_c0

        # Learning rate for this stage
        lr = lr_base * max(0.05, 1.0 - bi / len(beta_fracs))

        # Number of iterations: more for early stages, fewer for late
        if bi < 5:
            n_iters = 8000
        elif bi < 15:
            n_iters = 5000
        else:
            n_iters = 3000

        prev_best = best_score
        for it in range(n_iters):
            if time.monotonic() > deadline:
                break

            a = _softmax_from_z(z)
            g_a, max_c, smooth_val = _compute_grad_and_info(a, beta)
            scalar = float(np.dot(g_a, a))
            g_z = a * (g_a - scalar)

            # Clip gradient to prevent explosions
            grad_norm = np.linalg.norm(g_z)
            if grad_norm > 1e6:
                g_z = g_z * (1e6 / grad_norm)

            v = momentum * v + lr * g_z
            z = z - v
            z = z - z.mean()

            # Evaluate every 100 steps
            if (it + 1) % 100 == 0:
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    stuck_count = 0
                else:
                    stuck_count += 1

                # Adaptive: if stuck, reduce lr
                if stuck_count > 50:
                    lr *= 0.7
                    v *= 0.5
                    stuck_count = 0

    return best_a, best_score


def _surgical_peak_fix(a_init, deadline, n_rounds=50000):
    """Directly reduce peaks by adjusting contributing elements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(12345)

    for it in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        max_c = float(c.max())
        max_k = int(np.argmax(c))

        # Find contributing pairs at the peak lag
        i_min = max(0, max_k - (n - 1))
        i_max = min(n - 1, max_k)
        if i_min > i_max:
            continue

        indices = np.arange(i_min, i_max + 1)
        partners = max_k - indices
        products = a[indices] * a[partners]
        total_product = products.sum()

        if total_product < 1e-30:
            continue

        # Weight: contribution fraction of each pair
        weights = products / total_product

        # Find "safe" positions: those with low contribution to top peaks
        top_k_count = min(5, len(c))
        top_lags = np.argsort(c)[-top_k_count:]
        contrib = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            lo = max(0, kl - (n - 1))
            hi = min(n - 1, kl)
            for j in range(lo, hi + 1):
                p = kl - j
                if 0 <= p < n:
                    contrib[j] += a[p]

        # Strategy: reduce high-product pairs, add to safe positions
        # Pick the largest contributing pair
        idx = indices[int(np.argmax(weights))]
        partner = max_k - idx

        a_trial = a.copy()
        # Reduce the larger of the pair
        if a_trial[idx] >= a_trial[partner]:
            reduce_idx = idx
            keep_idx = partner
        else:
            reduce_idx = partner
            keep_idx = idx

        frac = rng.uniform(0.05, 0.25)
        delta = a_trial[reduce_idx] * frac
        a_trial[reduce_idx] -= delta

        # Add to a safe position (low contribution)
        safe_mask = contrib < np.median(contrib)
        safe_indices = np.where(safe_mask)[0]
        if len(safe_indices) == 0:
            safe_indices = np.arange(n)
        # Weight by inverse contribution
        safe_weights = 1.0 / (contrib[safe_indices] + 1e-10)
        safe_weights /= safe_weights.sum()
        add_idx = safe_indices[rng.choice(len(safe_indices), p=safe_weights)]
        a_trial[add_idx] += delta

        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        sc = _score_norm(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_sc


def _local_refinement(a_init, deadline, rng, n_trials=100000):
    """Local search with multiplicative perturbations and pair swaps."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_trials):
        if time.monotonic() > deadline:
            break

        a_trial = a.copy()
        strategy = it % 4

        if strategy == 0:
            # Small multiplicative perturbation on a random block
            block_size = int(rng.integers(3, min(32, n // 8)))
            start = int(rng.integers(0, n - block_size))
            mult = rng.uniform(0.85, 1.15, size=block_size)
            a_trial[start:start + block_size] *= mult
        elif strategy == 1:
            # Pair mass exchange
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            frac = rng.uniform(0.05, 0.3)
            d1 = a_trial[i1] * frac
            d2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - d1) + d2
            a_trial[i2] = max(0.0, a_trial[i2] - d2) + d1
        elif strategy == 2:
            # Gaussian-like perturbation in log space
            z = np.log(np.maximum(a_trial, 1e-15))
            z += 0.01 * rng.normal(size=n)
            z -= z.mean()
            a_trial = _softmax_from_z(z)
        else:
            # Targeted: reduce at peak-contributing positions, add elsewhere
            c = _full_convolve(a, a)
            peak_k = int(np.argmax(c))
            i_min = max(0, peak_k - (n - 1))
            i_max = min(n - 1, peak_k)
            if i_min <= i_max:
                indices = np.arange(i_min, i_max + 1)
                partners = peak_k - indices
                products = a[indices] * a[partners]
                top_idx = indices[int(np.argmax(products))]
                a_trial[top_idx] *= rng.uniform(0.7, 0.9)
                # Add to a random position far from the peak
                add_idx = int(rng.integers(n))
                a_trial[add_idx] += a_trial[top_idx] * 0.1  # partial compensation

        a_trial = np.maximum(a_trial, 1e-15)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s

        sc = _score_norm(a_trial)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_sc


def _try_different_n(a_best, deadline, rng):
    """Try interpolating to different lengths and re-optimizing briefly."""
    n_curr = len(a_best)
    best_a = a_best.copy()
    best_sc = _score_norm(a_best)

    candidate_ns = []
    for factor in [0.5, 1.5, 2]:
        target = int(n_curr * factor)
        target = max(1024, min(16384, _next_pow2(target)))
        if target != n_curr and target not in candidate_ns:
            candidate_ns.append(target)

    for target_n in candidate_ns:
        if time.monotonic() > deadline - 30:
            break
        # Interpolate
        x0 = np.linspace(0.0, 1.0, n_curr)
        x1 = np.linspace(0.0, 1.0, target_n)
        a_interp = np.interp(x1, x0, a_best)
        a_interp = np.maximum(a_interp, 1e-15)
        a_interp /= a_interp.sum()

        sc_interp = _score_norm(a_interp)
        if sc_interp < best_sc:
            best_sc = sc_interp
            best_a = a_interp.copy()

        # Brief optimization at new n
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 30, time.monotonic() + 60)
            a_opt, sc_opt = _nesterov_optimize(a_interp, dl,
                                                beta_start_frac=0.01,
                                                n_beta_levels=8,
                                                lr_start=0.08,
                                                iters_per_level=2000)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()

    return best_a, best_sc
