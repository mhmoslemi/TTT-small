        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _compute_conv_and_grad(a, beta):
    """
    Compute convolution and gradient of logsumexp(beta * conv) w.r.t. a.
    Returns (conv, grad, logsumexp_value).
    """
    n = len(a)
    conv = _full_convolve(a, a)  # length 2n-1
    cm = float(conv.max())
    
    if beta < 1e-15:
        # Just return the max gradient
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        c_shifted = (conv - cm) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = e / s
    
    logsumexp_val = cm + beta * float(np.log(w.sum() * (1.0 / len(conv))))  # Hmm, need to be careful
    
    # Actually, logsumexp = log(sum exp(beta * conv_k))
    # = log(sum exp(beta * (conv_k - cm))) + beta * cm
    # = log(sum w_k) + beta * cm  (since w_k = exp(beta*(conv_k-cm)) / sum exp(...))
    # = 0 + beta * cm + log((2n-1)) ... no wait
    # logsumexp(beta * conv) = log(sum_k exp(beta * conv_k))
    # = log(sum_k exp(beta * (conv_k - cm))) + beta * cm
    # Let S = sum_k exp(beta * (conv_k - cm))
    # = log(S) + beta * cm
    # And w_k = exp(beta*(conv_k-cm)) / S, so S = sum_k exp(beta*(conv_k-cm))
    
    # For the gradient, we need: d/dj [log(S) + beta*cm] = sum_k (d conv_k / da_j) * beta * exp(beta*conv_k) / sum_l exp(beta*conv_l)
    # = beta * sum_k w_k * (d conv_k / da_j) * ... 
    # Hmm, w_k = exp(beta*(conv_k-cm)) / S, and exp(beta*conv_k) / sum exp(beta*conv_l) = exp(beta*(conv_k-cm)) / sum exp(beta*(conv_l-cm)) = w_k.
    # So the gradient is beta * sum_k w_k * (d conv_k / da_j).
    
    # Now compute the gradient efficiently using FFT.
    # g_j = beta * sum_k w_k * (a_{k-j} + a_{j-k}) [with a_i = 0 for i outside [0,n-1]]
    
    # Method: 
    # Term 1: sum_k w_k * a_{k-j} for j = 0,...,n-1
    #   = cross-correlation of w and a, evaluated at lag j
    #   Using FFT: IFFT(conj(FFT(w)) * FFT(a)) gives cross-correlation
    #   But w has length 2n-1 and a has length n.
    
    # Let me use a different approach:
    # g_j = beta * [sum_k w_k * a_{k-j} + sum_k w_k * a_{j-k}]
    
    # First sum: for fixed j, sum over k from j to j+n-1 of w[k] * a[k-j]
    #   = dot(w[j:j+n], a)  (with clipping)
    
    # Second sum: for fixed j, sum over k from max(0, j-n+1) to min(j, 2n-2) of w[k] * a[j-k]
    #   = dot(w[max(0,j-n+1):min(j,2n-2)+1], a[j-(min(j,2n-2)+-1) : j-max(0,j-n+1)+1])
    #   This is the j-th element of the convolution of w and a (full convolution).
    
    # Full convolution of w (len 2n-1) and a (len n) has length 3n-2.
    # conv_wa[k] = sum_m w[m] * a[k-m]
    # For k = j (j in [0, n-1]): conv_wa[j] = sum_{m=0}^{j} w[m] * a[j-m] = sum_m w[m] * a[j-m]
    #   which is the second sum! ✓
    
    # For the first sum: sum_k w[k] * a[k-j] = sum_m w[m+j] * a[m] (reindex m = k-j)
    #   = dot(a, w[j:j+n]) ... this is the j-th element of the "anti-convolution"
    #   = (a * reverse(w))[2n-2-j]
    #   Where reverse(w)[k] = w[2n-2-k]
    
    # So: g_j = beta * [conv_a_wrev[2n-2-j] + conv_w_a[j]]
    # where conv_a_wrev = full_convolve(a, w[::-1]) and conv_w_a = full_convolve(w, a)
    # which equals full_convolve(a, w) by commutativity.
    
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w[::-1], N)
    
    conv_w_a = np.fft.irfft(A * W, N)[:3*n-2]  # full convolve of a and w
    conv_a_wr = np.fft.irfft(A * Wr, N)[:3*n-2]  # full convolve of a and w_rev
    
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = beta * (conv_w_a[j] + conv_a_wr[2*n-2-j])
    
    return conv, grad, cm

# = beta * (conv_w_a[0:n] + conv_a_wr[2n-2 : 2n-2-n+1 : -1])

def _compute_conv_and_grad_v2(a, beta):
    n = len(a)
    conv = _full_convolve(a, a)
    cm = float(conv.max())
    
    if beta < 1e-15:
        w = np.zeros(len(conv))
        w[int(np.argmax(conv))] = 1.0
    else:
        c_shifted = (conv - cm) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w = e / s
    
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w[::-1], N)
    
    conv_w_a = np.fft.irfft(A * W, N)[:3*n-2]
    conv_a_wr = np.fft.irfft(A * Wr, N)[:3*n-2]
    
    grad = beta * (conv_w_a[:n] + conv_a_wr[n-1:2*n-1][::-1])
    
    return conv, grad, cm

def _softmax_grad(z, a, grad_a):
    """Convert gradient w.r.t. a to gradient w.r.t. z (softmax parameterization)."""
    return grad_a * a - a * np.dot(grad_a, a)
