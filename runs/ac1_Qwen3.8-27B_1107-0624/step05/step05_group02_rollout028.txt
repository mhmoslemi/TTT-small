import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, beta):
    """Compute gradient of LSE(conv(a,a), beta) w.r.t. log(a) parameterization."""
    c = _full_convolve(a, a)
    # w = softmax(beta * c)
    c_shifted = beta * (c - c.max())
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0

        w = e / s

    wc = _full_convolve(w, a_rev)
    h = wc[len(a)-1 : 2*len(a)-1]
    W = float(np.dot(w, c))
    grad = a * (h - W)
    return grad, c, W

FFT-based gradient descent with log-sum-exp smoothing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses multi-beta annealing, Adam optimizer,
analytical gradients via cross-correlation, and multi-restart strategy.
"""
import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, beta):
    """Compute gradient of LSE(conv(a,a), beta) w.r.t. log-softmax params.
    Returns (grad, c, cmax)."""
    n = len(a)
    c = _full_convolve(a, a)
    cmax = float(c.max())
    c_shifted = beta * (c - cmax)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    grad = a * (h - W)
    return grad, c, cmax

def _gaussian_init(n, width):
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = np.exp(-((x - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _raised_cosine_init(n):
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _uniform_init(n):
    return np.ones(n, dtype=np.float64) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
