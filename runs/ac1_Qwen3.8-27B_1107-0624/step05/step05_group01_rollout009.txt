def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):

    c = _full_convolve(a, a)
    cm = float(c.max())

    max_s = float(shifted.max())

    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]

    grad = 4.0 * n * a * (h - W)

def _build_lp_matrix(a, peak_indices, n):

        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat

def _subgradient_momentum(a_init, deadline, n_steps=50000):
    """Subgradient method with heavy ball momentum for the exact max."""
    n = len(a_init)
    S = float(a_init.sum())
    a_cur = a_init.copy()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    mu = 0.9  # momentum
    lr = 0.01

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        
        # Find all peaks at max (within numerical tolerance)
        tol = 1e-10 * cm
        active = np.where(c >= cm - tol)[0]

        # Compute subgradient: average of gradients at active peaks
        grad = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                seg = a_cur[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
                grad[i_lo:i_hi + 1] += 2.0 * seg

        grad /= len(active)
        
        # Center the gradient (remove mean to preserve sum)
        grad = grad - grad.mean()

        # Heavy ball update
        v = mu * v - lr * grad
        a_new = a_cur + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new * (S / s_new)
        else:
            break

        a_cur = a_new

        # Decay learning rate
        if it % 1000 == 999:
            lr *= 0.8

        # Check improvement periodically
        if it % 50 == 49:
            sc = _score_norm(a_cur)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()

    return best_a, best_score

def _coord_descent(a_init, deadline, n_rounds=200):

                break

                    break

            for trial in range(5):

                d = d - d.mean()  # preserve sum approximately

                    s_cur = a_cur.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    if time.monotonic() < deadline - 30:
        c_b = _full_convolve(best_a, best_a)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]]
        lbfgs_time = min(100, (deadline - time.monotonic()) * 0.3)
        a_f, sc_f = _run_lbfgs(best_a, time.monotonic() + lbfgs_time,
                                beta_list=tiny_betas, maxiter_per_beta=8000)
        if sc_f < best_score:
            best_a, best_score = a_f, sc_f
            print(f"    L-BFGS improved to {best_score:.10f}")

    # Phase D: Subgradient momentum
    if time.monotonic() < deadline - 20:
        sg_time = min(60, (deadline - time.monotonic()) * 0.3)
        a_sg, sc_sg = _subgradient_momentum(best_a, time.monotonic() + sg_time, n_steps=30000)
        if sc_sg < best_score:
            best_a, best_score = a_sg, sc_sg
            print(f"    Subgradient improved to {best_score:.10f}")

    # Phase E: Adam polish
    if time.monotonic() < deadline - 10:
        a_ad, sc_ad = _adam_polish(best_a, deadline, lr=0.01)
        if sc_ad < best_score:
            best_a, best_score = a_ad, sc_ad
            print(f"    Adam improved to {best_score:.10f}")

    # Final LP polish
    if time.monotonic() < deadline - 5:
        a_final, sc_final = _lp_phase(best_a, deadline, K_max=50, max_rounds=10)
        if sc_final < best_score:
            best_a, best_score = a_final, sc_final

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-n optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP peak balancing, L-BFGS continuation, subgradient momentum,
    and Adam polish. Tries multiple n values.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, norm_score={sc_parent:.10f}, actual={_score_full(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _greedy_init(n_parent)
        update_best(parent, _score_norm(parent), "greedy_default")

    print(f"\n=== Phase 1: Main pipeline on n={n_parent} ===")
    main_deadline = min(deadline, t_start + 1200.0)
    a_main, sc_main = _run_pipeline_for_n(n_parent, parent, main_deadline)
    update_best(a_main, sc_main, "main_pipeline")

    # Try greedy init too
    if time.monotonic() < deadline - 200:
        a_g = _greedy_init(n_parent)
        sc_g = _score_norm(a_g)
        print(f"Greedy init: {sc_g:.10f}")
        if sc_g < best_score * 1.05:
            g_deadline = min(deadline, time.monotonic() + 150)
            a_gb, sc_gb = _run_pipeline_for_n(n_parent, a_g, g_deadline)
            update_best(a_gb, sc_gb, "greedy_pipeline")

    # Try different parametric inits
    if time.monotonic() < deadline - 150:
        for et in ['gauss', 'bimodal', 'sqrt']:
            if time.monotonic() > deadline - 60:
                break
            a_p = _parametric_init(n_parent, et)
            sc_p = _score_norm(a_p)
            if sc_p < best_score * 1.08:
                p_deadline = min(deadline, time.monotonic() + 80)
                a_pb, sc_pb = _run_pipeline_for_n(n_parent, a_p, p_deadline)
                update_best(a_pb, sc_pb, f"param_{et}")

    print(f"\n=== Phase 2: Try n={n_parent * 2} ===")
    n_double = n_parent * 2
    if n_double <= 16384 and time.monotonic() < deadline - 100:
        a_interp = _interp_profile(a_main if best_seq is a_main else parent, n_double)
        sc_interp = _score_norm(a_interp)
        print(f"Interpolated n={n_double}: {sc_interp:.10f}")
        if sc_interp < best_score * 1.02:
            update_best(a_interp, sc_interp, f"interp_n{n_double}")
            d_deadline = min(deadline, time.monotonic() + 120)
            a_db, sc_db = _run_pipeline_for_n(n_double, a_interp, d_deadline)
            update_best(a_db, sc_db, f"pipeline_n{n_double}")

    print(f"\n=== Phase 3: Try n={n_parent // 2} ===")
    n_half = n_parent // 2
    if n_half >= 2048 and time.monotonic() < deadline - 80:
        a_half = _interp_profile(a_main if best_seq is a_main else parent, n_half)
        sc_half = _score_norm(a_half)
        print(f"Interpolated n={n_half}: {sc_half:.10f}")
        if sc_half < best_score * 1.02:
            update_best(a_half, sc_half, f"interp_n{n_half}")
            h_deadline = min(deadline, time.monotonic() + 100)
            a_hb, sc_hb = _run_pipeline_for_n(n_half, a_half, h_deadline)
            update_best(a_hb, sc_hb, f"pipeline_n{n_half}")

    print(f"\n=== Phase 4: Final intensive LP + subgradient on best ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        # Intensive LP
        lp_int_time = min(60, (deadline - time.monotonic()) * 0.4)
        a_int, sc_int = _lp_phase(best_seq, time.monotonic() + lp_int_time, K_max=100, max_rounds=30)
        if sc_int < best_score:
            best_seq, best_score = a_int, sc_int
            print(f"  Intensive LP: {best_score:.10f}")

        # Subgradient
        if time.monotonic() < deadline - 10:
            sg_time = min(40, (deadline - time.monotonic()) * 0.3)
            a_sg, sc_sg = _subgradient_momentum(best_seq, time.monotonic() + sg_time, n_steps=50000)
            if sc_sg < best_score:
                best_seq, best_score = a_sg, sc_sg
                print(f"  Final subgradient: {best_score:.10f}")

        # One more tiny LP
        if time.monotonic() < deadline - 5:
            a_f2, sc_f2 = _lp_phase(best_seq, deadline, K_max=80, max_rounds=15)
            if sc_f2 < best_score:
                best_seq, best_score = a_f2, sc_f2
                print(f"  Final LP: {best_score:.10f}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_full(fb):.10f}")
    return [float(x) for x in fb]
