"""
Optimized step function sequence finder using smoothed gradient descent with 
Nesterov acceleration, LP-based large steps, and multi-scale exploration.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(n):
    """Build sparse matrix for LP: A[j, k] = 1 if j and k are valid for lag.
    A has shape (2n-1, n). A[k, j] = 1 if 0 <= k-j < n, i.e., j <= k < j+n."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            rows.extend([k] * (j_end - j_start + 1))
            cols.extend(range(j_start, j_end + 1))
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def _lp_step(a, A_rows, A_cols):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return blended result."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    
    # Build values for the LP matrix
    # A[k, j] * b[j] should give conv(a,b)[k]
    # But we need A[k,j] = a[k-j] (the coefficient), not just 0/1
    # conv(a,b)[k] = sum_j a[k-j] * b[j] (valid j)
    
    # So the LP matrix has entries A[k,j] = a[k-j] for valid (k,j)
    vals = []
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            j_range = np.arange(j_start, j_end + 1)
            # a[k-j] for j in j_range
            kv = k - j_range
            v = a[kv]
            vals.extend(v)
            rows.extend([k] * len(v))
            cols.extend(j_range)
    
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status != 0:
            return a, M
        b = np.maximum(res.x, 0.0)
    except Exception:
        return a, M
    
    if b.sum() < 1e-15:
        return a, M
    
    # Normalize b to have the same sum as a
    s_a = a.sum()
    b = b * (s_a / b.sum())
    
    # Try different blending factors
    best_t = 0.0
    best_sc = _score_norm(a) * (a.sum()**2)  # scale-invariant score * sum^2
    # Actually, let's just use the score directly
    best_sc = _score(a)
    
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_cand = (1.0 - t) * a + t * b
        a_cand = np.maximum(a_cand, 0)
        if a_cand.sum() < 1e-10:
            continue
        sc = _score(a_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t > 0:
        a = (1.0 - best_t) * a + best_t * b
        a = np.maximum(a, 0)
    
    return a, M


def _gradient_smoothed(a, beta):
    """Compute gradient of the smoothed max-convolution objective.
    Returns gradient w.r.t. a (not projected to simplex)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    # Softmax weights
    c_shift = (c - c_max) / max(beta, 1e-15)
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # Gradient: grad[j] = sum_k w[k] * d(conv(a,a)[k])/d(a[j])
    # For interior: d(conv(a,a)[k])/d(a[j]) = 2*a[k-j] (when 0<=k-j<n)
    # grad[j] = 2 * sum_k w[k] * a[k-j] = 2 * sum_i w[j+i] * a[i]
    # This is 2 * correlate(w, a, 'valid')[j]
    grad = 2.0 * np.correlate(w, a, 'valid')
    # 'valid' gives min(len(w),len(a)) - max(len(w),len(a)) + 1 values... 
    # Actually np.correlate(w, a, 'valid') gives len(w) - len(a) + 1 values
    # = (2n-1) - n + 1 = n values. 
    
    # But we need to handle boundaries. The 'valid' mode assumes w[j+i] is defined
    # for i=0..n-1, which is true since len(w) = 2n-1 and j is 0..n-1,
    # so j+i is 0..2n-2. Good.
    
    return grad


def _nesterov_sm_descend(a_init, deadline, beta_start, beta_end, max_iter, momentum=0.9):
    """Nesterov-accelerated smooth descent on the simplex."""
    n = len(a_init)
    s = a_init.sum()
    a_init = a_init / s  # normalize
    
    z = np.log(np.maximum(a_init, 1e-300))
    z -= z.mean()
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    v = np.zeros(n)  # momentum (for Nesterov)
    z_nest = z.copy()  # Nesterov lookahead
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Update beta (anneal)
        t_frac = it / max(max_iter, 1)
        beta = beta_start * (beta_end / beta_start) ** t_frac
        
        # Evaluate at Nesterov point
        a = _softmax(z_nest)
        
        # Compute gradient
        grad = _gradient_smoothed(a, beta)
        
        # Project to tangent space of simplex
        W = float(np.dot(a, grad))
        grad_proj = a * (grad - W)
        
        # Nesterov update
        v = momentum * v + grad_proj
        step = v  # Use momentum directly as step direction
        
        # Adaptive step size
        # Try a few step sizes
        best_step = 0.01
        best_f = _score_norm(a)
        for step_size in [0.01, 0.03, 0.05, 0.1, 0.2]:
            z_trial = z - step_size * step
            z_trial = np.clip(z_trial, -50, 50)
            a_trial = _softmax(z_trial)
            f_trial = _score_norm(a_trial)
            if f_trial < best_f:
                best_f = f_trial
                best_step = step_size
        
        # Update
        z = z - best_step * step
        z = np.clip(z, -50, 50)
        z -= z.mean()
        
        # Nesterov lookahead
        z_nest = z - momentum * best_step * step
        
        # Track best
        if it % 50 == 0:
            a_curr = _softmax(z)
            sc = _score(a_curr)
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
    
    # Final evaluation
    a_final = _softmax(z)
    sc = _score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=100):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    
    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Try a few values for a[j]
            current_val = a[j]
            if current_val < 1e-10:
                continue
            s_rest = a.sum() - current_val
            if s_rest < 1e-15:
                continue
            
            # The optimal value of a[j] (given others fixed) minimizes
            # max_k (a*a)[k] where a appears in at most 2n-1 terms.
            # (a*a)[k] = sum_{i != j} a[i]*a[k-i] + a[j]*a[k-j] + a[j]*a[j-k] (if valid)
            # = base[k] + a[j] * (a[k-j] + a[j-k]) (where valid terms)
            # We want to minimize max_k (base[k] + a[j] * coeff[k])
            # subject to a[j] >= 0 and sum constraint (a[j] <= s_rest)
            
            # This is a 1D minimax problem. We can solve it by bisection or
            # by trying a grid of values.
            
            # For efficiency, just try geometric spacings around current value
            candidates = [current_val * 0.5, current_val * 0.7, current_val * 0.85,
                         current_val, current_val * 1.15, current_val * 1.3,
                         current_val * 2.0, 0.0]
            
            best_v = current_val
            best_sc = _score(a)
            
            for v in candidates:
                if v < 0 or v > s_rest:
                    v = np.clip(v, 0, s_rest)
                a[j] = v
                sc = _score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_v = v
            a[j] = best_v
        
        if _score(a) < best_score:
            best_score = _score(a)
            best_a = a.copy()
    
    return best_a, best_score


def _try_larger_n(a_parent, n_target, deadline):

def _lp_step_min_t(a):
    """LP: minimize t s.t. conv(a_new, a) <= t, sum(a_new)=1, a_new>=0."""
    n = len(a)
    # Build A: A[k,j] = a[k-j] for valid (k,j)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k-n+1)
        j_end = min(k, n-1)
        if j_end >= j_start:
            jr = np.arange(j_start, j_end+1)
            vals.append(a[k-jr])  # array of length (j_end-j_start+1)
            rows.extend([k]*(j_end-j_start+1))
            cols.extend(jr)
    
    A_sparse = sp.csc_matrix((np.concatenate(vals), (np.array(rows), np.array(cols))), shape=(2*n-1, n))
    
    # LP: min t s.t. A_sparse @ a_new - t <= 0, sum(a_new) = 1, a_new >= 0
    # Variables: x = [a_new (n vars), t (1 var)]
    # c = [0,...,0, 1]
    c_obj = np.zeros(n+1)

    b_ub = np.zeros(2*n-1)

Optimized step function sequence minimizer using smoothed Nesterov gradient descent,
LP-based large steps, and multi-scale exploration. Minimizes 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_smoothed(a, beta):
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shift = (c - c_max) / max(beta, 1e-15)
    e = np.exp(c_shift)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad = 2.0 * np.correlate(w, a, 'valid')
    return grad
