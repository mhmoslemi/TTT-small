2. Evaluating at each breakpoint and at a[j] = 0

"""
Optimized step function minimizer using L-BFGS with log-sum-exp smoothing,
subgradient method with backtracking line search, and coordinate descent refinement.
Key improvements over baseline: backtracking ensures monotonic convergence,
adaptive beta schedule, and coordinate-level fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_exact(a, c, n):
    c_max = c.max()
    tol = 0.001 * c_max
    active = c >= (c_max - tol)
    n_active = active.sum()
    if n_active == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
        n_active = 1
    w = np.zeros_like(c)
    w[active] = 1.0 / n_active
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_phase(a_init, beta, n, max_iter=2000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init


def _subgrad_backtrack(a_init, deadline, max_steps=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.02
    stagnation = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_exact(a, c, n)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        grad_dir = -grad / grad_norm

        # Backtracking line search
        step = lr
        improved = False
        for bt in range(12):
            z_try = z + step * grad_dir
            z_try = np.clip(z_try, -50.0, 50.0)
            z_try = z_try - z_try.mean()
            a_try = _softmax(z_try)
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-12:
                z = z_try
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True
                lr = step * 1.1
                stagnation = 0
                break
            step *= 0.5

        if not improved:
            lr *= 0.7
            stagnation += 1
            if stagnation > 50:
                # Try a larger random kick
                z = z + 0.01 * np.random.randn(n)
                z = z - z.mean()
                stagnation = 0
                lr = 0.02
            if lr < 1e-8:
                break

        if i % 200 == 0:
            pass  # Progress is tracked implicitly

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _coord_descent(a_init, n_sweeps=3):
    n = len(a_init)
    a = a_init.copy()
    best_sc = _score(a)

    for sweep in range(n_sweeps):
        improved_any = False
        # Process in random order
        order = np.random.permutation(n)
        for j in order:
            old_val = a[j]
            if old_val < 1e-10:
                continue
            # Try a few perturbations
            candidates = [old_val * 0.85, old_val * 0.92, old_val * 1.08, old_val * 1.15, old_val * 0.7]
            best_local = old_val
            best_local_sc = _score(a)

            for cv in candidates:
                a[j] = cv
                sc = _score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = cv
            a[j] = best_local

            if best_local != old_val:
                improved_any = True

        sc_now = _score(a)
        if sc_now < best_sc:
            best_sc = sc_now
        if not improved_any:
            break

    return a, best_sc


def _make_cosine_profile(n):
    t = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_tapered_profile(n):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** 0.5
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_flat_top_profile(n, alpha=0.3):
    t = np.linspace(0, 1, n)
    a = np.ones(n)
    # Taper the edges
    edge = int(alpha * n)
    for i in range(edge):
        frac = i / edge
        a[i] = 0.5 * (1 - np.cos(np.pi * frac))
        a[n - 1 - i] = 0.5 * (1 - np.cos(np.pi * frac))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _lbfgs_schedule(a_start, deadline, min_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        # Estimate iterations we can afford
        est_time_per_iter = 0.003 * max(1.0, n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 4))))

        a_new = _lbfgs_phase(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc

    return current_a
