Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for non-negative sequences. Key insights:

1. **The optimal sequence must be asymmetric** (palindromic sequences have score ≥ 2 by Cauchy-Schwarz)
2. **Efficient gradient computation** via FFT enables thousands of iterations
3. **Power-mean continuation** smoothly approaches the max objective
4. **Beta-distribution profiles** with α≠β provide good asymmetric starts
5. **Simplex projection** (water-filling) maintains feasibility

"""
High-efficiency optimizer for min 2*n*max(conv(a,a))/sum(a)^2.
Uses power-mean gradient descent with FFT, asymmetric Beta initializations,
simplex projection, and multi-scale refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    if n == 0:
        return np.inf
    c = _fft_convolve(a, a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _score_sum1(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _simplex_project(a):
    """Project a onto {x: sum(x)=1, x>=0} via water-filling."""
    n = len(a)
    if a.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho_star = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[rho_star - 1] - 1.0) / rho_star
    return np.maximum(a - theta, 0.0)


def _power_mean_grad(a, p, c=None):
    """Compute gradient of (1/K * sum c[k]^p)^(1/p) w.r.t. a."""
    n = len(a)
    if c is None:
        c = _fft_convolve(a, a)
    K = len(c)
    c_clipped = np.maximum(c, 1e-30)
    cp = c_clipped ** p
    S = np.mean(cp)
    if S < 1e-300:
        return np.zeros(n)
    # factor = 2/K * S^((1-p)/p)
    log_factor = np.log(max(2.0 / K, 1e-300)) + (1.0 - p) * np.log(max(S, 1e-300)) / p
    factor = np.exp(log_factor)
    # w[k] = c[k]^(p-1)
    w = c_clipped ** (p - 1)
    # h[i] = sum_k w[k] * a[k-i] = conv(w, a_rev)[n-1+i]
    a_rev = a[::-1]
    h_full = _fft_convolve(w, a_rev)
    h = h_full[n - 1:2 * n - 1]
    return factor * h


def _lse_obj_grad(z, beta, n):
    """Log-sum-exp objective and gradient in log-domain."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = _fft_convolve(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        obj = 2.0 * n * cm
        grad = np.zeros(n)
    else:
        log_sum = np.log(s)
        obj = 2.0 * n * (cm + beta * log_sum)
        w = e / s
        a_rev = a[::-1]
        w_conv = _fft_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = np.dot(w, c)
        grad = 4.0 * n * a * (h - W)
    return obj, grad


def _make_beta_profile(n, alpha, beta):
    """Beta distribution profile: a[i] ∝ (i+1)^(alpha-1) * (n-1-i)^(beta-1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def _make_bspline_profile(n, order):
    """B-spline profile of given order via iterative convolution of box."""
    box = np.ones(max(2, n // order)) / (n // order)
    a = box.copy()
    for _ in range(order - 1):
        a = _fft_convolve(a, box)
        a = a[:n] if len(a) >= n else np.pad(a, (0, n - len(a)))
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def _make_asym_bspline(n, left_order, right_order):
    """Asymmetric B-spline: different orders on each side."""
    # Create left half with left_order, right half with right_order
    half = n // 2
    a_left = _make_bspline_profile(half + 1, left_order)[:half]
    a_right = _make_bspline_profile(n - half + 1, right_order)[::-1][:n - half]
    a = np.concatenate([a_left, a_right])
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def _make_exp_decay(n, tau_frac):
    """Exponential decay profile."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / (tau_frac * n))
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def _make_power_law(n, p):
    """Power law: a[i] ∝ (n-1-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (n - 1 - i + 1) ** p
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def _run_power_mean_descent(a_init, deadline, p_schedule, lr_init=0.05, label=""):
    """Power-mean gradient descent with Adam and simplex projection."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_sum1(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    
    for p_val in p_schedule:
        if time.monotonic() > deadline - 2.0:
            break
        # Adaptive number of iterations per p level
        time_per_p = max(2.0, (deadline - time.monotonic()) / max(1, len(p_schedule) - p_schedule.index(p_val)))
        max_iters_p = int(50000 * time_per_p / 30.0)
        max_iters_p = max(500, min(max_iters_p, 200000))
        
        lr = lr_init / (1.0 + 0.1 * p_schedule.index(p_val))
        
        for it in range(max_iters_p):
            if time.monotonic() > deadline - 2.0:
                break
            t += 1
            a = _simplex_project(np.exp(z - z.max()))
            c = _fft_convolve(a, a)
            
            if p_val <= 1.0:
                # Just use max gradient
                k_max = int(np.argmax(c))
                lo = max(0, k_max - (n - 1))
                hi = min(n - 1, k_max)
                grad = np.zeros(n)
                grad[lo:hi + 1] = 2.0 * a[k_max - hi:k_max - lo + 1][::-1]
                grad *= 2.0 * n
            else:
                grad = _power_mean_grad(a, p_val, c)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = np.clip(z, -50, 50)
            
            if t % 20 == 0:
                a_check = _simplex_project(np.exp(z - z.max()))
                sc = _score_sum1(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
        
        print(f"    p={p_val}: score={best_score:.10f} [{label}]")
    
    return best_a, best_score


def _run_lse_lbfgs(a_init, deadline, beta_schedule=None, max_wall=60.0):
    """L-BFGS with log-sum-exp continuation in log-domain."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_sum1(a_init)
    
    if beta_schedule is None:
        c0 = _fft_convolve(a_init, a_init)
        max_c0 = float(c0.max())
        beta_schedule = [b * max_c0 for b in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
    
    per_level = max(3.0, max_wall / len(beta_schedule))
    
    for beta in beta_schedule:
        if time.monotonic() > deadline - 2.0:
            break
        
        def obj_fn(z_):
            o, _ = _lse_obj_grad(z_, beta, n)
            return o
        
        def grad_fn(z_):
            _, g = _lse_obj_grad(z_, beta, n)
            return g
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _simplex_project(np.exp(result.x - result.x.max()))
                sc = _score_sum1(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _peak_local_search(a_init, deadline, n_trials=50000):
    """Peak-targeted local search with mass redistribution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_sum1(a)
    rng = np.random.default_rng(12345)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _fft_convolve(a, a)
        K = min(8, len(c))
        top_lags = np.argsort(c)[-K:]
        
        strategy = it % 4
        
        if strategy == 0:
            # Reduce contribution at top peak
            k_max = int(top_lags[-1])
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            if lo > hi:
                continue
            pairs = np.arange(lo, hi + 1)
            partners = k_max - pairs
            products = a[pairs] * a[partners]
            idx = pairs[int(np.argmax(products))]
            
            a_trial = a.copy()
            frac = rng.uniform(0.3, 0.7)
            delta = a_trial[idx] * frac
            a_trial[idx] = max(0.0, a_trial[idx] - delta)
            
            # Find safe position (low contribution to peaks)
            contrib = np.zeros(n)
            for kl in top_lags[:3]:
                kl = int(kl)
                lo2 = max(0, kl - (n - 1))
                hi2 = min(n - 1, kl)
                if lo2 <= hi2:
                    contrib[lo2:hi2 + 1] += a[kl - hi2:kl - lo2 + 1][::-1]
            safe = np.argsort(contrib)[:max(1, n // 10)]
            add_idx = int(safe[rng.integers(len(safe))])
            a_trial[add_idx] += delta
            a_trial = _simplex_project(a_trial)
            
        elif strategy == 1:
            # Block perturbation
            block_size = int(rng.integers(3, min(32, n // 10)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.6, 1.4, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = _simplex_project(a_trial)
            
        elif strategy == 2:
            # Pair swap
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt = min(a_trial[i1], a_trial[i2]) * frac
            a_trial[i1] -= amt
            a_trial[i2] += amt
            a_trial = _simplex_project(np.maximum(a_trial, 0.0))
            
        else:
            # Gaussian perturbation
            a_trial = a.copy()
            sigma = rng.uniform(0.01, 0.05)
            a_trial *= rng.lognormal(0, sigma, size=n)
            a_trial = _simplex_project(a_trial)
        
        sc = _score_sum1(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_sc


def _interp_to_n(a, n_new):
    """Interpolate profile to new length."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    if n0 == 1:
        return np.full(n_new, a[0])
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-15)
    return _simplex_project(p)
