"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses parametric initializations (power-law ramps) combined with
L-BFGS log-sum-exp smoothing and subgradient polishing.
Key improvement: multi-length search and analytically-informed starts.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_active(z, a, c, n, tol=0.001):
    c_max = c.max()
    active = c >= (c_max - tol * c_max)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_run(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

def _beta_schedule(a_start, deadline, min_remaining=50.0):
    n = len(a_start)
    c = _full_convolve(a_start, a_start)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a_start
    
    betas = max_c * np.array([0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125])
    current_a = a_start.copy()
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        # Estimate time per iteration
        est_time = 0.003 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time * 5))))
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        sc_old = _score(current_a)
        if sc < sc_old:
            current_a = a_new
    return current_a

def _ramp_init(n, p=1.0, max_val=1000.0):
    """Power-law ramp: a_j = max_val * (j/(n-1))^p"""
    t = np.linspace(0, 1, n)
    a = max_val * t**p
    a = np.maximum(a, 1e-10)
    return a

def _ramp_init_sym(n, p=1.0, max_val=1000.0):
    """Symmetric power-law: high at both ends, low in middle"""
    t = np.linspace(0, 1, n)
    a = max_val * (1 - 2*abs(t - 0.5) + 1)**p  # This gives max at ends
    # Actually let's do: a_j = max_val * max(t, 1-t)^p
    a = max_val * np.maximum(t, 1-t)**p
    a = np.maximum(a, 1e-10)
    return a
