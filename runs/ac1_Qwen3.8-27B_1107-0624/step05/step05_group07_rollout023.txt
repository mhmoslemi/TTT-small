Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50.0, 50.0)
    e = np.exp(vals)

    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    G = 4.0 * n * h
    grad = a * (G - W)
    return grad, a, c

            # Check score periodically
            if it % 50 == 0:
                a = _softmax(z)
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        # End of beta: update best
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"    LSE beta={beta:.6f}: score={sc:.10f}, best={best_score:.10f}")

    return best_a, best_score


def _peak_focused_obj(z, beta, n, w_peaks):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50.0, 50.0)
    e = np.exp(vals)
    weighted = w_peaks * e
    W = weighted.sum()
    if W < 1e-300:
        return 2.0 * n * cm
    lse = cm + beta * float(np.log(W))
    return 2.0 * n * lse


def _peak_focused_grad(z, beta, n, w_peaks):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50.0, 50.0)
    e = np.exp(vals)
    weighted = w_peaks * e
    W = weighted.sum()
    if W < 1e-300:
        w = e / e.sum()
    else:
        w = weighted / W
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    Wc = float(np.dot(w, c))
    G = 4.0 * n * h
    grad = a * (G - Wc)
    return grad


def _nesterov_peak_focused(a_init, deadline, n_peaks=10):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c = _full_convolve(_softmax(z), _softmax(z))
    cm = float(c.max())

    z_prev = z.copy()
    momentum = 0.9

    for phase in range(5):
        # Recompute peak weights
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        top_k = np.argsort(c)[-n_peaks:]
        w_peaks = np.ones(2 * n - 1)
        w_peaks[top_k] = 5.0
        w_peaks[top_k[:max(1, n_peaks // 2)]] = 10.0

        beta = (0.01 / (phase + 1)) * cm
        time_budget = min(40.0, (deadline - time.monotonic()) * 0.3)
        if time_budget < 5:
            break

        max_iter = int(time_budget * 40)
        for it in range(max_iter):
            if time.monotonic() > deadline - 60:
                break
            z_nest = z + momentum * (z - z_prev)
            grad = _peak_focused_grad(z_nest, beta, n, w_peaks)
            if not np.all(np.isfinite(grad)):
                break
            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-15:
                lr = min(1.0, 0.5 / (gnorm + 1e-10))
            else:
                lr = 0.5
            z_prev = z.copy()
            z = z - lr * grad
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()
            momentum = min(momentum + 0.0005 * it, 0.98)

            if it % 100 == 0:
                a = _softmax(z)
                sc = _score_norm(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        print(f"    Peak-focused phase {phase}: beta={beta:.6f}, score={sc:.10f}")

    return best_a, best_score


def _coord_1d_minimize(j, a, c_current):
    """Minimize max(conv) when only a[j] varies, using bisection on subgradient."""
    n = len(a)
    # base_c = c with a[j]=0: c - 2*a[j]*a_shifted - a[j]^2*delta
    x_old = a[j]
    # a_shifted[k] = a[k-j] for valid k
    # Build shifted array
    a_shift = np.zeros(2 * n - 1)
    # For k from j to j+n-1: a_shift[k] = a[k-j]
    for valid_k in range(j, min(j + n, 2 * n - 1)):
        a_shift[valid_k] = a[valid_k - j]
    # Actually we need: contribution of a[j] to c[k] is 2*a[j]*a[k-j] if 0<=k-j<n and k!=2j, or a[j]^2 if k=2j
    # So c[k] = base[k] + lin[k]*x + quad[k]*x^2
    # where lin[k] = 2*a[k-j] (if 0<=k-j<n and k!=2j), quad[k] = 1 (if k==2j)

    lin = np.zeros(2 * n - 1)
    quad = np.zeros(2 * n - 1)
    # Component from i=j: a[j]*a[k-j], valid if 0<=k-j<n
    lo = max(0, j)
    hi = min(2 * n - 1, j + n - 1)
    if hi > lo:
        idx = np.arange(lo, hi)
        lin[idx] = a[idx - j] if j != 0 else a[idx - j]
        # Wait, a[k-j] for k in [lo, hi): k-j ranges from lo-j to hi-j-1
        # lo = j (since j >= 0), so k-j ranges from 0 to hi-j-1
        # hi = j+n-1, so hi-j-1 = n-2. Hmm, let me redo.
        # k ranges from j to j+n-1, so k-j ranges from 0 to n-1. Valid (all in [0,n-1]).
        lin[lo:hi] = a[0:n - (hi - lo - 1) + 1] if hi - lo <= n else a[:hi-lo]
        # Actually simpler: for k in [j, j+n-1], a[k-j] is a[0], a[1], ..., a[n-1]
        # But we also need the component from i=k-j: a[k-j]*a[j], which is the same term
        # unless k-j = j (i.e., k = 2j)
    # Let me just do it directly:
    lin = np.zeros(2 * n - 1)
    quad = np.zeros(2 * n - 1)
    # For each k, the contribution of x=a[j] to c[k]:
    # From i=j: x * a[k-j] if 0 <= k-j < n
    # From i=k-j (if k-j != j and 0 <= k-j < n): a[k-j] * x
    if 0 <= 2 * j <= 2 * n - 2:
        quad[2 * j] = 1.0  # when k=2j, both terms give x*a[j]=x^2, totaling 2*x^2... 
        # Wait: from i=j: x * a[2j-j] = x * a[j] = x^2
        # From i=k-j=j: a[j] * x = x^2
        # Total: 2*x^2? No! When i=j and k-j=j, these are the SAME term.
        # In the sum c_k = sum_i a_i * a_{k-i}, when i=j and k-i=j, these are the same i value.
        # So it's just one term: a[j]*a[j] = x^2.
        quad[2 * j] = 1.0
        # Don't add to lin for this k
    # For k != 2j:
    for k_offset in range(n):
        k = j + k_offset
        if k <= 2 * n - 2 and k_offset != j:
            lin[k] = 2.0 * a[k_offset]
        k = j - k_offset
        if k >= 0 and k <= 2 * n - 2 and k_offset != j:
            if k != 2 * j:
                lin[k] = 2.0 * a[k_offset]

    # base = c_current - lin*x_old - quad*x_old^2
    base = c_current - lin * x_old - quad * x_old * x_old

    # Bisection to find x that minimizes max_k (base[k] + lin[k]*x + quad[k]*x^2)
    # f(x) is convex in x >= 0
    x_lo = 0.0
    x_hi = 3.0 * x_old if x_old > 0 else 1.0 / n

    # Evaluate f at boundaries
    def eval_f(x):
        return float(np.max(base + lin * x + quad * x * x))

    f_lo = eval_f(x_lo)
    f_hi = eval_f(x_hi)

    # Check if minimum is at boundary
    # Subgradient at x: sum_{k in S} (lin[k] + 2*quad[k]*x) / |S|
    def subgrad(x):
        vals = base + lin * x + quad * x * x
        mx = vals.max()
        tol = 1e-10 * max(1.0, abs(mx))
        active = np.where(vals >= mx - tol)[0]
        if len(active) == 0:
            return 0.0
        sg = np.sum(lin[active] + 2.0 * quad[active] * x) / len(active)
        return float(sg)

    # Bisection
    for _ in range(40):
        x_mid = 0.5 * (x_lo + x_hi)
        sg = subgrad(x_mid)
        if sg > 0:
            x_hi = x_mid
        else:
            x_lo = x_mid

    x_opt = 0.5 * (x_lo + x_hi)
    # Also check a few candidates
    candidates = [0.0, x_opt, x_old, 0.5 * x_old, 1.5 * x_old, 2.0 * x_old]
    candidates = [max(0.0, c) for c in candidates]
    best_x = x_old
    best_val = eval_f(x_old)
    for xc in candidates:
        v = eval_f(xc)
        if v < best_val:
            best_val = v
            best_x = xc

    return best_x


def _coord_descent_round(a, deadline, max_rounds=10):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c_current = _full_convolve(a, a)
        improved = 0.0
        for j in range(n):
            if time.monotonic() > deadline:
                break
            if a[j] < 1e-12 and rnd < max_rounds - 2:
                # Skip very small elements in early rounds
                continue
            x_new = _coord_1d_minimize(j, a, c_current)
            if x_new >= 0:
                old = a[j]
                a[j] = x_new
                # Update c_current efficiently
                # c changes by: (x_new - old) * (lin + ...) + ...
                # Simpler: just update the affected bins
                dx = x_new - old
                if abs(dx) > 1e-15:
                    # c[k] changes by dx * (2*a[k-j] - 2*old... ) 
                    # Actually: new contribution - old contribution
                    # = dx * a[k-j] (from i=j) + dx * a[k-j] (from i=k-j, if k-j!=j)
                    # Plus if k=2j: additional dx*x_new (from x^2 - old^2 = dx*(x_new+old))
                    for k in range(max(0, j - (n - 1)), min(2 * n - 2, j + (n - 1)) + 1):
                        if 0 <= k - j < n:
                            if k - j != j:
                                c_current[k] += 2.0 * dx * a[k - j]
                            else:
                                c_current[k] += 2.0 * dx * a[k - j] + dx * (x_new + old)
                    # Hmm this is getting complicated. Let me just recompute periodically.
                improved += abs(x_new - old)

        # Recompute c_current to avoid accumulated error
        c_current = _full_convolve(a, a)

        s = a.sum()
        if s > 1e-15:
            a = a / s
            c_current = _full_convolve(a, a)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    Coord round {rnd}: score={sc:.10f}, improved={improved:.6f}")
        else:
            print(f"    Coord round {rnd}: score={sc:.10f} (no improvement)")

    return best_a, best_score


def _make_random_profile(n, rng):
    """Generate a random non-negative profile."""
    # Mix of different structures
    choice = rng.integers(0, 5)
    if choice == 0:
        # Smooth random
        freqs = rng.uniform(0.1, 3.0, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 1.0, size=5)
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, p, amp in zip(freqs, phases, amps):
            a += amp * np.cos(2 * np.pi * f * t + p)
    elif choice == 1:
        # Gaussian bump
        center = rng.uniform(0.2, 0.8)
        width = rng.uniform(0.05, 0.3)
        t = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((t - center) / width) ** 2)
    elif choice == 2:
        # Cosine window variant
        t = np.linspace(0, 1, n)
        alpha = rng.uniform(0.5, 3.0)
        a = np.cos(np.pi * (t - 0.5) * alpha) ** 2
    elif choice == 3:
        # Sigmoid
        t = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        sharp = rng.uniform(5, 30)
        a = 1.0 / (1.0 + np.exp(-sharp * (t - center)))
    else:
        # Random Fourier (smooth)
        n_freqs = 8
        freqs = rng.integers(1, 5, size=n_freqs) * 2 + 1  # odd frequencies
        coeffs = rng.normal(size=n_freqs) * 0.3
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        for f, c in zip(freqs, coeffs):
            a += c * np.cos(2 * np.pi * f * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        t = np.linspace(0, 1, n_init)
        a = np.maximum(np.cos(np.pi * (2 * t - 1) / 2), 1e-12) ** 2
        parent = a / a.sum()
        update_best(parent, _score_norm(parent), "default")

    n_parent = len(parent)

    # Phase 1: Nesterov GD with log-sum-exp at parent n (600s)
    phase1_end = min(deadline, t_start + 550.0)
    print(f"\nPhase 1: Nesterov LSE (n={n_parent})")

    a_cur = parent.copy()
    c0 = _full_convolve(a_cur, a_cur)
    max_c0 = float(c0.max())

    # Beta schedule: from moderate to very small
    beta_schedule = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005,
                     0.004, 0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0006,
                     0.0005, 0.0004, 0.0003, 0.0002, 0.00015]
    beta_schedule = [b * max_c0 for b in beta_schedule]

    a_best, sc_best = _nesterov_gd(a_cur, phase1_end, beta_schedule, lr_init=0.8, momentum_init=0.9)
    update_best(a_best, sc_best, "nesterov_lse")

    # Phase 2: Peak-focused refinement (300s)
    phase2_end = min(deadline, t_start + 850.0)
    print(f"\nPhase 2: Peak-focused refinement")
    if time.monotonic() < phase2_end and best_seq is not None:
        a_pf, sc_pf = _nesterov_peak_focused(best_seq, phase2_end, n_peaks=10)
        update_best(a_pf, sc_pf, "peak_focused")

    # Phase 3: More Nesterov with smaller beta (200s)
    phase3_end = min(deadline, t_start + 1050.0)
    print(f"\nPhase 3: Fine Nesterov (small beta)")
    if time.monotonic() < phase3_end and best_seq is not None:
        c_cur = _full_convolve(best_seq, best_seq)
        max_c_cur = float(c_cur.max())
        fine_betas = [0.0001, 0.00005, 0.00003, 0.00002, 0.00001]
        fine_betas = [b * max_c_cur for b in fine_betas]
        a_fine, sc_fine = _nesterov_gd(best_seq, phase3_end, fine_betas, lr_init=0.3, momentum_init=0.95)
        update_best(a_fine, sc_fine, "fine_nesterov")

    # Phase 4: Coordinate descent (400s)
    phase4_end = min(deadline, t_start + 1450.0)
    print(f"\nPhase 4: Coordinate descent")
    if time.monotonic() < phase4_end and best_seq is not None:
        # Use a smaller n for coordinate descent if n is very large
        n_cd = len(best_seq)
        if n_cd > 4096:
            a_cd = _interp_profile(best_seq, 4096)
        else:
            a_cd = best_seq.copy()
        sc_cd_start = _score_norm(a_cd)
        a_cd_best, sc_cd_best = _coord_descent_round(a_cd, phase4_end, max_rounds=15)
        update_best(a_cd_best, sc_cd_best, "coord_descent")

        # If we downsampled, interpolate back
        if n_cd > 4096:
            a_cd_back = _interp_profile(a_cd_best, n_cd)
            sc_cd_back = _score_norm(a_cd_back)
            update_best(a_cd_back, sc_cd_back, "coord_descent_interp_back")

    # Phase 5: Final polish - alternating peak-focused + coord (remaining time)
    phase5_end = deadline
    print(f"\nPhase 5: Final polish")
    if time.monotonic() < phase5_end and best_seq is not None:
        # Quick peak-focused
        time_left = phase5_end - time.monotonic()
        if time_left > 30:
            a_pol, sc_pol = _nesterov_peak_focused(best_seq, time.monotonic() + time_left * 0.3, n_peaks=5)
            update_best(a_pol, sc_pol, "polish_pf")

        # Quick coord descent
        if time.monotonic() < phase5_end - 10 and best_seq is not None:
            a_pol2, sc_pol2 = _coord_descent_round(best_seq, phase5_end - 10, max_rounds=5)
            update_best(a_pol2, sc_pol2, "polish_cd")

        sc_2048 = _score_norm(a_2048)

            a_2048_opt, sc_2048_opt = _nesterov_gd(a_2048, min(deadline - 5, time.monotonic() + 60), betas_2048)
            update_best(a_2048_opt, sc_2048_opt, "downsample_2048_opt")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn_safe(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print("Evaluator returned non-finite!")
                    # Fallback
                    n_fb = 2048
                    t_fb = np.linspace(0, 1, n_fb)
                    a_fb = np.maximum(np.cos(np.pi * (2 * t_fb - 1) / 2), 1e-12) ** 2
                    a_fb = a_fb / a_fb.sum()
                    return [float(x) for x in a_fb]
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    t_fb = np.linspace(0, 1, n_fb)
    a_fb = np.maximum(np.cos(np.pi * (2 * t_fb - 1) / 2), 1e-12) ** 2
    a_fb = a_fb / a_fb.sum()
    print(f"FALLBACK: n={n_fb}")
    return [float(x) for x in a_fb]


def _score_fn_safe(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
