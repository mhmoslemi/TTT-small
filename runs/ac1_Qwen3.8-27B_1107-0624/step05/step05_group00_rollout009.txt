"""
Multi-phase optimizer combining FISTA, mirror descent, and L-BFGS
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w):
    """Compute gradient of sum(w_k * (a*a)_k) w.r.t. a"""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()

            break

        y = np.maximum(y, 0.0)

def _mirror_descent(a_init, deadline, max_iter):
    """Mirror descent with exact subgradient on the max-convolution"""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    best_score = _score_norm(a)
    best_a = a.copy()
    
    lr = 1.0 / np.sqrt(2.0 * n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find active set (positions where c is close to max)
        threshold = c_max - 1e-6 * c_max
        active = np.where(c >= threshold)[0]
        
        # Subgradient: average over active set
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        
        grad = _compute_grad(a, w)
        
        # Mirror descent step (exponential weighting)
        z = np.log(np.maximum(a, 1e-30))
        z = z - lr * grad
        a = _softmax_compute(z)
        
        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Decay learning rate
        lr = 1.0 / np.sqrt(2.0 * n * (it + 1))
    
    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

"""
Multi-phase optimizer: aggressive FISTA + mirror descent + Adam + L-BFGS
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_opt(a_init, beta, deadline, max_iter):

    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        if it % 80 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)

        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _mirror_descent_opt(a_init, deadline, max_iter):
    """Mirror descent using exact subgradient of max(conv(a,a)) on simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_score = _score_norm(a)
    best_a = a.copy()
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    lr_decay = 1.0 / np.sqrt(2.0 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Active set: positions within a small tolerance of max
        tol = 1e-7 * c_max
        active = np.where(c >= c_max - tol)[0]
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        grad = _compute_grad(a, w)
        lr = lr_decay / np.sqrt(it + 1)
        z = z - lr * grad
        a = _softmax_compute(z)
        if it % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        best_a = a.copy()

def _adam_opt(a_init, deadline, beta, lr_init, max_iter):

    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        e = np.exp((c - cm) / beta)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()
        if t % 25 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        if it % 4000 == 0:
            current_lr = max(0.0003, current_lr * 0.92)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

    from scipy.optimize import minimize
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    best_score = _score_norm(a_init)
    best_z = z0.copy()

        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta * np.log(s))

        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_sh = (c - cm) / beta
        e = np.exp(c_sh)
        s = float(e.sum())

            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0

            w = e / s
        a_rev = a[::-1]

        h = wc[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        correction = float(np.dot(a, grad_a))

    max_time = deadline - time.monotonic() - 3.0
    if max_time <= 1.0:
        return a_init, best_score

    try:
        result = minimize(
            obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
            options={'maxiter': max(50, int(max_time * 30)), 'gtol': 1e-12, 'maxls': 15}
        )
        z_opt = result.x
        a_opt = _softmax_compute(z_opt)
        sc_opt = _score_norm(a_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_z = z_opt.copy()
    except Exception as e:
        print(f"  L-BFGS error: {e}")

    a_best = _softmax_compute(best_z)
    return a_best, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    a_work = a0.copy()

    # === Phase 1: FISTA aggressive continuation ===
    print("Phase 1: FISTA aggressive beta continuation")
    phase1_end = min(deadline, t_start + 350.0)
    c_curr = _full_convolve(a_work, a_work)
    max_c0 = float(c_curr.max())
    beta = 0.03 * max_c0
    n_phases = 25

    for phase in range(n_phases):
        if time.monotonic() > phase1_end - 3:
            break
        remaining = phase1_end - time.monotonic() - 2
        if remaining < 1:
            break
        per_time = min(remaining / max(1, n_phases - phase), 25.0)
        max_it = max(3000, int(per_time * 4000))
        a_work, _ = _fista_opt(a_work, beta, time.monotonic() + per_time, max_it)
        a_work = a_work / a_work.sum()
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            if phase % 5 == 0:
                print(f"  FISTA p{phase} beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        beta = 0.35 * float(c_curr.max())
        if beta < 1e-10:
            beta = 1e-10

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Mirror descent (exact subgradient) ===
    print("Phase 2: Mirror descent exact subgradient")
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end - 10:
        a_md, _ = _mirror_descent_opt(a_work, phase2_end, max_iter=500000)
        sc_md = _score(a_md)
        if sc_md < best_score:
            best_score = sc_md
            best_seq = a_md.copy()
            print(f"  Mirror descent: score={sc_md:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_md

    # === Phase 3: Adam multi-beta fine ===
    print("Phase 3: Adam fine optimization")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        betas = [0.0008 * max_cc, 0.0002 * max_cc, 0.00005 * max_cc, 0.00001 * max_cc]
        per_beta = (phase3_end - time.monotonic() - 10) / len(betas)
        for bi, beta in enumerate(betas):
            if time.monotonic() > phase3_end - 10:
                break
            a_work, _ = _adam_opt(a_work, time.monotonic() + max(5.0, per_beta),
                                  beta, lr_init=0.006, max_iter=500000)
            sc = _score(a_work)
            if sc < best_score:
                best_score = sc
                best_seq = a_work.copy()
                print(f"  Adam beta={beta:.2e}: score={sc:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS refinement ===
    print("Phase 4: L-BFGS refinement")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end - 15:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        beta_lb = 0.00005 * max_cc
        a_lb, _ = _lbfgs_opt(a_work, phase4_end, beta_lb)
        sc_lb = _score(a_lb)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  L-BFGS: score={sc_lb:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = a_lb

    # === Phase 5: Restart exploration ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 20:
        n_restarts = 4
        per_r = (phase5_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            if i == 0:
                p = _perturb(a_work, 0.015, rng)
            elif i == 1:
                p = _make_profile(n_main, "raised_cosine")
                p = 0.8 * a_work + 0.2 * p
                p = np.maximum(p, 1e-12)
                p = p / p.sum()
            elif i == 2:
                shift = rng.integers(1, max(2, n_main // 512))
                p = np.roll(a_work, shift).copy()
                p = p / p.sum()
            else:
                p = _perturb(a_work, 0.03, rng)
            c_p = _full_convolve(p, p)
            max_cp = float(c_p.max())
            b_r = 0.0003 * max_cp
            p, _ = _fista_opt(p, b_r, time.monotonic() + per_r * 0.5, 40000)
            p, _ = _adam_opt(p, time.monotonic() + per_r * 0.5, b_r * 0.2, 0.004, 80000)
            sc_r = _score(p)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = p.copy()
                a_work = p
                print(f"  Restart {i}: score={sc_r:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-polish ===
    print("Phase 6: Final ultra-polish")
    if best_seq is not None and time.monotonic() < deadline - 12:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        per_t = deadline - time.monotonic() - 5
        if per_t > 8:
            # Final Adam with very tiny beta
            a_pol, _ = _adam_opt(a_pol, time.monotonic() + per_t * 0.6,
                                 0.000005 * max_cp, 0.0015, 200000)
            sc_p = _score(a_pol)
            if sc_p < best_score:
                best_score = sc_p
                best_seq = a_pol.copy()
                print(f"  Final Adam: score={sc_p:.10f}")
            # L-BFGS on remaining time
            if time.monotonic() < deadline - 8:
                c_f = _full_convolve(a_pol, a_pol)
                b_f = 0.000003 * float(c_f.max())
                a_pol, _ = _lbfgs_opt(a_pol, deadline - 3.0, b_f)
                sc_f = _score(a_pol)
                if sc_f < best_score:
                    best_score = sc_f
                    best_seq = a_pol.copy()
                    print(f"  Final L-BFGS: score={sc_f:.10f}")

    print(f"Done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
