        if score < best_score:

            best_d = d

3. Minimize the smooth function `g(d) = 2n*(c[k*] + 2d*s[k*]) / (S+d)^2` using Newton's method.
4. Check if the solution is valid (i.e., k* is still the argmax at the new d).
5. If not, iterate.

    n_conv = 2*n - 1
    # Current best (d=0)
    c_max_0 = c.max()
    k_star = np.argmax(c)
    
    # Newton iteration on g(d) = 2n*(c[k*] + 2*d*s[k*]) / (S+d)^2
    # g'(d) = 2n * [2*s[k*]*(S+d)^2 - (c[k*]+2d*s[k*]) * 2*(S+d)] / (S+d)^4
    #       = 2n * [2*s[k*]*(S+d) - 2*(c[k*]+2d*s[k*])] / (S+d)^3
    #       = 4n * [s[k*]*(S+d) - (c[k*]+2d*s[k*])] / (S+d)^3
    #       = 4n * [s[k*]*S + d*s[k*] - c[k*] - 2d*s[k*]] / (S+d)^3
    #       = 4n * [s[k*]*S - c[k*] - d*s[k*]] / (S+d)^3
    
    # Set g'(d) = 0: s[k*]*S - c[k*] - d*s[k*] = 0

"""
Hybrid optimizer combining L-BFGS (log-sum-exp smoothing), coordinate descent,
and FISTA for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key: efficient coordinate descent with analytical 1D steps + momentum methods.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_exact(a, c, n):
    c_max = c.max()
    tol = 0.001 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_phase(a_init, beta, n, deadline, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _fista_optimize(a_init, beta, n, max_iter=10000, deadline=None):
    """Nesterov-accelerated gradient descent in log-space with backtracking."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    t_k = 1.0
    L = 1.0
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for i in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_y = _softmax(y)
        grad = _smograd(y, beta, n)
        obj_y = _smoobj(y, beta, n)

        # Gradient step
        z_new = y - grad / L
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        # Backtracking line search
        obj_new = _smoobj(z_new, beta, n)
        backtracks = 0
        while obj_new > obj_y + 0.5 * L * np.dot(grad, z_new - y) and backtracks < 10:
            L *= 2.0
            z_new = y - grad / L
            z_new = np.clip(z_new, -60.0, 60.0)
            z_new = z_new - z_new.mean()
            obj_new = _smoobj(z_new, beta, n)
            backtracks += 1
        while obj_new < obj_y - 0.5 * L * np.dot(grad, z_new - y) and backtracks < 20:
            L = max(L / 1.5, 0.1)
            z_candidate = y - grad / L
            z_candidate = np.clip(z_candidate, -60.0, 60.0)
            z_candidate = z_candidate - z_candidate.mean()
            obj_candidate = _smoobj(z_candidate, beta, n)
            if obj_candidate < obj_y:
                z_new = z_candidate
                obj_new = obj_candidate
            backtracks += 1

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z_new + ((t_k - 1.0) / t_new) * (z_new - y)
        y = np.clip(y, -60.0, 60.0)
        y = y - y.mean()
        t_k = t_new

        if i % 200 == 0:
            a_check = _softmax(y)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(y)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _coord_descent_sweep(a, c, S, n, deadline=None):
    """One full coordinate descent sweep. Returns updated (a, c, S, score)."""
    t_start = time.monotonic()
    score = 2.0 * n * c.max() / (S * S)

    for i in range(n):
        if deadline is not None and (time.monotonic() - t_start) > 5.0:
            break

        ai = a[i]
        if ai < 1e-15:
            continue

        # Compute influence vector: s[k] = a[k-i] if 0<=k-i<n
        # conv(a,e_i)[k] = a[k-i] for valid indices
        s = np.zeros(2 * n - 1)
        # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k < i+n
        s[i:i + n] = a

        # Also need: the term from the "other side" of convolution
        # conv(a, e_i)[k] actually equals a[k-i] when 0<=k-i<n
        # But conv is symmetric, so conv(e_i, a)[k] = a[k-i] as well
        # Actually for conv(a,a) change: delta_c[k] = 2*d*a[k-i] + d^2*delta(k,2i)
        # Wait, let me reconsider.
        # c_new[k] = sum_j (a[j]+d*delta(j,i)) * (a[k-j]+d*delta(k-j,i))
        #          = c[k] + d*sum_j (delta(j,i)*a[k-j] + a[j]*delta(k-j,i))
        #            + d^2*delta(i,i')*delta(k-i,i'')
        # The linear term: d*(a[k-i] + a[i]) where valid indices apply
        # a[k-i] valid when 0<=k-i<n
        # a[i] valid when 0<=k-i<n (i.e., same condition since we need k-j=i => j=k-i)
        # Wait: delta(k-j,i) means j=k-i, and a[j]=a[k-i] is valid when 0<=k-i<n
        # So both terms give a[k-i], total linear: 2*d*a[k-i] (when 0<=k-i<n)
        # And the d^2 term: when j=i AND k-j=i, i.e., k=2i: d^2

        # So: c_new[k] = c[k] + 2*d*s[k] + d^2*(1 if k==2i else 0)
        # where s[k] = a[k-i] if 0<=k-i<n, else 0
        # This is what we have above. Good.

        # Try candidate d values
        if ai > 1e-10:
            ratios = [0.1, 0.3, 0.5, 0.8]
            d_candidates = [-ai * r for r in ratios] + [ai * r for r in ratios]
            d_candidates.append(-ai * 0.99)
        else:
            d_candidates = [S / n * 0.1, S / n * 0.5]

        best_d = 0.0
        best_sc = score

        for d in d_candidates:
            if ai + d < -1e-15:
                continue
            if S + d < 0.01:
                continue
            new_c_max = (c + 2.0 * d * s).max()
            if 2 * i < 2 * n - 1:
                val_at_2i = c[2 * i] + 2.0 * d * s[2 * i] + d * d
                if val_at_2i > new_c_max:
                    new_c_max = val_at_2i
            new_score = 2.0 * n * new_c_max / ((S + d) * (S + d))
            if new_score < best_sc - 1e-14:
                best_sc = new_score
                best_d = d

        if best_d != 0.0:
            # Update
            a[i] += best_d
            S += best_d
            c = c + 2.0 * best_d * s
            if 2 * i < 2 * n - 1:
                c[2 * i] += best_d * best_d
            score = best_sc
            c = np.maximum(c, 0.0)  # numerical safety

    return a, c, S, score


def _coord_descent(a, max_sweeps=200, deadline=None, verbose=False):
    """Multiple coordinate descent sweeps with random restarts of ordering."""
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    score = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng(999)

    for sw in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline - 5:
            break

        t0 = time.monotonic()
        a, c, S, score = _coord_descent_sweep(a, c, S, n, deadline)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose and sw % 10 == 0:
                print(f"    CoordDescent sweep {sw}: score={best_score:.10f} "
                      f"({time.monotonic()-t0:.3f}s/sweep)")
        else:
            # No improvement, try a random perturbation to escape
            if sw % 20 == 19:
                idx = rng.integers(0, n)
                a[idx] *= rng.uniform(0.5, 2.0)
                S = float(a.sum())
                c = _full_convolve(a, a)
                score = 2.0 * n * c.max() / (S * S)

    return best_a, best_score


def _coord_descent_verbose(a, max_sweeps=500, deadline=None, print_every=50):
    """Coordinate descent with more detailed logging and larger perturbations."""
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    score = 2.0 * n * c.max() / (S * S)
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng(777)
    no_improve = 0

    for sw in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline - 5:
            break

        t0 = time.monotonic()
        a, c, S, score = _coord_descent_sweep(a, c, S, n, deadline)

        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            # Larger random perturbation
            n_perturb = rng.integers(5, max(10, n // 20))
            idxs = rng.choice(n, size=n_perturb, replace=False)
            for idx in idxs:
                a[idx] *= rng.uniform(0.3, 3.0)
            a = np.maximum(a, 0)
            S = float(a.sum())
            if S < 0.01:
                S = 1.0
                a = a / a.sum() * S
            c = _full_convolve(a, a)
            score = 2.0 * n * c.max() / (S * S)
            no_improve = 0

        if sw % print_every == 0 and sw > 0:
            print(f"    CD verbose sweep {sw}: best={best_score:.10f}, "
                  f"no_improve={no_improve}, time={time.monotonic()-t0:.2f}s")

    return best_a, best_score


def _make_structured_init(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "triangle":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 2.0 * (2.0 * t - 1.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bilateral":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "barker_like":
        # Based on Barker sequence pattern [1,1,1,-1,1,-1,-1] scaled to non-neg
        base = np.array([3, 3, 3, 1, 3, 1, 1], dtype=float)
        repeats = max(1, n // len(base))
        a = np.tile(base, repeats)[:n]
        a = a / a.sum()
        return a
    else:
        return np.ones(n) / n


def _perturb_large(a, rng, scale=0.5):
    """Large random perturbation to escape local minima."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + scale * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, block_size=None):
    """Perturb a random block of the sequence."""
    n = len(a)
    if block_size is None:
        block_size = rng.integers(n // 8, n // 2)
    start = rng.integers(0, max(1, n - block_size))
    z = np.log(np.maximum(a, 1e-15))
    z[start:start + block_size] += rng.uniform(-1.0, 1.0, block_size)
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimizer: L-BFGS + coordinate descent + FISTA + random restarts.
    Minimizes 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_structured_init(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}, score={_score(a0):.10f}")
        update_best(a0, _score(a0), "init_uniform")

    n = n_main

    # --- Phase 1: L-BFGS with aggressive beta schedule ---
    print(f"\nPhase 1: L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    c_cur = _full_convolve(a0, a0)
    max_c = float(c_cur.max())
    betas = max_c * np.array([0.5, 0.25, 0.125, 0.0625, 0.03125,
                              0.015625, 0.0078, 0.0039, 0.002, 0.001, 0.0005])
    current_a = a0.copy()
    current_sc = sc0

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 300:
            break
        remaining = deadline - 300 - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.004 * (n / 2048.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 10))))
        a_new = _lbfgs_phase(current_a, beta, n, time.monotonic() + min(remaining, 60), max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-13:
            current_a = a_new
            current_sc = sc
        update_best(current_a, current_sc, f"p1_lbfgs_{i}")

    # --- Phase 2: FISTA ---
    print(f"\nPhase 2: FISTA (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        c_b = _full_convolve(current_a, current_a)
        mb = float(c_b.max())
        fista_beta = mb * 0.001
        a_fista, sc_fista = _fista_optimize(current_a, fista_beta, n,
                                            max_iter=20000,
                                            deadline=deadline - 200)
        update_best(a_fista, sc_fista, "fista")

    # --- Phase 3: Coordinate descent (main new technique) ---
    print(f"\nPhase 3: Coordinate Descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        cd_deadline = deadline - 80
        a_cd, sc_cd = _coord_descent(best_seq, max_sweeps=500, deadline=cd_deadline)
        update_best(a_cd, sc_cd, "coord_descent")

    # --- Phase 4: Interleave L-BFGS + CD ---
    print(f"\nPhase 4: Interleave (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        for cycle in range(5):
            if time.monotonic() > deadline - 40:
                break

            # Short L-BFGS
            if time.monotonic() < deadline - 30:
                c_b = _full_convolve(best_seq, best_seq)
                mb = float(c_b.max())
                beta = mb * 0.0005
                a_tmp = _lbfgs_phase(best_seq, beta, n, time.monotonic() + 20, max_iter=500)
                sc_tmp = _score(a_tmp)
                if sc_tmp < best_score:
                    best_seq = a_tmp
                    best_score = sc_tmp
                    update_best(a_tmp, sc_tmp, f"p4_lbfgs_{cycle}")

            # Coordinate descent
            if time.monotonic() < deadline - 25 and best_seq is not None:
                a_tmp2, sc_tmp2 = _coord_descent(best_seq, max_sweeps=50,
                                                 deadline=deadline - 25)
                update_best(a_tmp2, sc_tmp2, f"p4_cd_{cycle}")

    # --- Phase 5: Random restarts with large perturbations ---
    print(f"\nPhase 5: Random Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        n_restarts = min(6, max(1, int((deadline - 20 - time.monotonic()) / 60)))
        print(f"  {n_restarts} restarts available")

        for r in range(n_restarts):
            if time.monotonic() > deadline - 15:
                break

            if r == 0:
                p_start = _perturb_large(best_seq, rng, scale=0.3)
            elif r == 1:
                p_start = _perturb_block(best_seq, rng)
            elif r == 2:
                p_start = _perturb_large(best_seq, rng, scale=0.8)
            elif r == 3:
                p_start = _perturb_block(best_seq, rng, block_size=n // 16)
            elif r == 4:
                # Mirror perturbation
                p_start = best_seq.copy()
                half = n // 2
                p_start[half:] = best_seq[:half][::-1]
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb_large(best_seq, rng, scale=0.5)

            sc_start = _score(p_start)
            print(f"  Restart {r}: start_score={sc_start:.10f}")

            # Quick FISTA
            if time.monotonic() < deadline - 10:
                c_r = _full_convolve(p_start, p_start)
                mb_r = float(c_r.max())
                a_r, sc_r = _fista_optimize(p_start, mb_r * 0.001, n,
                                            max_iter=5000,
                                            deadline=deadline - 10)
                update_best(a_r, sc_r, f"restart_{r}_fista")

            # Coordinate descent on the result
            if time.monotonic() < deadline - 5 and best_seq is not None:
                a_r2, sc_r2 = _coord_descent(
                    a_r if sc_r < best_score else best_seq,
                    max_sweeps=30, deadline=deadline - 5)
                update_best(a_r2, sc_r2, f"restart_{r}_cd")

    # --- Phase 6: Final verbose coordinate descent ---
    print(f"\nPhase 6: Final CD Polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_final, sc_final = _coord_descent_verbose(
            best_seq, max_sweeps=200, deadline=deadline - 2, print_every=50)
        update_best(a_final, sc_final, "final_cd")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_structured_init(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_structured_init(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
