Let me try a completely different approach: **iterative peak shaving**.

Algorithm:
1. Start with a sequence (e.g., from parent or GMM init)
2. Compute convolution c = a*a
3. Find the peak position k* where c is maximized
4. The peak value is c[k*] = sum_i a[i]*a[k*-i]
5. To reduce this peak, we need to reduce some of the products a[i]*a[k*-i]
6. The most effective strategy is to reduce the largest contributing pair
7. But reducing a[i] might increase other peaks
8. So we need a balanced approach: reduce a[i] in proportion to how much it contributes to the peak vs. other peaks

This is essentially a "spectral erosion" or "peak shaving" algorithm.

A simpler version:
1. Find the top-k peaks in the convolution
2. For each peak, find the top contributing index pairs
3. Reduce the a[i] values that contribute most to multiple peaks simultaneously
4. Renormalize

Let me implement this as a local search:

```python
def _peak_shave(a, max_iter=10000):
    n = len(a)
    best_score = _score(a)
    a = a.copy()
    for it in range(max_iter):
        c = _full_convolve(a, a)
        cm = c.max()
        # Find top peaks
        peak_idx = np.argmax(c)
        # Find top contributors to this peak
        k = peak_idx
        valid_i = np.arange(max(0, k-n+1), min(n, k+1))
        products = a[valid_i] * a[k-valid_i]
        top_contributors = valid_i[np.argsort(products)[::-1][:10]]
        # For each contributor, try reducing it
        best_delta = 0
        best_new_score = best_score
        best_idx = -1
        for idx in top_contributors:
            for delta in [0.01, 0.005, 0.001, 0.002]:
                idx_frac = a[idx] * delta / a.sum()
                a_trial = a.copy()
                a_trial[idx] *= (1 - idx_frac)
                s_new = _score(a_trial)
                if s_new < best_new_score:
                    best_new_score = s_new
                    best_delta = idx_frac
                    best_idx = idx
        if best_idx >= 0:
            a[best_idx] *= (1 - best_delta)
            best_score = best_new_score
        else:
            # Try increasing elements that don't contribute to the peak
            non_peak = np.setdiff1d(np.arange(n), top_contributors)
            random_idx = non_peak[np.random.randint(len(non_peak))]
            a[random_idx] *= 1.01
            s_new = _score(a)
            if s_new > best_score:
                a[random_idx] /= 1.01
            else:
                best_score = s_new
    return a, best_score
```

```python
def _softmax_vec(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):

def _gmm_seq(n, n_comp, means, stds, weights):
    x = np.arange(n, dtype=np.float64)
    a = np.zeros(n)
    for k in range(n_comp):
        s = stds[k]
        if s < 1e-10:
            a[int(np.clip(means[k], 0, n - 1))] += weights[k]
        else:
            a += weights[k] * np.exp(-0.5 * ((x - means[k]) / s) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _make_ramp(n, slope=1.5):

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _run_lbfgs(a_init, deadline, rng, n_betas=15):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    max_c0 = float(c0.max())

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:

        time_budget = min(time_budget, 200.0)
        level_deadline = time.monotonic() + max(5.0, time_budget)

    lr = lr_init

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_vec(z)
        c = _full_convolve(a, a)

        # Adaptive log-sum-exp temperature
        cm = float(c.max())
        if cm < 1e-30:
            break
        sorted_c = np.sort(c)[::-1]
        k = min(10, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k] if k > 0 else sorted_c[0]
        beta = max(1e-10, 0.2 * gap) if gap > 0.01 * cm else max(1e-10, 0.05 * cm)

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()

        sc = _score_norm(_softmax_vec(z))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _softmax_vec(z).copy()

        if t % 3000 == 0:
            lr = max(0.005, lr * 0.85)

        if t % 15000 == 0:
            z = z + 0.05 * rng.normal(size=n)
            z = np.clip(z, -70, 70)
            z = z - z.mean()

    return best_a, best_score


def _peak_shave(a_init, deadline, rng):
    """Iterative peak-shaving: find dominant peak, reduce its contributors."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    # Track recent scores for convergence detection
    no_improve = 0
    patience = 500

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(c)[::-1][:top_k]

        # For the dominant peak, find contributors
        k = int(peak_indices[0])
        valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
        if len(valid_i) == 0:
            break
        products = a[valid_i] * a[k - valid_i]
        total_prod = products.sum()
        if total_prod < 1e-15:
            break

        # Weighted contribution fraction
        contrib_frac = products / total_prod
        top_frac = np.argsort(contrib_frac)[::-1][:10]

        # Try reducing top contributors
        improved = False
        for pos in top_frac:
            idx = int(valid_i[pos])
            # Compute how much this index contributes to other peaks too
            # (we want to reduce indices that contribute a lot to the peak
            #  but not too much to other peaks)
            for frac in [0.02, 0.01, 0.005, 0.002, 0.015]:
                a_trial = a.copy()
                a_trial[idx] *= (1.0 - frac)
                if a_trial[idx] < 0:
                    continue
                sc = _score_norm(a_trial)
                if np.isfinite(sc) and sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
                elif np.isfinite(sc) and sc < _score_norm(a) - 1e-10:
                    # Accept if it improves over current (though not over best)
                    a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Try opposite: increase a small element far from the peak
            if no_improve % 10 == 0:
                # Find elements with low contribution to the peak
                low_contrib = np.argsort(contrib_frac)[:5]
                for pos in low_contrib:
                    idx = int(valid_i[pos])
                    for frac in [0.02, 0.05, 0.1]:
                        a_trial = a.copy()
                        a_trial[idx] *= (1.0 + frac)
                        sc = _score_norm(a_trial)
                        if np.isfinite(sc) and sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            improved = True
                            break
                        elif np.isfinite(sc) and sc < _score_norm(a) - 1e-10:
                            a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break

            # Random perturbation if stuck
            if no_improve % 50 == 0:
                n_pert = rng.integers(1, 6)
                pert_idx = rng.choice(n, size=n_pert, replace=False)
                for idx in pert_idx:
                    direction = rng.choice([-1, 1])
                    for frac in [0.01, 0.02, 0.05]:
                        a_trial = a.copy()
                        a_trial[idx] *= (1.0 + direction * frac)
                        if a_trial[idx] >= 0:
                            sc = _score_norm(a_trial)
                            if np.isfinite(sc) and sc < _score_norm(a) - 1e-12:
                                a = a_trial.copy()
                                if sc < best_score:
                                    best_score = sc
                                    best_a = a.copy()
                                    no_improve = 0
                                break

            if no_improve > patience:
                if patience < 3000:
                    patience *= 1.5
                else:
                    break
        else:
            no_improve = 0

    return best_a, best_score


def _coord_descent(a_init, deadline, rng):
    """Coordinate descent: try small changes to individual elements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    no_improve = 0

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        peak_idx = int(np.argmax(c))

        # Focus on indices that contribute to the peak
        valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        products = a[valid_i] * a[peak_idx - valid_i]
        contrib_order = np.argsort(products)[::-1]

        # Try reducing top contributors and increasing low contributors
        candidates = list(contrib_order[:8]) + list(contrib_order[-5:])
        rng.shuffle(candidates)

        improved = False
        for pos in candidates[:12]:
            idx = int(valid_i[pos])
            direction = -1 if products[pos] > cm * 0.1 else 1
            for frac in [0.005, 0.01, 0.02]:
                a_trial = a.copy()
                a_trial[idx] *= (1.0 + direction * frac)
                if a_trial[idx] < 0:
                    continue
                sc = _score_norm(a_trial)
                if np.isfinite(sc) and sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
                elif np.isfinite(sc) and sc < _score_norm(a) - 1e-10:
                    a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(512, 1.5)
        update_best(a0, _score_norm(a0), "default_ramp")

    n_target = 4096

    # --- Phase 1: diverse initialization + quick optimization (15% time) ---
    phase1_end = t_start + (budget_s - 30.0) * 0.15
    print(f"\nPhase 1: Diverse initialization (deadline t={phase1_end - t_start:.0f}s)")

    # Try GMM initializations at target n
    gmm_candidates = []
    for i in range(30):
        g = _random_gmm(n_target, rng, n_comp=rng.integers(2, 7))
        gmm_candidates.append(g)
    # Pick top 3 by score
    gmm_scores = [_score_norm(g) for g in gmm_candidates]
    gmm_order = np.argsort(gmm_scores)
    top_gmm = [gmm_candidates[i] for i in gmm_order[:3]]
    print(f"  Best GMM init scores: {[f'{gmm_scores[i]:.6f}' for i in gmm_order[:5]]}")

    # Also try polynomial inits
    poly_candidates = []
    for i in range(20):
        d = int(rng.integers(3, 8))
        coeffs = rng.uniform(0.1, 3.0, d)
        poly_candidates.append(_polynomial_seq(n_target, coeffs, rng))
    poly_scores = [_score_norm(p) for p in poly_candidates]
    poly_order = np.argsort(poly_scores)
    top_poly = [poly_candidates[i] for i in poly_order[:3]]

    # Combine: try quick L-BFGS on top candidates
    all_inits = top_gmm + top_poly
    if parent is not None:
        # Upsample parent to n_target if needed
        p_up = _upsample(parent, n_target) if len(parent) != n_target else parent.copy()
        all_inits.append(p_up)

    best_init = None
    best_init_score = np.inf
    for init in all_inits:
        if time.monotonic() > phase1_end - 5:
            break
        quick_dl = min(phase1_end - 2, time.monotonic() + 8.0)
        a_opt, sc_opt = _run_lbfgs(init, quick_dl, rng, n_betas=6)
        if sc_opt < best_init_score:
            best_init_score = sc_opt
            best_init = a_opt
        update_best(a_opt, sc_opt, "phase1_init")

    if best_init is None:
        best_init = all_inits[0] if all_inits else _make_ramp(n_target, 1.5)
        best_init_score = _score_norm(best_init)

    # --- Phase 2: Multi-scale L-BFGS (35% time) ---
    phase2_end = t_start + (budget_s - 30.0) * 0.50
    print(f"\nPhase 2: Multi-scale L-BFGS (deadline t={phase2_end - t_start:.0f}s)")

    # Multi-scale: optimize at smaller n, upsample
    cur_n = 512
    a_cur = _upsample(best_init, cur_n)
    sc_cur = _score_norm(a_cur)
    print(f"  Start n={cur_n}: score_norm={sc_cur:.10f}")

    scales = [1024, 2048, 4096]
    for target_n in scales:
        if time.monotonic() > phase2_end - 10:
            break
        a_cur = _upsample(a_cur, target_n)
        sc_cur = _score_norm(a_cur)
        print(f"  Upsampled to n={target_n}: score_norm={sc_cur:.10f}")

        opt_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.8)
        a_opt, sc_opt = _run_lbfgs(a_cur, opt_dl, rng, n_betas=15)
        print(f"  After n={target_n} L-BFGS: score_norm={sc_opt:.10f}")
        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt
        update_best(a_cur, sc_cur, f"multiscale_{target_n}")

    # Direct L-BFGS from best_init at n_target
    if time.monotonic() < phase2_end - 15:
        dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        a_opt, sc_opt = _run_lbfgs(best_init, dl, rng, n_betas=20)
        update_best(a_opt, sc_opt, "direct_lbfgs")
        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    # --- Phase 3: Adam refinement (25% time) ---
    phase3_end = t_start + (budget_s - 30.0) * 0.75
    print(f"\nPhase 3: Adam refinement (deadline t={phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        # Adam from best
        gd_dl = min(phase3_end - 20, time.monotonic() + (phase3_end - time.monotonic()) * 0.6)
        a_gd, sc_gd = _adam_refine(best_seq, gd_dl, rng, lr_init=0.06)
        update_best(a_gd, sc_gd, "adam")

        # Peak shaving from best
        if time.monotonic() < phase3_end - 10:
            ps_dl = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
            a_ps, sc_ps = _peak_shave(best_seq, ps_dl, rng)
            update_best(a_ps, sc_ps, "peak_shave")

        # Adam from peak-shaved result
        if time.monotonic() < phase3_end - 5 and sc_ps < best_score:
            gd_dl2 = min(phase3_end - 3, time.monotonic() + 30.0)
            a_gd2, sc_gd2 = _adam_refine(a_ps, gd_dl2, rng, lr_init=0.03)
            update_best(a_gd2, sc_gd2, "adam_after_shave")

    print(f"\n  Phase 3 complete: best_score={best_score:.10f}")

    # --- Phase 4: Local search polish (25% time) ---
    phase4_end = deadline
    print(f"\nPhase 4: Local search polish (deadline t={phase4_end - t_start:.0f}s)")

    if time.monotonic() < phase4_end - 5 and best_seq is not None:
        # Coordinate descent
        cd_dl = min(phase4_end - 20, time.monotonic() + (phase4_end - time.monotonic()) * 0.4)
        a_cd, sc_cd = _coord_descent(best_seq, cd_dl, rng)
        update_best(a_cd, sc_cd, "coord_descent")

        # More peak shaving
        if time.monotonic() < phase4_end - 10:
            ps_dl2 = min(phase4_end - 5, time.monotonic() + (phase4_end - time.monotonic()) * 0.4)
            a_ps2, sc_ps2 = _peak_shave(best_seq, ps_dl2, rng)
            update_best(a_ps2, sc_ps2, "peak_shave2")

        # Final tiny L-BFGS
        if time.monotonic() < phase4_end - 3 and best_seq is not None:
            n_final = len(best_seq)
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            beta_tiny = max(1e-6 * max_c_final, 1e-14)
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta_tiny, n_final),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12,
                             'maxls': 30, 'maxfun': 20000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_vec(result.x)
                    sc_new = _score_norm(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_seq = a_new.copy()
                        print(f"  Final L-BFGS: {best_score:.10f}")
            except Exception:
                pass

        # One more coordinate descent pass
        if time.monotonic() < phase4_end - 3 and best_seq is not None:
            cd_dl2 = min(phase4_end - 1, time.monotonic() + 15.0)
            a_cd2, sc_cd2 = _coord_descent(best_seq, cd_dl2, rng)
            update_best(a_cd2, sc_cd2, "coord_descent_final")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(4096, 1.5)

        sc_final = _score(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        elapsed = time.monotonic() - t_start
        print(f"Total time: {elapsed:.1f}s")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
