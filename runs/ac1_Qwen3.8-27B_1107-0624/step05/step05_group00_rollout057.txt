"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses FISTA with adaptive beta-continuation,
Adam in log-space with aggressive scheduling, local coordinate search,
and diverse restart strategies.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1, so just 2*n*max(a*a)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Gradient of sum(w * (a*a)) with respect to a.
    (a*a)_k = sum_i a_i * a_{k-i}
    d/d a_j [sum_k w_k * (a*a)_k] = 2 * sum_k w_k * a_{k-j} (with valid indexing)
    This equals 2 * (w * rev(a))_j evaluated at the right positions.
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # wc has length 2n-1, we want positions n-1 to 2n-2 (length n)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_from_z(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 8.0 * n  # Lipschitz constant estimate (use a larger value for safety)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_conv(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_from_z(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            # Gradient of sum(w * (a*a)) w.r.t. z (where a = softmax(z)/sum(softmax(z))... 
            # Actually a = softmax(z) directly since we normalize)
            # d/dz_j [sum_k w_k * (a*a)_k] where a_i = exp(z_i)/sum(exp(z))
            # = 2 * sum_k w_k * a_{k-j} * (delta_{jk} - a_k)  ... this gets complicated
            # Simpler: compute grad w.r.t. a, then chain rule
            # grad_a = 2 * (w * rev(a))[n-1:2n-1]
            # grad_z = grad_a * a - a * sum(grad_a * a)  (for softmax parameterization)
            # But since we're on the simplex (sum a = 1), the gradient w.r.t. z is:
            # grad_z_j = grad_a_j - sum_i(grad_a_i * a_i)  (projection to tangent space)
            
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad_a = 2.0 * h
            # Project to tangent space of simplex
            grad_z = grad_a - np.dot(grad_a, a)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_from_z(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        # End of beta phase, optionally decay lr
        current_lr *= 0.85

    a_final = _softmax_from_z(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _local_search(a, deadline, n_trials=5000):
    """Local coordinate descent search: try adjusting pairs of coordinates."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    # Precompute convolution for fast updates
    # When we change a[i] by delta and a[j] by -delta (to maintain sum=1),
    # the change in convolution is:
    # (a*a)_k changes by: delta*(a_{k-i} + a_{k-j}) - delta*(a_{k-i} + a_{k-j}) ... 
    # This is complex, so let's just recompute for random pairs
    
    c = _full_convolve(a, a)
    
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        # Pick two random indices and a small perturbation
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        delta = rng.uniform(-0.01, 0.01) * best_a[i]
        
        if best_a[i] + delta < 0 or best_a[j] - delta < 0:
            continue
        
        a_trial = best_a.copy()
        a_trial[i] += delta
        a_trial[j] -= delta
        
        sc = _score_norm(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial
    
    return best_a, best_score

    profiles = []

def _interp_profile(p, n_new):

    if n0 == n_new:

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_from_z(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _grad_a_from_w(a, w):
    """Compute grad_a = 2*(w * rev(a))[n-1:2n-1], the gradient w.r.t. a
    of sum_k w_k * (a*a)_k."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return 2.0 * wc[n - 1:2 * n - 1]


def _grad_z_correct(a, w, n):
    """Correct gradient w.r.t. z for softmax parameterization a=softmax(z).
    grad_z_j = a_j * (grad_a_j - a·grad_a)
    """
    grad_a = _grad_a_from_w(a, w)
    dot_val = np.dot(a, grad_a)
    return a * (grad_a - dot_val)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max and adaptive step size."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    # Adaptive Lipschitz estimate
    L = 4.0 * n
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_a_from_w(y, w)
        a_next = _proj_simplex(y - step * grad)

        # Simple backtracking: if score increases, increase L
        if it > 0 and it % 50 == 0:
            sc = _score_norm(a_next)
            if sc > best_score * 1.05:
                L *= 1.5
                step = 1.0 / L
            elif sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with CORRECTED softmax gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_from_z(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            # CORRECT gradient: a * (grad_a - a·grad_a)
            grad_z = _grad_z_correct(a, w, n)
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_from_z(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

        # Decay LR between beta phases
        current_lr *= 0.7

    a_final = _softmax_from_z(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_v2(a_init, beta, deadline, max_iter):
    """Improved FISTA with steepest-descent fallback when momentum stalls."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    stale_iters = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _grad_a_from_w(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 30 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
                stale_iters = 0
            else:
                stale_iters += 1

        # If stuck for too long, reset momentum and try steeper step
        if stale_iters > 5:
            a_prev = a_curr.copy()
            t_val = 1.0
            step = min(step * 1.3, 1.0 / (n * 0.5))
            stale_iters = 0

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_from_z(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_main = 4096
    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial: max_conv={max_c0:.8f}", flush=True)

    # === Phase 1: FISTA coarse-to-fine (400s) ===
    print("=== Phase 1: FISTA coarse-to-fine ===", flush=True)
    phase1_end = t_start + 400.0
    a_curr = a0.copy()

    # Generate detailed beta schedule
    betas = []
    beta = 0.08 * max_c0
    while beta > 0.00002 * max_c0:
        betas.append(beta)
        beta *= 0.6  # Slower decay for more iterations per level
    if len(betas) < 20:
        while len(betas) < 20:
            betas.append(betas[-1] * 0.5)
    betas = betas[:25]

    for i, beta in enumerate(betas):
        if time.monotonic() > phase1_end - 3:
            break
        per_time = min(35.0, phase1_end - time.monotonic() - 2)
        if per_time < 3:
            break
        max_it = int(per_time * 2500)
        a_curr, sc = _fista_v2(a_curr, beta, time.monotonic() + per_time, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            if i % 3 == 0 or sc_full < best_score + 0.001:
                print(f"  FISTA beta={beta:.6f}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)

    print(f"End Phase 1: best={best_score:.10f}", flush=True)

    # === Phase 2: Adam log-space with correct gradient (500s) ===
    print("=== Phase 2: Adam log-space (corrected grad) ===", flush=True)
    phase2_end = t_start + 900.0
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    # Fine beta schedule for Adam
    betas_fine = []
    beta = 0.004 * max_cw
    while beta > 0.000002 * max_cw:
        betas_fine.append(beta)
        beta *= 0.5
    betas_fine = betas_fine[:18]

    per_time = phase2_end - time.monotonic() - 10
    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.08, max_iter_total=2000000
        )
        sc_full = _score(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)
        else:
            print(f"  Adam: no improvement (got {sc_full:.10f})", flush=True)

    print(f"End Phase 2: best={best_score:.10f}", flush=True)

    # === Phase 3: Multi-restart with diverse structures (500s) ===
    print("=== Phase 3: Multi-restart ===", flush=True)
    phase3_end = t_start + 1400.0
    if time.monotonic() < phase3_end - 30:
        n_restarts = 10
        per_restart = (phase3_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break

            a_base = best_seq / best_seq.sum()

            # Diverse initialization strategies
            if i == 0:
                p_start = _perturb(a_base, 0.015, rng)
            elif i == 1:
                n = len(a_base)
                t = np.linspace(0, 1, n)
                mod = 1.0 + 0.08 * np.cos(2 * np.pi * 5 * t)
                p_start = a_base * mod
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 2:
                n = len(a_base)
                t = np.linspace(0, 1, n)
                mod = 1.0 + 0.08 * np.cos(2 * np.pi * 7 * t + 0.5)
                p_start = a_base * mod
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i == 3:
                # Palindromic projection
                p_start = (a_base + a_base[::-1]) / (2.0 * a_base.sum())
            elif i == 4:
                p_start = _perturb(a_base, 0.04, rng)
            elif i == 5:
                # Taper ends more
                n = len(a_base)
                t = np.linspace(0, 1, n)
                taper = np.minimum(1.0, np.minimum(t, 1.0 - t) / 0.05 + 1.0)
                p_start = a_base * taper
                p_start = p_start / p_start.sum()
            elif i == 6:
                # Blend with Gaussian
                n = len(a_base)
                t = np.linspace(0, 1, n)
                gauss = np.exp(-0.5 * ((t - 0.5) / 0.35) ** 2)
                p_start = 0.6 * a_base + 0.4 * (gauss / gauss.sum())
            elif i == 7:
                # Blend with raised cosine
                n = len(a_base)
                t = np.linspace(0, 1, n)
                rc = 0.5 * (1 + np.cos(2 * np.pi * t))
                rc = np.maximum(rc, 1e-10)
                p_start = 0.5 * a_base + 0.5 * (rc / rc.sum())
            elif i == 8:
                # Anti-palindromic perturbation
                n = len(a_base)
                p_start = a_base + 0.05 * (a_base - a_base[::-1])
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # Frequency domain perturbation
                n = len(a_base)
                fa = np.fft.rfft(a_base)
                fa *= (1.0 + 0.03 * rng.standard_normal(fa.shape))
                p_start = np.fft.irfft(fa, n * 2)[:n]
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            # Optimize this starting point
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())

            # Short FISTA warmup
            beta_f = max_cs * 0.005
            time_fista = min(8.0, per_restart * 0.25)
            p_mid, _ = _fista_v2(p_start, beta_f, time.monotonic() + time_fista,
                                  int(time_fista * 2000))
            p_mid = p_mid / p_mid.sum()

            # Adam polish
            c_mid = _full_convolve(p_mid, p_mid)
            max_cm = float(c_mid.max())
            betas_a = [0.002 * max_cm, 0.0008 * max_cm, 0.0003 * max_cm,
                       0.0001 * max_cm, 0.00004 * max_cm]
            time_adam = per_restart * 0.7 - 3
            if time_adam > 5:
                p_best, _ = _adam_logspace(
                    p_mid, time.monotonic() + time_adam,
                    betas_a, lr_init=0.05, max_iter_total=100000
                )
                sc_full = _score(p_best)
                if sc_full < best_score:
                    best_score = sc_full
                    best_seq = p_best.copy()
                    print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)

    print(f"End Phase 3: best={best_score:.10f}", flush=True)

    # === Phase 4: Multi-n exploration (200s) ===
    print("=== Phase 4: Multi-n ===", flush=True)
    phase4_end = t_start + 1600.0
    if time.monotonic() < phase4_end - 20:
        for n_try in [n_main * 2, n_main // 2]:
            if time.monotonic() > phase4_end - 15:
                break
            if n_try == len(best_seq) or n_try < 512 or n_try > 16384:
                continue

            a_try = _interp_profile(best_seq, n_try)
            sc_try = _score(a_try)
            print(f"  n={n_try}: interpolated score={sc_try:.10f}", flush=True)

            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()

            # Optimize
            if time.monotonic() < phase4_end - 10:
                c_try = _full_convolve(a_try, a_try)
                max_ct = float(c_try.max())

                # FISTA
                beta_t = 0.004 * max_ct

                                            int(t_f * 2000))
                else:
                    a_t_opt = a_try

                # Adam
                c_t = _full_convolve(a_t_opt, a_t_opt)
                max_ct2 = float(c_t.max())
                betas_t = [0.002 * max_ct2, 0.0008 * max_ct2, 0.0003 * max_ct2,
                           0.0001 * max_ct2]
                t_a = phase4_end - time.monotonic() - 5
                if t_a > 5:
                    a_t_opt, _ = _adam_logspace(
                        a_t_opt / a_t_opt.sum(), time.monotonic() + t_a,
                        betas_t, lr_init=0.06, max_iter_total=200000
                    )
                    sc_t_full = _score(a_t_opt)
                    if sc_t_full < best_score:
                        best_score = sc_t_full
                        best_seq = a_t_opt.copy()
                        print(f"  n={n_try} optimized: score={sc_t_full:.10f}", flush=True)

    # === Phase 5: Final ultra-fine polish (remaining time) ===
    print("=== Phase 5: Final polish ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 5:

                     0.000003 * max_cp, 0.000001 * max_cp]

        if per_t > 10:

            if sc_pol_full < best_score:

        sc_final = _score(np.array(final))

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
