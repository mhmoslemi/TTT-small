    if beta_init is None:
        c = _full_convolve(a_init, a_init)
        cm = float(c.max())
        sorted_c = np.sort(c)[::-1]
        k = min(K_focus, len(sorted_c) - 1)
        gap = sorted_c[0] - sorted_c[k]
        beta = max(1e-10, 0.3 * gap) if gap > 0.01 * cm else max(1e-10, 0.05 * cm)
    else:
        beta = beta_init
    
    lr_start = lr
    patience = 0
    patience_limit = 5000
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta: decrease over time
        if it % 200 == 0:
            sorted_c = np.sort(c)[::-1]
            kk = min(K_focus, len(sorted_c) - 1)
            gap = sorted_c[0] - sorted_c[kk]
            if gap > 0.005 * cm:
                beta_new = max(1e-10, 0.3 * gap)
            else:
                beta_new = max(1e-10, 0.005 * cm)
            beta = beta_new
        
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        
        w_rev = w[::-1]
        ac = _full_convolve(a, w_rev)
        h = ac[n - 1:2 * n - 1]
        
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            if sc < best_score - 1e-10:
                patience = 0
            else:
                patience += 1
            best_score = sc
            best_a = a.copy()
        else:
            patience += 1
        
        # Learning rate decay
        if t % 5000 == 0:
            lr = max(0.001, lr_start * 0.8 ** (t // 5000))
        
        # Perturbation if stuck
        if patience > patience_limit:
            patience = 0
            noise_scale = 0.05 + 0.05 * rng.uniform()
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            lr = lr_start * 0.5
            patience_limit = max(2000, patience_limit // 2)
    
    return best_a, best_score


def _peak_surgery(a, deadline, rng, max_rounds=500, K=5):
    """Direct multiplicative peak equalization."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c, cm, top_k, top_vals = _conv_and_peaks(a_cur, K=K)
        
        # Weight peaks by how close they are to max
        thresholds = top_vals[:1] * 0.98  # peaks within 2% of max
        active_mask = top_vals >= thresholds[0]
        active_peaks = top_k[active_mask]
        active_vals = top_vals[active_mask]
        
        if len(active_peaks) == 0:
            active_peaks = top_k[:1]
            active_vals = top_vals[:1]
        
        # Compute pressure
        # Weights: peaks closer to max get higher weight
        max_v = active_vals[0]
        weights = np.maximum(0.1, 1.0 - (max_v - active_vals) / max_v)
        weights = weights / weights.sum()
        
        grad_a = _pressure_from_peaks(a_cur, active_peaks, weights)
        
        # Multiplicative update: reduce a[j] where pressure is high, increase where low
        mean_pressure = np.mean(grad_a)
        # Normalize pressure to have zero mean (maintain sum)
        pressure = grad_a - mean_pressure
        
        # Adaptive step size
        step = 0.01 * (1.0 + 0.5 * rng.uniform())
        
        z_trial = z - step * pressure / (np.abs(pressure).max() + 1e-10)
        z_trial = np.clip(z_trial, -60, 60)
        z_trial = z_trial - z_trial.mean()
        a_trial = _softmax_fn(z_trial)
        
        sc = _score_norm(a_trial)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a_trial.copy()
            z = z_trial.copy()
        else:
            # Try smaller step
            for s in [0.005, 0.002, 0.001]:
                z_trial = z - s * pressure / (np.abs(pressure).max() + 1e-10)
                z_trial = np.clip(z_trial, -60, 60)
                z_trial = z_trial - z_trial.mean()
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if np.isfinite(sc) and sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    break
    
    return best_a, best_score


def _lbfgs_refine(a_init, deadline, beta, n_betas=8):
    """L-BFGS with continuation on beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z = np.clip(z, -60, 60)
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    if beta <= 0:
        c = _full_convolve(a_init, a_init)
        beta = 0.01 * float(c.max())
    
    betas = np.geomspace(beta, max(beta * 1e-4, 1e-12), n_betas)
    
    for idx, b in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 120.0)
        dl = time.monotonic() + time_per
        
        maxiter = 3000 if b > 0.01 * float(_full_convolve(a_init, a_init).max()) else 8000
        
        try:
            result = minimize(
                _lse_obj, z, args=(b, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 80, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -60, 60)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _make_ramp(n, slope=2.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):

    deadline = t_start + max(0.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(4096, 2.0)
        update_best(a0, _score_norm(a0), "default_ramp")
    
    print(f"\nStarting optimization, deadline in {deadline - t_start:.0f}s")
    
    # === Phase 1: Fine optimization of parent (50% of time) ===
    phase1_end = t_start + (budget_s - 30.0) * 0.50
    print(f"\n=== Phase 1: Fine optimization (t={phase1_end - t_start:.0f}s) ===")
    
    a_cur = best_seq.copy()
    n_cur = len(a_cur)
    
    # 1a: Adam optimization with sharp beta
    adam_dl = min(phase1_end - 30, t_start + (budget_s - 30.0) * 0.25)

    a_adam, sc_adam = _adam_optimize(a_cur, adam_dl, rng, lr=0.08, K_focus=15, name="adam1")

    # 1b: L-BFGS refinement from Adam result

        c = _full_convolve(a_cur, a_cur)
        beta0 = 0.001 * float(c.max())
        lbfgs_dl = min(phase1_end - 30, time.monotonic() + (phase1_end - time.monotonic()) * 0.6)

    return [float(x) for x in fallback]
