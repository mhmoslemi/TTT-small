import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 4000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=200):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(12345)

    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Use a random subset of indices per round for large n
        if n > 2048:
            indices = rng.choice(n, size=min(512, n), replace=False)
        else:
            indices = np.arange(n)
        rng.shuffle(indices)

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try a few perturbations
            for factor in [0.97, 0.99, 1.01, 1.03]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-30:
                    a[j] = old_val
                    continue
                a[j] = new_val
                sc = _score_norm(a / s)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[j] = old_val
        # Renormalize
        a = a / a.sum()
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if not improved and r > 20:
            break
    return best_a, best_sc


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p * np.sqrt(n0))
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")

    # === Phase 1: FISTA with aggressive beta continuation ===
    print("Phase 1: FISTA aggressive beta continuation")
    phase1_end = min(deadline, t_start + 500.0)
    a_curr = a0.copy()

    betas_list = []
    beta = 0.05 * max_c0
    while beta > 1e-12 * max_c0:
        betas_list.append(beta)
        beta *= 0.4
    # Ensure we go very fine
    betas_list.append(1e-12 * max_c0)
    if len(betas_list) > 25:
        betas_list = betas_list[::len(betas_list) // 25] + [1e-12 * max_c0]

    for i, beta in enumerate(betas_list):
        if time.monotonic() > phase1_end - 5:
            break
        per_time = min(40.0, phase1_end - time.monotonic() - 3)
        if per_time < 2:
            break
        max_it = max(3000, int(per_time * 3000))
        a_curr, sc = _fista_smoothed(a_curr, beta, time.monotonic() + per_time, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.2e}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _convolve(a_curr, a_curr)
        max_c_curr = float(c_curr.max())

    print(f"End Phase 1: score={best_score:.10f}")

    # === Phase 2: Adam log-space fine optimization ===
    print("Phase 2: Adam log-space fine")
    phase2_end = min(deadline, t_start + 1000.0)
    c_curr = _convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas_fine = []
    beta = 0.001 * max_cc
    while beta > 1e-14 * max_c0:
        betas_fine.append(beta)
        beta *= 0.35
    if len(betas_fine) > 15:
        betas_fine = betas_fine[::len(betas_fine) // 15] + [1e-14 * max_c0]

    per_time = phase2_end - time.monotonic() - 20
    if per_time > 30:
        a_opt, sc_opt = _adam_logspace(
            a_curr, time.monotonic() + per_time,
            betas_fine, lr_init=0.01, max_iter_total=2000000
        )
        sc_opt_full = _score(a_opt)
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()

    a_work = best_seq.copy()
    a_work = np.clip(a_work, 0, 1000)
    a_work = a_work / a_work.sum()

    # === Phase 3: Multi-restart with diverse perturbations ===
    print("Phase 3: Multi-restart")
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end - 60:
        n_restarts = 8
        per_restart = (phase3_end - time.monotonic() - 30) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break
            if i % 4 == 0:
                p_start = _perturb(a_work, 0.02 + 0.005 * i, rng)
            elif i % 4 == 1:
                sz = n_main // 1024
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif i % 4 == 2:
                p_start = _perturb(a_work, 0.05, rng)
            else:
                # Circular shift of a block
                block = n_main // 4
                shift = rng.integers(1, block)
                p_start = a_work.copy()
                p_start[:block] = np.roll(a_work[:block], shift)
                p_start = p_start / p_start.sum()

            c_s = _convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs,
                  0.00005 * max_cs, 0.00001 * max_cs, 0.000005 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart * 0.7),
                bs, lr_init=0.008, max_iter_total=200000
            )
            # Quick FISTA polish
            c_pb = _convolve(p_best, p_best)
            max_cp = float(c_pb.max())
            p_best, sc_pb = _fista_smoothed(
                p_best, 0.00001 * max_cp,
                time.monotonic() + per_restart * 0.25, 50000
            )
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Coordinate descent polish ===
    print("Phase 4: Coordinate descent")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_cd = best_seq.copy()
        a_cd = a_cd / a_cd.sum()
        cd_deadline = min(deadline - 80, time.monotonic() + 150)
        a_cd, sc_cd = _coord_descent(a_cd, cd_deadline, max_rounds=500)
        sc_cd_full = _score(a_cd)
        print(f"  Coord descent: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
        if sc_cd_full < best_score:
            best_score = sc_cd_full
            best_seq = a_cd.copy()

    # === Phase 5: Try n=8192 ===
    print("Phase 5: n=8192 exploration")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 60 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interpolated n=8192: score={sc_big:.10f}")
        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        # Optimize the big version
        if time.monotonic() < phase5_end - 30:
            c_big = _convolve(a_big, a_big)
            max_cb = float(c_big.max())
            # FISTA on big
            beta_big = 0.01 * max_cb
            a_big_opt, sc_big_opt = _fista_smoothed(
                a_big, beta_big, time.monotonic() + 30.0, 50000
            )
            a_big_opt = a_big_opt / a_big_opt.sum()
            sc_big_opt_full = _score(a_big_opt)
            if sc_big_opt_full < best_score:
                best_score = sc_big_opt_full
                best_seq = a_big_opt.copy()

            # Adam on big
            if time.monotonic() < phase5_end - 15:
                c_bo = _convolve(a_big_opt, a_big_opt)
                max_cbo = float(c_bo.max())
                bs_big = [0.001 * max_cbo, 0.0005 * max_cbo, 0.0002 * max_cbo,
                          0.0001 * max_cbo, 0.00005 * max_cbo]
                per_t = phase5_end - time.monotonic() - 10
                if per_t > 20:
                    a_big_opt, sc = _adam_logspace(
                        a_big_opt, time.monotonic() + per_t,
                        bs_big, lr_init=0.006, max_iter_total=100000
                    )
                    sc_full = _score(a_big_opt)
                    if sc_full < best_score:
                        best_score = sc_full
                        best_seq = a_big_opt.copy()
                        print(f"  Optimized n=8192: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine Adam polish ===
    print("Phase 6: Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        n_best = len(best_seq)
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000001 * max_cp,
                     0.0000005 * max_cp, 0.0000001 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 15:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.002, max_iter_total=500000
            )
            sc_pol_full = _score(a_pol)
            print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
