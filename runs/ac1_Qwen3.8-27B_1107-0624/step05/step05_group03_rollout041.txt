Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses flat-convolution-inspired profiles, L-BFGS with logsumexp continuation,
coordinate descent, and multi-scale local search.

import numpy as np
from scipy.optimize import minimize
from scipy.special import comb
import time

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _make_profile(n, kind, **params):
    """Generate various profile shapes."""
    t = np.linspace(0.0, 1.0, n)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'triangular':
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) * (n - i)
    elif kind == 'flat_conv':
        # Central binomial inspired: a_i ~ 1/sqrt(i+1) for i=0..n-1
        i = np.arange(n, dtype=np.float64)
        gamma = params.get('gamma', 1.0)
        a = 1.0 / np.power(i + 1.0, gamma)
    elif kind == 'flat_conv_sym':
        # Symmetric version
        i = np.arange(n, dtype=np.float64)
        gamma = params.get('gamma', 1.0)
        d = np.minimum(i, n - 1 - i)
        a = 1.0 / np.power(d + 1.0, gamma)
    elif kind == 'bspline':
        order = params.get('order', 4)
        u = np.clip(t, 0.01, 0.99)
        a = np.power(u, order-1) * np.power(1-u, order-1)
    elif kind == 'raised_cos':
        alpha = params.get('alpha', 1.0)
        a = 0.5 * (1.0 + np.cos(2.0*np.pi*(t-0.5)*alpha))
    elif kind == 'w_profile':
        freq = params.get('freq', 2)
        depth = params.get('depth', 0.3)
        a = 1.0 + depth * np.cos(2.0*np.pi*freq*t)
    elif kind == 'bimodal':
        sep = params.get('separation', 0.3)
        w = params.get('width', 0.15)
        c1 = 0.5 - sep/2
        c2 = 0.5 + sep/2
        a = np.exp(-(t-c1)**2/(2*w**2)) + np.exp(-(t-c2)**2/(2*w**2))
    elif kind == 'power':
        p = params.get('p', 2.0)
        u = np.abs(2.0*t - 1.0)
        a = np.power(1.0-u, p)
    elif kind == 'arcsin':
        # Arcsine distribution: f(x) = 1/(pi*sqrt(x*(1-x)))
        u = np.clip(t, 1e-6, 1-1e-6)
        a = 1.0 / np.sqrt(u * (1.0-u))
    elif kind == 'inverse_sqrt':
        # 1/sqrt(x) type
        u = np.clip(t, 1e-6, 1.0)
        a = 1.0 / np.sqrt(u)
    elif kind == 'inverse_sqrt_sym':
        # Symmetric 1/sqrt
        i = np.arange(n, dtype=np.float64)
        d = np.minimum(i + 1, n - i)
        a = 1.0 / np.sqrt(d)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
