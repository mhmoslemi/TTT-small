import numpy as np

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    k = 1
    while k < out_n:
        k <<= 1
    N = max(2, k)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    """Project onto the probability simplex (sum=1, non-negative)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.where(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fista_simplex(a_init, beta, deadline, max_iter=5000):
    """FISTA with softmax-smoothed max on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = 1.0 / (4.0 * n / beta)

    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(2.0 * n * c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        a_next = _proj_simplex(y - step * grad)

        sc = float(2.0 * n * _full_convolve(a_next, a_next).max())
        if sc < best_score:
            best_score = sc
            best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    return best_a, best_score


def _adam_refine(a_init, deadline, beta, lr=0.01, max_iter=100000):
    """Adam in log-space for fine polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for _ in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _full_convolve(a, a)
        cm = float(c.max())
        w = np.exp((c - cm) / beta)
        ws = float(w.sum())
        if ws < 1e-300 or not np.isfinite(ws):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 1.0:
            update = update * (1.0 / un)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()
            sc = _score(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if t % 5000 == 0:
            lr = max(0.001, lr * 0.95)

    return best_a, best_score

                sc_p_np = _score_np(parent)
                print(f"Parent: n={len(parent)}, score_fft={sc_p:.10f}, score_np={sc_p_np:.10f}")
                try:
                    sc_p_eval = float(evaluate_sequence([float(x) for x in parent]))
                    print(f"Parent evaluator score: {sc_p_eval:.10f}")
                    if sc_p_eval < best_score:
                        best_score = sc_p_eval
                        best_seq = parent.copy()
                except Exception as ex:
                    print(f"Parent eval error: {ex}")
    except Exception as ex:
        print(f"No parent: {ex}")

    # Try uniform sequences of various lengths
    for n_try in [128, 256, 512, 1024, 2048, 4096, 8192, 16384, 32768]:
        if time.monotonic() > deadline:
            break
        a_uni = np.ones(n_try) / n_try
        sc_u = _score(a_uni)
        sc_u_np = _score_np(a_uni)
        print(f"Uniform n={n_try}: score_fft={sc_u:.10f}, score_np={sc_u_np:.10f}")
        try:
            sc_u_eval = float(evaluate_sequence([float(x) for x in a_uni]))
            print(f"  Evaluator: {sc_u_eval:.10f}")
            if sc_u_eval < best_score:
                best_score = sc_u_eval
                best_seq = a_uni.copy()
        except Exception as ex:
            print(f"  Eval error: {ex}")

    print(f"Best so far: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # If parent had a better score than uniform, try to optimize it
    if parent is not None and best_seq is not None and len(best_seq) != len(parent):
        # The evaluator preferred something other than uniform; try to improve parent
        a_work = parent.copy()
        a_work = a_work / a_work.sum()
        c_work = _full_convolve(a_work, a_work)
        max_c = float(c_work.max())

        # FISTA phase
        if time.monotonic() < deadline - 300:
            beta = 0.01 * max_c
            a_opt, _ = _fista_simplex(a_work, beta, time.monotonic() + min(60.0, deadline - time.monotonic() - 30), 20000)
            a_opt = a_opt / a_opt.sum()
            try:
                sc_opt = float(evaluate_sequence([float(x) for x in a_opt]))
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_seq = a_opt.copy()
                    print(f"  FISTA improved: {sc_opt:.10f}")
            except Exception:
                pass

        # Adam phase
        if time.monotonic() < deadline - 100:
            beta_fine = 0.001 * max_c
            a_opt2, _ = _adam_refine(a_opt, time.monotonic() + min(120.0, deadline - time.monotonic() - 50), beta_fine, lr=0.005, max_iter=50000)
            a_opt2 = a_opt2 / a_opt2.sum()
            try:
                sc_opt2 = float(evaluate_sequence([float(x) for x in a_opt2]))
                if sc_opt2 < best_score:
                    best_score = sc_opt2
                    best_seq = a_opt2.copy()
                    print(f"  Adam improved: {sc_opt2:.10f}")
            except Exception:
                pass

    # If uniform is best (score ~2), try a few non-uniform perturbations to check
    if best_score >= 1.99:
        n_best = len(best_seq)
        # Try a sinusoidal perturbation
        t_arr = np.linspace(0, 1, n_best)
        for amp in [0.01, 0.05, 0.1, 0.2, 0.5]:
            if time.monotonic() > deadline - 30:
                break
            a_pert = 1.0 + amp * np.cos(2.0 * np.pi * t_arr)
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()
            try:
                sc_pert = float(evaluate_sequence([float(x) for x in a_pert]))
                if sc_pert < best_score:
                    best_score = sc_pert
                    best_seq = a_pert.copy()
                    print(f"  Perturbed amp={amp}: {sc_pert:.10f}")
            except Exception:
                pass

        # Try tapered
        a_taper = np.sin(np.pi * t_arr) ** 0.5
        a_taper = np.maximum(a_taper, 1e-10)
        a_taper = a_taper / a_taper.sum()
        try:
            sc_tap = float(evaluate_sequence([float(x) for x in a_taper]))
            if sc_tap < best_score:
                best_score = sc_tap
                best_seq = a_taper.copy()
                print(f"  Tapered: {sc_tap:.10f}")
        except Exception:
            pass

    # Final: if we have time, do a focused optimization on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_final = best_seq.copy()
        a_final = np.clip(a_final, 0, 1000)
        a_final = a_final / a_final.sum()
        c_fin = _full_convolve(a_final, a_final)
        max_cf = float(c_fin.max())

        # Very fine Adam
        beta_ult = max(1e-6, 0.0001 * max_cf)
        a_pol, _ = _adam_refine(a_final, time.monotonic() + (deadline - time.monotonic() - 30), beta_ult, lr=0.002, max_iter=200000)
        a_pol = a_pol / a_pol.sum()
        try:
            sc_pol = float(evaluate_sequence([float(x) for x in a_pol]))
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Final polish: {sc_pol:.10f}")
        except Exception:
            pass

    # Final validation and return
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = _score(best_seq)

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256) / 256

    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Final evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Final eval error: {ex}")

    return [float(x) for x in final]
