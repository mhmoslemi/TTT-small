def _optimize_adaptive(a_init, deadline, base_lr=0.01, beta_init=None):
    """
    Adaptive-temperature gradient descent in log-space for minimizing
    max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15)).astype(np.float64)
    z = z - z.mean()
    
    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_M = float(c_init.max())
    
    if beta_init is None:
        # Choose initial beta so that ~10% of lags have significant weight
        c_sorted = np.sort(c_init)[::-1]
        threshold = c_sorted[min(int(0.1*n), len(c_sorted)-1)]
        beta = max(1e-6, (c_init.max() - threshold) / 2.0)
        beta = min(beta, best_M * 10)  # cap it
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = base_lr
    
    # Adaptive beta schedule
    beta_min = best_M * 1e-4
    beta_max = best_M * 50
    beta_increase = 1.05  # multiply beta by this each iteration (slow increase)
    
    no_improve_count = 0
    patience = 5000  # increase beta or reduce lr after this many steps without improvement
    
    while time.monotonic() < deadline:
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Compute smooth max weights
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Compute gradient
        grad = _conv_grad(w, a, n)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Adam update
        t_step += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        # Check improvement
        if t_step % 200 == 0:
            sc = float(_full_convolve(a, a).max())
            if sc < best_M:
                best_M = sc
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 200
        
        # Adaptive beta and lr
        if no_improve_count > patience:
            # Try increasing beta (sharper max approximation)
            if beta < beta_max:
                beta *= beta_increase ** (no_improve_count / patience)
            current_lr = max(0.001, current_lr * 0.9)
            no_improve_count = 0
        
        # Gradually increase beta
        if t_step % 5000 == 0 and beta < beta_max:
            beta = min(beta_max, beta * 1.02)
    
    # Final evaluation
    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_M:
        best_M = sc_final
        best_a = a_final.copy()
    
    return best_a, best_M


def _make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n)
    c = (n-1) / 2.0
    a = np.exp(-0.5 * ((x - c) / (sigma_frac * n / 2))**2)
    return a / a.sum()


def _make_cosine(n):
    x = np.arange(n)
    a = np.cos(np.pi * x / (2 * n))**2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _make_polynomial(n, degree=4):
    x = np.linspace(-1, 1, n)
    a = (1 - x**2) ** degree
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
