```python
def _project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) - (cv - target_sum) > 0)
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
