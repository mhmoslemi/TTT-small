def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1) / 1.0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

                                                      beta_scale=bs)
        print(f"  L-BFGS beta_scale={bs}: score={sc:.10f} (t={elapsed:.1f}s)")
        update_best(a_opt, f"LBFGS bs={bs}")
    
    # === Phase 2: FISTA with exact max (500s - 900s) ===
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: FISTA projected gradient ===")
    
    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_fista_init = best_seq / best_seq.sum()
        # Try multiple FISTA runs with different lr scales
        for lr_scale in [1.0, 0.5, 2.0]:
            if time.monotonic() > phase2_end - 20:
                break
            a_fista = a_fista_init.copy()
            # Manual FISTA with the specific lr
            n = len(a_fista)
            lr = lr_scale / (8.0 * n)
            y = a_fista.copy()
            t_k = 1.0
            best_fista = a_fista.copy()
            best_fista_sc = float(_score(a_fista))
            
            for it in range(20000):
                if time.monotonic() > phase2_end - 20:
                    break
                c = _convolve(y, y)
                idx = int(np.argmax(c))
                g = np.zeros(n)
                j_start = max(0, idx - (n - 1))
                j_end = min(n - 1, idx)
                g[j_start:j_end + 1] = 4.0 * n * a_fista[idx - np.arange(j_start, j_end + 1)]
                
                y_new = _proj_simplex(y - lr * g)
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
                y = y_new + ((t_k - 1.0) / t_new) * (y_new - a_fista)
                y = _proj_simplex(y)
                t_k = t_new
                a_fista = y_new
                
                if it % 1000 == 0:
                    sc = float(_score(a_fista))
                    if sc < best_fista_sc:
                        best_fista_sc = sc
                        best_fista = a_fista.copy()
                    if it % 5000 == 0:
                        print(f"    FISTA lr_scale={lr_scale} it={it}: score={sc:.10f}")
            
            sc_final = float(_score(a_fista))
            if sc_final < best_fista_sc:
                best_fista_sc = sc_final
                best_fista = a_fista.copy()
            update_best(best_fista, f"FISTA lr={lr_scale}")
    
    # === Phase 3: Multi-restart with diverse inits (900s - 1300s) ===
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 3: Multi-restart ===")
    
    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 6
        per_restart = (phase3_end - time.monotonic() - 30) / max(1, n_restarts)
        
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break
            
            # Different perturbation strategies
            if i == 0:
                # Small perturbation in z-space
                z_r = z0 + 0.05 * rng.standard_normal(len(z0))
            elif i == 1:
                # Medium perturbation
                z_r = z0 + 0.15 * rng.standard_normal(len(z0))
            elif i == 2:
                # Structured: random permutation of blocks
                block_size = max(1, len(a_best_norm) // 16)
                a_r = a_best_norm.copy()
                for b in range(0, len(a_r) - block_size, block_size):
                    if rng.random() < 0.3:
                        a_r[b:b+block_size] = a_r[b:b+block_size][::-1]
                z_r = np.log(np.maximum(a_r, 1e-15))
                z_r = z_r - z_r.mean()
            elif i == 3:
                # Blend with structured init
                a_struct = _make_structured_init(len(a_best_norm), rng)
                a_blend = 0.7 * a_best_norm + 0.3 * a_struct
                a_blend /= a_blend.sum()
                z_r = np.log(np.maximum(a_blend, 1e-15))
                z_r = z_r - z_r.mean()
            elif i == 4:
                # Blow up a random region
                z_r = z0.copy()
                center = rng.integers(0, len(z_r))
                width = len(z_r) // 10
                mask = np.zeros(len(z_r), dtype=bool)
                mask[max(0, center-width):min(len(z_r), center+width)] = True
                z_r[mask] += 0.5
                z_r = z_r - z_r.mean()
            else:
                # Large perturbation
                z_r = z0 + 0.5 * rng.standard_normal(len(z0))
            
            # L-BFGS from this restart
            remaining = min(per_restart, phase3_end - time.monotonic() - 15)
            if remaining < 15:
                break
            for bs in [0.02, 0.01, 0.005]:
                if time.monotonic() > phase3_end - 15:
                    break
                z_r, a_rot, sc_r, _ = _lbfgs_phase(z_r, time.monotonic() + remaining,
                                                    beta_scale=bs)
                remaining = phase3_end - time.monotonic() - 15
                if remaining < 10:
                    break
                update_best(a_rot, f"restart{i}_bs={bs}")
    
    # === Phase 4: n-variation (1300s - 1550s) ===
    phase4_end = min(deadline, t_start + 1550.0)
    print(f"\n=== Phase 4: n-variation ===")
    
    if best_seq is not None and time.monotonic() < phase4_end - 30:
        n_current = len(best_seq)
        target_ns = []
        if n_current < 4096:
            target_ns = [4096, 8192]
        elif n_current == 4096:
            target_ns = [8192, 16384]
        elif n_current == 8192:
            target_ns = [4096, 16384]
        elif n_current == 16384:
            target_ns = [8192]
        else:
            target_ns = [4096, 8192, 16384]
        
        for target_n in target_ns:
            if time.monotonic() > phase4_end - 40:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp, f"interp_n={target_n}")
            
            if time.monotonic() < phase4_end - 20:
                z_int = np.log(np.maximum(p_interp, 1e-15))
                z_int = z_int - z_int.mean()
                remaining = phase4_end - time.monotonic() - 15
                if remaining > 30:
                    for bs in [0.02, 0.01]:
                        if time.monotonic() > phase4_end - 15:
                            break
                        z_int, a_int, sc_int, _ = _lbfgs_phase(
                            z_int, time.monotonic() + (phase4_end - time.monotonic() - 15),
                            beta_scale=bs)
                        update_best(a_int, f"interp_n={target_n}_L")
    
    # === Phase 5: Final polish (1550s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_best_norm = best_seq / best_seq.sum()
        z_final = np.log(np.maximum(a_best_norm, 1e-15))
        z_final = z_final - z_final.mean()
        
        # Very fine L-BFGS
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            for bs in [0.005, 0.002, 0.001, 0.0005]:
                if time.monotonic() > deadline - 10:
                    break
                z_final, a_fin, sc_fin, _ = _lbfgs_phase(
                    z_final, time.monotonic() + (deadline - time.monotonic() - 10),
                    beta_scale=bs)
                update_best(a_fin, f"final_bs={bs}")
        
        # Quick FISTA polish
        if time.monotonic() < deadline - 5:
            a_fista_final = best_seq / best_seq.sum()
            n = len(a_fista_final)
            lr = 1.0 / (8.0 * n)
            y = a_fista_final.copy()
            t_k = 1.0
            best_fin = a_fista_final.copy()
            best_fin_sc = float(_score(a_fista_final))
            
            for it in range(5000):
                if time.monotonic() > deadline - 3:
                    break
                c = _convolve(y, y)
                idx = int(np.argmax(c))
                g = np.zeros(n)
                j_start = max(0, idx - (n - 1))
                j_end = min(n - 1, idx)
                g[j_start:j_end + 1] = 4.0 * n * a_fista_final[idx - np.arange(j_start, j_end + 1)]
                y_new = _proj_simplex(y - lr * g)
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
                y = y_new + ((t_k - 1.0) / t_new) * (y_new - a_fista_final)
                y = _proj_simplex(y)
                t_k = t_new
                a_fista_final = y_new
                if it % 500 == 0:
                    sc = float(_score(a_fista_final))
                    if sc < best_fin_sc:
                        best_fin_sc = sc
                        best_fin = a_fista_final.copy()
            
            sc_end = float(_score(a_fista_final))
            if sc_end < best_fin_sc:
                best_fin = a_fista_final.copy()
            update_best(best_fin, "final_FISTA")
    
    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
