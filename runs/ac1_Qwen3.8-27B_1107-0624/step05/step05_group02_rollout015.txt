                          jac=True, options={'maxiter': max_iter, 'gtol': 1e-8,
                                             'ftol': 1e-10, 'maxfun': max_iter * 3})
            if res.x is not None:
                z = res.x - np.mean(res.x)
                a = np.exp(z - np.max(z))
                a = a / a.sum()
                sc = _score(a)
                if sc < best_obj:
                    best_obj = sc
                    best_a = a.copy()
        except Exception as ex:
            print(f"  L-BFGS error at beta={beta:.6f}: {ex}")
    
    # Final evaluation
    a_final = np.exp(z - np.max(z))
    a_final = a_final / a_final.sum()
    sc_final = _score(a_final)
    if sc_final < best_obj:
        best_obj = sc_final
        best_a = a_final.copy()
    
    return best_a, best_obj


def _make_profile(n, profile_type='uniform'):
    if profile_type == 'uniform':
        return np.ones(n) / n
    elif profile_type == 'triangular':
        x = np.linspace(0, 1, n)
        return np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 1e-10
    elif profile_type == 'cosine':
        x = np.linspace(0, np.pi, n)
        return np.cos(0.5 * x) ** 2 + 1e-10
    elif profile_type == 'gaussian':
        x = np.linspace(0, 1, n)
        return np.exp(-4 * (x - 0.5) ** 2) + 1e-10
    elif profile_type == 'raised_cosine':
        x = np.linspace(0, 1, n)
        return 0.5 * (1 + np.cos(2 * np.pi * x)) + 1e-10
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return np.exp(z - np.max(z)) / np.sum(np.exp(z - np.max(z)))


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _lp_descent(f, max_steps=30, time_limit_per_lp=20):
    """LP-based descent: find direction g that reduces max convolution."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    
    # Build constraint matrix
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    for step in range(max_steps):
        if time.monotonic() > time_limit_per_lp + 0:  # This won't work, need deadline
            break
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'time_limit': time_limit_per_lp, 'presolve': True})
            if res.status != 0 or res.x is None:
                break
            g = np.maximum(res.x, 0)
            if g.sum() < 1e-15:
                break
        except:
            break
        
        g = g / g.sum()
        
        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())
            if M_new < M * (1 - 1e-10):
                f = f_new
                M = M_new
                c = c_new
                improved = True
                break
        
        if not improved:
            break
    
    return f, M
