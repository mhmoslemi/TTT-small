"""
Optimized search for minimum 2*n*max(conv(a,a))/sum(a)^2 using
iterated LP minimax, L-BFGS log-sum-exp continuation, and coordinate descent.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Returns 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Returns 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    """Build sparse matrix A such that (A @ g)[k] = (f * g)[k]."""
    n = len(f)
    # (f * g)[k] = sum_i f[i] * g[k-i]
    # For k in 0..2n-2, this is a linear function of g
    # (A @ g)[k] = sum_j A[k,j] * g[j]
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    row_idx = []
    col_idx = []
    data = []
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row_idx.append(k)
                col_idx.append(j)
                data.append(f[i])
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def _build_conv_matrix_fast(f):
    """Build sparse matrix more efficiently using vectorized operations."""
    n = len(f)
    # For each k (0 to 2n-2), the valid j range is max(0, k-n+1) to min(k, n-1)
    # A[k, j] = f[k-j] for valid j
    row_parts = []
    col_parts = []
    data_parts = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        j = np.arange(j_min, j_max + 1)
        i = k - j
        row_parts.append(np.full(len(j), k))
        col_parts.append(j)
        data_parts.append(f[i])
    row_idx = np.concatenate(row_parts)
    col_idx = np.concatenate(col_parts)
    data = np.concatenate(data_parts)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def _lp_minimax(f):
    """Solve LP: minimize t s.t. (f*g)[k] <= t, sum(g)=1, g>=0."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix_fast(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
            else:
                g_val = np.ones(n) / n
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None

def _lp_iterate(f, n_iters, deadline, step_size=0.3):
    """Iterated minimax with LP subproblems."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    f_cur = f.copy()
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        g, t_val = _lp_minimax(f_cur)
        if g is None:
            break
        f_new = (1 - step_size) * f_cur + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f_cur = f_new
        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    
    return best_a, best_score

def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # Gradient of sum_k c_k * w_k with respect to a
    # d/da_i [sum_k w_k * c_k] where c_k = sum_j a_j * a_{k-j}
    # = sum_k w_k * (sum_{j: k-j valid} a_{k-j} + a_{k-i}... )
    # Using the trick: grad_i = 2 * sum_k w_k * (a * a^R)[i+k] * (b-spline correction)
    # Simpler: use the fact that d c_k / d a_i = a[k-i] + a[i-k] (with bounds)
    # For the softmax parameterization, d a_i / d z_j = a_i (delta_ij - a_j)
    # So grad_z_j = sum_i [2n * 2 * a_i * (something)] * a_i * (delta_ij - a_j)
    
    # Actually, let's use the efficient formula from the previous code:
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_z = 4.0 * n * a * (h - W)
    return grad_z

def _run_lbfgs(a_init, deadline, beta_start=None):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if beta_start is None:
        beta_start = 0.1 * max_c0
    
    beta_list = [beta_start * (0.5 ** i) for i in range(30)]
    beta_list = [max(b, 1e-10 * max_c0) for b in beta_list]
    
    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 90.0)
        maxiter = max(100, int(time_per * 20))
        
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _coord_descent(a_init, deadline, max_rounds=50):
    """Coordinate descent on z = log(a) parameterization."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            base_score = _score_norm_fn(a)
            best_delta = 0.0
            best_local = base_score
            for delta in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.08, -0.08]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score

Minimize 2*n*max(conv(a,a))/sum(a)^2 using iterated LP minimax,
L-BFGS log-sum-exp continuation, and adaptive strategies.
Multiple starting points including uniform sequences and parent.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    n = len(a)

    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
            else:
                g_val = np.ones(n) / n
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters, deadline, step_size=0.3):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    f_cur = f.copy()
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        g, t_val = _lp_minimax(f_cur)
        if g is None:
            break
        f_new = (1 - step_size) * f_cur + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f_cur = f_new
        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    return best_a, best_score


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_z = 4.0 * n * a * (h - W)
    return grad_z


def _run_lbfgs(a_init, deadline, beta_start=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if beta_start is None:
        beta_start = 0.05 * max_c0
    beta_list = [beta_start * (0.6 ** i) for i in range(40)]
    beta_list = [max(b, 1e-12 * max_c0) for b in beta_list]
    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 120.0)
        maxiter = max(200, int(time_per * 15))
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _proj_subgrad(f, n_iters, deadline, lr=0.1):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    a_cur = f.copy()
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        # Subgradient of max_k c_k at a_cur
        # c_k = sum_i a_i * a[k-i], dc_k/da_j = a[k-j] + a[j-k] (with bounds)
        # The active set is where c_k = cm
        active = np.where(np.abs(c - cm) < 1e-10)[0]
        # Subgradient: sum of active gradients
        grad = np.zeros(n)
        for k in active:
            for j in range(n):
                i1 = k - j
                if 0 <= i1 < n:
                    grad[j] += a_cur[i1]
                i2 = j - k
                if 0 <= i2 < n:
                    grad[j] += a_cur[i2]
        grad *= 2.0
        # Projected gradient step on simplex
        a_new = a_cur - lr * grad / (lr * np.max(np.abs(grad)) + 1e-10)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a_cur = a_new
        lr *= 0.99
    return best_a, best_score


def _proj_subgrad_fast(f, n_iters, deadline):
    """Faster projected subgradient using FFT-based gradient computation."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    a_cur = f.copy()
    lr = 0.5
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        # Find active constraints (where c_k is close to max)
        tol = 0.01 * cm
        active = c > (cm - tol)
        if not np.any(active):
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(c)] = True
        # Gradient of sum of active c_k with respect to a
        # Using FFT: grad = A_active^T @ a_cur (where A_active is the conv matrix for active lags)
        # (f*g)[k] = sum_j f[k-j]*g[j], so d/dc_k/da_j = f[k-j] (with bounds)
        # Total grad_j = sum_{k active} a_cur[k-j] (with bounds)
        # This is (a_cur * 1_active)[j + n - 1] where 1_active is the indicator
        active_ext = np.zeros(2 * n - 1)
        active_ext[active] = 1.0
        conv_grad = _full_convolve(a_cur, active_ext)
        grad = conv_grad[n - 1:2 * n - 1]
        # Normalize step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step = lr / gnorm
        a_new = a_cur - step * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a_cur = a_new
        lr *= 0.995
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Phase 0: Evaluate baselines
    print("Phase 0: Baselines")
    for n_try in [256, 512, 1024, 2048, 4096, 8192]:
        u = np.ones(n_try) / n_try
        sc = _score_norm_fn(u)
        update_best(u, sc, f"uniform_n{n_try}")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_fn(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, _score_norm_fn(parent), "parent")

    # Phase 1: LP iterations (main workhorse)
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\nPhase 1: LP iterations (until t={phase1_end - t_start:.0f}s)")

    if parent is not None and len(parent) <= 4096:
        a_lp = parent.copy()
    elif parent is not None:
        a_lp = _interp_profile(parent, 4096)
    else:
        a_lp = np.ones(2048) / 2048

    update_best(a_lp, _score_norm_fn(a_lp), "lp_start")

    # Multiple LP rounds with different step sizes
    step_schedules = [
        (0.5, 15),
        (0.3, 20),
        (0.2, 20),
        (0.1, 20),
        (0.05, 20),
        (0.03, 20),
    ]
    for round_idx, (step, n_iters) in enumerate(step_schedules):
        if time.monotonic() > phase1_end:
            break
        a_lp, sc_cur = _lp_iterate(a_lp, n_iters=n_iters, deadline=phase1_end, step_size=step)
        update_best(a_lp, sc_cur, f"lp_r{round_idx}_s{step}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS refinement
    phase2_end = min(deadline, t_start + 1000.0)
    print(f"\nPhase 2: L-BFGS refinement (until t={phase2_end - t_start:.0f}s)")

    n_restarts = 4
    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 30:
            break
        if r == 0:
            a_cur = best_seq.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.02 * (r + 1)
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
        per_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) / (n_restarts - r))
        p_best, sc_best = _run_lbfgs(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}_n{len(a_cur)}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Explore different n values
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\nPhase 3: Different n exploration (until t={phase3_end - t_start:.0f}s)")

    n_current = len(best_seq)
    n_options = sorted(set([n_current, 1024, 2048, 4096, 8192]))
    for n_try in n_options:
        if time.monotonic() > phase3_end - 60:
            break
        if n_try == n_current:
            continue
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")
        if time.monotonic() < phase3_end - 40:
            per_dl = min(phase3_end - 30, time.monotonic() + 80.0)
            p_best, sc_best = _run_lbfgs(p_interp, per_dl)
            update_best(p_best, sc_best, f"lbfgs_n{n_try}")
        # Also try LP on this size
        if time.monotonic() < phase3_end - 30:
            per_dl2 = min(phase3_end - 20, time.monotonic() + 50.0)
            p_best2, sc_best2 = _lp_iterate(p_interp, n_iters=10, deadline=per_dl2, step_size=0.1)
            update_best(p_best2, sc_best2, f"lp_n{n_try}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Fine polishing
    phase4_end = deadline
    print(f"\nPhase 4: Fine polishing")

    # More LP with tiny steps
    if time.monotonic() < phase4_end - 60:
        per_dl = min(phase4_end - 50, time.monotonic() + 100.0)
        p_best, sc_best = _lp_iterate(best_seq, n_iters=30, deadline=per_dl, step_size=0.02)
        update_best(p_best, sc_best, "lp_fine")

    # Projected subgradient
    if time.monotonic() < phase4_end - 40:
        per_dl = min(phase4_end - 35, time.monotonic() + 60.0)
        p_best, sc_best = _proj_subgrad_fast(best_seq, n_iters=5000, deadline=per_dl)
        update_best(p_best, sc_best, "proj_subgrad")

    # Very fine L-BFGS
    if time.monotonic() < phase4_end - 20:
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_tiny = 1e-5 * max_c_final
        per_dl = min(phase4_end - 10, time.monotonic() + 40.0)
        p_best, sc_best = _run_lbfgs(best_seq, per_dl, beta_start=beta_tiny)
        update_best(p_best, sc_best, "ultra_fine_lbfgs")

    # One final LP round if time permits
    if time.monotonic() < phase4_end - 10:
        per_dl = min(phase4_end - 5, time.monotonic() + 20.0)
        p_best, sc_best = _lp_iterate(best_seq, n_iters=10, deadline=per_dl, step_size=0.01)
        update_best(p_best, sc_best, "final_lp")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback: uniform
    print("FALLBACK: return uniform")
    return [1.0 / 1024] * 1024
