def _next_pow2(n):

def _nesterov_phase(a_init, deadline, K=50, total_iters=200000):
    """Nesterov accelerated gradient descent with decaying beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    
    # Nesterov parameters
    gamma = 0.9  # momentum
    lr0 = 0.05
    beta0 = 0.01 * cm0
    beta_decay = 0.9995
    beta_min = 1e-10 * cm0
    
    t = 0
    no_improve = 0
    check_interval = 20
    
    for t in range(total_iters):
        if time.monotonic() > deadline:
            break
        
        beta = max(beta_min, beta0 * (beta_decay ** t))
        lr = lr0 / (1.0 + 0.01 * t)
        
        a_y = _softmax(y)
        grad_y = _softmin_grad(y, beta, n, K=K)
        
        # Nesterov update
        z = y + gamma * (y - z)
        z = z - lr * grad_y
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        y = z
        
        if t % check_interval == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                # Small perturbation
                z = z + 0.02 * np.random.randn(n)
                z = z - z.mean()
                y = z
                no_improve = 0
    
    return best_a, best_score


def _proj_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.full(n, target_sum / n)
    rho = rho[-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _peak_peel(a_init, deadline, max_iter=500000):
    """Projected subgradient descent on max of convolution."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c = _full_convolve(a, a)
    cm0 = float(c.max())
    lr = 0.1 * cm0 / n  # initial learning rate
    
    t = 0
    no_improve = 0
    
    for t in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        peak_idx = int(np.argmax(c))
        cm = float(c[peak_idx])
        
        # Compute gradient of c[peak_idx] w.r.t. a
        # dc_k/da_j = a[k-j] + a[j-k] (with valid index checks)
        # Use the convolution trick:
        # grad_a[j] = 2 * a[peak_idx - j] (approximately, with boundary handling)
        j_range = np.arange(n)
        idx1 = peak_idx - j_range  # k - j
        idx2 = j_range - peak_idx  # j - k
        grad_a = np.zeros(n)
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad_a[valid1] += a[idx1[valid1]]
        grad_a[valid2] += a[idx2[valid2]]
        
        # Adaptive learning rate
        lr_t = lr / (1.0 + 0.001 * t)
        
        # Subgradient step
        a_new = a - lr_t * grad_a
        a_new = _proj_simplex(a_new, 1.0)
        
        # Check if improved
        if t % 10 == 0:
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-15:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                lr = lr * 0.9
                no_improve = 0
        
        a = a_new
    
    return best_a, best_score


def _lp_peak_balance(a_init, deadline, K=50, max_rounds=20):
    """LP-based peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.9 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = sorted_idx[:300]
        
        # Build constraint matrix
        Kact = len(active)
        A_mat = np.zeros((Kact, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi+1] = a_init[k-i_lo:k-i_lo+(i_hi-i_lo+1)]
        
        b_vec = np.full(Kact, M * 1.001)
        
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)]*n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break
        
        current_score = 2.0 * n * M / (S * S)
        improved = False
        for t_blend in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t_blend) * best_a + t_blend * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm(a_trial)
                current_score = sc_trial
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _lbfgs_phase(a_init, deadline, beta_list=None, maxiter=5000):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]
    
    # Use top-K for the objective to make it more tractable
    K = min(100, 2*n-1)
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n, K),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _random_restart_local(a_best, deadline, n_restarts=10, perturb_scale=0.01):
    """Try random perturbations and do quick local optimization."""
    n = len(a_best)
    best_a = a_best.copy()
    best_score = _score_norm(a_best)
    
    rng = np.random.default_rng(12345)
    
    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        
        # Random perturbation in z-space
        z = np.log(np.maximum(a_best, 1e-15))
        z = z - z.mean()
        z = z + perturb_scale * rng.standard_normal(n)
        z = z - z.mean()
        a_pert = _softmax(z)
        
        # Quick L-BFGS
        c0 = _full_convolve(a_pert, a_pert)
        m0 = float(c0.max())
        beta = 0.01 * m0
        K = 50
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n, K),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': 2000, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxls': 50, 'maxfun': 5000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    print(f"  Restart {r}: improved to {sc:.10f}")
        except Exception:
            pass
    
    return best_a, best_score
