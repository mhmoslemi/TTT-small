"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, Nesterov-accelerated subgradient
with exact active sets, adaptive perturbations, and fine-tuning phases.
Key improvements: momentum-based subgradient, backtracking line search,
active-set-aware perturbations, and more aggressive beta continuation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, top_k=1):
    """Compute subgradient using top-k active peaks."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros(n)
    # Select top-k peaks
    vals = c.copy()
    # Only consider peaks within 1% of max as "active"
    tol = 0.01 * c_max
    active_mask = vals >= (c_max - tol)
    if active_mask.sum() < top_k:
        idx = np.argsort(vals)[-top_k:]
        active_mask = np.zeros_like(vals, dtype=bool)
        active_mask[idx] = True
    active_count = min(active_mask.sum(), top_k)
    # Get top-k indices
    c_sorted = np.argsort(c)[::-1][:active_count]
    w = np.zeros_like(c)
    w[c_sorted] = 1.0 / active_count
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _nesterov_subgrad(a_init, deadline, max_steps=10000, top_k=3):
    """Nesterov-accelerated subgradient with backtracking line search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    v = np.zeros(n)  # velocity
    best_a = a_init.copy()
    best_sc = _score(a_init)
    alpha = 0.01
    mu = 0.9  # momentum
    t_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Nesterov step
        y = z + mu * v
        a_y = _softmax(y)
        c_y = _full_convolve(a_y, a_y)
        g = _gradient_active(a_y, c_y, n, top_k)

        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-18:
            break

        # Backtracking line search
        a_trial = _softmax(y)
        sc_trial = _score(a_trial)

        step_alpha = alpha
        for _bt in range(20):
            z_new = y - step_alpha * g / (g_norm + 1e-30)
            a_new = _softmax(z_new)
            sc_new = _score(a_new)
            if sc_new < sc_trial * (1 - 1e-8) or _bt == 19:
                break
            step_alpha *= 0.5

        v = mu * v - step_alpha * g / (g_norm + 1e-30)
        z = z + v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        t_count += 1

        # Anneal momentum and step size
        mu = 0.9 + 0.05 * np.exp(-i / 500.0)
        alpha = max(0.001, 0.01 * np.exp(-i / 2000.0))

        if i % 100 == 0:
            a_check = _softmax(z)
            sc_check = _score(a_check)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lbfgs_phase(a_init, beta, n, deadline, max_iter=3000):
    """Run one L-BFGS-B phase."""
    z0 = np.log(np.maximum(a_init, 1e-18))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxcor': 40,
                'maxls': 100,
                'disp': False,
            }
        )
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _perturb_active_aware(a, rng, active_peaks, strength=0.02):
    """Perturb more at positions that contribute to active peaks."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    # Compute which positions contribute most to active peaks
    contrib = np.zeros(n)
    for k in active_peaks:
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                contrib[j] += a[j] * a[idx]
    # Normalize contribution
    cmax = contrib.max()
    if cmax > 1e-30:
        contrib = contrib / cmax
    # Higher perturbation where contribution is high
    noise = rng.standard_normal(n)
    z = z + strength * (0.5 + contrib) * noise
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3, amp=0.01):
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * rng.uniform(0.5, 2.0) * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_gaussian(a, rng, sigma=0.01):
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _get_top_peaks(c, k=5):
    """Get indices of top-k convolution peaks."""
    return np.argsort(c)[::-1][:k]


def _quick_optimize(a_init, deadline, n_phases=3):
    """Quick multi-phase L-BFGS optimization."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a_init

    remaining = deadline - time.monotonic()
    time_per_phase = max(5.0, remaining / n_phases)

    betas = [max_c * s for s in [0.05, 0.02, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        est_time = 0.004 * (n / 2048.0)
        max_iter = min(3000, max(30, int((deadline - 5 - time.monotonic()) / (est_time * 10))))
        a_new, ok = _lbfgs_phase(current_a, beta, n, deadline - 5, max_iter=max_iter)
        sc = _score(a_new)
        if sc < current_sc:
            current_a = a_new
            current_sc = sc

    return current_a


def _make_seed(n, kind="uniform", rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.8
        return np.maximum(a, 1e-10)
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        return np.maximum(a, 1e-10)
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        return np.maximum(a, 1e-10)
    elif kind == "random":
        if rng is None:
            rng = np.random.default_rng(0)
        a = rng.random(n) + 0.1
        return a
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer combining L-BFGS with log-sum-exp smoothing,
    Nesterov-accelerated subgradient with exact active sets, and
    active-set-aware stochastic perturbations for diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy() / parent.sum()  # normalize to sum=1 for optimization
        sc_raw = _score(parent)
        print(f"Parent: n={n_main}, raw_score={sc_raw:.10f}")
        update_best(parent, sc_raw, "parent")
    else:
        n_main = 2048
        a0 = _make_seed(n_main, "uniform")
        print(f"No parent, using uniform n={n_main}")
        update_best(a0, _score(a0), "init")

    # --- Phase 1: L-BFGS with beta continuation ---
    print(f"\nPhase 1: L-BFGS beta-continuation (t={time.monotonic() - t_start:.1f}s)")
    a_main = _quick_optimize(a0, deadline - 600, n_phases=5)
    # Restore original scale for evaluation
    if parent is not None:
        a_main_scaled = a_main * (parent.sum() / a_main.sum())
    else:
        a_main_scaled = a_main
    sc = _score(a_main_scaled)
    update_best(a_main_scaled, sc, "phase1_lbfgs")

    # Work in normalized space (sum=1) from here on
    a_work = a_main.copy() / a_main.sum() if a_main.sum() > 0 else a0

    # --- Phase 2: Nesterov subgradient ---
    print(f"\nPhase 2: Nesterov subgradient (t={time.monotonic() - t_start:.1f}s)")
    phase2_deadline = min(deadline - 400, time.monotonic() + 200)
    a_sg, sc_sg = _nesterov_subgrad(a_work, phase2_deadline, max_steps=8000, top_k=3)
    if parent is not None:
        a_sg_s = a_sg * (parent.sum())
    else:
        a_sg_s = a_sg
    update_best(a_sg_s, _score(a_sg_s), "phase2_nesterov")

    # --- Phase 3: Fine L-BFGS polishing ---
    print(f"\nPhase 3: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_best_n = best_seq.copy() / best_seq.sum()
        n = len(a_best_n)
        c_best = _full_convolve(a_best_n, a_best_n)
        mb = float(c_best.max())
        fine_betas = mb * np.array([0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 150:
                break
            remaining_time = deadline - 150 - time.monotonic()
            if remaining_time < 10:
                break
            est_t = 0.004 * (n / 2048.0)
            max_iter = min(5000, max(100, int(remaining_time / (est_t * 12))))
            a_new, ok = _lbfgs_phase(a_best_n, beta, n, deadline - 150, max_iter=max_iter)
            a_new_s = a_new * (best_seq.sum())
            sc = _score(a_new_s)
            update_best(a_new_s, sc, f"phase3_fine_{i}")
            if sc < best_score:
                a_best_n = a_new_s / a_new_s.sum()

    # --- Phase 4: More Nesterov on best ---
    print(f"\nPhase 4: Nesterov on best (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_best_n = best_seq / best_seq.sum()
        phase4_dl = min(deadline - 80, time.monotonic() + 150)
        a_sg2, sc_sg2 = _nesterov_subgrad(a_best_n, phase4_dl, max_steps=5000, top_k=5)
        a_sg2_s = a_sg2 * best_seq.sum()
        update_best(a_sg2_s, _score(a_sg2_s), "phase4_nesterov2")

    # --- Phase 5: Active-set-aware restarts ---
    print(f"\nPhase 5: Active-aware restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        a_best_n = best_seq / best_seq.sum()
        n = len(a_best_n)
        c_best = _full_convolve(a_best_n, a_best_n)
        top_peaks = _get_top_peaks(c_best, k=5)
        scale = best_seq.sum()

        n_restarts = min(5, max(2, int((deadline - 60 - time.monotonic()) / 120)))
        print(f"  Running {n_restarts} restarts")

        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break

            if i == 0:
                p_start = _perturb_active_aware(a_best_n, rng, top_peaks, strength=0.015)
            elif i == 1:
                p_start = _perturb_fourier(a_best_n, rng, n_modes=2, amp=0.005)
            elif i == 2:
                p_start = _perturb_gaussian(a_best_n, rng, sigma=0.008)
            elif i == 3:
                p_start = _perturb_active_aware(a_best_n, rng, top_peaks, strength=0.005)
            else:
                p_start = _perturb_fourier(a_best_n, rng, n_modes=5, amp=0.003)

            sc_start = _score(p_start * scale)
            print(f"  Restart {i}: score={sc_start:.10f} (best={best_score:.10f})")

            # Quick optimization
            dl_this = min(deadline - 40, time.monotonic() + 100)
            a_r = _quick_optimize(p_start, dl_this, n_phases=3)
            a_r_s = a_r * scale
            sc_r = _score(a_r_s)
            update_best(a_r_s, sc_r, f"restart_{i}")

            # Quick Nesterov polish
            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 35:
                a_r2, sc_r2 = _nesterov_subgrad(a_r_s / a_r_s.sum(), deadline - 35,
                                                max_steps=2000, top_k=3)
                a_r2_s = a_r2 * scale
                update_best(a_r2_s, _score(a_r2_s), f"restart_{i}_nesterov")

    # --- Phase 6: Final ultra-fine optimization ---
    print(f"\nPhase 6: Final ultra-fine (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final_n = best_seq / best_seq.sum()
        n = len(a_final_n)
        c_f = _full_convolve(a_final_n, a_final_n)
        mb = float(c_f.max())

        # Very small beta L-BFGS
        if time.monotonic() < deadline - 3:
            beta_final = mb * 0.000005
            a_fin, _ = _lbfgs_phase(a_final_n, beta_final, n, deadline - 3, max_iter=2000)
            a_fin_s = a_fin * best_seq.sum()
            update_best(a_fin_s, _score(a_fin_s), "final_ultra")

        # Final Nesterov
        if time.monotonic() < deadline - 2:
            a_fin2, _ = _nesterov_subgrad(a_final_n, deadline - 2, max_steps=1000, top_k=5)
            a_fin2_s = a_fin2 * best_seq.sum()
            update_best(a_fin2_s, _score(a_fin2_s), "final_nesterov")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_seed(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_seed(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
