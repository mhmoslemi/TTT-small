    fb = np.fft.rfft(b, N)

    return r

def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _softmin_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top K peaks
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:K_peaks]
        top_val = c[top_idx]
        # Weights: softmax over negative peak values (more weight on larger peaks)
        tw = top_val - top_val.max()
        tw = np.exp(tw / (top_val.max() * 0.1 + 1e-30))
        tw = tw / tw.sum()

        # Compute subgradient
        grad = _subgrad_z(a, top_idx, tw, top_val)

        # Adam update
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        # Periodic scoring and improvement tracking
        if t % 500 == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Perturbation when stuck
        if t % perturb_interval == 0 and no_improve > 20:
            scale = 0.05 * (1.0 + 0.5 * no_improve / 20.0)
            z = z + scale * rng.standard_normal(n)
            z = z - z.mean()
            no_improve = 0
            a_pert = _softmax(z)
            sc_pert = _score_norm(a_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert.copy()

        # Periodic LP
        if t % lp_interval == 0 and time.monotonic() < deadline - 30:
            a_cur = _softmax(z)
            c_cur = _full_convolve(a_cur, a_cur)
            M_cur = float(c_cur.max())
            sorted_idx_lp = np.argsort(c_cur)[::-1]
            active = sorted_idx_lp[:min(20, len(sorted_idx_lp))]
            # Include peaks within 95% of max
            thresh = 0.95 * M_cur
            addl = np.where(c_cur >= thresh)[0]
            active = np.unique(np.concatenate([active, addl]))
            if len(active) > 50:
                active = active[:50]
            b_lp = _lp_solve(a_cur, M_cur * 1.001, active, time_limit=10)
            if b_lp is not None and b_lp.sum() > 1e-15:
                S_cur = float(a_cur.sum())
                b_scaled = b_lp * (S_cur / b_lp.sum())
                for blend_t in [1.0, 0.5, 0.3, 0.2]:
                    a_trial = (1.0 - blend_t) * a_cur + blend_t * b_scaled
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    a_trial = a_trial / s_trial
                    sc_trial = _score_norm(a_trial)
                    if sc_trial < best_score - 1e-12:
                        best_score = sc_trial
                        best_a = a_trial.copy()
                        z = np.log(np.maximum(a_trial, 1e-15))
                        z = z - z.mean()
                        break

        # Progress print
        if t % print_iter == 0:
            print(f"    SG iter={t}, score={_score_norm(_softmax(z)):.10f}, "
                  f"best={best_score:.10f}, no_impr={no_improve}")

    return best_a, best_score


def _lp_phase(a_init, deadline, K=20, max_rounds=50):
    """Iterative LP peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        thresh = 0.92 * M
        addl = np.where(c >= thresh)[0]
        active = np.unique(np.concatenate([active, addl]))
        if len(active) > 60:
            active = active[:60]

        b_lp = _lp_solve(best_a, M * 1.001, active, time_limit=15)
        if b_lp is None or b_lp.sum() < 1e-15:
            break
        b_scaled = b_lp * (S / b_lp.sum())

        improved = False
        current_score = best_score
        for t_blend in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t_blend) * best_a + t_blend * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            sc_trial = _score_norm(a_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break
        print(f"    LP round {rnd+1}: score={best_score:.10f}")

    return best_a, best_score


def _adam_fine_polish(a_init, deadline, lr=0.01, K_peaks=3):
    """Fine-grained Adam with small learning rate for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    rng = np.random.default_rng(123)

    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        top_idx = sorted_idx[:K_peaks]
        top_val = c[top_idx]
        tw = top_val - top_val.max()
        tw = np.exp(tw / (top_val.max() * 0.05 + 1e-30))
        tw = tw / tw.sum()

        grad = _subgrad_z(a, top_idx, tw, top_val)
        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % 1000 == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 20:
                z = z + 0.02 * rng.standard_normal(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: exact subgradient with Adam, LP peak balancing,
    L-BFGS polish, and fine-grained subgradient refinement.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        update_best(parent, _score_norm(parent), "parent")
        print(f"Parent: n={n}, norm_score={best_score:.10f}, full_score={_score_full(parent):.10f}")
    else:
        n = 4096
        parent = _greedy_init(n)
        update_best(parent, _score_norm(parent), "greedy")
        print(f"Greedy init: n={n}, score={best_score:.10f}")

    a = best_seq.copy()

    # === Phase 1: Subgradient + Adam (800s) ===
    print("Phase 1: Subgradient + Adam")
    p1_end = min(deadline, t_start + 800.0)
    a_sg, sc_sg = _subgradient_phase(a, p1_end, rng, K_peaks=5, lr=0.05)
    update_best(a_sg, sc_sg, "subgrad_main")

    # Try a second round with different K
    if time.monotonic() < p1_end - 60:
        a_sg2, sc_sg2 = _subgradient_phase(a_sg, min(p1_end - 10, time.monotonic() + 200),
                                           rng, K_peaks=3, lr=0.03)
        update_best(a_sg2, sc_sg2, "subgrad_K3")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: LP peak balancing (200s) ===
    print("Phase 2: LP peak balancing")
    p2_end = min(deadline, t_start + 1050.0)
    if best_seq is not None:
        a_lp, sc_lp = _lp_phase(best_seq, p2_end, K=20, max_rounds=30)
        update_best(a_lp, sc_lp, "lp_phase")

        if time.monotonic() < p2_end - 30:
            a_lp2, sc_lp2 = _lp_phase(best_seq, p2_end, K=40, max_rounds=20)
            update_best(a_lp2, sc_lp2, "lp_K40")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: L-BFGS polish (400s) ===
    print("Phase 3: L-BFGS polish")
    p3_end = min(deadline, t_start + 1450.0)
    if best_seq is not None:
        a_lb, sc_lb = _run_lbfgs(best_seq, p3_end, maxiter_per_beta=5000)
        update_best(a_lb, sc_lb, "lbfgs_polish")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Fine subgradient polish (200s) ===
    print("Phase 4: Fine subgradient polish")
    p4_end = min(deadline, t_start + 1650.0)
    if best_seq is not None:
        a_fine, sc_fine = _adam_fine_polish(best_seq, p4_end, lr=0.008, K_peaks=3)
        update_best(a_fine, sc_fine, "fine_polish")

        if time.monotonic() < p4_end - 30:
            a_fine2, sc_fine2 = _adam_fine_polish(best_seq, p4_end, lr=0.003, K_peaks=2)
            update_best(a_fine2, sc_fine2, "fine_polish_low")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Try n*2 interpolation if time allows ===
    if time.monotonic() < deadline - 50 and best_seq is not None:
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"Phase 5: Interpolating to n={n_new}")
            x0 = np.linspace(0.0, 1.0, len(best_seq))
            x1 = np.linspace(0.0, 1.0, n_new)
            a_interp = np.interp(x1, x0, best_seq)
            a_interp = np.maximum(a_interp, 1e-12)
            a_interp = a_interp / a_interp.sum()
            sc_in = _score_norm(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")

            if time.monotonic() < deadline - 20:
                a_in_pol, sc_in_pol = _adam_fine_polish(a_interp, deadline - 10,
                                                        lr=0.02, K_peaks=5)
                update_best(a_in_pol, sc_in_pol, f"interp_polish_n{n_new}")

    # === Final validation ===
    print(f"\nFinal best_score (norm) = {best_score:.10f}")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
            # Scale back to have sum=1 for consistent scoring
            final_seq = final_seq  # already normalized
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using greedy fallback")
                    final_seq = _greedy_init(512)
            except Exception as ex:
                print(f"Eval error: {ex}")
                final_seq = _greedy_init(512)

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_full(fb):.10f}")
    return [float(x) for x in fb]
