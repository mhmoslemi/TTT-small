from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _contribution_to_peaks(a, top_k=10):
    """For each position i, compute how much it contributes to the top convolution peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    K = min(top_k, len(c))
    top_idx = np.argsort(c)[-K:]
    contrib = np.zeros(n)
    for k in top_idx:

        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            partner = k - j
            if 0 <= partner < n:
                contrib[j] += c[k] * a[partner]
    return contrib

    """Apply exponential tilting: reduce positions with high contribution."""
    z = np.log(np.maximum(a, 1e-15)) - contrib / temperature

    for it in range(n_iters):

        top_lags = np.argsort(c)[-K:]

        for k in top_lags:
            k = int(k)
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            w_val = c[k]
            for j in range(lo, hi+1):
                partner = k - j
                contrib[j] += w_val * a[partner]
        
        # Find the position with highest contribution
        src = int(np.argmax(contrib))

        if len(safe_positions) == 0:

        dst = int(safe_positions[rng.integers(len(safe_positions))])

                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

            else:
                w = e_ / s_
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            # Gradient of 2n * LSE w.r.t. z: sum_i 4n * a_i * (h_i - dot(w,c))
            W = float(np.dot(w, c_))
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            t0 = time.monotonic()
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': 20000,
                         'iprint': -1}
            )
            elapsed = time.monotonic() - t0
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _nesterov_accelerated(a_init, deadline, rng, lr=0.01, momentum=0.9, top_k=5):
    """Nesterov-accelerated gradient descent with top-k weight focusing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()
    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)

    momentum = 0.9
    t_nest = 1.0

    max_iters = 50000
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break

        a_y = _softmax_fn(y)
        c = _full_convolve(a_y, a_y)

        # Top-k weighting
        K = min(top_k, len(c))
        top_idx = np.argsort(c)[-K:]
        cm = c[top_idx[-1]]
        w = np.exp((c - cm) / 0.01)
        w /= w.sum()

        # Gradient
        a_rev = a_y[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W_val = float(np.dot(w, c))
        grad = 4.0 * n * a_y * (h - W_val)

        # Nesterov update
        t_nest = t_nest + 1.0
        m = (t_nest - 1.0) / (t_nest + momentum + 1.0)
        z_new = z - lr * grad
        y = (1.0 - m) * z_new + m * y

        # Periodic evaluation
        if (it + 1) % 20 == 0:
            a_curr = _softmax_fn(y)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_curr.copy()
                z = y.copy()

    return best_a, best_sc


def _peak_shave_vectorized(a_init, deadline, rng, n_trials=50000, top_k=5):
    """Vectorized peak-shaving local search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        K = min(top_k, len(c))
        top_idx = np.argsort(c)[-K:]

        # Compute contribution vector via convolution
        w = np.zeros(2 * n - 1)
        for k in top_idx:
            w[int(k)] = c[int(k)]
        contrib = _conv_grad_weights(a, w)

        # Strategy selection
        strategy = it % 4

        if strategy == 0:
            # Move mass from highest contribution to lowest
            src = int(np.argmax(contrib))
            # Safe positions: low contribution
            safe_threshold = np.percentile(contrib, 30)
            safe = np.where(contrib < safe_threshold)[0]
            if len(safe) == 0:
                continue
            dst = int(safe[rng.integers(len(safe))])
            frac = rng.uniform(0.1, 0.4)
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

        elif strategy == 1:
            # Block perturbation
            block_size = int(rng.integers(4, min(32, n // 16)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        elif strategy == 2:
            # Top-peak specific: reduce the max-lag contributors
            max_lag = int(top_idx[-1])
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                worst = int(pairs[np.argmax(products)])
                safe = np.where(contrib < np.percentile(contrib, 20))[0]
                if len(safe) > 0:
                    dst = int(safe[rng.integers(len(safe))])
                    frac = rng.uniform(0.2, 0.5)
                    amt = a[worst] * frac
                    a_trial = a.copy()
                    a_trial[worst] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s > 1e-10:
                        a_trial /= s
                        sc = _score_norm_fn(a_trial)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()
        else:
            # Symmetric pair swap
            i1 = int(rng.integers(n // 2))
            i2 = n - 1 - i1
            if i1 == i2:
                continue
            a_trial = a.copy()
            delta = rng.uniform(-1, 1) * 0.05 * a.mean()
            a_trial[i1] += delta
            a_trial[i2] -= delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a, best_sc


def _subgradient_polish(a_init, deadline, n_iters=10000, step_scale=0.01):
    """Subgradient method directly targeting the max convolution peak."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    t = 0

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1
        c = _full_convolve(a, a)
        max_k = int(np.argmax(c))
        max_val = c[max_k]

        # Subgradient: 2*a shifted by max_k
        i_min = max(0, max_k - (n - 1))
        i_max = min(n - 1, max_k)
        grad = np.zeros(n)
        grad[i_min:i_max + 1] = 2.0 * a[max_k - i_max:max_k - i_min + 1][::-1]

        # Adaptive step size
        step = step_scale / np.sqrt(t)
        a_trial = a - step * grad
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        sc = _score_norm_fn(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse init -> L-BFGS continuation -> Nesterov
    -> peak-shaving -> multi-restart -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_work = 8192
    if parent is not None:
        n_work = len(parent)
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={n_work}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        parent = _make_triangular(n_work)
        sc_p = _score_norm_fn(parent)
        print(f"No parent, triangular n={n_work}, score={sc_p:.10f}")
        update_best(parent, sc_p, "init")

    print(f"Working with n={n_work}")

    # === Phase 0: Diverse initialization (40s) ===
    phase0_end = min(deadline, t_start + 40.0)
    profiles = [
        _make_triangular(n_work),
        _make_uniform(n_work),
        _make_raised_cosine(n_work, 1.0),
        _make_raised_cosine(n_work, 2.0),
        _make_raised_cosine(n_work, 0.5),
        _make_power_profile(n_work, 1.0),
        _make_power_profile(n_work, 2.0),
        _make_power_profile(n_work, 4.0),
        _make_power_profile(n_work, 8.0),
        _make_bimodal(n_work, 0.3, 0.15),
        _make_bimodal(n_work, 0.5, 0.2),
        _make_bimodal(n_work, 0.4, 0.1),
        _make_w_profile(n_work, 2, 0.3),
        _make_w_profile(n_work, 3, 0.2),
        _make_w_profile(n_work, 4, 0.15),
        _make_w_profile(n_work, 5, 0.1),
        _make_quadratic(n_work, 0.5, -0.3),
        _make_quadratic(n_work, 1.0, -0.5),
        _make_quadratic(n_work, 0.8, 0.2),
        _make_spectral(n_work, n_fourier=32, seed=int(seed)),
        _make_spectral(n_work, n_fourier=64, seed=int(seed) + 1),
    ]

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # === Phase 1: L-BFGS continuation on top candidates (600s) ===
    phase1_end = min(deadline, t_start + 640.0)
    top_candidates = [candidates[0][1], candidates[1][1]] if len(candidates) >= 2 else [candidates[0][1]]
    if parent is not None:
        top_candidates.append(parent.copy())

    for ci, start_p in enumerate(top_candidates[:2]):
        if time.monotonic() > phase1_end - 60:
            break
        sc_start = _score_norm_fn(start_p)
        print(f"Phase 1.{ci}: L-BFGS from score={sc_start:.10f}")
        dl = min(phase1_end - 30, time.monotonic() + 300.0)
        a_best, sc_best = _lbfgs_logsumexp(start_p, dl, rng, max_wall_time=280.0, beta_scale=1.0, min_beta_frac=0.00003)
        update_best(a_best, sc_best, f"lbfgs_c{ci}")

    # Second L-BFGS pass with smaller beta on best
    if best_seq is not None and time.monotonic() < phase1_end - 30:
        dl = min(phase1_end - 10, time.monotonic() + 200.0)
        a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, rng, max_wall_time=180.0, beta_scale=0.4, min_beta_frac=0.00002)
        update_best(a_best, sc_best, "lbfgs_pass2")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov accelerated gradient (300s) ===
    phase2_end = min(deadline, t_start + 940.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: Nesterov accelerated")
        dl = min(phase2_end - 10, time.monotonic() + 250.0)
        a_best, sc_best = _nesterov_accelerated(best_seq, dl, rng, lr=0.008, top_k=3)
        update_best(a_best, sc_best, "nesterov_k3")

        if time.monotonic() < phase2_end - 30:
            dl2 = min(phase2_end - 10, time.monotonic() + 100.0)
            a_best2, sc_best2 = _nesterov_accelerated(best_seq, dl2, rng, lr=0.004, top_k=7)
            update_best(a_best2, sc_best2, "nesterov_k7")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak-shaving local search (400s) ===
    phase3_end = min(deadline, t_start + 1340.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Peak-shaving local search")
        dl = min(phase3_end - 10, time.monotonic() + 350.0)
        a_best, sc_best = _peak_shave_vectorized(best_seq, dl, rng, n_trials=80000, top_k=5)
        update_best(a_best, sc_best, "peak_shave")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Multi-restart (250s) ===
    phase4_end = min(deadline, t_start + 1590.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Multi-restart")
        n_restarts = 4
        per_restart = max(30.0, (phase4_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            noise = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase4_end - 10, time.monotonic() + per_restart)
            a_best, sc_best = _lbfgs_logsumexp(p_start, dl, rng,
                                               max_wall_time=per_restart - 10,
                                               beta_scale=0.6, min_beta_frac=0.0001)
            update_best(a_best, sc_best, f"restart{i}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Subgradient polish + final L-BFGS (150s) ===
    phase5_end = min(deadline, t_start + 1740.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        remaining = max(30.0, phase5_end - time.monotonic())
        print(f"Phase 5: Final polish, remaining={remaining:.0f}s")

        # Subgradient
        if remaining > 50:
            dl = min(phase5_end - 20, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _subgradient_polish(best_seq, dl, n_iters=20000, step_scale=0.005)
            update_best(a_best, sc_best, "subgrad")

        # Final L-BFGS with very small beta
        if time.monotonic() < phase5_end - 20:
            dl = min(phase5_end - 10, time.monotonic() + 80.0)
            a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, rng,
                                               max_wall_time=70.0,
                                               beta_scale=0.15, min_beta_frac=0.00001)
            update_best(a_best, sc_best, "final_lbfgs")

        # Final local search
        if time.monotonic() < phase5_end - 10:
            dl = min(phase5_end - 5, time.monotonic() + 40.0)
            a_best, sc_best = _peak_shave_vectorized(best_seq, dl, rng, n_trials=20000, top_k=3)
            update_best(a_best, sc_best, "final_local")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator
        if len(final_seq) <= 32768:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular(8192)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
