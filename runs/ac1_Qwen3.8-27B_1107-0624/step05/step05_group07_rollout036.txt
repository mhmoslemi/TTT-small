"""
Optimized ramp-initialization + LP minimax + L-BFGS continuation
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: non-palindromic (ramp) sequences achieve ~4/3 asymptotically,
much better than palindromic sequences which are bounded below by 2.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int64), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f, max_time=30):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters=100, deadline=None, step_size=0.3):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(f, f)
        M = float(c.max())
        g, t_val = _lp_minimax(f)
        if g is None:
            break
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        if it % 30 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003, 0.0002,
                 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002,
                 0.00001, 0.000005]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(300, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _single_beta_lbfgs(a_init, beta, deadline, maxiter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            _logsumexp_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_logsumexp_grad,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': maxiter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm_fn(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _make_ramp(n, power=1.0):
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** power
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s


def _make_ramp_variants(n):
    """Generate several ramp-like initializations."""
    variants = []
    # Standard ramp
    variants.append(_make_ramp(n, 1.0))
    # Power ramps
    for p in [0.5, 1.5, 2.0, 3.0]:
        variants.append(_make_ramp(n, p))
    # Exponential ramps
    for alpha in [0.1, 0.3, 0.5, 1.0]:
        t = np.arange(n, dtype=np.float64)
        a = np.exp(alpha * t / n)
        a = a / a.sum()
        variants.append(a)
    # Ramp with cosine modulation
    t = np.linspace(0, 1, n)
    a = (t + 0.1) * (0.5 + 0.5 * np.cos(np.pi * t))
    a = np.maximum(a, 1e-12)
    variants.append(a / a.sum())
    # Inverse ramp (concentrated at start)
    t = np.arange(n, dtype=np.float64)
    a = (n - t) ** 1.0
    variants.append(a / a.sum())
    return variants


def _interp_to_size(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _greedy_coord_descent(a_init, deadline, max_rounds=20):
    """Fast coordinate descent with smaller step grid."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    steps = [-0.2, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.2]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in steps:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Phase 0: Try ramp initializations at multiple sizes
    print("Phase 0: Ramp initializations")
    n_candidates = [2048, 3072, 4096]

    # First, try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc_p = _score_norm_fn(parent)
                update_best(parent, sc_p, "parent")
                print(f"  Parent loaded: n={len(parent)}, score={sc_p:.10f}")
    except Exception:
        pass

    # Try ramps at different sizes
    ramp_best = None
    ramp_best_score = np.inf
    for n in n_candidates:
        if time.monotonic() > deadline:
            break
        for var_idx, a_ramp in enumerate(_make_ramp_variants(n)):
            sc = _score_norm_fn(a_ramp)
            if sc < ramp_best_score:
                ramp_best_score = sc
                ramp_best = a_ramp.copy()
                update_best(a_ramp, sc, f"ramp_{var_idx}_n{n}")

    print(f"  Best ramp score: {ramp_best_score:.10f}")

    # Phase 1: LP minimax on the best ramp
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\nPhase 1: LP minimax iterations")

    n_lp = len(ramp_best)
    a_cur = ramp_best.copy()

    # Multiple rounds of LP with decreasing step size
    for lp_round in range(8):
        if time.monotonic() > phase1_end:
            break
        remaining_iters = max(5, int((phase1_end - time.monotonic()) / 5.0))
        step = max(0.1, 0.5 / (lp_round + 1))
        a_cur, sc_cur = _lp_iterate(a_cur, n_iters=min(remaining_iters, 50),
                                     deadline=phase1_end, step_size=step)
        update_best(a_cur, sc_cur, f"lp_round{lp_round}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS continuation at the best n
    phase2_end = min(deadline, t_start + 1200.0)
    print(f"\nPhase 2: L-BFGS fine continuation")

    n_curr = len(best_seq)
    a_start = best_seq.copy()

    # Run L-BFGS continuation
    per_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
    p_best, sc_best = _run_lbfgs_continuation(a_start, per_dl)
    update_best(p_best, sc_best, "lbfgs_main")

    # Second L-BFGS run from a slightly perturbed starting point
    if time.monotonic() < phase2_end - 60:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.02 * rng.normal(size=len(z))
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_perturbed = _softmax_fn(z)
        per_dl2 = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.4)
        p_best2, sc_best2 = _run_lbfgs_continuation(a_perturbed, per_dl2)
        update_best(p_best2, sc_best2, "lbfgs_perturbed")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Try to improve at a different n
    phase3_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 3: Cross-n exploration")

    if best_seq is not None:
        # Try interpolating to a different size and optimizing
        target_n = 4096 if len(best_seq) != 4096 else 3072
        if time.monotonic() < phase3_end - 60:
            a_interp = _interp_to_size(best_seq, target_n)
            sc_interp = _score_norm_fn(a_interp)
            update_best(a_interp, sc_interp, f"interp_n{target_n}")

            per_dl3 = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.7)
            p_best3, sc_best3 = _run_lbfgs_continuation(a_interp, per_dl3)
            update_best(p_best3, sc_best3, f"lbfgs_n{target_n}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Ultra-fine L-BFGS with tiny beta
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"\nPhase 4: Ultra-fine optimization")

    if best_seq is not None and time.monotonic() < phase4_end:
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        # Very tiny beta L-BFGS
        beta_tiny = 1e-6 * max_c_final
        per_dl4 = min(phase4_end - 10, time.monotonic() + 100.0)
        p_best4, sc_best4 = _single_beta_lbfgs(best_seq, beta_tiny, per_dl4, maxiter=15000)
        update_best(p_best4, sc_best4, "ultra_fine_lbfgs")

        # Even tinier
        if time.monotonic() < phase4_end - 20:
            beta_ultra = 5e-7 * max_c_final
            per_dl4b = min(phase4_end - 5, time.monotonic() + 50.0)
            p_best4b, sc_best4b = _single_beta_lbfgs(best_seq, beta_ultra, per_dl4b, maxiter=10000)
            update_best(p_best4b, sc_best4b, "ultra_ultra_lbfgs")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish
    phase5_end = deadline
    print(f"\nPhase 5: Coordinate descent polish")

    if best_seq is not None and time.monotonic() < phase5_end:
        per_dl5 = phase5_end - 10
        p_best5, sc_best5 = _greedy_coord_descent(best_seq, per_dl5, max_rounds=10)
        update_best(p_best5, sc_best5, "coord_descent")

        # Final tiny-beta L-BFGS
        if time.monotonic() < phase5_end - 15:
            c_fin = _full_convolve(best_seq, best_seq)
            mc = float(c_fin.max())
            beta_final = 1e-7 * mc
            per_dl5b = min(phase5_end - 5, time.monotonic() + 30.0)
            p_best5b, sc_best5b = _single_beta_lbfgs(best_seq, beta_final, per_dl5b, maxiter=5000)
            update_best(p_best5b, sc_best5b, "final_tiny_beta")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL SCORE: {sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 6000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"EVALUATOR SCORE: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        print("Evaluator gave better score!")
            except Exception as ex:
                print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback: simple ramp
    n_fb = 2048
    fallback = _make_ramp(n_fb, 1.0)
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
