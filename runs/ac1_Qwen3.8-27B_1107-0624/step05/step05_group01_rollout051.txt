**Structural insight:**

The objective is `2n * max_k (a*a)[k] / (sum a)^2`.

If we think of `a` as a probability distribution (sum=1), then `(a*a)[k]` is the probability of two independent draws differing by `k`. We want to minimize the maximum probability.

This is like wanting a distribution on `{0, ..., n-1}` such that the difference distribution is as flat as possible. The difference distribution has `2n-1` values, and we want them all to be as small as possible (ideally `1/(2n-1)`).

For a uniform distribution, the difference distribution is triangular (peak at 0), giving max = `1/n` and objective = `2`.

For the "optimal" distribution, we want the difference distribution to be as flat as possible. This is related to the concept of "difference sets" in combinatorics, but in the continuous/weighted setting.

A key question: what's the theoretical lower bound? 

We know `sum_k (a*a)[k] = 1` (for sum(a)=1). So the average of the `2n-1` values is `1/(2n-1)`. The max is at least the average, so `max >= 1/(2n-1)`, giving objective >= `2n/(2n-1)` ≈ 1.

But there are additional constraints. For example, `(a*a)[n-1] = sum_i a[i] a[n-1-i]`. If `a` is concentrated on one side, this can be small. But then other peaks will be large.

2*n*max(conv(a,a))/sum(a)^2 with non-negative a.
Uses log-sum-exp smoothing with adaptive beta, multiplicative (mirror) updates,
peak-targeted subgradient steps, and diverse random restarts.
"""
import numpy as np
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a)=1, returns 2n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_and_weights(c, beta):
    """Compute log-sum-exp of c with temperature beta, and the softmax weights."""
    cm = c.max()
    shifted = (c - cm) / beta
    ms = shifted.max()
    e = np.exp(shifted - ms)
    w = e / e.sum()
    lse = cm + beta * (ms + np.log(e.sum()))
    return float(lse), w

def _grad_lse(z, beta, n):
    """Gradient of 2n*lse(conv(softmax(z), softmax(z))) w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    lse, w = _lse_and_weights(c, beta)
    # grad of lse w.r.t. a: w convolved with reversed a
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return 0.0, np.zeros_like(z)
    return float(2.0 * n * lse), grad

def _nesterov_optimize(a_init, deadline, beta_init=None, max_iter=500000):
    """Nesterov momentum optimization on softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    c0 = _full_convolve(a_init, a_init)
    m0 = c0.max()
    if beta_init is None:
        beta = 0.05 * m0
    else:
        beta = beta_init
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    lr = 0.1  # base learning rate
    lr_decay = 0.9999
    beta_decay = 0.9995
    min_beta = 1e-8 * m0
    
    no_improve = 0
    check_every = 20
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Nesterov look-ahead
        z_look = z + mu * v
        
        a = _softmax(z_look)
        c = _full_convolve(a, a)
        lse_val, w = _lse_and_weights(c, beta)
        
        # Gradient
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Update
        v = mu * v - lr * grad
        z = z + v
        z = z - z.mean()
        z = np.clip(z, -100, 100)
        
        # Adaptive learning rate based on gradient norm
        gn = np.linalg.norm(grad)
        if gn > 1e-300:
            # Adjust lr to keep step size reasonable
            step_norm = np.linalg.norm(v)
            target_step = 0.5  # target relative step in z-space
            if step_norm > 0:
                lr *= max(0.5, min(2.0, target_step / (step_norm / lr + 1e-300)))
        
        # Decay
        lr *= lr_decay
        beta *= beta_decay
        beta = max(beta, min_beta)
        
        # Check score periodically
        if (it + 1) % check_every == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 200:
                # Restart momentum
                v = v * 0.5
                no_improve = 0
                lr *= 1.5
    
    return best_a, best_score

def _peak_targeted_subgradient(a_init, deadline, K=10, max_iter=200000):
    """Direct subgradient method targeting the top K peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    lr = 0.01
    v = np.zeros(n)
    mu = 0.85
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = best_a
        c = _full_convolve(a, a)
        M = c.max()
        
        # Get top K peak indices
        top_k = np.argsort(c)[::-1][:K]
        
        # Compute gradient as weighted combination of peak subgradients
        # Weight by how close each peak is to M
        weights = np.zeros(2*n-1)
        for k in top_k:
            weights[k] = (c[k] / M) ** 4  # concentrate on highest peaks
        
        # Gradient = sum_k weights[k] * d(c[k])/d(a)
        # d(c[k])/d(a)[i] = 2*a[k-i] for valid k-i
        grad = np.zeros(n)
        for k in top_k:
            wi = weights[k]
            # d(c[k])/d(a)[i] = 2*a[k-i] for 0 <= k-i < n
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi+1] += wi * 2.0 * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)][::-1]
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-300:
            break
        
        # Center the gradient (to maintain sum constraint approximately)
        grad_centered = grad - np.dot(grad, a) / max(a.sum(), 1e-300)
        
        # Momentum update
        v = mu * v - lr / gn * grad_centered
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score_norm(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        else:
            # Accept with some probability (simulated annealing)
            # Always reject for now to be safe
            pass
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr *= 0.995
            mu = min(mu + 0.0001, 0.95)
    
    return best_a, best_score

def _adapt_peak_subgrad(a_init, deadline, K=20, max_iter=300000):

    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c = _full_convolve(best_a, best_a)
    M0 = c.max()
    
    lr = 0.005 * M0 / np.max(best_a) if np.max(best_a) > 0 else 0.005
    v = np.zeros(n)
    mu = 0.9
    patience = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = best_a
        c = _full_convolve(a, a)
        M = c.max()
        
        # Get active peaks: top K or those > 0.8*M
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        thresh = 0.85 * M
        more = np.where(c >= thresh)[0]
        active = np.unique(np.concatenate([active, more]))
        if len(active) > 100:
            active = active[:100]
        
        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            # d(c[k])/d(a)[i] = 2*a[k-i] for valid indices
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                indices = np.arange(i_lo, i_hi + 1)
                vals = 2.0 * a[k - indices]
                grad[indices] += vals
        
        # Scale gradient by peak values (focus on high peaks)
        # Actually, let's weight by c[k]
        grad_weighted = np.zeros(n)
        for k in active:
            w = c[k] / M
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                indices = np.arange(i_lo, i_hi + 1)
                grad_weighted[indices] += w * 2.0 * a[k - indices]
        
        # Project gradient to maintain sum constraint
        # We want a_new to have the same sum as a
        # So we need sum(grad_step) = 0
        # Subtract the weighted average
        mean_g = np.dot(grad_weighted, a) / max(a.sum(), 1e-300)
        grad_proj = grad_weighted - mean_g
        
        # Step
        v = mu * v - lr * grad_proj
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score_norm(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                lr *= 0.5
                v *= 0.3
                patience = 0
                if lr < 1e-10:
                    break
        
        # Slowly decay learning rate
        if it % 200 == 0:
            lr *= 0.98
    
    return best_a, best_score

def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_smooth(c, beta):
    cm = float(c.max())
    shifted = (c - cm) / beta
    ms = float(shifted.max())
    e = np.exp(shifted - ms)
    w = e / e.sum()
    lse = cm + beta * (ms + float(np.log(e.sum())))
    return lse, w


def _lse_obj_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    lse, w = _lse_smooth(c, beta)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    val = float(2.0 * n * lse)
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return val, np.zeros_like(z)
    return val, grad


def _nesterov_smooth(a_init, deadline, beta_init=None, mu=0.93, lr_init=0.15):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    if beta_init is None:
        beta = 0.08 * m0
    else:
        beta = beta_init
    min_beta = 1e-6 * m0
    beta_decay = 0.9997
    lr = lr_init
    lr_decay = 0.99995

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    v = np.zeros(n)
    no_improve = 0
    check_every = 15

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        z_look = z + mu * v
        z_look = z_look - z_look.mean()
        val, grad = _lse_obj_grad(z_look, beta, n)

        if not np.all(np.isfinite(grad)):
            break

        v = mu * v - lr * grad
        z = z + v
        z = z - z.mean()
        z = np.clip(z, -80, 80)

        lr *= lr_decay
        beta = max(beta * beta_decay, min_beta)

        if (it + 1) % check_every == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-11:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 300:
                scale = 1.0 / (1.0 + 0.1 * no_improve / 300.0)
                v *= scale
                no_improve = 0

    return best_a, best_score


def _peak_subgrad_bt(a_init, deadline, K=15):
    """Peak-targeted subgradient with backtracking line search."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    mu = 0.88
    v = np.zeros(n)
    lr = 0.003
    it = 0

    while time.monotonic() < deadline:
        a = best_a
        c = _full_convolve(a, a)
        M = float(c.max())
        if M < 1e-15:
            break

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        thresh = 0.90 * M
        more = np.where(c >= thresh)[0]
        active = np.unique(np.concatenate([active, more]))
        if len(active) > 80:
            active = active[:80]

        grad = np.zeros(n)
        for k in active:
            w = c[k] / M
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                idx = np.arange(i_lo, i_hi + 1)
                grad[idx] += w * 2.0 * a[k - idx]

        mean_g = float(np.dot(grad, a)) / max(float(a.sum()), 1e-300)
        grad = grad - mean_g

        gn = float(np.linalg.norm(grad))
        if gn < 1e-300:
            break

        v = mu * v - lr / gn * grad
        a_try = a + v
        a_try = np.maximum(a_try, 0)
        s = float(a_try.sum())
        if s < 1e-15:
            v *= 0.3
            continue
        a_try = a_try / s

        sc = _score_norm(a_try)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_try.copy()
        else:
            # Backtrack
            for backtrack in range(5):
                t = 0.5 ** backtrack
                a_bt = a + t * v
                a_bt = np.maximum(a_bt, 0)
                s_bt = float(a_bt.sum())
                if s_bt < 1e-15:
                    break
                a_bt = a_bt / s_bt
                sc_bt = _score_norm(a_bt)
                if sc_bt < best_score - 1e-12:
                    best_score = sc_bt
                    best_a = a_bt.copy()
                    break
            else:
                lr *= 0.7
                v *= 0.5

        it += 1
        if it % 500 == 0:
            lr *= 0.97
            mu = min(mu + 0.0005, 0.95)

    return best_a, best_score


def _lp_balance(a_init, deadline, K=25, max_rounds=20):
    """LP-based peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        thresh = 0.92 * M
        additional = np.where(c >= thresh)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 150:
            active = active[:150]

        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(len(active), M * 1.0005)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = float(b_new.sum())
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = _score_norm(best_a)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc_trial = _score_norm(a_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.015, max_iter=300000):
    """Adam optimizer with log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.005 * cm0
    beta_decay = 0.9998
    min_beta = 1e-7 * cm0

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(beta * beta_decay, min_beta)

        shifted = (c - cm) / beta
        ms = float(shifted.max())
        e = np.exp(shifted - ms)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % 40 == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-11:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z += 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _lbfgs_polish(a_init, deadline, beta_init=None, maxiter=20000):
    """Final L-BFGS polish with very small beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    if beta_init is None:
        beta_init = 0.001 * m0

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    try:
        result = minimize(
            _lse_obj_grad, z,
            method='L-BFGS-B',
            args=(beta_init, n),
            jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxls': 100, 'maxfun': 50000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax(result.x)
            s_new = _score_norm(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score


def _try_symmetrize(a):
    """Try symmetrizing and see if it helps."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return a
    a_sym = a_sym / s
    sc_orig = _score_norm(a)
    sc_sym = _score_norm(a_sym)
    if sc_sym < sc_orig:
        return a_sym
    return a


def _greedy_init(n):
    b = np.ones(n)
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: Nesterov+LSE, peak subgradient, LP balance,
    diverse restarts, Adam polish, and L-BFGS final.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        parent = _try_symmetrize(parent)
        update_best(parent, _score_norm(parent), "parent_sym")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm(parent), "greedy_init")

    print(f"Starting: n={n}, best={best_score:.10f}")

    # === Phase 1: Nesterov + log-sum-exp (550s) ===
    print("Phase 1: Nesterov + log-sum-exp")
    p1_end = min(deadline, t_start + 550.0)

    a1, sc1 = _nesterov_smooth(best_seq, p1_end, mu=0.93, lr_init=0.12)
    update_best(a1, sc1, "nesterov_main")

    # Continue with smaller beta
    if time.monotonic() < p1_end - 60:
        c_tmp = _full_convolve(a1, a1)
        m_tmp = float(c_tmp.max())
        a2, sc2 = _nesterov_smooth(a1, p1_end, beta_init=0.02 * m_tmp, mu=0.95, lr_init=0.06)
        update_best(a2, sc2, "nesterov_small_beta")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Peak-targeted subgradient (300s) ===
    print("Phase 2: Peak-targeted subgradient")
    p2_end = min(deadline, t_start + 850.0)

    if time.monotonic() < p2_end:
        a3, sc3 = _peak_subgrad_bt(best_seq, p2_end, K=15)
        update_best(a3, sc3, "peak_subgrad")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP balancing (200s) ===
    print("Phase 3: LP balancing")
    p3_end = min(deadline, t_start + 1050.0)

    if time.monotonic() < p3_end:
        a_lp, sc_lp = _lp_balance(best_seq, p3_end, K=25, max_rounds=15)
        update_best(a_lp, sc_lp, "lp_balance")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Diverse restarts (300s) ===
    print("Phase 4: Diverse restarts")
    p4_end = min(deadline, t_start + 1350.0)

    n_restarts = 6
    remaining = max(0, p4_end - time.monotonic())
    per_restart = remaining / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > p4_end - 10:
            break

        # Different perturbation strategies
        if r == 0:
            # Gaussian noise
            a_pert = best_seq + rng.normal(0, 0.03 / np.sqrt(n), n)
        elif r == 1:
            # Multiplicative noise
            a_pert = best_seq * np.exp(rng.normal(0, 0.05, n))
        elif r == 2:
            # Random subset reduction
            mask = rng.random(n) < 0.2
            a_pert = best_seq.copy()
            a_pert[mask] *= 0.3
        elif r == 3:
            # Center boost
            center = n // 2
            width = n // 4
            a_pert = best_seq.copy()
            a_pert[center - width:center + width] *= 1.3
        elif r == 4:
            # Edge boost
            a_pert = best_seq.copy()
            a_pert[:n // 5] *= 1.5
            a_pert[-n // 5:] *= 1.5
        else:
            # Large random perturbation
            a_pert = best_seq + rng.normal(0, 0.1 / np.sqrt(n), n)

        a_pert = np.maximum(a_pert, 0)
        s_pert = a_pert.sum()
        if s_pert < 1e-15:
            continue
        a_pert = a_pert / s_pert

        t_end = time.monotonic() + max(10, per_restart - 5)
        a_r, sc_r = _nesterov_smooth(a_pert, t_end, mu=0.92, lr_init=0.08)
        update_best(a_r, sc_r, f"restart_{r}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Adam polish (200s) ===
    print("Phase 5: Adam polish")
    p5_end = min(deadline, t_start + 1550.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_a1, sc_a1 = _adam_polish(best_seq, p5_end, lr=0.012)
        update_best(a_a1, sc_a1, "adam_012")

        if time.monotonic() < p5_end - 30:
            a_a2, sc_a2 = _adam_polish(best_seq, p5_end, lr=0.005)
            update_best(a_a2, sc_a2, "adam_005")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: L-BFGS final polish (remaining) ===
    print("Phase 6: L-BFGS final")
    if time.monotonic() < deadline - 20 and best_seq is not None:
        c_f = _full_convolve(best_seq, best_seq)
        m_f = float(c_f.max())
        a_f, sc_f = _lbfgs_polish(best_seq, deadline - 10, beta_init=0.0005 * m_f)
        update_best(a_f, sc_f, "lbfgs_final")

        # Try symmetrized version too
        a_sym = _try_symmetrize(best_seq)
        if not np.allclose(a_sym, best_seq):
            update_best(a_sym, _score_norm(a_sym), "sym_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                    return [float(x) for x in final_seq]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score(fb):.10f}")
    return [float(x) for x in fb]
