"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA (smoothed max) for coarse optimization, Adam in log-space with aggressive
temperature schedule for fine optimization, exact subgradient line search for final polish,
and multi-restart exploration. Key improvements over previous: more temperature phases,
cosine LR decay, exact subgradient finishing, and better time allocation.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """Exact subgradient of 2n*max(c) w.r.t. z in log-space."""

    h = wc[n - 1:2 * n - 1]

def _fista_smoothed(a_init, beta_list, deadline, max_iter_per_beta):
    """FISTA with multiple beta (temperature) levels."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        L = 8.0 * n / beta
        step = 1.0 / L
        a_curr = best_a.copy() if _score_norm(best_a) < _score_norm(a_curr) else a_curr
        a_prev = a_curr.copy()
        t_val = 1.0

            if time.monotonic() > deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)

            if s_y > 1e-30:

            else:

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, lr_final, max_iter_total):
    """Adam optimizer in log-space with cosine LR decay and multiple temperature phases."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    total_iter = max_iter_total
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        phase_start_t = t
        # Estimate iterations for this phase
        remaining_beta = len(beta_schedule) - beta_schedule.index(beta)
        it_target = max(1000, (total_iter - t) // remaining_beta)

        it_count = 0
        while it_count < it_target:
            if time.monotonic() > deadline:
                break
            it_count += 1
            t += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine LR decay
            progress = t / max(total_iter, 1)
            current_lr = lr_final + 0.5 * (lr_init - lr_final) * (1.0 + np.cos(np.pi * progress))

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

                sc = _score_norm(a_chk)

    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        w = 0.15
        a = np.ones(n)
        mask_left = t < w
        mask_right = t > (1.0 - w)
        a[mask_left] = np.sin(np.pi * t[mask_left] / (2.0 * w)) ** 0.5
        a[mask_right] = np.sin(np.pi * (1.0 - t[mask_right]) / (2.0 * w)) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent loaded: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"Starting from scratch, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.8f}, n={n_main}")

    # === Phase 1: FISTA coarse-to-fine (first 15% of time) ===
    print("\n=== Phase 1: FISTA coarse-to-fine ===")
    phase1_end = t_start + 0.15 * (deadline - t_start)
    a_curr = a0.copy()

    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())
    betas_fista = []
    beta = 0.02 * max_cc
    while beta > 0.00005 * max_c0:
        betas_fista.append(beta)
        beta *= 0.5
    betas_fista = betas_fista[:12]

    per_beta_iter = max(2000, int((phase1_end - time.monotonic()) / max(len(betas_fista), 1) * 3000))
    per_beta_iter = min(per_beta_iter, 50000)

    a_curr, sc_curr = _fista_smoothed(a_curr, betas_fista, phase1_end, per_beta_iter)
    a_curr = a_curr / a_curr.sum()
    sc_curr_full = _score(a_curr)
    if sc_curr_full < best_score:
        best_score = sc_curr_full
        best_seq = a_curr.copy()
    print(f"End Phase 1: score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam log-space with aggressive schedule (next 50% of time) ===
    print("\n=== Phase 2: Adam log-space (aggressive) ===")
    phase2_end = t_start + 0.65 * (deadline - t_start)

    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    # More temperature phases, smaller final temperature
    betas_adam = []
    beta = 0.002 * max_cc
    while beta > 0.000005 * max_c0:
        betas_adam.append(beta)
        beta *= 0.6  # slower decrease for more gradual annealing
    betas_adam = betas_adam[:15]
    if len(betas_adam) < 8:
        # Ensure we have enough phases
        beta = 0.000005 * max_c0
        while len(betas_adam) < 10:
            betas_adam.append(beta)
            beta *= 0.5

    print(f"  Adam betas: {len(betas_adam)} phases, first={betas_adam[0]:.2e}, last={betas_adam[-1]:.2e}")

    a_opt, sc_opt = _adam_logspace(
        a_curr, phase2_end, betas_adam,
        lr_init=0.015, lr_final=0.001,
        max_iter_total=800000
    )
    a_opt = a_opt / a_opt.sum()
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
    print(f"End Phase 2: score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart exploration (next 15% of time) ===
    print("\n=== Phase 3: Multi-restart ===")
    phase3_end = t_start + 0.80 * (deadline - t_start)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    if time.monotonic() < phase3_end - 30:
        n_restarts = 4
        per_restart = (phase3_end - time.monotonic() - 15) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break

            if i == 0:
                p_start = _perturb_log(a_work, 0.02, rng)
            elif i == 1:
                p_start = _perturb_log(a_work, 0.04, rng)
            elif i == 2:
                # Structured perturbation: add low-frequency component
                z = np.log(np.maximum(a_work, 1e-15))
                freq = rng.uniform(0.5, 3.0)
                t_idx = np.linspace(0, 2 * np.pi, len(z))
                z = z + 0.03 * np.sin(freq * t_idx)
                z = z - z.mean()
                p_start = _softmax(z)
            else:
                p_start = _perturb_log(a_work, 0.06, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                bs, lr_init=0.01, lr_final=0.002,
                max_iter_total=80000
            )
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: score={sc_full:.10f} (no improvement)")

    # === Phase 4: Exact subgradient line search (next 7% of time) ===
    print("\n=== Phase 4: Exact subgradient line search ===")
    phase4_end = min(deadline, t_start + 0.92 * (deadline - t_start))
    a_sub = best_seq.copy()
    a_sub = a_sub / a_sub.sum()

    if time.monotonic() < phase4_end - 5:
        a_sub, sc_sub = _subgradient_line_search(a_sub, phase4_end, max_iter=200000)
        a_sub = a_sub / a_sub.sum()
        sc_sub_full = _score(a_sub)
        if sc_sub_full < best_score:
            best_score = sc_sub_full
            best_seq = a_sub.copy()
            print(f"  Subgradient: score={sc_sub_full:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Subgradient: score={sc_sub_full:.10f} (no improvement)")

    # === Phase 5: Final ultra-fine Adam polish ===
    print("\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        betas_ult = [0.00003 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp,
                     0.000002 * max_cp]

        per_t = deadline - time.monotonic() - 8
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.004, lr_final=0.0005,
                max_iter_total=300000
            )
            a_pol = a_pol / a_pol.sum()
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"\nFinal best score: {best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(4096, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
