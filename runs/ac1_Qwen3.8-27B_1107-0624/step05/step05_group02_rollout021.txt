    # Find hot positions
    hot = np.argsort(c)[-num_hot:]
    # Compute pressure on each a[j]
    pressure = np.zeros(n)
    for k in hot:
        # c[k] = sum_j a[j] * a[k-j] for valid j
        # Gradient of c[k] w.r.t. a[j] is a[k-j]
        j_range = np.arange(max(0, k-n+1), min(n, k+1))
        a_kj = a[k - j_range]
        pressure[j_range] += a_kj
    # Reduce a where pressure is high
    pressure_norm = pressure / max(pressure.max(), 1e-15)
    a_new = a * (1 - eta * pressure_norm)
    a_new = np.maximum(a_new, 0)
    # Renormalize
    a_new = a_new * (np.sqrt(2*n) / a_new.sum())
    return a_new

import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, w):
    """Compute gradient of sum_k w[k]*c[k] w.r.t. log-parameters z.
    Returns dL/dz[j] = 2*a[j]*(h[j] - W) where h = xcorr(w, a), W = dot(w, c)."""

    c = _full_convolve(a, a)
    W = float(np.dot(w, c))
    a_flipped = a[::-1]
    wc = _full_convolve(w, a_flipped)
    h = wc[n-1:2*n-1]
    return 2.0 * a * (h - W), c, W


def _soft_max_weights(c, num_hot, beta):
    """Compute soft-max weights over top `num_hot` entries of c."""
    k = min(num_hot, len(c))
    # Find threshold: k-th largest value

    s = e.sum()
    if s < 1e-300:
        w_hot = np.ones(k) / k
    else:
        w_hot = e / s
    
    w = np.zeros_like(c)
    w[idx] = w_hot
    return w

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    num_hot = max(4, int(num_hot_frac * (2 * n - 1)))

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        current_lr = lr
        patience = 0
        phase_best = best_val
        iters = 0

        while iters < 80000:
            if time.monotonic() > deadline:
                break
            iters += 1
            t_step += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            w = _soft_max_weights(c, num_hot, beta)
            h = _full_convolve(w, a[::-1])[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 2.0 * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > max_step:
                update *= max_step / un

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if cm < phase_best - 1e-12:
                phase_best = cm
                patience = 0
            else:
                patience += 1

            if cm < best_val:
                best_val = cm
                best_a = a.copy()
                if callback:
                    callback(a, cm, t_step)

            if patience > reheat_patience:
                # Reheat: perturb z to escape local minimum
                sigma = 0.05 * (beta / max_c0 + 0.01)
                z = z + sigma * np.random.default_rng(t_step).standard_normal(n)
                z = z - z.mean()
                z = np.clip(z, -50.0, 50.0)
                patience = 0
                current_lr = lr
                m_buf *= 0.5
                v_buf *= 0.5

            if t_step % 500 == 0:
                sc = float(_full_convolve(_softmax(z), _softmax(z)).max())
                if sc < best_val:
                    best_val = sc
                    best_a = _softmax(z).copy()
                    if callback:
                        callback(best_a, sc, t_step)

            if patience > reheat_patience * 4:
                break

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_val:
        best_val = sc_final
        best_a = a_final.copy()

    return best_a, best_val


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f, M, n, time_limit=45):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for optimal step function heights.
    Uses adaptive Adam + LP kicks + multi-restart + size exploration.
    """
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t0
            print(f"  ** NEW BEST: {best_score:.10f} (n={len(p)}) t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent / parent.sum()
        print(f"Parent loaded: n={n_main}, score={_score(a0):.10f}")
    else:
        n_main = 8192
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")

    update_best(a0)

    # === Phase 1: Fast Adam with adaptive soft-max (0 to 55% budget) ===
    phase1_end = t0 + 0.55 * budget_s
    print(f"\n=== Phase 1: Adam optimization (n={n_main}, until t={phase1_end - t0:.0f}s) ===")

    a_curr = a0.copy()
    step_count = 0

    def cb1(a, val, t):
        nonlocal step_count
        step_count = t
        if t % 5000 == 0:
            normed = a / a.sum() * np.sqrt(2 * len(a))
            print(f"    Adam t={t}, norm_score={val:.10f}, "
                  f"actual={_score(normed):.10f}, elapsed={time.monotonic()-t0:.0f}s")

    a1, val1 = _adam_run(a_curr, phase1_end - 5, lr=0.015,
                          num_hot_frac=0.05, max_step=0.3,
                          reheat_patience=6000, callback=cb1)
    update_best(a1)
    print(f"Phase 1 done: score={val1:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 2: Strategic LP kicks + Adam between kicks (55% to 78%) ===
    phase2_end = t0 + 0.78 * budget_s
    print(f"\n=== Phase 2: LP + Adam hybrid (until t={phase2_end - t0:.0f}s) ===")

    f = a1 / a1.sum()
    n_lp = n_main
    A_cache = None
    f_cached = None
    lp_kicks = 0
    stuck_total = 0

    while time.monotonic() < phase2_end - 60:
        c = _full_convolve(f, f)
        M = float(c.max())

        # LP step
        if A_cache is None or np.max(np.abs(f - f_cached)) > 1e-8:
            A_cache = _build_lp_matrix(f, n_lp)
            f_cached = f.copy()

        g = _lp_step(f, M, n_lp, time_limit=40)

        if g is None:
            stuck_total += 1
            if stuck_total > 8:
                print(f"  LP consistently failing, ending Phase 2 after {lp_kicks} kicks")
                break
        else:
            stuck_total = 0
            g_sum = float(g.sum())
            g_norm = g * (f.sum() / g_sum)

            # Backtracking line search
            improved = False
            for t_try in [0.8, 0.5, 0.3, 0.15, 0.07, 0.03]:
                f_new = (1 - t_try) * f + t_try * g_norm
                f_new = np.maximum(f_new, 0)
                fns = float(f_new.sum())
                if fns < 1e-10:
                    continue
                f_new = f_new / fns
                c_new = _full_convolve(f_new, f_new)
                M_new = float(c_new.max())
                if M_new < M * (1 - 1e-10):
                    f = f_new
                    M = M_new
                    improved = True
                    lp_kicks += 1
                    if lp_kicks % 2 == 0:
                        normed = f * np.sqrt(2 * n_lp)
                        print(f"  LP kick {lp_kicks}: M={M:.10f}, "
                              f"score={_score(normed):.10f}, t={time.monotonic()-t0:.0f}s")
                    update_best(f * np.sqrt(2 * n_lp))
                    break

            if not improved:
                stuck_total += 1

        # Adam polish between LP kicks
        if time.monotonic() < phase2_end - 80:
            a_tmp, _ = _adam_run(f, time.monotonic() + min(60, phase2_end - time.monotonic() - 70),
                                  lr=0.005, max_step=0.2, reheat_patience=4000)
            f = a_tmp / a_tmp.sum()
            A_cache = None
            f_cached = None
            update_best(f * np.sqrt(2 * n_lp))

    update_best(f * np.sqrt(2 * n_lp))
    print(f"Phase 2 done: {lp_kicks} LP kicks, best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 3: Multi-restart Adam polish (78% to 92%) ===
    phase3_end = t0 + 0.92 * budget_s
    print(f"\n=== Phase 3: Multi-restart polish (until t={phase3_end - t0:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 6
        per_restart = (phase3_end - time.monotonic() - 20) / max(1, n_restarts)
        sigmas = [0.02, 0.04, 0.06, 0.1, 0.15, 0.2]

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break
            remaining_i = min(per_restart, phase3_end - time.monotonic() - 10)
            if remaining_i < 30:
                break
            p_start = _perturb(a_best_norm, sigmas[i], rng)
            a_r, val_r = _adam_run(p_start, time.monotonic() + remaining_i,
                                    lr=0.008, max_step=0.25, reheat_patience=5000)
            update_best(a_r)
            print(f"  Restart {i+1} (sigma={sigmas[i]}): val={val_r:.10f}, "
                  f"best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 4: Size exploration (92% to 97%) ===
    phase4_end = t0 + 0.97 * budget_s
    print(f"\n=== Phase 4: Size exploration (until t={phase4_end - t0:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase4_end - 50:
        target_ns = []
        if n_main != 4096:
            target_ns.append(4096)
        if n_main != 16384:
            target_ns.append(16384)

        for target_n in target_ns:
            if time.monotonic() > phase4_end - 60:
                break
            p_interp = _interp_profile(best_seq / best_seq.sum(), target_n)
            update_best(p_interp * np.sqrt(2 * target_n))

            remaining = phase4_end - time.monotonic() - 40
            if remaining > 60:
                a_r, _ = _adam_run(p_interp, time.monotonic() + remaining,
                                    lr=0.01, max_step=0.25, reheat_patience=5000)
                update_best(a_r * np.sqrt(2 * target_n))
                print(f"  n={target_n}: best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 5: Final polish (97% to deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 8
        if remaining > 20:
            a_final, _ = _adam_run(a_best_norm, time.monotonic() + remaining,
                                    lr=0.003, num_hot_frac=0.02,
                                    max_step=0.1, reheat_patience=10000)
            update_best(a_final)

    # Final validation
    print(f"\n=== Final ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t0:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
