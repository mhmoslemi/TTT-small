"""
Peak-targeting hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses Adam with adaptive log-sum-exp + direct peak gradient, structured perturbations,
multi-scale exploration, and coordinate-descent polish.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Assumes sum(a)=1, returns 2*n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_gradient(z, beta, use_direct_peak, peak_fraction=0.0):
    """Compute gradient of log-sum-exp (or direct max) objective w.r.t. z.
    Returns (grad_z, c, cm) where c is the convolution and cm is its max."""
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n), c, cm

    if use_direct_peak:
        # Direct subgradient: only the argmax peak
        k_star = int(np.argmax(c))
        w = np.zeros_like(c)
        w[k_star] = 1.0
        g_factor = 1.0
    else:
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        g_factor = 1.0

    # g[m] = 4*n * sum_k w[k] * a[k-m]  (cross-correlation)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    # w has length 2n-1, a_rev has length n, result has length 3n-2
    g = 4.0 * n * w_conv[n - 1:2 * n - 1]

    grad_a = g
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z, c, cm


def _adam_step(z, grad_z, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m = b1 * m + (1 - b1) * grad_z
    v = b2 * v + (1 - b2) * (grad_z * grad_z)
    mh = m / (1 - b1 ** t)
    vh = v / (1 - b2 ** t)
    z_new = z - lr * mh / (np.sqrt(vh) + eps)
    return z_new, m, v


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _run_adam_hybrid(a_init, deadline, rng, lr_init=0.5, max_iter=500000,

def _coordinate_descent(a_init, deadline, rng, max_rounds=5000):
    """Targeted coordinate descent on top convolution peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top peaks
        n_peaks = min(3, len(c))
        peak_indices = np.argsort(c)[::-1][:n_peaks]

        improved = False
        for pk in peak_indices:
            # Find top contributors to c[pk]
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:6]

            for idx_pos in order[:3]:
                idx = valid_i[idx_pos]
                jdx = pk - idx
                if jdx < 0 or jdx >= n:
                    continue
                # Try reducing z[idx] (and optionally z[jdx])
                for delta in [0.002, 0.005, 0.01, 0.02]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    if idx != jdx:
                        z_trial[jdx] -= delta * 0.5
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing small values to redistribute
            small_idx = np.argsort(a_cur)[:3]
            for idx in small_idx:
                for delta in [0.005, 0.01]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

            if not improved and rnd > 50:
                # Small random restart
                z = z + 0.01 * rng.normal(size=n)
                z = z - z.mean()

    return best_a, best_score


def _run_lbfgs_refine(a_init, deadline, rng, n_betas=6):
    """L-BFGS with log-sum-exp continuation for final refinement."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    betas = np.geomspace(0.01 * max_c0, 1e-7 * max_c0, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        level_deadline = time.monotonic() + min(time_budget, 120.0)

        maxiter = min(5000, int((maxiter) * (time_budget / 30.0))) if time_budget > 0 else 1000
        maxiter = max(500, min(maxiter, 10000))

        def obj(z_in, b=beta):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm_in = float(c_in.max())
            if cm_in < 1e-30:
                return 0.0
            shifted = (c_in - cm_in) / b
            logsum = cm_in / b + float(np.log(np.sum(np.exp(shifted))))
            return 2.0 * n * b * logsum

        def grad(z_in, b=beta):

            if result.x is not None and np.all(np.isfinite(result.x)):

                    best_a = a_new.copy()

        except Exception:

    return best_a, best_score

            lr_init=0.5, max_iter=1000000,
            beta_start_frac=0.05, beta_end_frac=1e-6,
            beta_transition=30000, use_direct_from=10000
        )
        update_best(a_adam, sc_adam, "deep_adam")

        # Second Adam run with different hyperparams
        if time.monotonic() < phase2_end - 60:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            # Larger perturbation
            z = z + 0.05 * rng.normal(size=n_best)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            adam_dl2 = min(phase2_end - 10, time.monotonic() + 180.0)
            a_adam2, sc_adam2 = _run_adam_hybrid(
                a_pert, adam_dl2, rng,
                lr_init=0.3, max_iter=300000,
                beta_start_frac=0.02, beta_end_frac=1e-6,
                beta_transition=20000, use_direct_from=5000
            )
            update_best(a_adam2, sc_adam2, "adam_perturbed")

    print(f"  Phase 2 best: {best_score:.10f}")

    # --- Phase 3: L-BFGS refinement (25% remaining) ---
    phase3_end = t_start + (budget_s - 20.0) * 0.92
    print(f"\nPhase 3: L-BFGS refinement (until t={phase3_end - t_start:.0f}s)")

    if best_seq is not None and time.monotonic() < phase3_end - 10:
        lbfgs_dl = min(phase3_end - 5, time.monotonic() + (phase3_end - time.monotonic()) * 0.7)
        a_lbfgs, sc_lbfgs = _run_lbfgs_refine(best_seq, lbfgs_dl, rng, n_betas=4)
        update_best(a_lbfgs, sc_lbfgs, "lbfgs_refine")

    print(f"  Phase 3 best: {best_score:.10f}")

    # --- Phase 4: Coordinate descent polish + final Adam ---
    print(f"\nPhase 4: Coordinate descent + final polish")

    if best_seq is not None and time.monotonic() < deadline - 10:
        cd_dl = min(deadline - 30, time.monotonic() + 120.0)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_dl, rng, max_rounds=3000)
        update_best(a_cd, sc_cd, "coord_descent")

        # Final tiny Adam
        if time.monotonic() < deadline - 10:
            final_dl = min(deadline - 5, time.monotonic() + 60.0)
            a_final_adam, sc_final = _run_adam_hybrid(
                best_seq, final_dl, rng,
                lr_init=0.1, max_iter=100000,
                beta_start_frac=0.005, beta_end_frac=1e-8,
                beta_transition=5000, use_direct_from=1000
            )
            update_best(a_final_adam, sc_final, "final_adam")

    print(f"\n  Final best: {best_score:.10f}")

    # --- Return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _make_ramp(4096, 1.5)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
