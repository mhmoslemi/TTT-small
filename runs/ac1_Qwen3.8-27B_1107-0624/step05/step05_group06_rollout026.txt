from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix_sparse(a):
    """Build sparse (2n-1) x n matrix C such that C @ b = full_convolve(a, b)."""
    n = len(a)
    C = lil_matrix((2*n-1, n))
    for i in range(n):
        k_min = i
        k_max = min(i + n, 2*n-1)
        C[k_min:k_max, i] = a[:k_max-k_min]
    return C.tocsr()

def _solve_lp_direction(a, M, time_limit=30):
    """Solve: max sum(b) s.t. C @ b <= M, b >= 0 using HiGHS."""
    n = len(a)
    c_obj = -np.ones(n)  # minimize -sum = maximize sum
    
    C = _build_conv_matrix_sparse(a)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub,
                  bounds=[(0, None)]*n, method='highs',
                  options={'time_limit': time_limit})
    
    if not res.success or res.x is None:
        return None
    b_val = np.maximum(res.x, 0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

Uses scipy HiGHS solver for LP direction finding with line search,
Nesterov-accelerated subgradient for polishing, and Fourier phase
perturbation for diversification.

import time
from scipy.sparse import coo_matrix, csr_matrix
from scipy.optimize import linprog

    N = _next_pow2(out_n)

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

def _precompute_conv_indices(n):
    """Precompute row and column indices for the convolution matrix."""
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])
    cols = np.repeat(np.arange(n), n)

    w[active] = 1.0 / active.sum()
    # grad[i] = sum_j a[j] * w[i+j] for valid j
    # = correlate(w, a, 'valid')[i]
    grad = np.correlate(w, a, mode='valid')
    return grad


def _nesterov_step(a, v, lr, n):
    """One Nesterov accelerated subgradient step."""
    grad = _compute_subgrad(a)
    a_new = np.maximum(a - lr * (2 * grad - v), 1e-12)
    a_new = a_new / a_new.sum()
    v_new = grad
    return a_new, v_new


def _phase_perturb(a, rng):
    """Fourier phase randomization for diversification."""
    n = len(a)
    N = _next_pow2(2 * n)
    A = np.fft.rfft(a, N)
    for _ in range(10):
        phase = rng.uniform(0, 2 * np.pi, len(A))
        A_pert = A * np.exp(1j * phase)
        a_pert = np.fft.irfft(A_pert, N)[:n]
        a_pert = np.maximum(a_pert, 0)
        s = a_pert.sum()
        if s > 0.01:
            return a_pert / s
    return a.copy()


def _gaussian_perturb(a, rng, sigma=0.02):
    """Add Gaussian noise in log domain."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z += sigma * rng.standard_normal(n)
    z = z - z.mean()
    a_new = np.exp(z)
    a_new = a_new / a_new.sum()
    return a_new


def _make_uniform(n):
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-based iterative optimization with Nesterov subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def do_update(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"t={time.monotonic()-t_start:.1f}s")

    # --- Initialize ---
    try:
        parent = np.array(height_sequence_1, dtype=np.float64).ravel()
        parent = np.clip(parent, 0, 1000.0)
        if parent.sum() < 0.01:
            parent = _make_uniform(2048)
    except Exception:
        parent = _make_uniform(2048)

    n = len(parent)
    a = parent / parent.sum()
    sc = _score(a)
    print(f"Initial: n={n}, score={sc:.10f}")
    do_update(a, sc, "init")

    # Precompute indices
    row_idx, col_idx = _precompute_conv_indices(n)
    print(f"Precomputed conv matrix indices ({len(row_idx)} entries)")

    # --- Phase 1: LP-based iteration ---
    print(f"\nPhase 1: LP-based iteration (n={n})")
    M = float(_full_convolve(a, a).max())
    print(f"  Initial M = {M:.10f}")

    prev_score = sc
    stall_count = 0

    for iteration in range(300):
        if time.monotonic() > deadline - 150:
            break

        t0 = time.monotonic()
        b = _solve_lp(a, M, row_idx, col_idx, time_limit=25)
        lp_time = time.monotonic() - t0

        if b is None:
            print(f"  Iter {iteration}: LP failed ({lp_time:.1f}s)")
            # Try with slightly larger M
            M *= 1.01
            continue

        b = b / b.sum()

        # Line search
        best_t = 0.0
        best_sc_new = _score(a)
        for t in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
            a_new = (1 - t) * a + t * b
            sc_new = _score(a_new)
            if sc_new < best_sc_new:
                best_sc_new = sc_new
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * b
            M = float(_full_convolve(a, a).max())
            sc_new = _score(a)
            do_update(a, sc_new, f"lp_{iteration}_t{best_t:.2f}")
            print(f"  Iter {iteration}: t={best_t:.2f}, score={sc_new:.10f}, "
                  f"M={M:.8f}, lp_time={lp_time:.1f}s")
            stall_count = 0
        else:
            stall_count += 1
            print(f"  Iter {iteration}: no improvement ({lp_time:.1f}s)")

            if stall_count >= 3:
                # Perturb to escape
                a = _gaussian_perturb(a, rng, sigma=0.01)
                M = float(_full_convolve(a, a).max())
                stall_count = 0
                print(f"    Perturbed, new M={M:.8f}")

    # --- Phase 2: Nesterov accelerated subgradient ---
    print(f"\nPhase 2: Nesterov subgradient polishing")
    if time.monotonic() < deadline - 50:
        a_cur = best_seq.copy()
        v = np.zeros(n)
        lr = 0.05
        t_polyak = 0.5
        sg_best = _score(a_cur)

        for i in range(20000):
            if time.monotonic() > deadline - 50:
                break

            grad = _compute_subgrad(a_cur)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break

            # Nesterov step
            a_y = np.maximum(a_cur - lr * v, 1e-12)
            a_y = a_y / a_y.sum()
            grad_y = _compute_subgrad(a_y)
            a_new = np.maximum(a_cur - lr * grad_y, 1e-12)
            a_new = a_new / a_new.sum()
            v = grad_y

            # Adaptive learning rate
            sc_new = _score(a_new)
            if sc_new < sg_best:
                sg_best = sc_new
                do_update(a_new, sc_new, f"nest_{i}")
                lr = min(lr * 1.01, 0.1)
            else:
                lr = max(lr * 0.95, 0.001)

            a_cur = a_new

            if i % 500 == 0:
                print(f"    Nesterov iter {i}: score={sc_new:.10f}, lr={lr:.4f}")

    # --- Phase 3: Additional LP iterations from best ---
    print(f"\nPhase 3: Additional LP from best")
    if best_seq is not None and time.monotonic() < deadline - 30:
        a_cur = best_seq.copy()
        n_cur = len(a_cur)
        # Recompute indices if n changed
        if n_cur != n:
            row_idx, col_idx = _precompute_conv_indices(n_cur)

        M_cur = float(_full_convolve(a_cur, a_cur).max())
        for iteration in range(50):
            if time.monotonic() > deadline - 30:
                break
            b = _solve_lp(a_cur, M_cur, row_idx, col_idx, time_limit=20)
            if b is None:
                break
            b = b / b.sum()
            best_t = 0.0
            best_sc_new = _score(a_cur)
            for t in [1.0, 0.7, 0.4, 0.2]:
                a_new = (1 - t) * a_cur + t * b
                sc_new = _score(a_new)
                if sc_new < best_sc_new:
                    best_sc_new = sc_new
                    best_t = t
            if best_t > 0:
                a_cur = (1 - best_t) * a_cur + best_t * b
                M_cur = float(_full_convolve(a_cur, a_cur).max())
                do_update(a_cur, best_sc_new, f"lp2_{iteration}")
            else:
                break

    # --- Phase 4: Fourier perturb + quick LP ---
    print(f"\nPhase 4: Fourier perturbation exploration")
    if best_seq is not None and time.monotonic() < deadline - 20:
        for pi in range(3):
            if time.monotonic() > deadline - 20:
                break
            a_pert = _phase_perturb(best_seq, rng)
            sc_pert = _score(a_pert)
            if sc_pert < best_score + 0.01:
                do_update(a_pert, sc_pert, f"fourier_{pi}")
                # Quick LP on perturbed
                n_p = len(a_pert)
                if n_p == n:
                    M_p = float(_full_convolve(a_pert, a_pert).max())
                    b_p = _solve_lp(a_pert, M_p, row_idx, col_idx, time_limit=15)
                    if b_p is not None:
                        b_p = b_p / b_p.sum()
                        for t in [1.0, 0.5, 0.3]:
                            a_new = (1 - t) * a_pert + t * b_p
                            sc_new = _score(a_new)
                            if sc_new < _score(a_pert):
                                do_update(a_new, sc_new, f"fourier_lp_{pi}_t{t}")
                                break

    # --- Phase 5: Final subgradient sweep ---
    print(f"\nPhase 5: Final subgradient sweep")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin = best_seq.copy()
        n_fin = len(a_fin)
        for i in range(1000):
            if time.monotonic() > deadline - 5:
                break
            grad = _compute_subgrad(a_fin)
            gn = np.linalg.norm(grad)
            if gn < 1e-12:
                break
            lr_fin = 0.01 / (1 + i * 0.01)
            a_fin = np.maximum(a_fin - lr_fin * grad, 1e-12)
            a_fin = a_fin / a_fin.sum()
            if i % 100 == 0:
                sc_fin = _score(a_fin)
                do_update(a_fin, sc_fin, f"final_sg_{i}")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0, 1000.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
