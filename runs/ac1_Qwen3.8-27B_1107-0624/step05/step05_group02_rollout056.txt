Improved gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam with adaptive beta scheduling, multi-scale exploration, and structured restarts.

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)

def _make_uniform(n):
    return np.ones(n) / n


def _make_cosine(n):
    """U-shaped (cosine) profile."""
    i = np.arange(n)
    a = 1.0 + np.cos(2.0 * np.pi * i / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_triangular_inv(n):
    """Inverted triangle (U-shaped)."""
    i = np.arange(n)
    center = (n - 1) / 2.0
    a = 1.0 - (i - center) / (center + 1)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_power_law(n, p):
    """Power law profile: a_i ~ (1 - |2i/n - 1|)^p"""
    i = np.arange(n)
    x = np.abs(2.0 * i / n - 1.0)  # 0 at center, 1 at edges
    a = x ** p + 1e-10
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

Actually, let me think about it from a different angle. The convolution c = a * a (full) has DFT: C(f) = |A(f)|^2. So max(c) is the peak of the autocorrelation, which by the Wiener-Khinchin theorem is related to the flatness of the power spectrum |A(f)|^2.

If |A(f)|^2 is perfectly flat (= 1 for all f, since A(0)=sum(a)=1), then c would be a delta function and max(c) = c_0 = 1 (for a delta sequence). But that's not achievable with a >= 0.

The constraint a >= 0 in the time domain translates to a constraint on A(f) in the frequency domain. Specifically, a_i >= 0 for all i means that the inverse DFT of |A(f)|^2 (with appropriate phase) is non-negative.

Actually, I just realized something important. Let me re-examine the problem.

The score is: 2n * max(c) / (sum(a))^2

If I normalize a to have sum = 1, the score is 2n * max(c).

For n=8192, I need max(c) < 1.5027/(2*8192) ≈ 9.16e-5.

The average of c is 1/(2n-1) ≈ 6.1e-5. So I need the peak to be less than about 1.5x the average. That's a peak-to-average ratio (PAPR) of less than 1.5.

For a random sequence, the PAPR of the autocorrelation is typically O(log n) or O(sqrt(log n)) depending on the distribution. So we need to do significantly better than random.

For a Barker sequence (which has nearly zero off-peak autocorrelation), the PAPR is very low. But Barker sequences only exist for certain lengths and have specific structures (they're ±1 sequences, not non-negative).

For non-negative sequences, the constraint a >= 0 is more restrictive. The best we can do is have the autocorrelation be as flat as possible.

For the optimal solution, I'd expect a to be a smooth "bowl" shape (high at edges, low in middle) to minimize the center of the convolution while keeping the edges reasonable.

"""
Multi-phase gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Adam with adaptive beta scheduling, periodic kicks for local minimum escape,
multi-scale exploration, and structured restarts.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_optimize(a_init, deadline, lr=0.01, beta_start=0.05, beta_end=0.0001,
                   beta_half_life=10000, kick_interval=8000, kick_strength=0.08,
                   patience=4000, max_grad_norm=0.5, seed=0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    last_improve_step = 0
    kick_rng = np.random.default_rng(99999 + seed)

    while time.monotonic() < deadline:
        t_step += 1

        beta = beta_start * (beta_end / beta_start) ** (t_step / float(beta_half_life))
        beta = max(beta, beta_end)

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > max_grad_norm:
            update = update * (max_grad_norm / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t_step % 100 == 0:
            a_check = _softmax(z)
            sc = float(_full_convolve(a_check, a_check).max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                last_improve_step = t_step

        if patience > 0 and (t_step - last_improve_step) > patience:
            kick = kick_strength * kick_rng.standard_normal(n)
            z = z + kick
            z = z - z.mean()
            last_improve_step = t_step
            current_lr = min(0.04, current_lr * 1.2)

        if t_step % 8000 == 0:
            current_lr = max(0.001, current_lr * 0.92)

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _two_phase_adam(a_init, deadline, lr1=0.01, lr2=0.003,
                    beta1_start=0.05, beta1_end=0.002, beta2_start=0.002, beta2_end=0.00005,
                    split_frac=0.6, seed=0):
    """Run Adam in two phases: coarse then fine."""
    t_mid = time.monotonic() + split_frac * (deadline - time.monotonic())
    a1, _ = _adam_optimize(a_init, t_mid, lr=lr1, beta_start=beta1_start,
                           beta_end=beta1_end, beta_half_life=8000,
                           kick_interval=6000, kick_strength=0.1, patience=3000, seed=seed)
    remaining = deadline - time.monotonic()
    if remaining < 5:
        return a1, _score_norm(a1)
    a2, sc2 = _adam_optimize(a1, deadline, lr=lr2, beta_start=beta2_start,
                             beta_end=beta2_end, beta_half_life=10000,
                             kick_interval=5000, kick_strength=0.05, patience=5000, seed=seed + 1)
    return a2, _score_norm(a2)


def _make_uniform(n):
    return np.ones(n) / n


def _make_cosine_bowl(n):
    i = np.arange(n)
    a = 1.0 + np.cos(2.0 * np.pi * i / n)
    a = np.maximum(a, 1e-10)
    a = a + 0.3 * np.ones(n)
    return a / a.sum()


def _make_power_law(n, p):
    i = np.arange(n)
    x = np.abs(2.0 * i / n - 1.0)
    a = x ** p + 1e-10
    return a / a.sum()


def _make_gaussian_bowl(n, sigma_frac=0.3):
    i = np.arange(n)
    x = np.abs(2.0 * i / n - 1.0)
    a = np.exp(-(x ** 2) / (2.0 * sigma_frac ** 2))
    a = 1.0 - 0.8 * a
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
