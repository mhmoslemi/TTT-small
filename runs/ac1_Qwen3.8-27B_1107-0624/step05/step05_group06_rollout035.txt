Hybrid optimizer combining L-BFGS (log-sum-exp smoothing), LP-based
direction finding, and projected subgradient for minimizing
2n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, eye
from scipy.fft import rfft, irfft

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    """One L-BFGS step with given beta."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    fun = lambda z: _smoobj(z, beta, n)
    jac = lambda z: _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-15,
                                  'gtol': 1e-12, 'maxcor': 30, 'maxls': 50,
                                  'disp': False})
        return _softmax(result.x)

"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2 using:
1. L-BFGS with log-sum-exp smoothing and beta-continuation
2. LP-based direction finding (novel: solves max sum(b) s.t. conv(f,b)<=M, b>=0)
3. Projected subgradient with momentum and adaptive active set
4. Coordinate descent for fine-tuning

import numpy as np

from scipy.fft import rfft, irfft

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_optimize(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    fun = lambda z: _smoobj(z, beta, n)
    jac = lambda z: _smograd(z, beta, n)
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': 1e-15,
                                  'gtol': 1e-12, 'maxcor': 40, 'maxls': 80,
                                  'disp': False})
        return _softmax(result.x)
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a_init

def _build_lp_matrix(f, n):
    """Build sparse constraint matrix A where A[k,m] = f[k-m] if valid."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        m_lo = max(0, k - (n - 1))
        m_hi = min(n - 1, k)
        if m_hi >= m_lo:
            ks = np.full(m_hi - m_lo + 1, k)
            ms = np.arange(m_lo, m_hi + 1)
            vals = f[k - ms]
            rows.extend(ks.tolist())
            cols.extend(ms.tolist())
            data.extend(vals.tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _solve_lp(f, M, n, A_cache=None):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    if A_cache is not None:
        A = A_cache
    else:
        A = _build_lp_matrix(f, n)
    b_vec = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds,
                        method='highs', options={'presolve': True,
                                                  'time_limit': 60})
        if result.success:
            return result.x
    except Exception as e:
        print(f"    LP error: {e}")
    return None

            active[int(np.argmax(c))] = True

        a_new = np.maximum(a_new, 0)

        if s_new < 0.01:
