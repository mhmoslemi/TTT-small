    n = len(a)
    c = _full_convolve(a, a)

def _peak_directed_perturb(a, rng, strength=0.05):
    """Perturb the sequence to specifically target reducing the current peak."""
    n = len(a)
    c = _full_convolve(a, a)
    peak_idx = int(np.argmax(c))

    # Compute contribution of each a[i] to the peak
    contributions = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contributions[i] = a[j]
    # Total contribution = sum(a[i] * a[peak_idx-i]) = c[peak_idx]
    total = c[peak_idx]

    # In log-space, decrease high contributors and increase low ones
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    # Perturb: reduce peak contributors, boost others
    contrib_norm = contributions / max(total, 1e-15)
    noise = rng.standard_normal(n)
    z = z - strength * (contrib_norm - 1.0 / n) + strength * 0.5 * noise
    z = z - z.mean()
    return _softmax(z)

    c_cur = _full_convolve(best_seq, best_seq)
    max_cc = float(c_cur.max())

    for attempt in range(8):
        if time.monotonic() > phase4_end - 20:
            break

        # Alternate between peak-directed and random perturbation
        if attempt % 2 == 0:
            p_start = _peak_directed_perturb(best_seq, rng, strength=0.04 + 0.01 * attempt)
        else:
            p_start = _random_restart_perturb(best_seq, rng, sigma=0.02 + 0.005 * attempt)

        # Quick Adam refinement
        c_p = _full_convolve(p_start, p_start)
        beta_p = 0.005 * float(c_p.max())
        p_refined = _short_adam(p_start, iters=800, beta=beta_p, lr=0.01)
        sc_p = _score(p_refined)
        print(f"  Restart[{attempt}]: score={sc_p:.10f}, t={time.monotonic()-t_start:.0f}s")

        if np.isfinite(sc_p) and sc_p < best_score:
            best_score = sc_p
            best_seq = p_refined.copy()
            print(f"    NEW BEST: {best_score:.10f}")

    # === Phase 5: Final ultra-fine Adam (1650 - deadline) ===
    phase5_end = deadline - 5
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Final polish, target t={phase5_end - t_start:.0f}s")
        c_f = _full_convolve(best_seq, best_seq)
        max_cf = float(c_f.max())
        a_final, sc_final = _adam_loop(
            best_seq, phase5_end,
            beta_start=0.001 * max_cf,
            beta_end=0.000005 * max_cf,
            lr_init=0.004,
            lr_end=0.0001,
            max_iter=300000,
            momentum_schedule='exponential'
        )
        full_sc = _score(a_final)
        print(f"Phase 5 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_final.copy()

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
