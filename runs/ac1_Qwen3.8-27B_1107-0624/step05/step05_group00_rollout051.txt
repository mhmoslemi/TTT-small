"""
Multi-phase optimizer using log-space Adam with top-k smoothed max,
structural perturbations, and multi-length exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_conv(a):
    return _full_convolve(a, a)

def _compute_grad_a(a, w):
    """
    Compute gradient of sum_k w_k * (a*a)[k] w.r.t. a.
    w has length 2n-1.
    Returns array of length n.
    """
    n = len(a)
    w_rev = w[::-1]
    fc = _full_convolve(a, w_rev)  # length 3n-2
    g = fc[n-1:2*n-1][::-1]  # length n
    return 2.0 * g

def _compute_grad_z(a, grad_a):
    """Convert gradient from a-space to z-space (log-space)."""
    return a * (grad_a - np.dot(a, grad_a))

def _adam_logspace(a_init, deadline, betas, lr_init, lr_decay, max_iter, top_k_frac=1.0):
    """
    Adam optimizer in log-space with softmax-smoothed max.
    
    Parameters:
    - a_init: initial point on simplex (sum=1)
    - deadline: time by which to finish
    - betas: list of smoothing parameters (decreasing)
    - lr_init: initial learning rate
    - lr_decay: decay factor applied every 5000 steps
    - max_iter: maximum total iterations
    - top_k_frac: fraction of top values to weight (1.0 = all)
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    step_count = 0
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        remain = max_iter - t
        if remain <= 0:
            break
        
        phase_time = min(60.0, deadline - time.monotonic())
        phase_deadline = time.monotonic() + phase_time
        it_in_phase = 0
        
        while it_in_phase < remain:
            if time.monotonic() > phase_deadline or time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            step_count += 1
            
            a = _softmax(z)
            c = _compute_conv(a)
            cm = float(c.max())
            
            if top_k_frac < 1.0:
                # Use top-k weighted smoothing
                k = max(1, int(top_k_frac * len(c)))
                sorted_idx = np.argsort(c)[::-1][:k]
                w = np.zeros_like(c)
                c_sub = c[sorted_idx]
                c_sub_norm = c_sub - c_sub.max()
                e = np.exp(c_sub_norm / beta)
                s = e.sum()
                if s < 1e-300:
                    w[sorted_idx[0]] = 1.0
                else:
                    w[sorted_idx] = e / s
            else:
                c_shifted = (c - cm) / beta
                e = np.exp(c_shifted)
                s = float(e.sum())
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = e / s
            
            grad_a = _compute_grad_a(a, w)
            if not np.all(np.isfinite(grad_a)):
                break
            
            grad_z = _compute_grad_z(a, grad_a)
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if step_count % 5000 == 0:
                current_lr = max(0.0005, current_lr * lr_decay)
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _fista_simplex(a_init, deadline, betas, max_iter):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    
    global_step = 0
    for beta in betas:
        if time.monotonic() > deadline:
            break
        phase_deadline = min(deadline, time.monotonic() + 50.0)
        L = 4.0 * n / beta  # Lipschitz constant estimate
        step = 1.0 / L
        it = 0
        while it < max_iter and time.monotonic() < phase_deadline:
            it += 1
            global_step += 1
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c = _compute_conv(y)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            grad = _compute_grad_a(y, w)
            a_next = _proj_simplex(y - step * grad)
            
            if global_step % 100 == 0:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_frequency(a, rng):
    """Add a low-frequency sinusoidal perturbation."""
    n = len(a)
    freq = rng.integers(1, max(2, n // 256))
    phase = rng.uniform(0, 2 * np.pi)
    amplitude = 0.05 * rng.random()
    t = np.arange(n)
    pert = 1 + amplitude * np.sin(2 * np.pi * freq * t / n + phase)
    a_new = a * pert
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()
