def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

            # sum_k c_k^(p-1) * a_{i-k} = (c^(p-1) * flip(a))[i] (same by symmetry of convolution)
            # So total = 2 * (c^(p-1) * flip(a))[i]... 
            # Wait, let me be more careful.
            # c^(p-1) * a (convolution) gives sum_j c^(p-1)_j * a_{i-j}
            # We want sum_k c^(p-1)_k * a_{k-i} = sum_j c^(p-1)_{j+i} * a_j (sub j = k-i)
            # This is the cross-correlation, not convolution.
            # cross_corr(i) = sum_j c_j * a_{j-i} = (flip(c) * a)(i) = (c * flip(a))(2n-2-i)... 
            # This is getting confusing. Let me use the same formula as the previous code.
            # The previous code computes: h = w_conv[n-1:2*n-1] where w_conv = convolve(w, a_rev)
            # This gives h[i] = sum_j w[j] * a_rev[i-j+n-1] = sum_j w[j] * a[n-1-i+j]
            # Hmm, let me just verify with the previous code's formula.
            
            # From previous code: grad = 4*n * a * (h - W) where h = convolve(w, a[::-1])[n-1:2n-1]
            # This is the gradient of the log-sum-exp of c_k w.r.t. z (log-params)
            # For power-sum, it's different.
            
            # Let me just compute it directly using the formula:
            # d/da_i [sum_k c_k^p] = p * sum_k c_k^(p-1) * (a_{k-i} + a_{i-k}) [with boundary]
            # = p * 2 * sum_k c_k^(p-1) * a_{k-i} [with boundary] (by symmetry, since c is symmetric around n-1)
            # Wait, c_k is NOT symmetric in general (it's the auto-convolution, which IS symmetric: c_k = c_{2n-2-k})
            # So c_k^(p-1) is also symmetric.
            # sum_k c_k^(p-1) * a_{k-i} (with 0<=k-i<=n-1) 
            # Let j = k-i, so k = j+i: sum_j c_{j+i}^(p-1) * a_j (with 0<=j<=n-1, 0<=j+i<=2n-2)
            # This is the cross-correlation of c^(p-1) and a, evaluated at i.
            
            # Using FFT: cross_corr_a_c(i) = sum_j a_j * c_{j+i} = ifft(fft(a) * conj(fft(c)))[i]... 
            # Actually, cross-correlation can be computed as: ifft(conj(fft(a)) * fft(c))
            # Or: convolve(flip(a), c)[n-1:] gives the cross-correlation
            
            # Let me just use the convolution approach:
            # h[i] = sum_j a_j * c_{j+i} for j=0..n-1, j+i in 0..2n-2
            # This is (flip(a) * c)[n-1 + i] ... let me verify
            # (flip(a) * c)[m] = sum_j flip(a)_j * c_{m-j} = sum_j a_{n-1-j} * c_{m-j}
            # Let i' = n-1-j, so j = n-1-i': sum_{i'} a_{i'} * c_{m-n+1+i'}
            # For m = n-1+i: sum_{i'} a_{i'} * c_{i+i'}
            # Yes! So h = (flip(a) * c)[n-1 : 2n-1]
            
            flip_a = a_in[::-1]
            fc = _full_convolve(flip_a, c_pm1)
            h = fc[n-1:2*n-1]
            
            # Total gradient w.r.t. a: p * 2 * h (factor of 2 because of the two terms)
            # Wait, by symmetry of c: c_k = c_{2n-2-k}
            # sum_k c_k^(p-1) * a_{k-i} + sum_k c_k^(p-1) * a_{i-k}
            # Second sum: let j = i-k, so k = i-j: sum_j c_{i-j}^(p-1) * a_j
            # Since c is symmetric: c_{i-j} = c_{2n-2-i+j}
            # So second sum = sum_j c_{2n-2-i+j}^(p-1) * a_j
            # This is NOT the same as the first sum in general.
            # First sum: sum_j c_{i+j}^(p-1) * a_j
            # Second sum: sum_j c_{2n-2-i+j}^(p-1) * a_j
            # These are different unless i = n-1.
            
            # So the total gradient w.r.t. a_i is:
            # p * (sum_j c_{i+j}^(p-1) * a_j + sum_j c_{2n-2-i+j}^(p-1) * a_j)
            # = p * (h1[i] + h2[i])
            # where h1[i] = sum_j a_j * c_{i+j} and h2[i] = sum_j a_j * c_{2n-2-i+j}
            
            # h1 = (flip(a) * c)[n-1:] (as computed above)
            # h2[i] = sum_j a_j * c_{2n-2-i+j}
            # Let k = 2n-2-i+j, so j = k-2n+2+i: h2[i] = sum_k a_{k-2n+2+i} * c_k
            # This is (a * flip(c))[2n-2-i] = (a * c)[2n-2-i] (since c is symmetric, flip(c) = c)
            # Wait, c has length 2n-1, so flip(c) has the same values reversed.
            # c_k = c_{2n-2-k}, so flip(c)_k = c_{2n-2-k} = c_k. So c IS symmetric!
            # Therefore flip(c) = c, and (a * flip(c)) = (a * c).
            # h2[i] = (a * c)_{2n-2-i}
            
            # So total gradient w.r.t. a_i = p * (h1[i] + (a*c)_{2n-2-i})
            # where h1 = (flip(a) * c_pm1)[n-1:2n-1]
            # and (a*c_pm1) has length 2n-1
            
            ac = _full_convolve(a_in, c_pm1)
            h2 = ac[2*n-2 : n-1 : -1]  # (a*c)[2n-2-i] for i=0..n-1
            # Actually, ac has indices 0..2n-2. We want ac[2n-2-i] for i=0..n-1
            # That's ac[2n-2], ac[2n-3], ..., ac[n-1]
            h2 = ac[n-1:2*n-1][::-1]
            
            grad_a = p * (h1 + h2)  # gradient w.r.t. a
            
            # Now convert to gradient w.r.t. z (where a = softmax(z)):
            # d/dz_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
            Wa = float(np.dot(grad_a, a_in))
            grad_z = a_in * (grad_a - Wa)
            
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z
        
        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad,
                             options={'maxiter': 3000, 'ftol': 1e-18,
                                      'gtol': 1e-12, 'maxls': 50, 'maxfun': 10000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                s_new = _score_unnorm(a_new)
                if np.isfinite(s_new) and s_new < best_s:
                    best_s = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_s
