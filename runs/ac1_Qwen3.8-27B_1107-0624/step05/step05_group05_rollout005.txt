def _quick_adam(a_init, deadline, beta, lr, max_iter=100000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        beta_cur = beta * (1.0 + 0.1 * (t / max_iter))
        z, m_buf, v_buf, _ = _adam_iter(a, z, m_buf, v_buf, t, beta_cur, lr, b1, b2, eps)
        if t % 2000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    # Build LP indices
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    print(f"LP indices built: {len(rows)} nonzeros, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: LP iterations (0 - 180s)
    phase1_end = min(deadline, t_start + 180.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0

    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a, M = _lp_step(a, rows, cols)
        sc = _score_norm(a)
        lp_count += 1
        dt_lp = time.monotonic() - t0

        if sc < prev_sc - 1e-12:
            stall = 0
        else:
            stall += 1

        if lp_count % 3 == 0 or stall > 10:
            full_sc = _score(a)
            print(f"  LP[{lp_count}] score={full_sc:.10f} M={M:.8f} dt={dt_lp:.1f}s stall={stall} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()

        if stall > 30:
            print(f"  LP stalled at iter {lp_count}")
            break
        prev_sc = sc

    full_sc = _score(a)
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Adam with beta annealing (180 - 700s)
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        print(f"Phase 2: Adam optimization, max_c={max_cb:.6f}")

        # Multiple Adam runs with different beta schedules
        schedules = [
            (0.5e-3 * max_cb, 1e-5 * max_cb, 0.012, 2000000),
            (0.2e-3 * max_cb, 5e-6 * max_cb, 0.008, 3000000),
            (0.1e-3 * max_cb, 1e-6 * max_cb, 0.006, 3000000),
        ]
        a_current = best_seq.copy()
        for si, (b_start, b_end, lr, max_it) in enumerate(schedules):
            if time.monotonic() > phase2_end - 60:
                break
            sub_deadline = min(phase2_end, time.monotonic() + (phase2_end - time.monotonic()) * 0.4)
            print(f"  Adam schedule {si}: beta [{b_start:.2e} -> {b_end:.2e}], lr={lr}")
            a_polished, sc_pol = _adam_optimize(
                a_current, sub_deadline,
                beta_start=b_start, beta_end=b_end,
                lr=lr, max_iter=max_it
            )
            full_sc = _score(a_polished)
            print(f"    Result: score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a_polished.copy()
                a_current = a_polished.copy()

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: LP + Adam hybrid (700 - 1050s)
    phase3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase3_end:
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp3_count = 0
        hybrid_cycles = 0

        while time.monotonic() < phase3_end and hybrid_cycles < 20:
            # Do a few LP steps
            for _ in range(3):
                if time.monotonic() > phase3_end:
                    break
                a, M = _lp_step(a, rows, cols)
                sc = _score_norm(a)
                lp3_count += 1
                if sc < prev_sc - 1e-12:
                    stall = 0
                else:
                    stall += 1
                prev_sc = sc
            # Then do Adam refinement
            c_cur = _full_convolve(a, a)
            max_cc = float(c_cur.max())
            sub_dl = min(phase3_end - 5, time.monotonic() + 30)
            a, sc_adam = _quick_adam(a, sub_dl, beta=1e-4 * max_cc, lr=0.01, max_iter=50000)
            full_sc = _score(a)
            hybrid_cycles += 1
            print(f"  Hybrid[{hybrid_cycles}] LP={lp3_count} score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
            if stall > 15:
                break

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Multi-start perturbations (1050 - 1550s)
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end:
        n_restarts = 12
        print(f"Phase 4: {n_restarts} restarts")
        for attempt in range(n_restarts):
            if time.monotonic() > phase4_end - 30:
                break
            sigma = 0.02 + 0.01 * (attempt % 5)
            p_start = _perturb(best_seq, sigma, rng)

            # Quick LP
            prev_sc = _score_norm(p_start)
            ps_stall = 0
            for it in range(8):
                if time.monotonic() > phase4_end - 10:
                    break
                p_start, _ = _lp_step(p_start, rows, cols)
                sc = _score_norm(p_start)
                if sc < prev_sc - 1e-12:
                    ps_stall = 0
                else:
                    ps_stall += 1
                if ps_stall > 5:
                    break
                prev_sc = sc

            # Adam refinement
            c_ps = _full_convolve(p_start, p_start)
            max_cps = float(c_ps.max())
            sub_dl = min(phase4_end - 5, time.monotonic() + 25)
            p_start, _ = _quick_adam(p_start, sub_dl, beta=5e-5 * max_cps, lr=0.01, max_iter=40000)

            full_sc = _score(p_start)
            improved = "YES" if full_sc < best_score else "no"
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc:.10f} ({improved}) t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_start.copy()

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 5: Final ultra-fine polish (1550 - deadline)
    if time.monotonic() < deadline - 20:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        deadline5 = deadline - 10
        print(f"Phase 5: Final polish, max_c={max_cf:.8f}")

        # Very aggressive beta (close to true max)
        a_polished, sc_pol = _adam_optimize(
            best_seq, deadline5,
            beta_start=2e-5 * max_cf, beta_end=2e-7 * max_cf,
            lr=0.003, max_iter=5000000,
            check_interval=200, print_interval=10000
        )
        full_sc = _score(a_polished)
        print(f"Phase 5 result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_polished.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
