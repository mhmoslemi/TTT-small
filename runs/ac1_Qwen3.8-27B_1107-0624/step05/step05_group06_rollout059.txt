def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _coordinate_descent(a, deadline, max_rounds=20, top_k=50, verbose=False):
    """Coordinate descent: optimize each coordinate given all others fixed."""
    n = len(a)

        # Process coordinates in random order
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:

            # Optimize a[j] given all other coordinates fixed
            # f(a) = 2n * max_k conv_k / s^2
            # conv_k = sum_i a[i]*a[k-i]

        print(f"No parent, using uniform n={n_main}, score={_score(a0):.10f}")
        update_best(a0, _score(a0), "init_uniform")

    # --- Phase 1: Main L-BFGS with extended beta schedule ---
    print(f"\nPhase 1: L-BFGS beta-continuation (t={time.monotonic()-t_start:.1f}s)")
    a_main = _beta_schedule(a0, deadline, min_time_remaining=400.0)
    sc_main = _score(a_main)
    update_best(a_main, sc_main, "lbfgs_main")

    # --- Phase 2: Projected subgradient with backtracking ---
    print(f"\nPhase 2: Subgradient backtracking (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 250.0, time.monotonic() + 180.0)
        a_sg, sc_sg = _subgrad_backtrack(best_seq, sg_deadline, max_iter=8000, verbose=True)
        update_best(a_sg, sc_sg, "subgrad_bt")

    # --- Phase 3: Fine L-BFGS with very small beta ---
    print(f"\nPhase 3: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        c_best = _fft_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        n = len(best_seq)
        fine_betas = max_cb * np.array([0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 150:
                break
            remaining = deadline - 150 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(4000, max(100, int(remaining / 0.03)))
            a_new = _lbfgs_run(best_seq, beta, n, max_iter=max_iter)
            sc = _score(a_new)
            update_best(a_new, sc, f"fine_{i}")

    # --- Phase 4: Coordinate descent ---
    print(f"\nPhase 4: Coordinate descent (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        cd_deadline = min(deadline - 80.0, time.monotonic() + 60.0)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_deadline, max_rounds=10)
        update_best(a_cd, sc_cd, "coord_descent")

    # --- Phase 5: Multi-restart with diverse strategies ---
    print(f"\nPhase 5: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        remaining = deadline - 50 - time.monotonic()
        n_restarts = min(6, max(2, int(remaining / 100)))

        # Different perturbation strategies
        strategies = [
            ("gauss_small", lambda: _perturb_gaussian(best_seq, 0.02, rng)),
            ("gauss_med", lambda: _perturb_gaussian(best_seq, 0.05, rng)),
            ("harmonic", lambda: _perturb_harmonic(best_seq, rng, n_harmonics=3)),
            ("block", lambda: _perturb_block(best_seq, rng)),
            ("gauss_large", lambda: _perturb_gaussian(best_seq, 0.1, rng)),
            ("harmonic2", lambda: _perturb_harmonic(best_seq, rng, n_harmonics=5)),
        ]

        for i in range(n_restarts):
            if time.monotonic() > deadline - 45:
                break
            if i >= len(strategies):
                break
            name, fn = strategies[i]
            p_start = fn()
            sc_start = _score(p_start)

            # Quick L-BFGS from perturbed point
            if time.monotonic() < deadline - 40:
                a_r = _beta_schedule(p_start, deadline, min_time_remaining=40.0)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{name}")

    # --- Phase 6: Boundary exploration ---
    print(f"\nPhase 6: Boundary exploration (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        bd_deadline = min(deadline - 20.0, time.monotonic() + 30.0)
        a_bd, sc_bd = _boundary_explore(best_seq, bd_deadline, n_candidates=5)
        update_best(a_bd, sc_bd, "boundary")

    # --- Phase 7: Final combined polish ---
    print(f"\nPhase 7: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Quick subgradient
        if time.monotonic() < deadline - 10:
            a_f1, sc_f1 = _subgrad_backtrack(best_seq, deadline - 8, max_iter=2000, verbose=False)
            update_best(a_f1, sc_f1, "final_sg")

        # Very fine L-BFGS
        if time.monotonic() < deadline - 3:
            n = len(best_seq)
            c_b = _fft_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            if mb > 0:
                a_f2 = _lbfgs_run(best_seq, mb * 0.000005, n, max_iter=1000)
                update_best(a_f2, _score(a_f2), "final_lbfgs")

        # One more coordinate descent
        if time.monotonic() < deadline - 2:
            a_f3, sc_f3 = _coordinate_descent(best_seq, deadline - 2, max_rounds=5)
            update_best(a_f3, sc_f3, "final_cd")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
