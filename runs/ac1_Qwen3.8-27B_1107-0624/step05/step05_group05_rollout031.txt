Uses KL-divergence mirror descent on the simplex, adaptive temperature annealing,

import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_self(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _cross_corr(w, a):
    """Compute g[j] = sum_k w[k]*a[k-j] for j=0..n-1, where w has length 2n-1."""
    n = len(a)
    a_rev = a[::-1]
    out_n = len(w) + n - 1
    N = _next_pow2(out_n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(fw * fa, N)[:out_n]
    return conv[n - 1:2 * n - 1]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_self(a)
    return float(2.0 * n * c.max() / (s * s))

    """Score assuming sum(a)=1, ignoring the s^2 factor."""
    n = len(a)
    c = _convolve_self(a)
    return float(2.0 * n * c.max())

        else:
            w = e_c / s_e
        
        h = c_max + T * np.log(max(s_e, 1e-300))
        
        # Primal gradient: g_a[j] = 4*n*(g0[j] - h)
        g0 = _cross_corr(w, a)
        g_a = 4.0 * n * (g0 - h)
        
        # Convert to log-space gradient:
        # dx[j] = sum_k (da[k]/dx[j]) * g_a[k]
        # a[k] = exp(x[k]) / sum exp(x), so da[k]/dx[j] = a[k]*(delta[kj] - a[j])
        # dx[j] = sum_k a[k]*(delta[kj]-a[j]) * g_a[k] = a[j]*g_a[j] - a[j]*sum_k a[k]*g_a[k]
        # = a[j] * (g_a[j] - sum_k a[k]*g_a[k])
        W = float(np.dot(a, g_a))
        g_x = a * (g_a - W)
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * g_x
        v_buf = b2 * v_buf + (1.0 - b2) * (g_x * g_x)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        x = x - update
        x = np.clip(x, -30.0, 30.0)
        x = x - x.mean()
        
        # Check score
        if t % 500 == 0:
            x_shift2 = x - np.max(x)
            a_check = np.exp(x_shift2)
            a_check /= a_check.sum()
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    
    # Final
    x_shift = x - np.max(x)
    a_final = np.exp(x_shift)
    a_final /= a_final.sum()
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _upsample(a, factor=2):
    """Upsample by inserting zeros."""
    n = len(a)
    new_a = np.zeros(n * factor)
    new_a[::factor] = a
    return new_a / new_a.sum()


def _interp_upsample(a, n_new):
    """Interpolation-based upsample."""
    n0 = len(a)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    x = np.log(np.maximum(a, 1e-300))
    x = x + sigma * rng.standard_normal(len(a))
    x = x - x.mean()
    x_shift = x - np.max(x)
    e = np.exp(x_shift)
    return e / e.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        print(f"Parent loaded: n={n_parent}")
        a_parent = parent.copy()
        sc_parent = _score(a_parent)
        best_seq = a_parent.copy()
        best_score = sc_parent
        print(f"Parent score: {sc_parent:.10f}")
    else:
        n_parent = 2048
        a_parent = np.ones(n_parent) / n_parent
        sc_parent = _score(a_parent)
        best_seq = a_parent.copy()
        best_score = sc_parent
        print(f"No parent, uniform n={n_parent}, score={sc_parent:.10f}")
    
    # ============ Phase 1: Mirror Descent on parent (n=2048) ============
    print(f"\n=== Phase 1: Mirror Descent n={n_parent} ===")
    phase1_end = min(deadline, t_start + 350.0)
    if time.monotonic() < phase1_end:
        a = a_parent.copy()
        
        # Compute initial max for T calibration
        c_init = _convolve_self(a)
        max_c_init = float(c_init.max())
        
        # Multiple passes with decreasing T
        T_starts = [0.1 * max_c_init, 0.05 * max_c_init, 0.02 * max_c_init,
                    0.01 * max_c_init, 0.005 * max_c_init, 0.002 * max_c_init,
                    0.001 * max_c_init, 0.0005 * max_c_init, 0.0002 * max_c_init]
        
        for T_start in T_starts:
            if time.monotonic() > phase1_end - 5:
                break
            T_end = max(T_start * 0.01, 1e-8)
            
            # Estimate max iterations in remaining time
            remaining = phase1_end - time.monotonic()
            max_iter = int(remaining * 800)  # ~800 iters/sec estimate for n=2048
            max_iter = max(max_iter, 100)
            
            eta0 = 0.5 / n_parent
            eta_min = 0.05 / n_parent
            
            a, sc = _mirror_descent(a, T_start, T_end, max_iter, eta0, eta_min,
                                    phase1_end, verbose=True, verbose_every=1000)
            sc_full = _score(a)
            if np.isfinite(sc_full) and sc_full < best_score:
                best_score = sc_full
                best_seq = a.copy()
                print(f"  New best: {best_score:.10f}")
    
    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # ============ Phase 2: Adam in log-space on best (n=2048) ============
    print(f"\n=== Phase 2: Adam log-space n={n_parent} ===")
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        c_best = _convolve_self(best_seq)
        max_cb = float(c_best.max())
        
        # Multiple Adam runs with different settings
        configs = [
            (0.01 * max_cb, 0.0001 * max_cb, 500000, 0.05),
            (0.005 * max_cb, 0.00005 * max_cb, 500000, 0.03),
            (0.002 * max_cb, 0.00002 * max_cb, 500000, 0.02),
            (0.001 * max_cb, 0.00001 * max_cb, 300000, 0.01),

        for T0_a, Tend_a, max_it, lr_a in configs:
            if time.monotonic() > phase2_end - 30:
                break
            a_adam, sc_adam = _adam_logspace(best_seq, phase2_end - 10, T0_a, Tend_a, max_it, lr_a)
            sc_full = _score(a_adam)
            if np.isfinite(sc_full) and sc_full < best_score:
                best_score = sc_full
                best_seq = a_adam.copy()
                print(f"  Adam new best: {best_score:.10f} (T0={T0_a:.2e}, lr={lr_a})")
            else:
                print(f"  Adam: {sc_full:.10f} (not better than {best_score:.10f})")
    
    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # ============ Phase 3: Try larger n (4096) ============
    print(f"\n=== Phase 3: Try n=4096 ===")
    phase3_end = min(deadline, t_start + 1150.0)
    if time.monotonic() < phase3_end:

        # Initialize from upscaled parent
        a_big = _interp_upsample(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Upsampled n=4096, score={sc_big:.10f}")
        
        if np.isfinite(sc_big) and sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        a_big2 = _upsample(best_seq, 2)
        sc_big2 = _score(a_big2)
        print(f"  Zero-insert n=4096, score={sc_big2:.10f}")

        if sc_big2 < sc_big:
            a_big = a_big2
        
        c_big = _convolve_self(a_big)
        max_cb = float(c_big.max())
        
        # Mirror descent on n=4096
        T_starts_big = [0.05 * max_cb, 0.02 * max_cb, 0.01 * max_cb,
                        0.005 * max_cb, 0.002 * max_cb, 0.001 * max_cb]
        
        for T_start in T_starts_big:

            eta_min = 0.03 / n_big

                                        phase3_end, verbose=True, verbose_every=500)
            sc_full = _score(a_big)
            if np.isfinite(sc_full) and sc_full < best_score:
                best_score = sc_full
                best_seq = a_big.copy()
                print(f"  n=4096 new best: {best_score:.10f}")
        
        # Adam on n=4096
        if time.monotonic() < phase3_end - 30:

            max_cb2 = float(c_big2.max())
            a_adam_big, sc_a = _adam_logspace(

    # ============ Phase 4: Restart and refine ============
    print(f"\n=== Phase 4: Restarts ===")

    if time.monotonic() < phase4_end:

        c_cur = _convolve_self(best_seq)
        max_cc = float(c_cur.max())
        
        for attempt in range(5):

                break
            sigma = 0.02 + 0.01 * attempt
            a_pert = _perturb_log(best_seq, sigma, rng)

            if sc_pert < best_score:

                best_seq = a_pert.copy()
            
            # Quick mirror descent from perturbed point
            T_start = 0.005 * max_cc
            T_end = 0.0001 * max_cc
            max_iter = 3000
            eta0 = 0.3 / n_cur
            eta_min = 0.02 / n_cur
            a_pert, _ = _mirror_descent(a_pert, T_start, T_end, max_iter, eta0, eta_min,
                                        phase4_end - 15, verbose=False)
            sc_pert2 = _score(a_pert)
            if np.isfinite(sc_pert2) and sc_pert2 < best_score:

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # ============ Phase 5: Final ultra-fine polish ============
    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 20:
        n_final = len(best_seq)
        c_fin = _convolve_self(best_seq)
        max_cf = float(c_fin.max())

        a_fin, sc_fin = _adam_logspace(

        if time.monotonic() < deadline - 10:

            sc_fin2 = _score(a_fin2)

    if best_seq is not None:

        try:

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
