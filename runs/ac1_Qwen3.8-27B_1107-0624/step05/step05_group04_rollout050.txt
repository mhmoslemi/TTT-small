"""
Multi-scale Adam optimization with peak surgery and spectral shaping for minimizing
2*n*max(conv(a,a))/sum(a)^2. Key innovations:
1. Direct analytical gradient in log-space (fast, no L-BFGS dependency)
2. Aggressive peak surgery: identify top convolution peaks and reduce top contributors
3. Spectral flatness as secondary regularization
4. Adaptive multi-scale with time allocation based on improvement
5. Try both n=4096 and n=8192

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _grad_max_conv_z(z, beta):
    """Compute gradient of smooth max convolution w.r.t. z in log-space."""
    n = len(z)
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n), cm, a, c
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # h[j] = sum_k w[k] * a[k-j] for valid k-j
    # = correlate(w, a, 'valid')[j]
    h = np.correlate(w, a, mode='valid')
    dot_wc = float(np.dot(w, c))
    grad_z = 2.0 * a * (h - dot_wc)
    return grad_z, cm, a, c


def _adaptive_beta(c, cm):
    """Adaptively choose beta based on the spread of convolution values."""
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(30, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.02 * cm:
        return max(1e-12, 0.02 * cm)
    elif gap > 0.3 * cm:
        return max(1e-12, 0.05 * cm)
    else:
        return max(1e-12, 0.5 * gap)


def _adam_optimize(z_init, deadline, rng, lr=0.05, max_iter=100000, decay_every=5000, decay_factor=0.85):
    """Run Adam optimization in z-space."""
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    best_z = z.copy()
    
    a = _softmax(z)
    c = _fft_convolve(a, a)
    best_score = float(2.0 * n * c.max())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    current_lr = lr
    
    c_max = float(c.max())
    beta = _adaptive_beta(c, c_max)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        grad_z, cm, a, c = _grad_max_conv_z(z, beta)
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adaptive beta
        if t % 200 == 0:
            new_beta = _adaptive_beta(c, cm)
            if new_beta < beta * 0.3:
                beta = new_beta
            elif new_beta > beta * 3.0:
                beta = new_beta
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        # Track best
        a_new = _softmax(z)
        c_new = _fft_convolve(a_new, a_new)
        sc = float(2.0 * n * c_new.max())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_z = z.copy()
        
        # Decay learning rate
        if t % decay_every == 0:
            current_lr = max(0.001, current_lr * decay_factor)
        
        # Kick if stuck
        if t % 5000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_z, best_score


def _lbfgs_round(z_init, deadline, n_betas=10):
    """Run L-BFGS with log-sum-exp continuation."""
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    best_z = z.copy()
    
    a = _softmax(z)
    c = _fft_convolve(a, a)
    best_score = float(2.0 * n * c.max())
    cm = float(c.max())
    
    beta_start = 0.2 * cm
    beta_end = 1e-5 * cm
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    def obj(zv, beta_val):
        av = _softmax(zv)
        cv = _fft_convolve(av, av)
        cmv = float(cv.max())
        if cmv < 1e-30:
            return 0.0
        shifted = (cv - cmv) / beta_val
        return beta_val * float(np.log(np.sum(np.exp(shifted)))) + cmv
    
    def grad(zv, beta_val):
        g, _, _, _ = _grad_max_conv_z(zv, beta_val)
        # Add the bias term from cm
        av = _softmax(zv)
        cv = _fft_convolve(av, av)
        cmv = float(cv.max())
        shifted = (cv - cmv) / beta_val
        e = np.exp(shifted)
        w = e / e.sum()
        # The full gradient includes dc/dz which already accounts for cm
        return g
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + max(5.0, time_budget)
        
        maxiter = 2000 if beta > 0.05 * cm else 5000
        try:
            result = minimize(
                obj, z, args=(beta,),
                method='L-BFGS-B', jac=grad,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': maxiter * 4}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -100, 100)
                z_new = z_new - z_new.mean()
                a_new = _softmax(z_new)
                c_new = _fft_convolve(a_new, a_new)
                sc = float(2.0 * n * c_new.max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_z = z_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_z, best_score


def _peak_surgery(z, deadline, rng, max_rounds=500):
    """Directly reduce peaks by modifying z at contributing positions."""
    n = len(z)
    best_z = z.copy()
    a = _softmax(z)
    c = _fft_convolve(a, a)
    best_score = float(2.0 * n * c.max())
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(best_z)
        c = _fft_convolve(a, a)
        cm = float(c.max())
        
        # Find top peaks
        n_peaks = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        for pk in peak_indices:
            # Valid i range for b[pk] = sum a[i]*a[pk-i]
            i_min = max(0, pk - n + 1)
            i_max = min(n - 1, pk)
            if i_min > i_max:
                continue
            valid_i = np.arange(i_min, i_max + 1)
            contribs = a[valid_i] * a[pk - valid_i]
            
            # Try reducing the top contributors
            order = np.argsort(contribs)[::-1][:6]
            for pos in order:
                idx = valid_i[pos]
                # Try reducing this index
                for delta in [0.02, 0.05, 0.1, 0.2, 0.5]:
                    z_trial = best_z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax(z_trial)
                    c_trial = _fft_convolve(a_trial, a_trial)
                    sc = float(2.0 * n * c_trial.max())
                    if sc < best_score - 1e-10:
                        best_score = sc
                        best_z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            
            # Try increasing non-contributing positions
            if not improved:
                # Positions that don't contribute to this peak
                non_contrib = np.setdiff1d(np.arange(n), valid_i)
                if len(non_contrib) > 0:
                    # Pick positions with smallest current a value
                    small_mask = non_contrib[np.argsort(a[non_contrib])[:5]]
                    for idx in small_mask:
                        for delta in [0.05, 0.1, 0.2]:
                            z_trial = best_z.copy()
                            z_trial[idx] += delta
                            a_trial = _softmax(z_trial)
                            c_trial = _fft_convolve(a_trial, a_trial)
                            sc = float(2.0 * n * c_trial.max())
                            if sc < best_score - 1e-10:
                                best_score = sc
                                best_z = z_trial.copy()
                                improved = True
                                break
                        if improved:
                            break
            
            if improved:
                break
        
        if not improved and rnd > 50:
            # Try random perturbation
            z_trial = best_z + 0.05 * rng.normal(size=n)
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax(z_trial)
            c_trial = _fft_convolve(a_trial, a_trial)
            sc = float(2.0 * n * c_trial.max())
            if sc < best_score - 1e-10:
                best_score = sc
                best_z = z_trial.copy()
                improved = True
            if not improved and rnd > 200:
                break
    
    return best_z, best_score


def _upsample_z(z, target_n):
    """Upsample z-values from current length to target_n."""
    n = len(z)
    if n == target_n:
        return z.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    z_new = np.interp(x1, x0, z)
    z_new = z_new - z_new.mean()
    return z_new


def _make_ramp(n, exponent=2.0):

    t = np.linspace(0, 1, n)

    best_seq = None
    best_score = np.inf
    best_n = 0
    
    def update_best(a_seq, sc, label=""):
        nonlocal best_seq, best_score, best_n
        if a_seq is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(a_seq, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = len(best_seq)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={best_n}) [{label}] t={elapsed:.0f}s")
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent loaded: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        print("No parent available, using generated profiles")
    
    # --- Phase 1: Multi-scale from profiles (40% of time) ---
    phase1_end = t_start + (budget_s - 30.0) * 0.40
    print(f"\n=== Phase 1: Multi-scale optimization (deadline {phase1_end - t_start:.0f}s) ===")
    
    # If we have a parent, use it as the top-scale start
    if parent is not None:
        parent_n = len(parent)
        # Use parent as the start for the top scale
        # First optimize at parent's scale
        if parent_n >= 2048:
            z_p = np.log(np.maximum(parent, 1e-15))
            z_p = z_p - z_p.mean()
            sc_p = _score_norm(parent)
            
            # Quick Adam at parent scale
            adam_dl = min(phase1_end, t_start + 120.0)
            z_opt, sc_opt = _adam_optimize(z_p, adam_dl, rng, lr=0.03, decay_every=3000)
            a_opt = _softmax(z_opt)
            update_best(a_opt, sc_opt, "parent_adam")
            
            # Also try L-BFGS
            if time.monotonic() < phase1_end - 30:
                lbfgs_dl = min(phase1_end - 5, time.monotonic() + 180.0)
                z_lb, sc_lb = _lbfgs_round(z_opt, lbfgs_dl, n_betas=12)
                a_lb = _softmax(z_lb)
                update_best(a_lb, sc_lb, "parent_lbfgs")
            
            # Peak surgery
            if time.monotonic() < phase1_end - 10:
                ps_dl = min(phase1_end - 3, time.monotonic() + 60.0)
                z_ps, sc_ps = _peak_surgery(z_opt if sc_opt < sc_lb else z_lb, ps_dl, rng)
                a_ps = _softmax(z_ps)
                update_best(a_ps, sc_ps, "parent_surgery")
        else:
            # Parent is small, upsample to larger scales
            z_cur = np.log(np.maximum(parent, 1e-15))
            z_cur = z_cur - z_cur.mean()
            sc_cur = _score_norm(parent)
            
            # Upsample to 2048 or 4096
            target = 4096 if parent_n < 2048 else 4096
            z_up = _upsample_z(z_cur, target)
            a_up = _softmax(z_up)
            sc_up = _score_norm(a_up)
            print(f"  Upsampled parent to n={target}: score_norm={sc_up:.10f}")
            
            adam_dl = min(phase1_end - 30, t_start + 300.0)
            z_opt, sc_opt = _adam_optimize(z_up, adam_dl, rng, lr=0.05)
            a_opt = _softmax(z_opt)
            update_best(a_opt, sc_opt, "parent_upsampled_adam")
    else:
        # No parent: do full multi-scale from scratch
        # Start at n=128 with best profile
        n_start = 128
        profiles = _make_profiles(n_start, rng)
        best_prof = None
        best_prof_sc = np.inf
        for prof in profiles[:12]:
            sc = _score_norm(prof)
            if sc < best_prof_sc:
                best_prof_sc = sc
                best_prof = prof.copy()
        
        print(f"  Best initial profile at n={n_start}: score_norm={best_prof_sc:.10f}")
        
        z_cur = np.log(np.maximum(best_prof, 1e-15))
        z_cur = z_cur - z_cur.mean()
        sc_cur = best_prof_sc
        
        # Optimize at n=128
        adam_dl = min(phase1_end, t_start + 30.0)
        z_cur, sc_cur = _adam_optimize(z_cur, adam_dl, rng, lr=0.1, decay_every=2000)
        print(f"  After n=128 Adam: score_norm={sc_cur:.10f}")
        
        # Multi-scale up
        scales = [256, 512, 1024, 2048, 4096]
        for target_n in scales:
            if time.monotonic() > phase1_end - 15:
                break
            z_up = _upsample_z(z_cur, target_n)
            a_up = _softmax(z_up)
            sc_up = _score_norm(a_up)
            print(f"  Upsampled to n={target_n}: score_norm={sc_up:.10f}")
            
            # Adam at this scale
            alloc = min(60.0, (phase1_end - time.monotonic()) * 0.4)
            adam_dl = time.monotonic() + max(10.0, alloc)
            z_opt, sc_opt = _adam_optimize(z_up, adam_dl, rng, lr=0.05, decay_every=3000)
            
            # Quick L-BFGS
            if time.monotonic() < phase1_end - 20:
                lbfgs_dl = min(phase1_end - 5, time.monotonic() + 30.0)
                z_lb, sc_lb = _lbfgs_round(z_opt, lbfgs_dl, n_betas=6)
                if sc_lb < sc_opt:
                    sc_opt = sc_lb
                    z_opt = z_lb
            
            if sc_opt < sc_cur:
                z_cur = z_opt
                sc_cur = sc_opt
                a_cur = _softmax(z_cur)
                update_best(a_cur, sc_cur, f"scale_{target_n}")
            else:
                sc_cur = min(sc_cur, sc_opt)
    
    print(f"\n  Phase 1 done: best_score={best_score:.10f}, n={best_n}")
    
    # --- Phase 2: Fine optimization at best scale (30% of time) ---
    phase2_end = t_start + (budget_s - 30.0) * 0.70
    print(f"\n=== Phase 2: Fine optimization (deadline {phase2_end - t_start:.0f}s) ===")
    
    if best_seq is not None and time.monotonic() < phase2_end - 20:
        n_best = len(best_seq)
        z_best = np.log(np.maximum(best_seq, 1e-15))
        z_best = z_best - z_best.mean()
        
        # Long Adam run
        adam_dl = min(phase2_end - 60, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
        z_adam, sc_adam = _adam_optimize(z_best, adam_dl, rng, lr=0.02, decay_every=4000, decay_factor=0.9)
        a_adam = _softmax(z_adam)
        update_best(a_adam, sc_adam, "phase2_adam")
        
        # L-BFGS from Adam result
        if time.monotonic() < phase2_end - 30:
            lbfgs_dl = min(phase2_end - 10, time.monotonic() + 120.0)
            z_lb, sc_lb = _lbfgs_round(z_adam, lbfgs_dl, n_betas=15)
            a_lb = _softmax(z_lb)
            update_best(a_lb, sc_lb, "phase2_lbfgs")
        
        # Peak surgery
        if time.monotonic() < phase2_end - 10:
            z_cur = z_adam if sc_adam <= (sc_lb if 'sc_lb' in dir() else np.inf) else z_lb
            ps_dl = min(phase2_end - 5, time.monotonic() + 60.0)
            z_ps, sc_ps = _peak_surgery(z_cur, ps_dl, rng, max_rounds=500)
            a_ps = _softmax(z_ps)
            update_best(a_ps, sc_ps, "phase2_surgery")
    
    print(f"\n  Phase 2 done: best_score={best_score:.10f}, n={best_n}")
    
    # --- Phase 3: Try alternative scale n=8192 (15% of time) ---
    phase3_end = t_start + (budget_s - 30.0) * 0.85
    print(f"\n=== Phase 3: Try n=8192 (deadline {phase3_end - t_start:.0f}s) ===")
    
    if best_seq is not None and time.monotonic() < phase3_end - 60:
        # Upsample best to 8192
        z_8192 = _upsample_z(np.log(np.maximum(best_seq, 1e-15)), 8192)
        a_8192 = _softmax(z_8192)
        sc_8192 = _score_norm(a_8192)
        print(f"  Upsampled to n=8192: score_norm={sc_8192:.10f}")
        
        # Adam at 8192
        adam_dl = min(phase3_end - 20, time.monotonic() + (phase3_end - time.monotonic()) * 0.7)
        z_opt, sc_opt = _adam_optimize(z_8192, adam_dl, rng, lr=0.02, decay_every=3000)
        a_opt = _softmax(z_opt)
        update_best(a_opt, sc_opt, "n8192_adam")
        
        # Quick surgery
        if time.monotonic() < phase3_end - 10:
            ps_dl = min(phase3_end - 5, time.monotonic() + 30.0)
            z_ps, sc_ps = _peak_surgery(z_opt, ps_dl, rng, max_rounds=200)
            a_ps = _softmax(z_ps)
            update_best(a_ps, sc_ps, "n8192_surgery")
    
    print(f"\n  Phase 3 done: best_score={best_score:.10f}, n={best_n}")
    
    # --- Phase 4: Final polish (15% of time) ---
    print(f"\n=== Phase 4: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        z_final = np.log(np.maximum(best_seq, 1e-15))
        z_final = z_final - z_final.mean()
        
        # Very fine Adam
        fine_dl = min(deadline - 30, time.monotonic() + 120.0)
        z_fine, sc_fine = _adam_optimize(z_final, fine_dl, rng, lr=0.005, decay_every=2000, decay_factor=0.95)
        a_fine = _softmax(z_fine)
        update_best(a_fine, sc_fine, "final_adam")
        
        # Tiny-beta L-BFGS
        if time.monotonic() < deadline - 15:
            lbfgs_dl = min(deadline - 5, time.monotonic() + 60.0)
            z_tlb, sc_tlb = _lbfgs_round(z_fine, lbfgs_dl, n_betas=8)
            a_tlb = _softmax(z_tlb)
            update_best(a_tlb, sc_tlb, "final_lbfgs")
        
        # Final surgery
        if time.monotonic() < deadline - 5:
            z_s = z_fine if sc_fine <= (sc_tlb if 'sc_tlb' in dir() else np.inf) else z_tlb
            ps_dl = min(deadline - 2, time.monotonic() + 30.0)
            z_fs, sc_fs = _peak_surgery(z_s, ps_dl, rng, max_rounds=300)
            a_fs = _softmax(z_fs)
            update_best(a_fs, sc_fs, "final_surgery")
    
    # --- Final validation and return ---
    print(f"\n=== Final Results ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s_final = float(final_seq.sum())
        if s_final > 0.01:
            # Normalize to ensure good properties
            final_seq = final_seq / s_final
        else:
            final_seq = _make_ramp(256, 2.0)
        
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final_seq])
            print(f"evaluate_sequence score: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence failed: {e}")
        
        sc_final = _score(final_seq)
        print(f"Internal score: {sc_final:.10f}, n={len(final_seq)}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_ramp(256, 2.0)
    print(f"FALLBACK: n=256, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
