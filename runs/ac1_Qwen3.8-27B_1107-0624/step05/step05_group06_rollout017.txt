    """Solve max sum(b) s.t. b>=0, conv(f,b) <= M."""
    # conv(f,b)_k = sum_j f[j]*b[k-j] for valid j, k-j

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):

def _next_pow2(n):

    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fast(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f[j]
    return C

def _fw_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b>=0, conv(f,b) <= M. Returns (g, S)."""
    C = _build_conv_matrix(f)
    b_ub = M * np.ones(2 * n - 1)
    c_obj = -np.ones(n)
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')
    except Exception:
        return None, 0.0
    if not result.success:
        return None, 0.0
    b = np.maximum(result.x, 0)
    S = float(b.sum())
    if S < 1e-15:
        return None, 0.0
    g = b / S
    return g, S

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_run(a_init, beta, n, max_iter=3000, ftol=1e-15, gtol=1e-12):
    """Run L-BFGS-B on smoothed objective. Returns optimized a."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - np.mean(z0)

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 30, 'maxls': 100})
        return _softmax(result.x)
    except Exception:
        return a_init

    """Beta-continuation L-BFGS schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025,

        if remaining < 3.0:

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - np.mean(z)
    return _softmax(z)

def _try_length(a0, deadline, label, verbose=False):
    """Try optimizing a sequence of given length, return (best_a, best_sc)."""
    n = len(a0)
    if verbose:
        print(f"  [{label}] n={n}, init_score={_score_fast(a0):.10f}")
    best_a = a0.copy()
    best_sc = _score_fast(a0)

    # Phase A: L-BFGS
    if time.monotonic() < deadline - 10:
        a_lb = _lbfgs_schedule(a0, deadline, min_remaining=20.0)
        sc_lb = _score_fast(a_lb)
        if sc_lb < best_sc:
            best_a = a_lb
            best_sc = sc_lb
            if verbose:
                print(f"    [{label}] after L-BFGS: {best_sc:.10f}")

    # Phase B: FW steps
    if time.monotonic() < deadline - 5:
        a_fw, sc_fw = _fw_optimize(best_a, deadline, verbose=False)
        if sc_fw < best_sc:
            best_a = a_fw
            best_sc = sc_fw
            if verbose:
                print(f"    [{label}] after FW: {best_sc:.10f}")

    # Phase C: L-BFGS again
    if time.monotonic() < deadline - 5:
        a_lb2 = _lbfgs_schedule(best_a, deadline, min_remaining=5.0)
        sc_lb2 = _score_fast(a_lb2)
        if sc_lb2 < best_sc:
            best_a = a_lb2
            best_sc = sc_lb2

    # Phase D: Subgradient
    if time.monotonic() < deadline - 2:
        a_sg, sc_sg = _subgrad_polish(best_a, deadline, n_steps=2000, lr_init=0.02)
        if sc_sg < best_sc:
            best_a = a_sg
            best_sc = sc_sg

    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function. Uses LP-FW + L-BFGS + subgradient hybrid approach
    with multi-start diversification across different sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update(p, sc=None):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score_fast(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0 and parent_raw.sum() > 0.01:
            parent = parent_raw / parent_raw.sum()
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score_fast(parent):.10f}")
        _update(parent)
    else:
        n_def = 2048
        parent = _make_profile(n_def, "uniform")
        print(f"No parent, using uniform n={n_def}, score={_score_fast(parent):.10f}")
        _update(parent)

    # === Phase 1: Main L-BFGS from parent ===
    print(f"\n=== Phase 1: L-BFGS from parent (t={time.monotonic()-t_start:.1f}s) ===")
    n_main = len(parent)
    a_main = _lbfgs_schedule(parent, deadline, min_remaining=600.0)
    _update(a_main)

    # === Phase 2: LP-FW broad search ===
    print(f"\n=== Phase 2: LP-FW (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 300:
        fw_deadline = min(deadline - 300.0, time.monotonic() + 400.0)
        a_fw, sc_fw = _fw_optimize(best_seq, fw_deadline, verbose=True)
        _update(a_fw, sc_fw)

    # === Phase 3: L-BFGS refinement ===
    print(f"\n=== Phase 3: L-BFGS refine (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 200:
        a_r = _lbfgs_schedule(best_seq, deadline, min_remaining=200.0)
        _update(a_r)

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 100:
        a_sg, sc_sg = _subgrad_polish(best_seq, max(time.monotonic() + 150.0, deadline - 100.0),
                                      n_steps=5000, lr_init=0.03)
        _update(a_sg, sc_sg)

    # === Phase 5: Multi-restart diversification ===
    print(f"\n=== Phase 5: Multi-restart (t={time.monotonic()-t_start:.1f}s) ===")
    if time.monotonic() < deadline - 60:
        remaining = deadline - 60.0 - time.monotonic()
        n_restarts = min(6, max(1, int(remaining / 80.0)))
        strategies = [
            ("perturb_03", lambda: _perturb(best_seq, 0.03, rng)),
            ("perturb_01", lambda: None),  # placeholder
            ("tapered", None),
            ("cosine", None),
            ("gaussian", None),
            ("perturb_05", lambda: _perturb(best_seq, 0.05, rng)),
        ]

        for i in range(n_restarts):
            if time.monotonic() > deadline - 50:
                break
            alloc = (deadline - 50.0) / (n_restarts - i)
            sub_deadline = time.monotonic() + alloc * 0.8

            if i < 2:
                p = _perturb(best_seq, 0.02 + 0.02 * i, rng)
            elif i == 2:
                p = _make_profile(len(best_seq), "tapered")
                p = 0.5 * best_seq / best_seq.sum() + 0.5 * p
            elif i == 3:
                p = _make_profile(len(best_seq), "cosine")
                p = 0.5 * best_seq / best_seq.sum() + 0.5 * p
            elif i == 4:
                p = _make_profile(len(best_seq), "gaussian")
                p = 0.5 * best_seq / best_seq.sum() + 0.5 * p
            else:
                p = _perturb(best_seq, 0.05, rng)

            print(f"  Restart {i}: start={_score_fast(p):.10f}")
            a_r, sc_r = _try_length(p, sub_deadline, f"r{i}", verbose=False)
            _update(a_r, sc_r)

    # === Phase 6: Fine L-BFGS with very small beta ===
    print(f"\n=== Phase 6: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.0002, 0.0001, 0.00005, 0.00002])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 5:
                break
            max_iter = min(3000, max(200, int((deadline - 5 - time.monotonic()) / 0.03)))
            a_f = _lbfgs_run(best_seq, beta, n, max_iter=max_iter)
            _update(a_f)

    # === Phase 7: Final subgradient ===
    print(f"\n=== Phase 7: Final SG (t={time.monotonic()-t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 1.0,
                                        n_steps=3000, lr_init=0.015)
        _update(a_fin, sc_fin)

    # === Final validation ===
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score_fast(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(2048, "uniform")
    print(f"FALLBACK: score={_score_fast(fallback):.10f}")
    return [float(x) for x in fallback]
