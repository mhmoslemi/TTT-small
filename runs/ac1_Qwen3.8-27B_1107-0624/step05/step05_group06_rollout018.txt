Multi-scale L-BFGS optimization with adaptive beta-continuation, p-norm smoothing,

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

    s = float(a.sum())
    if s < 1e-15:

        if time.monotonic() > deadline:

def _subgrad_momentum(a_init, deadline, n_steps=5000, lr=0.02, momentum=0.9):
    """Subgradient polishing with momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(best_a)
    vel = np.zeros(n)
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 1e-5 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        step = lr / (1.0 + 0.001 * i)
        
        vel = momentum * vel - step * grad
        z = z + vel
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

    a_final = _softmax(z)
    sc_final = _score(a_final)

        best_a = a_final.copy()

def _upsample(a_small, n_large):
    """Upsample a sequence from small to large length using cubic interpolation."""
    from scipy.interpolate import interp1d
    n_small = len(a_small)
    if n_small == n_large:
        return a_small.copy()
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    f = interp1d(x_small, a_small, kind='cubic', fill_value='extrapolate')
    a_large = f(x_large)
    a_large = np.maximum(a_large, 0)
    s = a_large.sum()
    if s < 1e-15:
        return np.ones(n_large) / n_large
    return a_large / s

def _symmetrize(a):
    """Make a sequence palindromic (symmetric)."""
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        return a / max(a.sum(), 1e-15)
    return a_sym / s

"""
Multi-scale L-BFGS optimization with adaptive beta-continuation, p-norm smoothing,
momentum subgradient polishing, and symmetry exploitation for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _smoobj_pnorm(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-300:
        return 0.0
    sp = np.sum(c ** p)
    if sp < 1e-300:
        return 2.0 * n * c_max
    smoothed = sp ** (1.0 / p)
    return 2.0 * n * smoothed


def _smograd_pnorm(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-300:
        return np.zeros_like(z)
    sp = np.sum(c ** p)
    if sp < 1e-300:
        return np.zeros_like(z)
    w = (c ** (p - 1)) * (sp ** (1.0 / p - 1.0))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_subgrad(a, n, tol=1e-6):
    """Compute exact subgradient (active set) of 2n*max(conv(a,a)) w.r.t. a."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    active = c >= (c_max - tol * c_max)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 4.0 * n * h, c, c_max


def _lbfgs_run(a_init, n, beta, max_iter, ftol=1e-14, gtol=1e-10, maxcor=40):
    """Single L-BFGS-B run with LSE smoothing."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj_lse(z, beta, n)

    def jac(z):
        return _smoobj_lse_grad_wrapper(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                     'maxcor': maxcor, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x), int(result.nit), result.success
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        return a_init, 0, False


def _smoobj_lse_grad_wrapper(z, beta, n):
    return _smograd_lse(z, beta, n)


def _lbfgs_run_pnorm(a_init, n, p, max_iter, ftol=1e-14, gtol=1e-9, maxcor=30):
    """Single L-BFGS-B run with p-norm smoothing."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj_pnorm(z, p, n)

    def jac(z):

def _subgrad_momentum(a_init, deadline, n_steps=8000, lr=0.012, momentum=0.85, tol=1e-6):
    """Subgradient polishing with momentum in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(best_a)
    vel = np.zeros(n)
    no_improve = 0
    best_check_sc = best_sc

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        step = lr / (1.0 + 0.0005 * i)
        vel = momentum * vel - step * grad
        z = z + vel
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.9
                    momentum = min(0.95, momentum * 1.01)
                    no_improve = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_hann(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 - np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_blackman(n):
    t = np.linspace(0, 1, n)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _upsample(a_small, n_large):
    n_small = len(a_small)
    if n_small == n_large:
        return a_small.copy()
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    f = interp1d(x_small, a_small, kind='cubic', fill_value='extrapolate')
    a_large = f(x_large)
    a_large = np.maximum(a_large, 0)
    s = a_large.sum()
    if s < 1e-15:
        return np.ones(n_large) / n_large
    return a_large / s


def _random_perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=3, amp_scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _symmetrize(a):
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-15:
        total = a.sum()
        if total < 1e-15:
            return a
        return a / total
    return a_sym / s

        sc_sym = _score(a_sym)
        if sc_sym < best_score:
            update_best(a_sym, sc_sym, "symmetrized")
        else:
            print(f"  Symmetrized: {sc_sym:.10f} (no improvement over {best_score:.10f})")
        # Try L-BFGS from symmetrized point
        if sc_sym < best_score + 0.001 and time.monotonic() < deadline - 40:
            c_sym = _full_convolve(a_sym, a_sym)
            mb_sym = float(c_sym.max())
            beta_sym = mb_sym * 0.0003
            a_s2, _, _ = _lbfgs_run(a_sym, n, beta_sym, max_iter=2000)
            sc_s2 = _score(a_s2)
            update_best(a_s2, sc_s2, "sym_lbfgs")

    # --- Phase 5: Fine L-BFGS polishing ---
    print(f"\nPhase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 20:
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.00005, 0.00002, 0.00001, 0.000005])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 15:
                break
            remaining = deadline - 15 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(3000, max(100, int(remaining / 0.035)))
            a_f, niter, _ = _lbfgs_run(best_seq, n, beta, max_iter,
                                       ftol=1e-15, gtol=1e-11, maxcor=50)
            sc_f = _score(a_f)
            update_best(a_f, sc_f, f"fine_{i}")

    # --- Phase 6: Multi-restart with diverse strategies ---
    print(f"\nPhase 6: Multi-restart (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        n_restarts = min(4, max(1, int(remaining / 100)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break

            if i == 0:
                p_start = _random_perturb(best_seq, 0.015, rng)
            elif i == 1:
                p_start = _fourier_perturb(best_seq, rng, n_modes=3, amp_scale=0.015)
            elif i == 2:
                # Mix with a different profile
                profile = _make_blackman(n)
                p_start = 0.8 * best_seq + 0.2 * profile
                p_start = p_start / p_start.sum()
                p_start = _symmetrize(p_start)
            else:
                # Larger perturbation
                p_start = _random_perturb(best_seq, 0.04, rng)

            sc_start = _score(p_start)
            print(f"  Restart {i}: start_score={sc_start:.10f}")

            if sc_start < best_score + 0.005:
                a_r, sc_r = _run_lbfgs_schedule(p_start, n, deadline, min_remaining=20)
                update_best(a_r, sc_r, f"restart_{i}")

                # Quick subgradient polish
                if sc_r < best_score + 0.002 and time.monotonic() < deadline - 15:
                    a_r2, sc_r2 = _subgrad_momentum(a_r, deadline - 15,
                                                     n_steps=2000, lr=0.008,
                                                     momentum=0.85, tol=1e-7)
                    update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # --- Phase 7: Final fine-tuning cycle ---
    print(f"\nPhase 7: Final tuning (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # One last subgradient with very small steps
        a_fin, sc_fin = _subgrad_momentum(best_seq, deadline - 3,
                                           n_steps=2000, lr=0.003,
                                           momentum=0.8, tol=1e-8)
        update_best(a_fin, sc_fin, "final_sg")

        # Symmetrize one more time
        a_fin_sym = _symmetrize(a_fin)
        sc_fin_sym = _score(a_fin_sym)
        update_best(a_fin_sym, sc_fin_sym, "final_sym")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_hann(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_hann(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
