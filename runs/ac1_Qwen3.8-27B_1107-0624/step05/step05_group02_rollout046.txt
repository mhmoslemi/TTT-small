"""
Optimized soft-max gradient descent with LP initialization and adaptive temperature
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-based gradient computation, LP for initial descent, multi-restart,
and n-exploration to beat the previous bound of 1.5059.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, w, n):
    """Compute gradient of soft-max smoothed objective in z-space.
    Returns grad = 4*n*a*(H - W) where H_i = sum_k w_k * a_{k-i}, W = sum_k w_k * C_k.
    """
    N = _next_pow2(3 * n - 1)
    w_padded = np.zeros(N)
    w_padded[:2 * n - 1] = w
    a_padded = np.zeros(N)
    a_padded[:n] = a
    W_freq = np.fft.rfft(w_padded)
    A_freq = np.fft.rfft(a_padded)
    H = np.fft.irfft(W_freq * np.conj(A_freq), N)[:n]
    C = _full_convolve(a, a)
    W_val = float(np.dot(w, C))
    grad = 4.0 * n * a * (H - W_val)
    return grad


def _soft_max_weights(C, beta):
    """Compute soft-max weights over convolution values."""
    c_shifted = (C - C.max()) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
        return w
    return e / s


def _lp_step(f, n, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns g or None."""
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

        # Try larger n
        current_n = len(best_seq)
        target_ns = []
        if current_n <= 8192:
            target_ns.append(16384)
        if current_n <= 4096:
            target_ns.append(8192)

        for target_n in target_ns:
            if time.monotonic() > phase4_end - 80:
                break
            p_interp = _interp_profile(best_seq, target_n)
            _update_best(p_interp)
            print(f"  Interpolated to n={target_n}: score={_score_fn(p_interp):.8f}")

            remaining = phase4_end - time.monotonic() - 40
            if remaining > 100:
                a_r, _ = _adam_optimize_v2(p_interp, time.monotonic() + remaining, lr=0.004, beta_scale=0.003)
                _update_best(a_r)
                print(f"  Optimized n={target_n}: score={_score_fn(a_r):.8f}")

        # Also try smaller n (sometimes helps due to better conditioning)
        if current_n > 4096 and time.monotonic() < phase4_end - 100:
            p_small = _interp_profile(best_seq, 4096)
            _update_best(p_small)
            remaining = phase4_end - time.monotonic() - 30
            if remaining > 80:
                a_r, _ = _adam_optimize_v2(p_small, time.monotonic() + remaining, lr=0.006, beta_scale=0.004)
                _update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish (1550s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_final, _ = _adam_optimize(a_best_norm, time.monotonic() + remaining, lr=0.002,
                                        beta_init=0.002 * _full_convolve(a_best_norm, a_best_norm).max())
            _update_best(a_final)

    print(f"Phase 5 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score_fn(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
