from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return 2.0 * n * c.max() / (s * s)

    """Compute gradient of smooth max of conv(a,a) w.r.t. log-space variables z."""
    n = len(a)

def _adam_step(z, grad, m_buf, v_buf, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m_buf = b1 * m_buf + (1-b1) * grad
    v_buf = b2 * v_buf + (1-b2) * grad**2
    mh = m_buf / (1 - b1**t)
    vh = v_buf / (1 - b2**t)
    update = lr * mh / (np.sqrt(vh) + eps)
    # Clip update
    max_norm = 1.0
    norm = np.linalg.norm(update)
    if norm > max_norm:
        update *= max_norm / norm
    return z - update, m_buf, v_buf

    e = np.exp(z)
