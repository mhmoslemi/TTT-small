```
    N = next_pow2(2*n - 1)
    for _ in range(iters):

        if s > 0:

        a = a_new
    return a
```

- Or the derivative within a piece is zero

Within a region where conv_k* = c_k* + d_k* * a_j is the max:

Setting to zero: d_k*(S+a_j) = 2(c_k* + d_k*a_j)

d_k*S - 2*c_k* = d_k*a_j
a_j* = (d_k*S - 2*c_k*) / d_k = S - 2*c_k*/d_k

So the critical point in each linear piece is at a_j = S - 2*c_k*/d_k.

```python
Coordinate descent + log-sum-exp continuation for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key insight: changing a[j] by delta changes conv by delta * (2*a shifted by j).
This enables exact 1D line search per coordinate via golden section.
Combined with multi-scale continuation for global convergence.

import numpy as np
from scipy.optimize import minimize, brent
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _opscore(a, conv):
    """Score given pre-computed convolution. Assumes sum(a)=1."""
    n = len(a)
    return float(2.0 * n * np.max(conv))

def _golden_section_1d(f, a, b, tol=1e-10, max_iter=100):
    """Minimize f on [a, b] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2
```
