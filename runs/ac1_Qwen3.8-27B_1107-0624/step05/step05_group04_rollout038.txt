    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta_start = 0.3 * max_c0
    beta_end = 1e-6 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 200.0)
        level_deadline = time.monotonic() + time_budget
        maxiter = 500 if beta > 0.1 * max_c0 else (2000 if beta > 0.01 * max_c0 else 8000)
        maxfun = maxiter * 5
        try:
            result = minimize(
                _logsumexp_obj, z.copy(), args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    return best_a, best_score


def _coordinate_descent(a_init, deadline, rng, max_rounds=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a_cur = _softmax_fn(z)
    c_cur = _full_convolve(a_cur, a_cur)
    cm_cur = float(c_cur.max())

    step_sizes = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        if rnd % 200 == 0:
            pass

        # Find peak location
        peak_k = int(np.argmax(c_cur))

        # Find top contributors to the peak
        valid_i = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        if len(valid_i) == 0:
            continue
        prod = a_cur[valid_i] * a_cur[peak_k - valid_i]
        order = np.argsort(prod)[::-1][:12]

        improved = False
        for idx_pos in order:
            idx = valid_i[idx_pos]
            other = peak_k - idx
            if other < 0 or other >= n:
                continue
            # Try reducing idx
            for delta in step_sizes:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                sc = float(c_trial.max())
                if sc < cm_cur - 1e-14:
                    cm_cur = sc
                    c_cur = c_trial
                    z = z_trial.copy()
                    a_cur = a_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing the other index too
            for idx_pos in order[:6]:
                idx = valid_i[idx_pos]
                other = peak_k - idx
                if other < 0 or other >= n or other == idx:
                    continue
                for delta in step_sizes[:3]:
                    z_trial = z.copy()
                    z_trial[other] -= delta
                    a_trial = _softmax_fn(z_trial)
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc = float(c_trial.max())
                    if sc < cm_cur - 1e-14:
                        cm_cur = sc
                        c_cur = c_trial
                        z = z_trial.copy()
                        a_cur = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try increasing small values to spread energy
            small_idx = np.argsort(a_cur)[:8]
            for idx in small_idx:
                for delta in [0.1, 0.3, 0.5]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc = float(c_trial.max())
                    if sc < cm_cur - 1e-14:
                        cm_cur = sc
                        c_cur = c_trial
                        z = z_trial.copy()
                        a_cur = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if improved:
            sc_full = _score_norm_fn(a_cur)
            if sc_full < best_score:
                best_score = sc_full
                best_a = a_cur.copy()
        elif rnd > 200:
            break

    return best_a, best_score


def _run_adam_gd(a_init, deadline, rng, lr_init=0.1, max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break
        beta = _adaptive_beta(c)
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        if not np.all(np.isfinite(grad_z)):
            break
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 3000 == 0:
            lr = max(0.005, lr * 0.85)
        if t % 15000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
    return best_a, best_score


def _evaluate_candidates(deadline):
    """Quickly evaluate several analytical candidates at various lengths."""
    candidates = []
    lengths = [1024, 2048, 4096, 6144, 8192]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        # Power ramps
        for p in [0.7, 1.0, 1.3, 1.5, 2.0]:
            a = _make_ramp(n, p)
            sc = _score_norm_fn(a)
            candidates.append((sc, a, f"ramp_p{p}_n{n}"))
        # Oscillatory ramps
        for p in [1.0, 1.3]:
            for freq in [1.0, 2.0, 3.0]:
                for amp in [0.05, 0.1, 0.2, 0.3]:
                    a = _make_ramp_osc(n, p, freq, amp)
                    sc = _score_norm_fn(a)
                    candidates.append((sc, a, f"osc_p{p}_f{freq}_a{amp}_n{n}"))
        # Gaussian-modulated ramp
        for w in [0.3, 0.5, 0.8]:
            a = _make_gaussian_ramp(n, 1.0, w)
            sc = _score_norm_fn(a)
            candidates.append((sc, a, f"gauss_w{w}_n{n}"))
    candidates.sort(key=lambda x: x[0])
    return candidates


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Try to load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")

    # Phase 1: Evaluate analytical candidates (short time budget)
    print("\nPhase 1: Evaluating analytical candidates...")
    p1_deadline = min(deadline, t_start + 60.0)
    candidates = _evaluate_candidates(p1_deadline)
    if candidates:
        print(f"  Top 5 candidates:")
        for sc, a, name in candidates[:5]:
            print(f"    {sc:.10f} - {name}")
        # Update best from candidates
        for sc, a, name in candidates[:5]:
            update_best(a, sc, name)

    # Phase 2: Multi-scale optimization from best candidate
    print(f"\nPhase 2: Multi-scale L-BFGS optimization (deadline t={deadline - t_start:.0f}s)")
    p2_end = t_start + (budget_s - 30.0) * 0.55

    # Start from the best found so far (or a ramp if nothing good)
    if best_seq is not None:
        a_cur = best_seq.copy()
        sc_cur = best_score
    else:
        a_cur = _make_ramp(4096, 1.0)
        sc_cur = _score_norm_fn(a_cur)
        update_best(a_cur, sc_cur, "init_ramp")

    # If best is already large, just optimize at that size
    n_best = len(a_cur)
    if n_best <= 1024:
        # Multi-scale up
        scales = [2048, 4096, 8192]
        for target_n in scales:
            if time.monotonic() > p2_end - 20:
                break
            a_target = _upsample(a_cur, target_n)
            sc_target = _score_norm_fn(a_target)
            print(f"  Upsampled to n={target_n}: score={sc_target:.10f}")
            alloc = min(120.0, (p2_end - time.monotonic()) * 0.4)
            opt_deadline = time.monotonic() + alloc
            a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_deadline, rng, n_betas=12)
            print(f"  After n={target_n} L-BFGS: score={sc_opt:.10f}")
            if sc_opt < sc_cur:
                a_cur = a_opt
                sc_cur = sc_opt
                update_best(a_cur, sc_cur, f"multi_scale_{target_n}")
    else:
        # Already large, just optimize
        opt_deadline = p2_end - 10
        a_opt, sc_opt = _run_lbfgs_continuation(a_cur, opt_deadline, rng, n_betas=20)
        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt
            update_best(a_cur, sc_cur, "lbfgs_large")

    # Phase 3: Coordinate descent for peak flattening
    print(f"\nPhase 3: Coordinate descent (deadline t={deadline - t_start:.0f}s)")
    p3_end = t_start + (budget_s - 30.0) * 0.75
    if time.monotonic() < p3_end - 10 and best_seq is not None:
        cd_deadline = p3_end - 10
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_deadline, rng, max_rounds=20000)
        update_best(a_cd, sc_cd, "coord_descent")

        # Run coordinate descent again from the result
        if time.monotonic() < p3_end - 10:
            cd_deadline2 = p3_end - 5
            a_cd2, sc_cd2 = _coordinate_descent(best_seq, cd_deadline2, rng, max_rounds=10000)
            update_best(a_cd2, sc_cd2, "coord_descent_2")

    # Phase 4: Adam GD refinement
    print(f"\nPhase 4: Adam GD refinement")
    p4_end = t_start + (budget_s - 30.0) * 0.88
    if time.monotonic() < p4_end - 10 and best_seq is not None:
        gd_deadline = p4_end - 10
        a_gd, sc_gd = _run_adam_gd(best_seq, gd_deadline, rng, lr_init=0.05, max_iter=30000)
        update_best(a_gd, sc_gd, "adam_gd")

        # Coordinate descent after Adam
        if time.monotonic() < p4_end - 10:
            cd_dl = p4_end - 5
            a_cd3, sc_cd3 = _coordinate_descent(best_seq, cd_dl, rng, max_rounds=5000)
            update_best(a_cd3, sc_cd3, "coord_after_adam")

    # Phase 5: Final polish - tiny beta L-BFGS + coordinate descent
    print(f"\nPhase 5: Final polish")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = 1e-5 * max_c_final

        dbeta_deadline = time.monotonic() + min(60.0, (deadline - time.monotonic()) * 0.4)
        try:
            result = minimize(
                _logsumexp_obj, z.copy(), args=(beta_tiny,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 10000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                update_best(a_new, sc_new, "tiny_beta_lbfgs")
        except Exception as ex:
            print(f"  Tiny-beta L-BFGS failed: {ex}")

        # Final coordinate descent
        if time.monotonic() < deadline - 5 and best_seq is not None:
            cd_final_dl = deadline - 3
            a_cf, sc_cf = _coordinate_descent(best_seq, cd_final_dl, rng, max_rounds=10000)
            update_best(a_cf, sc_cf, "final_coord")

    # Phase 6: Try a different length if time remains
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\nPhase 6: Try alternate length")
        n_alt = len(best_seq)
        if n_alt == 4096:
            targets = [6144, 8192]
        elif n_alt == 8192:
            targets = [6144, 4096]
        else:
            targets = [4096, 8192]

        for target_n in targets:
            if time.monotonic() > deadline - 30:
                break
            a_alt = _upsample(best_seq, target_n)
            sc_alt = _score_norm_fn(a_alt)
            print(f"  Alternate n={target_n}: score={sc_alt:.10f}")
            if sc_alt < best_score * 1.01:  # Only optimize if competitive
                alloc = min(60.0, deadline - time.monotonic() - 10)
                if alloc > 20:
                    a_opt, sc_opt = _run_lbfgs_continuation(a_alt, time.monotonic() + alloc, rng, n_betas=8)
                    update_best(a_opt, sc_opt, f"alt_{target_n}")
                    if time.monotonic() < deadline - 15:
                        a_cd, sc_cd = _coordinate_descent(a_opt, deadline - 10, rng, max_rounds=3000)
                        update_best(a_cd, sc_cd, f"alt_cd_{target_n}")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.0)
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(4096, 1.0)
    print(f"FALLBACK: n=4096, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
