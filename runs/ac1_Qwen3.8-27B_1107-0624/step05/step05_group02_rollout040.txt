Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp gradient descent with FFT-based gradients,
adaptive temperature, and multi-restart strategy.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_lse(a, w):
    """Compute gradient of LSE objective.
    a: sequence of length n (sum = 1)
    w: weights of length 2n-1 (sum = 1, non-negative)
    Returns: gradient of length n
    """
    n = len(a)
    N = _next_pow2(3 * n - 2)
    fw = np.fft.rfft(w.astype(np.float64), N)
    fa = np.fft.rfft(a[::-1].astype(np.float64), N)
    full_conv = np.fft.irfft(fw * fa, N)
    return 2.0 * full_conv[n - 1:2 * n - 1]


def _adam_step(z, grad, m_buf, v_buf, t_step, lr, b1=0.9, b2=0.999, eps=1e-8):

    a = _softmax(z)
    grad_centered = grad - np.dot(grad, a)
    
    m_buf = b1 * m_buf + (1.0 - b1) * grad_centered
    v_buf = b2 * v_buf + (1.0 - b2) * grad_centered ** 2
    mh = m_buf / (1.0 - b1 ** t_step)
    vh = v_buf / (1.0 - b2 ** t_step)

    z = z - z.mean()
    z = np.clip(z, -80.0, 80.0)

"""
Multi-phase LSE gradient descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradients, adaptive temperature cooling, and multi-restart strategy.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_normalized(a):
    """Score for a normalized sequence (sum=1): 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))

def _softmax(z):
    z = z - np.max(z)
    z = np.clip(z, -50, 50)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_gradient(a, w, fft_size):
    """Compute gradient: 2 * correlate(w, a, 'valid') via FFT.
    a: length n, w: length 2n-1, both non-negative.
    Returns array of length n.
    """
    n = len(a)
    fw = np.fft.rfft(w, fft_size)
    fa = np.fft.rfft(a[::-1], fft_size)
    full_conv = np.fft.irfft(fw * fa, fft_size)
    return 2.0 * full_conv[n-1:2*n-1]

def _lse_optimize(z_init, deadline, tau_init=1.0, tau_min=0.005, lr_init=0.03, 
                  check_every=10, restore_every=200, verbose=True):
    """
    LSE gradient descent with Adam in log-space parameterization.
    z_init: initial log-space parameters (length n)
    deadline: absolute time to stop
    Returns: (best_a, best_score)
    """
    n = len(z_init)
    fft_size = _next_pow2(3 * n - 2)
    
    z = z_init.copy()
    a = _softmax(z)
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    tau = tau_init
    lr = lr_init
    no_improve = 0
    iter_count = 0
    besttau = tau
    
    while time.monotonic() < deadline:
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(np.max(c))
        
        # LSE weights
        c_shifted = (c - cm) / tau
        c_shifted = np.clip(c_shifted, -200, 0)
        e = np.exp(c_shifted)
        s = float(np.sum(e))
        if s < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient via FFT
        grad = _compute_gradient(a, w, fft_size)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Center gradient for log-space
        grad_c = grad - np.dot(grad, a)
        
        # Adam step
        t_step += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_c
        v_buf = b2 * v_buf + (1.0 - b2) * grad_c * grad_c
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)
        
        step = lr * mh / (np.sqrt(vh) + eps)
        sn = np.linalg.norm(step)
        if sn > 3.0:
            step *= 3.0 / sn
        
        z = z - step
        z = z - z.mean()
        z = np.clip(z, -100, 100)
        iter_count += 1
        
        # Periodic evaluation
        if iter_count % check_every == 0:
            a_chk = _softmax(z)
            sc = _score_normalized(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive cooling
            if no_improve >= 50:
                tau = max(tau_min, tau * 0.5)
                lr = max(0.002, lr * 0.6)
                no_improve = 0
                # Reheat occasionally
                if tau <= tau_min * 2 and iter_count % (restore_every * 2) == 0:
                    tau = tau_min * 10
                    lr = max(lr, 0.01)
            elif no_improve >= 30:
                tau = max(tau_min, tau * 0.7)
            
            # Slow cooling
            if iter_count % 1000 == 0:
                tau = max(tau_min, tau * 0.95)
        
        # Hard checkpoint every restore_every iterations
        if iter_count % restore_every == 0:
            a_chk = _softmax(z)
            sc = _score_normalized(a_chk)
            if sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
    
    # Final evaluation
    a_final = _softmax(z)
    sc_final = _score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    if verbose:
        print(f"    LSE done: {iter_count} iters, tau={tau:.4f}, score={best_score:.10f}")
    
    return best_a, best_score

def _perturb_z(z, sigma, rng):
    z_new = z + sigma * rng.standard_normal(len(z))
    z_new -= z_new.mean()
    return np.clip(z_new, -80, 80)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _make_initial(n, asymmetry=0.3, rng=None):
    """Create an initial asymmetric bump."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    center = 0.35 + 0.3 * asymmetry * rng.random()
    width = 0.15 + 0.1 * rng.random()
    a = np.exp(-((x - center) / width) ** 2)
    # Add asymmetry
    a *= (1.0 + 0.3 * rng.random() * (x - 0.5))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Phase 1: LSE gradient descent from parent (main workhorse, ~500s)
    Phase 2: Multi-restart with varied perturbations (~500s)
    Phase 3: Larger-n exploration (~300s)
    Phase 4: Final fine-tuning (~100s)
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p_norm = p / s
        sc = float(_score_normalized(p_norm))
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_seq = p.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}, t={elapsed:.0f}s)")
    
    # Load parent sequence
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        a0 = parent.copy()
        n = len(a0)
    else:
        n = 8192
        a0 = _make_initial(n, rng=rng)
    
    update_best(a0)
    print(f"Starting: n={n}, score={best_score:.10f}")
    
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.mean()
    
    # === Phase 1: Main optimization from parent ===
    print(f"\n=== Phase 1: Main LSE optimization ({500}s) ===")
    p1_end = min(deadline, t_start + 500.0)
    
    if time.monotonic() < p1_end - 5:
        a1, s1 = _lse_optimize(z0, p1_end - 3, tau_init=2.0, tau_min=0.005, 
                               lr_init=0.04, check_every=5, restore_every=500)
        update_best(a1)
    
    # Follow-up with different params from best so far
    if best_seq is not None and time.monotonic() < p1_end - 30:
        a_b = best_seq / best_seq.sum()
        z_b = np.log(np.maximum(a_b, 1e-15))
        z_b -= z_b.mean()
        remaining = p1_end - time.monotonic() - 5
        if remaining > 30:
            a1b, s1b = _lse_optimize(z_b, time.monotonic() + remaining, 
                                     tau_init=0.5, tau_min=0.002,
                                     lr_init=0.02, check_every=3, restore_every=300)
            update_best(a1b)
    
    print(f"Phase 1 complete: best={best_score:.10f}")
    
    # === Phase 2: Multi-restart ===
    print(f"\n=== Phase 2: Multi-restart ({500}s) ===")
    p2_end = min(deadline, t_start + 1000.0)
    
    if best_seq is not None and time.monotonic() < p2_end - 30:
        a_best_n = best_seq / best_seq.sum()
        n_b = len(a_best_n)
        z_best = np.log(np.maximum(a_best_n, 1e-15))
        z_best -= z_best.mean()
        
        # Different restart strategies
        restarts = [
            ("small_perturb", 0.02, 0.3, 0.015),
            ("med_perturb", 0.05, 0.2, 0.012),
            ("struct_perturb", 0.0, 0.15, 0.01),
            ("freq_perturb", 0.0, 0.15, 0.01),
            ("large_perturb", 0.1, 0.1, 0.008),
        ]
        
        n_restarts = len(restarts)
        per_time = (p2_end - time.monotonic() - 20) / n_restarts
        
        for idx, (name, sigma, tau_i, lr_i) in enumerate(restarts):
            if time.monotonic() > p2_end - 15:
                break
            
            elapsed_r = time.monotonic() - t_start
            remaining_r = min(per_time, p2_end - time.monotonic() - 10)
            if remaining_r < 15:
                break
            
            print(f"  Restart {idx+1}/{n_restarts} ({name}) starting at t={elapsed_r:.0f}s, duration={remaining_r:.0f}s")
            
            if name == "struct_perturb":
                z_p = z_best.copy()
                half = n_b // 2
                z_p[:half] += 0.04 * rng.standard_normal(half)
                z_p[half:] -= 0.04 * rng.standard_normal(n_b - half)
            elif name == "freq_perturb":
                a_p = a_best_n.copy()
                A = np.fft.rfft(a_p)
                noise_scale = 0.03
                noise = noise_scale * rng.standard_normal(len(A))
                noise[0] = 0  # Don't perturb DC
                A = A + noise
                a_p = np.fft.irfft(A, n_b)
                a_p = np.maximum(a_p, 1e-10)
                a_p = a_p / a_p.sum()
                z_p = np.log(a_p)
            else:
                z_p = _perturb_z(z_best, sigma, rng)
            
            z_p = np.clip(z_p, -80, 80)
            
            a_r, s_r = _lse_optimize(z_p, time.monotonic() + remaining_r,
                                     tau_init=tau_i, tau_min=0.003,
                                     lr_init=lr_i, check_every=5, restore_every=400)
            update_best(a_r)
    
    print(f"Phase 2 complete: best={best_score:.10f}")
    
    # === Phase 3: Larger n ===
    print(f"\n=== Phase 3: Larger n exploration ({350}s) ===")
    p3_end = min(deadline, t_start + 1350.0)
    
    if best_seq is not None and time.monotonic() < p3_end - 30:
        for target_n in [16384, 32768]:
            if time.monotonic() > p3_end - 40:
                break
            p_i = _interp_profile(best_seq, target_n)
            update_best(p_i)
            
            if time.monotonic() < p3_end - 40:
                z_i = np.log(np.maximum(p_i, 1e-15))
                z_i -= z_i.mean()
                remaining = p3_end - time.monotonic() - 15
                if remaining > 30:
                    print(f"  Optimizing n={target_n} for {remaining:.0f}s")
                    a_i, s_i = _lse_optimize(z_i, time.monotonic() + remaining,
                                             tau_init=0.3, tau_min=0.01,
                                             lr_init=0.01, check_every=10, restore_every=500)
                    update_best(a_i)
    
    print(f"Phase 3 complete: best={best_score:.10f}")
    
    # === Phase 4: Different n from scratch ===
    p4_end = min(deadline, t_start + 1550.0)
    print(f"\n=== Phase 4: Fresh n exploration ===")
    
    if time.monotonic() < p4_end - 30:
        for fresh_n in [4096, 8192]:
            if time.monotonic() > p4_end - 30:
                break
            a_f = _make_initial(fresh_n, asymmetry=0.5, rng=rng)
            update_best(a_f)
            
            if time.monotonic() < p4_end - 30:
                z_f = np.log(np.maximum(a_f, 1e-15))
                z_f -= z_f.mean()
                remaining = p4_end - time.monotonic() - 15
                if remaining > 30:
                    a_fr, s_fr = _lse_optimize(z_f, time.monotonic() + remaining,
                                               tau_init=1.0, tau_min=0.01,
                                               lr_init=0.03, check_every=5, restore_every=400)
                    update_best(a_fr)
    
    print(f"Phase 4 complete: best={best_score:.10f}")
    
    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_b = best_seq / best_seq.sum()
        z_f = np.log(np.maximum(a_b, 1e-15))
        z_f -= z_f.mean()
        remaining = deadline - time.monotonic() - 3
        if remaining > 10:
            a_f, s_f = _lse_optimize(z_f, deadline - 2,
                                     tau_init=0.02, tau_min=0.001,
                                     lr_init=0.003, check_every=3, restore_every=200)
            update_best(a_f)
    
    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score_normalized(final / s)
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={elapsed:.0f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check failed: {ex}")
        return [float(x) for x in final]
    
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK uniform: score={_score_normalized(fallback):.10f}")
    return [float(x) for x in fallback]
