"""
Optimized search for low-peak autocorrelation sequences using
smooth gradient descent with annealing, multiple restarts, and
adaptive n-size exploration. Improves upon LP-based approach with
faster convergence via second-order-inspired updates.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n*max(conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _full_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp(c, beta):
    """Compute log(sum(exp((c - cmax)/beta))) * beta + cmax as smooth max."""
    cmax = float(c.max())
    shifted = (c - cmax) / beta
    # For numerical stability
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    s = np.log(float(e.sum()))
    return cmax + beta * s


def _gradient_smooth_max(a, beta):
    """
    Compute gradient of 2n * smooth_max(conv(a,a)) w.r.t. a.
    smooth_max is log-sum-exp approximation.
    Returns (gradient, smooth_max_value)
    """
    n = len(a)
    c = _fft_convolve(a, a)
    cmax = float(c.max())
    shifted = (c - cmax) / beta
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    s = float(e.sum())
    w = e / s  # weights over convolution positions
    
    # d smooth_max / d a_i = sum_k w_k * d c_k / d a_i
    # d c_k / d a_i = 2 * a_{k-i} (if k-i in [0,n-1])
    # So gradient_i = 2 * sum_k w_k * a_{k-i}
    # This is 2 * (w * a_reversed) truncated to [n-1, 2n-2]
    
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n-1:2*n-1]  # This gives sum_k w_k * a_{k-i} for i=0..n-1
    
    smooth_val = cmax + beta * math.log(s)
    grad = 4.0 * n * h  # 2n * 2 * h
    
    return grad, smooth_val


def _gradient_hard_max(a):
    """
    Compute subgradient of 2n * max(conv(a,a)) w.r.t. a.
    Returns (subgradient, max_value, argmax)
    """
    n = len(a)
    c = _fft_convolve(a, a)
    kmax = int(np.argmax(c))
    cmax = float(c[kmax])
    
    # Subgradient: 4n * a_{kmax-i} for valid i
    # a_{kmax-i} is valid when 0 <= kmax-i < n
    lo = max(0, kmax - (n-1))
    hi = min(n-1, kmax)
    grad = np.zeros(n)
    idx = np.arange(lo, hi+1)
    grad[idx] = 4.0 * n * a[kmax - idx][::-1]
    
    return grad, cmax, kmax


def _rmsprop_step(z, grad, m_buf, v_buf, t, lr, rho=0.9, eps=1e-8, decay=0.0):
    """Standard RMSProp update in log-space."""
    m_buf = rho * m_buf + (1 - rho) * grad
    v_buf = rho * v_buf + (1 - rho) * (grad * grad)
    update = lr * m_buf / (np.sqrt(v_buf) + eps)
    z_new = z - update
    if decay > 0:
        z_new = z_new * (1 - decay)
    return z_new, m_buf, v_buf

        log_grad = grad / np.maximum(a, 1e-15)

            den = float(np.dot(prev_grad, prev_grad))

            v = log_grad + beta_cg * momentum * v

        for _ in range(5):

def _annealed_optimization(a_init, deadline, n_temps=15):

    Annealed smooth gradient descent.

    v_buf = np.zeros(n)

        # Each temperature phase gets proportional time
        # More time for lower temperatures

        while iterations < max_iter_per_beta and time.monotonic() < phase_deadline:
            t += 1
            iterations += 1
            
            a = _softmax(z)
            grad, smax = _gradient_smooth_max(a, beta)
            
            # RMSProp in log-space

            log_grad = np.clip(log_grad, -100, 100)
            
            m_buf = 0.9 * m_buf + 0.1 * log_grad
            v_buf = 0.9 * v_buf + 0.1 * (log_grad * log_grad)
            
            lr = 0.5 / (1.0 + 0.001 * t)
            update = lr * m_buf / (np.sqrt(v_buf) + 1e-8)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

                update *= 0.8 / un

            z = z - update
            z = np.clip(z, -60, 60)
            z = z - z.mean()

            if t % 300 == 0:
                a_ch = _softmax(z)
                sc = _score_norm(a_ch)
                if sc < best_score:
                    best_score = sc
                    best_a = a_ch.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_phase(a_init, deadline, max_lp_steps=30):
    """LP-based optimization with gradient polishing."""
    from scipy.sparse import csr_matrix
    n = len(a_init)
    f = a_init.copy() / a_init.sum()
    best_f = f.copy()
    best_score = _score_norm(f)

    A_cache = _build_lp_matrix(f, n)
    f_for_matrix = f.copy()
    current_t = 0.4
    stuck = 0
    steps = 0

    while time.monotonic() < deadline - 10 and steps < max_lp_steps:
        c = _fft_convolve(f, f)
        M = float(c.max())

        if np.max(np.abs(f - f_for_matrix)) > 1e-8:
            A_cache = _build_lp_matrix(f, n)
            f_for_matrix = f.copy()

        g = _lp_step(f, A_cache, M, time_limit=20)

        if g is None:
            stuck += 1
            if stuck > 8:
                break
            # Try gradient step instead
            grad, _ = _gradient_smooth(f, M * 0.1, c)
            f_new = f - 0.01 * grad / np.maximum(f.sum(), 1e-15)
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            f = f_new
            continue

        stuck = 0
        g_norm = g / g.sum()

        improved = False
        t_try = current_t
        for _ in range(5):
            f_new = (1 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / fs
            M_new = float(_fft_convolve(f_new, f_new).max())
            if M_new < M * (1 - 1e-10):
                f = f_new
                current_t = min(1.0, t_try * 1.4)
                improved = True
                steps += 1
                sc = _score_norm(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                if steps % 3 == 0:
                    print(f"    LP {steps}: M={M:.10f}->{M_new:.10f} score={sc:.10f} t={time.monotonic()-t_start_global:.0f}s")
                break
            else:
                t_try *= 0.5

        if not improved:
            current_t *= 0.5
            stuck += 1
            if stuck > 15:
                break

        # Brief gradient polish between LP steps
        if time.monotonic() < deadline - 30:
            a_pol = _lp_result_polish(f, deadline=min(deadline - 10, time.monotonic() + 8))
            sc_pol = _score_norm(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_f = a_pol.copy()
                f = a_pol

    return best_f, best_score


def _lp_result_polish(a_init, deadline):
    """Quick gradient polish (up to deadline)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    c = _fft_convolve(a_init, a_init)
    beta = float(c.max()) * 0.01

    for _ in range(100000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _fft_convolve(a, a)
        grad, _ = _gradient_smooth(a, beta, c)
        lg = grad / np.maximum(a, 1e-15)
        lg = np.clip(lg, -100, 100)
        m = 0.9 * m + 0.1 * lg
        v = 0.9 * v + 0.1 * lg * lg
        lr = 0.2 / (1.0 + 0.001 * t)
        upd = lr * m / (np.sqrt(v) + 1e-8)
        un = np.linalg.norm(upd)
        if un > 0.5:
            upd *= 0.5 / un
        z = z - upd
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        if t % 500 == 0:
            sc = _score_norm(_softmax(z))
            if sc < best_sc:
                best_sc = sc
                best_a = _softmax(z).copy()
    return best_a


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_to_n(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: LP descent + annealed gradient + multi-restart + larger n.
    """
    global t_start_global
    t_start_global = time.monotonic()
    deadline = t_start_global + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _report(score, n, tag=""):
        nonlocal best_seq, best_score
        print(f"  {tag} score={score:.10f} n={n} t={time.monotonic()-t_start_global:.1f}s")

    def _try_update(p, tag=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _full_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) {tag} t={time.monotonic()-t_start_global:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        print(f"Parent: n={n_main}, score={_full_score(a0):.10f}")
        _try_update(a0, "parent")
    else:
        n_main = 8192
        a0 = np.ones(n_main) / n_main
        print(f"No parent, uniform n={n_main}")
        _try_update(a0, "uniform")

    a_norm = a0 / a0.sum()

    # === Phase 1: LP-based optimization (0 - 450s) ===
    t1_end = min(deadline, t_start_global + 450.0)
    print(f"\n=== Phase 1: LP-based optimization (n={n_main}) ===")
    if time.monotonic() < t1_end - 30:
        a_lp, sc_lp = _lp_phase(a_norm, t1_end, max_lp_steps=25)
        _try_update(a_lp, "LP")
        a_norm = a_lp / a_lp.sum()
    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")

    # === Phase 2: Annealed gradient descent (450s - 850s) ===
    t2_end = min(deadline, t_start_global + 850.0)
    print(f"\n=== Phase 2: Annealed gradient descent ===")
    if time.monotonic() < t2_end - 30:
        a_ag, sc_ag = _annealed_descent(a_norm, t2_end, n_temps=10, base_lr=0.25)
        _try_update(a_ag, "Annealed")
        a_norm = a_ag / a_ag.sum()
    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")

    # === Phase 3: More LP from polished result (850s - 1150s) ===
    t3_end = min(deadline, t_start_global + 1150.0)
    print(f"\n=== Phase 3: LP refinement ===")
    if best_seq is not None and time.monotonic() < t3_end - 30:
        a_norm3 = best_seq / best_seq.sum()
        a_lp3, _ = _lp_phase(a_norm3, t3_end, max_lp_steps=15)
        _try_update(a_lp3, "LP2")
    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")

    # === Phase 4: Multi-restart (1150s - 1450s) ===
    t4_end = min(deadline, t_start_global + 1450.0)
    print(f"\n=== Phase 4: Multi-restart ===")
    if best_seq is not None and time.monotonic() < t4_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 4
        per_restart = (t4_end - time.monotonic() - 20) / max(1, n_restarts)
        sigmas = [0.02, 0.04, 0.07, 0.12]

        for i in range(n_restarts):
            if time.monotonic() > t4_end - 20:
                break
            a_pert = _perturb_log(a_best_norm, sigmas[i], rng)
            dl = time.monotonic() + min(per_restart, t4_end - time.monotonic() - 5)
            a_r, _ = _annealed_descent(a_pert, dl, n_temps=6, base_lr=0.2)
            _try_update(a_r, f"Restart{i+1}")
    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")

    # === Phase 5: Larger n (1450s - 1700s) ===
    t5_end = min(deadline, t_start_global + 1700.0)
    print(f"\n=== Phase 5: Larger n exploration ===")
    if best_seq is not None and len(best_seq) < 16384 and time.monotonic() < t5_end - 60:
        target_n = 16384
        p_interp = _interp_to_n(best_seq / best_seq.sum(), target_n)
        _try_update(p_interp, "Interp16384")
        remaining = t5_end - time.monotonic() - 20
        if remaining > 60:
            a_r, _ = _annealed_descent(p_interp, time.monotonic() + remaining, n_temps=5, base_lr=0.2)
            _try_update(a_r, "Opt16384")
    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start_global:.1f}s")

    # === Phase 6: Final polish (1700s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            a_final, _ = _annealed_descent(a_best_norm, time.monotonic() + remaining, n_temps=3, base_lr=0.1)
            _try_update(a_final, "Final")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _full_score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start_global:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_full_score(fallback):.10f}")
    return [float(x) for x in fallback]
