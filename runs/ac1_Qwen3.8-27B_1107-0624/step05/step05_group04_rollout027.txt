"""
Multi-start multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses low-sidelobe window initializations, L-BFGS with log-sum-exp continuation,
and peak-aware local search with structured perturbations.

import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def score_norm_fn(a):
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def logsumexp_obj(z, beta):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def logsumexp_grad(z, beta):
    n = len(z)
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # h_i = sum_k w[k] * a[k-i] for valid k-i
    a_rev = a[::-1]
    w_conv = full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def make_flat_top_window(n, order=4):
    """Harris flat-top window of given order."""

def next_pow2(n):
    k = 1
    while k < n:

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)

    return e / s

            gap = sorted_c[0] - sorted_c[kk]
            if gap < 0.05 * cm:
                beta = max(1e-10, 0.05 * cm)
            elif gap > 0.5 * cm:
                beta = max(1e-10, 0.01 * cm)
            else:
                beta = max(1e-10, 0.3 * gap)

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1.0 - b1) * grad_z
        v = b2 * v + (1.0 - b2) * (grad_z * grad_z)
        mh = m / (1.0 - b1 ** t_step)
        vh = v / (1.0 - b2 ** t_step)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        a = softmax_fn(z)

        sc = score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t_step % 5000 == 0:
            lr = max(0.003, lr * 0.85)
        if t_step % 30000 == 0:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()
            z = np.clip(z, -70.0, 70.0)

    return best_a, best_score


def peak_aware_search(a_init, deadline, rng, max_rounds=2000):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()
    no_improve = 0

    dl_values = [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = softmax_fn(z)
        c = full_convolve(a_cur, a_cur)
        cm = float(c.max())
        top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for pk in top_peaks:
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            if lo > hi:
                continue
            valid_i = np.arange(lo, hi + 1)
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            if prod.max() < 1e-20:
                continue
            order = np.argsort(prod)[::-1]
            for pos in order[:5]:
                idx = valid_i[pos]
                for delta in dl_values:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    small_idx = int(np.argmin(a_cur))
                    z_trial[small_idx] += delta * 0.5
                    a_trial = softmax_fn(z_trial)
                    sc = score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            small_idx = np.argsort(a_cur)[:3]
            for idx in small_idx:
                for delta in [0.003, 0.008, 0.015]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = softmax_fn(z_trial)
                    sc = score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 80:
                z = z + 0.04 * rng.normal(size=n)
                z = z - z.mean()
                z = np.clip(z, -60.0, 60.0)
                no_improve = 0

    return best_a, best_score


def run_cyclical_subgradient(a_init, deadline, rng):
    """Cyclical subgradient: cycle through convolution positions, reduce peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    a = a_init.copy()
    lr = 0.01
    total_iters = 0

    for epoch in range(200):
        if time.monotonic() > deadline:
            break
        c = full_convolve(a, a)
        target = float(np.median(c)) * 1.1
        peak_mask = c > target
        num_peaks = int(peak_mask.sum())
        if num_peaks == 0:
            break

        for _ in range(min(num_peaks, 20)):
            if time.monotonic() > deadline:
                break
            ck = full_convolve(a, a)
            peak_k = int(np.argmax(ck))
            if ck[peak_k] <= target:
                break
            lo = max(0, peak_k - n + 1)
            hi = min(n - 1, peak_k)
            if lo > hi:
                break
            valid_i = np.arange(lo, hi + 1)
            grad = 2.0 * a[peak_k - valid_i]
            # Normalize gradient
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-20:
                break
            grad = grad / gnorm
            # Update
            step = lr * ck[peak_k]
            a[lo:hi + 1] -= step * grad
            a[valid_i[0]] += step * grad.sum() / len(valid_i)
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s
            total_iters += 1
            if total_iters % 100 == 0:
                sc = score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        lr *= 0.95

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s_parent = parent_raw.sum()
            if s_parent > 0.01:
                parent = parent_raw / s_parent
    except Exception:
        parent = None

    if parent is not None:
        sc_p = score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = make_flat_top_window(256, 4)
        update_best(a0, score_norm_fn(a0), "default_flat_top")
        best_seq = a0.copy()
        best_score = score_norm_fn(a0)

    # Phase 1: Diverse initializations at n=128, quick optimize, pick top
    t_phase1 = t_start + (deadline - t_start) * 0.08
    print(f"\nPhase 1: Diverse init at n=128 (deadline t={t_phase1 - t_start:.0f}s)")

    n_init = 128
    profiles = make_diverse_profiles(n_init, rng)
    quick_results = []
    t_per_profile = 5.0

    for idx, prof in enumerate(profiles):
        if time.monotonic() > t_phase1:
            break
        sc = score_norm_fn(prof)
        t_prof_deadline = min(t_phase1, time.monotonic() + t_per_profile)
        a_opt, sc_opt = run_lbfgs_continuation(prof, t_prof_deadline, n_betas=6)
        quick_results.append((sc_opt, a_opt, f"profile_{idx}"))
        print(f"    Profile {idx}: {sc_opt:.10f}")

    quick_results.sort(key=lambda x: x[0])
    print(f"  Top 3: {[f'{r[0]:.8f}' for r in quick_results[:3]]}")

    # Phase 2: Multi-scale up for top 3 profiles
    t_phase2 = t_start + (deadline - t_start) * 0.72
    print(f"\nPhase 2: Multi-scale up (deadline t={t_phase2 - t_start:.0f}s)")

    scales = [256, 512, 1024, 2048, 4096]
    scale_times = {256: 30, 512: 60, 1024: 120, 2048: 180, 4096: 120}

    # Use parent's n to determine final target
    final_n = 4096
    if parent is not None:
        final_n = len(parent)

    for top_idx in range(min(3, len(quick_results))):
        sc_cur, a_cur, label = quick_results[top_idx]
        print(f"\n  Track {top_idx} ({label}): starting n={n_init}, score={sc_cur:.10f}")

        for target_n in scales:
            if target_n > final_n:
                break
            if time.monotonic() > t_phase2 - 10:
                break
            a_cur = upsample(a_cur, target_n)
            alloc = scale_times.get(target_n, 60)
            # Adjust time allocation for multiple tracks
            if top_idx > 0:
                alloc = int(alloc * 0.5)
            opt_deadline = min(t_phase2 - 5, time.monotonic() + alloc)
            a_opt, sc_opt = run_lbfgs_continuation(a_cur, opt_deadline, n_betas=12)
            print(f"    n={target_n}: {sc_opt:.10f}")
            if sc_opt < sc_cur:
                sc_cur = sc_opt
                a_cur = a_opt

        if sc_cur < best_score:
            update_best(a_cur, sc_cur, f"track_{top_idx}_end")
        else:
            a_cur = best_seq.copy()
            sc_cur = best_score

    print(f"\n  Phase 2 complete: best={best_score:.10f}")

    # Phase 3: Extended optimization at final scale
    t_phase3 = deadline - 30.0
    print(f"\nPhase 3: Extended optimization at n={len(best_seq)} (deadline t={t_phase3 - t_start:.0f}s)")

    if best_seq is not None:
        n_best = len(best_seq)

        # 3a: Fine L-BFGS continuation
        t_3a = t_start + (t_phase3 - t_start) * 0.35
        alloc_3a = min(t_3a, time.monotonic() + 300.0)
        a_3a, sc_3a = run_lbfgs_continuation(best_seq, alloc_3a, n_betas=25)
        update_best(a_3a, sc_3a, "fine_lbfgs")

        # 3b: Adam GD
        t_3b = t_start + (t_phase3 - t_start) * 0.60
        if time.monotonic() < t_3b:
            alloc_3b = min(t_3b, time.monotonic() + 200.0)
            a_3b, sc_3b = run_adam_gd(best_seq, alloc_3b, rng, lr_init=0.03)
            update_best(a_3b, sc_3b, "adam_gd")

        # 3c: Peak-aware search
        t_3c = t_start + (t_phase3 - t_start) * 0.85
        if time.monotonic() < t_3c:
            alloc_3c = min(t_3c, time.monotonic() + 150.0)
            a_3c, sc_3c = peak_aware_search(best_seq, alloc_3c, rng, max_rounds=3000)
            update_best(a_3c, sc_3c, "peak_search")

        # 3d: Cyclical subgradient
        if time.monotonic() < deadline - 40:
            alloc_3d = min(deadline - 40, time.monotonic() + 60.0)
            a_3d, sc_3d = run_cyclical_subgradient(best_seq, alloc_3d, rng)
            update_best(a_3d, sc_3d, "cyclical_sg")

        # 3e: One more Adam round from best
        if time.monotonic() < deadline - 20:
            alloc_3e = min(deadline - 20, time.monotonic() + 40.0)
            a_3e, sc_3e = run_adam_gd(best_seq, alloc_3e, rng, lr_init=0.005)
            update_best(a_3e, sc_3e, "adam_fine")

    # Phase 4: Final tiny-beta L-BFGS polish
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = len(best_seq)
        c_final = full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = max(1e-8, 1e-5 * max_c_final)
        try:
            result = minimize(
                logsumexp_obj, z, args=(beta_tiny,),
                method='L-BFGS-B', jac=logsumexp_grad,
                options={'maxiter': 30000, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxls': 50, 'maxfun': 100000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax_fn(result.x)
                sc_new = score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Tiny-beta polish: {best_score:.10f}")
        except Exception as ex:
            print(f"  Tiny-beta failed: {ex}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = make_flat_top_window(256, 4)
        sc_final = score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = make_flat_top_window(256, 4)
    print(f"FALLBACK: n=256, score={score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
