"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f, timeout=60):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, max_time=timeout)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
            else:
                return None, None
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters, deadline, step_size):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        g, t_val = _lp_minimax(f, timeout=30)
        if g is None:
            break
        f_new = (1.0 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    return best_a, best_score


def _peak_shaving(a, deadline, n_rounds=200):
    """Responsibility-based peak shaving: identify peaks, compute contributions,
    and reduce responsible coefficients while redistributing mass."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a_cur = a.copy()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        if cm < 1e-15:
            break
        # Find indices near the peak
        threshold = 0.95 * cm
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) == 0:
            break

        # Compute responsibility: for each a_j, how much does it contribute to peaks?
        resp = np.zeros(n)
        for k in peak_idx:
            # c[k] = sum_j a[j]*a[k-j] for valid j
            # Contribution of a[j] to c[k] is a[j]*a[k-j]
            for j in range(n):
                k_j = k - j
                if 0 <= k_j < n:
                    resp[j] += a_cur[j] * a_cur[k_j]

        # Total responsibility
        total_resp = resp.sum()
        if total_resp < 1e-15:
            break

        # Reduce high-responsibility coefficients, increase low-responsibility ones
        # Use a soft approach: a_new = a * (1 - alpha * (resp/total_resp - 1/n))
        alpha = 0.1
        factor = 1.0 - alpha * (resp / total_resp - 1.0 / n)
        factor = np.maximum(factor, 0.1)
        a_new = a_cur * factor
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a_cur = a_new

        # Decay alpha
        alpha = max(0.01, alpha * 0.98)

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005,
                 0.00003, 0.00002, 0.000015, 0.00001]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        maxiter = max(100, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxls': 30, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _subgradient_linesearch(a_init, deadline, max_iter=5000):
    """Exact subgradient method with backtracking line search on the max."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a_cur = a_init.copy()
    z = np.log(np.maximum(a_cur, 1e-15))
    z = z - z.mean()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        score = 2.0 * n * cm

        # Find the peak index (or indices near peak)
        peak_thresh = 0.999 * cm
        peak_set = set(np.where(c >= peak_thresh)[0].tolist())
        peak_list = list(peak_set)
        if not peak_list:
            break

        # Subgradient: average of gradients at peak indices
        # grad_j = sum_{k in peaks} (1/|peaks|) * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        for k in peak_list:
            for j in range(n):
                k_j = k - j
                if 0 <= k_j < n:
                    grad[j] += a[k_j]
        grad /= len(peak_list)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Backtracking line search in z-space
        # z_new = z - lr * grad (projected to maintain softmax normalization)
        lr = 0.1
        best_z_trial = z.copy()
        best_score_trial = score
        for bt in range(20):
            z_trial = z - lr * grad
            z_trial = np.clip(z_trial, -60, 60)
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax_fn(z_trial)
            sc_trial = _score_norm_fn(a_trial)
            if sc_trial < best_score_trial * (1 - 0.001):
                best_z_trial = z_trial.copy()
                best_score_trial = sc_trial
                break
            lr *= 0.7

        if best_score_trial < best_score:
            z = best_z_trial
            a_cur = _softmax_fn(z)
            best_score = best_score_trial
            best_a = a_cur.copy()
        else:
            # Stagnation: try smaller lr or break
            if lr < 1e-6:
                break

    return best_a, best_score


def _fine_coord_descent(a_init, deadline, max_rounds=100):
    """Very fine coordinate descent with small steps in z-space."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    # Adaptive step sizes
    base_steps = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        improved = False

        # Focus on coordinates that contribute to the peak
        cm = float(c.max())
        # Compute "importance" of each coordinate
        peak_idx = np.where(c >= 0.9 * cm)[0]
        import_scores = np.zeros(n)
        for k in peak_idx:
            for j in range(n):
                k_j = k - j
                if 0 <= k_j < n:
                    import_scores[j] += a[j] * a[k_j]

        # Process high-importance coordinates first
        order = np.argsort(-import_scores)
        n_try = min(n, 200)  # Only try top 200 per round

        for jj in range(n_try):
            if time.monotonic() > deadline:
                break
            j = order[jj]
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in base_steps:
                for sign in [1, -1]:
                    z_trial = z.copy()
                    z_trial[j] += sign * delta
                    z_trial = z_trial - z_trial.mean()
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_local:
                        best_local = sc
                        best_delta = sign * delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_score

def _spectral_peak_reduction(a, deadline, n_iters=100):
    """Fedorov-style spectral peak reduction for aperiodic convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a_cur = a.copy()
    N = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        # Compute spectrum
        A = np.fft.rfft(a_cur, N)
        # Compute time-domain convolution via spectrum
        C_spec = A * A
        c = np.fft.irfft(C_spec, N)[:2 * n - 1]
        cm = float(c.max())
        if cm < 1e-15:
            break

        # Find peak
        k_star = int(np.argmax(c))

        # In frequency domain, c[k] = (1/N) * sum_omega |A(omega)|^2 * exp(i*omega*k)
        # To reduce c[k_star], reduce |A|^2 where exp(i*omega*k_star) is positive
        freqs = np.fft.rfftfreq(N) * N  # in units of "cycles"
        # The phase at frequency omega for lag k_star is omega * k_star * 2*pi/N
        phase = 2.0 * np.pi * freqs * k_star / N
        cos_phase = np.cos(phase)

        # Create mask: reduce components that constructively interfere at k_star
        alpha = 0.05
        mask = 1.0 - alpha * np.maximum(cos_phase, 0)
        mask = np.sqrt(np.maximum(mask, 0.01))

        # Apply mask to spectrum magnitude
        A_new = A * mask
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_cur = a_new
        else:
            # Reduce alpha
            alpha = max(0.01, alpha * 0.95)

    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    print(f"Starting with n={n_parent}")

    # === Phase 1: LP Iterations (300s) ===
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\nPhase 1: LP iterations")

    # Work at n=2048 for speed
    n_lp = min(n_parent, 2048)
    if n_parent > n_lp:
        a_lp = _interp_profile(parent, n_lp)
    else:
        a_lp = parent.copy()
    update_best(a_lp, _score_norm_fn(a_lp), f"lp_init_n{n_lp}")

    a_cur = a_lp.copy()
    for lp_round in range(8):
        if time.monotonic() > phase1_end:
            break
        step = 0.8 / (lp_round + 1)
        a_cur, sc_cur = _lp_iterate(a_cur, n_iters=15, deadline=phase1_end, step_size=step)
        update_best(a_cur, sc_cur, f"lp_r{lp_round}_n{n_lp}")

    # Interpolate back
    if n_parent > n_lp:
        a_rec = _interp_profile(a_cur, n_parent)
        sc_rec = _score_norm_fn(a_rec)
        update_best(a_rec, sc_rec, f"lp_interp_n{n_parent}")
    else:
        a_rec = a_cur.copy()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Spectral Peak Reduction (150s) ===
    phase2_end = min(deadline, t_start + 450.0)
    print(f"\nPhase 2: Spectral peak reduction (n={len(best_seq)})")

    if time.monotonic() < phase2_end:
        a_spec_start = best_seq.copy() if len(best_seq) <= 4096 else _interp_profile(best_seq, 4096)
        per_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.8)
        p_best, sc_best = _spectral_peak_reduction(a_spec_start, per_dl, n_iters=200)
        update_best(p_best, sc_best, "spectral")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak Shaving (150s) ===
    phase3_end = min(deadline, t_start + 600.0)
    print(f"\nPhase 3: Peak shaving (n={len(best_seq)})")

    if time.monotonic() < phase3_end:
        a_shave_start = best_seq.copy()
        per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.8)
        p_best, sc_best = _peak_shaving(a_shave_start, per_dl, n_rounds=300)
        update_best(p_best, sc_best, "peak_shaving")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: L-BFGS Continuation (350s) ===
    phase4_end = min(deadline, t_start + 950.0)
    print(f"\nPhase 4: L-BFGS continuation")

    n_restarts = 4
    for r in range(n_restarts):
        if time.monotonic() > phase4_end - 30:
            break
        if r == 0:
            a_cur4 = best_seq.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.03 * (r + 1)
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur4 = _softmax_fn(z)

        per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) / (n_restarts - r))
        p_best, sc_best = _run_lbfgs_continuation(a_cur4, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}_n{len(a_cur4)}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Subgradient + Line Search (200s) ===
    phase5_end = min(deadline, t_start + 1150.0)
    print(f"\nPhase 5: Subgradient line search (n={len(best_seq)})")

    if time.monotonic() < phase5_end:
        per_dl = min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic()) * 0.85)
        p_best, sc_best = _subgradient_linesearch(best_seq, per_dl, max_iter=10000)
        update_best(p_best, sc_best, "subgrad_ls")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Fine Coordinate Descent (200s) ===
    phase6_end = min(deadline, t_start + 1350.0)
    print(f"\nPhase 6: Fine coordinate descent (n={len(best_seq)})")

    if time.monotonic() < phase6_end:
        per_dl = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic()) * 0.9)
        p_best, sc_best = _fine_coord_descent(best_seq, per_dl, max_rounds=200)
        update_best(p_best, sc_best, "fine_coord")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Random Perturbation Search (150s) ===
    phase7_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 7: Random perturbation search (n={len(best_seq)})")

    if time.monotonic() < phase7_end:
        per_dl = min(phase7_end - 10, time.monotonic() + (phase7_end - time.monotonic()) * 0.9)
        p_best, sc_best = _random_perturb_search(best_seq, per_dl, rng, n_trials=100, noise_scale=0.01)
        update_best(p_best, sc_best, "rand_perturb")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Final polish - alternate fine methods on remaining time ===
    phase8_end = deadline
    print(f"\nPhase 8: Final polish")

    if time.monotonic() < phase8_end and best_seq is not None:
        # One more round of fine coordinate descent
        per_dl = min(phase8_end - 30, time.monotonic() + 60.0)
        p_best, sc_best = _fine_coord_descent(best_seq, per_dl, max_rounds=50)
        update_best(p_best, sc_best, "final_coord")

        # One more subgradient
        if time.monotonic() < phase8_end - 30:
            per_dl = min(phase8_end - 20, time.monotonic() + 40.0)
            p_best, sc_best = _subgradient_linesearch(best_seq, per_dl, max_iter=5000)
            update_best(p_best, sc_best, "final_subgrad")

        # Tiny L-BFGS
        if time.monotonic() < phase8_end - 20:
            n_final = len(best_seq)
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_tiny = 2e-6 * max_c_final
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            per_dl = min(phase8_end - 10, time.monotonic() + 30.0)
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta_tiny, n_final),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-15,
                             'maxls': 30, 'maxfun': 15000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_seq = a_new.copy()
                        print(f"  FINAL L-BFGS: {best_score:.10f}")
            except Exception:
                pass

    print(f"  Phase 8 done: best={best_score:.10f}")

    # === Final Validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 8000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
