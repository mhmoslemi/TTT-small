"""
Hybrid optimizer combining direct subgradient descent with L-BFGS log-sum-exp
smoothing for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive active sets, line search, Nesterov acceleration, and diverse
initializations across multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _subgrad_direct(a, c, tol_frac):
    """Compute subgradient using active set. g[j] = 2*sum_k w[k]*a[k-j] for valid k-j."""
    n = len(a)
    c_max = float(c.max())
    if c_max < 1e-15:
        return np.zeros(n)
    active = c >= c_max * (1.0 - tol_frac)
    if active.sum() == 0:
        active = np.zeros(len(c), dtype=bool)
        active[int(np.argmax(c))] = True
    count = float(active.sum())
    # g[j] = (2/count) * sum_{k in active} a[k-j] where 0 <= k-j < n
    # Use FFT: g[j] = (2/count) * sum_m a[m] * v[m+j] where v[k]=1/count if active
    # Let's compute h = ifft(conj(fft(a_padded)) * fft(v_padded)) which gives cross-correlation
    N = _next_pow2(2 * n + 1)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    v = np.zeros(N)
    v[active] = 1.0 / count
    A = np.fft.rfft(a_pad)
    V = np.fft.rfft(v)
    # cross_corr[j] = sum_m a[m] * v[m+j] (linear, no aliasing with this padding)
    # = ifft(conj(A) * V) ... let me verify
    # ifft(conj(A) * V)[j] = sum_m a[m] * v[(m-j) mod N] -- this is circular
    # For linear: we need v[m+j], not v[m-j]
    # ifft(A * conj(V))[j] = sum_m a[(m+j) mod N] * v[m] -- also circular but different direction
    # Let me use: g[j] = sum_k v[k] * a[k-j] = sum_k v[k] * a[k-j]
    # Let m = k-j: g[j] = sum_m v[m+j] * a[m]
    # This is the cross-correlation (a star v)[j] = sum_m a[m] * v[m+j]
    # In FFT with zero-padding to N >= 3n: 
    # cross = ifft(conj(fft(a_pad, N)) * fft(v_pad, N)) -- no, this gives sum a[m]*v[m-j]
    # We need sum a[m]*v[m+j] = sum v[m]*a[m+j] = conj(circ_corr at -j) for real
    # Actually for real sequences: sum_m a[m]*v[m+j] = ifft(fft(a) * conj(fft(v)))[-j mod N]
    # = ifft(fft(a) * conj(fft(v)))[N-j] (for j > 0)
    # Let me just compute it directly for the active set (usually small)
    g = np.zeros(n)
    active_idx = np.where(active)[0]
    for k in active_idx:
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # g[j] += (2/count) * a[k-j] for j in [j_min, j_max]
            # a[k-j] for j=j_min..j_max is a[k-j_max]..a[k-j_min]
            g[j_min:j_max + 1] += a[k - j_max:k - j_min + 1][::-1]
    return 2.0 * g / count

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_phase(a_init, beta, n, max_iter, ftol=1e-14):
    """Run L-BFGS-B for one beta phase."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': ftol,
                'gtol': 1e-10,
                'maxcor': 30,
                'maxls': 80,
                'disp': False,
            }
        )
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init

def _subgrad_nesterov(a_init, deadline, max_steps=2000, tol_frac_init=0.005, tol_frac_final=0.0005):
    """Nesterov-accelerated subgradient with line search in original space."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a_init
    a = a / s
    best_a = a.copy()
    best_sc = _score(a)
    
    x_prev = a.copy()
    t = 1.0
    eta = 0.1
    failed_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        # Adaptive tolerance
        tol_frac = tol_frac_init + (tol_frac_final - tol_frac_init) * (i / max_steps)

        # Nesterov momentum
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t_prev - 1.0) / t
        y = a + momentum * (a - x_prev)
        y = np.maximum(y, 0.0)
        y_s = y.sum()
        if y_s < 1e-15:
            y = a.copy()
        else:
            y = y / y_s

        c = _full_convolve(y, y)
        g = _subgrad_direct(y, c, tol_frac)
        grad_norm = np.linalg.norm(g)
        if grad_norm < 1e-12:
            break

        # Line search (backtracking)
        found = False
        for ls_iter in range(20):
            cand = np.maximum(a - eta * g, 0.0)
            cs = cand.sum()
            if cs < 1e-15:
                eta *= 0.5
                continue
            cand = cand / cs
            sc = _score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
            if sc < _score(a) - 1e-10 * _score(a):
                a = cand
                found = True
                break
            eta *= 0.5

        x_prev = a.copy()
        if not found:
            failed_count += 1
            if failed_count > 100:
                break
        else:
            failed_count = max(0, failed_count - 1)
            if i % 50 == 0 and i > 0:
                eta = min(eta * 1.1, 0.3)

    return best_a, best_sc

def _lbfgs_schedule(a_start, deadline, min_time_remaining=30.0):
    """Run L-BFGS beta-continuation schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        est_time_per_iter = 0.004 * (n / 1024.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))
        a_new = _lbfgs_phase(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc

    return current_a

def _make_init(n, kind, rng):
    """Create various initialization profiles."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "geometric":
        r = rng.uniform(0.9, 0.99)
        a = r ** np.arange(n)
        return a / a.sum()
    elif kind == "ramp":
        a = np.arange(1, n + 1, dtype=float)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        a = 1.0 + 0.4 * np.cos(t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        a = 1.0 + 0.35 * np.cos(2 * t)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.15, 0.35)
        a = np.exp(-((t - center) / width) ** 2)
        return a / a.sum()
    elif kind == "two_level":
        a = np.ones(n)
        split = n // 2
        a[:split] = rng.uniform(1.5, 3.0)
        return a / a.sum()
    elif kind == "random":
        a = np.abs(rng.standard_normal(n))
        return a / a.sum()
    elif kind == "exponential_tail":
        a = np.exp(-0.5 * np.linspace(0, 3, n))
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):

    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

        max_cb = float(c_best.max())
        n = len(best_seq)
        fine_betas = max_cb * np.array([0.0004, 0.0002, 0.0001, 0.00005, 0.00002])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 80:
                break
            remaining = deadline - 80 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(3000, max(100, int(remaining / 0.03)))
            a_new = _lbfgs_phase(best_seq, beta, n, max_iter)
            _update(a_new, f"fine_{i}")

    # --- Phase 4: Active set polishing ---
    print(f"\nPhase 4: Active-set polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_pol, sc_pol = _active_set_polish(best_seq, deadline - 60, n_steps=2000, tol_frac=0.0005)
        _update(a_pol, "polish")

    # --- Phase 5: Diverse restarts ---
    print(f"\nPhase 5: Diverse restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        n_restarts = min(4, max(1, int(remaining / 120)))
        n_best = len(best_seq)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 35:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 2:
                # Fourier perturbation
                z = np.log(np.maximum(best_seq, 1e-15))
                t = np.arange(n_best)
                freq = rng.integers(1, max(2, n_best // 8))
                amp = 0.03 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * freq * t / n_best + phase)
                z = z - z.mean()
                p_start = _softmax(z)
            else:
                p_start = _perturb(best_seq, 0.01, rng)

            sc_start = _score(p_start)
            print(f"  Restart {i}: start_score={sc_start:.10f}")

            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=40.0)
            _update(a_r, f"restart_{i}")

            if time.monotonic() < deadline - 25:
                a_r2, _ = _active_set_polish(a_r, deadline - 25, n_steps=500, tol_frac=0.001)
                _update(a_r2, f"restart_{i}_polish")

    # --- Phase 6: Try different n values ---
    print(f"\nPhase 6: Different n (t={time.monotonic() - t_start:.1f}s)")
    try_ns = [1024, 4096]
    if time.monotonic() < deadline - 20:
        for n_try in try_ns:
            if time.monotonic() > deadline - 15:
                break
            # Resample best to new length
            if best_seq is not None:
                from numpy import interp
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n_try)
                a_resampled = interp(x_new, x_old, best_seq)
                a_resampled = np.maximum(a_resampled, 0.0)
                a_resampled = a_resampled / a_resampled.sum()
            else:
                a_resampled = np.ones(n_try) / n_try

            sc_rs = _score(a_resampled)
            print(f"  n={n_try}: resampled score={sc_rs:.10f}")
            _update(a_resampled, f"resample_n{n_try}")

            # Quick L-BFGS
            if time.monotonic() < deadline - 15 and sc_rs < best_score + 0.1:
                a_opt = _lbfgs_schedule(a_resampled, deadline, min_time_remaining=15.0)
                _update(a_opt, f"opt_n{n_try}")

    # --- Phase 7: Final polish ---
    print(f"\nPhase 7: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # One more L-BFGS with tiny beta
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 0.00003
        if time.monotonic() < deadline - 3:
            a_fin = _lbfgs_phase(best_seq, beta_final, n, max_iter=1000)
            _update(a_fin, "final_lbfgs")

        # Final active set
        if time.monotonic() < deadline - 2:
            a_fin2, _ = _active_set_polish(best_seq, deadline - 2, n_steps=500, tol_frac=0.0003)
            _update(a_fin2, "final_polish")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
