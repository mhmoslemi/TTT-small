"""
L-BFGS with adaptive log-sum-exp smoothing and multi-restart strategy
for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses analytic gradient in log-space with FFT-accelerated convolution.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _pow2(n):
    return 1 << (n - 1).bit_length()


def _convolve_fft(a, b):
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _pow2(out_len)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_len]


def _score_from_a(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _convolve_fft(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_and_grad_log(x, beta):
    """Compute score and gradient in log-space using log-sum-exp smoothing."""
    n = len(x)
    a = np.exp(x)
    s = a.sum()
    if s < 1e-15:
        return np.inf, np.zeros_like(x)

    c = _convolve_fft(a, a)

    # Log-sum-exp soft max
    c_max = c.max()
    shifted = (c - c_max) / beta
    exp_shifted = np.exp(shifted)
    exp_sum = exp_shifted.sum()
    if exp_sum < 1e-300 or not np.isfinite(exp_sum):
        F = c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        F = c_max + beta * np.log(exp_sum)
        w = exp_shifted / exp_sum

    # h_j = sum_m a[m] * w[m+j] for m=0..n-1
    # = conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_aw = _convolve_fft(a_rev, w)
    h = conv_aw[n - 1:2 * n - 1]

    score = 2.0 * n * F / (s * s)

    # dScore/dx_j = (4*n*a[j]/s^3) * (s*h[j] - F)
    grad = (4.0 * n * a / (s * s * s)) * (s * h - F)

    grad = grad + 2.0 * reg_coeff * (x - x.mean())
    score = score + reg_coeff * np.sum((x - x.mean()) ** 2)

    return score, grad

        sc, g = _score_and_grad_log(x, beta)
        if not np.isfinite(sc):
            return 1e10, np.zeros_like(x)
        return sc, g

    try:
        result = minimize(obj_grad, x_init, method='L-BFGS-B',

                                   'gtol': 1e-9, 'maxfun': maxfun,
                                   'maxls': 30, 'disp': False})

        if time.monotonic() > deadline - 120:
            break
        t0 = time.monotonic()
        x, nit = _lbfgs_optimize(x, beta, deadline, maxiter=400, maxfun=800)
        a_new = np.exp(x - x.mean())
        sc_new = _score_from_a(a_new)
        dt = time.monotonic() - t0
        improved = ""
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()
            improved = " *"
        print(f"  beta={beta:.6f}: score={sc_new:.10f} iters={nit} dt={dt:.1f}s{improved}")

    # === Phase 2: Multi-restart with perturbations ===
    print(f"\n--- Phase 2: Multi-Restart ---")
    n_restarts = 8
    for ri in range(n_restarts):
        if time.monotonic() > deadline - 80:
            break

        sigma = 0.02 + 0.04 * ri / n_restarts
        x_pert = np.log(best_seq + 1e-30) + sigma * rng.standard_normal(n)
        x_pert = x_pert - x_pert.mean()

        for beta in [0.002, 0.0005, 0.0001]:
            if time.monotonic() > deadline - 20:
                break
            x_pert, _ = _lbfgs_optimize(x_pert, beta, deadline, maxiter=200, maxfun=400)

        a_r = np.exp(x_pert - x_pert.mean())
        sc_r = _score_from_a(a_r)
        improved = ""
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            improved = " *"
        print(f"  Restart[{ri}] sigma={sigma:.3f}: score={sc_r:.10f}{improved}")

    # === Phase 3: Try different sequence lengths ===
    print(f"\n--- Phase 3: Different Lengths ---")
    for n_try in [1024, 4096, 8192]:
        if time.monotonic() > deadline - 60:
            break
        if n_try == n:
            continue

        # Interpolate best sequence to new length
        x0 = np.linspace(0, 1, n)
        x1 = np.linspace(0, 1, n_try)
        a_interp = np.interp(x1, x0, best_seq)
        a_interp = np.maximum(a_interp, 1e-12)
        a_interp = a_interp / a_interp.sum()

        x_interp = np.log(a_interp)
        x_interp = x_interp - x_interp.mean()

        for beta in [0.005, 0.001, 0.0002]:
            if time.monotonic() > deadline - 15:
                break
            x_interp, _ = _lbfgs_optimize(x_interp, beta, deadline, maxiter=200, maxfun=400)

        a_try = np.exp(x_interp - x_interp.mean())
        sc_try = _score_from_a(a_try)
        improved = ""
        if sc_try < best_score:
            best_score = sc_try
            best_seq = a_try.copy()
            n = n_try
            improved = " *"
        print(f"  n={n_try}: score={sc_try:.10f}{improved}")

    # === Phase 4: Focused refinement on best ===
    print(f"\n--- Phase 4: Focused Refinement ---")
    x_best = np.log(best_seq + 1e-30)
    x_best = x_best - x_best.mean()

    fine_betas = [0.0001, 0.00005, 0.00002, 0.00001]
    for beta in fine_betas:
        if time.monotonic() > deadline - 15:
            break
        t0 = time.monotonic()
        x_best, nit = _lbfgs_optimize(x_best, beta, deadline, maxiter=500, maxfun=1000)
        a_fine = np.exp(x_best - x_best.mean())
        sc_fine = _score_from_a(a_fine)
        dt = time.monotonic() - t0
        improved = ""
        if sc_fine < best_score:
            best_score = sc_fine
            best_seq = a_fine.copy()
            improved = " *"
        print(f"  Fine beta={beta:.7f}: score={sc_fine:.10f} iters={nit} dt={dt:.1f}s{improved}")

    # === Phase 5: Subgradient polishing (exact max) ===
    print(f"\n--- Phase 5: Subgradient Polishing ---")
    if time.monotonic() < deadline - 10:
        a_cur = best_seq.copy()
        n_cur = len(a_cur)
        lr_sub = 0.1 / n_cur
        for sub_iter in range(200):
            if time.monotonic() > deadline - 5:
                break
            c = _convolve_fft(a_cur, a_cur)
            c_max = c.max()
            # Find all lags within 1% of max (active set)
            active = c >= 0.99 * c_max
            w_sub = active.astype(np.float64)
            w_sub /= w_sub.sum()

            # Subgradient: g_j = (2*a_j/s) * h_j where h = cross-correlation
            s_cur = a_cur.sum()
            a_rev = a_cur[::-1]
            conv_aw = _convolve_fft(a_rev, w_sub)
            h = conv_aw[n_cur - 1:2 * n_cur - 1]
            g = 2.0 * a_cur * h / s_cur

            # Update in log-space
            x_sub = np.log(a_cur + 1e-30)
            x_sub -= lr_sub * g / (a_cur + 1e-30)
            x_sub = x_sub - x_sub.mean()
            a_candidate = np.exp(x_sub)
            sc_candidate = _score_from_a(a_candidate)

            if sc_candidate < _score_from_a(a_cur):
                a_cur = a_candidate
                if sc_candidate < best_score:
                    best_score = sc_candidate
                    best_seq = a_cur.copy()

        sc_polish = _score_from_a(a_cur)
        print(f"  Subgradient final: score={sc_polish:.10f}")

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score_from_a(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
