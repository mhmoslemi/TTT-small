"""
Multi-phase multiplicative gradient descent with adaptive peak focusing
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses softmax-smoothed max gradient, multiplicative updates, FFT-based
gradient computation, flat-spectrum diversification, and multi-length search.
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1. Returns 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _compute_grad_fft(a, c, beta):
    """Compute the smoothed gradient of max(c) w.r.t. a."""
    n = len(a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    shifted = np.clip(shifted, -60, 0)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # g[m] = 2 * sum_k w[k] * a[k-m] = 2 * cross_corr(w, a)[m]
    # cross_corr(w, a)[m] = ifft(fft(w) * conj(fft(a)))[m]
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    g = 2.0 * corr[:n]
    return g


def _compute_grad_topk(a, c, K):
    """Compute gradient focusing on top-K peaks."""
    n = len(a)
    # Get top-K indices
    if K >= len(c):
        topk = np.argsort(c)[::-1]
    else:
        # Use partition for efficiency
        part = np.argpartition(c, -K)[-K:]
        topk = part[np.argsort(c[part])[::-1]]
    
    w = np.zeros_like(c)
    w[topk] = 1.0 / K
    
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    g = 2.0 * corr[:n]
    return g


def _multiplicative_update(a, g, eta):
    """Apply multiplicative (exponentiated gradient) update."""
    # a_new = a * exp(-eta * g)
    # Clip to avoid underflow
    log_update = -eta * g
    log_update = np.clip(log_update, -20, 5)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a_new / s


def _flat_spectrum_update(a, power=0.7, n_iter=1):
    """Apply flat-spectrum update to diversify."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = A / (mag + 1e-30)
    # Compress dynamic range
    mag_new = mag ** power
    A_new = mag_new * phase
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def _interp(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.arange(n_old) / n_old
    x1 = np.arange(n_new) / n_new
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _perturb(a, sigma, rng):
    """Perturb log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()
