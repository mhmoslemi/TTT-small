"""
Peak-repulsion + L-BFGS hybrid optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses multiplicative update based on KKT-inspired peak repulsion,
combined with L-BFGS soft-max refinement and LP peak balancing.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _cross_corr(a, w):
    """Compute h[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1.
    a has length n, w has length 2n-1."""
    n = len(a)
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(A * np.conj(W), N)[:n]
    return R


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _peak_repulsion(a_init, deadline, alpha=1.0, T_init=None, cooling=0.9999,
                    min_T=None, max_iter=1000000, check_interval=10):
    """
    KKT-inspired multiplicative peak repulsion.
    Update: a_i <- a_i / h_i^alpha where h_i = sum_k w_k * a_{k-i}
    w_k = softmax(c_k / T) are adaptive peak weights.
    """
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    if S < 1e-15:
        return a_init, _score_norm_fn(a_init)
    a = a / S
    
    c = _full_convolve(a, a)
    M = float(c.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    
    if T_init is None:
        T_init = 0.005 * M
    T = T_init
    if min_T is None:
        min_T = 1e-8 * M
    
    no_improve = 0
    stagnation_limit = 200
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute peak weights
        c_shift = c / T - (c / T).max()
        w = np.exp(c_shift)
        w_sum = w.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Compute h_i = sum_k w_k * a_{k-i}
        h = _cross_corr(a, w)
        h = np.maximum(h, 1e-30)
        
        # Multiplicative update
        a_new = a / (h ** alpha)
        S_new = a_new.sum()
        if S_new < 1e-15 or not np.isfinite(S_new):
            T *= cooling
            continue
        a_new = a_new / S_new
        
        # Check improvement
        if (it + 1) % check_interval == 0:
            c_new = _full_convolve(a_new, a_new)
            M_new = float(c_new.max())
            sc_new = 2.0 * n * M_new
            if sc_new < best_score - 1e-15:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new.copy()
                c = c_new
                M = M_new
                no_improve = 0
            else:
                a = a_new.copy()
                c = c_new
                M = M_new
                no_improve += 1
            
            if no_improve > stagnation_limit:
                # Reset with larger temperature
                T = max(T * 10, 0.01 * M)
                no_improve = 0
        
        # Cool
        T = max(T * cooling, min_T)
    
    return best_a, best_score


def _softmax_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmax_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    # grad wrt a: 2n/beta * sum_k w_k * 2*a_{k-i} = 4n/beta * h_i
    # where h_i = sum_k w_k * a_{k-i}
    h = _cross_corr(a, w)
    grad_a = (4.0 * n / beta) * h
    # Project to simplex tangent: grad_z = a * (grad_a - dot(a, grad_a))
    Wa = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002,
                 0.0001, 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmax_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmax_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _lp_phase(a_init, deadline, K=50, max_rounds=20):
    """LP-based peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.98 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]
        
        # Build constraint matrix
        K_act = len(active)
        A_mat = np.zeros((K_act, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        
        b_vec = np.full(K_act, M * 1.0005)
        
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break
        
        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-14:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _coord_descent(a_init, deadline, max_iter=1000000):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    a = a / S
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    c = _full_convolve(a, a)
    M = float(c.max())
    
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick the coordinate that most contributes to the max peak
        peak_k = int(np.argmax(c))
        # The coordinates that contribute to c[peak_k] are i in [max(0, peak_k-n+1), min(n-1, peak_k)]
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        
        # Randomly pick among the contributing coordinates
        i = int(np.random.randint(i_lo, i_hi + 1))
        
        # Optimize a[i] to minimize max(c) (keeping others fixed)
        # c_k changes linearly/quadratically with a[i]
        # c_k += a[i] * (a_{k-i} + a_{k-i} - a[i] if k-i==i) for the affected k's
        # Actually, changing a[i] affects c_k for all k where a_{k-i} is non-zero

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _cross_corr(a, w):
    """Compute h[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1.
    a has length n, w has length 2n-1."""
    n = len(a)
    N = _next_pow2(3 * n - 2)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    R = np.fft.irfft(A * np.conj(W), N)[:n]
    return R

    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())
