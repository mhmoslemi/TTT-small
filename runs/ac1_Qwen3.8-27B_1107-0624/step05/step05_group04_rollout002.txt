"""
Optimization of step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Multi-scale L-BFGS with log-sum-exp continuation for a good starting point.
2. LP-based iterative refinement (key innovation): at each step, solve LP to find
   the direction maximizing total mass under convolution constraints, then mix.
3. Fine-tuning with small-beta L-BFGS and random perturbations.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags,csr_matrix
import time

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _logsumexp_obj(z, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _logsumexp_grad(z, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(len(a))
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 5.0]:

def _build_conv_matrix(f):
    n = len(f)
    diagonals = [np.full(n, f[k]) for k in range(n)]
    offsets = list(range(0, -n, -1))  # [0, -1, -2, ..., -(n-1)]
    C = diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    return C

def _lp_iteration(f, M):
    """Solve LP: max sum(b) s.t. C@b <= M, b >= 0. Return optimal b."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30})
        if result.success:
            b = result.x
            b = np.maximum(b, 0)
            if b.sum() > 1e-10:
                return b
    except Exception as e:
        print(f"  LP error: {e}")
    return None
