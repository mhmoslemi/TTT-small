"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Riemann-like projected gradient, L-BFGS with p-norm smoothing,
active-set refinement, diverse restarts, and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _p_norm_obj(z, beta, p, n):
    """Smooth p-norm approximation using softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _p_norm_grad(z, beta, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_active_set(a, c, n, tol_frac=0.001):
    """Compute subgradient using exact active set."""
    c_max = c.max()
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * np.ones_like(a) * (h - W)

def _riemann_iterate(a, n_iter=500, alpha_init=0.1, alpha_decay=0.9997, deadline=None):
    """Riemann-like projected gradient iterations for fast convergence."""
    n = len(a)
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    alpha = alpha_init
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        excess = np.maximum(c - c_max * 0.8, 0)
        excess_sum = excess.sum()
        
        if excess_sum < 1e-15:
            break
        
        w = excess / excess_sum
        a_rev = a[::-1]
        grad = _full_convolve(w, a_rev)
        h = grad[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        direction = h - W
        
        a_new = a - alpha * direction
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = _score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            alpha = min(0.3, alpha * 1.05)
        else:
            alpha *= alpha_decay
        
        a = a_new
        
        if i % 200 == 0 and i > 0:
            pass
    
    return best_a

def _lbfgs_softmax(a_init, beta, n, max_iter=5000, deadline=None):
    """L-BFGS-B optimization in softmax space."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _p_norm_obj(z, beta, 1.0, n)
    
    def jac(z):
        return _p_norm_grad(z, beta, 1.0, n)
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'maxcor': 40,
                'maxls': 100,
                'disp': False,
            }
        )
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False

def _subgrad_direct(a_init, deadline, max_steps=3000, lr_init=0.02):
    """Subgradient polishing with direct non-negative parameterization."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        direction = h - W
        
        grad_norm = np.linalg.norm(direction)
        if grad_norm < 1e-15:
            break
        
        step = lr / (1.0 + 0.005 * i)
        a_new = a - step * direction
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        if i % 20 == 0:
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new
                lr = min(0.1, lr * 1.02)
            else:
                a = a_new
                lr *= 0.92
        else:
            a = a_new
    
    a_final = np.maximum(a, 0)
    s_f = a_final.sum()
    if s_f > 1e-15:
        a_final = a_final / s_f
        sc_f = _score(a_final)
        if sc_f < best_sc:
            best_sc = sc_f
            best_a = a_final.copy()
    
    return best_a, best_sc

def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    """Full L-BFGS beta-continuation schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start
    
    betas = max_c * np.array([0.3, 0.15, 0.07, 0.035, 0.015, 0.007, 0.003, 0.0015, 0.0007, 0.0003])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        est_iter_time = 0.004 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_iter_time * 6))))
        
        a_new, success = _lbfgs_softmax(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc
    
    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quadratic":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = 0.1
        for i in range(int(n * taper)):
            frac = i / (n * taper)
            a[i] = 0.3 + 0.7 * frac
            a[n - 1 - i] = 0.3 + 0.7 * frac
        return a / a.sum()
    else:
        return np.ones(n) / n

def _symmetrize(a):
    """Make sequence palindromic."""
    n = len(a)
    if n % 2 == 0:
        half = a[:n//2]
        return np.concatenate([half, half[::-1]])
    else:
        half = a[:n//2]
        mid = a[n//2:n//2+1]
        return np.concatenate([half, mid, half[::-1]])

def _random_init(n, rng, kind="gaussian"):
    if kind == "gaussian":
        a = np.abs(rng.standard_normal(n))
    elif kind == "exponential":
        a = rng.exponential(1.0, n)
    elif kind == "fourier":
        n_modes = max(1, n // 16)
        a = np.ones(n)
        t = np.arange(n)
        for _ in range(n_modes):
            freq = rng.integers(1, n // 2 + 1)
            phase = rng.uniform(0, 2 * np.pi)
            a = a + 0.1 * np.cos(2 * np.pi * freq * t / n + phase)
        a = np.abs(a)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _active_set_refine(a, deadline, max_iter=200):
    """Iteratively refine using shrinking active set tolerance."""
    n = len(a)
    current_a = a.copy()
    current_sc = _score(a)
    
    for tol_frac in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(current_a, current_a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        
        if active.sum() < max(2, len(c) // 100):
            active = c >= (c_max - tol_frac * 10 * c_max)
        
        if active.sum() == 0:
            continue
        
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Do a few subgradient steps with this fixed active set
        a_local = current_a.copy()
        for step in range(20):
            if time.monotonic() > deadline:
                break
            a_rev = a_local[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, _full_convolve(a_local, a_local)))
            direction = h - W
            grad_norm = np.linalg.norm(direction)
            if grad_norm < 1e-15:
                break
            lr = 5.0 / (grad_norm * n)
            a_local = a_local - lr * direction
            a_local = np.maximum(a_local, 0)
            s = a_local.sum()
            if s < 1e-15:
                break
            a_local = a_local / s
        
        sc = _score(a_local)
        if sc < current_sc:
            current_a = a_local
            current_sc = sc
    
    return current_a, current_sc

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: Riemann pre-pass, L-BFGS with log-sum-exp,
    active-set refinement, diverse restarts, multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, using uniform n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "init")

    # --- Phase 1: Riemann pre-optimization ---
    print(f"\nPhase 1: Riemann pre-optimization (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 180
    a_riemann = _riemann_iterate(a0, n_iter=2000, alpha_init=0.05, alpha_decay=0.9998, deadline=p1_deadline)
    sc_r = _score(a_riemann)
    update_best(a_riemann, sc_r, "riemann")
    
    # Use the better of parent and riemann as starting point
    a_start = a_riemann if sc_r < _score(a0) else a0

    # --- Phase 2: L-BFGS main optimization ---
    print(f"\nPhase 2: L-BFGS main (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = min(deadline, t_start + 500)
    a_lbfgs = _lbfgs_schedule(a_start, p2_deadline, min_time_remaining=300.0)
    sc_lbfgs = _score(a_lbfgs)
    update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")

    # --- Phase 3: Subgradient + active set refinement ---
    print(f"\nPhase 3: Subgradient + active set (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        p3_deadline = min(deadline - 400, time.monotonic() + 150)
        a_sg, sc_sg = _subgrad_direct(best_seq, p3_deadline, max_steps=2000, lr_init=0.015)
        update_best(a_sg, sc_sg, "subgrad")
        
        p3b_deadline = min(deadline - 300, time.monotonic() + 100)
        a_as, sc_as = _active_set_refine(best_seq, p3b_deadline)
        update_best(a_as, sc_as, "active_set")

    # --- Phase 4: Fine L-BFGS on best ---
    print(f"\nPhase 4: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 300:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 250:
                break
            remaining = deadline - 250 - time.monotonic()
            if remaining < 20:
                break
            max_iter = min(3000, max(200, int(remaining / 0.03)))
            a_new, _ = _lbfgs_softmax(best_seq, beta, n, max_iter)
            sc = _score(a_new)
            update_best(a_new, sc, f"fine_{i}")

    # --- Phase 5: Multi-restart with diverse initializations ---
    print(f"\nPhase 5: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        remaining = deadline - 60 - time.monotonic()
        n_restarts = min(5, max(2, int(remaining / 120)))
        
        init_strategies = [
            ("perturb_small", lambda: _perturb(best_seq, 0.02, rng)),
            ("perturb_med", lambda: _perturb(best_seq, 0.05, rng)),
            ("tapered", lambda: _make_profile(n, "tapered")),
            ("quadratic", lambda: _make_profile(n, "quadratic")),
            ("random_gauss", lambda: _random_init(n, rng, "gaussian")),
            ("symmetric", lambda: _symmetrize(best_seq)),
            ("flat_taper", lambda: _make_profile(n, "flat_taper")),
        ]
        
        for i in range(min(n_restarts, len(init_strategies))):
            if time.monotonic() > deadline - 50:
                break
            name, gen_fn = init_strategies[i]
            a_init = gen_fn()
            sc_init = _score(a_init)
            print(f"  Restart {i} ({name}): score={sc_init:.10f}")
            
            # Quick Riemann
            rd = _riemann_iterate(a_init, n_iter=300, alpha_init=0.03, alpha_decay=0.999, 
                                   deadline=min(deadline - 40, time.monotonic() + 30))
            sc_rd = _score(rd)
            
            # Quick L-BFGS
            rd_deadline = min(deadline - 40, time.monotonic() + 60)
            a_r = _lbfgs_schedule(rd, rd_deadline, min_time_remaining=30)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{name}")

    # --- Phase 6: Try different lengths ---
    print(f"\nPhase 6: Multi-length exploration (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        for n_try in [3072, 4096, 1536]:
            if time.monotonic() > deadline - 25:
                break
            a_try = _make_profile(n_try, "tapered")
            sc_try = _score(a_try)
            print(f"  n={n_try}: init score={sc_try:.10f}")
            
            rd_try = _riemann_iterate(a_try, n_iter=300, alpha_init=0.03, alpha_decay=0.999,
                                       deadline=min(deadline - 20, time.monotonic() + 40))
            sc_rd = _score(rd_try)
            
            a_try2 = _lbfgs_schedule(rd_try, min(deadline - 20, time.monotonic() + 60), min_time_remaining=20)
            sc_final = _score(a_try2)
            update_best(a_try2, sc_final, f"n={n_try}")

    # --- Phase 7: Final polish ---
    print(f"\nPhase 7: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        # Final L-BFGS with smallest beta
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 0.00003
        a_fin, _ = _lbfgs_softmax(best_seq, beta_final, n, max_iter=2000)
        sc_fin = _score(a_fin)
        update_best(a_fin, sc_fin, "final_lbfgs")
        
        # Final subgradient
        if time.monotonic() < deadline - 5:
            a_fin2, sc_fin2 = _subgrad_direct(best_seq, deadline - 3, max_steps=1000, lr_init=0.01)
            update_best(a_fin2, sc_fin2, "final_subgrad")
        
        # Final active set
        if time.monotonic() < deadline - 3:
            a_fin3, sc_fin3 = _active_set_refine(best_seq, deadline - 2, max_iter=100)
            update_best(a_fin3, sc_fin3, "final_active")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
