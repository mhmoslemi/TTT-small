"""
Optimization of non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses parametric geometric initialization, aggressive L-BFGS continuation,
Adam polish with exact line search, and coordinate descent refinement.
Key insight: optimal shape is near-geometric with ratio ~1+2/n, giving score ~1.448.
"""
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _geometric_init(n, ratio):
    """Generate geometric sequence a[j] = ratio^j, normalized."""
    a = np.power(ratio, np.arange(n, dtype=np.float64))
    a = a / a.sum()
    return a


def _parametric_init(n, family, params):
    t = np.linspace(0.0, 1.0, n)
    if family == 'geom':
        r = params[0]
        a = np.power(r, np.arange(n, dtype=np.float64))
    elif family == 'geom_taper':
        r, p = params
        a = np.power(r, np.arange(n, dtype=np.float64)) * np.power(1.0 - t, p)
    elif family == 'exp':
        alpha = params[0]
        a = np.exp(alpha * t)
    elif family == 'power':
        p = params[0]
        a = np.power(t + 1.0/n, p)
    elif family == 'rational':
        beta = params[0]
        a = 1.0 / (1.0 + beta * t)
    elif family == 'sine':
        p = params[0]
        a = np.sin(0.5 * np.pi * t) ** p
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    return a


def _param_score(family, params, n):
    a = _parametric_init(n, family, params)
    return _score(a)


def _optimize_params(family, param_ranges, n, max_iter=200):
    """Optimize parametric family parameters using Nelder-Mead."""
    x0 = [lo + 0.3 * (hi - lo) for lo, hi in param_ranges]

    def obj(x):
        return _param_score(family, x, n)

    res = minimize(obj, x0, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
    return res.x, res.fun


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000, verbose=False):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [max(b * m0, 1e-15) for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-22,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                    if verbose:
                        print(f"    L-BFGS beta={beta:.2e}: score={s_new:.10f}")
        except Exception:
            pass
    return best_a, best_score


def _adam_line_search(a_init, deadline, lr=0.01, max_iter=500000, verbose=False):
    """Adam with line search on the log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    no_improve = 0
    check_interval = 20

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.005 * (0.95 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        direction = mh / (np.sqrt(vh) + eps)

        # Line search: try several step sizes
        best_z = z
        best_obj = _softmin_obj(z, beta, n)
        for step_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = lr * step_mult
            z_trial = z - alpha * direction
            z_trial = np.clip(z_trial, -100, 100)
            z_trial = z_trial - z_trial.mean()
            obj_trial = _softmin_obj(z_trial, beta, n)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_z = z_trial

        z = best_z

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _coord_descent_polish(a_init, deadline, max_sweeps=50, verbose=False):
    """
    Coordinate descent: for each coordinate, find optimal value via 1D search.
    Uses the fact that the 1D subproblem is convex.
    """
    n = len(a_init)
    a = a_init.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cur_max = float(c.max())
        improved = 0

        # Randomize order for diversity
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() > deadline:
                break
            i = int(idx)
            ai = a[i]
            if ai < 1e-15:
                continue

            # Affected lags: i to min(i+n-1, 2n-2)
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            if k_hi < k_lo:
                continue

            # Unaffected max
            if k_lo > 0 or k_hi < 2 * n - 2:
                mask = np.ones(2 * n - 1, dtype=bool)
                mask[k_lo:k_hi + 1] = False
                C_unaff = float(c[mask].max()) if mask.any() else 0.0
            else:
                C_unaff = 0.0

            # Linear coefficients for affected lags (excluding 2i)
            # For lag k in [i, i+n-1]: coefficient is 2*a[k-i] (if k != 2i)
            # Quadratic term at lag 2i: x^2
            k_2i = 2 * i
            has_quad = (k_2i >= k_lo and k_2i <= k_hi)

            # Get the linear coefficients
            # For lag k: a[k-i] where k ranges from i to i+n-1
            # That's a[0] to a[n-1]
            lin_coefs = 2.0 * a  # a[k-i] for k=i,...,i+n-1 gives a[0],...,a[n-1]
            lin_base = c[k_lo:k_hi + 1].copy()

            # Remove the quadratic contribution from linear_base at position k_2i
            if has_quad:
                pos_2i = k_2i - k_lo
                # At this position, the "linear" formula gives 2*a[i]*x (wrong)
                # Actually the true formula is x^2, not 2*a[i]*x
                # So lin_base at this pos should be c[k_2i] - a[i]^2 (base without quad)
                lin_base[pos_2i] = c[k_2i] - ai * ai
                lin_coefs_local = lin_coefs.copy()
                lin_coefs_local[pos_2i] = 0.0  # no linear term at 2i
            else:
                lin_coefs_local = lin_coefs

            # Now minimize max(C_unaff, max_j(lin_base[j] + lin_coefs_local[j]*x),
            #                  c[2i]-a[i]^2 + x^2 if has_quad)
            # subject to x >= 0

            # Upper bound for x: where x^2 > cur_max + C_unaff
            x_max = max(2.0 * ai, np.sqrt(cur_max + C_unaff + 1.0), 10.0 * ai)

            # Use scipy minimize_scalar for the 1D problem
            def obj_1d(x):
                if x < 0:
                    x = 0.0
                vals = lin_base + lin_coefs_local * x
                if has_quad:
                    pos_2i = k_2i - k_lo
                    vals[pos_2i] = c[k_2i] - ai * ai + x * x
                m = float(vals.max())
                if C_unaff > m:
                    m = C_unaff
                return m

            # Ternary search (convex function)
            lo = 0.0
            hi = x_max
            # Check if function is decreasing at 0 (should increase x)
            # or increasing (should stay at or near 0)
            f_mid_lo = obj_1d(0.1 * hi)
            f_0 = obj_1d(0.0)
            if f_mid_lo > f_0:
                hi = 0.1 * hi

            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_g = lo + (1.0 - gr) * (hi - lo)
            b_g = lo + gr * (hi - lo)
            fa = obj_1d(a_g)
            fb = obj_1d(b_g)
            for _ in range(60):
                if fa < fb:
                    hi = b_g
                    b_g = a_g
                    fb = fa
                    a_g = lo + (1.0 - gr) * (hi - lo)
                    fa = obj_1d(a_g)
                else:
                    lo = a_g
                    a_g = b_g
                    fa = fb
                    b_g = lo + gr * (hi - lo)
                    fb = obj_1d(b_g)

            x_opt = (lo + hi) / 2.0
            new_val = obj_1d(x_opt)

            if new_val < cur_max - 1e-14:
                a[i] = max(0.0, x_opt)
                improved += 1

        if improved == 0:
            break

        new_score = _score_norm(a)
        if new_score < best_score:
            best_score = new_score
            if verbose:

    return a.copy(), best_score
