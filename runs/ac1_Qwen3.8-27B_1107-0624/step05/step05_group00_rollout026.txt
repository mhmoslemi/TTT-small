"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA on simplex, L-BFGS in log-space, Adam with cosine annealing,
Fourier-based multi-restart, and multi-length exploration.

import numpy as np
import time
import scipy.optimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _compute_grad_z(a, c, w, n):
    """Gradient of smooth max of conv(a,a) w.r.t. log-space parameter z = log(n*a)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on the simplex for minimizing smooth max of conv(a,a)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()
    best_iter = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]

            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
                best_iter = it
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

    """Adam optimizer in log-space with cosine-annealed learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    best_iter = 0
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve_count = 0
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        phase_start_t = t
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):

            current_lr = lr_min + 0.5 * (lr_init - lr_min) * (1 + np.cos(np.pi * progress))
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            
            if t % 20 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    best_iter = t
                    no_improve_count = 0
                else:
                    no_improve_count += 20
            
            if no_improve_count > convergence_patience and it_in_phase > 100:
                break
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _lbfgs_smoothed(z_init, beta, deadline, max_iter=5000):
    """L-BFGS on log-space parameterization with smoothed max objective."""
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    
    best_z = z.copy()
    best_score = np.inf
    
    def objective_and_grad(z_flat):
        nonlocal best_z, best_score
        if time.monotonic() > deadline:
            return np.inf, np.zeros_like(z_flat)
        a = _softmax_compute(z_flat)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            obj = 2.0 * n * cm
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            obj = 2.0 * n * (cm + beta * np.log(s))
            w = e / s
        
        grad = _compute_grad_z(a, c, w, n)
        
        if obj < best_score:
            best_score = obj
            best_z = z_flat.copy()
        
        return obj, grad
    
    # L-BFGS-B on the log-space parameters
    options = {
        'maxiter': max_iter,
        'gtol': 1e-10,
        'ftol': 1e-12,
        'maxfun': max_iter
    }
    
    try:
        result = scipy.optimize.minimize(
            objective_and_grad, z, method='L-BFGS-B',
            jac=True, options=options,
            callback=lambda x: (time.monotonic() > deadline)
        )
    except Exception:
        pass
    
    a_best = _softmax_compute(best_z)
    return a_best, best_score

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian_envelope":
        t = np.linspace(0, 1, n, endpoint=False)
        center = 0.5
        a = np.exp(-((t - center) ** 2) / (2 * 0.15 ** 2))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "two_phasor":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 3 * t) + 0.1 * np.cos(2.0 * np.pi * 5 * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _fourier_perturb(a, freqs, amplitudes, phase, rng):
    """Perturb a by adding Fourier components."""
    n = len(a)
    z = np.zeros(n, dtype=np.complex128)
    for i, (f, amp) in enumerate(zip(freqs, amplitudes)):
        z += amp * np.exp(1j * (2 * np.pi * f * np.arange(n) / n + phase[i]))
    a_pert = a + np.real(z)
    a_pert = np.maximum(a_pert, 1e-12)
    return a_pert / a_pert.sum()

**Alternative approach: Water-filling / power allocation**

The problem is: minimize `max_k (a*a)[k]` subject to `sum a = 1, a >= 0`.

Think of it this way: we have `a` as a "signal" and `a*a` as its "autocorrelation." We want to minimize the peak of the autocorrelation. This is equivalent to: given the constraint on the "energy" (sum = 1), minimize the peak of the "power spectral density" autocorrelation.

In the frequency domain, the autocorrelation is the inverse Fourier transform of `|A(k)|^2` where `A` is the DFT of `a`. So we want to minimize the peak of the inverse DFT of `|A(k)|^2`.

This is related to the " flattening" of the power spectrum. If we could make `|A(k)|^2` flat, then the autocorrelation would be a single delta (at lag 0), which is optimal. But we can't make it perfectly flat because `a >= 0` and `sum a = 1` constrain the spectrum.

**New idea: Start from a Zadoff-Chu-like sequence and project onto the feasible set.**

Zadoff-Chu sequences have perfectly flat autocorrelation (for the complex case). For the real, non-negative case, we can't achieve this, but we can start from a projection of such a sequence.

Actually, let me think about this differently. The optimal solution should have the property that ALL the convolution values at the max are "balanced" - i.e., the gradient of the max pushes all active components equally. This is the KKT condition.

At the optimum, if multiple convolution values are at the maximum, the gradient of the objective is a convex combination of the individual gradients. For the solution to be stationary (on the simplex), this gradient must be constant (equal to the Lagrange multiplier for the sum constraint) on the support of `a`.

This gives us a nice characterization: at the optimum, `sum_k w_k * (a * e_k)[i] = lambda` for all i in the support, where `w` is the probability distribution on the active convolution values.

This is a fixed-point equation that we could try to solve iteratively!

**Fixed-point iteration:**
1. Start with `a_0` (normalized, sum=1, non-negative)
2. Compute `c = conv(a, a)`
3. Find the max values and set `w` = uniform on argmax (or softmax for smoothing)
4. Compute `g = 2 * (w * a_rev)[n-1:2n-1]` (the gradient)
5. Find `lambda = mean(g[i] for i in support)`
6. Update: `a_new[i] = a[i] * exp(-eta * (g[i] - lambda))` for i in support, 0 otherwise
7. Normalize: `a_new = a_new / sum(a_new)`
8. Repeat

This is essentially the multiplicative weight update / exponential weights algorithm applied to this problem. It's related to the "sinkron" or "power iteration" for finding the eigenvector.

Actually, this is very similar to what the log-space Adam is doing! The difference is that instead of using adaptive learning rates (Adam), we use a fixed step with the exponential update. And instead of the softmax-smoothed max, we use the exact max (or a very sharp approximation).

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines FISTA on simplex with aggressive beta schedule, Adam in log-space with cosine
annealing, fixed-point iteration, structured multi-restart, and multi-length exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a = len(a)
    out_n = n_a + n_b - 1 if n_b is not None else 2 * n_a - 1
    if n_b is None:
        n_b = n_a
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()) / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * float(c.max()))


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 100 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, lr_min, max_iter_total, patience=2000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    best_t = 0

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain = max_iter_total - t
        if remain <= 0:
            break
        phase_start = t
        it_phase = 0
        while it_phase < remain:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_sh = (c - cm) / beta
            e = np.exp(c_sh)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            progress = min(1.0, (t - phase_start) / max(1, remain))
            lr = lr_min + 0.5 * (lr_init - lr_min) * (1.0 + np.cos(np.pi * progress))

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.3:
                update = update * (0.3 / un)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 25 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    best_t = t
                    no_improve = 0
                else:
                    no_improve += 25

            if no_improve > patience and it_phase > 200:
                break

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fixed_point_iter(a_init, deadline, max_iter, beta_0=0.001):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()

    c_init = _full_convolve(a, a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        beta = beta_0 * cm * (0.95 ** it)
        if beta < 1e-10 * cm:
            beta = 1e-10 * cm

        w = np.exp((c - cm) / beta)
        ws = float(w.sum())
        if ws < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws

        grad = _compute_grad(a, w)
        support = a > 1e-10
        if support.sum() == 0:
            break
        lam = float(grad[support].mean())

        eta = 0.1
        a_new = a * np.exp(-eta * (grad - lam))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            break
        a = a_new / s

        if it % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.sin(np.pi * (t + 0.5)) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _fourier_perturb(a, freqs, amps, phases, rng):
    n = len(a)
    t_arr = np.arange(n, dtype=np.float64)
    z = 0.0
    for i in range(len(freqs)):
        z = z + amps[i] * np.cos(2.0 * np.pi * freqs[i] * t_arr / n + phases[i])
    new_a = a + z
    new_a = np.maximum(new_a, 1e-12)
    return new_a / new_a.sum()


def _random_perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

        final_list = [float(x) for x in final_arr]
        sc_final = _score(final_arr)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_list)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence(final_list))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return final_list

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
