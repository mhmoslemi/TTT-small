```python
import numpy as np
import time

def _proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _convolve_fft(a, b=None):
    """Fast linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    # Clean up tiny negative values
    r[np.abs(r) < 1e-12] = 0
    return r

def _score(a):
    """Score = 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _convolve_fft(a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _convolve_fft(a)
    return 2.0 * n * c.max()

def _subgradient_topk(a, k_lag, n):
    """Subgradient of conv(a,a)_{k_lag} w.r.t. a.
    Returns 2 * (a[k_lag - j] for valid j), i.e., 2 * reversed_shifted(a, k_lag)."""
    grad = np.zeros(n)
    # (grad)_m = 2 * a[k_lag - m] if 0 <= k_lag - m < n
    lo = max(0, k_lag - n + 1)
    hi = min(n - 1, k_lag)
    for m in range(lo, hi + 1):
        grad[m] = 2.0 * a[k_lag - m]
    return grad
```

```
    a_new = a - alpha * v
    
    # Project to simplex
    a = proj_simplex(a_new)
    
    # Track best
    sc = score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
```

```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a, b=None):
    n = len(a)
    if b is None:
        b = a
        m = n
    else:
        m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _convolve_fft(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _convolve_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgradient_for_lag(a, k_lag):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k_lag - n + 1)
    hi = min(n - 1, k_lag)
    if lo > hi:
        return grad
    grad[lo:hi + 1] = 2.0 * a[k_lag - hi:k_lag - lo + 1][::-1]
    return grad


def _topk_subgradient(a, top_lags, weights=None):
    n = len(a)
    grad = np.zeros(n)
    for k_lag in top_lags:
        grad += _subgradient_for_lag(a, int(k_lag))
    if weights is not None and len(weights) > 0:
        # Already summed, so no extra scaling needed if weights sum to 1
        # But we already summed unweighted, so divide by count
        pass
    return grad / len(top_lags)


def _make_uniform(n):
    return np.ones(n) / n


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)

def _subgradient_phase(a_init, deadline, rng, max_iters=None):
    """Core subgradient optimization with momentum and simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = _make_uniform(n)
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_sc = _score_norm(a)

    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum factor
    alpha = 0.05  # learning rate
    alpha_min = 1e-6
    K_start = 2
    K_end = 8
    K_current = K_start

    patience = 0
    patience_limit = 2000
    stall_count = 0

    it = 0
    eval_count = 0
    if max_iters is None:
        max_iters = 50_000_000

    while it < max_iters:
        if time.monotonic() > deadline - 5.0:
            break

        it += 1
        eval_count += 1

        # Compute convolution
        c = _convolve_fft(a)
        max_c = float(c.max())
        sc = 2.0 * n * max_c

        # Track best
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
            stall_count = 0
        else:
            patience += 1

        # Adaptive K
        K_current = min(K_end, K_start + it // 5000)

        # Find top K lags
        if K_current == 1:
            top_k = np.array([int(np.argmax(c))])
        else:
            K_actual = min(K_current, 2 * n - 1)
            top_k = np.argsort(c)[-K_actual:]

        # Compute subgradient (average of top K)
        g = _topk_subgradient(a, top_k)

        # Nesterov-style momentum
        v = beta * v + (1.0 - beta) * g

        # Step size adaptation
        # Decay slowly over time
        alpha = max(alpha_min, 0.05 / (1.0 + 0.0001 * it))

        # If stuck, try a bigger step (restart momentum)
        if patience > patience_limit:
            v = g.copy()  # Reset momentum to current gradient
            alpha = min(0.1, alpha * 3.0)
            patience = 0
            stall_count += 1
            if stall_count > 20:
                # Major restart: go back to best, add noise
                a = best_a.copy()
                noise = 0.01 * rng.normal(size=n)
                a = _proj_simplex(a + noise)
                v = np.zeros(n)
                stall_count = 0

        # Update
        a_new = a - alpha * v
        a = _proj_simplex(a_new)

        # Periodic detailed evaluation (every 500 iters)
        if it % 500 == 0:
            sc_check = _score_norm(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    return best_a, best_sc


def _subgradient_with_linesearch(a_init, deadline, rng):
    """Subgradient with backtracking line search for better step sizes."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_sc = _score_norm(a)

    v = np.zeros(n)
    beta = 0.85
    alpha = 0.02
    K = 3

    it = 0
    max_iters = 2_000_000
    no_improve = 0

    while it < max_iters:
        if time.monotonic() > deadline - 5.0:
            break
        it += 1

        c = _convolve_fft(a)
        sc = 2.0 * n * float(c.max())

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            # Reset
            a = best_a.copy()
            v = np.zeros(n)
            alpha = 0.03
            no_improve = 0
            K = min(K + 1, 10)

        # Top K subgradient
        K_actual = min(K, 2 * n - 1)
        top_k = np.argsort(c)[-K_actual:]
        g = _topk_subgradient(a, top_k)

        # Momentum
        v = beta * v + (1.0 - beta) * g

        # Backtracking line search
        alpha_try = alpha
        best_alpha = alpha_try
        best_trial_sc = _score_norm(a)

        for ls in range(6):
            trial = _proj_simplex(a - alpha_try * v)
            trial_sc = _score_norm(trial)
            if trial_sc < best_trial_sc:
                best_trial_sc = trial_sc
                best_alpha = alpha_try
            if trial_sc < best_sc + 1e-8:
                break
            alpha_try *= 0.5

        a = _proj_simplex(a - best_alpha * v)
        alpha = best_alpha * 1.05  # Slightly increase for next time

    return best_a, best_sc


def _peak_local_search(a_init, deadline, rng, n_trials=100000):
    """Peak-targeted local search exploiting partner structure."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 3.0:
            break

        c = _convolve_fft(a)
        max_lag = int(np.argmax(c))
        max_val = c[max_lag]

        # Find top 3 lags
        top3 = np.argsort(c)[-3:]

        if it % 4 == 0:
            # Strategy 1: Reduce mass at hot pair, add to cold position
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min > i_max:
                continue
            pairs = np.arange(i_min, i_max + 1)
            partners = max_lag - pairs
            products = a[pairs] * a[partners]
            idx = pairs[int(np.argmax(products))]

            delta = a[idx] * rng.uniform(0.3, 0.7)
            a_trial = a.copy()
            a_trial[idx] -= delta
            a_trial[idx] = max(0.0, a_trial[idx])

            # Find safe position (low contribution to top lags)
            contrib = np.zeros(n)
            for kl in top3:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    contrib[lo:hi + 1] += a[kl - hi:kl - lo + 1][::-1]

            # Pick from bottom quartile of contribution
            threshold = np.percentile(contrib, 25)
            safe_mask = contrib <= threshold
            if not np.any(safe_mask):
                safe_mask = np.ones(n, dtype=bool)
            safe_indices = np.where(safe_mask)[0]
            if len(safe_indices) == 0:
                continue
            add_idx = int(safe_indices[rng.integers(len(safe_indices))])
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial /= ss
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        elif it % 4 == 1:
            # Strategy 2: Block multiplicative perturbation
            block_size = int(rng.integers(3, min(20, n // 10)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        elif it % 4 == 2:
            # Strategy 3: Antisymmetric perturbation around peak lag
            # If peak is at lag k, try making a more "antisymmetric" around k/2
            center = max_lag / 2.0
            a_trial = a.copy()
            # Slightly reduce elements symmetric around center, increase others
            n_half = n // 2
            for _ in range(5):
                j = int(rng.integers(0, n_half))
                partner = n - 1 - j  # Partner for lag n-1
                if j != partner:
                    frac = rng.uniform(0.1, 0.3)
                    amt = a_trial[j] * frac
                    a_trial[j] -= amt
                    a_trial[j] = max(0.0, a_trial[j])
                    a_trial[partner] += amt
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        else:
            # Strategy 4: Pair swap (exchange mass between two random positions)
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt = min(a_trial[i1], a_trial[i2]) * frac
            a_trial[i1] -= amt
            a_trial[i2] += amt
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, rng, n_sweeps=50):
    """Coordinate descent: optimize each element (or pair) at a time."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 3.0:
            break

        # Random order
        order = rng.permutation(n)

        for idx in order:
            if time.monotonic() > deadline - 3.0:
                break

            # Binary search for optimal value of a[idx]
            # Fix all other elements, optimize a[idx]
            a_rest_sum = a.sum() - a[idx]
            if a_rest_sum < 1e-10:
                continue

            # The contribution of a[idx] to conv at lag k is:
            # (conv)_{k} includes a[idx] * a[k-idx] (if valid)
            # So increasing a[idx] increases conv at lags idx+j for all valid j
            # We want to find the value that minimizes max conv
            # This is a 1D problem - use ternary search or binary search

            lo = 0.0
            hi = a[idx] * 3.0 + 0.01

            # We need sum to stay 1, so if we increase a[idx] by delta,
            # we decrease others proportionally. Let's just try a few values.
            candidates = [a[idx] * 0.5, a[idx] * 0.75, a[idx] * 1.25, a[idx] * 1.5,
                          a[idx] + 0.001, a[idx] - 0.001, 0.0, a[idx] * 2.0]
            candidates = [max(0.0, min(c, 1.0)) for c in candidates if 0.0 <= c <= 1.0]

            best_trial = None
            best_trial_sc = _score_norm(a)

            for cand in candidates:
                if abs(cand - a[idx]) < 1e-10:
                    continue
                a_trial = a.copy()
                delta = cand - a[idx]
                a_trial[idx] = cand
                # Redistribute: subtract delta/n from all other elements
                other_indices = np.ones(n, dtype=bool)
                other_indices[idx] = False
                a_trial[other_indices] -= delta / (n - 1)
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial /= ss
                sc = _score_norm(a_trial)
                if sc < best_trial_sc:
                    best_trial_sc = sc
                    best_trial = a_trial.copy()

            if best_trial is not None:
                a = best_trial
                if best_trial_sc < best_sc:
                    best_sc = best_trial_sc
                    best_a = a.copy()

    return best_a, best_sc


def _multi_scale_refine(a_init, deadline, rng):
    """Downsample, optimize, upsample, refine."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    # Downsample to n/4
    n_coarse = max(256, n // 4)
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, n_coarse)
    a_coarse = np.interp(x1, x0, a_init)
    a_coarse = np.maximum(a_coarse, 1e-12)
    a_coarse /= a_coarse.sum()

    # Optimize at coarse scale
    dl_coarse = min(deadline - 60, time.monotonic() + 120.0)
    a_c_best, sc_c_best = _subgradient_phase(a_coarse, dl_coarse, rng)

    # Upsample
    a_up = np.interp(x0, x1, a_c_best)
    a_up = np.maximum(a_up, 1e-12)
    a_up /= a_up.sum()
    sc_up = _score_norm(a_up)

    if sc_up < best_sc:
        best_sc = sc_up
        best_a = a_up.copy()

    # Refine at full scale
    if time.monotonic() < deadline - 30:
        dl_fine = deadline - 20
        a_best, sc_best = _subgradient_phase(best_a, dl_fine, rng)
        if sc_best < best_sc:
            best_sc = sc_best
            best_a = a_best.copy()

    return best_a, best_sc


def _adam_method(a_init, deadline, rng, lr=0.01, k_top=5):
    """Adam optimizer with top-K smoothed gradient in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    max_iters = 500_000
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        t += 1

        # Softmax
        z_shift = z - z.max()
        e = np.exp(z_shift)
        s = e.sum()
        if s < 1e-300:
            a = np.ones(n) / n
        else:
            a = e / s

        # Compute conv and top-K gradient
        c = _convolve_fft(a)
        sc = 2.0 * n * float(c.max())

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        K = min(k_top, 2 * n - 1)
        top_k = np.argsort(c)[-K:]
        g = _topk_subgradient(a, top_k)

        # Scale gradient for log-space parameterization
        # d(a)/d(z_i) = a_i (delta_ij - a_j), so grad_z = grad_a / a (approximately)
        # More precisely: if f = sum_j g_j * a_j, then df/dz_i = g_i * a_i - a_i * sum_j g_j * a_j
        # For our subgradient: grad_a = g, so grad_z_i = g_i * a_i - a_i * <g, a>
        ga = float(np.dot(g, a))
        grad_z = a * (g - ga)

        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()

        # Decay LR
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.95)

    return best_a, best_sc

        _make_raised_cosine(n_work, 1.0),
        _make_raised_cosine(n_work, 2.0),
        _make_bimodal(n_work, 0.4, 0.15),
        _make_bimodal(n_work, 0.5, 0.2),
        _make_bimodal(n_work, 0.3, 0.1),
    ]
    # Also try perturbed parent
    if parent is not None:
        z = np.log(np.maximum(parent, 1e-15))
        z = z - z.mean()
        for noise_level in [0.05, 0.1, 0.2]:
            z_pert = z + noise_level * rng.normal(size=n_work)
            z_pert = z_pert - z_pert.mean()
            e = np.exp(z_pert - z_pert.max())
            profiles.append(e / e.sum())

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > p1_end:
            break
        p = np.maximum(p, 1e-12)
        p = p / p.sum()
        sc = _score_norm(p)
        candidates.append((sc, p.copy()))
        update_best(p, sc, f"init{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Subgradient with momentum (600s) ---
    p2_end = min(deadline, t_start + 630.0)
    if best_seq is not None and time.monotonic() < p2_end:
        print(f"Phase 2: Subgradient, start score={_score_norm(best_seq):.10f}")
        dl = p2_end - 5.0
        a_best, sc_best = _subgradient_phase(best_seq, dl, rng)
        update_best(a_best, sc_best, "subgrad")

        # Also try from the best profile candidate
        if candidates and len(candidates) > 1 and time.monotonic() < p2_end - 100:
            a_cand = candidates[1][1].copy()
            dl2 = min(p2_end - 50, time.monotonic() + 150.0)
            a_best2, sc_best2 = _subgradient_phase(a_cand, dl2, rng)
            update_best(a_best2, sc_best2, "subgrad_cand")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Adam refinement (300s) ---
    p3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < p3_end:
        print(f"Phase 3: Adam, start score={_score_norm(best_seq):.10f}")
        dl = p3_end - 5.0
        a_best, sc_best = _adam_method(best_seq, dl, rng, lr=0.015, k_top=5)
        update_best(a_best, sc_best, "adam_k5")

        if time.monotonic() < p3_end - 60:
            dl2 = p3_end - 5.0
            a_best2, sc_best2 = _adam_method(best_seq, dl2, rng, lr=0.008, k_top=3)
            update_best(a_best2, sc_best2, "adam_k3")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Peak-targeted local search (400s) ---
    p4_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < p4_end:
        print(f"Phase 4: Local search, start score={_score_norm(best_seq):.10f}")
        dl = p4_end - 5.0
        a_best, sc_best = _peak_local_search(best_seq, dl, rng, n_trials=300000)
        update_best(a_best, sc_best, "local_search")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-scale refinement (200s) ---
    p5_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < p5_end:
        print(f"Phase 5: Multi-scale, start score={_score_norm(best_seq):.10f}")
        dl = p5_end - 5.0
        a_best, sc_best = _multi_scale_refine(best_seq, dl, rng)
        update_best(a_best, sc_best, "multi_scale")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Second subgradient pass from best (200s) ---
    p6_end = min(deadline, t_start + 1730.0)
    if best_seq is not None and time.monotonic() < p6_end:
        print(f"Phase 6: Second subgradient, start score={_score_norm(best_seq):.10f}")
        dl = p6_end - 5.0
        a_best, sc_best = _subgradient_phase(best_seq, dl, rng)
        update_best(a_best, sc_best, "subgrad2")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final intensive local search (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"Phase 7: Final polish, remaining={deadline - time.monotonic():.0f}s")
        dl = deadline - 10.0
        remaining = dl - time.monotonic()

        if remaining > 60:
            a_best, sc_best = _peak_local_search(best_seq, dl, rng, n_trials=200000)
            update_best(a_best, sc_best, "final_local")

        if time.monotonic() < deadline - 30:
            # Very fine Adam
            dl2 = deadline - 15.0
            a_best2, sc_best2 = _adam_method(best_seq, dl2, rng, lr=0.003, k_top=2)
            update_best(a_best2, sc_best2, "final_adam")

    print(f"All phases done: best={best_score:.10f}")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator if reasonable size
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
                if np.isfinite(sc_eval) and sc_eval < sc_final:
                    # Evaluator found better (shouldn't happen but check)
                    pass
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular(4096)
    sc_fb = _score_full(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
```
