import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _stochastic_refine(a, deadline, rng, n_tries=5000, sigma_init=0.02):

            break

        z = z - z.mean()
        
        # Random perturbation
        if rng.random() < 0.5:
            # Global perturbation
            z += sigma * rng.standard_normal(n)
        else:
            # Local perturbation (random subset)
            k = max(1, n // 16)
            idx = rng.choice(n, size=k, replace=False)
            z[idx] += sigma * rng.standard_normal(k)

        sc = _score_norm(candidate)

            best_a = candidate
            best_score = sc
            sigma = max(0.001, sigma * 1.1)  # Increase slightly on success
        else:
            sigma = max(0.001, sigma * 0.999)  # Decrease on failure
    
    return best_a, best_score


def _coordinate_descent(a, deadline, n_passes=50):

    max_cc = float(c_curr.max())

    betas = [0.02 * max_cc, 0.01 * max_cc, 0.005 * max_cc, 0.002 * max_cc, 0.001 * max_cc]
    for beta in betas:
        if time.monotonic() > phase1_end - 5:
            break
        per_time = min(25.0, phase1_end - time.monotonic() - 3)
        if per_time < 5:
            break
        a_curr, sc = _fista_smoothed(a_curr, beta, time.monotonic() + per_time, 20000)
        a_curr = a_curr / a_curr.sum()
        sc_score = _score(a_curr)
        if sc_score < best_score:
            best_score = sc_score
            best_seq = a_curr.copy()
            print(f"  FISTA beta={beta:.6f}: score={sc_score:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())

    print(f"End Phase 1: best_score={best_score:.10f}")
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # Phase 2: Extended Adam with aggressive annealing and plateau detection
    print("Phase 2: Adam main optimization")
    phase2_end = min(deadline, t_start + 800.0)
    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    # Beta schedule: start moderately small, decay slowly
    betas_fine = []
    beta = 0.002 * max_cc
    while beta > 0.0000005 * max_cc:
        betas_fine.append(beta)
        beta *= 0.65
    if len(betas_fine) > 12:
        betas_fine = betas_fine[:12]

    per_time = phase2_end - time.monotonic() - 10
    a_opt, sc_opt = _adam_logspace(
        a_work, time.monotonic() + per_time,
        betas_fine, lr_init=0.006, max_iter_total=1200000,
        lr_decay=0.99, decay_period=3000,
        plateau_patience=150, plateau_reduction=0.75
    )
    a_opt = a_opt / a_opt.sum()
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam main: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
    else:
        print(f"  Adam main: {sc_opt_full:.10f} (no improvement over {best_score:.10f})")

    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    # Phase 3: Multi-restart diversification
    print("Phase 3: Multi-restart")
    phase3_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < phase3_end - 60:
        n_restarts = 10
        per_restart = (phase3_end - time.monotonic() - 30) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break

            r = rng.random()
            if r < 0.25:
                p_start = _perturb(a_work, 0.015 + 0.003 * i, rng)
            elif r < 0.45:
                sz = len(a_work)
                shift = int(rng.integers(1, max(2, sz // 2048)))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif r < 0.65:
                z = np.log(np.maximum(a_work, 1e-15))
                z = z - z.mean()
                freq = int(rng.integers(1, 12))
                t_arr = np.arange(len(a_work))
                phase_r = rng.random() * 2 * np.pi
                amp = 0.02 + 0.005 * rng.random()
                z += amp * np.cos(2 * np.pi * freq * t_arr / len(a_work) + phase_r)
                p_start = _softmax_compute(z)
            elif r < 0.85:
                k = max(8, len(a_work) // 128)
                idx = rng.choice(len(a_work), size=k, replace=False)
                p_start = a_work.copy()
                p_start[idx] *= (1.0 + 0.08 * rng.standard_normal(k))
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()
            else:
                # Interpolation with slight noise
                n_mid = len(a_work) // 2
                p_down = _interp_profile(a_work, n_mid)
                p_start = _interp_profile(p_down, len(a_work))
                p_start = _perturb(p_start, 0.01, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0003 * max_cs, 0.0001 * max_cs, 0.00003 * max_cs,
                  0.00001 * max_cs, 0.000003 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                bs, lr_init=0.004, max_iter_total=80000,
                lr_decay=0.995, decay_period=4000,
                plateau_patience=100, plateau_reduction=0.7
            )
            p_best = p_best / p_best.sum()
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                a_work = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Larger n exploration
    print("Phase 4: Larger n exploration")
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        n_target = 8192
        if len(best_seq) < n_target:
            a_large = _interp_profile(best_seq, n_target)
            sc_large = _score(a_large)
            print(f"  Interpolated n={n_target}: score={sc_large:.10f}")
            if sc_large < best_score:
                best_score = sc_large
                best_seq = a_large.copy()
                a_work = a_large.copy()

            if time.monotonic() < phase4_end - 20:
                c_l = _full_convolve(a_large, a_large)
                max_cl = float(c_l.max())
                beta_l = 0.003 * max_cl
                a_large_opt, _ = _fista_smoothed(a_large, beta_l, time.monotonic() + 15.0, 15000)
                betas_l = [0.001 * max_cl, 0.0004 * max_cl, 0.00015 * max_cl,
                           0.00005 * max_cl, 0.00002 * max_cl]
                per_t = phase4_end - time.monotonic() - 10
                if per_t > 15:
                    a_large_opt, sc_l = _adam_logspace(
                        a_large_opt, time.monotonic() + per_t,
                        betas_l, lr_init=0.004, max_iter_total=80000,
                        lr_decay=0.995, decay_period=5000,
                        plateau_patience=100, plateau_reduction=0.75
                    )
                    a_large_opt = a_large_opt / a_large_opt.sum()
                    sc_l_full = _score(a_large_opt)
                    if sc_l_full < best_score:
                        best_score = sc_l_full
                        best_seq = a_large_opt.copy()
                        a_work = a_large_opt.copy()
                        print(f"  Optimized n={n_target}: score={sc_l_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Stochastic refinement
    print("Phase 5: Stochastic refinement")
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_stoch, sc_stoch = _stochastic_refine(
            best_seq, time.monotonic() + min(120.0, deadline - time.monotonic() - 15),
            rng, n_tries=15000, sigma_init=0.01
        )
        a_stoch = a_stoch / a_stoch.sum()
        sc_stoch_full = _score(a_stoch)
        if sc_stoch_full < best_score:
            best_score = sc_stoch_full
            best_seq = a_stoch.copy()
            print(f"  Stochastic: score={sc_stoch_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Final ultra-fine Adam polish
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp,
                     0.000001 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.0015, max_iter_total=400000,
                lr_decay=0.998, decay_period=8000,
                plateau_patience=200, plateau_reduction=0.8
            )
            a_pol = a_pol / a_pol.sum()
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final validation
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
