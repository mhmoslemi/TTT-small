    return r

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

    """Compute gradient of c_k = sum_i a[i]*a[k-i] w.r.t. a.
    Returns 2*a[k-i] (shifted)."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            g[i] += 2.0 * a[j]
    return g

    # and a scaling to match the target sum

    if n == 0:
        return v

    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]

        return np.ones(n) / n
    rho = rho[-1]

    i_max = min(n - 1, k)
    if i_min <= i_max:
        idx = np.arange(i_min, i_max + 1)
        jdx = k - idx
        g[idx] = 2.0 * a[jdx]

    c = full_convolve(a_full, a_full)

    vals = np.clip((c - cm) / beta, -50, 50)

                best_score = sc
                best_a = a_try.copy()
            a = a_try
            lr *= 0.9999
    return best_a, best_score


def run_coord_descent(a_init, deadline, max_rounds=30):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    rng = np.random.default_rng(999)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_vec(z)
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = score_norm(a)
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_vec(z_trial)
                sc = score_norm(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_vec(z)
                sc = score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def make_initial_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [2.0, 4.0, -2.0, -4.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 4.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.5 + u**2)
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        profiles = make_initial_profiles(n_init)
        parent = profiles[0]
        update_best(parent, score_norm(parent), "default")

    n_parent = len(parent)
    print(f"Starting best: {best_score:.10f}")

    # ============================================================
    # Phase 1: Try n=8192 via interpolation + block CD + LSE (400s)
    # ============================================================
    phase1_end = min(deadline, t_start + 400.0)
    print(f"\nPhase 1: n=8192 exploration (t_end={phase1_end - t_start:.0f}s)")

    n_big = 8192
    a_big = interp_profile(parent, n_big)
    sc_big = score_norm(a_big)
    update_best(a_big, sc_big, f"interp_n{n_big}")

    if time.monotonic() < phase1_end - 60:
        dl1 = min(phase1_end - 30, time.monotonic() + 200.0)
        a_big, sc_big2 = run_block_cd(a_big, dl1, block_size=40, n_rounds=30, beta=0.02)
        update_best(a_big, sc_big2, f"block_cd_n{n_big}")

    if time.monotonic() < phase1_end - 20:
        dl2 = min(phase1_end - 10, time.monotonic() + 150.0)
        a_big2, sc_big3 = run_lse_lbfgs(a_big, dl2, beta_start_ratio=0.03, n_betas=10)
        update_best(a_big2, sc_big3, f"lse_n{n_big}")

    print(f"  Phase 1 done: best={best_score:.10f} (n={len(best_seq) if best_seq is not None else 0})")

    # ============================================================
    # Phase 2: Block CD at n=4096 (300s)
    # ============================================================
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\nPhase 2: Block CD at n={n_parent}")

    a_cur = best_seq.copy() if len(best_seq) == n_parent else interp_profile(best_seq, n_parent)
    sc_cur = score_norm(a_cur)
    update_best(a_cur, sc_cur, f"phase2_init_n{n_parent}")

    if time.monotonic() < phase2_end - 30:
        dl3 = min(phase2_end - 20, time.monotonic() + 280.0)
        a_cur, sc_cur2 = run_block_cd(a_cur, dl3, block_size=25, n_rounds=80, beta=0.015)
        update_best(a_cur, sc_cur2, f"block_cd_n{n_parent}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # ============================================================
    # Phase 3: LSE L-BFGS continuation (300s)
    # ============================================================
    phase3_end = min(deadline, t_start + 1000.0)
    print(f"\nPhase 3: LSE L-BFGS fine continuation")

    n_opt = len(best_seq)
    a_cur = best_seq.copy()

    for restart in range(3):
        if time.monotonic() > phase3_end - 30:
            break
        if restart == 0:
            a_start = a_cur.copy()
        else:
            z = np.log(np.maximum(a_cur, 1e-15))
            z += 0.03 * (restart + 1) * rng.normal(size=n_opt)
            z = np.clip(z, -60, 60)
            a_start = _softmax_vec(z)
            a_start = a_start / a_start.sum()

        per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) / (3 - restart))
        a_res, sc_res = run_lse_lbfgs(a_start, per_dl, beta_start_ratio=0.02, n_betas=12)
        update_best(a_res, sc_res, f"lse_r{restart}_n{n_opt}")
        a_cur = best_seq.copy()

    print(f"  Phase 3 done: best={best_score:.10f}")

    # ============================================================
    # Phase 4: Subgradient method with restarts (250s)
    # ============================================================
    phase4_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 4: Subgradient method")

    if time.monotonic() < phase4_end - 30:
        dl5 = min(phase4_end - 20, time.monotonic() + 230.0)
        a_sub, sc_sub = run_subgradient(best_seq, dl5, n_restarts=4, iters_per_restart=3000)
        update_best(a_sub, sc_sub, "subgradient")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # ============================================================
    # Phase 5: Try different initializations at n=4096 (200s)
    # ============================================================
    phase5_end = min(deadline, t_start + 1450.0)
    print(f"\nPhase 5: Diverse initializations")

    profiles = make_initial_profiles(n_parent)
    n_profiles_to_try = min(4, len(profiles))
    for pi in range(n_profiles_to_try):
        if time.monotonic() > phase5_end - 30:
            break
        a_prof = profiles[pi].copy()
        # Blend with current best
        a_blend = 0.3 * a_prof + 0.7 * best_seq
        a_blend = np.maximum(a_blend, 0)
        a_blend = a_blend / a_blend.sum()

        per_dl = min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic()) / (n_profiles_to_try - pi))
        a_res, sc_res = run_block_cd(a_blend, per_dl, block_size=25, n_rounds=20, beta=0.02)
        update_best(a_res, sc_res, f"diversify_{pi}_n{n_parent}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # ============================================================
    # Phase 6: Coordinate descent polish (remaining time)
    # ============================================================
    phase6_end = deadline
    print(f"\nPhase 6: Coordinate descent polish")

    if time.monotonic() < phase6_end - 20 and best_seq is not None:
        dl7 = min(phase6_end - 10, time.monotonic() + (phase6_end - time.monotonic()) * 0.6)
        a_cd, sc_cd = run_coord_descent(best_seq, dl7, max_rounds=20)
        update_best(a_cd, sc_cd, "coord_descent")

    # Final tiny beta L-BFGS
    if time.monotonic() < phase6_end - 15 and best_seq is not None:
        n_fin = len(best_seq)
        c_fin = full_convolve(best_seq, best_seq)
        max_c_fin = float(c_fin.max())
        beta_tiny = 1e-5 * max_c_fin
        z_fin = np.log(np.maximum(best_seq, 1e-15))
        z_fin = z_fin - z_fin.mean()
        dl8 = min(phase6_end - 5, time.monotonic() + 40.0)

        def obj_fin(z_in):
            a_s = _softmax_vec(z_in)
            return lse_obj_flat(a_s, beta_tiny)

        def grad_fin(z_in):
            a_s = _softmax_vec(z_in)
            return lse_grad_flat(a_s, beta_tiny)

        try:
            res = minimize(obj_fin, z_fin, method='L-BFGS-B', jac=grad_fin,
                           options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_fin = _softmax_vec(res.x)
                sc_fin = score_norm(a_fin)
                update_best(a_fin, sc_fin, "tiny_beta_final")
        except Exception:
            pass

    print(f"\nFINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")

    # Validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = score_fn(final_seq)
        print(f"Validated score: {sc_final:.10f}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = make_initial_profiles(n_fb)[2]
    print(f"FALLBACK: n={n_fb}, score={score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
