    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()
    best_score = float(_full_convolve(a, a).max())
    
    c = _full_convolve(a, a)

    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:

def _adam_optimize(a_init, deadline, lr=0.01):
    """Adam optimizer with log-sum-exp smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    beta_scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    max_it_per_phase = 50000

    for beta_scale in beta_scales:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * max_c_init
        iters = 0
        while iters < max_it_per_phase:
            if time.monotonic() > deadline:
                break
            iters += 1
            t_step += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            c_shifted = (c - cm) / beta
            e = np.exp(np.minimum(c_shifted, 0))
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_step % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if iters % 20000 == 0:
                current_lr = max(0.001, current_lr * 0.9)

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _make_uniform(n):
    return np.ones(n) / n

def _perturb_structured(a, sigma, rng):
    """Structured perturbation using cosine basis for low-frequency changes."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    
    # Add perturbation in cosine basis (low-frequency)
    num_modes = min(n // 4, 200)
    modes = rng.integers(1, num_modes + 1, size=min(10, num_modes))
    for m in modes:
        t = np.arange(n) / n
        cos_term = np.cos(2 * np.pi * m * t)
        z += sigma * rng.standard_normal() * cos_term
    
    # Add small random perturbation
    z += sigma * 0.1 * rng.standard_normal(n)
    
    z = z - z.mean()
    return _softmax(z)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines subgradient descent with multiplicative updates (FFT-accelerated),
LP-based direction finding for escaping local minima, and Adam fine-tuning.
Key innovation: efficient subgradient computation via cross-correlation using FFT.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_subgradient(a, temperature):
    """Compute smoothed subgradient of max(conv(a,a)) with respect to a.
    
    Returns (gradient, convolution, max_conv).
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Compute softmax weights over convolution values
    c_shifted = (c - c_max) / temperature
    e = np.exp(np.minimum(c_shifted, 0))
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        idx = int(np.argmax(c))
        w = np.zeros_like(c)
        w[idx] = 1.0
    else:
        w = e / s
    
    # term2[i] = sum_k w[k] * a[i-k] = (a * w)[i] for i in [0, n-1]
    # term1[i] = sum_k w[k] * a[k-i] = (w * a_rev)[n-1+i] for i in [0, n-1]
    a_rev = a[::-1]
    N = _next_pow2(3 * n + 10)
    Fa = np.fft.rfft(a, N)
    Fw = np.fft.rfft(w, N)
    Fa_rev = np.fft.rfft(a_rev, N)
    
    conv_aw = np.fft.irfft(Fa * Fw, N)
    term2 = conv_aw[:n]
    
    conv_w_arev = np.fft.irfft(Fw * Fa_rev, N)
    term1 = conv_w_arev[n - 1:2 * n - 1]
    
    g = term1 + term2
    return g, c, c_max


def _subgradient_descent(a_init, deadline, lr_init=0.5, lr_min=0.005,
                         temp_init=0.05, temp_min=0.0005):
    """Subgradient descent with multiplicative updates."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, float('inf')
    a = a / s
    
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_score = float(c.max())
    current_max = best_score
    
    eta = lr_init
    temperature = temp_init
    it = 0
    no_improve_count = 0
    max_no_improve = 2000
    
    while time.monotonic() < deadline:
        it += 1
        
        g, c, current_max = _compute_subgradient(a, temperature)
        
        # Center gradient for sum constraint
        g = g - g.mean()
        
        # Multiplicative update
        exponent = -eta * g
        exponent = np.clip(exponent, -40, 40)
        a_new = a * np.exp(exponent)
        
        s_new = a_new.sum()
        if s_new < 1e-15 or not np.isfinite(s_new):
            eta *= 0.5
            continue
        a_new = a_new / s_new
        
        c_new = _full_convolve(a_new, a_new)
        c_new_max = float(c_new.max())
        
        if c_new_max < current_max * (1 - 1e-14):
            a = a_new
            current_max = c_new_max
            no_improve_count = 0
            eta = max(lr_min, eta * 0.9995)
            temperature = max(temp_min, temperature * 0.999)
            
            if current_max < best_score:
                best_score = current_max
                best_a = a.copy()
        else:
            no_improve_count += 1
            eta *= 0.7
            if eta < lr_min:
                eta = lr_min
                temperature = max(temp_min, temperature * 0.95)
            
            if no_improve_count > max_no_improve:
                # Restart with perturbation
                z = np.log(np.maximum(a, 1e-15))
                z += 0.05 * np.random.standard_normal(n)
                z = z - z.mean()
                a = _softmax(z)
                c = _full_convolve(a, a)
                current_max = float(c.max())
                eta = min(lr_init, eta * 2)
                no_improve_count = 0
                max_no_improve = max(500, max_no_improve // 2)
        
        if it % 5000 == 0:
            sc = 2.0 * n * current_max
            print(f"    SubGrad iter {it}: max_c={current_max:.12f}, "
                  f"score≈{sc:.10f}, eta={eta:.6f}, temp={temperature:.6f}")
    
    return best_a, best_score


def _lp_solve_single(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M elementwise, g >= 0."""
    k_arr = np.arange(2 * n - 1)
    j_arr = np.arange(n)
    diff = k_arr[:, np.newaxis] - j_arr[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 45, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _adam_optimize(a_init, deadline, lr=0.01):
    """Adam optimizer with log-sum-exp smoothed max for fine-tuning."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    beta_scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    max_it_per_phase = 50000

    for beta_scale in beta_scales:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * max_c_init
        iters = 0
        while iters < max_it_per_phase:
            if time.monotonic() > deadline:
                break
            iters += 1
            t_step += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            c_shifted = (c - cm) / beta
            e = np.exp(np.minimum(c_shifted, 0))
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_step % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if iters % 20000 == 0:
                current_lr = max(0.001, current_lr * 0.9)

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _perturb_structured(a, sigma, rng):
    """Structured perturbation using cosine basis for low-frequency changes."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    
    num_modes = min(n // 4, 300)
    num_picks = min(15, num_modes)
    modes = rng.integers(1, num_modes + 1, size=num_picks)
    for m in modes:
        t = np.arange(n) / n
        cos_term = np.cos(2 * np.pi * m * t)
        z += sigma * rng.standard_normal() * cos_term
    
    z += sigma * 0.1 * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _try_update_best(best_score, best_seq, p):
    """Try to update best. Returns (best_score, best_seq, improved)."""
    if p is None:
        return best_score, best_seq, False
    p = np.asarray(p, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s < 0.01:
        return best_score, best_seq, False
    sc = _score(p)
    if np.isfinite(sc) and sc < best_score:
        return sc, p.copy(), True
    return best_score, best_seq, False


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining subgradient descent (multiplicative updates),
    LP-based direction finding, and Adam fine-tuning to minimize
    2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _report(label):
        nonlocal best_seq, best_score
        if best_seq is not None:
            print(f"  [{label}] best_score={best_score:.10f} (n={len(best_seq)}) "
                  f"t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_uniform(n_main)
        print(f"No parent, using uniform n={n_main}")
        best_seq = a0.copy()
        best_score = _score(a0)

    # === Phase 1: Subgradient descent (multiplicative updates) ===
    phase1_end = min(deadline, t_start + 450.0)
    print(f"\n=== Phase 1: Subgradient descent (n={n_main}) ===")
    
    a_sub, sc_sub = _subgradient_descent(
        a0, phase1_end - 10,
        lr_init=0.5, lr_min=0.005,
        temp_init=0.05, temp_min=0.0005
    )
    best_score, best_seq, improved = _try_update_best(best_score, best_seq, a_sub)
    if improved:
        print(f"  SubGrad improved to {best_score:.10f}")
    _report("Phase 1 end")

    # === Phase 2: LP-based optimization ===
    phase2_end = min(deadline, t_start + 850.0)
    print(f"\n=== Phase 2: LP-based optimization ===")
    
    f = best_seq / best_seq.sum()
    n_lp = len(f)
    
    A_cache = None
    f_for_matrix = None
    current_t = 0.3
    lp_steps = 0
    stuck_count = 0
    
    while time.monotonic() < phase2_end - 30:
        c = _full_convolve(f, f)
        M = float(c.max())
        
        if A_cache is None or f_for_matrix is None or np.max(np.abs(f - f_for_matrix)) > 1e-8:
            k_arr = np.arange(2 * n_lp - 1)
            j_arr = np.arange(n_lp)
            diff = k_arr[:, np.newaxis] - j_arr[np.newaxis, :]
            mask = (diff >= 0) & (diff < n_lp)
            rows, cols = np.where(mask)
            vals = f[diff[rows, cols]]
            A_cache = csr_matrix((vals, (rows, cols)), shape=(2 * n_lp - 1, n_lp))
            f_for_matrix = f.copy()
        
        g = _lp_solve_single(f, M, n_lp)
        
        if g is None:
            stuck_count += 1
            if stuck_count > 8:
                print(f"  LP stuck after {lp_steps} steps")
                break
            g = f.copy()
        else:
            stuck_count = 0
        
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_norm = g * float(f.sum()) / g_sum
        
        improved = False
        t_try = current_t
        for _ in range(8):
            f_new = (1 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            f_new_sum = float(f_new.sum())
            if f_new_sum < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / f_new_sum
            
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())
            
            if M_new < M * (1 - 1e-12):
                f = f_new
                current_t = min(1.0, t_try * 1.3)
                improved = True
                lp_steps += 1
                if lp_steps % 5 == 0:
                    sc = _score(f)
                    best_score, best_seq, imp = _try_update_best(best_score, best_seq, f)
                    if imp:
                        print(f"  LP step {lp_steps}: score={best_score:.10f} t={time.monotonic()-t_start:.1f}s")
                break
            else:
                t_try *= 0.5
        
        if not improved:
            current_t *= 0.5
            stuck_count += 1
            if stuck_count > 15:
                break
    
    best_score, best_seq, imp = _try_update_best(best_score, best_seq, f)
    _report(f"Phase 2 end ({lp_steps} LP steps)")

    # === Phase 3: Adam fine-tuning ===
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 3: Adam fine-tuning ===")
    
    if best_seq is not None and time.monotonic() < phase3_end - 60:
        a_init = best_seq / best_seq.sum()
        remaining = phase3_end - time.monotonic() - 30
        a_best, sc_best = _adam_optimize(a_init, time.monotonic() + remaining, lr=0.01)
        best_score, best_seq, imp = _try_update_best(best_score, best_seq, a_best)
        if imp:
            print(f"  Adam improved to {best_score:.10f}")
    
    # Second Adam with different learning rate
    if best_seq is not None and time.monotonic() < deadline - 300:
        a_init = best_seq / best_seq.sum()
        remaining = min(120.0, deadline - time.monotonic() - 300)
        if remaining > 30:
            a_best2, _ = _adam_optimize(a_init, time.monotonic() + remaining, lr=0.005)
            best_score, best_seq, imp = _try_update_best(best_score, best_seq, a_best2)
            if imp:
                print(f"  Adam2 improved to {best_score:.10f}")
    
    _report("Phase 3 end")

    # === Phase 4: Multi-restart with structured perturbations ===
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Multi-restart ===")
    
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 3
        per_restart = (phase4_end - time.monotonic() - 30) / max(1, n_restarts)
        sigmas = [0.02, 0.05, 0.08]
        
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 30:
                break
            p_start = _perturb_structured(a_best_norm, sigmas[i], rng)
            deadline_i = min(time.monotonic() + per_restart, phase4_end - time.monotonic() - 20)
            
            a_r1, _ = _subgradient_descent(p_start, deadline_i - 20,
                                            lr_init=0.3, lr_min=0.005,
                                            temp_init=0.03, temp_min=0.001)
            best_score, best_seq, imp = _try_update_best(best_score, best_seq, a_r1)
            if imp:
                print(f"  Restart {i+1}: improved to {best_score:.10f}")
            
            if time.monotonic() < deadline_i - 5:
                a_r2, _ = _adam_optimize(a_r1, deadline_i, lr=0.008)
                best_score, best_seq, imp = _try_update_best(best_score, best_seq, a_r2)
                if imp:
                    print(f"  Restart {i+1} Adam: improved to {best_score:.10f}")
    
    _report("Phase 4 end")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 15
        if remaining > 40:
            a_final, _ = _adam_optimize(a_best_norm, time.monotonic() + remaining, lr=0.003)
            best_score, best_seq, imp = _try_update_best(best_score, best_seq, a_final)
            if imp:
                print(f"  Final Adam: improved to {best_score:.10f}")
        
        if time.monotonic() < deadline - 30:
            remaining2 = deadline - time.monotonic() - 25
            if remaining2 > 30:
                source = a_final if 'a_final' in dir() and a_final is not None else a_best_norm
                a_final2, _ = _subgradient_descent(source, time.monotonic() + remaining2,
                                                    lr_init=0.1, lr_min=0.003,
                                                    temp_init=0.01, temp_min=0.001)
                best_score, best_seq, imp = _try_update_best(best_score, best_seq, a_final2)
                if imp:
                    print(f"  Final SubGrad: improved to {best_score:.10f}")
    
    _report("Phase 5 end")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
