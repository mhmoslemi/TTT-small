def _lse_score_and_grad(a, beta):
    """Compute the smoothed score and gradient efficiently."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        score = 2.0 * n * c_max
    else:
        w = e / s
        score = 2.0 * n * (c_max + beta * np.log(s))
    
    # Gradient
    w_rev = w[::-1]
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    Wr = np.fft.rfft(w_rev, N)
    A = np.fft.rfft(a, N)
    wa = np.fft.irfft(W * A, N)[:3*n-2]
    war = np.fft.irfft(Wr * A, N)[:3*n-2]
    grad = (2.0 * n / beta) * (wa[:n] + war[n-1:n-1+n])
    return score, grad, w


def _adamw_optimize(a_init, deadline, beta_start=1.0, beta_end=0.01, lr=0.01, 
                    weight_decay=0.0, display_every=1000):
    """AdamW optimizer with log-sum-exp smoothed objective and beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Total iterations estimate
    total_time = max(1.0, deadline - time.monotonic())
    # Estimate iterations per second
    c = _full_convolve(a_init, a_init)
    t0 = time.monotonic()
    for _ in range(100):
        _ = _full_convolve(a_init, a_init)
    it_per_sec = 100.0 / (time.monotonic() - t0)
    total_iters = int(total_time * it_per_sec * 0.9)
    
    for it in range(max(1, total_iters)):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta
        progress = it / max(1, total_iters)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        a = _softmax(z)
        score, grad, w = _lse_score_and_grad(a, beta)
        
        if not np.all(np.isfinite(grad)):
            break
        
        t_step += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)
        
        # AdamW update
        update = lr * (mh / (np.sqrt(vh) + eps) + weight_decay * z)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if (it + 1) % display_every == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"    iter {it+1}/{total_iters}, beta={beta:.5f}, "
                  f"score={sc:.10f}, best={best_score:.10f}, "
                  f"t={time.monotonic()-time.monotonic():.0f}s")
    
    # Final check

def _adamw_phased(a_init, deadline, phases=None):
    """Multi-phase AdamW with decreasing beta."""
    if phases is None:
        # (beta_start, beta_end, lr, fraction_of_time)
        phases = [
            (5.0, 1.0, 0.01, 0.2),
            (1.0, 0.2, 0.008, 0.25),
            (0.2, 0.05, 0.005, 0.25),
            (0.05, 0.01, 0.003, 0.2),
            (0.01, 0.002, 0.001, 0.1),
        ]
    
    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(a)
    
    for beta_s, beta_e, lr, frac in phases:
        phase_deadline = t0 + frac * total_time
        if time.monotonic() > phase_deadline - 5:
            continue
        a, sc = _adamw_optimize(a, phase_deadline, beta_start=beta_s, 
                                beta_end=beta_e, lr=lr)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  Phase done: score={sc:.10f}, t={time.monotonic()-t0:.1f}s")
    
    return best_a, best_score


def _cosine_window(n):
    """Raised cosine window."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    return w / w.sum()


def _hamming_window(n):
    x = np.arange(n)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    return w / w.sum()


def _flat_top_window(n):
    x = np.arange(n)
    w = 1.0 - 1.93 * np.cos(2 * np.pi * x / (n - 1)) + 1.29 * np.cos(4 * np.pi * x / (n - 1)) - 0.39 * np.cos(6 * np.pi * x / (n - 1))
    w = np.maximum(w, 0)
    return w / w.sum()


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_descent(f, M, time_limit=20):
    """LP-based descent direction."""
    n = len(f)
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _active_set_refine(a, max_iter=50):
    """
    Active set method: identify which convolution lags are at the maximum,
    then solve a smaller optimization problem.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    for _ in range(max_iter):
        c = _full_convolve(a, a)
        M = c.max()
        
        # Identify active set (lags where c[k] is close to M)
        threshold = M * 0.999
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            break
        if len(active) == 2 * n - 1:
            # All active, no further refinement possible
            break
        
        # Use the active set to compute a better gradient
        # At the KKT optimum, for the active set S:
        # The "pressure" from active constraints should be balanced
        # Use a weighted smoothed max focusing on active constraints
        w = np.zeros_like(c)
        w[active] = 1.0 / len(active)
        
        # Gradient computation with restricted weights
        w_rev = w[::-1]
        N = _next_pow2(3 * n)
        W = np.fft.rfft(w, N)
        Wr = np.fft.rfft(w_rev, N)
        A = np.fft.rfft(a, N)
        wa = np.fft.irfft(W * A, N)[:3*n-2]
        war = np.fft.irfft(Wr * A, N)[:3*n-2]
        grad = 2.0 * n * (wa[:n] + war[n-1:n-1+n])
        
        # Try a small Newton step: we want to reduce M
        # The Hessian of the max (approximately) involves the second derivative
        # For now, just do a gradient step with line search
        a_old = a.copy()
        for step_size in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_new = a - step_size * grad / (grad * grad).sum() * a.mean()
            a_new = np.maximum(a_new, 0)
            if a_new.sum() < 1e-15:
                continue
            a_new = a_new / a_new.sum()
            c_new = _full_convolve(a_new, a_new)
            M_new = c_new.max()
            if M_new < M:
                a = a_new
                break
    
    return a


def _try_n(a_best, n_target, deadline):
    """Interpolate to target n and optimize."""
    if len(a_best) == n_target:
        a = a_best.copy()
    else:
        x0 = np.linspace(0, 1, len(a_best))
        x1 = np.linspace(0, 1, n_target)
        a = np.interp(x1, x0, a_best)
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    sc = _score(a)
    print(f"    n={n_target}: interpolated score={sc:.10f}")
    
    # Optimize at this n
    a_opt, sc_opt = _adamw_optimize(a, deadline, beta_start=1.0, beta_end=0.01, lr=0.008)
    return a_opt, sc_opt
