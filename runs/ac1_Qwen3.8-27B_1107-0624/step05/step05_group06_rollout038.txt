"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, coordinate descent, and
active-set targeting. Explores multiple sequence lengths and initializations.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_optimize(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                     'maxcor': 50, 'maxls': 200, 'disp': False}
        )
        a_out = _softmax(result.x)
        return a_out, result.success
    except:
        return a_init, False

def _coord_descent(a_init, deadline, max_passes=5000, beta_smooth=0.001):
    """Coordinate descent in the simplex for minimizing max conv."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    c = _full_convolve(a, a)
    best_a = a.copy()
    best_max = float(c.max())
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        c_max = c.max()
        if c_max < best_max - 1e-15:
            best_max = c_max
            best_a = a.copy()
        
        # Compute weights for active set (smoothed)
        threshold = c_max - beta_smooth * c_max
        active = c >= threshold
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        
        # For each coordinate, compute the gradient and take a step
        # grad_i = sum_k w_k * a_{k-i} (for valid k)
        # This is the convolution of w with reversed a, centered
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = wc[n-1:2*n-1]
        
        # Also need to account for the normalization constraint
        # If we change a_i, the sum changes, so we need to renormalize
        # Approximate: the effective gradient in the simplex is
        # (grad_i - mean(grad))
        grad_centered = grad - grad.mean()
        
        # Adaptive step size
        step = 0.01 / (1 + p * 0.01)
        
        # Update: a_i <- a_i - step * grad_centered_i
        # But we need to handle the constraint a_i >= 0
        # Use a multiplicative update instead
        a_new = a * np.exp(-step * grad_centered)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Check if this improved
        c_new = _full_convolve(a_new, a_new)
        if c_new.max() < c_max - 1e-15:
            a = a_new
            c = c_new
        else:
            # Try smaller step
            step *= 0.5
            a_new = a * np.exp(-step * grad_centered)
            a_new = np.maximum(a_new, 1e-15)
            a_new = a_new / a_new.sum()
            c_new = _full_convolve(a_new, a_new)
            if c_new.max() < c_max - 1e-15:
                a = a_new
                c = c_new
    
    return best_a, best_max

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, coordinate descent with incremental
convolution updates, active-set targeting, and multi-length exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return c_max
    return c_max + beta * np.log(s)


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return a * (h - W)


def _lbfgs_optimize(a_init, beta, n, max_iter=5000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                     'maxcor': 50, 'maxls': 300, 'disp': False}
        )
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _lbfgs_schedule(a_start, deadline, min_remaining=60.0):
    """Run L-BFGS with adaptive beta continuation."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
                              0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(5000, max(100, int(remaining / 0.03)))
        a_new, success = _lbfgs_optimize(current_a, beta, n, max_iter)
        sc = _score(a_new)
        if sc < current_sc:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.2e}: score={sc:.10f} (iter={i})")

    return current_a


def _coord_descent_cd(a_init, deadline, max_passes=100000, tol=1e-12):
    """
    Fast coordinate descent with incremental convolution updates.
    Works in the simplex (sum=1, a>=0) using multiplicative updates.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    c = _full_convolve(a, a).copy()
    best_a = a.copy()
    best_cmax = float(c.max())
    c_max = float(c.max())

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        # Compute active set weights
        threshold = c_max - 0.005 * c_max
        active = c >= threshold
        n_active = active.sum()
        if n_active == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
            n_active = 1
        w = np.zeros(len(c))
        w[active] = 1.0 / n_active

        # Compute gradient: grad_i = sum_k w_k * a_{k-i}
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = wc[n - 1:2 * n - 1]

        # Center the gradient (for simplex constraint)
        grad_c = grad - grad.mean()

        # Adaptive step size
        base_step = 0.05 / (1.0 + p * 0.005)

        # Multiplicative update (keeps a > 0)
        for trial in range(3):
            a_new = a * np.exp(-base_step * grad_c)
            a_new = np.maximum(a_new, 1e-18)
            a_new = a_new / a_new.sum()

            # Compute new convolution
            c_new = _full_convolve(a_new, a_new)
            c_new_max = float(c_new.max())

            if c_new_max < c_max - tol:
                # Update convolution incrementally for efficiency
                delta_a = a_new - a
                c = c + _full_convolve(delta_a, a_new) + _full_convolve(a, delta_a) \
                    - _full_convolve(delta_a, delta_a)
                a = a_new
                c_max = c_new_max
                break
            else:
                base_step *= 0.5

        if c_max < best_cmax - 1e-15:
            best_cmax = c_max
            best_a = a.copy()

        # Check convergence
        if p > 100 and best_cmax - c_max < 1e-14:
            break

    return best_a, 2.0 * n * best_cmax


def _subgrad_polish(a_init, deadline, max_steps=5000, lr_init=0.1):
    """Subgradient method in z-space with exact active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        active = c >= (c_max - 0.001 * c_max)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = a * (h - W)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = lr / (1.0 + i * 0.002)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_chk = _softmax(z)
            sc = _score(a_chk)
            if sc < best_sc:
                best_sc = sc
                best_a = a_chk.copy()
                lr *= 1.02
            else:
                lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=5):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = 0.05 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _make_init(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "random":
        rng = np.random.default_rng(12345)
        a = rng.random(n) + 0.01
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t)**0.5 + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n
