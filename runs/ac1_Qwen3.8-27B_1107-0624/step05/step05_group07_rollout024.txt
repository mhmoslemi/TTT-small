The LP iterated minimax works as follows:
- Given current f (normalized to sum 1), find g* = argmin_{g >= 0, sum g = 1} max_k (f*g)[k]
- Update: f ← (1-t)f + t*g*

This is a form of mirror descent on the simplex. The LP gives the "best response" g that minimizes the worst-case convolution with f.

The key question is: how many iterations of this are needed, and what step size to use?

For a convex problem, mirror descent with step size t_k = O(1/sqrt(k)) gives O(1/sqrt(k)) convergence. But in practice, we can do much better with adaptive step sizes.

Let me think about a different approach: **Frank-Wolfe (conditional gradient) method**.

The Frank-Wolfe method for minimizing a convex function f(x) over a convex set C:
- At each iteration, find s = argmin_{s in C} ∇f(x) · s
- Update: x ← (1-t)x + t*s

In our case, C is the simplex {g >= 0, sum g = 1} and f(g) = max_k (f_cur * g)[k] (where f_cur is fixed during one iteration of the outer loop).

The gradient of max_k (f_cur * g)[k] with respect to g is:
- If k* = argmax_k (f_cur * g)[k], then ∇_g f = (f_cur * g)[k*]'s gradient = the k*-th row of the convolution matrix.
- The k*-th row of the convolution matrix applied to g gives (f_cur * g)[k*].

So ∇_g f = (A[g] · f_cur)_k* where A is the convolution operator. More precisely, if we define C_f(g) = f * g (the convolution), then ∇_g max_k C_f(g)[k] is the row of the convolution matrix corresponding to argmax.

The Frank-Wolfe step would be: find the vertex of the simplex that minimizes the inner product with this gradient. That is, find j that minimizes ∇_g f[j], and set s = e_j (the j-th standard basis vector).

But this is just gradient descent projected to the simplex, which is the same as coordinate descent. Not very helpful.

- L-BFGS with aggressive log-sum-exp beta continuation
- Multi-scale (n) exploration with interpolation
- Focused asymmetric initializations (ramp/step shapes)

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int64), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate_adaptive(f, n_max_iters=500, deadline=None):
    """LP iterated minimax with adaptive step size."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    t_step = 0.4
    no_improve_count = 0
    max_no_improve = 30

    for it in range(n_max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g, t_val = _lp_solve(f)
        if g is None:
            break

        # Adaptive step size: use multiplicative update
        # f_new = f^(1-t) * g^t, normalized
        if t_step > 0.01:
            f_new = np.power(np.maximum(f, 1e-30), 1 - t_step) * np.power(np.maximum(g, 1e-30), t_step)
        else:
            f_new = (1 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
            no_improve_count = 0
            # Slightly increase step if improving
            t_step = min(0.5, t_step * 1.05)
        else:
            no_improve_count += 1
            t_step *= 0.85

        f = f_new
        if it % 50 == 0:

            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}, step={t_step:.4f}{elapsed}")

        if no_improve_count > max_no_improve and t_step < 0.02:
            break

    return best_a, best_score

        if it % 50 == 0:
            print(f"    LP(linear) iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")

    return best_a, best_score


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, verbose=True):
    """L-BFGS with aggressive beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Aggressive beta schedule - go much lower than before
    beta_fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005,
                  0.004, 0.003, 0.002, 0.0015, 0.001, 0.0007, 0.0005,
                  0.0003, 0.0002, 0.0001, 0.00007, 0.00005,
                  0.00003, 0.00002, 0.00001, 0.000007, 0.000005,
                  0.000003, 0.000002, 0.000001, 0.0000007, 0.0000005]
    beta_list = [b * max_c0 for b in beta_fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_left = deadline - time.monotonic()
        time_per = time_left / max(1, remaining)
        time_per = min(time_per, 90.0)
        time_per = max(time_per, 5.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(500, int(time_per * 50))
        maxiter = min(maxiter, 50000)

        if verbose and idx % 5 == 0:
            print(f"    LSE beta={beta:.2e} ({beta/max_c0:.2e}*max_c), maxiter={maxiter}")

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-30, 'gtol': 1e-16,
                         'maxls': 100, 'maxfun': maxiter * 10,
                         'iprint': -1}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception as e:
            if verbose:
                print(f"      L-BFGS error at beta idx {idx}: {e}")

    return best_a, best_score
