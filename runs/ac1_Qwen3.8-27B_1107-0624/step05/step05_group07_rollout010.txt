Wait, for the coordinate descent, let me think about what the 1D subproblem looks like.

If we fix all of `a` except `a[j]`, then:
- `(a*a)[k]` changes only for `k` such that there exists `i` with `i = j` or `k-i = j`, i.e., `k = j + i` or `k = j + (k-j)` ... 
- More precisely: `(a*a)[k] = sum_i a[i]*a[k-i]`. The terms involving `a[j]` are:
  - `i = j`: gives `a[j]*a[k-j]`
  - `k-i = j`, i.e., `i = k-j`: gives `a[k-j]*a[j]`
  
  So the change in `(a*a)[k]` when we change `a[j]` from `v` to `v+δ` is:
  `2*a[k-j]*δ + 0` (for `i=j` and `i=k-j` being different)
  or `2*a[j]*δ + δ^2` (when `k = 2j`, i.e., `i = j = k-j`)
  
  Wait, when `k = 2j`: the term `i=j` gives `a[j]*a[j]`, and `i=k-j=j` is the same term. So it's just `a[j]^2` and the derivative is `2*a[j]`, and the second derivative is 1.
  
  When `k ≠ 2j`: the terms `i=j` and `i=k-j` are different, so we have `a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j]`. The derivative w.r.t. `a[j]` is `2*a[k-j]`, and the second derivative is 0.

So the 1D subproblem for `a[j]` is:
`minimize max_k [(a*a)[k] + 2*(a[k-j] if k≠2j else a[j])*δ + (δ^2 if k=2j else 0)]`
subject to `a[j] + δ >= 0` and `sum(d) = 0` (normalization constraint, which means other elements must be adjusted)

The normalization constraint makes this more complex. If we ignore it (or handle it by renormalizing after), the subproblem is:
`minimize max_k [c[k] + 2*a[k-j]*v - 2*a[k-j]*a[j] + (v^2 - a[j]^2) * 1_{k=2j}]`

where `v = a[j] + δ` is the new value and `c[k]` is the current autocorrelation.

This simplifies to:
For `k ≠ 2j`: `c[k] + 2*a[k-j]*(v - a[j])`
For `k = 2j`: `c[2j] - 2*a[j]^2 + v^2`

We want to find `v >= 0` that minimizes the maximum of these.

This is a 1D minimax problem! We can solve it with bisection:
- For a given `t`, check if there exists `v >= 0` such that all constraints are satisfied
- The constraints are:
  - For `k ≠ 2j`: `c[k] + 2*a[k-j]*(v - a[j]) <= t` → `v <= (t - c[k])/(2*a[k-j]) + a[j]` (if `a[k-j] > 0`) or `v >= (t - c[k])/(2*a[k-j]) + a[j]` (if `a[k-j] < 0`, but `a >= 0` so this is always `a[k-j] >= 0`)
  - For `k = 2j`: `v^2 <= t - c[2j] + 2*a[j]^2` → `v <= sqrt(t - c[2j] + 2*a[j]^2)` (if the argument is positive)

So for a given `t`, the feasible set for `v` is:
`v >= 0`
`v <= min over k≠2j of [(t - c[k])/(2*a[k-j]) + a[j]]` (for `a[k-j] > 0`)
`v <= sqrt(max(0, t - c[2j] + 2*a[j]^2))`

This is feasible iff the upper bound is >= 0.

We can binary search on `t` to find the minimum feasible `t`, which gives the optimal `v`.

This is a very clean 1D optimization! And it's exact (not approximate).

After updating `a[j]`, we need to renormalize (divide by the new sum). This slightly changes the problem, but it's a small perturbation.

```python
"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP minimax iterations, exact subgradient mirror descent,
and exact 1D coordinate descent for final polish.
"""
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
from scipy.sparse import csr_matrix
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Full score: 2n * max(conv(a,a)) / sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(f):
    """Build the convolution matrix for cross-correlation with f."""
    n = len(f)
    # (f*g)[k] = sum_i f[i] * g[k-i]
    # For k = 0,...,2n-2 and g indexed 0,...,n-1:
    # (f*g)[k] = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} f[i] * g[k-i]
    # = sum_{j: 0<=j<n, 0<=k-j<n} f[k-j] * g[j]
    # So the matrix A has A[k,j] = f[k-j] if 0<=k-j<n, else 0
    row_idx = []
    col_idx = []
    data = []
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row_idx.append(k)
                col_idx.append(j)
                data.append(f[i])
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    return A


def _lp_minimax(f, solver='CLARABEL'):
    """Solve: minimize t s.t. (f*g)[k] <= t, sum(g)=1, g>=0"""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
        if g.value is not None and np.all(np.isfinite(g.value)):
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 0:
                g_val = g_val / gs
            return g_val, float(t.value), prob
    except Exception as e:
        print(f"  LP failed ({solver}): {e}")
    return None, None, None


def _lp_iterate_improved(f, deadline, n_iters=200, use_line_search=True):
    """LP-based minimax iteration with line search."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm(f)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        g, t_val, _ = _lp_minimax(f)
        if g is None:
            break
        
        if use_line_search:
            # Try different step sizes
            best_t = 0.5
            best_sc = _score_norm(f)
            for alpha in [0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 1.0]:
                f_new = (1 - alpha) * f + alpha * g
                f_new = np.maximum(f_new, 0)
                fs = f_new.sum()
                if fs < 1e-15:
                    continue
                f_new = f_new / fs
                sc = _score_norm(f_new)
                if sc < best_sc:
                    best_sc = sc
                    best_t = alpha
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs > 0:
                f_new = f_new / fs
            sc = best_sc
        else:
            f_new = 0.5 * f + 0.5 * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            sc = _score_norm(f_new)
        
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        
        f = f_new
        if it % 20 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}")
    
    return best_a, best_score


def _subgrad_mirror_descent(a_init, deadline, top_k=30, eta_init=1.0):
    """Mirror descent with exact subgradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z -= z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    eta = eta_init
    it = 0
    
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top_k active lags
        if top_k < len(c):
            idx = np.argsort(c)[::-1][:top_k]
        else:
            idx = np.where(c >= c_max)[0]
        
        mu = np.zeros(2*n-1)
        mu[idx] = 1.0 / len(idx)
        
        # Compute subgradient: grad[j] = 2 * sum_k mu[k] * a[k-j]
        N = _next_pow2(2*n)
        mu_pad = np.zeros(N)
        mu_pad[:2*n-1] = mu
        a_pad = np.zeros(N)
        a_pad[:n] = a
        
        MU = np.fft.rfft(mu_pad)
        AP = np.fft.rfft(a_pad)
        grad = 2.0 * np.fft.irfft(MU * np.conj(AP), N)[:n]
        
        # Step size: normalize by the "energy" of the gradient
        # For mirror descent with entropy, the natural step is:
        # eta_k = (f(a_k) - f*) / <grad, a_k>
        # We don't know f*, so use a heuristic:
        gradient_dot = float(np.dot(grad, a))
        if abs(gradient_dot) < 1e-30:
            break
        eta_cur = eta * c_max / max(abs(gradient_dot), 1e-30)
        eta_cur = min(eta_cur, 1.0)
        
        z = z - eta_cur * grad / max(c_max * n, 1e-10)
        z = np.clip(z, -100, 100)
        z -= z.mean()
        
        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        it += 1
        if it % 10000 == 0:
            eta *= 0.95
            print(f"    MD iter {it}: score={sc:.10f}, best={best_score:.10f}")
    
    return best_a, best_score


def _coord_descent_exact(a_init, deadline, max_rounds=50):
    """Coordinate descent with exact 1D subproblem solution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = best_a.copy()
        c = _full_convolve(a, a)
        c_max = float(c.max())
        improved = False
        
        # Process coordinates in order of "importance" (those contributing most to peaks)
        for j in range(n):
            if time.monotonic() > deadline:
                break
            
            # 1D subproblem: find optimal v = a[j] new value
            # For k != 2j: c[k] + 2*a[k-j]*(v - a[j]) <= t
            # For k = 2j: c[2j] - 2*a[j]^2 + v^2 <= t  (wait, need to be careful)

            # c[k] includes term a[j]*a[k-j] (when 0<=k-j<n)
            # and term a[k-j]*a[j] (same thing, since convolution is of a with itself)
            # Actually c[k] = sum_i a[i]*a[k-i], and the terms involving a[j] are:
            #   i=j: a[j]*a[k-j] (if 0<=k-j<n)
            #   k-i=j => i=k-j: a[k-j]*a[j] (if 0<=k-j<n)
            # So total contribution of a[j] to c[k] is 2*a[j]*a[k-j] (if k!=2j)
            # or a[j]^2 (if k=2j)

            # For k where 0<=k-j<n (i.e., j<=k<j+n):
            #   old: c[k] includes 2*a[j]*a[k-j] (or a[j]^2 if k=2j)
            #   new: c[k] includes 2*v*a[k-j] (or v^2 if k=2j)
            #   change: 2*(v-a[j])*a[k-j] (or (v^2-a[j]^2) if k=2j)

            # For a given t, find feasible v >= 0:
            #   For all k where j<=k<j+n and k != 2j:

            #     => v <= (t - c[k]) / (2*a[k-j]) + a[j]  (if a[k-j] > 0)
            #     => c[k] <= t  (if a[k-j] = 0, constraint is always satisfied or not)
            #   For k = 2j (if 0<=2j<2n-1):
            #     c[2j] - a[j]^2 + v^2 <= t  (the a[j]^2 term is replaced by v^2)
            #     => v^2 <= t - c[2j] + a[j]^2
            #     => v <= sqrt(t - c[2j] + a[j]^2)
            
            aj = a[j]
            
            # Upper bound on t: current c_max
            t_lo = float(c.max()) - 1e-6  # We want to at least maintain
            t_hi = float(c.max())
            
            # Actually, let's do it differently:
            # Find the minimum t such that there exists v >= 0 satisfying all constraints
            # t_hi = c_max (current max, which is always feasible with v = a[j])
            # t_lo = some lower bound
            
            # Bisection:
            t_lo = 0.0
            t_hi = c_max + 1e-10
            
            for _ in range(40):  # 40 bisection steps gives ~1e-12 precision
                t_mid = (t_lo + t_hi) / 2
                
                # Find max feasible v
                v_max = np.inf
                
                # Constraints from k != 2j
                # k ranges from j to j+n-1 (where 0<=k-j<n)
                # But k also must be in [0, 2n-2]
                k_min = max(0, j)
                k_max = min(2*n-2, j+n-1)
                
                for k in range(k_min, k_max+1):
                    if k == 2*j:
                        continue
                    akj = a[k-j]
                    if akj > 1e-15:

                        bound = (t_mid - c[k]) / (2.0 * akj) + aj
                        if bound < v_max:
                            v_max = bound
                    else:
                        # c[k] must be <= t_mid (no v-dependence)
                        if c[k] > t_mid:
                            v_max = -1  # Infeasible
                            break
                
                # Constraint from k = 2j
                if 0 <= 2*j <= 2*n-2:

                    if arg >= 0:
                        bound = np.sqrt(arg)
                        if bound < v_max:
                            v_max = bound
                    else:
                        v_max = -1  # Infeasible
                
                # Also v >= 0
                if v_max < 0:
                    t_lo = t_mid  # Need larger t
                else:
                    t_hi = t_mid  # t_mid is feasible
            
            # Optimal v (use the largest feasible v at t_hi? No, we want to minimize t)
            # At t_hi (minimum feasible t), find v:
            t_opt = t_hi
            v_max = np.inf
            k_min = max(0, j)
            k_max = min(2*n-2, j+n-1)
            for k in range(k_min, k_max+1):
                if k == 2*j:
                    continue
                akj = a[k-j]
                if akj > 1e-15:
                    bound = (t_opt - c[k]) / (2.0 * akj) + aj
                    if bound < v_max:
                        v_max = bound
            if 0 <= 2*j <= 2*n-2:
                arg = t_opt - c[2*j] + aj*aj
                if arg >= 0:
                    v_max = min(v_max, np.sqrt(arg))
            
            v_new = max(0, v_max)
            # Also try v = 0 and v = v_max to see which gives better max
            # Actually, at the optimal t, any v in [0, v_max] is feasible
            # We want to choose v that minimizes the actual max (not just satisfies t)
            # For simplicity, use v = v_max (or v = 0 if v_max < aj*0.1)
            # Actually, let's just use v = v_max
            
            if v_new != aj:
                a[j] = v_new
                # Renormalize
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
        if (rnd+1) % 5 == 0:
            print(f"    CD round {rnd+1}: best={best_score:.10f}")
    
    return best_a, best_score
```

```python

    k_min = max(0, j)
    k_max = min(2*n-2, j+n-1)
    
    ks = np.arange(k_min, k_max+1)
    akj = a[ks - j]  # a[k-j] for each k
    
    # Exclude k = 2j from the linear constraints
    mask = ks != 2*j
    
    # Linear constraints: c[k] + 2*(v-aj)*akj <= t => v <= (t-c[k])/(2*akj) + aj
    # For akj > 0: v <= (t - c[k]) / (2*akj) + aj
    # For akj = 0: c[k] <= t (independent of v)
    
    # Quadratic constraint (k=2j): c[2j] - aj^2 + v^2 <= t => v <= sqrt(t - c[2j] + aj^2)
    
    # Binary search on t
    t_lo = 0.0
    t_hi = float(c.max()) + 1e-10
    
    for _ in range(45):
        t_mid = (t_lo + t_hi) / 2
        
        # Check feasibility
        feasible = True
        v_max = np.inf
        
        # Linear constraints (vectorized)
        akj_pos = akj[mask]
        c_pos = c[ks[mask]]
        valid = akj_pos > 1e-15
        
        # For valid (akj > 0): v <= (t - c[k]) / (2*akj) + aj
        if np.any(valid):
            bounds = (t_mid - c_pos[valid]) / (2.0 * akj_pos[valid]) + aj
            v_max = min(v_max, float(np.min(bounds)))
        
        # For invalid (akj = 0): need c[k] <= t
        if np.any(~valid):
            if np.any(c_pos[~valid] > t_mid):
                feasible = False
        
        # Quadratic constraint
        if 2*j >= k_min and 2*j <= k_max:
            arg = t_mid - c[2*j] + aj*aj
            if arg < 0:
                feasible = False
            else:
                v_max = min(v_max, np.sqrt(arg))
        
        if not feasible or v_max < 0:
            t_lo = t_mid
        else:
            t_hi = t_mid
    
    return t_hi, v_max
```
