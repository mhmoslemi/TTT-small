= w1*a_flipped[3] + w2*a_flipped[2] + w3*a_flipped[1] + w4*a_flipped[0]
But a_flipped[3] = 0 (out of bounds). So:
= w2*a0 + w3*a1 + w4*a2 ✓

Great, so the formula works: grad[m] = 2 * (w_full * a_flipped)[m + n - 1].

Now let me write the full solution. I'll aim for a clean, efficient implementation with the following strategy:

Phase 1: FISTA with adaptive beta (coarse to fine) on the simplex for n=4096.
Phase 2: Adam in log-space for fine optimization at n=4096.
Phase 3: Interpolate to n=8192 and optimize.
Phase 4: Final polish.

Let me also add a "top-k" trick: instead of computing the softmax over all 2n-1 = 8191 convolution coefficients, only use the top k (e.g., k=50) coefficients. This makes each iteration faster and focuses the optimization on the most important constraints.

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA with top-k smoothed max on simplex, Adam log-space polishing, and multi-n exploration.
Key improvements: adaptive step size, top-k approximation, n=8192 exploration, better schedules.
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_weights(c, beta, top_k=None):
    if top_k is not None and top_k < len(c):
        k = min(top_k, len(c))
        idx = np.argpartition(-c, k - 1)[:k]
        c_top = c[idx]
        c_shift = c_top - c_top.max()
        w_top = np.exp(c_shift / beta)
        s = w_top.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[idx[np.argmax(c_top)]] = 1.0
        else:
            w = np.zeros_like(c)
            w[idx] = w_top / s
    else:
        c_shift = c - c.max()
        w = np.exp(c_shift / beta)
        s = w.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / s
    return w


def _grad_smoothed(a, w):
    n = len(a)
    a_flipped = a[::-1]
    w_a = _full_convolve(w, a_flipped)
    g = w_a[n - 1:2 * n - 1] * 2.0
    return g


def _fista_topk(a_init, beta_list, top_k_list, deadline, max_total_iter=200000):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    total_iter = 0

    for pi, (beta, top_k) in enumerate(zip(beta_list, top_k_list)):
        phase_deadline = min(deadline, time.monotonic() + max(30.0, (deadline - time.monotonic()) / max(1, len(beta_list) - pi) * 1.5))
        max_iter_phase = max(2000, int((phase_deadline - time.monotonic()) * 3000))

        c_curr = _full_convolve(a_curr)
        L = 4.0 * n / beta * 1.5
        step = 1.0 / L

        for it in range(max_iter_phase):
            if time.monotonic() > phase_deadline:
                break
            total_iter += 1

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y)
            w = _compute_weights(c_y, beta, top_k)
            grad = _grad_smoothed(y, w)

            a_next = _proj_simplex(y - step * grad)

            if it % 200 == 0 or it < 5:
                sc = _score_norm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score, total_iter


def _adam_logspace(a_init, deadline, beta_init, top_k, lr_init=0.01, max_iter=300000, lr_decay=0.995, decay_every=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    beta = beta_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = np.exp(z - z.max())
        a = a / a.sum()

        c = _full_convolve(a)
        cm = c.max()
        c_shifted = (c - cm) / beta
        w = _compute_weights(c, beta, top_k)

        a_flipped = a[::-1]
        w_a = _full_convolve(w, a_flipped)
        h = w_a[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 50 == 0:
            a_chk = np.exp(z - z.max())
            a_chk = a_chk / a_chk.sum()
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

        if t % decay_every == 0:
            current_lr = max(0.0005, current_lr * lr_decay)
            beta *= 0.7

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "hann":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = np.cos(np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_like":
        t = np.linspace(-1, 1, n)
        a = np.sinc(2.0 * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
