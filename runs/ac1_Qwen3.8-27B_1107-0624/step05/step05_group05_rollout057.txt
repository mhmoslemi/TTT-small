def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

        vh = v_buf / (1.0 - b2 ** t)

            update = update * (0.3 / update_norm)
        z = z - update

        z = z - z.mean()
