Now let me think about the Nesterov subgradient method more carefully:

The objective is `f(a) = max_k {conv(a,a)[k]}` on the simplex `{a >= 0, sum(a) = 1}`.

Standard Nesterov accelerated gradient (for smooth functions):
- x_{k+1} = (x_k - x_{k-1}) * (k/(k+3)) + x_k - alpha_k * grad(f)(x_k)
- (simplified: y_k = x_k + (t_k - 1)/(t_{k+1}) * (x_k - x_{k-1}), x_{k+1} = proj(y_k - alpha_k * grad(y_k)))

For subgradient methods, Nesterov acceleration can be adapted:
- t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
- beta_k = (t_k - 1) / t_{k+1}
- y_k = x_k + beta_k * (x_k - x_{k-1})
- g_k = subgradient at y_k
- x_{k+1} = proj_simplex(y_k - alpha_k * g_k)

The step size alpha_k can be:
- Constant (conservative)
- Decreasing: alpha_k = C / sqrt(k)
- Adaptive: based on the Lipschitz constant of f

For our problem, the Lipschitz constant of f(a) = max_k conv(a,a)[k] on the simplex is bounded by:
L = max_k ||grad conv(a,a)[k]||_2

For conv(a,a)[k], the gradient has at most min(k+1, 2n-1-k, n) non-zero entries, each of magnitude at most 2*max(a). So L ≤ 2*max(a)*sqrt(n) ≈ 2*sqrt(n) (since max(a) ≤ 1 for our normalized sequences).

This is not super tight, but gives us a starting point for the step size.

Uses multi-scale continuation with top-k peak balancing and coordinate descent polish.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]

def _simplex_project(v):
    n = len(v)
    if n <= 1:
        return np.ones(n) / max(1, n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = np.max(np.where(cond)[0])
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0.0)

    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""

1. x_0 = x_1 (initial)
2. For k = 0, 1, 2, ...:

   - x_{k+1} = proj_C(y_k - alpha_k * g_k) where g_k = grad f(y_k)

"""
Nesterov-accelerated subgradient method with top-k peak balancing for minimizing
the maximum autocorrelation of non-negative step functions. Uses multi-scale continuation,
mirror descent alternation, and coordinate descent polish.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    N = _next_pow2(n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n]


def _simplex_project(v):
    n = len(v)
    if n <= 0:
        return v
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    k = int(np.max(np.where(cond)[0]))
    theta = cssv[k] / (k + 1)
    return np.maximum(v - theta, 0.0)


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_actual(a):
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)


def _top_k_subgradient(a, c, k_top=5):
    n = len(a)
    peak_idx = np.argsort(c)[::-1][:k_top]
    peak_vals = c[peak_idx]
    total = peak_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = peak_vals / total
    g = np.zeros(n)
    for pk, wi in zip(peak_idx, w):
        if wi < 1e-12:
            continue
        pk = int(pk)
        lo = max(0, pk - n + 1)
        hi = min(n, pk + 1)
        if lo < hi:
            idx = np.arange(lo, hi)
            g[lo:hi] += 2.0 * wi * a[pk - idx]
    return g


def _make_ramp(n):
    t = np.linspace(0, 1, n)
    a = 1.0 + t
    return a / a.sum()


def _make_bell(n, width=0.3):
    t = np.linspace(-1, 1, n)
    a = np.exp(-t ** 2 / (2 * width ** 2))
    return a / a.sum()


def _make_u_shape(n, c=1.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + c * (t - 0.5) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_two_peak(n):
    t = np.linspace(0, 1, n)
    a = np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2)
    return a / a.sum()


def _make_cosine(n, c=0.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + c * np.cos(2.0 * np.pi * t)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_profiles(n):
    profiles = []
    profiles.append(('ramp', _make_ramp(n)))
    profiles.append(('bell', _make_bell(n)))
    profiles.append(('u_shape', _make_u_shape(n)))
    profiles.append(('two_peak', _make_two_peak(n)))
    profiles.append(('cosine', _make_cosine(n)))
    profiles.append(('uniform', np.ones(n) / n))
    return profiles


def _nesterov_subgrad(a_init, deadline, rng, n_peaks=5, alpha_init=None,
                       decay=0.9995, perturb_every=5000, perturb_mag=0.005,
                       restart_iters=20000, verbose_every=20000):
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    a_prev = a.copy()
    t_nest = 1.0

    best_a = a.copy()
    best_score = _score_norm(a)
    best_iter = 0

    if alpha_init is None:
        alpha_init = 2.0 / np.sqrt(n)
    alpha = alpha_init

    iteration = 0
    iter_since_improve = 0

    while time.monotonic() < deadline:
        iteration += 1

        # Nesterov momentum
        t_nest = (1.0 + np.sqrt(1.0 + 4.0 * t_nest ** 2)) / 2.0
        beta = (t_nest - 1.0) / t_nest
        a_look = a + beta * (a - a_prev)
        a_look = _simplex_project(a_look)

        # Compute gradient
        c = _fft_convolve(a_look, a_look)
        g = _top_k_subgradient(a_look, c, n_peaks)

        # Update
        a_prev = a.copy()
        a_new = a_look - alpha * g
        a = _simplex_project(a_new)

        # Check improvement (every 50 iters to save time)
        if iteration % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                best_iter = iteration
                iter_since_improve = 0
                if verbose_every and iteration % verbose_every == 0:
                    print(f"      iter {iteration}: score={best_score:.10f}")
            else:
                iter_since_improve += 50

            # Restart if stuck
            if iter_since_improve > restart_iters:
                t_nest = 1.0
                a_prev = a.copy()
                alpha = alpha_init * max(0.1, alpha / alpha_init)
                a = a + perturb_mag * rng.normal(size=n)
                a = _simplex_project(a)
                iter_since_improve = 0
                print(f"      RESTART at iter {iteration}, alpha={alpha:.6f}")

        # Decay
        alpha *= decay

        # Perturbation
        if perturb_every and iteration % perturb_every == 0:
            a = a + perturb_mag * rng.normal(size=n)
            a = _simplex_project(a)

    # Final check
    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _mirror_descent(a_init, deadline, rng, n_peaks=5, alpha_init=None,
                    decay=0.9995, perturb_every=5000, perturb_mag=0.005):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    if alpha_init is None:
        alpha_init = 1.0 / np.sqrt(n)
    alpha = alpha_init

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        c = _fft_convolve(a, a)
        g = _top_k_subgradient(a, c, n_peaks)

        # Multiplicative (mirror) update
        log_alpha_g = -alpha * g
        log_alpha_g = np.clip(log_alpha_g, -50, 50)
        a_new = a * np.exp(log_alpha_g)
        s = a_new.sum()
        if s < 1e-15 or not np.isfinite(s):
            alpha *= 0.5
            continue
        a = a_new / s

        if iteration % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

        alpha *= decay

        if perturb_every and iteration % perturb_every == 0:
            a = a + perturb_mag * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()

    sc = _score_norm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s < 1e-15:
        return np.ones(target_n) / target_n
    return new_a / s

def _coordinate_descent(a, deadline, max_passes=50, init_delta=0.001):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(42)
    delta = init_delta
    stale = 0

    for pass_num in range(max_passes):
        if time.monotonic() > deadline:
            break

        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break

            # Try decrease
            a_test = best_a.copy()
            old_val = a_test[i]
            a_test[i] = max(0.0, old_val - delta)
            s = a_test.sum()
            if s > 1e-15:
                a_test = a_test / s
                sc = _score_norm(a_test)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_test.copy()
                    improved = True
                    continue

            # Try increase
            a_test = best_a.copy()
            a_test[i] = old_val + delta
            s = a_test.sum()
            if s > 1e-15:
                a_test = a_test / s
                sc = _score_norm(a_test)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_test.copy()
                    improved = True

        if improved:
            stale = 0
        else:
            stale += 1
            delta *= 0.7
            if stale > 5 or delta < 1e-7:
                break

    return best_a, best_score
