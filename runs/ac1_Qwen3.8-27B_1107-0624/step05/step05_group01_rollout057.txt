def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    return float(np.dot(a[i_lo:i_hi+1], a[k-i_hi:k-i_lo+1]))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)

    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)

```python
"""
Convex optimization via cutting planes with LP solves for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Exploits the convexity of the problem (max of convex quadratics over simplex)
for guaranteed global convergence.
"""
import numpy as np
from scipy.optimize import linprog, minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(np.abs(c)) / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _conv_peak_info(a):
    """Return (max_val, peak_index, full_convolution)."""
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    return float(c[k_star]), k_star, c


def _grad_fk(a, k):
    """Gradient of f_k(a) = sum_i a[i]*a[k-i]. 
    grad[i] = 2*a[k-i] if k-i in [0,n-1], else 0."""
    n = len(a)
    grad = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo <= i_hi:
        # grad[i] = 2*a[k-i] for i in [i_lo, i_hi]
        # k-i ranges from k-i_hi to k-i_lo
        grad[i_lo:i_hi+1] = 2.0 * a[k-i_hi:k-i_lo+1]
    return grad


def _fk_value(a, k):
    """Value of f_k(a)."""
    n = len(a)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo > i_hi:
        return 0.0
    return float(np.dot(a[i_lo:i_hi+1], a[k-i_hi:k-i_lo+1]))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _quick_lbfgs(a_init, deadline, max_seconds=120):
    """Quick L-BFGS with log-sum-exp to get a better starting point."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    beta_list = [m0 * b for b in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _cutting_plane_lp(a_init, deadline, max_rounds=500):
    """Solving the convex problem via cutting plane method with LP.
    
    Problem: minimize M subject to f_k(a) <= M for all k, sum(a)=1, a>=0.
    Each cut is a linear approximation of f_k(a) <= M at a previous iterate.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Initialize cuts with the top peaks
    c_init = _full_convolve(a_init, a_init)
    peak_indices = np.argsort(c_init)[::-1][:10]

    cuts = []
    for k in peak_indices:
        grad = _grad_fk(a_init, k)
        fk_val = float(c_init[k])
        rhs = float(np.dot(grad, a_init)) - fk_val

        # Since grad @ a_init = 2*fk_val (by the gradient formula), rhs = fk_val
        # Wait: grad @ a = sum_i grad[i]*a[i] = sum_i 2*a[k-i]*a[i] = 2*f_k(a)
        # So rhs = 2*f_k(a_init) - f_k(a_init) = f_k(a_init)
        # Cut: 2 * (a[k-i] * a[i] sum) <= M + f_k(a_init)
        # i.e., 2 * f_k(a) / ... hmm wait.
        # Actually: grad @ a = sum_i grad[i] * a[i] = sum_i 2*a[k-i] * a[i] = 2 * f_k(a)
        # So the cut is: 2 * f_k(a) <= M + rhs = M + f_k(a_init)
        # This is correct as a valid relaxation: since f_k is convex,
        # f_k(a) >= f_k(a_init) + grad/2 @ (a - a_init)
        # = f_k(a_init) + f_k(a) - f_k(a_init) ... that's circular.
        
        # Let me re-derive. The first-order approximation of f_k at a_init:
        # f_k(a) >= f_k(a_init) + (grad_f_k(a_init) / 2) @ (a - a_init)
        # Wait, the gradient I computed is df_k/da, not (1/2)df_k/da.
        # f_k(a) = sum a[i]*a[k-i]. The gradient is [2*a[k-i]] (as I computed).
        # The first-order Taylor: f_k(a) >= f_k(a_init) + grad @ (a - a_init)
        # where grad = [2*a[k-i]]
        # So: f_k(a) >= f_k(a_init) + 2*grad_half @ (a - a_init)
        # where grad_half = [a[k-i]]
        # Alternatively: f_k(a) >= f_k(a_init) + grad @ (a - a_init)
        # where grad[i] = 2*a_init[k-i]

        # f_k(a_init) + grad @ (a - a_init) <= M
        # i.e., grad @ a <= M + grad @ a_init - f_k(a_init)
        # = M + 2*f_k(a_init) - f_k(a_init) = M + f_k(a_init)
        
        # So the cut is: grad @ a <= M + f_k(a_init)
        # where grad[i] = 2*a_init[k-i] (or 0)
        
        # Let me verify: at a = a_init, grad @ a_init = 2*f_k(a_init), 
        # and M + f_k(a_init) should be >= 2*f_k(a_init) for the cut to be feasible.
        # This requires M >= f_k(a_init), which is true since M >= max_k f_k(a_init) >= f_k(a_init).
        
        cuts.append((k, grad, fk_val))
    
    # Now iterate
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        C = len(cuts)
        
        # Build LP:
        # Variables: a[0..n-1], M (total n+1)
        # Objective: minimize M -> c = [0,...,0, 1]
        # Equality: sum(a) = 1 -> A_eq = [1,...,1, 0], b_eq = [1]
        # Inequality: for each cut t: grad_t @ a - M <= rhs_t
        #   A_ub row t: [grad_t[0], ..., grad_t[n-1], -1]
        #   b_ub[t] = rhs_t = f_k_t(a_t)  (as derived above)

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_eq[0, -1] = 0.0
        b_eq = np.array([1.0])
        
        A_ub = np.zeros((C, n + 1))
        b_ub = np.zeros(C)
        for t, (k_t, grad_t, fk_t) in enumerate(cuts):
            A_ub[t, :n] = grad_t
            A_ub[t, n] = -1.0
            b
