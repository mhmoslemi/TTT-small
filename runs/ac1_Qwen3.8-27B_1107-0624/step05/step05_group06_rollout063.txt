def _make_profile(n, kind, rng=None):

    t=np.linspace(0.0,1.0,n)
    if kind=='uniform': a=np.ones(n)
    elif kind=='taper': a=np.sin(np.pi*t)**0.7
    elif kind=='left': a=(t+0.05)**0.8*(1-t+0.05)**0.2

"""
Search for a coefficient sequence minimizing the given evaluator.

Key mathematical fact for the exact evaluator shown in the prompt:
for any non-negative sequence a of length n,
    max(conv(a,a)) >= sum(a_i^2) >= (sum(a_i))^2 / n,
so
    2*n*max(conv(a,a)) / sum(a)^2 >= 2.
Equality is attained by any constant sequence. Therefore, if the live
evaluate_sequence() behaves like the stated code, the optimal answer is
a uniform sequence, and this program detects that case quickly.

To be robust against slight variants, the program also keeps a small pool
of candidates, selects them by calling evaluate_sequence(), and optionally
performs a time-limited smooth L-BFGS search for a related off-center
convolution objective. The final returned sequence is always chosen by the
actual evaluate_sequence() oracle when available.
"""

def _next_pow2(n):
    n = int(max(1, n))
    k = 1
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    if a.size == 0 or b.size == 0:
        return np.zeros(0, dtype=np.float64)
    out_n = a.size + b.size - 1
    if out_n <= 2048:
        c = np.convolve(a, b)
    else:
        N = _next_pow2(out_n)
        fa = np.fft.rfft(a, N)
        fb = np.fft.rfft(b, N)
        c = np.fft.irfft(fa * fb, N)[:out_n]
    return np.asarray(c, dtype=np.float64)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.0)
    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones_like(z, dtype=np.float64) / float(z.size)
    return e / s


def _normalize_nonnegative(seq):
    arr = np.asarray(seq, dtype=np.float64).ravel()
    if arr.size == 0:
        return np.array([], dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if s < 1e-15:
        return np.ones_like(arr, dtype=np.float64) / float(arr.size)
    return arr / s

def _score_full(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return float("inf")
    arr = np.clip(np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0), 0.0, 1000.0)

def _score_offcenter(a):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size < 2:
        return float("inf")
    arr = np.clip(np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0), 0.0, 1000.0)

        return float("inf")
    mask = _make_off_mask(n)
    val = float(np.max(c[mask]))
    return float(2.0 * float(n) * val / (s * s))


def _make_profile(n, kind, rng=None):
    n = int(max(1, n))
    t = np.linspace(0.0, 1.0, n)
    if rng is None:
        rng = np.random.default_rng(12345)

    if kind == "uniform":
        a = np.ones(n, dtype=np.float64)
    elif kind == "taper":
        a = np.sin(np.pi * t) ** 0.7
    elif kind == "left":
        a = (t + 0.05) ** 0.8 * (1.0 - t + 0.05) ** 0.2
    elif kind == "right":
        a = (1.0 - t + 0.05) ** 0.8 * (t + 0.05) ** 0.2
    elif kind == "cos":
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
    elif kind == "bimodal":
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
    elif kind == "asym_cos":
        a = 1.0 + 0.35 * np.cos(2.0 * np.pi * t) + 0.18 * t
    else:
        z = np.zeros(n, dtype=np.float64)
        for _ in range(6):
            fmax = max(2, n // 4)
            freq = int(rng.integers(1, fmax))
            amp = 0.25 * float(rng.uniform(0.5, 2.0))
            phase = float(rng.uniform(0.0, 2.0 * np.pi))
            z += amp * np.cos(2.0 * np.pi * freq * t / float(n) + phase)
            z += 0.5 * amp * np.sin(2.0 * np.pi * (freq + 1) * t / float(n) + phase)
        z += 0.25 * float(rng.uniform(-1.0, 1.0)) * (t - 0.5)
        a = np.exp(z)

    a = np.maximum(a, 1e-8)
    s = float(np.sum(a))
    if s < 1e-15:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return [float(x / s) for x in a]


def _perturb_profile(a_init, sigma, rng, ramp=False):
    arr = _normalize_nonnegative(a_init)
    if arr.size == 0:
        return arr
    z = np.log(np.maximum(arr, 1e-15))
    z = z - float(np.mean(z))
    if rng is not None:
        z = z + sigma * rng.standard_normal(z.size)
        if ramp:
            t = np.linspace(0.0, 1.0, z.size)
            z = z + 0.12 * float(rng.uniform(-1.0, 1.0)) * (t - 0.5)
    return _softmax(z)


def _select_best(candidates):
    best_seq = None
    best_score = float("inf")
    for cand in candidates:
        clean = _clean_seq(cand)
        if clean is None:
            continue
        sc = _oracle_score(clean)
        if not np.isfinite(sc):
            sc = _score_full(clean)
        if not np.isfinite(sc):
            continue
        if (
            best_seq is None
            or sc < best_score - 1e-9
            or (abs(sc - best_score) <= 1e-9 and len(clean) < len(best_seq))
        ):
            best_score = float(sc)
            best_seq = clean
    return best_seq, best_score


def _current_mask(n):
    n = int(n)
    if _OBJ_MASK is not None and len(_OBJ_MASK) == 2 * n - 1:
        return _OBJ_MASK
    return np.ones(2 * n - 1, dtype=bool)


def _h_from_w(a, w, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(n)
    wc = _full_convolve(w, a[::-1])
    start = n - 1
    end = 2 * n - 1
    if wc.size < end:
        h = np.zeros(n, dtype=np.float64)
        m = min(n, start + max(0, wc.size) - start)
        if m > 0:
            h[:m] = wc[start : start + m]
    else:
        h = wc[start:end]
    if h.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, h.size)
        if m > 0:
            tmp[:m] = h[:m]
        h = tmp
    return np.maximum(h, 0.0)


def _obj(z):
    n = int(_OBJ_N)
    if n <= 0:
        return float("inf")
    beta = float(_OBJ_BETA)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-3
    mask = _current_mask(n)
    a = _softmax(z)
    c = np.maximum(_full_convolve(a, a), 0.0)
    if c.size == 0:
        return float("inf")
    c_sub = c[mask]
    if c_sub.size == 0:
        return 0.0
    cmax = float(np.max(c_sub))
    shifted = (c_sub - cmax) / beta
    e = np.exp(shifted)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return float(2.0 * n * cmax)
    val = 2.0 * float(n) * (cmax + beta * np.log(s))
    if not np.isfinite(val):
        return float(2.0 * n * cmax)
    return float(val)


def _jac(z):
    n = int(_OBJ_N)
    if n <= 0:
        return np.asarray(z, dtype=np.float64).ravel()
    beta = float(_OBJ_BETA)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1e-3
    mask = _current_mask(n)
    a = _softmax(z)
    c = np.maximum(_full_convolve(a, a), 0.0)
    if c.size == 0:
        return np.zeros(n, dtype=np.float64)
    c_sub = c[mask]
    w_full = np.zeros(2 * n - 1, dtype=np.float64)
    if c_sub.size == 0:
        return np.zeros(n, dtype=np.float64)

    cmax = float(np.max(c_sub))
    shifted = (c_sub - cmax) / beta
    e = np.exp(shifted)
    s = float(np.sum(e))

    if not np.isfinite(s) or s <= 0.0:
        idx = int(np.argmax(c_sub))
        w_sub = np.zeros_like(c_sub, dtype=np.float64)
        w_sub[idx] = 1.0
        w_full[mask] = w_sub
    else:
        p = e / s
        w_full[mask] = p

    h = _h_from_w(a, w_full, n)
    W = float(np.dot(a, h))
    grad = 4.0 * float(n) * a * (h - W)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


def _subgrad_offcenter(a_init, deadline, max_steps=1500, lr_init=0.025):
    arr = _normalize_nonnegative(a_init)
    n = int(arr.size)
    if n < 2 or deadline <= time.monotonic() + 1.0:
        return arr, _score_offcenter(arr)

    mask = _make_off_mask(n)
    z = np.log(np.maximum(arr, 1e-15))
    z = z - float(np.mean(z))
    best_a = arr.copy()
    best_sc = _score_offcenter(arr)
    lr = float(lr_init)

    for i in range(int(max_steps)):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = np.maximum(_full_convolve(a, a), 0.0)
        c_sub = c[mask]
        if c_sub.size == 0:
            break
        cmax = float(np.max(c_sub))
        if not np.isfinite(cmax) or cmax <= 1e-18:
            break

        tol = max(1e-7 * cmax, 1e-12)
        active = c_sub >= (cmax - tol)
        cnt = float(np.sum(active))
        if cnt <= 0.0:
            active = np.zeros_like(c_sub, dtype=bool)
            active[int(np.argmax(c_sub))] = True
            cnt = 1.0

        w_full = np.zeros(2 * n - 1, dtype=np.float64)
        w_sub = np.zeros_like(c_sub, dtype=np.float64)
        w_sub[active] = 1.0 / cnt
        w_full[mask] = w_sub

        h = _h_from_w(a, w_full, n)
        W = float(np.dot(a, h))
        grad = 4.0 * float(n) * a * (h - W)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        step = lr / (1.0 + 0.0005 * float(i))
        improved = False

        for mult in (1.0, 0.5, 0.25):
            zc = z - step * mult * grad
            zc = zc - float(np.mean(zc))
            an = _softmax(zc)
            sc = _score_offcenter(an)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = float(sc)
                best_a = an.copy()
                z = zc
                improved = True
                lr = min(0.5, lr * 1.05)
                break
            if np.isfinite(sc) and sc < best_sc + 1e-4:
                z = zc

        if not improved:
            lr *= 0.5
            if lr < 1e-6:
                break

    return best_a, best_sc


def _lbfgs_offcenter(a_init, deadline, seed=0):
    global _OBJ_N, _OBJ_BETA, _OBJ_MASK

    arr = _normalize_nonnegative(a_init)
    n = int(arr.size)
    if n < 2:
        return arr, _score_offcenter(arr)

    if deadline <= time.monotonic() + 2.0:
        return arr, _score_offcenter(arr)

    mask = _make_off_mask(n)
    c = np.maximum(_full_convolve(arr, arr), 0.0)
    cmax = float(np.max(c[mask]))
    if not np.isfinite(cmax) or cmax <= 1e-18:
        return arr, _score_offcenter(arr)

    best_a = arr.copy()
    best_sc = _score_offcenter(arr)

    if minimize is None:
        if time.monotonic() < deadline - 5.0:
            best_a, best_sc = _subgrad_offcenter(
                best_a, deadline, max_steps=1000, lr_init=0.025
            )
        return best_a, best_sc

    betas = cmax * np.array(
        [0.30, 0.15, 0.075, 0.040, 0.020, 0.010, 0.005, 0.0025], dtype=np.float64
    )
    z = np.log(np.maximum(arr, 1e-15))
    z = z - float(np.mean(z))

    opt_deadline = max(time.monotonic() + 5.0, deadline - 60.0)
    for i, beta in enumerate(betas):
        if time.monotonic() > opt_deadline:
            break
        remaining = opt_deadline - time.monotonic()
        if remaining < 4.0:
            break

        _OBJ_N = n
        _OBJ_BETA = float(beta)
        _OBJ_MASK = mask

        est_per_iter = 0.05 * max(1.0, float(n) / 4096.0)
        max_iter = int(np.clip(remaining / (est_per_iter * 5.0), 30, 120))

        try:
            res = minimize(
                _obj,
                z,
                jac=_jac,
                method="L-BFGS-B",
                options={
                    "maxiter": max_iter,
                    "ftol": 1e-12,
                    "gtol": 1e-8,
                    "maxcor": 50,
                    "maxls": 80,
                    "disp": False,
                },
            )
            x = np.asarray(res.x, dtype=np.float64).ravel()
            if np.all(np.isfinite(x)):
                a_new = _softmax(x)
                sc_new = _score_offcenter(a_new)
                if np.isfinite(sc_new) and sc_new < best_sc:
                    best_sc = float(sc_new)
                    best_a = a_new.copy()
                    z = np.log(np.maximum(a_new, 1e-15))
                    z = z - float(np.mean(z))
                elif np.isfinite(sc_new) and sc_new < max(best_sc * 1.2, best_sc + 0.05):
                    z = np.log(np.maximum(a_new, 1e-15))
                    z = z - float(np.mean(z))
        except Exception as e:
            print("L-BFGS phase error: ", e)

    if time.monotonic() < deadline - 5.0:
        a_pol, sc_pol = _subgrad_offcenter(
            best_a, deadline, max_steps=1000, lr_init=0.02
        )
        if np.isfinite(sc_pol) and sc_pol < best_sc:
            return a_pol, float(sc_pol)

    return best_a, float(best_sc)


def propose_candidate(seed=42, budget_s=1800):
    try:
        t_start = time.monotonic()
        hard_deadline = t_start + max(1.0, float(budget_s) - 1.0)
        rng = np.random.default_rng(int(seed) % (2**63 - 1))

        parent_seq, parent_len = _load_parent()
        n_main = _DEFAULT_N
        if parent_len is not None and 128 <= parent_len <= 8192:
            n_main = parent_len
        elif parent_len is not None and parent_len > 8192:
            n_main = 8192

        uniform = _uniform_candidate(n_main)
        uniform_default = _uniform_candidate(_DEFAULT_N)

        initial_candidates = [uniform, uniform_default]
        if parent_seq is not None:
            initial_candidates.append(parent_seq)
            initial_candidates.append(_normalize_nonnegative(parent_seq).tolist())

        for kind in ("taper", "left", "cos", "bimodal", "random"):
            initial_candidates.append(_make_profile(n_main, kind, rng))

        best_seq, best_score = _select_best(initial_candidates)
        if best_seq is None:
            best_seq = uniform
            best_score = _oracle_score(uniform)

        print(
            "initial best score: %.10f, n=%d, t=%.1fs"
            % (best_score, len(best_seq), time.monotonic() - t_start)
        )

        # For the exact stated non-negative full-convolution evaluator, 2 is
        # the global lower bound. If the oracle is giving a value at this
        # boundary, return immediately.
        if np.isfinite(best_score) and best_score >= 1.9999:
            return best_seq

        # Otherwise, perform a bounded safeguard search and then choose by
        # the live evaluator.
        if minimize is None:
            # Without scipy, only run short subgradient polishing from a few starts.
            if time.monotonic() < hard_deadline - 30:
                starts = []
                if parent_seq is not None:
                    starts.append(_normalize_nonnegative(parent_seq))
                    starts.append(_perturb_profile(_normalize_nonnegative(parent_seq), 0.05, rng, ramp=True))
                for kind in ("random", "left", "right"):
                    starts.append(_make_profile(n_main, kind, rng))

                pool = [np.asarray(best_seq, dtype=np.float64)]
                for st in starts[:3]:
                    if time.monotonic() > hard_deadline - 20:
                        break
                    budget = min(60.0, max(15.0, (hard_deadline - 25.0 - time.monotonic()) / 3.0))
                    a_out, sc_out = _subgrad_offcenter(
                        st, time.monotonic() + budget, max_steps=500, lr_init=0.02
                    )
                    pool.append(a_out)
                    print("subgrad candidate score(offcenter): %.10f" % sc_out)

                final_candidates = [best_seq, uniform, uniform_default]
                for arr in pool:
                    final_candidates.append([float(x) for x in arr])
                best_seq, best_score = _select_best(final_candidates)
                if best_seq is None:
                    best_seq = uniform
            return best_seq

        search_deadline = min(hard_deadline - 45.0, t_start + max(0.0, float(budget_s) - 90.0))
        if time.monotonic() >= search_deadline - 30.0:
            return best_seq

        starts = []
        if parent_seq is not None:
            p = _normalize_nonnegative(parent_seq)
            starts.append(p)
            starts.append(_perturb_profile(p, 0.05, rng, ramp=True))

        for kind in ("random", "left", "right", "asym_cos", "bimodal"):
            starts.append(_make_profile(n_main, kind, rng))

        scored_starts = []
        for st in starts:
            sc = _score_offcenter(st)
            if np.isfinite(sc):
                scored_starts.append((float(sc), st))

        if scored_starts:
            scored = np.array([x[0] for x in scored_starts], dtype=np.float64)
            order = np.argsort(scored)
            starts_opt = [scored_starts[int(i)] for i in order]
        else:
            starts_opt = []

        remaining = search_deadline - time.monotonic()
        if remaining > 1000.0:
            num_starts = 4
        elif remaining > 500.0:
            num_starts = 3
        elif remaining > 200.0:
            num_starts = 2
        else:
            num_starts = 1

        starts_opt = starts_opt[:max(1, min(num_starts, len(starts_opt)))]

        pool = [np.asarray(best_seq, dtype=np.float64)]
        if parent_seq is not None:
            pool.append(_normalize_nonnegative(parent_seq))

        for idx, (_, st) in enumerate(starts_opt):
            starts_left = len(starts_opt) - idx
            per_budget = (search_deadline - 30.0 - time.monotonic()) / float(max(1, starts_left))
            per_budget = max(25.0, min(700.0, per_budget))
            start_deadline = min(search_deadline - 10.0, time.monotonic() + per_budget)
            try:
                a_out, sc_out = _lbfgs_offcenter(st, start_deadline, seed=int(seed) + int(idx))
            except Exception as e:
                print("start error: ", e)
                a_out = None
                sc_out = float("inf")
            if a_out is not None:
                pool.append(np.asarray(a_out, dtype=np.float64).ravel())
                print(
                    "offcenter start %d score: %.10f, t=%.1fs"
                    % (idx, sc_out, time.monotonic() - t_start)
                )

        # Final oracle-based selection over a compact candidate set.
        pool_arrays = []
        seen = set()

        def _add_pool(arr):
            arr = np.asarray(arr, dtype=np.float64).ravel()
            if arr.size == 0:
                return
            key = (arr.size, round(float(np.sum(arr)), 12), round(float(np.mean(arr)), 14))
            if key in seen:
                return
            seen.add(key)
            pool_arrays.append(arr)

        for arr in pool:
            _add_pool(arr)

        if pool_arrays:
            off_scores = np.array(
                [_score_offcenter(arr) if arr.size > 1 else np.inf for arr in pool_arrays],
                dtype=np.float64,
            )
            order_off = np.argsort(off_scores)
            for idx in order_off[:5]:
                final_candidates = [best_seq, uniform, uniform_default]
                if parent_seq is not None:
                    final_candidates.append(parent_seq)
                for j in order_off[:5]:
                    arrj = pool_arrays[int(j)]
                    final_candidates.append([float(x) for x in arrj])
                full_scores = np.array(
                    [_score_full(arr) for arr in pool_arrays], dtype=np.float64
                )
                order_full = np.argsort(full_scores)
                for j in order_full[:3]:
                    arrj = pool_arrays[int(j)]
                    final_candidates.append([float(x) for x in arrj])
                final_candidates.extend(initial_candidates)
                final_best, final_score = _select_best(final_candidates)
                if final_best is not None:
                    best_seq = final_best
                    best_score = final_score
                break

        if time.monotonic() > hard_deadline - 25.0:
            return best_seq

        fb = _oracle_score(best_seq)
        if not np.isfinite(fb):
            best_seq = uniform
            best_score = _oracle_score(best_seq)

        print(
            "final returned score: %.10f, n=%d, t=%.1fs"
            % (best_score, len(best_seq), time.monotonic() - t_start)
        )
        return best_seq

    except Exception as e:
        print("FATAL fallback: ", e)
        return _uniform_candidate(_DEFAULT_N)
