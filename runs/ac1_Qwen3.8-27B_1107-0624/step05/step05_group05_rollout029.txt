"""
Optimized step function search with corrected gradient computation.
Uses mirror descent in log-space with proper gradient for the smoothed
max-convolution objective, combined with LP-based large-step updates
and Adam polishing. Key improvement: correct gradient that accounts
for both correlation directions (S1 and S2 terms).
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, w):
    """Compute the correct gradient of sum_k w_k * c_k with respect to a.
    grad_j = 2 * (S1_j + S2_j) where:
    S1_j = sum_i a_i * w_{j+i}  (correlation with w shifted forward)
    S2_j = sum_{i=1}^{j} a_i * w_{j-i}  (correlation with w shifted backward)
    """
    n = len(a)
    # S1_j = (convolve(flip(a), w))_{j+n-1}
    flip_a = a[::-1]
    conv_fw = _full_convolve(flip_a, w)
    S1 = conv_fw[n-1:2*n-1]  # length n, S1_j at index j
    
    # S2_j = (convolve(a, w))_j - a_0 * w_j
    conv_aw = _full_convolve(a, w)
    S2 = conv_aw[:n] - a[0] * w[:n]
    
    return 2.0 * (S1 + S2)


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M


def _adam_optimize(a_init, deadline, beta_start, beta_end, lr=0.01, max_iter=500000):
    """Adam optimization in log-space with beta annealing and correct gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Anneal beta
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        
        # Compute softmax weights
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Compute correct gradient
        grad = _compute_gradient(a, w)
        
        # Natural gradient (mirror descent step in z-space)
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            if t % 5000 == 0:
                if best_score < prev_best * 0.999:
                    current_lr = min(0.05, current_lr * 1.15)
                else:
                    current_lr = max(0.001, current_lr * 0.9)
                prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

"""
Optimized step function search with corrected gradient computation.
Key improvement over previous: proper gradient accounting for both
correlation directions (S1 and S2 terms), combined with Adam in log-space,
beta annealing, LP-based large steps, and multi-restart exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, w):
    """Compute correct gradient of sum_k w_k * c_k w.r.t. a.
    grad_j = 2 * (S1_j + S2_j) where S1 is forward correlation, S2 is backward.
    """
    n = len(a)
    flip_a = a[::-1]
    conv_fw = _full_convolve(flip_a, w)
    S1 = conv_fw[n - 1:2 * n - 1]
    conv_aw = _full_convolve(a, w)
    S2 = conv_aw[:n] - a[0] * w[:n]
    return 2.0 * (S1 + S2)


def _compute_gradient_simple(a, w):
    """Simpler (but still correct) gradient via direct correlation."""
    n = len(a)
    # Use the fact that grad_j = 2 * sum_k w_k * a_{|k-j|} for |k-j| < n
    # This equals 2 * (cross_correlation of a and w) evaluated appropriately
    # Alternative: compute via FFT of the full expression
    # Actually let's just use the two-convolution method which is O(n log n)
    flip_a = a[::-1]
    S1 = _full_convolve(flip_a, w)[n - 1:2 * n - 1]
    conv_aw = _full_convolve(a, w)
    S2 = conv_aw[:n] - a[0] * w[:n]
    return 2.0 * (S1 + S2)


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=25):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=25):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f.copy(), M


def _adam_optimize(a_init, deadline, beta_start, beta_end, lr=0.01, max_iter=500000):
    """Adam optimization in log-space with beta annealing and correct gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    log_ratio = np.log(max(beta_end, 1e-12) / max(beta_start, 1e-12))

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Exponential annealing of beta
        frac = t / max_iter
        beta = beta_start * np.exp(log_ratio * frac)
        if beta < 1e-12:
            beta = 1e-12

        # Compute softmax weights over c
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        # Compute correct gradient
        grad = _compute_gradient(a, w)

        # Natural gradient for simplex constraint
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if t % 1000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            if t % 10000 == 0:

                    current_lr = max(0.0005, current_lr * 0.85)
                else:
                    current_lr = max(0.001, current_lr * 0.95)
                prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

            beta_start=0.002 * max_cb,
            beta_end=0.00005 * max_cb,
            lr=0.006,
            max_iter=1000000
        )
        full_sc = _score(a_polished)
        print(f"Phase 3 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s", flush=True)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_polished.copy()

    # Phase 4: Multiple perturbed restarts with Adam (1200 - 1550s)
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end:
        n_restarts = 6
        attempt_time = (phase4_end - time.monotonic()) / n_restarts - 5
        for attempt in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            sigma = 0.015 + 0.015 * attempt
            p_start = _perturb(best_seq, sigma, rng)
            c_p = _full_convolve(p_start, p_start)
            max_cp = float(c_p.max())
            dl = min(phase4_end - 5, time.monotonic() + max(30.0, attempt_time))
            a_p, sc_p = _adam_optimize(
                p_start, dl,
                beta_start=0.001 * max_cp,
                beta_end=0.00002 * max_cp,
                lr=0.008,
                max_iter=200000
            )
            full_sc = _score(a_p)
            print(f"  Restart[{attempt}] sigma={sigma:.4f}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a_p.copy()

    # Phase 5: Try a different sequence length if time permits (1550 - 1700s)
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 30:
        for n_try in [4096, 1024]:
            if time.monotonic() > phase5_end - 30:
                break
            print(f"  Trying n={n_try}...", flush=True)
            a_new, sc_new = _try_different_n(best_seq, n_try, min(phase5_end - 5, time.monotonic() + 50))
            if sc_new is not None:
                full_sc = _score(a_new)
                print(f"  n={n_try}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s", flush=True)
                if np.isfinite(full_sc) and full_sc < best_score:
                    best_score = full_sc
                    best_seq = a_new.copy()
                    # Update LP indices for new n
                    n = len(best_seq)
                    rows, cols = _build_lp_indices(n)

    # Phase 6: Final ultra-fine Adam (1700 - deadline)
    if time.monotonic() < deadline - 10:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        a_final, sc_final = _adam_optimize(
            best_seq, deadline - 5,
            beta_start=0.0002 * max_cf,
            beta_end=0.000005 * max_cf,
            lr=0.003,
            max_iter=500000
        )
        full_sc = _score(a_final)
        print(f"Phase 6 final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s", flush=True)
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_final.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"Eval error: {ex}", flush=True)
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
