def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-300 or not np.isfinite(s):

"""
Multi-scale spectral optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines parametric initialization, FFT-based subgradient methods with spectral

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _subgradient_conv_max(a, S, top_k=20):
    """Compute subgradient of max_k (a*a)[k] using top-k active set."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    # Get top-k indices
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    # Use weights proportional to proximity to max
    vals = c[top_idx]
    weights = np.exp((vals - cm) / max(1e-10, cm * 0.01))
    weights = weights / weights.sum()
    # Build w vector
    w = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        w[idx] = weights[i]
    # Compute subgradient: g[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    N = _next_pow2(3 * n - 2)
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(W * np.conj(A)).real[:n]
    return 2.0 * cross, top_idx, cm


def _spectral_flatten(a, strength=0.3):
    """Apply spectral flattening to reduce peak autocorrelation."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    power = np.abs(A) ** 2
    # Compute target: geometric mean of power (log-flat)
    log_power = np.log(power + 1e-30)
    log_mean = np.mean(log_power)
    # Blend toward flat
    log_power_new = (1 - strength) * log_power + strength * log_mean
    power_new = np.exp(log_power_new)
    # Keep phase, adjust amplitude
    A_new = A * (np.sqrt(power_new) / (np.abs(A) + 1e-30))
    # Transform back
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()

    eta = 0.1
    m = 0.9
    grad_ema = np.zeros(n)
    
    for it in range(max_iter):

        vals = c[top_idx]

        weights = weights / weights.sum()
        w = np.zeros(2 * n - 1)

        grad = 2.0 * cross
        
        # Normalize gradient

        # Update with momentum

def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    """L-BFGS with fine beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 80.0)

        maxiter = max(300, int(time_per * 25))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):

                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score

def _coord_descent_refined(a_init, deadline, max_rounds=100):
    """Faster coordinate descent with line search."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    deltas = [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in deltas:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _parametric_init(n, rng):
    """Generate diverse parametric initializations."""
    t = np.linspace(0, 1, n)
    candidates = []
    
    # Exponential decay from left
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-alpha * t)
        candidates.append(a / a.sum())
    
    # Exponential decay from right
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-alpha * (1 - t))
        candidates.append(a / a.sum())
    
    # Polynomial decay
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = (1 - t) ** p + 0.01
        candidates.append(a / a.sum())
        a = t ** p + 0.01
        candidates.append(a / a.sum())
    
    # Gaussian centered at various positions
    for center in [0.2, 0.3, 0.5, 0.7, 0.8]:
        for sigma in [0.1, 0.15, 0.2, 0.3]:
            a = np.exp(-((t - center) / sigma) ** 2)
            candidates.append(a / a.sum())
    
    # Cosine windows
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * (t - 0.5)), 0) ** p + 0.01
        candidates.append(a / a.sum())
    
    # Skewed cosine
    for skew in [0.3, 0.5, 0.7]:
        a = np.maximum(np.cos(np.pi * (t - skew)), 0) ** 2 + 0.01
        candidates.append(a / a.sum())
    
    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 10, size=5)
        coeffs = rng.normal(size=5)
        a = 1.0 + sum(c * np.cos(2 * np.pi * f * t + rng.uniform(0, 2*np.pi)) 
                      for c, f in zip(coeffs, freqs))
        a = np.maximum(a, 0.01)
        candidates.append(a / a.sum())
    
    return candidates


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _nesterov_subgradient(a_init, deadline, max_iter=30000):
    """Nesterov-accelerated subgradient method."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_prev = z.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    momentum = 0.9
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Subgradient with top-k
        top_k = min(50, 2 * n - 1)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        vals = c[top_idx]
        temp = max(1e-10, cm * 0.05)
        weights = np.exp((vals - cm) / temp)
        weights = weights / weights.sum()
        w = np.zeros(2 * n - 1)
        for i, idx in enumerate(top_idx):
            w[idx] = weights[i]
        
        N = _next_pow2(3 * n - 2)
        W = np.fft.fft(w, N)
        A = np.fft.fft(a, N)
        cross = np.fft.ifft(W * np.conj(A)).real[:n]
        grad = 2.0 * cross
        
        # Normalize
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Nesterov update
        t = it + 1
        momentum = (1 + 1 / np.sqrt(t)) / 3  # Nesterov momentum
        z_new = z - (0.5 / np.sqrt(t)) * grad
        z_new = momentum * z_new + (1 - momentum) * z_prev
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        z_prev = z
        z = z_new
        
        a_new = _softmax_fn(z)
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        if it % 5000 == 0:
            print(f"    Nesterov iter {it}: score={sc:.10f}, best={best_score:.10f}")
    
    return best_a, best_a and best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        t = np.linspace(0, 1, n_init)
        parent = np.exp(-2.0 * t) + 0.1
        parent = parent / parent.sum()
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    print(f"\nStarting optimization, target n up to 4096")

    # Phase 1: Multi-scale initialization and subgradient (300s)
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\nPhase 1: Multi-scale subgradient (deadline: {phase1_end - t_start:.0f}s)")
    
    # Start at the parent n and try subgradient
    a_cur = parent.copy()
    sc_cur = _score_norm_fn(a_cur)
    
    # Try spectral flattening first
    a_flat = _spectral_flatten(a_cur, strength=0.3)
    sc_flat = _score_norm_fn(a_flat)
    if sc_flat < sc_cur:
        update_best(a_flat, sc_flat, "spectral_flat")
        a_cur = a_flat
    else:
        a_flat2 = _spectral_flatten(a_cur, strength=0.15)
        sc_flat2 = _score_norm_fn(a_flat2)
        if sc_flat2 < sc_cur:
            update_best(a_flat2, sc_flat2, "spectral_flat_mild")
            a_cur = a_flat2
    
    # Subgradient method
    p_best, sc_best = _run_subgradient_spectral(a_cur, min(phase1_end, t_start + 200), rng)
    update_best(p_best, sc_best, "subgradient_phase1")
    
    # Nesterov subgradient
    if time.monotonic() < phase1_end - 50:
        p_best, sc_best = _nesterov_subgradient(best_seq, min(phase1_end - 10, t_start + 250))
        update_best(p_best, sc_best, "nesterov_phase1")
    
    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS continuation (500s)
    phase2_end = min(deadline, t_start + 800.0)
    print(f"\nPhase 2: L-BFGS continuation (n={len(best_seq)})")

    n_restarts = 4
    per_restart_time = (phase2_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 30:
            break

        if r == 0:
            a_cur = best_seq.copy()
        elif r == 1:
            # Spectral flatten then optimize
            a_cur = _spectral_flatten(best_seq, strength=0.1)
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.03 * (r + 1)
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)

        per_dl = min(phase2_end - 10, time.monotonic() + per_restart_time)
        p_best, sc_best = _run_lbfgs_continuation(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}_n{len(a_cur)}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Try different n values (400s)
    phase3_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 3: Multi-n exploration")

    # Try upsampling to larger n
    for n_target in [4096, 2048]:
        if time.monotonic() > phase3_end - 100:
            break
        if len(best_seq) == n_target:
            continue
        p_interp = _interp_profile(best_seq, n_target)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_target}")
        
        if time.monotonic() < phase3_end - 60:
            per_dl = min(phase3_end - 20, time.monotonic() + 150.0)
            p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl)
            update_best(p_best, sc_best, f"lbfgs_n{n_target}")
    
    # Try parametric inits at n=4096
    if time.monotonic() < phase3_end - 80:
        n_p = 4096
        inits = _parametric_init(n_p, rng)
        # Evaluate all inits quickly
        best_init = None
        best_init_sc = np.inf
        for init in inits:
            sc = _score_norm_fn(init)
            if sc < best_init_sc:
                best_init_sc = sc
                best_init = init
        if best_init is not None:
            update_best(best_init, best_init_sc, "parametric_init")
            if time.monotonic() < phase3_end - 40:
                per_dl = min(phase3_end - 10, time.monotonic() + 100.0)
                p_best, sc_best = _run_lbfgs_continuation(best_init, per_dl)
                update_best(p_best, sc_best, "lbfgs_parametric")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Fine L-BFGS with tiny beta (300s)
    phase4_end = min(deadline, t_start + 1580.0)
    print(f"\nPhase 4: Fine optimization")

    if time.monotonic() < phase4_end and best_seq is not None:
        # Tiny beta L-BFGS
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        
        for beta_frac in [1e-4, 5e-5, 1e-5, 5e-6]:
            if time.monotonic() > phase4_end - 20:
                break
            beta_tiny = beta_frac * max_c_final
            per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.4)
            p_best, sc_best = _single_beta_lbfgs(best_seq, beta_tiny, per_dl, maxiter=8000)
            update_best(p_best, sc_best, f"tiny_beta_{beta_frac}")
            if p_best is not None and sc_best < best_score:
                best_seq = p_best
                c_final = _full_convolve(best_seq, best_seq)
                max_c_final = float(c_final.max())

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Coordinate descent polish (remaining)
    phase5_end = deadline
    print(f"\nPhase 5: Coordinate descent polish")

    if time.monotonic() < phase5_end and best_seq is not None:
        per_dl = min(phase5_end - 15, time.monotonic() + (phase5_end - time.monotonic()) * 0.7)
        p_best, sc_best = _coord_descent_refined(best_seq, per_dl, max_rounds=50)
        update_best(p_best, sc_best, "coord_descent")

        # Final micro-opt with L-BFGS
        if time.monotonic() < phase5_end - 15:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_ultra = 1e-5 * max_c_final
            per_dl2 = min(phase5_end - 5, time.monotonic() + 40.0)
            p_best2, sc_best2 = _single_beta_lbfgs(best_seq, beta_ultra, per_dl2, maxiter=5000)
            update_best(p_best2, sc_best2, "ultra_final")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    t = np.linspace(0, 1, n_fb)
    fallback = np.exp(-2.0 * t) + 0.1
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
