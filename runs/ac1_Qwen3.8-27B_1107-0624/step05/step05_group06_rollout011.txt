"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with adaptive log-sum-exp smoothing, shrinkage post-processing,
multi-n exploration, and diverse perturbation restarts.
Key improvements over baseline: adaptive beta, support reduction, multi-n search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(np.clip(shifted, -700, 0))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(z, a, c, n, tol_frac=0.002):
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):

def _newton_active_set(a_init, deadline, max_iter=200):
    """Newton's method on the active set subproblem (with sum=1 constraint)."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        n_active = int(active.sum())

        # Build Hessian for the active set subproblem:
        # minimize sum_{k in active} b_k  subject to sum(a)=1, a>=0
        # b_k = sum_i a[i]*a[k-i], gradient: 2*(P_k a)_j where (P_k a)_j = a[k-j]
        # Hessian: 2 * sum_{k in active} P_k (symmetric part)
        # Actually Hessian of b_k is 2*P_k where (P_k)_{i,j} = 1 if i+j=k
        # So H = 2 * sum_{k in active} P_k

        # Compute Hessian * v efficiently using FFT:
        # (Hv)_i = 2 * sum_{k in active} sum_j P_k(i,j) * v_j
        # = 2 * sum_{k in active} v[k-i] (if 0<=k-i<=n-1)
        # = 2 * (conv(v, active))[n-1-i]... let me think more carefully.

        # (P_k v)_i = v[k-i] if 0 <= k-i <= n-1, else 0
        # sum_k (P_k v)_i = sum_k v[k-i] where 0<=k<=2n-2 and 0<=k-i<=n-1
        # = sum_{j: 0<=j<=n-1} v[j] * (number of k such that k=j+i and k in active)
        # = sum_j v[j] * active[j+i] (where active is defined on 0..2n-2)

        # So (Hv)_i = 2 * sum_j v[j] * active[i+j]
        # This is 2 * (conv_flip(v, active))[i] where conv_flip reverses one input.
        # Actually: sum_j v[j] * active[i+j] is a cross-correlation.
        # Let's compute it via FFT: it's the "correlation" of v and active.
        # corr(i) = sum_j v[j] * active[i+j]

        # For the Newton step, we need to solve (H + mu*I) dx = -grad + lambda*1
        # where lambda enforces sum(dx) = 0 (since we're on the sum=1 simplex).

        # Let's just use the gradient and do a line search instead of full Newton
        # (computing H*v is O(n log n) via FFT, but inverting H is harder).

        grad = _gradient_active(np.log(np.maximum(a, 1e-30)), a, c, n, tol_frac=0.001)
        # Project gradient onto the tangent space of {a: sum(a)=1}
        grad_t = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad_t)
        if gnorm < 1e-12:
            break

        # Backtracking line search
        alpha = 1.0
        a_new = a.copy()
        f_old = float(c.max())
        for ls in range(30):
            a_try = a - alpha * grad_t
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                alpha *= 0.5
                continue
            a_try = a_try / s_try
            c_try = _full_convolve(a_try, a_try)
            f_try = float(c_try.max())
            if f_try < f_old - 1e-12 * alpha * gnorm:
                a_new = a_try
                break
            alpha *= 0.5

        a = a_new
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def _shrinkage(a, deadline, n_try=50):
    """Try zeroing out the smallest elements and re-optimizing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    # Sort elements by value (ascending) and try zeroing the smallest ones
    a_sorted_idx = np.argsort(a)

    # Try zeroing bottom 1%, 2%, 5%, 10%, 20%
    fractions = [0.01, 0.02, 0.05, 0.10, 0.15, 0.20]
    for frac in fractions:
        if time.monotonic() > deadline:
            break
        n_zero = max(1, int(n * frac))
        a_try = a.copy()
        a_try[a_sorted_idx[:n_zero]] = 0.0
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / a_try.sum()
        sc_try = _score(a_try)
        if sc_try < best_sc - 1e-10:
            best_sc = sc_try
            best_a = a_try.copy()
            # Re-optimize from this point
            z = np.log(np.maximum(best_a, 1e-30))
            z = z - z.mean()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "skewed":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * t)
        a = a + 0.1
        return a / a.sum()
    elif kind == "cos2":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "wheatstone":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):

def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    t = np.arange(n)
    for _ in range(n_modes):

        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block_swap(a, rng, n_blocks=4):
    """Swap random blocks of the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    block_size = n // (2 * n_blocks)
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) < block_size:
            continue
        tmp = z[i:i+block_size].copy()
        z[i:i+block_size] = z[j:j+block_size]
        z[j:j+block_size] = tmp
    return _softmax(z)


def _run_lbfgs(a_init, deadline, min_time_remaining=30.0):
    """Run L-BFGS with adaptive beta-continuation."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    # Adaptive beta schedule: start large, decrease based on gap
    c_sorted = np.sort(c_cur)[::-1]
    gap = c_sorted[0] - c_sorted[min(10, len(c_sorted)-1)] if len(c_sorted) > 10 else c_sorted[0]
    base_beta = max(gap, max_c * 0.01)

    betas = base_beta * np.array([1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])

    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-30))
    z = z - z.mean()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * max(1, n / 2048.0)
        max_iter = min(5000, max(30, int(remaining / (est_time_per_iter * 10))))

        def fun(z_local):
            return _smoobj(z_local, beta, n)

        def jac(z_local):
            return _smograd(z_local, beta, n)

        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-11,
                    'maxcor': 40,
                    'maxls': 80,
                }
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc_new
                z = np.log(np.maximum(a_new, 1e-30))
                z = z - z.mean()
        except Exception:
            pass

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy search: L-BFGS + adaptive beta + shrinkage + multi-n + restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    n_parent = 0
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
                n_parent = len(parent)
    except Exception:
        pass

    if parent is not None:
        print(f"Parent loaded: n={n_parent}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_parent = 2048
        parent = _make_profile(n_parent, "uniform")
        print(f"No parent, using uniform n={n_parent}")
        update_best(parent, _score(parent), "init")

    # --- Strategy 1: Main L-BFGS from parent ---
    print(f"\n[Phase 1] Main L-BFGS from parent (t={time.monotonic()-t_start:.1f}s)")
    a_main = _run_lbfgs(parent, deadline, min_time_remaining=400.0)
    update_best(a_main, _score(a_main), "lbfgs_main")

    # Subgradient polish
    if best_seq is not None and time.monotonic() < deadline - 300:
        sg_deadline = min(deadline - 300, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_max_steps=3000, lr_init=0.02)
        update_best(a_sg, sc_sg, "sg_polish_1")

    # --- Strategy 2: Shrinkage ---
    print(f"\n[Phase 2] Shrinkage (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_shr, sc_shr = _shrinkage(best_seq, deadline - 250)
        update_best(a_shr, sc_shr, "shrinkage")
        if sc_shr < best_score:
            # Re-optimize after shrinkage
            a_shr2 = _run_lbfgs(a_shr, deadline, min_time_remaining=200.0)
            update_best(a_shr2, _score(a_shr2), "lbfgs_post_shrink")

    # --- Strategy 3: Try different n values ---
    print(f"\n[Phase 3] Multi-n search (t={time.monotonic()-t_start:.1f}s)")
    n_candidates = [1024, 1536, 3072, 4096]
    # Also try resizing the parent
    if n_parent not in n_candidates and n_parent != 2048:
        n_candidates.append(n_parent)

    for n_try in n_candidates:
        if time.monotonic() > deadline - 150:
            break
        # Initialize from parent (resampled) or from a profile
        if parent is not None and n_parent != n_try:
            # Resample parent to new length
            x_old = np.linspace(0, 1, n_parent)
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, parent)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()
        else:
            a_init = parent.copy() if n_parent == n_try else _make_profile(n_try, "tapered")

        sc_init = _score(a_init)
        a_opt = _run_lbfgs(a_init, deadline, min_time_remaining=100.0)
        sc_opt = _score(a_opt)
        update_best(a_opt, sc_opt, f"n_{n_try}")

        # Quick subgradient
        if sc_opt < best_score + 0.01 and time.monotonic() < deadline - 60:
            a_opt2, sc_opt2 = _subgrad_polish(a_opt, min(deadline - 50, time.monotonic() + 60), n_max_steps=1000, lr_init=0.015)
            update_best(a_opt2, sc_opt2, f"n_{n_try}_sg")

    # --- Strategy 4: Diverse restarts from best ---
    print(f"\n[Phase 4] Diverse restarts (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 50:
        n_restarts = 3
        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)
            else:
                p_start = _perturb_block_swap(best_seq, rng, n_blocks=3)

            sc_start = _score(p_start)
            a_r = _run_lbfgs(p_start, deadline, min_time_remaining=30.0)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_{i}")

            if sc_r < best_score + 0.005 and time.monotonic() < deadline - 15:
                a_r2, sc_r2 = _subgrad_polish(a_r, min(deadline - 10, time.monotonic() + 30), n_max_steps=500, lr_init=0.01)
                update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # --- Strategy 5: Newton active set on best ---
    print(f"\n[Phase 5] Newton active set (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_newton, sc_newton = _newton_active_set(best_seq, min(deadline - 5, time.monotonic() + 60), max_iter=100)
        update_best(a_newton, sc_newton, "newton")

    # --- Strategy 6: Final fine polish ---
    print(f"\n[Phase 6] Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        # Very fine L-BFGS
        c_best = _full_convolve(best_seq, best_seq)
        mb = float(c_best.max())
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        beta_final = mb * 0.0001
        remaining = deadline - 3 - time.monotonic()
        if remaining > 5:
            max_iter = min(2000, max(50, int(remaining / 0.05)))
            try:
                result = minimize(
                    lambda z_local: _smoobj(z_local, beta_final, n),
                    z,
                    jac=lambda z_local: _smograd(z_local, beta_final, n),
                    method='L-BFGS-B',
                    options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxcor': 40}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_max_steps=300, lr_init=0.008)
            update_best(a_fin2, sc_fin2, "final_sg")

    # --- Final validation and return ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
