import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    c = _full_convolve(a, a)

def _lbfgs_optimize(a_init, beta_schedule, deadline, max_iter_per_phase=10000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        # Define objective and gradient for this beta
        def func(z):
            val, _ = _smoothed_obj_and_grad(z, beta, n)
            return val
        
        def grad(z):
            _, g = _smoothed_obj_and_grad(z, beta, n)
            return g
        
        # Run L-BFGS
        res = minimize(func, z, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter_per_phase,
                                'maxfun': max_iter_per_phase,
                                'gtol': 1e-8,
                                'iprint': 0})
        
        # Evaluate the true (unsmoothed) objective
        a_new = _softmax(res.x)
        score_new = _score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        z = res.x
        z = np.clip(z, -50, 50)
        z = z - z.mean()
    
    return best_a, best_score

"""
Gradient-based optimization with L-BFGS continuation for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses log-sum-exp smoothing with decreasing beta, Adam warmup,
and multi-restart strategy to find near-optimal flat-autocorrelation sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _smoothed_obj(z, beta, n):
    """Compute log-sum-exp smoothed objective value."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_c = np.exp(shifted)
    log_sum_exp = c_max + np.log(np.sum(exp_c)) / beta
    return 2.0 * n * log_sum_exp


def _smoothed_grad(z, beta, n):
    """Compute gradient of log-sum-exp smoothed objective w.r.t. z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_c = np.exp(shifted)
    w = exp_c / np.sum(exp_c)
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_step(z, beta, n, m_buf, v_buf, t_step, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Single Adam step for the smoothed objective in log-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = np.max(c)
    shifted = beta * (c - c_max)
    shifted = np.clip(shifted, -700, 700)
    exp_c = np.exp(shifted)
    w = exp_c / np.sum(exp_c)
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    
    m_buf = b1 * m_buf + (1.0 - b1) * grad
    v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
    m_hat = m_buf / (1.0 - b1 ** t_step)
    v_hat = v_buf / (1.0 - b2 ** t_step)
    
    update = lr * m_hat / (np.sqrt(v_hat) + eps)
    update_norm = np.linalg.norm(update)
    if update_norm > 1.0:
        update = update * (1.0 / update_norm)
    
    return z - update, m_buf, v_buf, t_step


def _run_adam(z_init, deadline, n, beta_schedule, lr=0.01, max_iter_per_phase=50000):
    """Run Adam with a schedule of beta values."""
    z = z_init.copy()
    z = z - z.mean()
    
    best_z = z.copy()
    best_a = _softmax(z)
    best_score = _score(best_a)

    v_buf = np.zeros(n)
    t_step = 0
    current_lr = lr
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        iters = 0
        while iters < max_iter_per_phase:
            if time.monotonic() > deadline:
                break
            iters += 1
            
            z, m_buf, v_buf, t_step = _adam_step(z, beta, n, m_buf, v_buf, t_step, current_lr)
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if iters % 200 == 0:
                a = _softmax(z)
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_z = z.copy()
                    best_a = a.copy()
            
            if iters % 5000 == 0:
                current_lr = max(0.002, current_lr * 0.95)
        
        # Reset Adam state at beta transitions
        m_buf = np.zeros(n)
        v_buf = np.zeros(n)
        t_step = 0
        current_lr = lr
    
    return best_z, best_a, best_score


def _run_lbfgs(z_init, beta, n, deadline, max_iter=5000, gtol=1e-8):
    """Run L-BFGS-B on the smoothed objective for a given beta."""
    z = z_init.copy()
    
    def func(z_local):
        return _smoothed_obj(z_local, beta, n)
    
    def grad(z_local):
        return _smoothed_grad(z_local, beta, n)
    
    try:
        res = minimize(func, z, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'gtol': gtol, 'iprint': 0,
                                'maxfun': max_iter * 3})
        z_opt = res.x
    except Exception:
        z_opt = z
    
    z_opt = np.clip(z_opt, -50.0, 50.0)
    z_opt = z_opt - z_opt.mean()
    
    a_opt = _softmax(z_opt)
    score = _score(a_opt)
    
    return z_opt, a_opt, score


def _perturb_z(z, sigma, rng):
    """Perturb z (log-space) with Gaussian noise."""
    z_new = z + sigma * rng.standard_normal(len(z))
    z_new = z_new - z_new.mean()
    return z_new


def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
