"""
Optimized search for minimum 2*n*max(conv(a,a))/sum(a)^2 using:
- Log-sum-exp smooth gradient with L-BFGS
- LP-based iterated minimax
- Asymmetric profile initialization
- Multi-scale (different n) exploration
- Continuation method on temperature parameter

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assume sum(a)=1, return 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lsexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _lsexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # Compute r = cross_correlation(w, a) using FFT
    # r[m] = sum_{l=0}^{n-1} w[m+l] * a[l]
    # This equals (w * a_rev)[m + n - 1] where * is full convolution
    a_rev = a[::-1]
    N = _next_pow2(3 * n - 2)
    W_fft = np.fft.rfft(w, N)
    A_rev_fft = np.fft.rfft(a_rev, N)
    conv = np.fft.irfft(W_fft * A_rev_fft, N)
    r = conv[n-1 : 2*n-1]
    grad = 4.0 * n * a * (r - W)
    return grad

def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def _lp_minimax(f):
    """Solve: min t s.t. (f*g)[k] <= t, sum(g)=1, g>=0"""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
                return g_val, float(t.value)
    except Exception as e:
        print(f"  LP failed: {e}")
    return None, None

def _lp_iterate(f, deadline, n_iters=50, step_sizes=None):

    n = len(f)
    best_a = f.copy()
    best_score = _score_norm(f)

        step_sizes = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05] * 10

            break
        g, t_val = _lp_minimax(f)
        if g is None:
            break
        step = step_sizes[it % len(step_sizes)]
        f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        if it % 25 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}")
    
    return best_a, best_score

def _make_asymmetric_profiles(n):
    """Generate asymmetric profile initializations"""
    t = np.linspace(0, 1, n)
    profiles = []
    # Geometric/exponential decay (one-sided)
    for rate in [2, 3, 4, 5, 8, 12]:
        a = np.exp(-rate * t)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    # One-sided bump
    for width in [0.1, 0.2, 0.3, 0.5]:
        a = np.exp(-((t - width/2) / width)**2)
        a = np.maximum(a, 1e-15)
        profiles.append(a / a.sum())
    # Power law
    for p in [2, 3, 5, 10]:
        a = (1 - t)**p + 1e-15
        profiles.append(a / a.sum())
    # Ramp
    a = t + 1e-15
    profiles.append(a / a.sum())
    a = (1-t) + 1e-15
    profiles.append(a / a.sum())
    # Quadratic
    a = t**2 + 1e-15
    profiles.append(a / a.sum())
    a = (1-t)**2 + 1e-15
    profiles.append(a / a.sum())
    # Cubic
    a = t**3 + 1e-15
    profiles.append(a / a.sum())
    return profiles

def _make_profiles(n):
    """Generate various symmetric and asymmetric profiles"""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Cosine bell (symmetric)
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    # Asymmetric
    profiles.extend(_make_asymmetric_profiles(n))
    return profiles

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _run_lbfgs_continuation(a_init, deadline, beta_list=None, n_restarts_inner=1):

def _subgradient_descent(a_init, deadline, n_steps=50000, lr_init=0.1):
    """Subgradient descent with momentum on the max function"""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for it in range(n_steps):
        if time.monotonic() > deadline:
            break
        t_step += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Use softmax with adaptive temperature
        beta = 0.01 * cm
        vals = (c - cm) / beta
        vals = np.clip(vals, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        
        a_rev = a[::-1]
        N = _next_pow2(3 * n - 2)
        w_conv = np.fft.irfft(np.fft.rfft(w, N) * np.fft.rfft(a_rev, N), N)
        r = w_conv[n-1:2*n-1]
        
        grad = 4.0 * n * a * (r - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_step)
        vh = v / (1 - b2 ** t_step)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay learning rate
        if t_step % 5000 == 0:
            lr = max(0.01, lr * 0.85)
        
        # Periodic noise
        if t_step % 20000 == 0:
            z = z + 0.02 * np.random.default_rng(t_step).standard_normal(n)
            z = z - z.mean()
    
    return best_a, best_score
