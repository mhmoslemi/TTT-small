"""
Optimized step function search using central binomial initialization,
L-BFGS with adaptive log-sum-exp smoothing, active-set subgradient polishing,
and multi-scale restart strategy for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _central_binomial(n):
    b = np.ones(n)
    for k in range(1, n):
        b[k] = b[k - 1] * (2 * k - 1) / (2 * k)
    b = b / b.sum()
    return b


def _obj_val(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if beta <= 1e-15:
        return float(2.0 * n * c_max)
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(2.0 * n * c_max)
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _obj_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    if beta <= 1e-15:
        w = np.zeros_like(c)
        idx = np.argmax(c)
        w[idx] = 1.0
    else:
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
    h = _full_convolve(w, a[::-1])[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_subgrad(a, n, max_steps=2000, lr=0.02, tol=0.002):
    """Subgradient descent using exact active set (positions within tol*max of max)."""
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_sc = _score(a)
    lr_cur = lr

    for i in range(max_steps):
        cur_a = _softmax(z)
        c = _full_convolve(cur_a, cur_a)
        c_max = c.max()
        active = c >= (c_max - tol * c_max)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        h = _full_convolve(w, cur_a[::-1])[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * cur_a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = lr_cur / (1.0 + 0.005 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0 or i == max_steps - 1:
            sc = _score(_softmax(z))
            if sc < best_sc:
                best_sc = sc
                best_a = _softmax(z).copy()
                lr_cur *= 0.98
            else:
                lr_cur *= 0.85
                if lr_cur < 1e-6:
                    break

    return best_a, best_sc


def _lbfgs_phase(a_init, beta, n, max_iter=2000):
    """Run L-BFGS-B for one beta value."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _obj_val(z, beta, n)

    def jac(z):
        return _obj_grad(z, beta, n)

    try:
        res = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxcor': 40, 'maxls': 200,
                                'disp': False})
        a_out = _softmax(res.x)
        return a_out
    except Exception:
        return a_init

def _perturb_z(a, sigma, rng):
    """Perturb in z-space (log-space)."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_structure(a, rng, n_modes=3):
    """Structured perturbation using Fourier modes."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        amp = rng.uniform(0.005, 0.05)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _tapered_init(n, exponent=0.5):
    """Create a tapered (non-uniform) initial sequence."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** exponent
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _bimodal_init(n):
    """Create a bimodal initial sequence."""
    t = np.linspace(0, 1, n)
    a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()
