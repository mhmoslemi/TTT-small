"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, LP-based direction finding,
subgradient polishing with line search, and multi-restart diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_exact(a, c, n, tol_frac=0.001):
    """Exact subgradient using active set."""
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)  # gradient of max(conv) w.r.t. a, times 2n for the full objective (with sum=1 fixed)

def _score_and_grad(a):
    n = len(a)
    c = _full_convolve(a, a)
    sc = _score(a)
    g = _gradient_exact(a, c, n)
    return sc, g, c

def _project_nonneg(a):
    return np.maximum(a, 0.0)

def _normalize(a):
    s = a.sum()
    if s < 1e-30:
        return a
    return a / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.25 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * (2*t - 1)**4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-15, gtol=1e-12):
    """Run L-BFGS-B for one beta phase."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0

def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    """Full L-BFGS beta-continuation schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start
    
    # Beta schedule: start with moderate, end with very small
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break
        
        # Estimate time per iteration
        est_iter_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(5000, max(100, int(remaining / (est_iter_time * 12))))
        
        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc
    
    return current_a

def _subgrad_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000):
    """Subgradient method with adaptive step size in direct space."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_sy = float(1.0 / np.sqrt(n))  # Polyak-like step size
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        sc, g, c = _score_and_grad(a)
        
        # Project gradient
        g = g - g.mean()  # maintain sum=1
        
        # Check convergence
        if np.linalg.norm(g) < 1e-10:
            break
        
        # Step
        a_new = a - step_sy * g
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        sc_new = _score(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        if sc_new < sc:
            a = a_new
            step_sy *= 1.01  # slight increase
        else:
            step_sy *= 0.9  # decrease
        
        # Decay step size
        step_sy *= (1.0 - 0.0001)
        step_sy = max(step_sy, 1e-8)
    
    return best_a, best_sc

def _subgrad_polish_z(a_init, deadline, tol_frac=1e-3, max_steps=5000, lr_init=0.05):
    """Subgradient polishing in z-space (log space) with exact active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        step = lr / (1.0 + i * 0.0005)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
            elif sc > best_sc * 1.001:
                lr *= 0.92
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.005 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

def _perturb_block(a, rng, n_blocks=4):
    """Perturb by adding random block structure."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = n // n_blocks
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
        z[start:end] += rng.uniform(-0.02, 0.02)
    z = z - z.mean()
    return _softmax(z)

def _try_target_n(a_best, target_n, deadline, min_time=30.0):
    """Try to optimize a sequence of a different length by interpolation."""
    n_cur = len(a_best)
    if target_n == n_cur:
        return a_best
    
    # Interpolate to target length
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a_best)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    
    # Optimize
    sc0 = _score(a_interp)
    a_opt = _lbfgs_schedule(a_interp, deadline, min_time)
    sc_opt = _score(a_opt)
    
    return a_opt

def _coordinate_descent(a, deadline, max_rounds=100):
    """Simple coordinate descent: try adjusting pairs of adjacent elements."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        for i in range(0, n - 1, max(1, n // 1000)):
            if time.monotonic() > deadline:
                break
            
            # Try moving mass from i to i+1 and vice versa
            for delta in [0.01, 0.005, 0.001]:
                a_try = best_a.copy()
                s = a_try[i] + a_try[i+1]
                for frac in [0.3, 0.5, 0.7]:
                    a_try[i] = s * (1 - frac + delta)
                    a_try[i+1] = s * (frac - delta)
                    if a_try[i] < 0 or a_try[i+1] < 0:
                        continue
                    a_try = a_try / a_try.sum()
                    sc_try = _score(a_try)
                    if sc_try < best_sc - 1e-12:
                        best_a = a_try.copy()
                        best_sc = sc_try
                        improved = True
        
        if not improved and rnd > 10:
            break
    
    return best_a, best_sc

"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing and beta-continuation,
subgradient polishing with tight active sets, direct-space polishing,
multi-restart diversification, and local coordinate search.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, tol_frac=1e-4):
    c_max = c.max()
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * n * (h - W)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered2":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 1.0
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "quartic":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.2 * (2.0 * t - 1.0) ** 4
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_run(a_init, beta, n, max_iter, ftol=1e-16, gtol=1e-13):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': ftol,
                                   'gtol': gtol, 'maxcor': 50,
                                   'maxls': 50, 'disp': False})
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _lbfgs_schedule(a_start, deadline, min_time_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-30:
        return a_start

    betas = max_c * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005,
                              0.0025, 0.001, 0.0005, 0.0002,
                              0.0001, 0.00005, 0.00002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 10:
            break

        est_iter_time = 0.003 * (n / 2048.0) ** 1.5
        max_iter = min(8000, max(100, int(remaining / (est_iter_time * 15))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-15:
            current_a = a_new
            current_sc = sc

    return current_a


def _subgrad_z_polish(a_init, deadline, tol_frac=1e-4, max_steps=10000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    stall_count = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.0003)
        z = z - step * grad
        z = np.clip(z, -90.0, 90.0)
        z = z - z.mean()

        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.03
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 3:
                    lr *= 0.7
                    if lr < 1e-6:
                        break
                else:
                    lr *= 0.93

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _subgrad_direct_polish(a_init, deadline, tol_frac=1e-5, max_steps=30000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    step_size = 1.0 / np.sqrt(n)
    stall = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        g = _gradient_active(a, c, n, tol_frac)
        g = g - g.mean()

        if np.linalg.norm(g) < 1e-14:
            break

        a_try = a - step_size * g
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-30:
            step_size *= 0.5
            continue
        a_try = a_try / s
        sc_try = _score(a_try)

        if sc_try < best_sc:
            best_sc = sc_try
            best_a = a_try.copy()
            stall = 0

        if sc_try < _score(a) - 1e-15:
            a = a_try
            step_size = min(step_size * 1.005, 0.1)
        else:
            stall += 1
            step_size *= 0.85

        step_size *= (1.0 - 1e-5)
        step_size = max(step_size, 1e-10)

        if stall > 200:
            break

    return best_a, best_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = 0.004 * rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(4, n // n_blocks)
    for _ in range(n_blocks):
        start = int(rng.integers(0, max(1, n - block_size)))
        end = min(n, start + int(block_size * rng.uniform(0.5, 1.5)))
        z[start:end] += rng.uniform(-0.015, 0.015)
    z = z - z.mean()
    return _softmax(z)


def _local_search(a, deadline, max_rounds=200):
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    step = 0.005

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        idx = int((rnd * 997) % n)
        jdx = int((rnd * 997 + 7) % n)
        if idx == jdx:
            jdx = (jdx + 1) % n

        for delta_frac in [0.3, 0.5, 0.7]:
            s_pair = best_a[idx] + best_a[jdx]
            new_val_i = s_pair * (1.0 - delta_frac + step)
            new_val_j = s_pair * (delta_frac - step)
            if new_val_i < 0 or new_val_j < 0:
                continue

            a_try = best_a.copy()
            a_try[idx] = new_val_i
            a_try[jdx] = new_val_j
            a_try = a_try / a_try.sum()
            sc_try = _score(a_try)

            if sc_try < best_sc - 1e-13:
                best_a = a_try.copy()
                best_sc = sc_try
                improved = True

        if not improved and rnd > 20:
            step *= 0.7
            if step < 1e-6:
                break

    return best_a, best_sc


def _interpolate_to_n(a, target_n):
    n_cur = len(a)
    t_old = np.linspace(0, 1, n_cur)
    t_new = np.linspace(0, 1, target_n)
    a_interp = np.interp(t_new, t_old, a)
    a_interp = np.maximum(a_interp, 1e-10)
    a_interp = a_interp / a_interp.sum()
    return a_interp


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining L-BFGS with log-sum-exp smoothing,
    subgradient polishing (both z-space and direct space), local coordinate
    search, and multi-restart diversification to minimize
    2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def try_update(p, label=""):
        nonlocal best_seq, best_score_val
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score_val:
            prev = best_score_val
            best_seq = p.copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] "
                  f"improvement={prev - best_score_val:.2e} t={elapsed:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent: n={len(parent)}, score={_score(parent):.10f}")
        try_update(parent, "parent")
    else:
        a0 = _make_profile(2048, "uniform")
        print(f"No parent. Uniform n=2048, score={_score(a0):.10f}")
        try_update(a0, "init_uniform")

    # === Phase 1: Main L-BFGS ===
    print(f"\n=== Phase 1: Main L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    a_p1 = _lbfgs_schedule(best_seq, deadline, min_time_remaining=500.0)
    try_update(a_p1, "phase1_lbfgs")

    # === Phase 2: Z-space subgradient (tight) ===
    print(f"\n=== Phase 2: Z-space subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 400:
        a_sg, sc_sg = _subgrad_z_polish(best_seq, deadline - 400,
                                        tol_frac=1e-4, max_steps=10000, lr_init=0.04)
        try_update(a_sg, "phase2_sg_z")

    # === Phase 3: Direct-space subgradient ===
    print(f"\n=== Phase 3: Direct subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 250:
        a_sg2, sc_sg2 = _subgrad_direct_polish(best_seq, deadline - 250,
                                               tol_frac=1e-5, max_steps=30000)
        try_update(a_sg2, "phase3_sg_direct")

    # === Phase 4: Fine L-BFGS ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 150:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2e-5, 1e-5, 5e-6, 2e-6])

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 100:
                break
            remaining = deadline - 100 - time.monotonic()
            if remaining < 10:
                break
            max_iter = min(5000, max(200, int(remaining / 0.004)))
            a_new, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_new, f"fine_{i}")

    # === Phase 5: Local search ===
    print(f"\n=== Phase 5: Local search (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a_ls, sc_ls = _local_search(best_seq, deadline - 100, max_rounds=500)
        try_update(a_ls, "local_search")

    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_configs = [
            ("gauss_002", "gauss", 0.02, 0),
            ("gauss_005", "gauss", 0.05, 0),
            ("gauss_010", "gauss", 0.10, 0),
            ("fourier_3", "fourier", 0, 3),
            ("fourier_5", "fourier", 0, 5),
            ("block_4", "block", 0, 4),
            ("block_8", "block", 0, 8),
        ]

        for name, ptype, sigma, modes in restart_configs:
            if time.monotonic() > deadline - 50:
                break

            if ptype == "gauss":
                p_start = _perturb_gaussian(best_seq, sigma, rng)
            elif ptype == "fourier":
                p_start = _perturb_fourier(best_seq, rng, modes)
            else:
                p_start = _perturb_block(best_seq, rng, modes)

            sc_start = _score(p_start)
            print(f"  Restart [{name}]: start={sc_start:.10f}")

            a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=40.0)
            try_update(a_r, f"restart_{name}")

            if time.monotonic() < deadline - 35 and _score(a_r) < best_score_val + 0.005:
                a_r2, _ = _subgrad_z_polish(a_r, deadline - 35,
                                            tol_frac=1e-4, max_steps=3000, lr_init=0.03)
                try_update(a_r2, f"restart_{name}_sg")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Different lengths (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        for target_n in [1024, 3072, 4096]:
            if time.monotonic() > deadline - 50:
                break
            if target_n == len(best_seq):
                continue
            print(f"  Trying n={target_n}...")
            a_interp = _interpolate_to_n(best_seq, target_n)
            a_try = _lbfgs_schedule(a_interp, deadline, min_time_remaining=30.0)
            try_update(a_try, f"n_{target_n}")

    # === Phase 8: Final aggressive polish cycle ===
    print(f"\n=== Phase 8: Final polish (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        for rnd in range(4):
            if time.monotonic() > deadline - 5:
                break

            # Tiny-beta L-BFGS
            n = len(best_seq)
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            beta = mb * 1e-6
            max_iter = 3000
            a_f, _, _ = _lbfgs_run(best_seq, beta, n, max_iter)
            try_update(a_f, f"final_lbfgs_{rnd}")

            # Tight subgradient
            if time.monotonic() < deadline - 5:
                a_s, _ = _subgrad_z_polish(best_seq, deadline - 5,
                                           tol_frac=1e-5, max_steps=5000, lr_init=0.015)
                try_update(a_s, f"final_sg_{rnd}")

            # Local search
            if time.monotonic() < deadline - 5:
                a_l, _ = _local_search(best_seq, deadline - 5, max_rounds=300)
                try_update(a_l, f"final_ls_{rnd}")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
