"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2 over non-negative
sequences. Combines L-BFGS with adaptive log-sum-exp smoothing, Nesterov-accelerated
subgradient polishing in z-space (simplex), Fourier-based diverse initializations,
and local block perturbations to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_exact(a, c, n, tol_frac=0.001):
    """Compute subgradient of max_k c_k with respect to p (on simplex)."""
    c_max = c.max()
    thresh = c_max - tol_frac * c_max
    active = c >= thresh
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    k_active = active.sum()
    w = np.zeros_like(c)
    w[active] = 1.0 / k_active
    # g[i] = 2 * sum_k w[k] * p[k-i] (zero extension)
    # = 2 * (w conv p)[appropriate slice]
    # Using FFT for efficiency:
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * h - 2.0 * W

def _nesterov_subgrad(a_init, deadline, n_max_steps=20000, eta0=0.01):
    """Nesterov-accelerated subgradient descent in z-space on the simplex."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    p_prev = a_init.copy()
    p_curr = a_init.copy()
    t0 = 0.0
    
    for t in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation
        if t > 0:
            beta_prev = (1.0 + np.sqrt(1.0 + 4.0 / max(1, t - 1) ** 2)) / 2.0
            beta_curr = (1.0 + np.sqrt(1.0 + 4.0 / t ** 2)) / 2.0
            momentum = (beta_prev - 1.0) / beta_curr
            y = p_curr + momentum * (p_curr - p_prev)
            y = _project_simplex_vec(y)
        else:
            y = p_curr.copy()
        
        # Compute subgradient at y
        c = _full_convolve(y, y)
        g = _gradient_exact(y, c, n, tol_frac=0.002)
        
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        eta = eta0 / (1.0 + 0.5 * np.sqrt(t))
        
        # Update in z-space
        z = np.log(np.maximum(y, 1e-15))
        z = z - eta * g
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()
        p_new = _softmax(z)
        
        p_prev = p_curr.copy()
        p_curr = p_new
        t0 = (1.0 + np.sqrt(1.0 + 4.0 / (t + 1) ** 2)) / 2.0 * t0 + 1.0
        
        # Track best
        if t % 20 == 0:
            sc = _score(p_curr)
            if sc < best_sc:
                best_sc = sc
                best_a = p_curr.copy()
    
    sc_final = _score(p_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = p_curr.copy()
    
    return best_a, best_sc

def _nesterov_subgrad_v2(z_init, deadline, n_max_steps=50000, eta0=0.005):
    """Nesterov subgradient directly in z-space (simpler and faster)."""
    n = _N_global
    best_a = _softmax(z_init.copy())
    best_sc = _score(best_a)
    
    z_prev = z_init.copy()
    z_curr = z_init.copy()
    
    for t in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        # Nesterov extrapolation in z-space
        if t > 0:
            beta_t = (1.0 + np.sqrt(1.0 + 4.0 / (t + 1) ** 2)) / 2.0
            beta_tm1 = (1.0 + np.sqrt(1.0 + 4.0 / t ** 2)) / 2.0
            momentum = (beta_tm1 - 1.0) / beta_t
            z_y = z_curr + momentum * (z_curr - z_prev)
        else:
            z_y = z_curr.copy()
        
        y = _softmax(z_y)
        c = _full_convolve(y, y)
        g = _gradient_exact(y, c, n, tol_frac=0.001)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        eta = eta0 / (1.0 + 0.3 * np.log1p(t))
        
        z_new = z_y - eta * g
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        
        z_prev = z_curr.copy()
        z_curr = z_new
        
        if t % 10 == 0:
            a_check = _softmax(z_curr)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z_curr)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _project_simplex_vec(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def _lbfgs_run(z0, beta, n, max_iter=2000):
    """Run L-BFGS-B for one beta value."""
    def fun(z):
        return _smoobj(z, beta, n)
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-16,
                'gtol': 1e-12,
                'maxcor': 40,
                'maxls': 200,
                'disp': False,
            }
        )
        return _softmax(result.x)
    except Exception:
        return _softmax(z0.copy())

def _lbfgs_schedule(a_start, deadline, min_time=20.0, n_phases=8):
    """Adaptive L-BFGS beta-continuation schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start
    
    z = np.log(np.maximum(a_start, 1e-15))
    z = z - z.mean()
    
    # Adaptive beta: start at 0.5*max_c, reduce by factor 0.6 each phase
    beta = max_c * 0.5
    current_a = a_start.copy()
    current_sc = _score(current_a)
    
    for phase in range(n_phases):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break
        
        # Scale max_iter based on available time
        est_time_per_iter = 0.004 * (n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time_per_iter * 10))))
        
        a_new = _lbfgs_run(z, beta, n, max_iter)
        sc = _score(a_new)
        
        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc
            z = np.log(np.maximum(current_a, 1e-15))
            z = z - z.mean()
        
        beta *= 0.55
    
    return current_a

def _fourier_init(n, rng, kind=0):
    """Generate initial sequence with flat-ish power spectrum."""
    if kind == 0:
        # Random phase, flat amplitude
        N = 2 * n
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        P = np.exp(1j * phases)
        P[0] = 1.0  # Ensures real and positive DC
        p = np.real(np.fft.irfft(P * (N // 2 + 1), N)[:n])
    elif kind == 1:
        # Tapered amplitude in frequency domain
        N = 2 * n
        freqs = np.arange(N // 2 + 1)
        amp = 1.0 / (1.0 + (freqs / (n * 0.3)) ** 2)
        phases = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        P = amp * np.exp(1j * phases)
        P[0] = amp[0]
        p = np.real(np.fft.irfft(P * (N // 2 + 1), N)[:n])
    elif kind == 2:
        # Band-limited: only certain frequencies
        N = 2 * n
        P = np.zeros(N // 2 + 1, dtype=complex)
        num_bands = max(1, n // 8)
        for _ in range(num_bands):
            f = rng.integers(1, N // 2)
            P[f] += np.exp(1j * rng.uniform(0, 2 * np.pi))
        P[0] = num_bands
        p = np.real(np.fft.irfft(P * (N // 2 + 1), N)[:n])
    else:
        p = np.ones(n) / n
    
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return p / s

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "edge":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb_random(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_block(a, rng, block_size=32):
    """Replace a random block with a uniform or random profile."""
    n = len(a)
    start = rng.integers(0, n - block_size)
    a_new = a.copy()
    profile_type = rng.integers(0, 3)
    if profile_type == 0:
        a_new[start:start+block_size] = 1.0 / block_size
    elif profile_type == 1:
        a_new[start:start+block_size] = rng.dirichlet(np.ones(block_size))
    else:
        t = np.linspace(0, 1, block_size)
        a_new[start:start+block_size] = np.sin(np.pi * t) ** 0.5
        a_new[start:start+block_size] = np.maximum(a_new[start:start+block_size], 1e-10)
        a_new[start:start+block_size] = a_new[start:start+block_size] / a_new[start:start+block_size].sum()
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _local_search(a_init, deadline, rng, n_trials=200):

    n = len(a_init)

        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
