Multi-phase optimizer combining FISTA, subgradient averaging, and coordinate descent
for minimizing the autocorrelation peak of non-negative sequences.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)

def _proj_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 0:
        return v
    # Project onto {x >= 0, sum(x) = target_sum}
    # Scale to work with sum=1, then scale back
    if target_sum != 1.0:
        return target_sum * _proj_simplex(v / target_sum, 1.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _convolve_grad(a, k):
    """Compute gradient of (a*a)[k] with respect to a."""
    n = len(a)
    # ∂(a*a)[k]/∂a[j] = 2*a[k-j] if 0 <= k-j < n, else 0
    # This is 2 * (a reversed and appropriately shifted)
    # a[k-j] for j=0,...,n-1 is a[k], a[k-1], ..., a[k-n+1]
    # We need to extract a[k-k+1 .. k] from the "extended" a (zero-padded)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    grad = np.zeros(n)
    if lo <= hi:

        # k-j ranges from k-(hi) to k-(lo), i.e., from lo to hi in a's indexing... 
        # Actually: for j in [0, n-1], k-j is in [k-(n-1), k]
        # We need 0 <= k-j < n, i.e., k-n+1 <= j <= k
        # So j ranges from max(0, k-n+1) to min(n-1, k)
        j_lo = max(0, k - (n - 1))
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            # a[k-j] for j = j_lo, ..., j_hi
            # k-j ranges from k-j_hi to k-j_lo
            indices = k - np.arange(j_lo, j_hi + 1)
            grad[j_lo:j_hi+1] = 2.0 * a[indices]
    return grad

def _subgradient_max(a, K=10):
    """Compute subgradient (average of top-K convolution coefficient gradients)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    # Find top K indices
    top_k = np.argsort(c)[::-1][:K]
    grad = np.zeros(n)
    for k in top_k:
        grad += _convolve_grad(a, int(k))
    grad /= K
    return grad, c, c_max

    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    L = 4.0 * n / beta  # Lipschitz constant estimate
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient: 2 * sum_k w[k] * (a * a_rev_k)[n-1:]
        # where a * a_rev_k is the cross-correlation
        # Actually, the gradient of sum_k w[k] * (a*a)[k] w.r.t. a
        # = sum_k w[k] * 2 * a[k-i] (with bounds)
        # = 2 * cross_correlation(w, a) (appropriately defined)

        # grad[i] = 2 * sum_k w[k] * a[k-i] (with a[x]=0 for x out of [0,n-1])
        # This is 2 * (w * a_rev)[n-1+i] where a_rev is a reversed...
