Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direct peak optimization, multi-scale continuation, and coordinate descent.

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Score assuming sum(a) = 1: returns 2n * max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    c = _full_convolve(a, a)
    cm = float(c.max())

        return 0.0

    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum
