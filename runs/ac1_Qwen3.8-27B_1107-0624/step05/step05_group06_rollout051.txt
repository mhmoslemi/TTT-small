"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, direct subgradient with 
Barzilai-Borwein step sizes, coordinate descent polishing, and 
multi-restart diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize
import itertools

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _active_set_gradient(z, a, c, n, rel_tol=0.001):
    """Gradient using exact active set (coeffs within rel_tol of max)."""
    c_max = c.max()
    tol = rel_tol * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_phase(z0, beta, n, max_iter, deadline):
    """Run one L-BFGS phase."""
    def fun(z):
        return _smoobj(z, beta, n)
    def jac(z):
        return _smograd(z, beta, n)
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'gtol': 1e-12,
                'maxcor': 50,
                'maxls': 200,
                'disp': False,
            }
        )
        return result.x, result.success, int(result.nit)
    except Exception:
        return z0, False, 0

def _bb_subgradient(a_init, deadline, n_steps=5000, rel_tol=0.0005):
    """Subgradient method with Barzilai-Borwein step sizes and exact active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    
    # BB state
    prev_z = None
    prev_grad = None
    bb_alpha = 0.05  # initial step
    
    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_set_gradient(z, a, c, n, rel_tol)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Barzilai-Borwein step size
        if prev_z is not None and prev_grad is not None:
            sz = z - prev_z
            sg = grad - prev_grad
            denom_y = np.dot(sg, sg)
            denom_s = np.dot(sz, sz)
            if denom_y > 1e-20:
                alpha_y = np.dot(sz, sg) / denom_y
            else:
                alpha_y = bb_alpha
            if denom_s > 1e-20:
                alpha_s = denom_s / max(np.dot(sz, sg), 1e-20)
            else:
                alpha_s = bb_alpha
            # Alternate between BB_y and BB_s
            if i % 2 == 0:
                bb_alpha = max(1e-10, min(alpha_y, 1.0))
            else:
                bb_alpha = max(1e-10, min(alpha_s, 1.0))
        
        z = z - bb_alpha * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        prev_z = z.copy()
        prev_grad = grad.copy()
        
        if i % 20 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _coordinate_polish(a_init, deadline, n_sweeps=50):
    """
    Coordinate descent: for each coordinate, find optimal value
    given the sum constraint. Uses parametric update.
    Actually, since sum(a)=1 is a constraint, we'll use a different approach:
    For pairs of coordinates (i, j), redistribute mass between them.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(0, n, max(1, n // 500)):
            if time.monotonic() > deadline:
                break
            # Try moving mass from a[i] to a[(i+1)%n]
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                trial = a.copy()
                amt = frac * a[i]
                trial[i] -= amt
                trial[(i+1) % n] += amt
                if trial[i] < 0 or trial[(i+1)%n] < 0:
                    continue
                sc = _score(trial)
                if sc < best_sc - 1e-15:
                    a = trial.copy()
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            break
    
    return best_a, best_sc

def _perturb_z(a, sigma, rng):

def _perturb_fourier_z(a, rng, n_modes=3):
    """Add random Fourier mode perturbations in z-space."""
    n = len(a)

    return z

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, Barzilai-Borwein subgradient
with exact active set tracking, targeted peak-reduction nudging, and

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _active_set_gradient(a, c, n, rel_tol=0.0005):
    """Exact subgradient using active set within rel_tol of max."""
    c_max = c.max()
    tol = rel_tol * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        return _smoobj(z, beta, n)

        return _smograd(z, beta, n)

                'maxls': 200,
                'disp': False,

            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_set_gradient(a, c, n, rel_tol)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        # Barzilai-Borwein step size
        if prev_z is not None and prev_grad is not None:
            sz = z - prev_z
            sg = grad - prev_grad
            dot_ss = np.dot(sz, sz)
            dot_sg = np.dot(sz, sg)
            dot_gg = np.dot(sg, sg)
            if dot_gg > 1e-20:
                alpha_y = dot_sg / dot_gg
            else:
                alpha_y = bb_alpha
            if dot_ss > 1e-20 and dot_sg > 1e-20:
                alpha_s = dot_ss / dot_sg
            else:
                alpha_s = bb_alpha
            # Alternate with damping
            if i % 2 == 0:
                bb_alpha = max(1e-12, min(0.5 * alpha_y, 0.5))
            else:
                bb_alpha = max(1e-12, min(0.5 * alpha_s, 0.5))

        # Nesterov step
        if use_nesterov:
            momentum = beta_hop * momentum + (1 - beta_hop) * (-bb_alpha * grad)
            z = z + momentum
        else:
            z = z - bb_alpha * grad

        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        prev_z = z.copy()
        prev_grad = grad.copy()

        # Periodic check
        if i % 15 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()

    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _plain_subgradient(a_init, deadline, n_steps=5000, rel_tol=0.0005, lr_init=0.02):
    """Plain subgradient with adaptive learning rate (no BB)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _active_set_gradient(a, c, n, rel_tol)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / (1.0 + 0.01 * i)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 15 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
                if patience > 10:
                    lr *= 0.7
                    patience = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _targeted_nudge(a_init, deadline, n_iters=200):
    """
    Targeted peak reduction: identify elements contributing most to the
    autocorrelation peak and nudge them to reduce the peak.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        peak_idx = int(np.argmax(c))

        # Find elements contributing most to the peak
        # (a*a)[peak_idx] = sum_i a[i] * a[peak_idx - i]
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        # Elements with highest contribution that we can reduce
        # Move mass from high-contrib elements to low-contrib ones
        top_k = min(10, n // 4)
        top_idx = np.argsort(contribs)[-top_k:]
        bottom_idx = np.argsort(contribs)[:top_k]

        moved = False
        for ti in top_idx:
            if a[ti] < 1e-10:
                continue
            for bi in bottom_idx:
                if bi == ti:
                    continue
                # Try small transfer
                for frac in [0.005, 0.01, 0.02]:
                    amt = frac * a[ti]
                    if a[ti] - amt < 0:
                        continue
                    trial = a.copy()
                    trial[ti] -= amt
                    trial[bi] += amt
                    sc = _score(trial)
                    if sc < best_sc - 1e-15:
                        a = trial.copy()
                        best_sc = sc
                        best_a = a.copy()
                        moved = True
                        break
                if moved:
                    break
            if moved:
                break

        if not moved:
            # Try random small perturbation on high-contrib elements
            rng_local = np.random.default_rng(it)
            trial = a.copy()
            for ti in top_idx[:5]:
                if a[ti] > 1e-6:
                    delta = rng_local.uniform(-0.01, 0.005) * a[ti]
                    trial[ti] = max(0, a[ti] + delta)
            # Renormalize
            s = trial.sum()
            if s > 0.01:
                trial = trial / s
                sc = _score(trial)
                if sc < best_sc - 1e-15:
                    a = trial.copy()
                    best_sc = sc
                    best_a = a.copy()
                    moved = True

        if not moved:
            break

    return best_a, best_sc


def _perturb_z_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return z


def _perturb_z_fourier(a, rng, n_modes=3, amp_range=(0.005, 0.04)):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = rng.uniform(*amp_range)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return z


def _perturb_z_local(a, rng, n_regions=3):
    """Localized perturbations at random positions."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(n_regions):
        center = rng.integers(0, n)
        width = rng.integers(n // 16, n // 4)
        mask = np.zeros(n)
        lo = max(0, center - width)
        hi = min(n, center + width)
        t_local = np.arange(lo, hi)
        mask[lo:hi] = np.exp(-0.5 * ((t_local - center) / max(1, width // 3)) ** 2)
        amp = rng.uniform(0.01, 0.05)
        z[lo:hi] += amp * mask[lo:hi] * rng.standard_normal(hi - lo)
    z = z - z.mean()
    return z


def _run_beta_schedule(z0, n, deadline, min_remaining=20.0):
    """Run L-BFGS beta continuation."""
    a_cur = _softmax(z0)
    c_cur = _full_convolve(a_cur, a_cur)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return z0

    betas = max_c * np.array([0.15, 0.075, 0.0375, 0.019, 0.0095, 0.0048, 0.0024, 0.0012, 0.0006])
    z_cur = z0.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:
            break
        est_ms = 4.0 * (n / 2048.0)
        max_iter = min(4000, max(30, int(remaining * 1000 / (est_ms * 4))))

        z_new, success, niter = _lbfgs_phase(z_cur, beta, n, max_iter, deadline)
        z_cur = z_new

    return z_cur


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cos":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t) + 0.1 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS log-sum-exp -> BB subgradient ->
    targeted nudging -> multi-restart -> final squeeze.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        p[p < 0] = 0
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Phase 0: Load parent or create initial ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_main}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_main = 2048
        parent = _make_profile(n_main, "uniform")
        update_best(parent, _score(parent), "init_uniform")
        print(f"No parent, uniform n={n_main}, score={_score(parent):.10f}")

    # === Phase 1: L-BFGS beta continuation (moderate time) ===
    print(f"\n=== Phase 1: L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    phase1_deadline = deadline - 500.0
    if time.monotonic() < phase1_deadline:
        z0 = np.log(np.maximum(parent, 1e-15))
        z0 = z0 - z0.mean()
        z1 = _run_beta_schedule(z0, n_main, phase1_deadline, min_remaining=100.0)
        a1 = _softmax(z1)
        sc1 = _score(a1)
        update_best(a1, sc1, "lbfgs_p1")

    # === Phase 2: BB Subgradient (heavy) ===
    print(f"\n=== Phase 2: BB Subgradient (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None:
        sg_deadline = min(deadline - 250.0, time.monotonic() + 200.0)
        a2, sc2 = _bb_subgradient(best_seq, sg_deadline, n_steps=10000, rel_tol=0.0003, use_nesterov=True)
        update_best(a2, sc2, "bb_subgrad")

    # === Phase 3: Targeted nudging ===
    print(f"\n=== Phase 3: Targeted Nudge (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        nudge_deadline = min(deadline - 80.0, time.monotonic() + 60.0)
        a3, sc3 = _targeted_nudge(best_seq, nudge_deadline, n_iters=500)
        update_best(a3, sc3, "targeted_nudge")

    # === Phase 4: Fine L-BFGS with tiny betas ===
    print(f"\n=== Phase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 120:
        n = len(best_seq)
        z_b = np.log(np.maximum(best_seq, 1e-15))
        z_b = z_b - z_b.mean()
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([5e-5, 2.5e-5, 1e-5, 5e-6, 2e-6])

        z_f = z_b
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 60:
                break
            remaining = deadline - 60 - time.monotonic()
            if remaining < 2:
                break
            max_iter = min(2000, max(30, int(remaining / 0.025)))
            z_f, _, _ = _lbfgs_phase(z_f, beta, n, max_iter, deadline)
            a_f = _softmax(z_f)
            sc_f = _score(a_f)
            update_best(a_f, sc_f, f"fine_{i}")

    # === Phase 5: BB Subgradient round 2 (tighter) ===
    print(f"\n=== Phase 5: BB Round 2 (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        sg_deadline2 = min(deadline - 20.0, time.monotonic() + 80.0)
        a5, sc5 = _bb_subgradient(best_seq, sg_deadline2, n_steps=8000, rel_tol=0.0001, use_nesterov=False)
        update_best(a5, sc5, "bb_round2")

    # === Phase 6: Multi-restart ===
    print(f"\n=== Phase 6: Multi-restart (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 25:
        n = len(best_seq)
        z_best = np.log(np.maximum(best_seq, 1e-15))
        z_best = z_best - z_best.mean()

        perturb_strategies = [
            ("gauss_001", lambda: _perturb_z_gaussian(best_seq, 0.005, rng)),
            ("gauss_003", lambda: _perturb_z_gaussian(best_seq, 0.015, rng)),
            ("fourier_2", lambda: _perturb_z_fourier(best_seq, rng, n_modes=2)),
            ("fourier_4", lambda: _perturb_z_fourier(best_seq, rng, n_modes=4, amp_range=(0.003, 0.02))),
            ("local_3", lambda: _perturb_z_local(best_seq, rng, n_regions=3)),
            ("local_5", lambda: _perturb_z_local(best_seq, rng, n_regions=5)),
            ("gauss_0005", lambda: _perturb_z_gaussian(best_seq, 0.002, rng)),
        ]

        n_restarts = min(len(perturb_strategies), max(1, int((deadline - 25 - time.monotonic()) / 40)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            label, perturb_fn = perturb_strategies[i]
            z_r = perturb_fn()
            a_r0 = _softmax(z_r)
            sc_r0 = _score(a_r0)
            print(f"  Restart {i} ({label}): start={sc_r0:.10f}")

            # Quick LBFGS
            rem = deadline - 10 - time.monotonic()
            if rem > 8:
                z_r = _run_beta_schedule(z_r, n, deadline, min_remaining=10.0)
                a_r = _softmax(z_r)
                sc_r = _score(a_r)
                update_best(a_r, sc_r, f"restart_{label}_lbfgs")

                # Brief BB
                if time.monotonic() < deadline - 5:
                    a_r2, sc_r2 = _bb_subgradient(a_r, deadline - 4, n_steps=800, rel_tol=0.0003, use_nesterov=True)
                    update_best(a_r2, sc_r2, f"restart_{label}_bb")

    # === Phase 7: Final squeeze ===
    print(f"\n=== Phase 7: Final Squeeze (t={time.monotonic() - t_start:.1f}s) ===")
    if best_seq is not None and time.monotonic() < deadline - 2:
        # Very tight BB
        a_fin, sc_fin = _bb_subgradient(best_seq, deadline - 1, n_steps=3000, rel_tol=0.00005, use_nesterov=False)
        update_best(a_fin, sc_fin, "final_bb")

        # One more targeted nudge
        if time.monotonic() < deadline - 1:
            a_tn, sc_tn = _targeted_nudge(best_seq, deadline - 0.5, n_iters=100)
            update_best(a_tn, sc_tn, "final_nudge")

    # === Final validation ===
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        final[final < 0] = 0
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
