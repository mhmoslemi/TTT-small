import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _convolve_self(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    r = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return r

def _cgh_iterate(a, n_iter, flatten_power=1.0, clip=True):
    """CGH / Gerchberg-Saxton for flat spectrum."""
    n = len(a)
    current = a / a.sum()
    for i in range(n_iter):
        # FFT
        fa = np.fft.rfft(current, _next_pow2(2 * n - 1))
        # Flatten magnitude
        mag = np.abs(fa)
        power = np.sum(mag ** 2)
        target_mag = np.sqrt(power / len(mag)) ** flatten_power
        # Actually, for perfect flatness: |A| = sqrt(total_power / num_bins)
        # Let's use: |A_new| = (|A|^alpha * C) normalized to have same total power
        phase = fa / (mag + 1e-300)
        
        # Option 1: Perfect flat
        new_mag = np.full_like(mag, np.sqrt(power / len(mag)))
        # Option 2: Partial flattening
        # new_mag = mag ** (1-flatten_power) * (np.sqrt(power / len(mag)) ** flatten_power)
        
        fa_new = new_mag * phase
        a_new = np.fft.irfft(fa_new, _next_pow2(2 * n - 1))[:n]
        
        if clip:
            a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        current = a_new
    return current

"""
Hybrid optimization for minimum peak convolution sequences.
Combines CGH (Gerchberg-Saxton error reduction), projected subgradient with
backtracking, LP-based descent, and Adam polishing to minimize
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: CGH provides fast initial convergence by alternating
frequency-domain flattening with time-domain non-negativity clipping,
followed by precise subgradient/LP refinement.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv_self(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _score(a):
    c = _fft_conv_self(a)
    return _score_from_conv(a, c)


def _normalize(a):

    if s < 1e-30:

def _cgh_optimize(a, n_iter, relaxation=1.0, deadline=None):
    """CGH Error Reduction: alternate flat-spectrum projection and non-negativity clipping."""
    n = len(a)
    N = _next_pow2(2 * n - 1)
    current = np.maximum(a, 0).copy()
    s = current.sum()
    if s < 1e-30:
        return np.ones(n) / n
    current = current / s

    best = current.copy()
    best_c = _fft_conv_self(best)
    best_max = float(best_c.max())

    # Precompute flat target magnitude
    for i in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        fa = np.fft.rfft(current, N)
        mag = np.abs(fa)
        phase = fa / (mag + 1e-300)

        # Target: flat spectrum with same total power
        total_power = float(np.sum(mag ** 2))
        target_mag = np.full_like(mag, np.sqrt(total_power / len(mag)))

        # Relaxed update
        new_mag = (1.0 - relaxation) * mag + relaxation * target_mag
        fa_new = new_mag * phase
        a_new = np.fft.irfft(fa_new, N)[:n]

        # Time domain constraint: non-negativity + normalization
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-30:
            a_new = current
        else:
            a_new = a_new / s

        current = a_new

        # Check improvement every 500 iterations
        if (i + 1) % 500 == 0 or i == n_iter - 1:
            c = _fft_conv_self(current)
            cm = float(c.max())
            if cm < best_max:
                best_max = cm
                best = current.copy()

    return best


def _subgradient_peak(a, top_k=5):
    """Compute subgradient of max(conv(a,a)) using top-k peaks.
    Returns (subgradient_vector, conv, max_value).
    """
    n = len(a)
    c = _fft_conv_self(a)
    c_max = float(c.max())

    # Find top-k peaks
    if top_k >= len(c):
        peaks = np.arange(len(c))
    else:
        peaks = np.argsort(c)[::-1][:top_k]

    # Weight by softmax over peaks
    peak_vals = c[peaks]
    beta = 0.01 * c_max + 1e-10
    w_peak = np.exp((peak_vals - peak_vals.max()) / beta)
    w_peak = w_peak / w_peak.sum()

    # Subgradient: s[j] = 2 * sum_{k in peaks} w_k * a[k-j] (with bounds 0<=k-j<n)
    s = np.zeros(n)
    for idx, k in enumerate(peaks):
        w = w_peak[idx]
        # s[j] += 2*w*a[k-j] for j where 0<=k-j<n, i.e., k-n+1 <= j <= k
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # a[k-j] for j in [j_start, j_end] => a[k-j_end : k-j_start+1]
            a_slice = a[k - j_end:k - j_start + 1]
            s[j_start:j_end + 1] += 2.0 * w * a_slice[::-1]

    return s, c, c_max

def _proj_subgradient(a, deadline, lr=None, max_iter=100000, top_k=5, use_smooth=False, smooth_beta=None):
    """Projected subgradient method for minimizing max(conv(a,a)) on the simplex."""
    n = len(a)
    current = np.maximum(a, 0).copy()
    s = current.sum()
    if s < 1e-30:
        current = np.ones(n) / n
    else:
        current = current / s

    c = _fft_conv_self(current)
    best = current.copy()
    best_max = float(c.max())

    if lr is None:
        lr = 1.0 / np.sqrt(n)

    t = 0
    stalled = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        if use_smooth:
            g, c, c_max = _subgradient_smooth(current, smooth_beta)
        else:
            g, c, c_max = _subgradient_peak(current, top_k=top_k)

        if not np.all(np.isfinite(g)):
            break

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            new_a = current - step * g
            new_a = np.maximum(new_a, 0)
            ns = new_a.sum()
            if ns < 1e-15:
                step *= 0.5
                continue
            new_a = new_a / ns
            new_c = _fft_conv_self(new_a)
            new_max = float(new_c.max())
            if new_max < c_max * (1 - 1e-10):
                current = new_a
                improved = True
                if new_max < best_max:
                    best_max = new_max
                    best = current.copy()
                break
            else:
                step *= 0.5

        if not improved:
            stalled += 1
            step *= 0.9
            if stalled > 100:
                # Reset with current as step direction
                lr = step * 10
                stalled = 0
        else:
            stalled = 0
            # Gradually increase step size
            lr = min(lr * 1.001, 2.0 / np.sqrt(n))

    return best, best_max


def _adam_optimize(a_init, deadline, lr=0.005, max_inner_iter=50000):
    """Adam optimizer with log-sum-exp smoothed max for fine-tuning."""

    b1, b2, eps = 0.9, 0.999, 1e-8

    beta_scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

        beta = beta_scale * best_score

            t_step += 1

            c = _fft_conv_self(a)
            cm = float(c.max())

            e = np.exp(c_shifted)
            s_exp = float(e.sum())
            if s_exp < 1e-300 or not np.isfinite(s_exp):

                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_exp

            # Gradient computation using FFT
            N = _next_pow2(3 * n)
            W = np.fft.rfft(w, N)
            A = np.fft.rfft(a, N)
            wc = np.fft.irfft(np.conj(W) * A, N)[:n]

            W_total = float(np.dot(w, c))
            grad = 2.0 * n * (a * wc - a * W_total)

            # Project gradient onto tangent space (sum of gradient = 0)
            grad = grad - grad.mean()

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_descent(f, M, n, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    # Build sparse constraint matrix
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _interp_to_length(a, n_new):
    """Interpolate sequence to new length."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-30:
        return np.ones(n_new) / n_new
    return p / s


def _make_profile(n, shape='uniform', seed=0):
    """Create various starting profiles."""
    rng = np.random.default_rng(seed)
    if shape == 'uniform':
        return np.ones(n) / n
    elif shape == 'cosine':
        x = np.arange(n)
        a = 1.0 + np.cos(2 * np.pi * x / n)
        return a / a.sum()
    elif shape == 'triangle':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2 * x - 1)
        a = a ** 2
        return a / a.sum()
    elif shape == 'bimodal':
        x = np.linspace(0, 1, n)
        a = np.exp(-50 * (x - 0.2) ** 2) + np.exp(-50 * (x - 0.8) ** 2)
        return a / a.sum()
    elif shape == 'random_smooth':
        a = rng.uniform(0.5, 1.0, n)
        # Smooth
        for _ in range(3):
            a = np.convolve(a, np.ones(5) / 5, mode='same')
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        return np.ones(n) / n

                current_t *= 0.5
                stuck += 1

        update_best(f)
        print(f"  LP done: {lp_steps} steps, best={best_score:.10f}")

    # === Phase 5: Adam polishing + multi-restart ===
    phase5_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 5: Adam polish + restarts ===")
    if best_seq is not None and time.monotonic() < phase5_end - 30:
        a_best = best_seq / best_seq.sum()
        remaining = phase5_end - time.monotonic() - 20
        n_restarts = 4
        per_restart = max(30, remaining / n_restarts)
        sigmas = [0.02, 0.04, 0.07, 0.1]

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 20:
                break
            # Perturb in log-space
            z = np.log(np.maximum(a_best, 1e-15))
            z = z + sigmas[i] * rng.standard_normal(len(a_best))
            z = z - z.mean()
            a_pert = _softmax(z)
            a_pert = a_pert / a_pert.sum()

            dl_i = min(time.monotonic() + per_restart, phase5_end - 15)
            a_r, sc_r = _adam_optimize(a_pert, deadline=dl_i, lr=0.008)
            update_best(a_r)

        # Final Adam from best
        if time.monotonic() < phase5_end - 30:
            a_final, _ = _adam_optimize(a_best, deadline=phase5_end - 15, lr=0.003)
            update_best(a_final)

    # === Phase 6: Larger n exploration ===
    phase6_end = min(deadline, t_start + 1750.0)
    print(f"\n=== Phase 6: Larger n exploration ===")
    if best_seq is not None and time.monotonic() < phase6_end - 30:
        target_ns = [16384]
        for tn in target_ns:
            if time.monotonic() > phase6_end - 40:
                break
            if len(best_seq) >= tn:
                continue
            a_interp = _interp_to_length(best_seq, tn)
            update_best(a_interp)
            if time.monotonic() < phase6_end - 60:
                dl = min(time.monotonic() + 60, phase6_end - 20)
                a_big, _ = _adam_optimize(a_interp, deadline=dl, lr=0.005)
                update_best(a_big)

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_final = best_seq / best_seq.sum()
        dl = deadline - 5
        a_polished, _ = _adam_optimize(a_final, deadline=dl, lr=0.001)
        update_best(a_polished)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
