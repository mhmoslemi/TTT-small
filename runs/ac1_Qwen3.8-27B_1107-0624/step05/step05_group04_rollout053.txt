"""
Improved multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Top-k peak objective for smoother landscape during mid-optimization

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Score with sum(a) assumed to be 1 (or close)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _topk_score(a, k):
    """Sum of top k values of convolution."""
    c = _full_convolve(a, a)
    sorted_c = np.sort(c)[::-1]
    return float(sorted_c[:k].sum()), sorted_c[:k], c

def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    # Ramps with different slopes
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponentials
    for rate in [1.0, 2.0, 3.0, 5.0, 8.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Quadratic
    a = 1.0 + 2.0 * t + 0.5 * (t - 0.5)**2
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Sine-based
    a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    return profiles

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _grad_topk(z, beta_topk, k):
    """Gradient of top-k sum (smoothed) w.r.t. z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    n = len(a)
    
    # Smoothed top-k: use softmax over sorted values
    # For simplicity, use the indicator of top-k positions

    w = np.zeros(len(c))
    w[topk_idx] = 1.0 / k

    offset = float(np.dot(a, grad_a))

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_end = 1e-5 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 300.0)
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 3000)
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    w[topk_idx] = 1.0 / k

    w_conv = _full_convolve(a_rev, w)

    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    for c in [-0.5, 0.5, 1.0, 2.0, -1.0]:
        a = 1.0 + 2.0 * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    a = 1.0 + 2.0 * t + 0.3 * np.sin(4 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = 1.0 + 3.0 * t - 0.5 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    return profiles


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _downsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if target_n == 1:
        return np.array([1.0])
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _run_lbfgs(a_init, deadline, rng, n_betas=15):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.5 * max_c0
    beta_end = 1e-5 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 300.0)
        level_deadline = time.monotonic() + time_budget

        if beta > 0.1 * max_c0:
            maxiter = 200
        elif beta > 0.01 * max_c0:
            maxiter = 800
        else:
            maxiter = 2000

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 30, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_topk_adam(a_init, deadline, rng, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    n_sched = 40
    k_schedule = []
    for i in range(n_sched):
        frac = i / (n_sched - 1)
        k = max(1, int(round(np.ceil((1 - frac * 0.9) * int(n * 0.2) + frac * 0.9 * 1))))
        k_schedule.append(max(1, k))

    k_idx = 0
    steps_per_k = max(20, 20000 // len(k_schedule))

    for it in range(200000):
        if time.monotonic() > deadline:
            break

        if k_idx < len(k_schedule) - 1 and (it + 1) % steps_per_k == 0:
            k_idx += 1
        k = k_schedule[min(k_idx, len(k_schedule) - 1)]

        t += 1
        a = _softmax_fn(z)
        grad_z = _grad_topk(z, k)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 100 == 0:
            a = _softmax_fn(z)
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if t % 5000 == 0:
            lr = max(0.001, lr_init * (0.92 ** (t // 5000)))

        if t % 20000 == 0:
            z = z + 0.03 * rng.normal(size=n)
            z = z - z.mean()

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _spectral_flatten(a_init, deadline, rng, n_iter=20000):
    """Iterative spectral flattening with non-negativity projection."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    a = a_init.copy()

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Compute spectrum
        a_padded = np.zeros(N)
        a_padded[:n] = a
        A = np.fft.rfft(a_padded)
        P = np.abs(A) ** 2

        # Compression: pull P toward its median
        p_med = np.median(P[P > 0]) if np.any(P > 0) else 1.0
        if p_med < 1e-30:
            break

        # Alpha: compression strength (start gentle, increase over time)
        alpha = 0.7 + 0.2 * min(1.0, it / 5000.0)

        # Compress: P_new = P^alpha normalized to preserve total energy
        P_new = P ** alpha
        # Preserve zero values (where P was zero)
        P_new[P == 0] = 0

        # Update magnitudes, preserve phases
        scale = np.sqrt(P_new / np.maximum(P, 1e-30))
        scale[P == 0] = 0
        A_new = A * scale

        # Transform back
        a_new_full = np.fft.irfft(A_new, N)
        a_new = np.maximum(a_new_full[:n], 0.0)
        s = a_new.sum()
        if s < 1e-15:
            # Revert
            a_new = a.copy()
            continue
        a_new = a_new / s

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        # Occasionally add noise to escape local minima
        if it % 500 == 0:
            a = a + 0.01 * rng.normal(size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score


def _combinatorial_search(a_init, deadline, rng, max_moves=50000):
    """Local search: transfer mass between index pairs to reduce peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a = a_init.copy()

    c = _full_convolve(a, a)
    peak_val = float(c.max())

    for move in range(max_moves):
        if time.monotonic() > deadline:
            break

        # Recompute peaks periodically
        if move % 50 == 0:
            c = _full_convolve(a, a)
            peak_val = float(c.max())
            peak_indices = np.argsort(c)[::-1][:10]
            # Identify high-mass indices that contribute to peaks
            contrib_score = np.zeros(n)
            for pk in peak_indices[:5]:
                valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
                if len(valid_i) == 0:
                    continue
                prod = a[valid_i] * a[pk - valid_i]
                contrib_score[valid_i] += prod
            high_idx = np.argsort(contrib_score)[::-1][:20]
            low_idx = np.argsort(a)[:20]

        # Try a random move
        if move % 50 == 0:
            src = high_idx[rng.integers(len(high_idx))]
            dst = low_idx[rng.integers(len(low_idx))]
        else:
            src = rng.integers(n)
            dst = rng.integers(n)

        if src == dst:
            continue
        if a[src] < 1e-10:
            continue

        # Try several transfer amounts
        fracs = [0.01, 0.05, 0.1, 0.2, 0.5]
        for frac in fracs:
            amt = frac * a[src]
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            if a_trial[src] < 0 or a_trial[dst] < 0:
                continue

            # Quick score check (only top peaks)
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = float(2.0 * n * c_trial.max())

            if sc_trial < best_score - 1e-12:
                best_score = sc_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                break
        else:
            # No improvement, try reverse direction occasionally
            if rng.random() < 0.3:
                src, dst = dst, src
                if a[src] < 1e-10:
                    continue
                amt = 0.05 * a[src]
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] >= 0:
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc_trial = float(2.0 * n * c_trial.max())
                    if sc_trial < best_score - 1e-12:
                        best_score = sc_trial
                        best_a = a_trial.copy()
                        a = a_trial.copy()

        # Occasional large perturbation
        if move % 5000 == 0:
            a = a + 0.02 * rng.normal(size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s

    return best_a, best_score


def _peak_equalize_v2(a_init, deadline, rng, max_rounds=2000):
    """Improved peak equalization: target all near-max peaks simultaneously."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    lr = 0.01
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find all peaks within 2% of max
        threshold = cm * 0.98
        active_peaks = np.where(c >= threshold)[0]

        if len(active_peaks) == 0:
            break

        # Compute gradient: reduce all active peaks
        w = np.zeros(len(c))
        w[active_peaks] = 1.0 / len(active_peaks)

        a_rev = a_cur[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a_cur, grad_a))
        grad_z = a_cur * (grad_a - offset)

        # Adam update
        if not hasattr(_peak_equalize_v2, '_m'):
            _peak_equalize_v2._m = np.zeros(n)
            _peak_equalize_v2._v = np.zeros(n)
            _peak_equalize_v2._t = 0

        _peak_equalize_v2._t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        _peak_equalize_v2._m = b1 * _peak_equalize_v2._m + (1 - b1) * grad_z
        _peak_equalize_v2._v = b2 * _peak_equalize_v2._v + (1 - b2) * (grad_z ** 2)
        mh = _peak_equalize_v2._m / (1 - b1 ** _peak_equalize_v2._t)
        vh = _peak_equalize_v2._v / (1 - b2 ** _peak_equalize_v2._t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if rnd % 100 == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

        if rnd % 2000 == 0:
            lr = max(0.001, lr * 0.8)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    # Reset state for peak_equalize_v2
    _peak_equalize_v2._m = None
    _peak_equalize_v2._v = None
    _peak_equalize_v2._t = 0

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s", flush=True)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}", flush=True)
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(256, 2.0)
        update_best(a0, _score_norm_fn(a0), "default_ramp")
        best_seq = a0.copy()
        best_score = _score_norm_fn(a0)

    total_budget = deadline - t_start

    # Phase 1: Multi-scale warm start (35% of time)
    phase1_frac = 0.35
    phase1_end = t_start + total_budget * phase1_frac
    print(f"\nPhase 1: Multi-scale (deadline +{phase1_end - t_start:.0f}s)", flush=True)

    n_start = 128
    profiles = _make_profiles(n_start)
    a_cur = None
    sc_cur = np.inf

    for prof in profiles:
        sc = _score_norm_fn(prof)
        if sc < sc_cur:
            sc_cur = sc
            a_cur = prof.copy()

    # If parent available, downsample it for warm start
    if parent is not None:
        a_ds = _downsample(parent, n_start)
        sc_ds = _score_norm_fn(a_ds)
        if sc_ds < sc_cur:
            a_cur = a_ds
            sc_cur = sc_ds

    print(f"  Starting profile at n={n_start}: score={sc_cur:.10f}", flush=True)

    # Quick L-BFGS at n_start
    quick_dl = min(phase1_end, t_start + 12.0)
    a_cur, sc_cur = _run_lbfgs(a_cur, quick_dl, rng, n_betas=8)
    print(f"  After n={n_start} L-BFGS: {sc_cur:.10f}", flush=True)

    # Multi-scale up
    scales = [256, 512, 1024, 2048, 4096]
    phase1_betas = [8, 10, 12, 14, 16]

    for i, target_n in enumerate(scales):
        if time.monotonic() > phase1_end - 15:
            break

        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)

        remaining_phase1 = phase1_end - time.monotonic()
        if remaining_phase1 < 20:
            break

        alloc = remaining_phase1 * (0.8 / (len(scales) - i))
        opt_dl = min(phase1_end - 5, time.monotonic() + alloc)

        a_opt, sc_opt = _run_lbfgs(a_target, opt_dl, rng, n_betas=phase1_betas[i])

        # Add a quick top-k pass
        if time.monotonic() < phase1_end - 10:
            topk_dl = min(phase1_end - 5, time.monotonic() + 10.0)
            a_opt2, sc_opt2 = _run_topk_adam(a_opt, topk_dl, rng, lr_init=0.03)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2

        print(f"  n={target_n}: {sc_target:.10f} -> {sc_opt:.10f}", flush=True)

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")
    else:
        a_cur = best_seq.copy()
        sc_cur = best_score

    print(f"  Phase 1 done: best={best_score:.10f}", flush=True)

    # Phase 2: Spectral flattening (20% of time)
    phase2_frac = 0.20
    phase2_end = t_start + total_budget * (phase1_frac + phase2_frac)
    print(f"\nPhase 2: Spectral flattening (deadline +{phase2_end - t_start:.0f}s)", flush=True)

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        _peak_equalize_v2._m = None
        _peak_equalize_v2._v = None
        _peak_equalize_v2._t = 0

        sf_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        a_sf, sc_sf = _spectral_flatten(best_seq, sf_dl, rng)
        update_best(a_sf, sc_sf, "spectral")

    # Phase 3: Top-k Adam refinement (20% of time)
    phase3_frac = 0.20
    phase3_end = t_start + total_budget * (phase1_frac + phase2_frac + phase3_frac)
    print(f"\nPhase 3: Top-k Adam (deadline +{phase3_end - t_start:.0f}s)", flush=True)

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        remaining = phase3_end - time.monotonic()
        for lr_try in [0.05, 0.02, 0.01]:
            if time.monotonic() > phase3_end - 15:
                break
            dl = min(phase3_end - 10, time.monotonic() + remaining * 0.35)
            a_r, sc_r = _run_topk_adam(best_seq, dl, rng, lr_init=lr_try)
            update_best(a_r, sc_r, f"topk_lr{lr_try}")

    # Phase 4: Peak equalization (10% of time)
    phase4_frac = 0.10
    phase4_end = t_start + total_budget * (phase1_frac + phase2_frac + phase3_frac + phase4_frac)
    print(f"\nPhase 4: Peak equalization (deadline +{phase4_end - t_start:.0f}s)", flush=True)

    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        _peak_equalize_v2._m = None
        _peak_equalize_v2._v = None
        _peak_equalize_v2._t = 0
        pe_dl = min(phase4_end - 5, time.monotonic() + (phase4_end - time.monotonic()))
        a_pe, sc_pe = _peak_equalize_v2(best_seq, pe_dl, rng)
        update_best(a_pe, sc_pe, "peak_eq")

    # Phase 5: Combinatorial search (10% of time)
    phase5_frac = 0.05
    phase5_end = t_start + total_budget * (phase1_frac + phase2_frac + phase3_frac + phase4_frac + phase5_frac)
    print(f"\nPhase 5: Combinatorial (deadline +{phase5_end - t_start:.0f}s)", flush=True)

    if time.monotonic() < phase5_end - 5 and best_seq is not None:
        cs_dl = min(phase5_end - 3, time.monotonic() + (phase5_end - time.monotonic()))
        a_cs, sc_cs = _combinatorial_search(best_seq, cs_dl, rng)
        update_best(a_cs, sc_cs, "combinatorial")

    # Phase 6: Final L-BFGS polish
    print(f"\nPhase 6: Final polish", flush=True)
    if time.monotonic() < deadline - 3 and best_seq is not None:
        fin_dl = min(deadline - 1, time.monotonic() + min(120.0, deadline - time.monotonic()))
        a_fin, sc_fin = _run_lbfgs(best_seq, fin_dl, rng, n_betas=25)
        update_best(a_fin, sc_fin, "final_lbfgs")

        # One more spectral pass if time
        if time.monotonic() < deadline - 3:
            sf_dl2 = min(deadline - 1, time.monotonic() + 30.0)
            a_sf2, sc_sf2 = _spectral_flatten(best_seq, sf_dl2, rng, n_iter=5000)
            update_best(a_sf2, sc_sf2, "spectral_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 2.0)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}", flush=True)
        return [float(x) for x in final_seq]

    fallback = _make_ramp(256, 2.0)
    return [float(x) for x in fallback]
