def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _self_conv(a):
    return _fft_convolve(a, a)

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1, returns 2n * max(c)"""
    n = len(a)
    c = _self_conv(a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smooth_max_and_grad(a, beta):
    """Returns (smooth_max_value, gradient) for beta*logsumexp(c/beta) where c = a*a"""
    n = len(a)
    c = _self_conv(a)
    c_shifted = c - c.max()
    e = np.exp(c_shifted / beta)
    s = e.sum()
    w = e / s  # softmax weights
    val = float(c.max() + beta * np.log(s))
    
    # grad[j] = 2 * sum_m w[m] * a[m-j]
    # = 2 * (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    grad = 2.0 * wc[n-1:2*n-1]  # positions n-1 to 2n-2
    
    return val, grad

```python
def _adam_smooth_max(a_init, deadline, beta_schedule, lr=0.01, max_iter=10000000):
    """Optimize using Adam on the smooth max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax(z)
        beta = beta_schedule  # or computed from t
        
        val, grad = _smooth_max_and_grad(a, beta)
        
        # Project gradient onto simplex tangent space
        # The constraint is sum(exp(z)) = const, so the effective gradient in z-space is:
        # grad_z[j] = a[j] * (grad[j] - dot(a, grad))
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1**t)
        vh = v_buf / (1.0 - b2**t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update for stability
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        # Check improvement periodically
        if t % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Final check
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score
```

Uses fast FFT-based Adam on log-sum-exp smoothed max, LP active-set refinement,

import time
from scipy.optimize import linprog
import scipy.sparse as sp

    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    return _fft_convolve(a, a)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _self_conv(a)

def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_max_grad(a, beta):
    n = len(a)
    c = _self_conv(a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / max(beta, 1e-30)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    val = c_max + beta * np.log(max(s, 1e-300))
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    return val, grad


def _adam_smooth_max(a_init, deadline, beta_start, beta_end, lr=0.01,
                     max_iter=10000000, check_interval=200, lr_min=0.001, lr_max=0.05):
    n = len(a_init)

        frac = min(1.0, t / max_iter)

def _build_lp_sparse(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    return rows, cols


def _lp_step(f, rows, cols, time_limit=30):
    n = len(f)
    c = _self_conv(f)
    M = float(c.max())
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return f, M
        g_0 = np.maximum(res.x, 0.0)
    except Exception:
        return f, M
    if g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t_val) * f + t_val * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t_val
    if best_t == 0.0:
        for t_val in [0.02, 0.05, 0.15]:
            f_cand = (1.0 - t_val) * f + t_val * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M
