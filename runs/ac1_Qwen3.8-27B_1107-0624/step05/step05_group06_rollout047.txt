def _upsample(a_in, n_out):
    """Upsample using linear interpolation in log-space."""
    n_in = len(a_in)
    t_in = np.linspace(0, 1, n_in)
    t_out = np.linspace(0, 1, n_out)
    log_a = np.log(np.maximum(a_in, 1e-15))
    log_a_out = np.interp(t_out, t_in, log_a)
    a_out = np.exp(log_a_out)
    return a_out / a_out.sum()

        a_sg2, sc_sg2 = _subgrad_polish(best_seq, sg_deadline, n_steps=3000,
                                        lr_init=0.02, mode="exact")
        update_best(a_sg2, sc_sg2, "exact_sg")

    # Phase 4: Multi-length exploration (n=4096)
    print(f"\n--- Phase 4: Multi-length exploration ---")
    if time.monotonic() < deadline - 250:
        n_explore = 4096
        a_explore = _upsample(best_seq, n_explore) if best_seq is not None \
            else _make_structural_init(n_explore, "cosine_bump")
        sc_explore = _score(a_explore)
        print(f"  Upsampled n={n_explore}, initial score={sc_explore:.10f}")

        # Run a shorter beta-continuation on the longer sequence
        explore_deadline = deadline - 100.0
        a_explore_opt = _run_beta_continuation(a_explore, explore_deadline,
                                               min_remaining=60.0, label="explore_4096")
        update_best(a_explore_opt, None, "explore_4096")

        # Quick subgradient on the best of the two
        if best_seq is not None and time.monotonic() < deadline - 80:
            sg_d = min(deadline - 80.0, time.monotonic() + 40.0)
            a_explore_sg, sc_explore_sg = _subgrad_polish(
                a_explore_opt, sg_d, n_steps=1500, lr_init=0.02, mode="peak")
            update_best(a_explore_sg, sc_explore_sg, "explore_4096_sg")

    # Phase 5: Fine L-BFGS on best sequence
    print(f"\n--- Phase 5: Fine L-BFGS ---")
    if best_seq is not None and time.monotonic() < deadline - 60:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.00003, 0.000015, 0.00001, 0.000005])

        z_fin = _log_map(best_seq)
        z_fin = z_fin - z_fin.mean()

        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - 40 - time.monotonic()
            if remaining < 5:
                break
            max_iter = min(2000, max(50, int(remaining / 0.05)))
            z_fin, _, _ = _lbfgs_phase(z_fin, beta, n, max_iter)
            a_fin = _softmax(z_fin)
            update_best(a_fin, None, f"fine_{i}")

    # Phase 6: Combinatorial local search
    print(f"\n--- Phase 6: Combinatorial local search ---")
    if best_seq is not None and time.monotonic() < deadline - 30:
        cs_deadline = min(deadline - 25.0, time.monotonic() + 60.0)
        a_cs, sc_cs = _combinatorial_local_search(best_seq, cs_deadline,
                                                  n_trials=3000)
        update_best(a_cs, sc_cs, "combinatorial")

    # Phase 7: Final micro-polish
    print(f"\n--- Phase 7: Final micro-polish ---")
    if best_seq is not None and time.monotonic() < deadline - 5:
        micro_deadline = deadline - 3.0
        a_micro, sc_micro = _subgrad_polish(best_seq, micro_deadline,
                                            n_steps=1000, lr_init=0.005,
                                            mode="peak")
        update_best(a_micro, sc_micro, "micro_polish")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_structural_init(256, "cosine_bump")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_structural_init(256, "cosine_bump")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
