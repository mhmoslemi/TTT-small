"""
Multi-phase convex optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA on simplex with continuation, then Adam in log-space with adaptive
learning rate, multi-n exploration, and ultra-fine polishing.
Key improvements: adaptive LR, more aggressive beta schedule, n=8192 exploration,
longer fine-phase time, and better time management.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        c_shifted = (c_y - c_max) / beta
        w = np.exp(c_shifted)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, lr_decay=0.995, decay_period=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    prev_score = best_score
    patience = 0
    max_patience = 20000

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_step = 0.3
            if update_norm > max_step:
                update = update * (max_step / update_norm)

            z_new = z - update
            z_new = np.clip(z_new, -50.0, 50.0)
            z_new = z_new - z_new.mean()
            z = z_new

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    patience = 0
                else:
                    patience += 50

                if patience > max_patience and sc > prev_score * 1.01:
                    current_lr *= 0.9
                    patience = 0

            prev_score = _score_norm(_softmax_compute(z)) if t % 50 == 0 else prev_score

            if it_in_phase % decay_period == 0:
                current_lr = max(0.0005, current_lr * lr_decay)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asymmetric":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.3) * (t - 0.3) * n * 0.1)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _fista_betas(a, max_cc, n_betas=15):
    c = _full_convolve(a, a)
    max_c = float(c.max())
    betas = []
    beta = 0.1 * max_c

        betas.append(beta)
        beta *= 0.5
    return betas


def _adam_betas(max_cc, n_betas=8, start_frac=0.005):
    betas = []
    beta = start_frac * max_cc
    for _ in range(n_betas):
        betas.append(beta)
        beta *= 0.4
    return betas

        print(f"Evaluator-consistent score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluate_sequence score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
