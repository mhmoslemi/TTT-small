def _cd_step(a, conv, k):

    Optimize coordinate k of a given current convolution conv.
    Returns (new_a_k, delta_conv) where delta_conv is the change to add to conv.

        if q_idx >= 0:
            vals = vals.copy()
            vals[q_idx] += t * t

        vals = R + B * t
        if q_idx >= 0:
            vals = vals.copy()
            vals[q_idx] += t * t

        if q_idx >= 0 and active[q_idx]:

def _lbfgs_run(a_init, beta, n, max_iter=3000):
    """Run L-BFGS-B for one beta."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    try:
        result = minimize(
            _smoobj, z0, jac=_smograd, method='L-BFGS-B',
            args=(beta, n),
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11,
                     'maxcor': 50, 'maxls': 100}
        )
        return _softmax(result.x)
    except Exception:
        return a_init


def _adaptive_lbfgs(a_init, deadline, min_time=30.0):
    """
    Adaptive beta-continuation L-BFGS.
    Starts with moderate beta and decreases adaptively based on progress.
    """
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a_init
    
    beta = max_c * 0.2
    current_a = a_init.copy()
    current_sc = _score(current_a)
    stalled = 0
    
    while time.monotonic() + min_time < deadline:
        remaining = deadline - min_time - time.monotonic()
        if remaining < 10:
            break
        
        max_iter = min(2000, max(100, int(remaining / 0.003)))
        a_new = _lbfgs_run(current_a, beta, n, max_iter)
        sc_new = _score(a_new)
        
        if sc_new < current_sc - 1e-10:
            current_a = a_new
            current_sc = sc_new
            stalled = 0
            # Decrease beta (make smoothing tighter)
            beta *= 0.5
            print(f"    LBFGS: beta={beta:.6e}, score={current_sc:.10f}")
        else:
            stalled += 1
            if stalled >= 3:
                break
            beta *= 0.7
    
    return current_a


def _subgrad_correct(a, n_iter=2000, lr=0.01, deadline=None):
    """
    Subgradient method with exact active set, in the probability simplex.
    Uses normalized gradient and projection onto {a>=0, sum=1}.
    """
    n = len(a)
    a = a.copy() / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for i in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Active set: lags within 0.5% of max
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient of max(w . c) with respect to z
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        
        # Normalized step
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = lr / (1.0 + 0.01 * i)
        z = z - step * grad / gn
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 200 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a


def _perturb_z(a, sigma, rng):
    """Perturb in z-space (log space)."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)


def _perturb_structure(a, rng, n_harmonics=5):
    """Perturb with random harmonic structure in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_harmonics):
        freq = rng.integers(1, max(2, n // 3))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _make_tapered_profile(n, sharpness=1.0):
    """Create a tapered (sinusoidal) profile."""
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** sharpness
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid CD + L-BFGS optimizer with adaptive beta continuation.
    Uses exact coordinate descent as complement to smoothed L-BFGS.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def _update(p, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent loaded: n={n_main}, score={sc0:.10f}")
        _update(a0, "parent")
    else:
        n_main = 2048
        a0 = _make_tapered_profile(n_main, sharpness=0.8)
        print(f"No parent, using tapered n={n_main}, score={_score(a0):.10f}")
        _update(a0, "init_tapered")
    
    n = n_main
    
    # --- Phase 1: Coordinate Descent Warmup ---
    print(f"\nPhase 1: CD warmup (t={time.monotonic()-t_start:.1f}s)")
    phase1_deadline = min(deadline - 600.0, t_start + 60.0)
    a_cd1 = _coordinate_descent(a0, n_passes=30, seed=seed,
                                time_budget=phase1_deadline - t_start, t_start=t_start)
    _update(a_cd1, "cd_warmup")
    
    # --- Phase 2: Main Adaptive L-BFGS ---
    print(f"\nPhase 2: Adaptive L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    phase2_deadline = min(deadline - 200.0, time.monotonic() + 600.0)
    a_main = _adaptive_lbfgs(best_seq, phase2_deadline, min_time=100.0)
    _update(a_main, "main_lbfgs")
    
    # --- Phase 3: CD Polish ---
    print(f"\nPhase 3: CD polish (t={time.monotonic()-t_start:.1f}s)")
    phase3_deadline = min(deadline - 100.0, time.monotonic() + 120.0)
    a_cd2 = _coordinate_descent(best_seq, n_passes=50, seed=seed + 1,
                                time_budget=phase3_deadline - t_start, t_start=t_start)
    _update(a_cd2, "cd_polish")
    
    # --- Phase 4: Fine L-BFGS ---
    print(f"\nPhase 4: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 60:
        phase4_deadline = min(deadline - 30.0, time.monotonic() + 300.0)
        a_fine = _adaptive_lbfgs(best_seq, phase4_deadline, min_time=30.0)
        _update(a_fine, "fine_lbfgs")
    
    # --- Phase 5: Subgradient polishing ---
    print(f"\nPhase 5: Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 20:
        phase5_deadline = min(deadline - 10.0, time.monotonic() + 60.0)
        a_sg = _subgrad_correct(best_seq, n_iter=3000, lr=0.005, deadline=phase5_deadline)
        _update(a_sg, "subgrad")
    
    # --- Phase 6: Multi-restart ---
    print(f"\nPhase 6: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 30:
        remaining = deadline - 30 - time.monotonic()
        n_restarts = min(5, max(1, int(remaining / 120)))
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            
            if i == 0:
                p_start = _perturb_z(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_structure(best_seq, rng, n_harmonics=3)
            elif i == 2:
                p_start = _perturb_z(best_seq, 0.01, rng)
            elif i == 3:
                p_start = _perturb_structure(best_seq, rng, n_harmonics=7)
            else:
                p_start = _make_tapered_profile(n, sharpness=rng.uniform(0.5, 2.0))
            
            print(f"  Restart {i}: start_score={_score(p_start):.10f}")
            
            # Short CD
            r_deadline = min(deadline - 25 - time.monotonic(), 30.0)
            p_cd = _coordinate_descent(p_start, n_passes=10, seed=seed + 100 + i,
                                       time_budget=r_deadline, t_start=time.monotonic())
            _update(p_cd, f"restart_{i}_cd")
            
            # Short L-BFGS
            r_deadline2 = min(deadline - 25, time.monotonic() + 60.0)
            p_lbfgs = _adaptive_lbfgs(p_cd, r_deadline2, min_time=10.0)
            _update(p_lbfgs, f"restart_{i}_lbfgs")
            
            # Short subgradient
            if time.monotonic() < deadline - 20:
                p_sg = _subgrad_correct(best_seq if _score(best_seq) < _score(p_lbfgs) else p_lbfgs,
                                        n_iter=500, lr=0.003,
                                        deadline=min(deadline - 15, time.monotonic() + 20.0))
                _update(p_sg, f"restart_{i}_sg")
    
    # --- Phase 7: Final CD + L-BFGS cycle ---
    print(f"\nPhase 7: Final cycle (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Final CD
        a_fin_cd = _coordinate_descent(best_seq, n_passes=10, seed=seed + 999,
                                       time_budget=20.0, t_start=time.monotonic())
        _update(a_fin_cd, "final_cd")
        
        # Final L-BFGS with tiny beta
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 0.00001
        if time.monotonic() < deadline - 3:
            a_fin = _lbfgs_run(best_seq, beta_final, n, max_iter=500)
            _update(a_fin, "final_lbfgs")
    
    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_tapered_profile(256, 1.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    # Fallback
    fallback = _make_tapered_profile(256, 1.0)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
