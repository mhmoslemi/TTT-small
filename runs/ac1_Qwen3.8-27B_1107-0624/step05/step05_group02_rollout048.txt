            # But since C_k = sum_j a_j a_{k-j}, dC_k/da_i = a_{k-i} (from first factor) + a_{k-i} (from second factor, when j = k-i gives a_{k-i}*a_i, derivative w.r.t. a_i gives a_{k-i})
            # Wait: C_k = sum_j a_j * a_{k-j}
            # dC_k/da_i: takes terms where j=i (gives a_{k-i}) or k-j=i i.e. j=k-i (gives a_i * 1 = a_i... no)
            # Let me be more careful: C_k = sum_j a_j * a_{k-j}
            # The term involving a_i appears when j=i: a_i * a_{k-i}, and when k-j=i, i.e., j=k-i: a_{k-i} * a_i
            # So dC_k/da_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (if i != k-i)
            # Or dC_k/da_i = a_i (if i = k-i, i.e., 2i = k)
            
            # The gradient of sum_k w_k * C_k is:
            # grad_i = sum_k w_k * dC_k/da_i = sum_k w_k * (a_{k-i} + a_{k-i}) [with appropriate bounds]
            # = 2 * sum_k w_k * a_{k-i} = 2 * (w * a_rev_shifted)[i]
            # where a_rev_shifted accounts for the shift
            
            # More precisely, define h_i = sum_k w_k * a_{k-i} where the sum is over valid k-i
            # h = convolve(w, a_flipped) shifted...
            
            # Let's use: h_i = sum_k w_k * a_{k-i} for valid indices
            # This is (w * a)[shifted]... let me think in terms of cross-correlation
            # h_i = sum_k w_k * a_{k-i} = (w cross-correlated with a) at lag i
            # = convolve(w, a_flipped)[i + len(a) - 1]
            
            a_flipped = a[::-1]
            wc = _full_convolve(w, a_flipped)
            # wc has length len(w) + len(a) - 1 = (2n-1) + n - 1 = 3n-2
            # h_i = wc[i + (n-1)] for i = 0, ..., n-1
            h = wc[n - 1:2 * n - 1]
            
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            # Normalization by (sum a)^2 - but since we're in log space, sum a changes
            # Actually, the objective is 2n * max(C) / (sum a)^2
            # In log space with softmax parameterization, sum a = sum exp(z) / sum exp(z) = 1
            # Wait, _softmax_z normalizes to sum=1, so the objective simplifies to 2n * max(C)
            # The gradient of 2n * (w . C) w.r.t. z is:
            # = 2n * sum_k w_k * dC_k/da_i * da_i/dz_i
            # da_i/dz_i = a_i (a_i - delta_ii) for softmax
            # This gets complicated. Let's just use the gradient w.r.t. a and adjust.
            
            # Simpler: the gradient w.r.t. z for the softmax parameterization is:
            # grad_z_i = a_i * (grad_a_i - mean(grad_a))
            # where grad_a is the gradient w.r.t. a (with the constraint that a is on the simplex)
            
            # Actually, for the objective f(a) = 2n * (w.C) (with sum a = 1):
            # The gradient on the simplex is: grad_a_i - (sum_j a_j * grad_a_j) * (projecting to tangent space)
            # But since we're using the softmax parameterization, we can just use:
            # grad_z_i = a_i * grad_a_i (the "exponential" gradient)
            # and Adam will handle the rest.
            
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = cur_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if iters % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_mc:
                    best_mc = sc
                    best_a = a.copy()
                    if verbose:
                        print(f"      Adam it={iters}: max_C={sc:.10f}, best={best_mc:.10f}")

            if iters % 5000 == 0:
                cur_lr = max(0.0005, cur_lr * 0.95)

    a_final = _softmax_z(z)
    mc_final = float(_full_convolve(a_final, a_final).max())
    if mc_final < best_mc:
        best_a = a_final.copy()

    return best_a


def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_raised_cosine(n):
    i = np.arange(n)
    a = np.cos(np.pi * (i + 0.5) / n) ** 2
    return a / a.sum()


def _make_random(n, rng):
    a = np.abs(rng.standard_normal(n))
    return a / a.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    z_shift = z - np.max(z)
    e = np.exp(z_shift)
    return e / e.sum()
