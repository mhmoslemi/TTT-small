import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _gradient_smoothed(a, c, beta):
    """Compute gradient of 2n * logsumexp(beta * c) / sum(a)^2 with respect to a.
    Actually, we work with the log parameterization, so let's compute d/dz.
    
    We compute: grad_a = 4n * sum_k w_k * a[k-i] (valid)
    where w_k = exp(beta*c[k]) / sum_j exp(beta*c[j])
    """
    n = len(a)
    c_shifted = c - c.max()
    e = np.exp(beta * c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    # grad_a[i] = 4n * sum_j w[j+i] * a[j] for j=0..n-1
    # This is cross-correlation of w (length 2n-1) and a (length n)
    # cross_corr[i] = sum_j w[j+i] * a[j]
    # Using FFT: 
    # Pad a to length 2n-1 (or next pow2), compute FFT
    # cross_corr = ifft(fft(w) * conj(fft(a_padded)))... 
    
    # Actually, let's use the formula directly:
    # grad_a[i] = 4n * sum_{j=0}^{n-1} w[j+i] * a[j]
    # This is the (i-th) element of the "unnormalized cross-correlation"
    
    # Using FFT: 
    # Let A = fft(a, N) where N >= 2n-1
    # Let W = fft(w, N)  
    # Then the circular cross-correlation is ifft(W * conj(A))
    # But we want the linear (not circular) cross-correlation
    
    # Actually, for the linear cross-correlation:
    # (corr)(i) = sum_j w[j+i] * a[j] for j=0..n-1
    # This is the same as: for each i, it's the inner product of w[i:i+n] with a
    # = (w * reverse(a))[i + n - 1] ... let me think
    
    # Standard cross-correlation: (w ⋆ a)[i] = sum_j w[j] * a[j+i]
    # We want: sum_j w[j+i] * a[j] = sum_k w[k] * a[k-i] (reindexing j→k-i)
    # = (w ⋆ a)[i] ... no wait
    # sum_j w[j+i] * a[j] = sum_k w[k] * a[k-i] where k=j+i
    # This is the cross-correlation (w ⋆ a)[i] = sum_k w[k] a[k-i]
    
    # Using FFT: cross-correlation can be computed as ifft(fft(w) * conj(fft(a_padded)))
    # where a_padded is a padded to length 2n-1 or more
    
    N = _next_pow2(2 * n)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:2*n-1] = w
    
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    corr = np.fft.irfft(W * np.conj(A), N)
    # corr[i] = sum_k w_pad[k] * a_pad[(k-i) mod N]
    # For i in 0..n-1, and since both are zero-padded to N >= 2n-1:
    # corr[i] = sum_{k=0}^{2n-2} w[k] * a_pad[(k-i) mod N]
    # For 0 ≤ i ≤ n-1 and 0 ≤ k ≤ 2n-2:
    # (k-i) mod N = k-i if k≥i, else k-i+N
    # a_pad[m] = a[m] if 0≤m<n, else 0
    # So we need 0 ≤ k-i < n (for k≥i) or k-i+N < n (impossible since N≥2n-1 > n)
    # So corr[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i] = sum_{j=0}^{n-1} w[j+i] * a[j] ✓
    
    grad_a = 4.0 * n * corr[:n]
    return grad_a

def _soft_max_objective(a, beta):
    """Compute the smoothed objective 2n * logsumexp(beta*c) / (beta * sum(a)^2)
    Actually, let's just compute 2n * (max c approximated by logsumexp) / sum^2
    For optimization, we minimize logsumexp(beta*c)/beta which approximates max(c)
    The score is 2n * max(c) / sum^2, so we minimize logsumexp(beta*c)/beta / sum^2 * 2n
    """
    n = len(a)
    s = a.sum()
    c = _full_convolve(a, a)
    c_shifted = c - c.max()
    logsumexp = c.max() + np.log(np.exp(beta * c_shifted).sum()) / beta
    return float(2.0 * n * logsumexp / (s * s))

def _adam_smooth(a_init, beta_schedule, deadline, lr=0.01, beta_min=50.0):
    """Optimize using Adam with log-sum-exp smoothing and beta annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        iters_this_beta = 0
        max_iters = 5000  # per beta level
        while iters_this_beta < max_iters:
            if time.monotonic() > deadline:
                break
            iters_this_beta += 1
            t_step += 1
            
            a = _softmax(z)
            c = _full_convolve(a, a)
            
            # Compute weights
            c_shifted = c - c.max()
            e = np.exp(beta * c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            
            # Compute gradient with respect to a
            N = _next_pow2(2 * n)
            a_pad = np.zeros(N)
            a_pad[:n] = a
            w_pad = np.zeros(N)
            w_pad[:2*n-1] = w
            grad_a = 4.0 * n * np.fft.irfft(np.fft.rfft(w_pad) * np.conj(np.fft.rfft(a_pad)), N)[:n]
            
            # Transform to z-space
            g_a_mean = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - g_a_mean)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            # Adam update
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            m_hat = m_buf / (1.0 - b1**t_step)
            v_hat = v_buf / (1.0 - b2**t_step)
            
            update = lr * m_hat / (np.sqrt(v_hat) + eps)
            # Clip update
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = z - z.mean()
            
            # Check improvement periodically
            if t_step % 100 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"    beta={beta:.1f}, iter={t_step}, score={sc:.10f}")
    
    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_a = a_final.copy()
        best_score = sc_final
    
    return best_a, best_score

def _coord_descent(a, max_sweeps=10, deadline=None):
    """Coordinate descent: for each i, find best a[i] >= 0 to minimize max(c)."""
    n = len(a)
    a = a.copy()
    s = a.sum()
    a = a / s  # normalize
    
    for sweep in range(max_sweeps):
        if deadline and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        current_max = c.max()
        
        for i in range(n):
            if deadline and time.monotonic() > deadline:
                    break
            
            old_val = a[i]
            if old_val < 1e-15:
                # Try a small positive value
                candidate = 1e-10
                # Check if it helps
                # c changes by: 2*candidate*a[k-i] for k where valid, plus candidate^2 for k=2i
                # This is a small perturbation, skip for now
                continue
            
            # We want to find t >= 0 that minimizes max_k c_new[k]
            # where c_new[k] = c[k] + delta_k
            # delta[k] = (t^2 - old_val^2) if k=2i
            #            2*(t - old_val)*a[k-i] if k != 2i and 0 <= k-i < n
            #            0 otherwise
            
            #pre-compute the linear coefficients
            # For each k, delta[k] = alpha_k * t + beta_k * t^2
            # alpha[k] = 2*a[k-i] if k != 2i and 0<=k-i<n, else 0
            #            2*old_val if k==2i (from the -old_val^2 term... wait)
            
            # Let me redo: c_new[k] = c[k] + (t - old_val) * lin_k + (t^2 - old_val^2) * quad_k
            # where quad_k = 1 if k=2i, else 0
            # and lin_k = 2*a[k-i] - 2*quad_k*old_val ... 
            # 
            # Actually: the change in c[k] when a[i] changes from old to new:
            # ∂c[k]/∂a[i] = 2*a[k-i] (if valid)
            # So c_new[k] ≈ c[k] + (new - old) * 2*a[k-i] (for the linear part)
            # Plus the quadratic correction: if k=2i, there's a (new^2 - old^2) term
            # 
            # More precisely: c[k] contains the term a[i]*a[k-i] (from j=i) and a[k-i]*a[i] (from j=k-i)
            # When we change a[i] to t:
            # If k=2i: the term was a[i]^2 = old^2, now it's t^2. Change: t^2 - old^2
            # If k≠2i (and valid): the terms were 2*a[i]*a[k-i] = 2*old*a[k-i], now 2*t*a[k-i]. Change: 2*(t-old)*a[k-i]
            # 
            # So: c_new[k] = c[k] + (t^2 - old^2) if k=2i
            #                       c[k] + 2*(t-old)*a[k-i] if valid and k≠2i
            #                       c[k] otherwise
            
            # For bisection: we want to find t that minimizes max_k c_new[k]
            # The function is convex in t (max of convex quadratics/linears)
            # Use ternary search on [0, 2*old_val] (or [0, 3*old_val] for safety)
            
            # Precompute the "base" and coefficients
            k_range = np.arange(2*n-1)
            valid = (k_range - i) >= 0
            valid &= (k_range - i) < n
            # Lin part: 2*a[k-i] for valid, excluding k=2i
            lin = np.zeros(2*n-1)
            lin[valid & (k_range != 2*i)] = 2.0 * a[k_range[valid & (k_range != 2*i)] - i]
            # Quad part: 1 for k=2i
            quad = np.zeros(2*n-1)
            if 0 <= 2*i <= 2*n-2:
                quad[2*i] = 1.0
            
            base = c.copy()
            
            # Ternary search for t in [0, t_max]
            t_max = max(2.0 * old_val, 1e-6)
            lo, hi = 0.0, t_max
            for _ in range(50):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                f1 = np.max(base + (m1 - old_val) * lin + (m1**2 - old_val**2) * quad)
                f2 = np.max(base + (m2 - old_val) * lin + (m2**2 - old_val**2) * quad)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            t_opt = (lo + hi) / 2
            # Check if it's an improvement
            c_test = base + (t_opt - old_val) * lin + (t_opt**2 - old_val**2) * quad
            if c_test.max() < current_max * (1 - 1e-12):
                a[i] = t_opt
                c = c_test
                current_max = c.max()
        
        # After sweep, normalize
        s = a.sum()
        if s > 0:
            a = a / s
    
    return a

"""
FFT-accelerated gradient optimization with log-sum-exp smoothing, beta-annealing,
multiplicative-weight mirror descent, and coordinate descent polishing for
minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses spectral gradient computation (O(n log n) per step) and multi-n exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, beta, n, N_fft, a_pad, w_buf):
    """Compute gradient of smoothed objective w.r.t. log-parameter z.
    Returns grad_z and the weights w (stored in w_buf for reuse)."""
    c_shifted = c - c.max()
    z2 = beta * c_shifted
    e = np.exp(z2)
    s_e = float(e.sum())
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e

    w_buf[:2 * n - 1] = w
    w_buf[2 * n - 1:] = 0.0
    a_pad[:n] = a
    a_pad[n:] = 0.0

    W = np.fft.rfft(w_buf)
    A = np.fft.rfft(a_pad)
    corr = np.fft.irfft(W * np.conj(A), N_fft)[:n]
    grad_a = 4.0 * n * corr

    g_mean = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - g_mean)
    return grad_z


def _adam_optimize(a_init, deadline, lr=0.005, betas=None):
    """Adam optimization in log-space with beta annealing."""
    n = len(a_init)
    N_fft = _next_pow2(2 * n)
    a_pad = np.zeros(N_fft)
    w_buf = np.zeros(N_fft)

    if betas is None:
        betas = [200, 500, 1000, 5000, 20000, 100000]

    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    time_per_beta = (deadline - time.monotonic()) / len(betas)

    for bi, beta in enumerate(betas):
        beta_deadline = time.monotonic() + time_per_beta
        iters = 0
        while time.monotonic() < beta_deadline:
            a = _softmax(z)
            c = _full_convolve(a, a)

            grad_z = _compute_grad_z(a, c, beta, n, N_fft, a_pad, w_buf)

            if not np.all(np.isfinite(grad_z)):
                break

            t_step += 1
            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            m_hat = m_buf / (1.0 - b1 ** t_step)
            v_hat = v_buf / (1.0 - b2 ** t_step)

            update = lr * m_hat / (np.sqrt(v_hat) + eps)
            un = np.linalg.norm(update)
            if un > 0.3:
                update = update * (0.3 / un)

            z = z - update
            z = z - z.mean()
            iters += 1

            if iters % 200 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    if iters % 2000 == 0:
                        print(f"    beta={beta:.0f}, iter={iters}, score={sc:.10f}, t={time.monotonic():.0f}")

        # Check at end of each beta phase
        a = _softmax(z)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score, z


def _mirror_descent(a_init, deadline, lr=0.01, betas=None):
    """Mirror descent (multiplicative weights) in log-space with beta annealing."""
    n = len(a_init)
    N_fft = _next_pow2(2 * n)
    a_pad = np.zeros(N_fft)
    w_buf = np.zeros(N_fft)

    if betas is None:
        betas = [200, 1000, 5000, 20000, 100000]

    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    time_per_beta = (deadline - time.monotonic()) / len(betas)
    current_lr = lr

    for beta in betas:
        beta_deadline = time.monotonic() + time_per_beta
        iters = 0
        while time.monotonic() < beta_deadline:
            a = _softmax(z)
            c = _full_convolve(a, a)

            grad_z = _compute_grad_z(a, c, beta, n, N_fft, a_pad, w_buf)

            if not np.all(np.isfinite(grad_z)):
                break

            # Multiplicative weights: z_i -= lr * grad_i / a_i (in probability space)
            # In log-space: z_i -= lr * (grad_a_i / a_i - mean)
            # Actually for mirror descent with KL: a_i *= exp(-eta * grad_a_i) / Z
            # In log-space: z_i -= eta * grad_a_i
            # But grad_a_i / a_i is the "natural gradient"
            step = grad_z / np.maximum(a, 1e-15)
            step = step - step.mean()

            # Adaptive lr
            step_norm = np.linalg.norm(step)
            if step_norm > 0:
                effective_lr = current_lr * 0.1 / step_norm
            else:
                effective_lr = current_lr

            z = z - effective_lr * step
            z = z - z.mean()
            iters += 1

            if iters % 500 == 0:
                sc = _score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if iters % 5000 == 0:
                current_lr *= 0.95

        a = _softmax(z)
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _coord_descent_polish(a_init, deadline, max_sweeps=20):
    """Coordinate descent: for each i, find optimal a[i] to minimize max(c)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score(a)

    c = _full_convolve(a, a)
    current_max = float(c.max())

    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Randomize order
        order = rng.permutation(n)

        for idx in range(n):
            i = order[idx]
            if time.monotonic() > deadline:
                break

            old_val = a[i]

            # Compute the delta structure
            k_range = np.arange(2 * n - 1)
            valid = (k_range >= i) & (k_range < i + n)
            is_diag = (k_range == 2 * i)

            # c_new[k] = c[k] + (t-old)*2*a[k-i] (if valid, not diag) + (t^2-old^2) (if diag)
            # Use bisection on subgradient for speed
            if old_val < 1e-15 and not valid.any():
                continue

            # Precompute linear coefficients
            lin = np.zeros(2 * n - 1)
            mask_lin = valid & ~is_diag
            if mask_lin.any():
                lin[mask_lin] = 2.0 * a[k_range[mask_lin] - i]
            if is_diag.any():
                # For k=2i: c changes by (t^2 - old^2), linearized as 2*old*(t-old) + (t-old)^2
                # For bisection, treat as: base + (t-old)*2*old + (t-old)^2
                # The "gradient" at t is 2*old + 2*(t-old) = 2*t
                pass

            # Bisection: find t in [0, 3*old_val + 1e-6] minimizing max(c_new)
            lo = 0.0
            hi = max(3.0 * old_val, old_val + 1e-6, 1e-8)

            for _ in range(30):
                mid = (lo + hi) / 2.0
                delta = mid - old_val
                c_test = c + delta * lin
                if is_diag.any():
                    c_test = c_test.copy()
                    c_test[is_diag] += delta * (2.0 * old_val) + delta * delta
                    # Actually: (t^2 - old^2) where t = old + delta = old^2 + 2*old*delta + delta^2 - old^2
                current_test = float(c_test.max())

                # Subgradient check: is the active constraint pushing t up or down?
                active_k = int(np.argmax(c_test))
                if active_k == 2 * i:
                    grad_at_t = 2.0 * mid
                elif 0 <= active_k - i < n:
                    grad_at_t = 2.0 * a[active_k - i]
                else:
                    grad_at_t = 0.0

                if grad_at_t < -1e-14:
                    lo = mid
                elif grad_at_t > 1e-14:
                    hi = mid
                else:
                    lo = mid
                    hi = mid
                    break

            t_opt = (lo + hi) / 2.0

            # Verify and apply
            delta = t_opt - old_val
            c_new = c + delta * lin
            if is_diag.any():
                c_new[is_diag] += delta * 2.0 * old_val + delta * delta

            if c_new.max() < current_max * (1.0 - 1e-13):
                a[i] = max(0.0, t_opt)
                c = c_new
                current_max = float(c.max())

        # Normalize after each sweep
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

        a_init_cd = best_seq / best_seq.sum()
        a_cd, sc_cd = _coord_descent_polish(a_init_cd, phase3_end, max_sweeps=50)
        update_best(a_cd)

    # === Phase 4: Second Adam run from CD result (1200s - 1400s) ===
    phase4_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 4: Second Adam run ===")

    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        a_init2 = best_seq / best_seq.sum()
        betas_p4 = [5000, 20000, 100000, 500000]
        a_opt2, sc_opt2, _ = _adam_optimize(a_init2, phase4_end, lr=0.003, betas=betas_p4)
        update_best(a_opt2)

    # === Phase 5: Multi-n exploration (1400s - 1650s) ===
    phase5_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 5: Multi-n exploration ===")

    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        target_ns = []
        if n_main <= 4096:
            target_ns = [2 * n_main, 4 * n_main]
        elif n_main <= 8192:
            target_ns = [4 * n_main] + ([2 * n_main] if 2 * n_main <= 32768 else [])
        else:
            target_ns = [16384] if n_main > 16384 else [2 * n_main]

        for target_n in target_ns:
            if time.monotonic() > phase5_end - 40:
                break
            if target_n > 65536:
                continue
            p_interp = _interp_profile(best_seq / best_seq.sum(), target_n)
            update_best(p_interp)

            remaining = phase5_end - time.monotonic() - 20
            if remaining > 60:
                betas_pn = [1000, 5000, 20000]
                a_r, _s, _ = _adam_optimize(p_interp, time.monotonic() + remaining * 0.7,
                                             lr=0.005, betas=betas_pn)
                update_best(a_r)

    # === Phase 6: Multi-restart (1650s - 1770s) ===
    phase6_end = min(deadline, t_start + 1770.0)
    print(f"\n=== Phase 6: Multi-restart ===")

    if time.monotonic() < phase6_end - 10 and best_seq is not None:
        a_norm = best_seq / best_seq.sum()
        sigmas = [0.02, 0.05, 0.1]
        n_restarts = 3
        per_restart = (phase6_end - time.monotonic() - 10) / max(1, n_restarts)

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            p_start = _perturb(a_norm, sigmas[i], rng)
            deadline_i = time.monotonic() + min(per_restart, phase6_end - time.monotonic() - 5)
            betas_pr = [5000, 50000]
            a_r, _s, _ = _adam_optimize(p_start, deadline_i, lr=0.005, betas=betas_pr)
            update_best(a_r)

    # === Final polish ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        a_final_init = best_seq / best_seq.sum()
        a_cd2, _ = _coord_descent_polish(a_final_init, deadline, max_sweeps=10)
        update_best(a_cd2)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
