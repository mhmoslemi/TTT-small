import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c.max()
    return float(2.0 * n * beta * np.log(s))


def _lse_grad(z, n, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _sumc2_obj(z, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.dot(c, c))


def _sumc2_grad(z, n):

    a_rev = a[::-1]
    wc = _full_convolve(c, a_rev)
    h = wc[n - 1:2 * n - 1]
    s2 = float(np.dot(c, c))
    return 4.0 * n * a * (h - s2)


def _adam_optimize(a_init, deadline, lr=0.01, beta_schedule=None, n_peaks=1):
    """Adam optimizer with log-sum-exp smoothed max and optional top-k subgradient mixing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    if beta_schedule is None:

    for beta_scale in beta_schedule:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * max_c_init
        min_beta = 1e-6 * max_c_init
        if beta < min_beta:
            beta = min_beta
        iters = 0
        phase_limit = max(2000, int((deadline - time.monotonic()) / max(1, len(beta_schedule)) / 2))

        while iters < phase_limit:
            if time.monotonic() > deadline:
                break
            iters += 1
            t_step += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            # LSE gradient
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            # Mix with top-k subgradient for extra descent pressure
            if n_peaks > 1 and iters % 3 == 0:
                top_k = np.argpartition(c, -min(n_peaks, len(c)))[-min(n_peaks, len(c)):]
                sub_grad_sum = np.zeros(n)
                for k_star in top_k:
                    ak = a[max(0, k_star - n + 1):k_star + 1]
                    # subgradient: 2*a_i * (a_{k-i} - c_{k*})
                    # a_{k-i} for i in [0,n-1] is a convolved
                    vals = _full_convolve(a[::-1], np.zeros(2*n-1))
                    # More efficient:
                    # For the subgradient at peak k*: grad_i = 2*a_i*(a_{k*-i} - c_{k*})
                    # a_{k*-i} is a shifted version
                    lo = max(0, k_star - n + 1)
                    hi = min(n - 1, k_star)
                    idx = np.arange(lo, hi + 1)
                    shifted = a[idx]
                    # Place into full array
                    full_shifted = np.zeros(n)
                    full_shifted[0:len(shifted)] = shifted
                    sub_grad_sum += 2.0 * a * (full_shifted - c[k_star])
                grad = 0.7 * grad + 0.3 * sub_grad_sum / len(top_k)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

                a_check = _softmax(z)
                sc = float(_full_convolve(a_check, a_check).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()

            if iters % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:

        best_a = a_final.copy()

def _lbfgs_optimize(a_init, deadline, n, obj_func, grad_func, maxiter=5000, ftol=1e-15, gtol=1e-12):
    """L-BFGS optimization in softmax space."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())

    def obj_fn(z):
        return obj_func(z, n)

    def grad_fn(z):
        return grad_func(z, n)

    try:
        res = minimize(obj_fn, z0, jac=grad_fn, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'maxfun': maxiter * 2,
                                'ftol': ftol, 'gtol': gtol, 'maxls': 50})
        if res.success or res.fun < best_score:
            a_res = _softmax(res.x)
            sc = float(_full_convolve(a_res, a_res).max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_res.copy()
    except Exception:
        pass

    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _structured_perturb(a, n_modes, rng):
    """Add sinusoidal perturbations to create diverse starting points."""
    z = np.log(np.maximum(a, 1e-15))
    x = np.linspace(0, 2 * np.pi * n_modes, len(z), endpoint=False)
    for m in range(1, n_modes + 1):
        amp = 0.05 * rng.random()
        phase = 2 * np.pi * rng.random()
        z = z + amp * np.sin(m * x + phase)
    z = z - z.mean()
    return _softmax(z)


def _endpoint_weighted(n, alpha, rng):
    """Create a sequence with endpoint weighting: a_i ∝ (1 + alpha * |2i/n - 1|^p)."""
    x = np.linspace(0, 1, n)
    w = 1.0 + alpha * np.abs(2.0 * x - 1.0)
    w = w + 0.01 * rng.standard_normal(n)
    w = np.maximum(w, 0.01)
    return w / w.sum()

            f_norm = a_new.copy()
            sc_check = _score(f_norm)
            update_best(f_norm)
            print(f"    Improved: M={sc_new:.10f}, score={sc_check:.10f}")
        else:
            print(f"    No improvement, trying next beta...")

    update_best(f_norm)
    print(f"Phase 1 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam fine-tuning (500s - 900s) ===
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: Adam fine-tuning ===")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_init = best_seq / best_seq.sum()
        remaining = phase2_end - time.monotonic() - 20
        a_best, sc_best = _adam_optimize(a_init, time.monotonic() + remaining,
                                         lr=0.008, n_peaks=3)
        update_best(a_best)

        # Second Adam run with different params
        if time.monotonic() < phase2_end - 60:
            remaining2 = phase2_end - time.monotonic() - 15
            a_best2, sc_best2 = _adam_optimize(a_best, time.monotonic() + remaining2,
                                               lr=0.005, n_peaks=5)
            update_best(a_best2)

    print(f"Phase 2 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 3: L-BFGS on sum(c^2) objective (900s - 1100s) ===
    phase3_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 3: L-BFGS on sum(c^2) ===")

    if best_seq is not None and time.monotonic() < phase3_end - 20:
        a_init = best_seq / best_seq.sum()
        a_new, sc_new = _lbfgs_optimize(a_init, time.monotonic() + (phase3_end - time.monotonic() - 15),
                                         n_main, _sumc2_obj, _sumc2_grad,
                                         maxiter=2000, ftol=1e-16, gtol=1e-14)
        # Check actual max score
        sc_actual = _score(a_new)
        if np.isfinite(sc_actual) and sc_actual < best_score:
            update_best(a_new)
        else:
            # The sum(c^2) minimizer might not directly improve max, but use it as init for Adam
            if time.monotonic() < phase3_end - 30:
                a_polished, _ = _adam_optimize(a_new, time.monotonic() + 40, lr=0.005, n_peaks=3)
                update_best(a_polished)

    print(f"Phase 3 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse initializations (1100s - 1450s) ===
    phase4_end = min(deadline, t_start + 1450.0)
    print(f"\n=== Phase 4: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 25) / max(1, n_restarts)

        restart_configs = [
            ('perturb_003', lambda: _perturb(a_best_norm, 0.03, rng)),
            ('perturb_008', lambda: _perturb(a_best_norm, 0.08, rng)),
            ('perturb_015', lambda: _perturb(a_best_norm, 0.15, rng)),
            ('struct_2', lambda: _structured_perturb(a_best_norm, 2, rng)),
            ('struct_5', lambda: _structured_perturb(a_best_norm, 5, rng)),
            ('struct_10', lambda: _structured_perturb(a_best_norm, 10, rng)),
        ]

        for name, gen_fn in restart_configs:
            if time.monotonic() > phase4_end - 25:
                break
            p_start = gen_fn()
            deadline_i = min(time.monotonic() + per_restart - 5, phase4_end - 20)
            a_r, sc_r = _adam_optimize(p_start, deadline_i, lr=0.008, n_peaks=3)
            update_best(a_r)

        # Try endpoint-weighted initializations
        if time.monotonic() < phase4_end - 60:
            for alpha in [0.5, 1.0, 2.0]:
                if time.monotonic() > phase4_end - 30:
                    break
                p_init = _endpoint_weighted(n_main, alpha, rng)
                # Blend with best
                p_init = 0.5 * a_best_norm + 0.5 * p_init
                p_init = p_init / p_init.sum()
                deadline_i = min(time.monotonic() + 50, phase4_end - 25)
                a_r, _ = _adam_optimize(p_init, deadline_i, lr=0.008, n_peaks=3)
                update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Different n exploration (1450s - 1680s) ===
    phase5_end = min(deadline, t_start + 1680.0)
    print(f"\n=== Phase 5: Different n exploration ===")

    if best_seq is not None and time.monotonic() < phase5_end - 40:
        target_ns = []
        bn = best_n
        for tn in [4096, 8192, 16384, 32768]:
            if tn != bn:
                target_ns.append(tn)

        # Sort by time budget feasibility
        target_ns.sort(key=lambda x: abs(x - bn))

        for target_n in target_ns[:3]:  # Try at most 3 different n values
            if time.monotonic() > phase5_end - 40:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            # Optimize at new n
            remaining = phase5_end - time.monotonic() - 30
            if remaining > 60:
                # First try L-BFGS
                if remaining > 120:
                    c_curr = _full_convolve(p_interp, p_interp)
                    M_curr = float(c_curr.max())
                    beta = 0.01 * M_curr

                    def lse_obj_n(z):
                        return _lse_obj(z, target_n, beta)

                    def lse_grad_n(z):
                        return _lse_grad(z, target_n, beta)

                    a_new, sc_new = _lbfgs_optimize(p_interp, time.monotonic() + min(60, remaining - 40),
                                                     target_n, lse_obj_n, lse_grad_n,
                                                     maxiter=2000, ftol=1e-15, gtol=1e-13)
                    update_best(a_new)

                # Then Adam
                if time.monotonic() < phase5_end - 20:
                    a_r, _ = _adam_optimize(a_new if 'a_new' in dir() else p_interp,
                                            time.monotonic() + (phase5_end - time.monotonic() - 10),
                                            lr=0.005, n_peaks=3)
                    update_best(a_r)
            print(f"  n={target_n}: done, current best={best_score:.10f}")

    print(f"Phase 5 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final polish (1680s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_final, _ = _adam_optimize(a_best_norm, time.monotonic() + remaining,
                                        lr=0.002, n_peaks=5)
            update_best(a_final)

    print(f"Phase 6 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
