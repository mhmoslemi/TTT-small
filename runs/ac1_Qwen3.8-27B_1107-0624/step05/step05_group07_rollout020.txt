# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score_norm(a):
    """Score with sum(a) normalized: 2*n*max(conv(a,a))"""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def score(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def objective(z, beta, n, top_k=None):
    """Log-sum-exp objective, optionally restricted to top-k lags."""
    a = softmax_fn(z)
    c = full_convolve(a, a)
    if top_k is not None and top_k < len(c):
        # Find top-k indices
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        cm = c_top.max()
        vals = (c_top - cm) / beta
        vals = np.clip(vals, -50, 50)
        lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    else:
        cm = c.max()
        vals = (c - cm) / beta
        vals = np.clip(vals, -50, 50)
        lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def gradient(z, beta, n, top_k=None):
    """Gradient of log-sum-exp objective."""
    a = softmax_fn(z)
    c = full_convolve(a, a)
    if top_k is not None and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_use = np.zeros_like(c)
        c_use[idx] = c[idx]
        cm = c[idx].max()
        vals = (c_use[idx] - cm) / beta
        vals = np.clip(vals, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        # w is over the top-k lags
        W = float(np.dot(w, c[idx]))
        # Build full w vector
        w_full = np.zeros_like(c)
        w_full[idx] = w
    else:
        cm = c.max()
        vals = (c - cm) / beta
        vals = np.clip(vals, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        w_full = w
    
    a_rev = a[::-1].copy()
    w_conv = full_convolve(w_full, a_rev)
    h = w_conv[n - 1:2 * n - 1]

Active-set spectral optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations: dynamic top-k active set, Adam with line search,
spectral flattening initialization, and exact subgradient polishing.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n, active_idx=None):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if active_idx is not None:
        c_use = c[active_idx]
    else:
        c_use = c
    cm = c_use.max()
    vals = np.clip((c_use - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _lse_grad(z, beta, n, active_idx=None):

    if active_idx is not None:

        cm = c_use.max()

        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c_use))
        # Build full weight vector (sparse)
        w_full = np.zeros(2 * n - 1)
        w_full[active_idx] = w
    else:

        vals = np.clip((c - cm) / beta, -50, 50)

        w = e / e.sum()

    """Get indices of top-k values in c."""
    if k >= len(c):
        return np.arange(len(c))
    return np.argpartition(c, -k)[-k:]

def _adam_optimize(z0, n, beta, deadline, top_k_adapt=True, 
                   lr=0.1, max_iter=50000, active_window=100):
    """Adam optimizer for the LSE objective with dynamic active set."""
    m = len(z0)
    best_a = _softmax(z0).copy()
    best_score = _score_norm(best_a)
    
    z = z0.copy()
    m_hat = np.zeros(m)
    v_hat = np.zeros(m)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Initial active set
    a_init = _softmax(z0)
    c_init = _full_convolve(a_init, a_init)
    active_idx = _top_k_indices(c_init, min(active_window, 2*n-1))
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Compute gradient with current active set
        grad = _lse_grad(z, beta, n, active_idx)
        
        # Adam update
        m_hat = b1 * m_hat + (1 - b1) * grad
        v_hat = b2 * v_hat + (1 - b2) * (grad * grad)
        m_hat_ub = m_hat / (1 - b1**t)
        v_hat_ub = v_hat / (1 - b2**t)
        
        z_new = z - lr * m_hat_ub / (np.sqrt(v_hat_ub) + eps)
        z_new = np.clip(z_new, -100, 100)
        z_new = z_new - z_new.mean()
        
        # Evaluate
        a_new = _softmax(z_new)
        sc = _score_norm(a_new)
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_new.copy()
        
        # Adapt active set periodically
        if top_k_adapt and t % 500 == 0:
            a_cur = _softmax(z)
            c_cur = _full_convolve(a_cur, a_cur)
            active_idx = _top_k_indices(c_cur, min(active_window, 2*n-1))
        
        # Decay learning rate
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.85)
    
    return best_a, best_score

def _subgradient_polish(a_init, n, deadline, max_iter=10000, lr=0.01):
    """Exact subgradient method on the active set (top lags)."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(best_a)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = c.max()
        
        # Find active set (lags within 1% of max)
        threshold = cm * 0.999
        active = np.where(c >= threshold)[0]
        
        # Subgradient = average of subgradients at active lags
        # For each active lag m, subgrad[j] = 2*a[m-j] (if 0<=m-j<n)
        # Weighted average: grad[j] = sum_m w_m * 2*a[m-j]
        W = float(c[active].mean())  # average max value
        # h[j] = sum_m (1/|active|) * a[m-j]
        h = np.zeros(n)
        for m in active:
            # a[m-j] valid for j in [max(0,m-n+1), min(n-1,m)]
            j_lo = max(0, m - n + 1)
            j_hi = min(n - 1, m)
            h[j_lo:j_hi+1] += a[m - j_hi:m - j_lo + 1][::-1] / len(active)
        
        grad_z = 4.0 * n * a * (h - W)
        
        # SGD with momentum
        z = z - lr * grad_z
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        a_new = _softmax(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # Adapt learning rate
        if it % 1000 == 0:
            lr = max(0.0001, lr * 0.9)
    
    return best_a, best_score

"""
Active-set spectral optimization for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses dynamic top-k active set, Adam with line search, spectral flattening
initialization (prospect method), multi-n exploration, and exact subgradient
polishing for final refinement.
"""
import numpy as np
import time
import math


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n, active_idx=None):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if active_idx is not None:
        c_use = c[active_idx]
    else:
        c_use = c
    cm = c_use.max()
    vals = np.clip((c_use - cm) / beta, -60, 60)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n, active_idx=None):
    a = _softmax(z)
    c = _full_convolve(a, a)
    full_len = 2 * n - 1
    if active_idx is not None:
        c_use = c[active_idx]
        cm = c_use.max()
        vals = np.clip((c_use - cm) / beta, -60, 60)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c_use))
        w_full = np.zeros(full_len)
        w_full[active_idx] = w
    else:
        cm = c.max()
        vals = np.clip((c - cm) / beta, -60, 60)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        w_full = w
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w_full, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _top_k_indices(c, k):
    if k >= len(c):
        return np.arange(len(c))
    return np.argpartition(c, -k)[-k:]


def _spectral_init(n, rng, iters=300, blend=0.3):
    a = rng.random(n) + 0.05
    a = a / a.sum()
    for it in range(iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        dc = mag[0]
        if len(mag) > 1:
            non_dc_mean = mag[1:].mean()
        else:
            non_dc_mean = 0.0
        target_mag = np.full_like(mag, non_dc_mean)
        target_mag[0] = dc
        new_mag = (1.0 - blend) * mag + blend * target_mag
        new_mag = np.maximum(new_mag, 1e-15)
        A_new = A / (mag + 1e-30) * new_mag
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 1e-15)
        a = a_new / a_new.sum()
    return a


def _adam_with_active_set(z0, n, beta, deadline, initial_top_k=None,
                          min_top_k=64, adapt_every=300, lr=0.15,
                          max_iter=100000):
    m = len(z0)
    best_a = _softmax(z0).copy()
    best_score = _score_norm(best_a)
    z = z0.copy()
    m_hat = np.zeros(m)
    v_hat = np.zeros(m)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    a_cur = _softmax(z0)
    c_cur = _full_convolve(a_cur, a_cur)
    full_len = 2 * n - 1
    if initial_top_k is None:
        initial_top_k = full_len
    cur_k = min(initial_top_k, full_len)
    active_idx = _top_k_indices(c_cur, cur_k)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        grad = _lse_grad(z, beta, n, active_idx)
        if not np.all(np.isfinite(grad)):
            break
        m_hat = b1 * m_hat + (1.0 - b1) * grad
        v_hat = b2 * v_hat + (1.0 - b2) * (grad * grad)
        m_hat_ub = m_hat / (1.0 - b1 ** t)
        v_hat_ub = v_hat / (1.0 - b2 ** t)
        z_new = z - lr * m_hat_ub / (np.sqrt(v_hat_ub) + eps)
        z_new = np.clip(z_new, -100, 100)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_new.copy()
        else:
            z = z_new.copy()

        if t % 2000 == 0:
            lr = max(0.005, lr * 0.88)

        if adapt_every > 0 and t % adapt_every == 0:
            a_check = _softmax(z)
            c_check = _full_convolve(a_check, a_check)
            cur_k = max(min_top_k, cur_k // 2)
            active_idx = _top_k_indices(c_check, cur_k)

    return best_a, best_score


def _beta_continuation_adam(a_init, deadline, n_max_betas=20):
    n = len(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if max_c0 < 1e-10:
        return best_a, best_score

    fracs = [0.5, 0.3, 0.2, 0.15, 0.1, 0.08, 0.06, 0.05, 0.04, 0.03,
             0.025, 0.02, 0.015, 0.012, 0.01, 0.008, 0.006, 0.005,
             0.004, 0.003, 0.0025, 0.002, 0.0015, 0.001, 0.0008]
    fracs = fracs[:n_max_betas]

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    n_betas = len(fracs)
    for idx, frac in enumerate(fracs):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 120.0)
        beta = frac * max_c0
        level_dl = time.monotonic() + time_per

        a_cur, sc_cur = _adam_with_active_set(
            z, n, beta, level_dl,
            initial_top_k=2 * n,
            min_top_k=32,
            adapt_every=200,
            lr=0.12,
            max_iter=int(time_per * 80)
        )
        if np.isfinite(sc_cur) and sc_cur < best_score:
            best_score = sc_cur
            best_a = a_cur.copy()
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()

    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_iter=20000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(best_a)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = c.max()
        threshold = cm * 0.998
        active = np.where(c >= threshold)[0]
        W = float(c[active].mean())
        w_full = np.zeros(2 * n - 1)
        w_full[active] = 1.0 / len(active)
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w_full, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        m_hat = b1 * m_hat + (1.0 - b1) * grad
        v_hat = b2 * v_hat + (1.0 - b2) * (grad * grad)
        m_hat_ub = m_hat / (1.0 - b1 ** t)
        v_hat_ub = v_hat / (1.0 - b2 ** t)
        z_new = z - lr * m_hat_ub / (np.sqrt(v_hat_ub) + eps)
        z_new = np.clip(z_new, -100, 100)
        z_new = z_new - z_new.mean()

        a_new = _softmax(z_new)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        z = z_new

        if it % 3000 == 0:
            lr = max(0.001, lr * 0.85)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_refine(a_init, beta, deadline, maxiter=5000):
    from scipy.optimize import minimize
    n = len(a_init)

    def obj(z):
        return _lse_obj(z, beta, n)

    def grad(z):
        return _lse_grad(z, beta, n)

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    try:
        result = minimize(
            obj, z, method='L-BFGS-B', jac=grad,
            options={'maxiter': maxiter, 'ftol': 1e-18,
                     'gtol': 1e-12, 'maxls': 40, 'maxfun': maxiter * 4}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax(result.x)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score
