"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, exact top-k active set subgradient,
variance-reduction rebalancing, and adaptive restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    thr = c_max - 1e-4 * c_max

    if active.sum() > k:
        idx = np.argsort(c)[::-1][:k]
        active = np.zeros_like(c, dtype=bool)
        active[idx] = True
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _grad_variance(a, c, n, k=5):
    """Gradient of variance of top-k conv values (to equalize them)."""
    c_max = c.max()
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter=5000, ftol=1e-15, gtol=1e-12):
    """Run L-BFGS-B for one beta phase."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': ftol,
                'gtol': gtol,
                'maxcor': 50,
                'maxls': 200,
                'disp': False,
            }
        )
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _subgrad_refined(a_init, deadline, n_steps=10000, lr_init=0.05, k_active=2):
    """Subgradient with exact top-k active set and adaptive step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _grad_topk(a, c, n, k=k_active)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        # Polyak-style step size
        step = lr / (1.0 + 0.01 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.1)
            else:
                lr *= 0.85

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _variance_rebalance(a_init, deadline, n_steps=500, lr_init=0.02, k=5):
    """Reduce variance of top-k convolution values to flatten the peak."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _grad_variance(a, c, n, k=k)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break

        step = lr / (1.0 + 0.02 * i)
        z = z - step * grad
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 50 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb_gaussian(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_smooth(a, rng, n_modes=3, max_amp=0.02):
    """Smooth perturbation using low-frequency Fourier modes."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        amp = max_amp * rng.uniform(0.3, 1.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_local(a, rng, n_perturb=50):
    """Perturb a random contiguous block."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    start = rng.integers(0, n - n_perturb)
    block = 0.03 * rng.standard_normal(n_perturb)
    # Taper the block
    taper = np.sin(np.pi * np.arange(n_perturb) / n_perturb) ** 0.5
    block *= taper
    z[start:start + n_perturb] += block
    z = z - z.mean()
    return _softmax(z)


def _optimize_schedule(a_start, deadline, min_remaining=60.0, aggressive=False):
    """Run a full beta-continuation L-BFGS schedule."""
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    if aggressive:
        betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                                  0.0005, 0.00025, 0.00012, 0.00006, 0.00003])
    else:
        betas = max_c * np.array([0.1, 0.05, 0.025, 0.01, 0.005, 0.0025,
                                  0.001, 0.0005, 0.00025])

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        est_iter_time = 0.004 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_iter_time * 10))))

        a_new, success, niter = _lbfgs_run(current_a, beta, n, max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-13:
            current_a = a_new
            current_sc = sc
            print(f"    beta_phase {i}: score={sc:.10f} ({niter} iters)")

    return current_a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS beta-continuation + top-k subgradient
    + variance rebalancing + adaptive restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"dt={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = np.ones(n_main) / n_main
        print(f"No parent, uniform n={n_main}")
        update_best(a0, _score(a0), "init_uniform")

    # Phase 1: Aggressive L-BFGS
    print(f"\nPhase 1: Aggressive L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    a1 = _optimize_schedule(a0, deadline, min_remaining=400.0, aggressive=True)
    sc1 = _score(a1)
    update_best(a1, sc1, "lbfgs_aggr")

    # Phase 2: Top-k subgradient polish (k=1, the true argmax)
    print(f"\nPhase 2: Top-1 subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        dl = min(deadline - 300.0, time.monotonic() + 180.0)
        a2, sc2 = _subgrad_refined(best_seq, dl, n_steps=15000, lr_init=0.03, k_active=1)
        update_best(a2, sc2, "subgrad_k1")

    # Phase 3: Variance rebalancing
    print(f"\nPhase 3: Variance rebalance (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        dl = min(deadline - 180.0, time.monotonic() + 90.0)
        a3, sc3 = _variance_rebalance(best_seq, dl, n_steps=300, lr_init=0.01, k=4)
        update_best(a3, sc3, "var_rebal_k4")

    # Phase 4: Fine L-BFGS after rebalancing
    print(f"\nPhase 4: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 100:
        a4 = _optimize_schedule(best_seq, deadline, min_remaining=80.0, aggressive=True)
        sc4 = _score(a4)
        update_best(a4, sc4, "lbfgs_fine")

    # Phase 5: Top-2 subgradient (balanced between peak and runner-up)
    print(f"\nPhase 5: Top-2 subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        dl = min(deadline - 50.0, time.monotonic() + 60.0)
        a5, sc5 = _subgrad_refined(best_seq, dl, n_steps=8000, lr_init=0.02, k_active=2)
        update_best(a5, sc5, "subgrad_k2")

    # Phase 6: Restart with smooth perturbation + quick optimize
    print(f"\nPhase 6: Smooth-perturb restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        for i in range(3):
            if time.monotonic() > deadline - 30:
                break
            if i == 0:
                p_start = _perturb_smooth(best_seq, rng, n_modes=2, max_amp=0.01)
            elif i == 1:
                p_start = _perturb_smooth(best_seq, rng, n_modes=4, max_amp=0.015)
            else:
                p_start = _perturb_local(best_seq, rng, n_perturb=100)

            sc_s = _score(p_start)
            # Quick optimize
            a_r = _optimize_schedule(p_start, deadline, min_remaining=20.0, aggressive=True)
            sc_r = _score(a_r)
            update_best(a_r, sc_r, f"restart_smooth_{i}")

    # Phase 7: Final ultra-fine polish
    print(f"\nPhase 7: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very small step subgradient on the best
        n = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        lr = 0.005
        for i in range(3000):
            if time.monotonic() > deadline - 3:
                break
            a = _softmax(z)
            c = _full_convolve(a, a)
            grad = _grad_topk(a, c, n, k=1)
            gn = np.linalg.norm(grad)
            if gn < 1e-17:
                break
            step = lr / (1.0 + 0.005 * i)
            z = z - step * grad
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()
            if i % 200 == 0:
                a_chk = _softmax(z)
                sc_chk = _score(a_chk)
                if sc_chk < best_score:
                    best_seq = a_chk.copy()
                    best_score = sc_chk
                    print(f"    final_polish iter {i}: {best_score:.10f}")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
