Optimized multi-scale minimization of 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp continuation with L-BFGS, subgradient refinement,

import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def score_norm_fn(a):
    """Assumes sum(a) = 1, returns 2*max(conv)"""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def logsumexp_obj(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def logsumexp_grad(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # grad of beta*log(sum(exp((c-cm)/beta))) with respect to z
    # dc/dz_j = 2 * (sum_k w[k] * a[k-j]) - (sum_j 2*a[j]*(sum_k w[k]*a[k-j]))
    # The first term: for each k, a[k-j] for valid k-j
    # This is a cross-correlation of w with a
    a_rev = a[::-1]
    w_conv = full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]  # h[j] = sum_k w[k] * a[k-j]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

def make_ramp(n, slope=1.5, start=0.0):
    t = np.linspace(0, 1, n)
    a = start + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_sine(n):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * (t + 0.5/n))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_gaussian(n, sigma_frac=0.3):
    t = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-0.5 * ((t - center) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_cosine_window(n):
    t = np.linspace(0, 1, n)
    a = 0.54 + 0.46 * np.cos(2 * np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_quadratic(n):
    t = np.linspace(0, 1, n)
    a = t ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_blackman(n):
    t = np.linspace(0, 1, n)
    a = 0.42 - 0.5 * np.cos(2*np.pi*t) + 0.08 * np.cos(4*np.pi*t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def get_initial_profiles(n):
    """Generate diverse initial profiles."""
    profiles = []
    profiles.append(('ramp_1', make_ramp(n, 1.0)))
    profiles.append(('ramp_2', make_ramp(n, 2.0)))
    profiles.append(('ramp_3', make_ramp(n, 3.0)))
    profiles.append(('ramp_05', make_ramp(n, 0.5)))
    profiles.append(('sine', make_sine(n)))
    profiles.append(('gauss_02', make_gaussian(n, 0.2)))
    profiles.append(('gauss_03', make_gaussian(n, 0.3)))
    profiles.append(('gauss_04', make_gaussian(n, 0.4)))
    profiles.append(('cosine', make_cosine_window(n)))
    profiles.append(('quad', make_quadratic(n)))
    profiles.append(('blackman', make_blackman(n)))
    # Reversed ramps
    profiles.append(('ramp_rev_1', make_ramp(n, 1.0)[::-1].copy()))
    profiles.append(('ramp_rev_2', make_ramp(n, 2.0)[::-1].copy()))
    return profiles

def run_lbfgs_continuation(a_init, deadline, rng, n_betas=15):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    
    c0 = full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_start = 0.5 * max_c0
    beta_end = max(1e-8 * max_c0, 1e-15)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 200.0)
        level_deadline = time.monotonic() + time_budget
        
        if beta > 0.1 * max_c0:
            maxiter = 300
        elif beta > 0.01 * max_c0:
            maxiter = 1000
        else:
            maxiter = 5000
        
        maxfun = maxiter * 5
        
        try:
            result = minimize(
                logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -100, 100)
                z_new = z_new - z_new.mean()
                a_new = softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def run_subgradient(a_init, deadline, rng, lr_init=0.01, n_iter=500000):
    """Subgradient method for the max of convolution."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    
    # Adaptive parameters
    lr = lr_init
    decay = 0.9999
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        
        a = softmax_fn(z)
        c = full_convolve(a, a)
        cm = float(c.max())
        
        # Find active set (top peaks)
        threshold = cm - 0.1 * cm  # within 10% of max
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute subgradient
        # For each active k, (P_k a)_i = a[k-i] if 0 <= k-i < n
        grad = np.zeros(n)
        for k in active:
            # (P_k a)_i = a[k-i] for valid i
            valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
            if len(valid_i) > 0:
                grad[valid_i] += a[k - valid_i]
        
        grad = 2.0 * grad
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum
        v = momentum * v + (1 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-15:
            break
        v = v / v_norm
        
        # Update
        z = z - lr * v
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Periodically check score
        if it % 100 == 0:
            a_check = softmax_fn(z)
            sc = score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        
        # Decay learning rate
        lr = max(0.001, lr * decay)
        
        # Random kick if stuck
        if it % 50000 == 0 and it > 0:
            z = z + 0.05 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score

def run_adam_gd(a_init, deadline, rng, lr_init=0.1, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    c = full_convolve(a_init, a_init)
    beta = 0.1 * float(c.max())
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = softmax_fn(z)
        c = full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta: decrease over time
        if it % 500 == 0:
            beta = max(1e-8 * cm, beta * 0.95)
        
        if cm < 1e-30:
            break
        
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        
        a_rev = a[::-1]
        w_conv = full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = softmax_fn(z)
        
        sc = score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay learning rate
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)
        
        # Kick if stuck
        if t % 20000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score

def peak_surgery(a_init, deadline, rng, max_rounds=500):
    """Targeted reduction of peak-contributing elements."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = softmax_fn(z)
        c = full_convolve(a_cur, a_cur)
        cm = float(c.max())
        
        # Find top peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices[:3]:
            # Contributions: a[i] * a[pk-i] for valid i
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:10]
            for idx_pos in order[:5]:
                idx = valid_i[idx_pos]
                for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = softmax_fn(z_trial)
                    sc = score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
