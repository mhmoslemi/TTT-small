"""
Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Spectral projection: iteratively shape DFT magnitude to flatten convolution spectrum
2. Aggressive L-BFGS log-sumexp continuation with 25+ fine levels

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _conv_and_max(a):
    c = _full_convolve(a, a)
    return c, float(c.max())


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):
    t = np.linspace(0.0, 1.0, n)
    center1 = 0.5 - separation / 2.0
    center2 = 0.5 + separation / 2.0
    a = np.exp(-(t - center1) ** 2 / (2 * width ** 2)) + np.exp(-(t - center2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bspline_profile(n, order=4):
    t = np.linspace(0.0, 1.0, n)
    u = t * 0.9 + 0.05
    u = np.clip(u, 0.01, 0.99)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _spectral_shape(a, strength=0.3, gamma=0.5):
    """Shape the DFT magnitude to flatten the convolution spectrum.
    
    Reduces the L4 norm of AC spectral components by compressing magnitude peaks.
    """
    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    
    # Keep DC component
    A_dc = A[0].copy()
    
    # Compute magnitude of AC components
    if len(A) > 1:
        mags = np.abs(A[1:])
        if mags.max() > 1e-300:
            # Compress magnitude: reduce peaks
            # target = geometric mean (robust)
            log_mags = np.log(np.maximum(mags, 1e-30))
            target_log = np.mean(log_mags)
            target_mag = np.exp(target_log)
            
            # Blend: new_mag = |A|^alpha * target^(1-alpha)
            new_mags = (np.maximum(mags, 1e-30) ** gamma) * (target_mag ** (1.0 - gamma))
            
            # Apply strength: interpolate between original and shaped
            new_mags = (1.0 - strength) * mags + strength * new_mags
            
            # Reconstruct with original phases
            phases = A[1:] / np.maximum(mags, 1e-30)
            A[1:] = phases * new_mags
    
    A[0] = A_dc
    
    # IFFT and extract
    a_new = np.fft.irfft(A, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s * a.sum()
    else:
        a_new = a.copy()
    
    return a_new


def _spectral_iterate(a, deadline, max_iters=200, strength=0.2, gamma=0.6):
    """Iterative spectral shaping."""
    a_curr = a.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a_curr)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        a_new = _spectral_shape(a_curr, strength=strength, gamma=gamma)
        sc = _score_norm_fn(a_new)
        
        if sc < best_sc * (1.0 + 1e-10):
            best_sc = sc
            best_a = a_new.copy()
            a_curr = a_new.copy()
            if it % 20 == 0:
                print(f"    Spectral iter {it}: score={sc:.10f}")
    
    return best_a, best_sc


def _lbfgs_logsumexp(a_init, deadline, beta_scale=1.0, min_beta_frac=0.00005,
                     n_levels=None, per_level_max=30.0):
    """L-BFGS with logsumexp continuation - aggressive schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if n_levels is None:
        n_levels = 20
    
    # Create fine geometric schedule
    ratio = (min_beta_frac / 0.1) ** (1.0 / (n_levels - 1))
    beta_starts = [0.1 * (ratio ** i) for i in range(n_levels)]
    # Ensure last is exactly min_beta_frac
    beta_starts[-1] = min_beta_frac
    betas = [b * max_c0 * beta_scale for b in beta_starts]
    
    # Filter out betas that are too similar
    filtered_betas = [betas[0]]
    for b in betas[1:]:
        if b < filtered_betas[-1] * 0.7:
            filtered_betas.append(b)
    betas = filtered_betas
    
    if len(betas) < 3:
        betas = [0.1 * max_c0 * beta_scale, 0.01 * max_c0 * beta_scale,
                 min_beta_frac * max_c0 * beta_scale]
    
    total_time_available = max(1.0, deadline - time.monotonic())
    per_level = max(2.0, total_time_available / len(betas))
    
    for beta_idx, beta in enumerate(betas):
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())

            s_ = e_.sum()

                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_

            a_rev = a_[::-1]

            g = 4.0 * n * a_ * (h - W)

                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,

                    best_a = a_new.copy()
                    z = z_new.copy()
                    if beta_idx % 5 == 0:

def _adam_peak_targeted(a_init, deadline, lr=0.015, k_top=5, lr_decay=0.98):
    """Adam with peak-targeted gradient and adaptive learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    max_iters = 80000
    current_lr = lr
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a_curr = _softmax_fn(z)
        
        # Compute top-K targeted gradient
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argpartition(c, -k_actual)[-k_actual:]
        
        # Weight: use softmax of top-K values (sharper)
        top_vals = c[top_idx]
        top_max = top_vals.max()
        w_top = np.exp((top_vals - top_max) / (top_max * 0.1 + 1e-10))
        w_top = w_top / w_top.sum()
        
        # Full weight vector (mostly zeros, with weight at top positions)
        w = np.zeros(len(c))
        for idx_i, idx in enumerate(top_idx):
            w[int(idx)] = w_top[idx_i]
        
        # Compute gradient
        W = float(np.dot(w, c))
        a_rev = a_curr[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a_curr * (h - W)
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - current_lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Check score periodically
        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        # Decay learning rate
        if t % 1000 == 0:
            current_lr = max(0.001, current_lr * lr_decay)
    
    return best_a, best_score


def _sim_anneal_local_search(a_init, deadline, n_trials=50000, T_init=0.01, T_min=0.0001):
    """Simulated annealing local search targeting convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    c = _full_convolve(a, a)
    current_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Temperature schedule
        frac = float(it) / float(n_trials)
        T = T_init * (T_min / T_init) ** frac
        
        c = _full_convolve(a, a)
        max_lag = int(np.argmax(c))
        
        # Choose move type
        mv_type = it % 4
        
        if mv_type == 0:
            # Reduce top contributor at max lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                idx = pairs[int(np.argmax(products))]
                
                a_trial = a.copy()
                frac_move = 0.1 + 0.3 * T * 100
                frac_move = min(0.4, frac_move)
                delta = a_trial[idx] * frac_move
                a_trial[idx] = max(0.0, a_trial[idx] - delta)
                
                # Add to a low-contribution position
                contrib = np.zeros(n)
                top_lags = np.argpartition(c, -5)[-5:]
                for kl in top_lags:
                    kl = int(kl)
                    lo = max(0, kl - (n - 1))
                    hi = min(n - 1, kl)
                    contrib[lo:hi + 1] += a[kl - lo:kl - lo + (hi - lo + 1)][::-1]
                
                safe = np.argsort(contrib)
                n_safe = max(1, n // 10)
                add_idx = int(safe[np.random.randint(n_safe)])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc_trial = _score_norm_fn(a_trial)
                    
                    if sc_trial < current_sc or np.random.random() < np.exp(-(sc_trial - current_sc) / max(T, 1e-10)):
                        a = a_trial.copy()
                        current_sc = sc_trial
                        if sc_trial < best_sc:
                            best_sc = sc_trial
                            best_a = a.copy()
        
        elif mv_type == 1:
            # Block perturbation
            block_size = max(2, int(np.random.randint(3, min(20, n // 8))))
            start = int(np.random.randint(0, n - block_size))
            a_trial = a.copy()
            mult = 1.0 + 0.2 * T * 50 * np.random.randn(block_size)
            mult = np.clip(mult, 0.5, 1.5)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc_trial = _score_norm_fn(a_trial)
            
            if sc_trial < current_sc or np.random.random() < np.exp(-(sc_trial - current_sc) / max(T, 1e-10)):
                a = a_trial.copy()
                current_sc = sc_trial
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a.copy()
        
        elif mv_type == 2:
            # Pair swap
            i1 = int(np.random.randint(n))
            i2 = int(np.random.randint(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = 0.1 + 0.2 * T * 30
            frac = min(0.3, frac)
            amt = min(a_trial[i1], a_trial[i2]) * frac
            a_trial[i1] -= amt
            a_trial[i2] += amt
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc_trial = _score_norm_fn(a_trial)
            
            if sc_trial < current_sc or np.random.random() < np.exp(-(sc_trial - current_sc) / max(T, 1e-10)):
                a = a_trial.copy()
                current_sc = sc_trial
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a.copy()
        
        else:
            # Random small perturbation
            idx = int(np.random.randint(n))
            a_trial = a.copy()
            delta = a_trial[idx] * 0.1 * np.random.randn()
            a_trial[idx] = max(0.0, a_trial[idx] + delta)
            # Compensate elsewhere
            idx2 = int(np.random.randint(n))
            if idx2 != idx:
                a_trial[idx2] = max(0.0, a_trial[idx2] - delta)
            a_trial = np.maximum(a_trial, 1e-15)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc_trial = _score_norm_fn(a_trial)
                
                if sc_trial < current_sc or np.random.random() < np.exp(-(sc_trial - current_sc) / max(T, 1e-10)):
                    a = a_trial.copy()
                    current_sc = sc_trial
                    if sc_trial < best_sc:
                        best_sc = sc_trial
                        best_a = a.copy()
    
    return best_a, best_sc


def _directed_peak_reduction(a_init, deadline, n_rounds=500):
    """Directly reduce the top convolution peaks by adjusting contributing elements."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        max_val = c.max()
        max_lag = int(np.argmax(c))
        
        # Find top 3 contributing pairs at the max lag
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        if i_min > i_max:
            continue
        
        pairs = np.arange(i_min, i_max + 1)
        partners = max_lag - pairs
        products = a[pairs] * a[partners]
        
        # Get top 3 pairs
        top3 = np.argsort(products)[-3:]
        
        # Compute contribution score for each position (across all top lags)
        top_lags = np.argpartition(c, -8)[-8:]
        contrib = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            lo = max(0, kl - (n - 1))
            hi = min(n - 1, kl)
            for j in range(lo, hi + 1):
                partner = kl - j
                if 0 <= partner < n:
                    contrib[j] += a[partner]
        
        # For each top pair, try to reduce it
        for t_idx in top3:
            idx = int(pairs[t_idx])
            partner_idx = int(partners[t_idx])
            
            # Determine which to reduce (the one with higher contribution to other peaks)
            if contrib[idx] >= contrib[partner_idx]:
                reduce_idx = idx
            else:
                reduce_idx = partner_idx
            
            a_trial = a.copy()
            # Reduce the element
            reduction = a_trial[reduce_idx] * 0.15
            a_trial[reduce_idx] -= reduction
            a_trial[reduce_idx] = max(0.0, a_trial[reduce_idx])
            
            # Add to lowest contribution position
            safe_positions = np.argsort(contrib)
            # Pick from bottom 20%
            n_safe = max(1, n // 5)
            add_candidates = safe_positions[:n_safe]
            add_idx = int(add_candidates[np.random.randint(len(add_candidates))])
            a_trial[add_idx] += reduction
            
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    c = _full_convolve(a, a)  # update
                    # Recompute contrib
                    top_lags = np.argpartition(c, -8)[-8:]
                    contrib = np.zeros(n)
                    for kl in top_lags:
                        kl = int(kl)
                        lo = max(0, kl - (n - 1))
                        hi = min(n - 1, kl)
                        for j in range(lo, hi + 1):
                            partner = kl - j
                            if 0 <= partner < n:
                                contrib[j] += a[partner]
    
    return best_a, best_sc


def _coarse_to_fine_optimize(a_coarse, n_fine, deadline):
    """Optimize at coarse resolution, then refine at fine resolution."""
    a_fine = _interp_profile(a_coarse, n_fine)
    sc = _score_norm_fn(a_fine)
    
    # Polish at fine resolution
    remaining = max(5.0, deadline - time.monotonic())
    alloc = min(remaining * 0.6, 120.0)
    dl = time.monotonic() + alloc
    a_best, sc_best = _lbfgs_logsumexp(a_fine, dl, beta_scale=0.5)
    
    if sc_best < sc:
        a_fine = a_best
        sc = sc_best
    
    return a_fine, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining spectral shaping, L-BFGS continuation,
    Adam peak-targeting, simulated annealing, and directed peak reduction.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    np.random.seed(int(seed) % (2**31 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}, t={elapsed:.0f}s) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_work = 4096
        parent = _make_triangular_weight(n_work)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using triangular n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    print(f"Working with n={n_work}")
    
    # --- Phase 0: Quick diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    profiles = [
        _make_bimodal_profile(n_work, 0.3, 0.12),
        _make_bimodal_profile(n_work, 0.5, 0.15),
        _make_bimodal_profile(n_work, 0.4, 0.1),
        _make_w_profile(n_work, 2, 0.3),
        _make_w_profile(n_work, 3, 0.25),
        _make_w_profile(n_work, 4, 0.3),
        _make_power_profile(n_work, 1.5),
        _make_power_profile(n_work, 2.0),
        _make_power_profile(n_work, 3.0),
        _make_bspline_profile(n_work, 3),
        _make_bspline_profile(n_work, 4),
        _make_bspline_profile(n_work, 5),
    ]
    
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        update_best(p, sc, f"prof{idx}")
    
    # Also try spectral shaping on parent
    if time.monotonic() < phase0_end - 5:
        a_spec, sc_spec = _spectral_iterate(parent, phase0_end - 5, max_iters=50, strength=0.3, gamma=0.5)
        update_best(a_spec, sc_spec, "spectral_init")
    
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # --- Phase 1: Spectral shaping (60s) ---
    phase1_end = min(deadline, t_start + 90.0)
    if best_seq is not None:
        print("Phase 1: Spectral shaping")
        a_spec, sc_spec = _spectral_iterate(best_seq, phase1_end, max_iters=100, strength=0.25, gamma=0.6)
        update_best(a_spec, sc_spec, "spectral")
        
        # Try different strength
        if time.monotonic() < phase1_end - 15:
            a_spec2, sc_spec2 = _spectral_iterate(best_seq, phase1_end, max_iters=50, strength=0.4, gamma=0.4)
            update_best(a_spec2, sc_spec2, "spectral_strong")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Main L-BFGS continuation (700s) ---
    phase2_end = min(deadline, t_start + 790.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: L-BFGS continuation (main)")
        dl = phase2_end - 5.0
        a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, beta_scale=1.0,
                                           min_beta_frac=0.00003, n_levels=25,
                                           per_level_max=30.0)
        update_best(a_best, sc_best, "lbfgs_main")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Second L-BFGS pass with different params (200s) ---
    phase3_end = min(deadline, t_start + 990.0)

        print("Phase 3: L-BFGS second pass")
        dl = phase3_end - 5.0
        a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, beta_scale=0.4,
                                           min_beta_frac=0.00001, n_levels=15)
        update_best(a_best, sc_best, "lbfgs_fine")
        
        # Spectral + L-BFGS combo
        if time.monotonic() < phase3_end - 30:
            a_spec, sc_spec = _spectral_iterate(best_seq, time.monotonic() + 20, max_iters=30, strength=0.3)
            if sc_spec < best_score:
                update_best(a_spec, sc_spec, "spectral_pre_lbfgs")
            dl2 = phase3_end - 5.0
            a_best2, sc_best2 = _lbfgs_logsumexp(a_spec, dl2, beta_scale=0.3,
                                                  min_beta_frac=0.00001, n_levels=10)
            update_best(a_best2, sc_best2, "spec+lbfgs")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Adam peak-targeted (250s) ---
    phase4_end = min(deadline, t_start + 1240.0)
    if best_seq is not None and time.monotonic() < phase4_end:

        dl = phase4_end - 5.0
        a_best, sc_best = _adam_peak_targeted(best_seq, dl, lr=0.01, k_top=3, lr_decay=0.97)
        update_best(a_best, sc_best, "adam_k3")
        
        if time.monotonic() < phase4_end - 20:
            dl2 = phase4_end - 5.0
            a_best2, sc_best2 = _adam_peak_targeted(best_seq, dl2, lr=0.005, k_top=7, lr_decay=0.96)
            update_best(a_best2, sc_best2, "adam_k7")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Simulated annealing local search (200s) ---
    phase5_end = min(deadline, t_start + 1440.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Simulated annealing")
        dl = phase5_end - 5.0
        a_best, sc_best = _sim_anneal_local_search(best_seq, dl, n_trials=40000,
                                                    T_init=0.005, T_min=0.00005)
        update_best(a_best, sc_best, "sim_anneal")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Directed peak reduction (150s) ---
    phase6_end = min(deadline, t_start + 1590.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Directed peak reduction")
        dl = phase6_end - 5.0
        a_best, sc_best = _directed_peak_reduction(best_seq, dl, n_rounds=300)
        update_best(a_best, sc_best, "directed")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Multi-restart (100s) ---
    phase7_end = min(deadline, t_start + 1690.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        print("Phase 7: Multi-restart")
        n_restarts = 3
        per_restart = max(15.0, (phase7_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            noise = 0.01 + 0.01 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase7_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _lbfgs_logsumexp(p_start, dl, beta_scale=0.5,
                                                min_beta_frac=0.00005, n_levels=10)
            update_best(a_best, sc_best, f"restart{i}")
    
    print(f"Phase 7 done: best={best_score:.10f}")
    
    # --- Phase 8: Final intensive polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 8: Final polishing, remaining={remaining:.0f}s")
        
        # 8a: Very fine L-BFGS
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.4)
            a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, beta_scale=0.2,
                                                min_beta_frac=0.00001, n_levels=15)

        if time.monotonic() < deadline - 30:

                                                        T_init=0.002, T_min=0.00002)

    # --- Final validation and return ---
    if best_seq is not None:

        if len(final_seq) <= 16384:

                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular_weight(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
