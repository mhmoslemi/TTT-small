def next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

    c_max = c.max()
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_one(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxcor': 40, 'maxls': 200})
        return _softmax(result.x)

"""
Step function coefficient optimization using L-BFGS with aggressive beta-continuation,
subgradient polishing with adaptive active sets, annealing-style restarts, and
multi-resolution search across different sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_exact(a, c, n, tol_frac=0.0005):
    c_max = c.max()
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_one(a_init, beta, n, max_iter):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxcor': 40, 'maxls': 200})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _beta_schedule(a_start, deadline, min_remaining=10.0):
    n = len(a_start)
    c = _full_convolve(a_start, a_start)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a_start.copy()

    fracs = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003,
             0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
    betas = max_c * np.array(fracs)
    current = a_start.copy()

    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 3:

        max_iter = min(6000, max(100, int(remaining / (est_time * 12))))
        a_new = _lbfgs_one(current, beta, n, max_iter)
        sc_new = _score(a_new)
        sc_cur = _score(current)
        if sc_new < sc_cur:
            current = a_new

    return current


def _subgrad_polish(a_init, deadline, max_steps=10000, lr_init=0.05):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    no_improve = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_exact(a, c, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step = lr / (1.0 + 0.002 * i)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_ch = _softmax(z)
            sc = _score(a_ch)
            if sc < best_sc:
                best_sc = sc
                best_a = a_ch.copy()
                no_improve = 0
                if i > 200:
                    lr *= 0.96
            else:
                no_improve += 1
                lr *= 0.82
                if no_improve > 20:
                    break

    a_f = _softmax(z)
    sc_f = _score(a_f)
    if sc_f < best_sc:
        best_sc = sc_f
        best_a = a_f.copy()
    return best_a, best_sc


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)

        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_mixed(a, rng):

    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    # Local noise
    z = z + 0.01 * rng.standard_normal(n)
    # Global Fourier mode

    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _make_seed(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-((t - 0.3) / 0.18) ** 2) + np.exp(-((t - 0.7) / 0.18) ** 2)
        return a / a.sum()
    elif kind == "abs_gauss":
        a = np.abs(rng.standard_normal(n)) + 0.05

    elif kind == "hamming":
        t = np.arange(n)
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * t / n)

                a_r3, sc_r3 = _local_search(a_r, deadline - 18, n_trials=30, sigma=0.002)
                _update(a_r3, sc_r3, f"restart_{i}_ls")

    # Phase 6: Try n=4096 if time allows
    print(f"\n[Phase 6] Try n=4096 (t={time.monotonic() - t0:.1f}s)")
    if time.monotonic() < deadline - 120:
        n_alt = 4096
        # Upsample parent to n_alt or use fresh init
        if best_seq is not None:
            from scipy.interpolate import interp1d
            t_old = np.linspace(0, 1, len(best_seq))
            t_new = np.linspace(0, 1, n_alt)
            f_interp = interp1d(t_old, best_seq, kind='cubic', bounds_error=False,
                                fill_value="extrapolate")
            a_alt = np.maximum(f_interp(t_new), 0.0)
            a_alt = a_alt / a_alt.sum()
        else:
            a_alt = _make_seed(n_alt, "uniform", rng)

        sc_alt = _score(a_alt)
        print(f"  n=4096 init score: {sc_alt:.10f}")

        if sc_alt < best_score + 0.1:  # Only worth it if competitive
            a_alt_opt = _beta_schedule(a_alt, min(deadline - 30, time.monotonic() + 80),
                                       min_remaining=5.0)
            _update(a_alt_opt, None, "n4096")

    # Phase 7: Final polishing cycle
    print(f"\n[Phase 7] Final polish (t={time.monotonic() - t0:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Subgradient
        a_f1, sc_f1 = _subgrad_polish(best_seq, deadline - 3, max_steps=2000, lr_init=0.015)
        _update(a_f1, sc_f1, "final_sg1")

        # One more fine L-BFGS
        if time.monotonic() < deadline - 2:
            n = len(best_seq)
            c = _full_convolve(best_seq, best_seq)
            mc = float(c.max())
            a_f2 = _lbfgs_one(best_seq, mc * 0.00001, n, 500)
            _update(a_f2, None, "final_lbfgs")

        # Final local search
        if time.monotonic() < deadline - 1:
            a_f3, sc_f3 = _local_search(best_seq, deadline - 1, n_trials=20, sigma=0.001)
            _update(a_f3, sc_f3, "final_ls")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t0:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fb = _make_seed(256, "uniform", rng)
    print(f"FALLBACK: score={_score(fb):.10f}")
    return [float(x) for x in fb]
