```python
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)

        return np.inf
    c = _full_convolve(a, a)

            c_top = c[idx]
            c_shift = (c_top - c_top.max()) / beta
            e = np.exp(c_shift)
            w_full[idx] = e / e.sum()
        else:

            e = np.exp(c_shift)
            w_full = e / e.sum()
```

1. Sort a in descending order: a_(1) >= a_(2) >= ... >= a_(n)
2. Find ρ = max{ j : a_(j) - (sum_{i=1}^{j} a_(i) - 1) / j > 0 }
3. θ = (sum_{i=1}^{ρ} a_(i) - 1) / ρ
4. Return max(a - θ, 0)

```python

    sorted_a = np.sort(a)[::-1]
    cssv = np.cumsum(sorted_a)

    vals = (cssv - 1.0) / np.arange(1, n + 1)
    mask = sorted_a > vals
    if mask.any():
        rho = np.max(np.where(mask)) + 1
    else:
        rho = n
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(a - theta, 0.0)
```

```python
    """Upsample a sequence by a factor using interpolation."""
    n = len(a)
    new_n = n * factor
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)

    padded = np.zeros(new_n * factor)
    padded[:n] = a
    return padded.reshape(new_n, factor).mean(axis=1) / (padded.sum() / (new_n * factor))
```

Multi-resolution gradient optimization with Top-K soft-max for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses coarse-to-fine optimization, Adam with adaptive beta, and multi-restart.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_z(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _upsample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _adam_optimize(a_init, deadline, top_k=None, beta_init_scale=0.1, lr_init=0.01):
    """Adam optimizer with soft-max smoothed max and optional top-k focus."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_cmax = float(c_init.max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    beta_scale = beta_init_scale

    max_c_init = best_cmax

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_z(z)

        c = _full_convolve(a, a)
        max_c = float(c.max())
        beta = max(beta_scale * max_c_init, 1e-10)

        # Compute soft-max weights (optionally top-k)
        if top_k is not None and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            w = np.zeros(len(c))
            c_top = c[idx]
            c_shift = (c_top - c_top.max()) / beta
            e = np.exp(c_shift)
            w[idx] = e / e.sum()
        else:
            c_shift = (c - max_c) / beta
            e = np.exp(c_shift)
            w = e / e.sum()

        # Gradient
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Periodic check and adaptation
        if t % 200 == 0:
            sc = float(_full_convolve(a, a).max())
            if np.isfinite(sc) and sc < best_cmax:
                best_cmax = sc
                best_a = a.copy()
            # Slowly decrease beta
            beta_scale *= 0.98
            beta_scale = max(beta_scale, 0.001)
            # Slowly decrease lr
            lr = max(0.001, lr * 0.995)

    # Final check
    a_final = _softmax_z(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_cmax:
        best_cmax = sc_final
        best_a = a_final.copy()

    return best_a, best_cmax


def _adam_multi_beta(a_init, deadline):
    """Run Adam with multiple beta scales, decreasing over time."""
    n = len(a_init)
    best_a = a_init.copy()
    best_cmax = float(_full_convolve(a_init, a_init).max())

    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())

    # Phases with different beta scales
    beta_scales = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    phase_time = (deadline - time.monotonic()) / len(beta_scales)
    
    for i, bs in enumerate(beta_scales):
        phase_deadline = time.monotonic() + phase_time
        if time.monotonic() > deadline:
            break
        
        a_result, cmax_result = _adam_optimize(
            best_a, phase_deadline, top_k=None,
            beta_init_scale=bs, lr_init=0.01 * (0.9 ** i)
        )
        if cmax_result < best_cmax:
            best_cmax = cmax_result
            best_a = a_result
        print(f"  Beta phase {i} (scale={bs}): cmax={cmax_result:.12f}, "
              f"score={_score(best_a):.10f}, time={time.monotonic():.1f}s")

    return best_a, best_cmax


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_z(z)
