def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):

    e = np.exp(z)

    except Exception:

def _subgradient_polish(a_init, deadline, max_iter=50000):
    """Subgradient method with Barzilai-Borwein step size."""
    n = len(a_init)
    a_cur = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    S = float(a_cur.sum())
    
    prev_a = a_cur.copy()
    prev_grad = None
    step = 0.01
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Subgradient of max(conv) at k_star: 2*a[k_star - j] (if valid)
        grad = np.zeros(n)
        for j in range(n):
            k_j = k_star - j
            if 0 <= k_j < n:
                grad[j] = 2.0 * a_cur[k_j]
        
        # Gradient of the full score 2n*M/S^2:
        # d/dj [2n*conv[k*]/S^2] = 2n * grad_j / S^2 - 2n*conv[k*] * a_j / S^3
        # = (2n/S^2) * (grad_j - M*a_j/S)
        grad_full = grad - M * a_cur / S
        
        # Project: subtract mean to maintain sum
        grad_proj = grad_full - np.mean(grad_full)
        
        # BB step size
        if prev_grad is not None:
            delta_a = a_cur - prev_a
            delta_g = grad_proj - prev_grad
            num = float(np.dot(delta_a, delta_g))
            den = float(np.dot(delta_g, delta_g))
            if den > 1e-30 and abs(num) > 1e-30:
                step_bb = abs(num) / den
                step = 0.5 * (step + np.clip(step_bb, 1e-8, 0.1))
        
        # Update
        a_new = a_cur - step * grad_proj
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        prev_a = a_cur.copy()
        prev_grad = grad_proj.copy()
        a_cur = a_new
    
    return best_a, best_score
