def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        
        # Get top K peak indices
        peak_idx = np.argsort(c)[::-1][:top_k]
        peaks = c[peak_idx]
        pm = float(peaks.max())
        
        if pm < 1e-300:
            break
        
        # Adaptive beta
        beta = max(1e-8 * pm, cm0 * 0.01 * (0.98 ** (t // 100)))
        
        # Log-sum-exp of top K peaks
        sh = (peaks - pm) / beta
        e = np.exp(sh - sh.max())
        w = e / e.sum()
        
        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            wi = w[idx]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_a[i_lo:i_hi+1] += wi * 2.0 * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        
        # Gradient w.r.t. z (softmax)
        Wa = float(np.dot(grad_a, a))
        grad_z = 2.0 * n * a * (grad_a - Wa)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()
        
        # Periodic check
        if t % 100 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
    
    return best_a, best_score

`max_k c[k] = sup_{|φ(ω)|<=1} |∫ φ(ω) |A(ω)|^2 dω|`

So minimizing max_k c[k] is equivalent to minimizing `sup_{|φ|≤1} |∫ φ |A|^2|`. This is a semi-infinite program, but we can approximate it with a finite number of constraint functions.

Option 2: **p-norm approximation**. Instead of the max (which is the ∞-norm), use the p-norm for large p:
`||c||_p = (sum_k c[k]^p)^(1/p)`

For p → ∞, this approaches the max. For finite p, it's smooth and differentiable. The gradient is:
`d/d a_i ||c||_p = (1/p) * ||c||_p^(1-p) * sum_k p * c[k]^(p-1) * dc[k]/da[i]`

This might be easier to optimize than the log-sum-exp because the p-norm has a simpler structure.

Let me try yet another approach: **Nesterov's accelerated gradient method** for the log-sum-exp surrogate.

Nesterov's method has a convergence rate of O(1/t^2) for smooth convex functions, compared to O(1/t) for standard gradient descent. This could give a significant speedup.

The Nesterov update is:
```
y_t = x_t + (t-1)/(t+2) * (x_t - x_{t-1})
x_{t+1} = y_t - (1/L) * ∇f(y_t)
```

where L is the Lipschitz constant of the gradient.

For our problem, L depends on n and the current point, but we can use a backtracking line search to find a good step size.

"""
Optimized step function search using Nesterov OGV with adaptive beta,
LP-based peak balancing, top-K peak targeting, and subgradient polish.
Targets score < 1.5027 for n=8192 non-negative sequences.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _parametric_init(n, etype='binom_rev'):
    t = np.linspace(0, 1, n)
    if etype == 'binom_rev':
        b = np.zeros(n)
        b[0] = 1.0
        for j in range(1, n):
            b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
        a = b[::-1].copy()
    elif etype == 'exp3':
        a = np.exp(3.0 * (t - 0.3))
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.4))
    elif etype == 'exp7':
        a = np.exp(7.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(np.maximum(0.01 + 0.99 * t, 0.001))
    elif etype == 'linear':
        a = 0.1 + 0.9 * t
    elif etype == 'quadratic':
        a = 0.05 + 0.3 * t + 0.65 * t**2
    elif etype == 'cubic':
        a = 0.03 + 0.1 * t + 0.2 * t**2 + 0.67 * t**3
    elif etype == 'power03':
        a = np.maximum(0.02 + 0.98 * t, 0.001) ** 0.3
    elif etype == 'power07':
        a = np.maximum(0.02 + 0.98 * t, 0.001) ** 0.7
    elif etype == 'inv_sqrt':
        j = np.arange(n, dtype=np.float64) + 1.0
        a = 1.0 / np.sqrt(j)
        a = a[::-1].copy()
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _obj_grad_topk(z, beta, n, top_k):
    """Objective and gradient using top-K log-sum-exp surrogate."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if len(c) <= top_k:
        peak_idx = np.arange(len(c))
    else:
        peak_idx = np.argpartition(c, -top_k)[-top_k:]
    peaks = c[peak_idx]
    pm = float(peaks.max())
    if pm < 1e-300:
        return 0.0, np.zeros_like(z)
    sh = (peaks - pm) / beta
    max_sh = float(sh.max())
    e = np.exp(sh - max_sh)
    w = e / e.sum()
    lse = pm + beta * (max_sh + float(np.log(e.sum())))
    obj = float(2.0 * n * lse)

    grad_a = np.zeros(n)
    for idx in range(len(peak_idx)):
        k = peak_idx[idx]
        wi = w[idx]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            seg_len = i_hi - i_lo + 1
            grad_a[i_lo:i_hi + 1] += wi * 2.0 * a[k - i_lo:k - i_lo + seg_len]

    Wa = float(np.dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return obj, np.zeros_like(z)
    return obj, grad_z


def _nesterov_oge(z_init, deadline, beta_fracs, top_k=5, max_beta_iters=3000):
    """Nesterov OGV with adaptive beta continuation."""
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)

    a0 = best_a.copy()
    c0 = _full_convolve(a0, a0)
    m0 = float(c0.max())

    beta_list = [m0 * b for b in beta_fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        x = z.copy()
        x_prev = z.copy()
        step = 0.5 / (n * beta + 1.0)
        t = 0

        for it in range(max_beta_iters):
            if time.monotonic() > deadline:
                break
            t += 1
            momentum = (t - 1) / (t + 2.0) if t > 1 else 0.0
            y = x + momentum * (x - x_prev)

            obj_y, grad_y = _obj_grad_topk(y, beta, n, top_k)
            if not np.all(np.isfinite(grad_y)):
                break

            # Backtracking line search
            x_new = None
            grad_sq = float(np.dot(grad_y, grad_y))
            if grad_sq < 1e-30:
                x_new = y.copy()
            else:
                for ls in range(15):
                    s_try = step
                    x_trial = y - s_try * grad_y
                    x_trial = x_trial - x_trial.mean()
                    obj_trial, _ = _obj_grad_topk(x_trial, beta, n, top_k)
                    if obj_trial <= obj_y + 0.5 * s_try * grad_sq:
                        x_new = x_trial
                        step = min(step * 1.5, 10.0)
                        break
                    step *= 0.5
                if x_new is None:
                    x_new = y - step * grad_y
                    x_new = x_new - x_new.mean()

            x_prev = x
            x = x_new

            # Periodic score check
            if t % 200 == 0:
                a_cur = _softmax_fn(x)
                sc = _score_norm_fn(a_cur)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_cur.copy()

        # Update z and check improvement
        a_new = _softmax_fn(x)
        sc_new = _score_norm_fn(a_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            z = x.copy()
        else:
            # No improvement, try to continue from current x anyway
            z = x.copy()

    return best_a, best_score


def _adam_peak(z_init, deadline, top_k=3, lr=0.01, decoupled=False):
    """Adam optimizer with top-K peak targeting."""
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_int = 200

    a0 = best_a.copy()
    c0 = _full_convolve(a0, a0)
    m0 = float(c0.max())
    beta = 0.01 * m0

    for it in range(5000000):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        if len(c) <= top_k:
            peak_idx = np.arange(len(c))
        else:
            peak_idx = np.argpartition(c, -top_k)[-top_k:]
        peaks = c[peak_idx]
        pm = float(peaks.max())
        if pm < 1e-300:
            break

        # Adaptive beta
        beta = max(1e-10 * pm, m0 * 0.01 * (0.985 ** (t // 200)))

        sh = (peaks - pm) / beta
        max_sh = float(sh.max())
        e = np.exp(sh - max_sh)
        w = e / e.sum()

        grad_a = np.zeros(n)
        for idx in range(len(peak_idx)):
            k = peak_idx[idx]
            wi = w[idx]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                seg_len = i_hi - i_lo + 1
                grad_a[i_lo:i_hi + 1] += wi * 2.0 * a[k - i_lo:k - i_lo + seg_len]

        Wa = float(np.dot(grad_a, a))
        grad_z = 2.0 * n * a * (grad_a - Wa)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()

        if t % check_int == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0
                # Reset momentum
                m[:] = 0
                v[:] = 0

    return best_a, best_score


def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            seg_len = i_hi - i_lo + 1
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + seg_len]
    return A_mat


def _lp_phase(a_init, deadline, K=50, max_rounds=20):
    """LP-based peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Active peaks: top K plus those within 97% of max
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.97 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]

        A_mat = _build_lp_matrix(best_a, active, n)
        b_vec = np.full(len(active), M * 1.0005)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})
            if res.status != 0:
                if time.monotonic() > deadline:
                    break
                continue
            b_new = res.x
            sum_b = float(b_new.sum())
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            if time.monotonic() > deadline:
                break
            continue

        # Find best blend
        current_score = 2.0 * n * M / (S * S)
        improved = False
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = float(a_trial.sum())
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _subgrad_polish(a_init, deadline, num_steps=50000):
    """Subgradient method directly on the max peak."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    step_size = 1e-4
    t = 0
    avg_grad = np.zeros(n)

    for it in range(num_steps):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)

        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        # Subgradient at the peak
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        grad_a = np.zeros(n)
        seg_len = i_hi - i_lo + 1
        grad_a[i_lo:i_hi + 1] = 2.0 * a[k_star - i_lo:k_star - i_lo + seg_len]

        # Convert to z-space gradient
        Wa = float(np.dot(grad_a, a))
        grad_z = 2.0 * n * a * (grad_a - Wa)

        if not np.all(np.isfinite(grad_z)):
            break

        # Averaging
        avg_grad = (1.0 - 1.0/t) * avg_grad + (1.0/t) * grad_z
        # Use averaged gradient for better convergence
        z = z - step_size / np.sqrt(t) * avg_grad
        z = z - z.mean()
        z = np.clip(z, -50, 50)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _try_n_value(a_seed, n_target, deadline, label):
    """Optimize at a specific n value using continuation method."""
    a = _interp_profile(a_seed, n_target)
    sc = _score_norm_fn(a)
    print(f"  {label}: n={n_target}, initial norm_score={sc:.10f}")

    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    # Nesterov with continuation
    c0 = _full_convolve(a, a)
    m0 = float(c0.max())
    fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    beta_list = [m0 * b for b in fracs]

    best_a, best_sc = _nesterov_oge(z, deadline, beta_list, top_k=5, max_beta_iters=2000)
    return best_a, best_sc

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov OGV with adaptive beta, LP peak balancing, Adam polish,
subgradient averaging, and coarse-to-fine restarts.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a

def _obj_grad(z, beta, n, top_k):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if len(c) <= top_k:
        pi = np.arange(len(c))
    else:
        pi = np.argpartition(c, -top_k)[-top_k:]
    peaks = c[pi]
    pm = float(peaks.max())
    if pm < 1e-300:
        return 0.0, np.zeros_like(z)
    sh = (peaks - pm) / beta
    mx = float(sh.max())
    e = np.exp(sh - mx)
    w = e / e.sum()
    lse = pm + beta * (mx + float(np.log(e.sum())))
    obj = float(2.0 * n * lse)
    ga = np.zeros(n)
    for idx in range(len(pi)):
        k = pi[idx]
        wi = w[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            sl = hi - lo + 1
            ga[lo:hi + 1] += wi * 2.0 * a[k - lo:k - lo + sl]
    Wa = float(np.dot(ga, a))
    gz = 2.0 * n * a * (ga - Wa)
    if not np.all(np.isfinite(gz)):
        return obj, np.zeros_like(z)
    return obj, gz
