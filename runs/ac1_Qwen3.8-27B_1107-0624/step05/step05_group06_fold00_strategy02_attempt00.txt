def evaluate_sequence(seq):
    """Evaluates a sequence definition with handling for empty or default arguments."""
    if not isinstance(seq, list):
        return np.inf
    if len(seq) == 0:
        return 0.0
    n = len(seq)
    sum_a = sum(seq)
    if sum_a < 0.01:
        return np.inf
    conv = [0.0] * (2*n - 1)
    for i in range(2*n - 1):
        sum_val = sum((seq[j] * seq[i-j] if i-j >= 0 and i-j < n else 0) for j in range(max(0, i-n+1), min(i+1, n)))
        conv[i] = sum_val
    max_b = max(conv)
    return 4.0 * n * max_b / sum_a**2

def gradient_exact(z, a, c):
    n = len(a)
    c_max = max(c)
    tol = 0.001 * c_max
    active = (c >= (c_max - tol)) if c_max is not None else [True] * n
    if not active.any():
        active = [np.argmax(c) != None]
        if len(active) == 0:
            active = [0] * n
    w = np.zeros(c.shape)
    w[~active] = 0.0
    w[active] = 1.0 / np.count_nonzero(active)
    a_rev = a[::-1]
    wc = np.zeros(len(w) + len(a_rev) - 1)
    for i in range(len(w)):
        for j in range(len(a_rev)):
            wc[i+j] += w[i] * a_rev[j]
    h = wc[n-1:2*n-1]
    W = np.dot(w, c)
    grad = 8.0 * np.array([(a_val / a_sum)**2 * (a_sum **0.5) * w_val * (h_val - W) for a_val, a_sum, w_val, h_val in zip(a, np.sum(a), W, h)]).reshape(len(a),)
    grad[0] = grad[0] if type(grad[0]) is float else float(grad[0])
    return grad

def perturb_fourier(a, sigma=0.03, max_shift=0.01, n_modes=3):
    """Perturb the sequence a using Fourier modes with randomization."""
    if n_modes <= 0:
        return np.clip(a / np.sum(a), 0.001, 1000.0) * np.sqrt(2*len(a))
    n = len(a)
    z = np.log(np.maximum(a, 1e-15)) - np.mean(np.log(np.maximum(a, 1e-15)))
    t = np.arange(n) / n
    for _ in range(n_modes):
        freq_index = np.random.choice(min(20, n//2) if n>0 else 1, 1)[0]
        freq = 0.5 + np.random.random() * 0.5 * freq_index
        phase = 2 * np.pi * np.random.random()
        amp = np.sqrt(np.random.rand()) * 0.5
        shift_amount = amp * np.cos(phase + amp * freq * t)
        # Weight towards the start to avoid invalid values
        shift = z + shift_amount
        # Apply clipping to prevent going out of bounds
        shift = np.clip(shift, float('-inf'), float('inf'))
        perturbed = np.exp(shift / (100.0 if n_modes > 1 else 50.0)) / (np.sum(np.exp(shift)) / n) * np.sqrt(2 * n)
        a = perturbed if np.random.rand() < 0.5 else a
    return a

            current_sequence = perturb_fourier(np.ones(n_length) / n_length * np.sqrt(2*n_length))
            s_val = sum(current_sequence)
            if s_val < 0.01:
                current_sequence = current_sequence * (0.01 / s_val) if s_val != 0 else current_sequence
                s_val = sum(current_sequence)
            score = evaluate_sequence(current_sequence)
            candidates.append((current_sequence, score))
        else:
            # Further iterations - use more seeds if possible to explore more
            if len(falldown_candidates) < 3:
                new_cand = np.random.rand(n_length) * np.sqrt(2*n_length)
                new_cand = np.clip(new_cand, 0.001, 50)
                s_val = sum(new_cand)
                assert s_val >= 0.01, "Failed to normalize:"
                score = evaluate_sequence(new_cand)
                falldown_candidates.append((new_cand, score))
            else:
                break
    
    best_seq = None
    best_score = np.inf
    
    for cand, score in candidates:
        if score == np.inf or score < 0:
            continue
        best_seq = cand
        best_score = score
        break
    
    for cand, score in falldown_candidates:
        if score < best_score and score != np.inf:
            best_seq = cand
            best_score = score
    
    # Adjust based on time and progress
    time_remaining = time_budget - time.time() + 60
    if time_remaining < 10:
        # Not enough time, return the best candidate we have
        pass
    elif time_remaining >= 60:
        # Run longer iterations
        for cand, score in candidates:
            if cand is not None and score < best_score and score != np.inf:
                perturbed_cand = perturb_fourier(cand, n_length)
                s_val = sum(perturbed_cand)
                if s_val < 0.01:
                    perturbed_cand = perturbed_cand * (0.01 / s_val)
                    falldown_candidates.append((perturbed_cand, evaluate_sequence(perturbed_cand)))
                score_new = evaluate_sequence(perturbed_cand)
                if score_new < best_score and score_new != np.inf:
                    best_score = score_new
                    best_seq = perturbed_cand
        
        for cand, score in falldown_candidates:
            score_new = evaluate_sequence(perturb_fourier(cand, n_length))
            if score_new < best_score:
                best_score = score_new
                best_seq = cand
    
    # Final scoring
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq)
        if final_score > best_score or not np.isfinite(final_score):
            best_score = final_score
    
    # Return the best sequence and its score
    return best_seq.tolist()
