def _smooth_max_grad(a, epsilon):
    """Compute gradient of epsilon * log(sum exp(b[k]/epsilon)) where b = conv(a,a)."""
    n = len(a)
    b = _full_convolve(a, a)  # length 2n-1
    # Softmax weights
    b_shifted = (b - b.max()) / epsilon
    w = np.exp(b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Uniform weights (fallback)
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w = w / w_sum
    
    # Gradient = 2 * cross_correlation(w, a)
    N = 1
    while N < len(w) + n - 1:
        N *= 2
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad_raw = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * grad_raw
    
    # Project onto sum(a)=1: subtract mean
    grad = grad - grad.mean()
    return grad, b.max()

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

"""
Optimized step function coefficient search using:
1. FFT-based smooth-max gradient (softmax continuation with Adam)
2. Peak-targeted structured local search
3. Multi-restart with controlled perturbations
4. Skew-direction exploitation for convolution peak reduction

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_beta_profile(n, alpha, beta):
    """Sample from a Beta-like distribution shape."""
    t = np.linspace(0.01, 0.99, n)
    a = t ** (alpha - 1) * (1.0 - t) ** (beta - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_gaussian_profile(n, center, width):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-(t - center) ** 2 / (2.0 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _smooth_max_grad(a, epsilon):
    """Compute gradient of epsilon*log(sum exp(b[k]/epsilon)) where b=conv(a,a).
    Returns (gradient_projected, b_max)."""
    n = len(a)
    b = _full_convolve(a, a)
    b_max = float(b.max())
    b_shifted = (b - b_max) / epsilon
    w = np.exp(b_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = w / w_sum

    # Gradient = 2 * cross_correlation(w, a)
    # result[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    N = 1
    while N < (2 * n - 1) + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad_raw = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * grad_raw

    # Project onto sum(a)=1: subtract mean
    grad = grad - grad.mean()
    return grad, b_max


def _topk_smooth_grad(a, k_top, epsilon):
    """Hybrid: use top-k lags with softmax weighting for a sharper gradient."""
    n = len(a)
    b = _full_convolve(a, a)
    b_max = float(b.max())

    # Get top-k lags
    k_actual = min(k_top, 2 * n - 1)
    top_idx = np.argsort(b)[-k_actual:]

    # Softmax over top-k
    b_top = b[top_idx]
    b_top_shifted = (b_top - b_top.max()) / epsilon
    w_top = np.exp(b_top_shifted)
    w_top = w_top / w_top.sum()

    # Build sparse weight vector
    w = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        w[int(idx)] = w_top[i]

    # Same FFT-based gradient computation
    N = 1
    while N < (2 * n - 1) + n - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    grad_raw = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * grad_raw
    grad = grad - grad.mean()
    return grad, b_max


def _adam_continuation(a_init, deadline, rng, lr_base=0.03, k_top=None):
    """Adam optimizer with softmax continuation."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    b0 = float(_full_convolve(a, a).max())
    if b0 < 1e-15:
        return best_a, best_score

    # Epsilon schedule
    eps_fracs = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps_adam = 0.8, 0.999, 1e-8
    lr = lr_base

    for eps_frac in eps_fracs:
        if time.monotonic() > deadline - 3:
            break
        epsilon = b0 * eps_frac

        # Adaptive iteration count
        iters = max(50, int(1500 * eps_frac + 200))

        for it in range(iters):
            if time.monotonic() > deadline - 3:
                break
            t += 1

            if k_top is not None:
                grad, b_max = _topk_smooth_grad(a, k_top, epsilon)
            else:
                grad, b_max = _smooth_max_grad(a, epsilon)

            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            mhat = m / (1 - b1 ** t)
            vhat = v / (1 - b2 ** t)

            a_new = a - lr * mhat / (np.sqrt(vhat) + eps_adam)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / s

            a = a_new

            if t % 15 == 0:
                sc = _score_norm_fn(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        # Decay LR slightly between phases
        lr = max(0.005, lr * 0.9)

    return best_a, best_score


def _peak_local_search(a_init, deadline, rng, n_trials=50000):

    n = len(a_init)
    a = a_init.copy()

    no_improve = 0
    max_no_improve = 5000

        if no_improve > max_no_improve:

            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            no_improve = 0

        top_lags = np.argsort(c)[-3:]
        max_lag = int(top_lags[-1])

        move_type = it % 4

        if move_type == 0:
            # Strategy: Reduce peak-contributing elements, add to safe positions
            # Responsibility of element j for lag k: proportional to a[k-j] (if valid)
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min > i_max:
                continue
            pairs = np.arange(i_min, i_max + 1)
            resp = a[max_lag - pairs]
            if resp.max() < 1e-15:
                continue
            # Find top responsible and bottom responsible
            top_resp_idx = pairs[int(np.argmax(resp))]
            # Safe positions: low total responsibility across all top lags
            total_resp = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    total_resp[lo:hi + 1] += a[kl - hi:kl - lo + 1][::-1]
            safe_candidates = np.argsort(total_resp)[:max(1, n // 10)]
            add_idx = int(safe_candidates[rng.integers(len(safe_candidates))])

            if add_idx == top_resp_idx:
                continue

            frac = rng.uniform(0.05, 0.3)
            delta = a[top_resp_idx] * frac
            a_trial = a.copy()
            a_trial[top_resp_idx] -= delta
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    no_improve = 0
                else:
                    no_improve += 1

        elif move_type == 1:
            # Strategy: Block multiplicative perturbation
            block_size = int(rng.integers(3, min(32, n // 4)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.8, 1.2, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                no_improve = 0
            else:
                no_improve += 1

        elif move_type == 2:
            # Strategy: Skew perturbation (exploit asymmetry to reduce middle peak)
            t = np.linspace(-1.0, 1.0, n)
            direction = t * a  # Skew direction
            direction -= direction.mean()  # Keep sum unchanged
            scale = rng.uniform(0.01, 0.05)
            a_trial = a + scale * direction
            a_trial = np.maximum(a_trial, 1e-10)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                no_improve = 0
            else:
                no_improve += 1

        else:
            # Strategy: Pair mass exchange targeting peak lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min > i_max:
                no_improve += 1
                continue
            # Find the pair with max product
            pairs = np.arange(i_min, i_max + 1)
            products = a[pairs] * a[max_lag - pairs]
            top_pair = int(np.argmax(products))
            idx1 = int(pairs[top_pair])
            idx2 = int(max_lag - pairs[top_pair])
            if idx1 == idx2:
                no_improve += 1
                continue

            # Reduce both, add to random safe positions
            frac = rng.uniform(0.1, 0.3)
            d1 = a[idx1] * frac
            d2 = a[idx2] * frac
            a_trial = a.copy()
            a_trial[idx1] -= d1
            a_trial[idx2] -= d2
            # Distribute to 2 random positions far from peak
            safe_pos = np.random.choice(n, size=2, replace=False)
            a_trial[int(safe_pos[0])] += d1
            a_trial[int(safe_pos[1])] += d2
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    no_improve = 0
                else:
                    no_improve += 1

    return best_a, best_sc


def _simulated_annealing(a_init, deadline, rng, T0=0.01, T1=0.0001):
    """Simulated annealing with peak-targeting moves."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    curr_sc = best_sc

    total_iters = 30000
    no_improve = 0

    for it in range(total_iters):
        if time.monotonic() > deadline - 2:
            break

        # Temperature schedule
        T = T0 * (T1 / T0) ** (it / total_iters)

        # Generate trial
        move_type = rng.integers(4)
        a_trial = a.copy()

        if move_type == 0:
            # Single element perturbation
            idx = int(rng.integers(n))
            delta = rng.normal(0, 0.1) * a[idx]
            a_trial[idx] += delta
            a_trial[idx] = max(0.0, a_trial[idx])
        elif move_type == 1:
            # Pair exchange
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            swap_amt = min(a[i1], a[i2]) * rng.uniform(0.1, 0.5)
            a_trial[i1] -= swap_amt
            a_trial[i2] += swap_amt
            a_trial[i1] = max(0.0, a_trial[i1])
            a_trial[i2] = max(0.0, a_trial[i2])
        elif move_type == 2:
            # Block smooth (replace block with its mean)
            block_size = int(rng.integers(5, min(20, n // 10)))
            start = int(rng.integers(0, n - block_size))
            block_mean = a_trial[start:start + block_size].mean()
            a_trial[start:start + block_size] = block_mean
        else:
            # Skew
            t = np.linspace(-1.0, 1.0, n)
            direction = t * a
            direction -= direction.mean()
            scale = rng.normal(0, 0.02)
            a_trial += scale * direction

        a_trial = np.maximum(a_trial, 1e-10)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s

        sc_trial = _score_norm_fn(a_trial)
        delta_sc = sc_trial - curr_sc

        if delta_sc < 0 or rng.random() < np.exp(-delta_sc / max(T, 1e-15)):
            a = a_trial
            curr_sc = sc_trial
            if sc_trial < best_sc:
                best_sc = sc_trial
                best_a = a_trial.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_a, best_sc

        if time.monotonic() < phase2_end - 30:
            dl4 = min(phase2_end - 25, time.monotonic() + 80.0)
            a_best4, sc_best4 = _adam_continuation(best_seq, dl4, rng, lr_base=0.01, k_top=2)
            update_best(a_best4, sc_best4, "adam_top2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Peak-targeted local search (400s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"Phase 3: Peak local search")
        dl = phase3_end - 5
        a_best, sc_best = _peak_local_search(best_seq, dl, rng, n_trials=100000)
        update_best(a_best, sc_best, "peak_search_1")

        # Second pass
        if time.monotonic() < phase3_end - 30:
            dl2 = phase3_end - 5
            a_best2, sc_best2 = _peak_local_search(best_seq, dl2, rng, n_trials=80000)
            update_best(a_best2, sc_best2, "peak_search_2")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Simulated annealing (200s) ---
    phase4_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: Simulated annealing")
        dl = phase4_end - 5
        a_best, sc_best = _simulated_annealing(best_seq, dl, rng, T0=0.005, T1=0.00005)
        update_best(a_best, sc_best, "sa_1")

        if time.monotonic() < phase4_end - 30:
            dl2 = phase4_end - 5
            a_best2, sc_best2 = _simulated_annealing(best_seq, dl2, rng, T0=0.002, T1=0.00002)
            update_best(a_best2, sc_best2, "sa_2")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (200s) ---
    phase5_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"Phase 5: Multi-restart")
        n_restarts = 5
        per_restart = max(20.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise_scale = 0.01 + 0.01 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_scale * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            # Quick Adam + local search
            dl = min(phase5_end - 5, time.monotonic() + per_restart * 0.6)
            a_best, sc_best = _adam_continuation(p_start, dl, rng, lr_base=0.02, k_top=3)

            if time.monotonic() < phase5_end - 5:
                dl2 = min(phase5_end - 5, time.monotonic() + per_restart * 0.4)
                a_best, sc_best = _peak_local_search(a_best, dl2, rng, n_trials=10000)

            update_best(a_best, sc_best, f"restart{i}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Try different n (150s) ---
    phase6_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        print(f"Phase 6: Try different n (current={n_curr})")

        # Try 2x
        if n_curr <= 8192:
            target_n = n_curr * 2
            if time.monotonic() < phase6_end - 60:
                p_interp = _interp_profile(best_seq, target_n)
                sc_interp = _score_norm_fn(p_interp)
                print(f"  Interp to n={target_n}: score={sc_interp:.10f}")
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

                if time.monotonic() < phase6_end - 30:
                    dl = min(phase6_end - 25, time.monotonic() + 40.0)
                    a_best, sc_best = _adam_continuation(p_interp, dl, rng, lr_base=0.02, k_top=3)
                    update_best(a_best, sc_best, f"adam_n{target_n}")

        # Try 0.5x
        if n_curr >= 8192:
            target_n = n_curr // 2
            if time.monotonic() < phase6_end - 30:
                p_interp = _interp_profile(best_seq, target_n)
                sc_interp = _score_norm_fn(p_interp)
                print(f"  Interp to n={target_n}: score={sc_interp:.10f}")
                update_best(p_interp, sc_interp, f"interp_n{target_n}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final intensive polishing (remaining ~300s) ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")

        # 7a: Very fine Adam
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _adam_continuation(best_seq, dl, rng, lr_base=0.008, k_top=2)
            update_best(a_best, sc_best, "final_adam_fine")

        # 7b: Peak local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _peak_local_search(best_seq, dl, rng, n_trials=60000)
            update_best(a_best, sc_best, "final_peak")

        # 7c: Another Adam pass
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _adam_continuation(best_seq, dl, rng, lr_base=0.005, k_top=2)
            update_best(a_best, sc_best, "final_adam2")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = float(2.0 * len(final_seq) * _full_convolve(final_seq, final_seq).max() / (s * s))
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite: {sc_eval}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular_weight(8192)
    sc_fb = float(2.0 * len(fallback) * _full_convolve(fallback, fallback).max())
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
