"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key differences from previous approach:
1. Geometric ramp initializations with optimized ratio
2. Direct peak-shaving via subgradient on top peaks

4. Adaptive peak gradient descent with annealing
5. Combinatorial pair-swap surgery on peak contributors

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _geometric_ramp(n, r):
    a = np.power(r, np.arange(n, dtype=np.float64))
    return a / a.sum()


def _power_ramp(n, p):
    t = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(t, p)
    return a / a.sum()


def _make_initial_profiles(n):
    profiles = []
    for r in [1.001, 1.002, 1.005, 1.01, 1.02, 1.05]:
        profiles.append(_geometric_ramp(n, r))
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_power_ramp(n, p))
    t = np.linspace(0, 1, n)
    for slope in [1.0, 2.0, 3.0]:
        a = 1.0 + slope * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _peak_gradient(a, c, T, top_k=5):
    n = len(a)
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros(2 * n - 1)
    for k in idx:
        excess = c[k] - T
        if excess > 0:
            w[k] = 2.0 * excess
    if np.all(w == 0):
        return np.zeros(n)
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad_a = conv_aw[n - 1:2 * n - 1][::-1]
    return grad_a


def _run_peak_shave(a_init, deadline, rng, top_k=5, lr_init=0.01, max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    T = None
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_count += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        if T is None:
            T = cm * 0.999
        else:
            T = max(cm * 0.9999, T * 0.99999)

        grad = _peak_gradient(a, c, T, top_k)
        offset = float(np.dot(a, grad))
        grad_z = a * (grad - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)

        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60, 60)
        z_new = z_new - z_new.mean()
        a_new = _softmax_fn(z_new)

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:

            best_a = a_new.copy()
            no_improve = 0
            z = z_new.copy()
        else:
            no_improve += 1
            if no_improve > 5000:
                z = z + 0.02 * rng.normal(size=n)
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                no_improve = 0

        if t_count % 5000 == 0:
            lr = max(0.001, lr * 0.85)

    return best_a, best_score

        a8192 = _upsample(best_seq, n8192)
        sc8192 = _score_norm(a8192)
        print(f"  Upsampled to n=8192: {sc8192:.10f}")

        if sc8192 < best_score * 1.001:
            # Quick peak shave at 8192
            shave_dl = min(deadline - 10, time.monotonic() + 120.0)
            a8192_opt, sc8192_opt = _run_peak_shave(a8192, shave_dl, rng, top_k=5, lr_init=0.03)
            _update_best(a8192_opt, sc8192_opt, "n8192_shave")

    # Phase 6: Final polish
    print(f"\nPhase 6: Final polish")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_tiny = max(1e-7 * max_c_final, 1e-12)

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxls': 30, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    _update_best(a_new, sc_new, "final_lbfgs")
        except Exception:
            pass

        # One more peak shave round
        if time.monotonic() < deadline - 5:
            shave_dl2 = min(deadline - 3, time.monotonic() + 30.0)
            a_final, sc_final = _run_peak_shave(best_seq, shave_dl2, rng, top_k=3, lr_init=0.005)
            _update_best(a_final, sc_final, "final_shave")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _geometric_ramp(256, 1.01)

        sc_eval = _score_full(final_seq)
        print(f"\nFINAL: score={sc_eval:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _geometric_ramp(256, 1.01)
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
