"""
Multi-phase optimizer: L-BFGS beta-continuation + coordinate descent polish
for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses structured multi-scale optimization with golden-section coordinate descent.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unit(a):
    """Score for unit-sum sequence: 2*n*max_c."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_weights(c, beta):
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = e.sum()

        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

    inv_phi2 = inv_phi * inv_phi

    def obj_and_grad(z):
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        a = e / e.sum()
        c = _full_convolve(a, a)
        c_max = c.max()
        log_sum = c_max / beta + np.log(np.sum(np.exp((c - c_max) / beta)))
        val = beta * log_sum
        w = np.exp((c - c_max) / beta)
        w = w / w.sum()
        grad_a = _grad_smoothed(a, c, w, beta)
        grad_a_centered = grad_a - np.mean(grad_a)
        grad_z = a * grad_a_centered
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
        return val, grad_z

    try:
        res = minimize(obj_and_grad, z, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': 100000, ' iprint': 0, 'maxls': 50})
        z_opt = res.x
    except Exception:
        z_opt = z

    z_opt = z_opt - z_opt.max()
    e = np.exp(z_opt)
    a_opt = e / e.sum()
    return a_opt


def _cd_polish(a, deadline, verbose=True):
    n = len(a)
    a = a.copy()
    c = _full_convolve(a, a)
    best_max = float(c.max())
    sweep = 0
    no_improve = 0
    vals_buf = np.empty(n)

    while time.monotonic() < deadline - 2.0:
        sweep += 1
        improved = 0

        for i in range(n):
            if time.monotonic() > deadline - 3.0:
                break
            x = a[i]
            if x < 1e-14:
                continue

            C = c[i:i + n] - 2.0 * x * a
            if 2 * i < 2 * n - 1:
                C[i] = c[2 * i] - x * x

            x_upper = max(2.0 * x, 1.3 * float(np.max(a)))

            def _fcd(x_new):
                np.multiply(a, 2.0 * x_new, out=vals_buf)
                vals_buf += C
                if 2 * i < 2 * n - 1:
                    vals_buf[i] = C[i] + x_new * x_new
                return float(vals_buf.max())

            lo, hi = 0.0, x_upper
            c1 = hi - inv_phi * (hi - lo)
            c2 = lo + inv_phi * (hi - lo)
            f1 = _fcd(c1)
            f2 = _fcd(c2)

            for _ in range(35):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - inv_phi * (hi - lo)
                    f1 = _fcd(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + inv_phi * (hi - lo)
                    f2 = _fcd(c2)

            x_new = (lo + hi) * 0.5
            f_new = _fcd(x_new)
            f_old = _fcd(x)

            if f_new < f_old - 1e-14:
                delta = 2.0 * (x_new - x) * a
                if 2 * i < 2 * n - 1:
                    delta[i] -= (x_new - x) * (x_new - x)
                a[i] = x_new
                c[i:i + n] += delta
                improved += 1

        current_max = float(c.max())
        if current_max < best_max - 1e-15:
            best_max = current_max
            if verbose and (sweep <= 5 or sweep % 20 == 0):
                print(f"    CD sweep {sweep}: max_c={best_max:.12f}, score={_score(a):.10f}")
        if improved == 0:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

    return a


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()


def _make_initial(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triangular":
        t = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * t - 1.0)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    global inv_phi
    inv_phi = (np.sqrt(5.0) - 1.0) / 2.0
    inv_phi2 = inv_phi * inv_phi

    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        best_a = parent.copy()
        best_n = len(best_a)
        best_score = _score(best_a)
        print(f"Parent loaded: n={best_n}, score={best_score:.10f}")
    else:
        best_n = 4096
        best_a = _make_initial(best_n, "uniform")
        best_score = _score(best_a)
        print(f"No parent, uniform n={best_n}, score={best_score:.10f}")

    # === Phase 1: L-BFGS beta continuation on original length ===
    print(f"\n=== Phase 1: L-BFGS beta continuation (n={best_n}) ===")
    phase1_end = t_start + 250.0
    a_work = best_a.copy()
    c_work = _full_convolve(a_work, a_work)
    max_c_work = float(c_work.max())

    beta = 0.1 * max_c_work
    min_beta = 0.00005 * max_c_work
    prev_score = _score_unit(a_work)

    while beta > min_beta and time.monotonic() < phase1_end - 10:
        a_work = _lbfgs_logspace(a_work, beta, time.monotonic() + 120, maxiter=1500)
        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_a = a_work.copy()
            print(f"  beta={beta:.8f}: score={sc:.10f} (t={time.monotonic()-t_start:.0f}s)")
        c_work = _full_convolve(a_work, a_work)
        max_c_work = float(c_work.max())
        beta = max(min_beta, beta * 0.4)

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Try n=8192 ===
    print(f"\n=== Phase 2: L-BFGS on n=8192 ===")
    phase2_end = t_start + 500.0
    n_8192 = 8192
    a_8192 = _interp_profile(a_work, n_8192)
    sc_8192 = _score(a_8192)
    print(f"  Interpolated n=8192: score={sc_8192:.10f}")

    if sc_8192 < best_score + 0.01:
        c_8192 = _full_convolve(a_8192, a_8192)
        max_c_8192 = float(c_8192.max())
        beta = 0.05 * max_c_8192
        min_beta = 0.00003 * max_c_8192

        while beta > min_beta and time.monotonic() < phase2_end - 15:
            a_8192 = _lbfgs_logspace(a_8192, beta, time.monotonic() + 100, maxiter=1000)
            sc = _score(a_8192)
            if sc < best_score:
                best_score = sc
                best_a = a_8192.copy()
                print(f"  n=8192 beta={beta:.8f}: score={sc:.10f} (t={time.monotonic()-t_start:.0f}s)")
            c_8192 = _full_convolve(a_8192, a_8192)
            max_c_8192 = float(c_8192.max())
            beta = max(min_beta, beta * 0.35)

    # Also try optimizing n=4096 further if it's still the best
    if len(best_a) == best_n and time.monotonic() < phase2_end - 20:
        a_4096 = best_a.copy()
        c_t = _full_convolve(a_4096, a_4096)
        max_ct = float(c_t.max())
        beta = 0.01 * max_ct
        min_beta = 0.00001 * max_ct
        while beta > min_beta and time.monotonic() < phase2_end - 10:
            a_4096 = _lbfgs_logspace(a_4096, beta, time.monotonic() + 80, maxiter=1000)
            sc = _score(a_4096)
            if sc < best_score:
                best_score = sc
                best_a = a_4096.copy()
                print(f"  n=4096 beta={beta:.8f}: score={sc:.10f}")
            beta *= 0.3

    print(f"  Phase 2 done: best={best_score:.10f}, n={len(best_a)}")

    # === Phase 3: Multi-restart diversification ===
    print(f"\n=== Phase 3: Multi-restart ===")
    phase3_end = t_start + 850.0
    if time.monotonic() < phase3_end - 30:
        n_restarts = 8
        per_restart = (phase3_end - time.monotonic() - 20) / n_restarts
        a_base = best_a.copy()

        for r in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break

            if r % 4 == 0:
                p_start = _perturb(a_base, 0.02 + 0.005 * r, rng)
            elif r % 4 == 1:
                sz = len(a_base)
                shift = int(rng.integers(1, max(2, sz // 512)))
                p_start = np.roll(a_base, shift).copy()
                p_start = p_start / p_start.sum()
            elif r % 4 == 2:
                p_start = _perturb(a_base, 0.04, rng)
            else:
                # Frequency-domain perturbation
                A = np.fft.rfft(a_base)
                freq = np.fft.rfftfreq(len(a_base))
                mask = (freq > 0) & (freq < len(a_base) / 4.0)
                A[mask] *= (1.0 + 0.03 * rng.standard_normal(int(mask.sum())))
                p_start = np.fft.irfft(A, len(a_base))
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            c_r = _full_convolve(p_start, p_start)
            max_cr = float(c_r.max())
            beta_r = 0.005 * max_cr

            p_opt = _lbfgs_logspace(p_start, beta_r, time.monotonic() + max(5.0, per_restart * 0.6),
                                    maxiter=500)
            beta_r2 = 0.001 * max_cr
            p_opt = _lbfgs_logspace(p_opt, beta_r2, time.monotonic() + max(5.0, per_restart * 0.35),
                                    maxiter=300)

            sc_r = _score(p_opt)
            if sc_r < best_score:
                best_score = sc_r
                best_a = p_opt.copy()
                print(f"  Restart {r}: score={sc_r:.10f} (t={time.monotonic()-t_start:.0f}s}")
            else:
                print(f"  Restart {r}: score={sc_r:.10f} (not improved)")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Coordinate descent polish ===
    print(f"\n=== Phase 4: Coordinate descent polish ===")
    phase4_end = deadline - 30.0
    a_cd = best_a.copy()
    a_cd = a_cd / a_cd.sum()

    # Split CD time: first half on current best, second half potentially retry
    cd_deadline = time.monotonic() + (phase4_end - time.monotonic()) * 0.7
    a_cd = _cd_polish(a_cd, cd_deadline, verbose=True)
    sc_cd = _score(a_cd)
    if sc_cd < best_score:
        best_score = sc_cd
        best_a = a_cd.copy()
        print(f"  After CD: score={sc_cd:.10f}")

    # === Phase 5: Final L-BFGS polish + validate ===
    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 10:
        a_pol = best_a.copy()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        beta_final = 0.0001 * max_cp

        a_pol = _lbfgs_logspace(a_pol, beta_final, time.monotonic() + 60, maxiter=500)
        sc_pol = _score(a_pol)
        if sc_pol < best_score:
            best_score = sc_pol
            best_a = a_pol.copy()
            print(f"  Final L-BFGS: score={sc_pol:.10f}")

        # One more CD pass if time allows
        if time.monotonic() < deadline - 15:
            a_cd2 = best_a.copy() / best_a.sum()
            a_cd2 = _cd_polish(a_cd2, deadline - 5.0, verbose=False)
            sc_cd2 = _score(a_cd2)
            if sc_cd2 < best_score:
                best_score = sc_cd2
                best_a = a_cd2.copy()
                print(f"  Final CD: score={sc_cd2:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s_final = float(final.sum())
    if s_final < 0.01:
        final = _make_initial(256, "uniform")
    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
