"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines subgradient, log-sum-exp L-BFGS, p-norm optimization,
and coordinate descent with symmetry exploitation.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0)

def _subgradient_step(a, lr, n_options=None):
    """One subgradient step for minimizing max(conv(a,a)) on the simplex."""
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    # Subgradient of g_{k*}(a) = sum_j a_j a_{k*-j}
    # d/da_i [sum_j a_j a_{k*-j}] = a_{k*-i} + a_{i-k*} (when valid)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            val += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            val += a[j2]
        grad[i] = val
    a_new = _project_simplex(a - lr * grad)
    return a_new, float(c.max())

def _subgradient_descent(a_init, deadline, n_iters=50000, lr_init=0.1):
    """Subgradient descent with decreasing step size."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = lr_init
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        a_new, cur_max = _subgradient_step(a, lr)
        sc = 2.0 * n * cur_max
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        # Decrease learning rate
        lr = lr_init / (1 + 0.01 * it)
    return best_a, best_score

def _pnorm_obj(z, p, n):
    """p-norm objective in log-parameterization."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    val = np.sum(c_pos ** p) ** (1.0 / p)
    return 2.0 * n * val

def _pnorm_grad(z, p, n):
    """Gradient of p-norm objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    S = np.sum(c_pos ** p)
    S_inv = S ** (1.0 / p - 1.0)
    # d/d a_i of (sum c_k^p)^{1/p} = S^{1/p - 1} * sum_k p * c_k^{p-1} * dc_k/da_i
    # dc_k/da_i = a_{k-i} + a_{i-k} (when valid)
    # = sum_k p * c_k^{p-1} * (a_{k-i} + a_{i-k})
    # Let w_k = p * c_k^{p-1} * S^{1/p - 1}
    # Then grad_a_i = sum_k w_k * (a_{k-i} + a_{i-k})
    # = (w * a_reversed shifted) evaluated at i
    # This is a convolution of w with a (reversed)!
    w = p * c_pos ** (p - 1) * S_inv
    # grad_a = conv(w, a_reversed)[n-1:2n-1]... let me be more careful
    # sum_k w_k * a_{k-i} is the convolution (w * a_reversed) at position (n-1-i) + something
    # Actually: sum_k w_k * a_{k-i} = (w shifted) convolved with a
    # Let me compute it directly using FFT
    a_rev = a[::-1].copy()
    # sum_k w_k * a_{k-i} for i in [0, n-1]:
    # This is the cross-correlation of w (length 2n-1) and a (length n)
    # = (w * a_rev)[n-1:2n-1+n-1]... 
    # Let me just compute it: for each i, sum_k w_k * a_{k-i} where 0<=k-i<n
    # = sum_j w_{j+i} * a_j where j = k-i, so k = j+i
    # This is the (i-th) element of the "anti-convolution"
    # Actually it's: (full_convolve(w, a_rev))[n-1:] gives us sum_k w_k a_{n-1-k+n-1-i}... 
    
    # Let me just do it with the convolution formula:
    # h_i = sum_k w_k * a_{k-i} (valid range)
    # This is the cross-correlation, which equals the convolution with a reversed:
    # h = _full_convolve(w, a_rev)
    # h[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-(k-j)] = sum_j w[j] * a[n-1-k+j]
    # We want h_i = sum_k w_k * a[k-i]
    # Let j = k-i: sum_j w[j+i] * a[j]
    # This is the (i-th) element of the "sliding dot product"
    # = _full_convolve(w, a)[i]... no
    # _full_convolve(w, a)[m] = sum_j w[j] * a[m-j]
    # We want sum_j w[j+i] * a[j] = sum_k w[k] * a[k-i] (letting k=j+i)
    # This is the cross-correlation at lag i
    # = _full_convolve(w, a)[n-1+i]... not quite
    # 
    # OK let me just use: h = np.correlate(w, a, mode='full')
    # np.correlate(w, a, 'full')[k] = sum_j w[j] * a[j+k - (n-1)]... 
    
    # I'll just compute it directly for clarity, using FFT:
    N_fft = _next_pow2(2 * n + 1)
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    # cross-correlation: sum_j w[j] * conj(a[j-i]) = sum_j w[j] * a[j-i] (real)
    # = ifft(fft(w) * conj(fft(a)))
    # But we need: sum_k w[k] * a[k-i]
    # = ifft(fft(w) * conj(fft(a)))[i]... let me verify
    # ifft(fft(w) * conj(fft(a)))[i] = sum_k w[k] * a[k-i] (circular)
    # For linear, we need N_fft large enough
    corr = np.fft.irfft(W * np.conj(A), N_fft)
    h = corr[:n]  # h[i] = sum_k w[k] * a[k-i] for i=0,...,n-1
    
    # Similarly, sum_k w_k * a_{i-k} = sum_k w_k * a[k-i] (by symmetry of the sum)
    # Wait: a_{i-k} = a_rev[k-i+... ]... 
    # sum_k w_k * a_{i-k}: let j = i-k, so k = i-j
    # = sum_j w[i-j] * a[j]
    # This is the cross-correlation at lag -i = the "other side"
    # = corr[n-1-i+n-1]... 
    # Actually: ifft(fft(w) * conj(fft(a)))[i] gives sum_k w[k] * a[(k-i) mod N]
    # For the second term sum_k w[k] * a[i-k], this is:
    # ifft(conj(fft(w)) * fft(a))[i] = conjugate of ifft(fft(w) * conj(fft(a)))[-i]
    # For real signals, this is just corr[-i mod N] = corr[N-i] (for i>0) or corr[0] (for i=0)
    
    # For real sequences: second term = h[n-1-i + ... ]
    # Actually, let me just compute both:
    # Term 1: h1[i] = sum_k w[k] * a[k-i] (k-i in [0,n-1])
    # Term 2: h2[i] = sum_k w[k] * a[i-k] (i-k in [0,n-1])
    # Note h2[i] = h1[i] if w and a are symmetric... in general they're different
    
    # h2[i] = sum_j w[i-j] * a[j] = cross-correlation at lag i (the other convention)
    # = corr2[i] where corr2 = ifft(conj(fft(w)) * fft(a))
    # = conj(corr[N-i]) for real signals = corr[-i] (for i in 0..n-1, -i maps to N-i)
    
    # For our case, since w and a are both real:
    # corr[i] = sum_k w[k] * a[(k-i) mod N]
    # For i in [0, n-1] and N > 2n-1, (k-i) mod N = k-i when k >= i, and N+k-i when k < i
    # Since w[k] = 0 for k >= 2n-1 and a[j] = 0 for j >= n:
    # corr[i] = sum_{k=i}^{2n-2} w[k] * a[k-i] (the N+k-i terms have a[N+k-i] = 0)
    # = sum_{k=i}^{2n-2} w[k] * a[k-i] = sum_j w[j+i] * a[j] (letting j = k-i)
    
    # So h1[i] = corr[i] for i in [0, n-1] ✓
    
    # For h2[i] = sum_k w[k] * a[i-k]:
    # = sum_{k=0}^{i} w[k] * a[i-k] (since a[i-k] = 0 for i-k < 0 or >= n)
    # This is the convolution (w * a)[i] = sum_k w[k] * a[i-k]
    # = _full_convolve(w, a)[i]
    
    # So: h2 = _full_convolve(w, a)[:n]
    # And: h1 = corr[:n] = ifft(fft(w)*conj(fft(a)))[:n]
    
    h2 = _full_convolve(w, a)[:n]
    # h1[i] = sum_k w[k]*a[k-i], from the corr computation
    # h1 = corr[:n]
    
    grad_a = 2.0 * (h1 + h2)  # Wait, is it 2*(h1+h2) or h1+h2?
    # dc_k/da_i = a[k-i] + a[i-k] (when both valid)
    # sum_k w_k * dc_k/da_i = sum_k w_k * a[k-i] + sum_k w_k * a[i-k] = h1[i] + h2[i]
    grad_a = h1 + h2
    
    # Now convert from grad_a to grad_z via softmax chain rule
    # a_i = exp(z_i) / sum_j exp(z_j)
    # da_i/dz_j = a_i * (delta_ij - a_j)
    # grad_z_j = sum_i grad_a_i * da_i/dz_j = a_j * (grad_a_j - sum_i grad_a_i * a_i)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)
    
    # Multiply by the outer factor: 2*n
    grad_z *= 2.0 * n
    
    return grad_z

def _pnorm_optimize(a_init, deadline, p_list=None):
    """Optimize using p-norm with continuation on p."""
    n = len(a_init)
    if p_list is None:
        p_list = [4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048]
    
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for p in p_list:
        if time.monotonic() > deadline:
            break
        remaining = len(p_list) - p_list.index(p)
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 60.0)
        dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 20))
        
        try:
            result = minimize(
                _pnorm_obj, z, args=(p, n),
                method='L-BFGS-B', jac=_pnorm_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001,
                 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        beta_list = [b * max_c0 for b in fracs]
    
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = len(beta_list) - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        dl = time.monotonic() + time_per
        maxiter = max(300, int(time_per * 40))
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
                    print(f"    LSE beta={beta:.2e}: score={best_score:.10f}")
        except Exception as e:
            print(f"    LSE beta={beta:.2e} failed: {e}")
    
    return best_a, best_score

def _coord_descent_fine(a_init, deadline, max_rounds=100):
    """Fine coordinate descent on the softmax parameterization."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    deltas = [-0.02, -0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01, 0.02]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        cur_score = _score_norm_fn(a)
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = cur_score
            for delta in deltas:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                cur_score = best_local
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = _softmax_fn(z).copy()
        if not improved:
            break
        if rnd % 10 == 0:
            print(f"    CD round {rnd}: score={best_score:.10f}")
    
    return best_a, best_score

def _symmetrize(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 0:
        a_sym /= s
    return a_sym

def _make_spectral_init(n, rng):
    """Initialize using spectral method (flat magnitude response)."""
    # Try to find a non-negative sequence with flat |DFT|^2
    # Using iterative projection: alternate between time-domain non-negativity and frequency-domain flatness
    best = None
    best_score = np.inf
    for trial in range(20):
        a = rng.dirichlet(np.ones(n))
        for it in range(200):
            # Project to flat spectrum (in the sense of equalizing |A|^2)
            N_fft = _next_pow2(2 * n - 1)
            A = np.fft.rfft(a, N_fft)
            mag = np.abs(A)
            mag_target = mag.mean()
            # Scale each frequency bin to have target magnitude
            phase = np.angle(A)
            A_new = mag_target * np.exp(1j * phase)
            # Ensure DC is correct: A_new[0] = sum(a) = 1
            A_new[0] = 1.0
            # Ensure symmetry for real signal
            A_new = (A_new + np.conj(A_new[::-1])) / 2 if len(A_new) > 1 else A_new
            # Actually for rfft, the symmetry is automatic
            a_new = np.fft.irfft(A_new, N_fft)[:n]
            # Project to simplex
            a_new = _project_simplex(a_new)
            if np.abs(a_new.sum() - 1) > 0.01:
                a_new = a_new / a_new.sum()
            a = a_new
            # Check convergence
            if trial == 0 and it % 50 == 0:
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best = a.copy()
        
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    
    return best if best is not None else np.ones(n) / n

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

"""
Multi-method optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines subgradient on simplex, p-norm L-BFGS continuation, log-sum-exp L-BFGS,
spectral initialization, symmetry exploitation, and fine coordinate descent.
Key innovations: direct simplex projection, p-norm with stable normalization,
adaptive restarts, and multi-scale n exploration.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(ind[cond]))
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def _subgradient_step_simplex(a, lr):
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    a_rev = a[::-1]
    # dc_k/da_i = a[k-i] + a[i-k] for valid indices
    # sum effect: h1[i] = sum_k w_k * a[k-i], h2[i] = sum_k w_k * a[i-k]
    # where w is one-hot at k_star
    # h1[i] = a[k_star - i] if 0 <= k_star - i < n else 0
    # h2[i] = a[i - k_star] if 0 <= i - k_star < n else 0
    for i in range(n):
        val = 0.0
        j1 = k_star - i
        if 0 <= j1 < n:
            val += a[j1]
        j2 = i - k_star
        if 0 <= j2 < n:
            val += a[j2]
        grad[i] = val
    a_new = _project_simplex(a - lr * grad)
    return a_new, float(c.max())


def _subgradient_descent(a_init, deadline, n_iters=100000, lr_init=0.05, decay=0.005):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    for it in range(n_iters):
        if time.monotonic() > deadline:

        lr = lr_init / (1.0 + decay * it)
        a_new, cur_max = _subgradient_step_simplex(a, lr)
        sc = 2.0 * n * cur_max
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def _pnorm_obj(z, p, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-300:
        return 0.0
    c_norm = c / c_max
    S = float(np.sum(c_norm ** p))
    if S < 1e-300:
        return 2.0 * n * c_max
    val = c_max * S ** (1.0 / p)
    return 2.0 * n * val


def _pnorm_grad(z, p, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if c_max < 1e-300:
        return np.zeros(n)
    c_norm = c / c_max
    S = float(np.sum(c_norm ** p))
    if S < 1e-300:
        return np.zeros(n)
    S_1p = S ** (1.0 / p - 1.0)
    w = c_norm ** (p - 1) * S_1p
    # grad_a_i = 2n * sum_k w_k * (a[k-i] + a[i-k])
    N_fft = _next_pow2(2 * n + 1)
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    h1 = np.fft.irfft(W * np.conj(A), N_fft)[:n]

    grad_a = 2.0 * n * (h1 + h2)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)

def _interp_profile(p0, n_new):

    profiles.append(a / n)
    for _ in range(5):
        a = rng.dirichlet(np.ones(n) * 0.3)
        profiles.append(a)
    return profiles


def _multi_start_optimize(profiles, deadline_per, n):
    best_a = None
    best_score = np.inf
    for i, p in enumerate(profiles):
        if time.monotonic() > deadline_per:
            break
        time_left = deadline_per - time.monotonic()
        if time_left < 30:
            break
        time_alloc = min(45.0, time_left / max(1, len(profiles) - i))
        dl = time.monotonic() + time_alloc
        p_norm = p / p.sum()
        # Quick subgradient
        a_sub, sc_sub = _subgradient_descent(p_norm, dl, n_iters=5000, lr_init=0.03, decay=0.01)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer combining subgradient, p-norm, log-sum-exp L-BFGS,
    spectral init, symmetry, and coordinate descent for minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_work = 4096

    if parent is not None:
        if len(parent) != n_work:
            parent = _interp_profile(parent, n_work)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = np.ones(n_work) / n_work
        update_best(parent, _score_norm_fn(parent), "uniform")

    # === Phase 1: Spectral + multi-start (250s) ===
    phase1_end = min(deadline, t_start + 250.0)
    print(f"\nPhase 1: Spectral init + multi-start (n={n_work})")

    # Spectral initialization
    a_spec = _spectral_init(n_work, rng, n_trials=5, n_iters=200)
    sc_spec = _score_norm_fn(a_spec)
    update_best(a_spec, sc_spec, "spectral")

    # Symmetrized spectral
    a_spec_sym = _symmetrize(a_spec)
    sc_spec_sym = _score_norm_fn(a_spec_sym)
    update_best(a_spec_sym, sc_spec_sym, "spectral_sym")

    # Multi-start from diverse profiles
    profiles = _make_profiles(n_work, rng)
    # Add parent and spectral to profiles
    profiles.append(parent)
    profiles.append(a_spec)
    profiles.append(a_spec_sym)

    if time.monotonic() < phase1_end - 30:
        a_ms, sc_ms = _multi_start_optimize(profiles, phase1_end, n_work)
        if a_ms is not None:
            update_best(a_ms, sc_ms, "multi_start")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Subgradient refinement (200s) ===
    phase2_end = min(deadline, t_start + 450.0)
    print(f"\nPhase 2: Subgradient refinement")

    a_sub, sc_sub = _subgradient_descent(best_seq.copy(), phase2_end,
                                          n_iters=200000, lr_init=0.03, decay=0.003)
    update_best(a_sub, sc_sub, "subgradient")

    a_sub_sym = _symmetrize(a_sub)
    sc_sub_sym = _score_norm_fn(a_sub_sym)
    update_best(a_sub_sym, sc_sub_sym, "subgradient_sym")

    if time.monotonic() < phase2_end - 20:
        a_sub2, sc_sub2 = _subgradient_descent(a_sub_sym, phase2_end,
                                                n_iters=100000, lr_init=0.02, decay=0.005)
        update_best(a_sub2, sc_sub2, "subgradient_sym2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: p-norm L-BFGS (400s) ===
    phase3_end = min(deadline, t_start + 850.0)
    print(f"\nPhase 3: p-norm L-BFGS (n={len(best_seq)})")

    p_list = [16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192]
    a_pn, sc_pn = _pnorm_optimize(best_seq.copy(), phase3_end, p_list=p_list)
    update_best(a_pn, sc_pn, "pnorm")

    # Symmetrized p-norm
    a_pn_sym = _symmetrize(a_pn)
    sc_pn_sym = _score_norm_fn(a_pn_sym)
    update_best(a_pn_sym, sc_pn_sym, "pnorm_sym")

    # Run p-norm on symmetrized
    if time.monotonic() < phase3_end - 60:
        a_pn2, sc_pn2 = _pnorm_optimize(a_pn_sym, min(phase3_end - 30, time.monotonic() + 120),
                                         p_list=[64, 128, 256, 512, 1024])
        update_best(a_pn2, sc_pn2, "pnorm_sym2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Log-sum-exp L-BFGS with ultra-fine beta (400s) ===
    phase4_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 4: Log-sum-exp L-BFGS")

    # Multiple restarts
    for r in range(5):
        if time.monotonic() > phase4_end - 60:
            break
        if r == 0:
            a_cur = best_seq.copy()
        elif r == 1:
            a_cur = _symmetrize(best_seq)
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.01 * (r - 1)
            z = z + noise * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
        time_left = phase4_end - 60 - time.monotonic()
        if time_left < 30:
            break
        time_alloc = min(60.0, time_left / (5 - r))
        dl = time.monotonic() + time_alloc
        a_lse, sc_lse = _lbfgs_continuation(a_cur, dl)
        update_best(a_lse, sc_lse, f"lse_r{r}")

    # Symmetrize and polish
    a_sym_best = _symmetrize(best_seq)
    sc_sym_best = _score_norm_fn(a_sym_best)
    update_best(a_sym_best, sc_sym_best, "sym_best")

    if time.monotonic() < phase4_end - 40:
        dl = min(phase4_end - 20, time.monotonic() + 80)
        a_lse_sym, sc_lse_sym = _lbfgs_continuation(a_sym_best, dl)
        update_best(a_lse_sym, sc_lse_sym, "lse_sym")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: n-value exploration (150s) ===
    phase5_end = min(deadline, t_start + 1400.0)
    print(f"\nPhase 5: n-value exploration")

    for n_try in [2048, 4096, 8192]:
        if time.monotonic() > phase5_end - 40:
            break
        if n_try == len(best_seq):
            continue
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")
        if time.monotonic() < phase5_end - 60:
            dl = min(phase5_end - 40, time.monotonic() + 50)
            a_opt, sc_opt = _lbfgs_continuation(p_interp, dl, max_beta_iters=8)
            update_best(a_opt, sc_opt, f"opt_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Fine coordinate descent (150s) ===
    phase6_end = min(deadline, t_start + 1550.0)
    print(f"\nPhase 6: Coordinate descent polish")

    if best_seq is not None and time.monotonic() < phase6_end:
        per_dl = min(phase6_end - 20, time.monotonic() + (phase6_end - time.monotonic()) * 0.8)
        a_cd, sc_cd = _coord_descent_fine(best_seq, per_dl, max_rounds=60)
        update_best(a_cd, sc_cd, "coord_descent")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final ultra-fine optimization (remaining) ===
    print(f"\nPhase 7: Final optimization")

    if best_seq is not None and time.monotonic() < deadline - 20:
        # Ultra-small beta
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_ultra = 5e-7 * max_c_final
        per_dl = min(deadline - 15, time.monotonic() + 50)
        a_final, sc_final = _lbfgs_continuation(
            best_seq, per_dl, beta_list=[beta_ultra, 2e-7 * max_c_final])
        update_best(a_final, sc_final, "ultra_final")

        # Symmetrized final
        a_sym_final = _symmetrize(best_seq)
        sc_sym_final = _score_norm_fn(a_sym_final)
        update_best(a_sym_final, sc_sym_final, "sym_final")

        # Final coord descent
        if time.monotonic() < deadline - 10:
            per_dl2 = min(deadline - 10, time.monotonic() + 30)
            a_cd2, sc_cd2 = _coord_descent_fine(best_seq, per_dl2, max_rounds=30)
            update_best(a_cd2, sc_cd2, "cd_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        return [float(x) for x in final_seq]

    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
