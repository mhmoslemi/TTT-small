"""
Multi-scale optimization with Nesterov momentum and adaptive peak equalization.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Multi-scale L-BFGS with log-sum-exp continuation (scales 128→4096)
2. Nesterov accelerated subgradient descent in the low-temperature regime
3. Adaptive peak equalization via coordinated moves
4. Final coordinate descent polish
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _lse_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _lse_grad(z, beta):
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # Compute h = a * w (circular-like, but we need the "centered" part)
    # h_i = sum_k w[k] * a[k-i] for valid indices
    # This is the cross-correlation of w and a, extracted at the right offset
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _subgradient_at_max(a):
    """Compute subgradient of max_k conv(a,a)_k w.r.t. a, in z-space (softmax param)."""
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    # Subgradient of conv_{k*} w.r.t. a_j
    # For j where k_star - j is in [0, n-1]:
    #   if 2*j != k_star: contrib = 2 * a[k_star - j]
    #   if 2*j == k_star: contrib = a[k_star - j]
    grad_a = np.zeros(n)
    valid_j = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
    for j in valid_j:
        kj = k_star - j
        if 0 <= kj < n:
            if 2 * j == k_star:
                grad_a[j] = a[kj]
            else:
                grad_a[j] = 2.0 * a[kj]
    # Convert to z-space gradient (softmax Jacobian)
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(30, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.02 * cm:
        return max(1e-10, 0.02 * cm)
    elif gap > 0.4 * cm:
        return max(1e-10, 0.005 * cm)
    else:
        return max(1e-10, 0.2 * gap)


def _upsample_cubic(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    # Use scipy's interpolation if available, else fall back to linear
    try:
        from scipy.interpolate import CubicSpline
        x0 = np.linspace(0.0, 1.0, n)
        x1 = np.linspace(0.0, 1.0, target_n)
        cs = CubicSpline(x0, a, bc_type='not-a-knot')
        new_a = cs(x1)
    except Exception:
        x0 = np.linspace(0.0, 1.0, n)
        x1 = np.linspace(0.0, 1.0, target_n)
        new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.3 * max_c0
    beta_end = max(1e-7 * max_c0, 1e-12)
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = max(1, len(betas) - idx)
        time_budget = (deadline - time.monotonic()) / remaining
        time_budget = min(time_budget, 200.0)
        level_deadline = time.monotonic() + time_budget

        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 5000)
        maxfun = maxiter * 5

        try:
            def obj_wrapper(z_w):
                return _lse_obj(z_w, beta)

            def jac_wrapper(z_w):
                return _lse_grad(z_w, beta)

            result = minimize(
                obj_wrapper, z.copy(),
                method='L-BFGS-B', jac=jac_wrapper,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': maxfun,
                         'iprint': -1}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -100, 100)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_nesterov(a_init, deadline, rng, lr_init=0.05, momentum=0.9):
    """Nesterov accelerated gradient descent on z-space using LSE approximation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    v = np.zeros(n)
    t = 0
    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)
    lr = lr_init
    patience = 0

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        if t % 50 == 0:
            beta = _adaptive_beta(c)
            # Decay learning rate
            lr = max(0.001, lr_init * 0.9995 ** t)

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        # Nesterov: lookahead step
        z_look = z - momentum * v
        a_look = _softmax_fn(z_look)
        c_look = _full_convolve(a_look, a_look)
        cm_look = float(c_look.max())
        if cm_look > 1e-30:
            shifted_look = (c_look - cm_look) / beta
            e_look = np.exp(shifted_look)
            w_look = e_look / e_look.sum()
            a_look_rev = a_look[::-1]
            w_conv_look = _full_convolve(a_look_rev, w_look)
            h_look = w_conv_look[n - 1:2 * n - 1]
            grad_a_look = 2.0 * h_look
            offset_look = float(np.dot(a_look, grad_a_look))
            grad_z_look = a_look * (grad_a_look - offset_look)
            if np.all(np.isfinite(grad_z_look)):
                grad_z = grad_z_look

        # Gradient norm clipping
        gnorm = np.linalg.norm(grad_z)
        if gnorm > 10.0:
            grad_z = grad_z * (10.0 / gnorm)

        v = momentum * v + lr * grad_z
        z = z - v
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % 20 == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1

            if patience > 50:
                # Reset with perturbation
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z = z + 0.05 * rng.normal(size=n)
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                v = np.zeros(n)
                patience = 0

    return best_a, best_score


def _peak_equalize(a, deadline, rng, max_rounds=500):
    """Targeted peak reduction by adjusting top contributors to the convolution max."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top peaks
        top_k = min(10, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices[:5]:
            # Find contributors to this peak
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:10]

            for idx_pos in order[:5]:
                idx = valid_i[idx_pos]
                # Try reducing this element
                for delta in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

            # Also try increasing small elements (to spread energy)
            if not improved and rnd % 3 == 0:
                small_idx = np.argsort(a_cur)[:3]
                for idx in small_idx:
                    for delta in [0.005, 0.01, 0.02]:
                        z_trial = z.copy()
                        z_trial[idx] += delta
                        a_trial = _softmax_fn(z_trial)
                        sc = _score_norm_fn(a_trial)
                        if sc < best_score - 1e-13:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            improved = True
                            break
                    if improved:
                        break

            if improved:
                break

        if not improved and rnd > 50:
            # Try a moderate perturbation and reassess
            if rnd % 20 == 0:
                z = z + 0.03 * rng.normal(size=n)
                z = np.clip(z, -80, 80)
                z = z - z.mean()

        if not improved and rnd > 200:
            break

    return best_a, best_score


def _coordinate_descent(a, deadline, rng):
    """1D coordinate descent on z-space, trying to improve by optimizing one coordinate at a time."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    # Get peaks to focus on
    c = _full_convolve(a, a)
    top_peaks = np.argsort(c)[::-1][:5]

    # Build list of important indices (those contributing to top peaks)
    important = set()
    for pk in top_peaks:
        valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
        if len(valid_i) > 0:
            prod = a[valid_i] * a[pk - valid_i]
            top_idx = valid_i[np.argsort(prod)[::-1][:5]]
            important.update(top_idx.tolist())

    if not important:
        important = set(range(n))

    # Shuffle for randomization
    idx_list = list(important)
    rng.shuffle(idx_list)

    stall_count = 0
    for idx in idx_list:
        if time.monotonic() > deadline:
            break
        if stall_count > 100:
            break

        # Try a range of values for z[idx]
        best_delta = 0.0
        best_sc = best_score
        for delta in np.linspace(-0.3, 0.3, 21):
            if delta == 0:
                continue
            z_trial = z.copy()
            z_trial[idx] += delta
            if z_trial[idx] > 80 or z_trial[idx] < -80:
                continue
            a_trial = _softmax_fn(z_trial)
            sc = _score_norm_fn(a_trial)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_delta = delta

        if best_delta != 0.0:
            z[idx] += best_delta
            a_check = _softmax_fn(z)
            sc_check = _score_norm_fn(a_check)
            if sc_check < best_score:
                best_score = sc_check
                best_a = a_check.copy()
                stall_count = 0
            else:
                z[idx] -= best_delta
                stall_count += 1
        else:
            stall_count += 1

    return best_a, best_score


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 4.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Sigmoid-like
    for center in [0.3, 0.5, 0.7]:
        for width in [0.2, 0.4]:
            a = 1.0 / (1.0 + np.exp(-10 * (t - center)))
            a += 0.1
            a = np.maximum(a, 1e-12)
            profiles.append(a / a.sum())
    return profiles


def _run_final_lbfgs(a_init, deadline, rng, n_betas=20):
    """Extended L-BFGS with very fine continuation for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.1 * max_c0
    beta_end = max(1e-8 * max_c0, 1e-14)
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = max(1, len(betas) - idx)
        time_budget = (deadline - time.monotonic()) / remaining
        time_budget = min(time_budget, 150.0)
        level_deadline = time.monotonic() + time_budget

        maxiter = 200 if beta > 0.05 * max_c0 else (800 if beta > 0.005 * max_c0 else 3000)
        maxfun = maxiter * 5

        try:
            def obj_w(z_w):
                return _lse_obj(z_w, beta)

            def jac_w(z_w):
                return _lse_grad(z_w, beta)

            result = minimize(
                obj_w, z.copy(),
                method='L-BFGS-B', jac=jac_w,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': maxfun, 'iprint': -1}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -100, 100)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent loaded: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(128, 2.0)
        update_best(a0, _score_norm_fn(a0), "default_ramp")

    # --- Phase 1: Multi-scale optimization (0-50% of time) ---
    phase1_end = t_start + (budget_s - 20.0) * 0.50
    print(f"\n=== Phase 1: Multi-scale (t=0 to {phase1_end - t_start:.0f}s) ===")

    # Start from best profile at n=64
    n_start = 64
    profiles = _make_profiles(n_start)
    a_cur = None
    sc_cur = np.inf
    for prof in profiles:
        sc = _score_norm_fn(prof)
        if sc < sc_cur:
            sc_cur = sc
            a_cur = prof.copy()
    print(f"  Best profile at n={n_start}: {sc_cur:.10f}")

    # Quick L-BFGS at n=64
    quick_dl = min(phase1_end - 5, t_start + 10.0)
    a_cur, sc_cur = _run_lbfgs_continuation(a_cur, quick_dl, rng, n_betas=6)
    print(f"  After n=64 opt: {sc_cur:.10f}")

    # If parent is available and at a higher scale, we'll also run from parent
    # Multi-scale up
    scales = [128, 256, 512, 1024, 2048, 4096]
    time_per_scale = {128: 15, 256: 25, 512: 40, 1024: 70, 2048: 130, 4096: 250}

    for target_n in scales:
        if time.monotonic() > phase1_end - 10:
            break

        a_target = _upsample_cubic(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)
        alloc = time_per_scale.get(target_n, 30)
        opt_dl = min(phase1_end - 5, time.monotonic() + float(alloc))
        n_beta = 10 if target_n <= 512 else 14
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_dl, rng, n_betas=n_beta)
        print(f"  n={target_n}: upsample={sc_target:.8f} -> opt={sc_opt:.8f}")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")
    else:
        a_cur = best_seq.copy()
        sc_cur = best_score

    print(f"  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Extended L-BFGS at full resolution (50-70%) ---
    phase2_end = t_start + (budget_s - 20.0) * 0.70
    print(f"\n=== Phase 2: Fine L-BFGS (t={phase2_end - t_start:.0f}s to {phase2_end - t_start:.0f}s) ===")

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_best = len(best_seq)
        p2_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        a_p2, sc_p2 = _run_final_lbfgs(best_seq, p2_dl, rng, n_betas=25)
        update_best(a_p2, sc_p2, "phase2_lbfgs")

        # Perturbed restart
        if time.monotonic() < phase2_end - 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.08 * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            p2_dl2 = min(phase2_end - 5, time.monotonic() + 60.0)
            a_p2b, sc_p2b = _run_final_lbfgs(a_pert, p2_dl2, rng, n_betas=12)

    phase3_end = t_start + (budget_s - 20.0) * 0.90

        a_nest, sc_nest = _run_nesterov(best_seq, nest_dl, rng, lr_init=0.03, momentum=0.92)

        if time.monotonic() < phase3_end - 10:
            pe_dl = min(phase3_end - 5, time.monotonic() + 50.0)

            update_best(a_pe, sc_pe, "peak_eq")

            a_nest2, sc_nest2 = _run_nesterov(best_seq, nest_dl2, rng, lr_init=0.01, momentum=0.95)

    print(f"  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Coordinate Descent + Final L-BFGS (90-98%) ---
    phase4_end = t_start + (budget_s - 20.0) * 0.98
    print(f"\n=== Phase 4: Coord Descent + Final (t={phase4_end - t_start:.0f}s) ===")

    if time.monotonic() < phase4_end - 5 and best_seq is not None:
        # Coordinate descent
        cd_dl = min(phase4_end - 20, time.monotonic() + 30.0)
        a_cd, sc_cd = _coordinate_descent(best_seq, cd_dl, rng)
        update_best(a_cd, sc_cd, "coord_descent")

        # Final tiny-beta L-BFGS
        if time.monotonic() < phase4_end - 3:
            n_fin = len(best_seq)
            c_fin = _full_convolve(best_seq, best_seq)
            max_c_fin = float(c_fin.max())
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            beta_tiny = max(1e-6 * max_c_fin, 1e-15)

            try:
                def obj_fin(z_w):
                    return _lse_obj(z_w, beta_tiny)

                def jac_fin(z_w):
                    return _lse_grad(z_w, beta_tiny)

                result = minimize(
                    obj_fin, z.copy(),
                    method='L-BFGS-B', jac=jac_fin,
                    options={'maxiter': 20000, 'ftol': 1e-18, 'gtol': 1e-13,
                             'maxls': 50, 'maxfun': 50000, 'iprint': -1}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(np.clip(result.x, -100, 100))
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_seq = a_new.copy()
                        print(f"  Final L-BFGS: {best_score:.10f}")
            except Exception:
                pass

    print(f"  Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Final validation ---
    print(f"\n=== Final Validation ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
