import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    """Full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    """Compute the score for a sequence (sum can be anything)."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    """Score assuming sum=1: 2*n*max(conv)."""
    n = len(a)

FISTA-based optimizer with logsumexp continuation, symmetry exploitation,
multiplicative weight polishing, and adaptive beta scheduling.
Exploits the fact that a symmetric optimal exists to halve the problem dimension.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _make_symmetric_full(a_half):
    n = len(a_half)
    if n % 2 == 1:
        full = np.concatenate([a_half, a_half[-2::-1]])
    else:
        full = np.concatenate([a_half, a_half[::-1]])
    return full


def _half_from_full(a_full):

def _softmax_fn(z):

    s = e.sum()

    shifted = (c - cm) * beta
    e = np.exp(shifted)
    s = e.sum()

    return cm + np.log(s) / beta, w

def _make_profile_sin(n, p=1.0):
    """Sinusoidal profile raised to power p."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_power(n, p=1.0):
    """(1-u^2)^p profile where u = 2i/(n-1)-1."""
    i = np.arange(n, dtype=np.float64)
    u = 2.0 * i / (n - 1) - 1.0
    a = (1.0 - u * u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_tri(n):
    """Triangular: i*(n-1-i)."""
    i = np.arange(n, dtype=np.float64)

def _interp_to_n(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _multiplicative_polish(a_full, deadline, max_iters=50000, eta_init=0.05):
    """Multiplicative weight updates to reduce max convolution."""
    n = len(a_full)
    a = a_full.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    eta = eta_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        c = _fft_convolve(a, a)
        M = c.max()
        
        # Find top K lags
        K = min(3, len(c))
        top_lags = np.argsort(c)[-K:]
        
        # Compute "pressure" at each position
        pressure = np.zeros(n)
        for k in top_lags:
            k = int(k)
            w_k = (c[k] - M + 1e-10) / (K * M)  # normalized weight
            # Actually use: for top lag, all contributing pairs get pressure
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                partner = k - j
                if 0 <= partner < n:
                    pressure[j] += w_k * 0.5
                    pressure[partner] += w_k * 0.5
        
        # Multiplicative update: decrease high pressure, increase low pressure
        # Use exponential: a <- a * exp(-eta * (pressure - mean(pressure)))
        p_mean = pressure.mean()
        update = np.exp(-eta * (pressure - p_mean))
        a = a * update
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if it % 100 == 0:
            sc = _score_norm_fn(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            else:
                eta *= 0.95
        
        # Decay eta
        if it % 2000 == 0:
            eta = max(eta * 0.9, 0.002)
    
    return best_a, best_sc

def _run_fista_with_schedule(a_full_init, deadline, beta_scale=1.0):
    """Run FISTA with adaptive beta schedule."""
    n_full = len(a_full_init)
    a_half = _half_from_full(a_full_init)
    n_h = len(a_half)
    
    z = np.log(np.maximum(a_half, 1e-15))
    z = z - z.mean()
    
    # Compute initial max conv to set beta scale
    c = _fft_convolve(a_full_init, a_full_init)
    M0 = c.max()
    
    # Beta schedule: from small (smooth) to large (sharp)
    n_levels = 12
    betas = []
    for i in range(n_levels):
        frac = i / max(1, n_levels - 1)
        # Exponential schedule
        b = beta_scale * M0 * (0.1 + 0.9 * (10.0 ** (frac * 4.0 - 1.0)))
        betas.append(b)
    # Remove duplicates and ensure increasing
    betas = sorted(set(betas))
    # Limit to reasonable range
    betas = [b for b in betas if b > 1e-6]
    if not betas:
        betas = [beta_scale * M0]
    
    return _fista_logsumexp(z, betas, deadline, step_scale=1.0)

                a_best, sc_best = _run_fista_with_schedule(p_large, dl, beta_scale=1.0)
                _update(a_best, sc_best, f"fista_n{target_n}")
    
    # Phase 7: Multi-restart with perturbation (200s)
    phase7_end = min(deadline, t_start + 1580.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_restarts = 3
        per_restart = max(20.0, (phase7_end - time.monotonic()) / max(1, n_restarts))
        print(f"Phase 7: Multi-restart, current best={best_score:.10f}")
        
        for ri in range(n_restarts):
            if time.monotonic() > phase7_end - 10:
                break
            noise_level = 0.01 + 0.01 * ri
            a_half = _half_from_full(best_seq)
            z = np.log(np.maximum(a_half, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=len(z))
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            a_h = _softmax_fn(z)
            a_f = _make_symmetric_full(a_h)
            a_f = a_f / a_f.sum()
            
            dl = min(phase7_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _run_fista_with_schedule(a_f, dl, beta_scale=0.8)
            _update(a_best, sc_best, f"restart{ri}")
    
    # Phase 8: Final intensive polishing (remaining time)
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(30.0, deadline - time.monotonic())
        print(f"Phase 8: Final polishing, remaining={remaining:.0f}s")
        
        # 8a: More multiplicative
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _multiplicative_polish(best_seq, dl, max_iters=30000, eta_init=0.015)
            _update(a_best, sc_best, "final_mux")
        
        # 8b: Direct descent
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _direct_max_descent(best_seq, dl, max_iters=20000)
            _update(a_best, sc_best, "final_direct")
        
        # 8c: Final FISTA
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _run_fista_with_schedule(best_seq, dl, beta_scale=0.3)
            _update(a_best, sc_best, "final_fista")
    
    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            fallback = _make_profile_tri(8192)
            final_seq = fallback
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        # Validate with evaluator
        if len(final_seq) <= 32768:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
                if np.isfinite(sc_eval) and sc_eval < sc_final:
                    sc_final = sc_eval
            except Exception as ex:
                print(f"Evaluator error: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_profile_tri(8192)
    print(f"FALLBACK: n=8192, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
