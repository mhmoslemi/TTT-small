Optimized LP + Subgradient + Nesterov-Adam hybrid for minimizing
2*n*max(conv(a,a))/sum(a)^2. Key improvements: golden-section line search,
adaptive LP bounds, subgradient fallback, and Nesterov-accelerated Adam in log-space.

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_core(a):
    """Returns 2*n*max(a*a) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v, s=1.0):
    """Project v onto simplex {x: sum(x)=s, x>=0}."""
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1)
    cond = v_sorted * idx > (cssv[-1] - s)
    if not np.any(cond):
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_lp_matrices(n):
    """Build sparse A matrix for LP: (f*b)_k = sum_i A[k,i] * b_i where A[k,i]=f_i if i in range."""
    # We'll build this on-the-fly since it depends on f
    return None


def _solve_lp(f, M, n):
    """Solve max 1'b s.t. (f*b)_k <= M for all k, b >= 0."""
    # Build sparse A: A[k, i] = f[i] for valid (k,i) pairs
    # (f*b)_k = sum_i f[i] * b[k-i] ... wait no
    # (f*b)_k = sum_i f[i] * b[k-i] where b is indexed 0..n-1
    # So A[k, j] = f[k-j] if 0 <= k-j < n, i.e., j in [k-n+1, k] and 0 <= j < n
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_lo > j_hi:
            continue
        js = np.arange(j_lo, j_hi + 1)
        # A[k, j] = f[k - j]
        vals = f[k - js]
        rows_list.append(np.full(len(js), k, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        vals_list.append(vals)
    
    if not rows_list:
        return None
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status != 0:
            return None
        x = np.maximum(res.x, 0.0)
        if x.sum() < 1e-15:
            return None
        return x / x.sum()
    except Exception:
        return None


def _golden_section_search(cf, cgm, cgg, max_iter=60):
    """
    Golden section search to minimize h(t) = max_k [(1-t)^2*cf[k] + 2t(1-t)*cgm[k] + t^2*cgg[k]]
    over t in [0, 1].
    cf = f*f, cgm = f*g (cross), cgg = g*g
    """
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    def h(t):
        val = (1.0 - t) ** 2 * cf + 2.0 * t * (1.0 - t) * cgm + t * t * cgg
        return float(np.max(val))
    
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = h(c)
    fd = h(d)
    
    for _ in range(max_iter):
        if b - a < 1e-12:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = h(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = h(d)
    
    t_opt = (a + b) / 2.0
    return t_opt, h(t_opt)


def _lp_iteration(a, n, use_adaptive_M=False):
    """One LP iteration: solve LP, line search, update."""
    c_ff = _full_convolve(a, a)
    M = float(c_ff.max())
    
    # Try with slightly reduced M for adaptive approach
    M_try = M * 0.999 if use_adaptive_M else M
    g = _solve_lp(a, M_try, n)
    
    if g is None:
        # Fallback: try with original M
        g = _solve_lp(a, M, n)
        if g is None:
            return a, M, 0.0
    
    # Compute line search components
    c_fg = _full_convolve(a, g)
    c_gg = _full_convolve(g, g)
    
    t_opt, h_val = _golden_section_search(c_ff, c_fg, c_gg)
    
    if t_opt <= 1e-10:
        return a, M, 0.0
    
    a_new = (1.0 - t_opt) * a + t_opt * g
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    
    return a_new, M, t_opt


def _subgradient_step(a, n, eta):
    """One projected subgradient step on the peak of a*a."""
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    
    # Subgradient of c_{k*} w.r.t. a_i:
    # g_i = a_{k*-i} if 0 <= k*-i < n, else 0; plus a_i if 2i == k*
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] += a[j]
        if 2 * i == k_star:
            g[i] += a[i]
    
    # Projected gradient step
    a_new = _proj_simplex(a - eta * g)
    return a_new


def _adam_nesterov(a_init, deadline, n):
    """Nesterov-accelerated Adam in log-space with softmax parameterization."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_core(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    v_buf_adj = np.zeros(n)  # For Nesterov lookahead
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.005
    beta_val = 0.01  # Temperature for softmax weighting
    
    # Nesterov momentum
    m_nest = np.zeros(n)
    gamma_nest = 0.9
    
    prev_best = best_score
    cool_count = 0
    
    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adjust beta based on peak concentration
        if t < 20000:
            beta_val = max(0.001, 0.01 * (0.9999 ** (t // 1000)))
        else:
            beta_val = max(0.0001, beta_val * 0.999)
        
        # Compute softmax-weighted gradient
        c_shifted = (c - c_max) / max(beta_val, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        
        #/projected gradient in z-space
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Nesterov: use momentum-predicted position
        m_nest = gamma_nest * m_nest + update
        update_nest = gamma_nest * m_nest
        
        # Clip update
        update_norm = np.linalg.norm(update_nest)
        if update_norm > 0.3:
            update_nest = update_nest * (0.3 / update_norm)
        
        z = z - update_nest
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if t % 500 == 0:
            sc = _score_core(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                cool_count = 0
            else:
                cool_count += 1
                if cool_count > 5:
                    lr = max(0.0005, lr * 0.9)
        
        # Adaptive LR
        if t % 5000 == 0:
            if best_score < prev_best * 0.9995:
                lr = min(0.02, lr * 1.05)
            else:
                lr = max(0.0005, lr * 0.95)
            prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_core(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if lp_count % 10 == 0:
            print(f"  LP[{lp_count}] score={sc:.10f} M={M:.10f} t_step={t_step:.4f} dt={dt:.1f}s stall={stall} elapsed={time.monotonic()-t_start:.0f}s")
        
        if stall > 50:
            print(f"  LP stalled at iter {lp_count}")
            break
        prev_sc = sc
    
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, elapsed={time.monotonic()-t_start:.0f}s")
    
    # Phase 2: Subgradient refinement (500 - 800s)
    phase2_end = min(deadline_total, t_start + 800.0)
    if time.monotonic() < phase2_end:
        a = best_a.copy()
        print(f"Phase 2: Subgradient refinement")
        eta = 0.01 / n  # Initial step size
        sg_count = 0
        sg_stall = 0
        prev_sc = _score_core(a)
        
        for it in range(500000):
            if time.monotonic() > phase2_end:
                break
            a = _subgradient_step(a, n, eta)
            sg_count += 1
            
            if sg_count % 200 == 0:
                sc = _score_core(a)
                if sc < prev_sc - 1e-8:
                    sg_stall = 0
                    # Slowly increase eta if making progress
                    eta = min(eta * 1.001, 0.05 / n)
                else:
                    sg_stall += 1
                    eta = max(eta * 0.99, 1e-6 / n)
                
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                
                if sg_count % 2000 == 0:
                    print(f"  SG[{sg_count}] score={sc:.10f} eta={eta:.2e} stall={sg_stall} elapsed={time.monotonic()-t_start:.0f}s")
                
                if sg_stall > 50:
                    break
                prev_sc = sc
    
    print(f"Phase 2 done: best={best_score:.10f}, elapsed={time.monotonic()-t_start:.0f}s")
    
    # Phase 3: Nesterov-Adam in log-space (800 - 1200s)
    phase3_end = min(deadline_total, t_start + 1200.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Nesterov-Adam polishing")
        a_polished, sc_polished = _adam_nesterov(best_a, phase3_end - 5, n)
        sc_full = _score_core(a_polished)
        print(f"  Adam done: score={sc_full:.10f}, elapsed={time.monotonic()-t_start:.0f}s")
        if sc_full < best_score:
            best_score = sc_full
            best_a = a_polished.copy()
    
    # Phase 4: LP again from best point (1200 - 1500s)
    phase4_end = min(deadline_total, t_start + 1500.0)
    if time.monotonic() < phase4_end:
        a = best_a.copy()
        print(f"Phase 4: LP refinement from best point")
        prev_sc = _score_core(a)
        stall = 0
        lp4_count = 0
        while time.monotonic() < phase4_end:
            a, M, t_step = _lp_iteration(a, n, use_adaptive_M=True)
            lp4_count += 1
            sc = _score_core(a)
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if lp4_count % 10 == 0:
                print(f"  LP4[{lp4_count}] score={sc:.10f} t={t_step:.4f} stall={stall} elapsed={time.monotonic()-t_start:.0f}s")
            if stall > 40:
                break
            prev_sc = sc
        print(f"Phase 4 done: {lp4_count} iters, best={best_score:.10f}")
    
    # Phase 5: Try different length (1500 - 1650s)
    phase5_end = min(deadline_total, t_start + 1650.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Try different sequence lengths")
        for n_try in [1024, 4096]:
            if time.monotonic() > phase5_end - 60:
                break
            print(f"  Trying n={n_try}")
            a_try, sc_try = _try_different_n(best_a, n_try, phase5_end - 30)
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
                n = n_try
                print(f"  Improved! n={n_try}, score={sc_try:.10f}")
            else:
                print(f"  No improvement at n={n_try}: {sc_try:.10f}")
    
    # Phase 6: Final Adam polish (1650 - deadline)
    if time.monotonic() < deadline_total - 20:
        n_final = len(best_a)
        print(f"Phase 6: Final polish (n={n_final})")
        a_final, sc_final = _adam_nesterov(best_a, deadline_total - 10, n_final)
        sc_check = _score_core(a_final)
        if sc_check < best_score:
            best_score = sc_check
            best_a = a_final.copy()
        print(f"  Final Adam: score={sc_check:.10f}")
    
    # Final validation and return
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
        print("WARNING: sum too small, using fallback")
    
    sc_final = _score_core(final)
    print(f"FINAL: n={len(final)}, score_norm={sc_final:.10f}, time={time.monotonic()-t_start:.1f}s")
    
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    
    return [float(x) for x in final]
