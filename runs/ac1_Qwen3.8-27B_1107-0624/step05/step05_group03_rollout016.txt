import numpy as np
from scipy.optimize import minimize
import time

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max()

Combines FISTA with top-K smoothing, frequency-domain shaping (GS algorithm),

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _softmax_vec(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _make_triangular(n):

    a = (i + 1) * (n - i)
    return a / a.sum()

def _make_bspline(n, order=4):

def _gs_shaping(a_init, deadline, alpha=0.5, iters=5000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    N_fft = _next_pow2(2*n - 1)
    
    for it in range(iters):
        if time.monotonic() > deadline - 5:
            break
        
        # Forward transform
        A = np.fft.rfft(a, N_fft)
        power = np.abs(A)**2
        
        # Target: partially flattened power spectrum
        mean_power = power.mean()
        # Blend: power_new = power^alpha * mean_power^(1-alpha)
        # This reduces the dynamic range of the power spectrum
        power_new = np.power(power, alpha) * np.power(mean_power, 1.0 - alpha)
        
        # Update
        A_new = np.sqrt(power_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        
        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

# Pair-transfer local search
def _pair_local_search(a_init, deadline, n_trials=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    rng = np.random.default_rng(12345)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 2:
            break
        
        c = _fft_convolve(a, a)
        max_lag = int(np.argmax(c))
        
        # Find responsible positions for the max lag
        lo = max(0, max_lag - (n-1))
        hi = min(n-1, max_lag)
        
        if lo > hi:
            continue
        
        # Products at the max lag
        pairs = np.arange(lo, hi+1)
        partners = max_lag - pairs
        products = a[pairs] * a[partners]
        
        # Pick the most responsible position
        p = pairs[np.argmax(products)]
        
        # Find safe positions (low contribution to top lags)
        top_lags = np.argsort(c)[-3:]
        contrib = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            lo2 = max(0, kl - (n-1))
            hi2 = min(n-1, kl)
            for j in range(lo2, hi2+1):
                partner = kl - j
                if 0 <= partner < n:
                    contrib[j] += a[partner]
        
        # Pick from least contributing positions
        safe_mask = contrib < np.percentile(contrib, 25)
        safe_idx = np.where(safe_mask)[0]
        if len(safe_idx) == 0:
            safe_idx = np.arange(n)
        
        q = safe_idx[rng.integers(len(safe_idx))]
        
        if p == q:
            continue
        
        # Try several transfer fractions
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            delta = a[p] * frac
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[p] = a[p] - delta
            a_trial[q] = a[q] + delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            
            sc = _score_norm(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                break
            elif sc < _score_norm(a):
                # Accept even if not improving global best (hill climbing)
                a = a_trial.copy()
    
    return best_a, best_score

# L-BFGS with logsumexp
def _lbfgs_smooth(a_init, deadline, beta_scale=1.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z -= z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _fft_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    betas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * beta_scale for b in betas]
    
    per_level = max(5.0, (deadline - time.monotonic()) / len(betas))
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            a_ = _softmax_vec(z_)
            c_ = _fft_convolve(a_, a_)
            cm = float(c_.max())
            c_shifted = (c_ - cm) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300 or not np.isfinite(s_):
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s_))
        
        def grad(z_):
            a_ = _softmax_vec(z_)
            c_ = _fft_convolve(a_, a_)
            cm = float(c_.max())
            c_shifted = (c_ - cm) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            
            # dL/da_j = 2n * sum_k w_k * a_[k-j] (with bounds)
            grad_a = np.zeros(n)
            for k in range(len(c_)):
                if w[k] < 1e-15:
                    continue
                lo = max(0, k - (n-1))
                hi = min(n-1, k)
                grad_a[lo:hi+1] += w[k] * a_[k-hi:k-lo+1][::-1]
            grad_a *= 2.0 * n
            
            mean_g = np.dot(a_, grad_a)
            grad_z = a_ * (grad_a - mean_g)
            
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_)
            return grad_z
        
        try:
            result = minimize(
                obj, z.copy(), jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_vec(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score
