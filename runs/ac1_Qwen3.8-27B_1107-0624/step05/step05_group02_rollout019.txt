    out_n = n + m - 1
    N = next_fast_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_normalized(a):
    """Assume sum(a) = 1. Return 2*n*max(conv)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())

def _lse_objective_and_grad(z, beta):
    """
    Objective and gradient for the log-sum-exp smoothed objective.
    z is the log-parameterization, a = softmax(z).
    Objective: (1/beta) * log(sum_k exp(beta * c[k])) where c = conv(a,a)

    a = _softmax(z)
    c = _fft_convolve(a, a)

    c_max = c.max()
    log_sum = c_max + np.log(np.exp(beta * (c - c_max)).sum())

    w /= w.sum()

    # grad_a[m] = 2 * sum_j w[m+j] * a[j]

    # grad_a[m] = 2 * sum_j W[m+j] * a[j]
    # This is a cross-correlation. 
    # (W * a_rev)[k] = sum_j W[j] * a_rev[k-j] = sum_j W[j] * a[n-1-(k-j)] = sum_j W[j] * a[n-1-k+j]

    # = sum_l W[l] * a[l-m] (where l ranges from m to m+n-1)

    # (W * a_rev)[k] = sum_j W[j] * a_rev[k-j] = sum_j W[j] * a[n-1-k+j]

    # (softmax Jacobian)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_z

"""
Adaptive log-sum-exp gradient descent with L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import next_fast_len


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_fast_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score_normalized(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _lse_obj_grad(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    obj = (c_max + np.log(s)) / beta
    
    a_rev = a[::-1]
    Wa = _fft_convolve(w, a_rev)
    grad_a = 2.0 * Wa[n-1:2*n-1]
    
    grad_z = a * (grad_a - float(np.dot(a, grad_a)))
    return obj, grad_z


def _gradient_descent(a_init, deadline, beta_start=50.0, beta_end=2.0, lr=0.5, momentum=0.9):
    """Gradient descent in log-space with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    v = np.zeros(n)  # momentum velocity
    best_a = a_init.copy()
    best_score = _score_normalized(a_init)
    
    total_time = max(1.0, deadline - time.monotonic())
    t0 = time.monotonic()
    step = 0
    patience = 0
    max_patience = 5000
    
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        obj, grad = _lse_obj_grad(z, beta)
        
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -30, 30)
        z = z - z.mean()
        
        step += 1
        if step % 200 == 0:
            a = _softmax(z)
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            
            if step % 2000 == 0:
                print(f"    GD step {step}: score={sc:.10f}, best={best_score:.10f}, beta={beta:.2f}, t={elapsed:.1f}s")
            
            if patience > max_patience:
                # Restart momentum
                v = np.zeros(n)
                patience = 0
                lr = max(0.01, lr * 0.7)
    
    a_final = _softmax(z)
    sc_final = _score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _lbfgs_polish(a_init, time_limit=120, beta=10.0):
    """L-BFGS polishing in log-space."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    t_start = time.monotonic()
    
    def fun_grad(z):
        if time.monotonic() - t_start > time_limit:
            raise ValueError("Time limit exceeded")
        obj, grad = _lse_obj_grad(z, beta)
        return obj, grad
    
    try:
        res = minimize(fun_grad, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': 5000, 'maxfun': 10000,
                                'ftol': 1e-15, 'gtol': 1e-12})
        a_result = _softmax(res.x)
        sc = _score_normalized(a_result)
        return a_result, sc
    except Exception as e:
        print(f"    L-BFGS failed: {e}")
        return a_init, _score_normalized(a_init)


def _enforce_symmetry(a):
    """Average with reverse to enforce symmetry."""
    return (a + a[::-1]) / 2.0


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

"""
Adaptive log-sum-exp gradient descent with L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. Log-sum-exp smoothed objective with adaptive beta schedule
2. L-BFGS for precise local optimization (exploits smooth structure)
3. Symmetry enforcement (problem is invariant under reversal)
4. Multi-scale beta annealing for global-to-local convergence
5. Subgradient tracking for the argmax lag
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import next_fast_len


def _fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_fast_len(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_full(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max()) / (s * s)


def _lse_obj_grad(z, beta):
    """
    Log-sum-exp smoothed objective and gradient in log-space.
    z parameterizes a = softmax(z).
    Objective: (1/beta) * log(sum_k exp(beta * c[k]))
    """
    n = len(z)
    a = _softmax(z)
    c = _fft_convolve(a, a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    obj = (c_max + np.log(s)) / beta

    # Gradient w.r.t. a: grad_a[m] = 2 * sum_j w[m+j] * a[j]
    # Computed as 2 * (w * a_rev)[n-1+m]
    a_rev = a[::-1]
    Wa = _fft_convolve(w, a_rev)
    grad_a = 2.0 * Wa[n - 1:2 * n - 1]

    # Chain rule through softmax: grad_z[i] = a[i] * (grad_a[i] - sum(a*grad_a))
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)

    # Ensure orthogonality to all-ones (numerical safety)
    grad_z -= np.mean(grad_z)

    return obj, grad_z


def _gradient_descent(a_init, deadline, beta_start=100.0, beta_end=2.0,
                      lr=0.5, momentum=0.9, sym_period=500):
    """
    Gradient descent in log-space with adaptive beta and momentum.
    Periodically enforces symmetry.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    v = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score_normalized(a_init)

    t0 = time.monotonic()
    total_time = max(1.0, deadline - t0)
    step = 0
    patience = 0
    max_patience = 3000
    lr_schedule_factor = 1.0

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        beta = beta_start * (beta_end / beta_start) ** progress

        obj, grad = _lse_obj_grad(z, beta)

        # Adaptive learning rate: scale inversely with beta
        effective_lr = lr * lr_schedule_factor / max(1.0, beta / 10.0)
        effective_lr = min(effective_lr, 2.0)

        v = momentum * v - effective_lr * grad
        z = z + v
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        step += 1

        # Periodic evaluation
        if step % 100 == 0:
            a = _softmax(z)
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

            if step % 2000 == 0:
                print(f"    GD step {step}: score={sc:.10f}, best={best_score:.10f}, "
                      f"beta={beta:.2f}, lr={effective_lr:.4f}, t={elapsed:.1f}s")

            # Plateau: reduce lr and reset momentum
            if patience > max_patience:
                v *= 0.5
                lr_schedule_factor *= 0.7
                patience = 0
                if lr_schedule_factor < 0.05:
                    break

        # Symmetry enforcement periodically
        if sym_period > 0 and step % sym_period == 0:
            a_curr = _softmax(z)
            a_sym = (a_curr + a_curr[::-1]) / 2.0
            a_sym = np.maximum(a_sym, 1e-15)
            a_sym = a_sym / a_sym.sum()
            z = np.log(a_sym)
            z = z - z.mean()
            v = np.zeros(n)

    a_final = _softmax(z)
    sc_final = _score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_descent(a_init, deadline, lr=0.1, momentum=0.95):
    """
    Subgradient method: directly target the argmax lag.
    Faster per-step but less smooth than log-sum-exp.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score_normalized(a_init)

    t0 = time.monotonic()
    step = 0

    while time.monotonic() < deadline:
        a = _softmax(z)
        c = _fft_convolve(a, a)
        k_max = int(np.argmax(c))

        # Subgradient at k_max: grad_a[m] = 2*a[k_max-m] (if valid)
        grad_a = np.zeros(n)
        for m in range(n):
            j = k_max - m
            if 0 <= j < n:
                grad_a[m] = 2.0 * a[j]

        # Convert to z-space gradient
        mean_grad = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - mean_grad)
        grad_z -= np.mean(grad_z)

        # Scale learning rate
        gnorm = np.linalg.norm(grad_z)
        if gnorm > 1e-15:
            grad_z = grad_z / gnorm

        v = momentum * v - lr * grad_z
        z = z + v
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        step += 1
        if step % 200 == 0:
            a = _softmax(z)
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        if step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGD step {step}: score={sc:.10f}, best={best_score:.10f}, t={elapsed:.1f}s")
            lr *= 0.85

    return best_a, best_score


def _lbfgs_polish(a_init, time_limit=120, beta=10.0):
    """L-BFGS polishing in log-space using log-sum-exp objective."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    t_start = time.monotonic()
    call_count = [0]

    def fun_grad(z):
        call_count[0] += 1
        if time.monotonic() - t_start > time_limit:
            raise ValueError("Time limit exceeded")
        return _lse_obj_grad(z, beta)

    try:
        res = minimize(fun_grad, z0, method='L-BFGS-B', jac=True,
                       options={'maxiter': 10000, 'maxfun': 20000,
                                'ftol': 1e-18, 'gtol': 1e-14,
                                'maxls': 20})
        a_result = _softmax(res.x)
        a_result = np.maximum(a_result, 0)
        a_result = a_result / a_result.sum()
        sc = _score_normalized(a_result)
        print(f"    L-BFGS: {call_count[0]} evals, score={sc:.10f}, t={time.monotonic()-t_start:.1f}s")
        return a_result, sc
    except Exception as e:
        print(f"    L-BFGS failed: {e}")
        return a_init, _score_normalized(a_init)


def _enforce_symmetry(a):
    """Average with reverse to enforce reversal symmetry."""
    return (a + a[::-1]) / 2.0


def _interp_profile(p, n_new):
    """Interpolate profile to new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

            best_seq, best_score = _try_update(best_seq, best_score, a_r, t_start)

            # Quick L-BFGS
            if time.monotonic() < phase3_end - 50:
                tl = min(80.0, phase3_end - time.monotonic() - 30)
                if tl > 30:
                    a_r2, _ = _lbfgs_polish(a_r, time_limit=tl, beta=8.0)
                    a_r2 = _enforce_symmetry(a_r2)
                    a_r2 = a_r2 / a_r2.sum()
                    best_seq, best_score = _try_update(best_seq, best_score, a_r2, t_start)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Subgradient + larger n ===
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Subgradient + larger n ===")

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        a_best_norm = _enforce_symmetry(best_seq / best_seq.sum())

        # Subgradient method on current best
        if time.monotonic() < phase4_end - 120:
            sg_end = min(time.monotonic() + 120, phase4_end - 60)
            print(f"  Subgradient descent (120s)")
            a_sg, _ = _subgradient_descent(a_best_norm, sg_end, lr=0.05, momentum=0.95)
            a_sg = _enforce_symmetry(a_sg)
            a_sg = a_sg / a_sg.sum()
            best_seq, best_score = _try_update(best_seq, best_score, a_sg, t_start)

        # Try larger n
        if len(best_seq) < 16384 and time.monotonic() < phase4_end - 80:
            target_n = 16384
            print(f"  Interpolating to n={target_n}")
            p_interp = _interp_profile(best_seq, target_n)
            p_interp = _enforce_symmetry(p_interp)
            best_seq, best_score = _try_update(best_seq, best_score, p_interp, t_start)

            remaining = phase4_end - time.monotonic() - 60
            if remaining > 120:
                a_r, _ = _gradient_descent(p_interp, time.monotonic() + min(remaining - 30, 250),
                                           beta_start=50.0, beta_end=3.0,
                                           lr=0.3, momentum=0.9, sym_period=500)
                a_r = _enforce_symmetry(a_r)
                a_r = a_r / a_r.sum()
                best_seq, best_score = _try_update(best_seq, best_score, a_r, t_start)

                if time.monotonic() < phase4_end - 50:
                    tl = min(100.0, phase4_end - time.monotonic() - 30)
                    if tl > 40:
                        a_r2, _ = _lbfgs_polish(a_r, time_limit=tl, beta=8.0)
                        a_r2 = _enforce_symmetry(a_r2)
                        a_r2 = a_r2 / a_r2.sum()
                        best_seq, best_score = _try_update(best_seq, best_score, a_r2, t_start)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = _enforce_symmetry(best_seq / best_seq.sum())

        # Final L-BFGS with low beta
        remaining = deadline - time.monotonic() - 10
        if remaining > 60:
            tl = min(150.0, remaining - 5)
            print(f"  Final L-BFGS (beta=3, limit={tl:.0f}s)")
            a_final, _ = _lbfgs_polish(a_best_norm, time_limit=tl, beta=3.0)
            a_final = _enforce_symmetry(a_final)
            a_final = a_final / a_final.sum()
            best_seq, best_score = _try_update(best_seq, best_score, a_final, t_start)

            a_best_norm = a_final

        # Very short GD with tiny beta
        if time.monotonic() < deadline - 20:
            a_f2, _ = _gradient_descent(a_best_norm, deadline - 8,
                                        beta_start=3.0, beta_end=0.5,
                                        lr=0.1, momentum=0.95, sym_period=0)
            a_f2 = _enforce_symmetry(a_f2)
            a_f2 = a_f2 / a_f2.sum()
            best_seq, best_score = _try_update(best_seq, best_score, a_f2, t_start)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score_full(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
