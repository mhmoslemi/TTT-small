import time

    c = np.zeros(n + 1)

"""
Direct convex LP solver for minimum peak auto-correlation of non-negative sequences.
For each candidate length n, solves the linear program:
  min t
  s.t. (a*a)_k <= t for all k = 0,...,2n-2
       sum(a) = 1
       a >= 0
This gives the exact optimal for fixed n. We try multiple n values and pick the best score 2*n*t*.
"""
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp
import time


def _build_conv_matrix(n):
    """Build sparse matrix A where (A@a)_k = (a*a)_k for k=0,...,2n-2."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        count = j_end - j_start + 1
        rows_parts.append(np.full(count, k, dtype=np.int32))
        cols_parts.append(np.arange(j_start, j_end + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    data = np.ones(len(rows), dtype=np.float64)
    return sp.csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp(n, time_limit=600):
    """Solve min t s.t. (a*a)_k <= t, sum(a)=1, a>=0."""
    A_conv = _build_conv_matrix(n)

    # Variables: [a_0, ..., a_{n-1}, t], objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: (a*a)_k - t <= 0 for each k
    neg_I = -sp.identity(2 * n - 1, format='csc')
    A_ub = sp.hstack([A_conv, neg_I], format='csc')
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': float(time_limit), 'presolve': True,
                           'dual_feasibility_tolerance': 1e-10,
                           'primal_feasibility_tolerance': 1e-10})

    if res.status != 0:
        return None, None, res.status

    a = res.x[:n]
    t_val = float(res.x[n])
    return a, t_val, 0


def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1]
    return float(2.0 * n * c.max() / (s * s))


def _try_lp(n, time_limit, deadline):
    """Attempt to solve LP for given n, return (sequence, score) or (None, inf)."""
    if time.monotonic() > deadline - 10:
        return None, np.inf
    t0 = time.monotonic()
    a, t_val, status = _solve_lp(n, time_limit=time_limit)
    dt = time.monotonic() - t0
    if a is None:
        print(f"    n={n}: LP failed (status={status}), time={dt:.1f}s")
        return None, np.inf
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return None, np.inf
    a = a / s
    sc = _score(a)
    print(f"    n={n}: t*={t_val:.12f}, score={sc:.10f}, time={dt:.1f}s")
    return a, sc
