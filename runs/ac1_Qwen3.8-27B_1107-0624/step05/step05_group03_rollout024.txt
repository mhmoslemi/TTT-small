import numpy as np
from scipy.optimize import minimize, minimize_scalar
from scipy.fft import rfft, irfft
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Score assuming sum=1: 2n * max_conv."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    """Get the top K convolution values and their indices."""
    c = _full_convolve(a, a)
    idx = np.argpartition(c, -K)[-K:]
    idx = idx[np.argsort(c[idx])[::-1]]
    return c[idx], idx

    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(200000):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        max_c = c.max()
        
        # Find active lags
        active = np.where(c >= max_c * 0.999)[0]  # Within 0.1% of max
        if len(active) > K:
            active = np.argsort(c)[-K:]
        
        # Compute gradient of average active lag
        grad_a = np.zeros(n)
        for k in active:
            k = int(k)
            # d(c_k)/d(a[j]) = 2*a[k-j] for 0 <= k-j < n
            for j in range(0, min(k+1, n)):
                grad_a[k-j] += 2.0 * a[j]
        grad_a /= len(active)
        
        # Convert to z-gradient
        mean_grad = np.dot(a, grad_a)
        grad_z = a * (grad_a - mean_grad)
        grad_z *= 2.0 * n  # Scale for the score = 2n * max_c
        
        # Adam update
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        # Adaptive learning rate
        lr_t = lr * np.sqrt(1 - b2**t) / (1 - b1**t)
        lr_t = max(0.001, lr_t * (1.0 / (1 + 0.0001 * t)))
        
        z -= lr_t * mh / (np.sqrt(vh) + eps)
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        # Check every 10 iterations
        if it % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score

"""
Optimized step-function minimizer using exact active-set gradient methods,
log-sum-exp continuation with L-BFGS, and greedy peak-shaving local search.
Key innovations: (1) exact subgradient computation targeting the true max
convolution entries (no smoothing bias), (2) adaptive active-set size that
grows as optimization converges, (3) two-phase strategy: coarse gradient
descent followed by fine L-BFGS polishing, (4) greedy single-element moves
that directly reduce the identified peak convolution entry.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _active_set_grad(a, c, max_c, K, tol_frac):
    """Compute gradient as average of top-K (or all near-max) convolution entries.
    Returns (grad_a, num_active)."""
    n = len(a)
    threshold = max_c * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    if len(active) > K:
        # Keep top K
        active = np.argsort(c[active])[-K:] + np.where(c >= threshold)[0][np.argsort(c[active])[-K:]]
        # Simpler: just take top K from all
        active = np.argsort(c)[-K:]
    
    grad_a = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad_a[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    num_active = len(active)
    if num_active > 0:
        grad_a /= num_active
    return grad_a, num_active


def _make_profile_linear(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_bimodal(n, sep, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2.0
    c2 = 0.5 + sep / 2.0
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_w(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_power(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profile_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _adam_active_set(a_init, deadline, rng, K_init=5, K_grow=2, tol_init=0.005,
                     lr_init=0.015, max_iters=500000):
    """Softmax gradient descent with exact active set, adaptive K and learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    K = K_init
    tol = tol_init
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        max_c = float(c.max())
        
        # Adaptive: grow K and shrink tol as we converge
        if it % 5000 == 0 and it > 0:
            K = min(K + 1, 20)
            tol = max(tol * 0.8, 0.0001)
            lr = max(lr * 0.92, 0.0005)
        
        grad_a, num_active = _active_set_grad(a, c, max_c, K, tol)
        
        # Convert to z-gradient
        mean_g = np.dot(a, grad_a)
        grad_z = a * (grad_a - mean_g)
        grad_z *= 2.0 * n
        
        # Safety: if gradient is too large or non-finite
        gnorm = np.linalg.norm(grad_z)
        if not np.isfinite(gnorm) or gnorm > 1e6:
            grad_z = np.zeros(n)
            gnorm = 1.0
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        step = lr * mh / (np.sqrt(vh) + eps)
        # Clip step for stability
        step_norm = np.linalg.norm(step)
        max_step = 5.0
        if step_norm > max_step:
            step *= max_step / step_norm
        
        z = z - step
        z = z - z.mean()
        z = np.clip(z, -60, 60)
        
        # Evaluate periodically
        if t % 5 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score

            return grad_z

def _greedy_peak_shave(a_init, deadline, rng, n_trials=50000):
    """Greedy local search: move mass from peak-contributing positions to safe positions."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()

        hi = min(n - 1, max_k)
        if lo > hi:
            continue
        
        pairs = np.arange(lo, hi + 1)
        partners = max_k - pairs
        products = a[pairs] * a[partners]
        peak_idx = pairs[int(np.argmax(products))]
        
        # Strategy 1: Reduce peak_idx, add to a low-contribution position

            # Compute contribution of each position to top lags

            a_trial[i1] = max(0.0, a_trial[i1])

        _make_profile_linear(n_work, 0.5),
        _make_profile_linear(n_work, 1.0),
        _make_profile_linear(n_work, 1.5),
        _make_profile_quadratic(n_work, 0.3, 0.2),
        _make_profile_quadratic(n_work, 0.5, -0.1),
        _make_profile_quadratic(n_work, 0.8, 0.1),
        _make_profile_bimodal(n_work, 0.3, 0.15),
        _make_profile_bimodal(n_work, 0.5, 0.2),
        _make_profile_bimodal(n_work, 0.4, 0.1),
        _make_profile_w(n_work, 2, 0.3),
        _make_profile_w(n_work, 2, 0.5),
        _make_profile_w(n_work, 3, 0.3),
        _make_profile_w(n_work, 4, 0.2),
        _make_profile_w(n_work, 1, 0.4),
        _spectral_init(n_work),
    ]
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # Best starting point for main optimization
    start_a = candidates[0][1].copy() if candidates else parent.copy()
    if best_seq is not None:
        start_a = best_seq.copy()
    
    # --- Phase 1: Main Adam active-set optimization (500s) ---
    phase1_end = min(deadline, t_start + 530.0)
    print(f"Phase 1: Adam active-set, start={_score_norm_fn(start_a):.10f}")
    
    a_best, sc_best = _adam_active_set(start_a, phase1_end, rng,
                                        K_init=5, tol_init=0.005, lr_init=0.012)
    update_best(a_best, sc_best, "adam_main")
    
    # Second pass with different hyperparams
    if time.monotonic() < phase1_end - 60:
        a_best2, sc_best2 = _adam_active_set(start_a, phase1_end, rng,
                                              K_init=10, tol_init=0.002, lr_init=0.008)
        update_best(a_best2, sc_best2, "adam_main2")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: L-BFGS log-sum-exp polishing (400s) ---
    phase2_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"Phase 2: L-BFGS continuation, current={best_score:.10f}")
        
        a_best, sc_best = _lbfgs_logsumexp(best_seq, phase2_end - 5, rng,
                                            beta_scale=1.0, max_wall_time=200)
        update_best(a_best, sc_best, "lbfgs_1")
        
        if time.monotonic() < phase2_end - 50:
            a_best2, sc_best2 = _lbfgs_logsumexp(best_seq, phase2_end - 5, rng,
                                                  beta_scale=0.5, max_wall_time=150)
            update_best(a_best2, sc_best2, "lbfgs_2")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Refined Adam with small K (250s) ---
    phase3_end = min(deadline, t_start + 1180.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"Phase 3: Refined Adam, current={best_score:.10f}")
        
        a_best, sc_best = _adam_active_set(best_seq, phase3_end - 5, rng,
                                            K_init=3, tol_init=0.001, lr_init=0.004)
        update_best(a_best, sc_best, "adam_refined")
        
        if time.monotonic() < phase3_end - 40:
            a_best2, sc_best2 = _adam_active_set(best_seq, phase3_end - 5, rng,
                                                  K_init=2, tol_init=0.0005, lr_init=0.002)
            update_best(a_best2, sc_best2, "adam_fine")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Greedy peak shaving (200s) ---
    phase4_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: Greedy peak shaving, current={best_score:.10f}")
        
        a_best, sc_best = _greedy_peak_shave(best_seq, phase4_end - 5, rng, n_trials=40000)
        update_best(a_best, sc_best, "peak_shave")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Multi-restart (200s) ---
    phase5_end = min(deadline, t_start + 1580.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"Phase 5: Multi-restart, current={best_score:.10f}")
        
        n_restarts = 5
        per_restart = max(15.0, (phase5_end - time.monotonic() - 10) / n_restarts)
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            
            noise = 0.01 + 0.008 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase5_end - 10, time.monotonic() + per_restart * 0.6)
            a_best, sc_best = _adam_active_set(p_start, dl, rng,
                                                K_init=5, lr_init=0.01)
            update_best(a_best, sc_best, f"restart{i}_adam")
            
            if time.monotonic() < phase5_end - 15:
                dl2 = min(phase5_end - 10, time.monotonic() + per_restart * 0.3)
                a_best2, sc_best2 = _greedy_peak_shave(a_best, dl2, rng, n_trials=5000)
                update_best(a_best2, sc_best2, f"restart{i}_local")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Multi-scale (150s) ---
    phase6_end = min(deadline, t_start + 1730.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print(f"Phase 6: Multi-scale, current={best_score:.10f}")
        a_best, sc_best = _multi_scale_polish(best_seq, phase6_end - 5, rng)
        update_best(a_best, sc_best, "multi_scale")
    
    # --- Phase 7: Final polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(20.0, deadline - time.monotonic() - 10)
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")
        
        # Very fine Adam
        if remaining > 40:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.5)
            a_best, sc_best = _adam_active_set(best_seq, dl, rng,
                                                K_init=3, tol_init=0.0003, lr_init=0.001)
            update_best(a_best, sc_best, "final_adam")
        
        # Final local search
        if time.monotonic() < deadline - 25:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.4)
            a_best, sc_best = _greedy_peak_shave(best_seq, dl, rng, n_trials=15000)
            update_best(a_best, sc_best, "final_local")
        
        # Tiny L-BFGS
        if time.monotonic() < deadline - 25:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, rng,
                                                beta_scale=0.3, max_wall_time=remaining * 0.3)
            update_best(a_best, sc_best, "final_lbfgs")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_norm_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator if size permits
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_profile_triangular(8192)
    sc_fb = _score_norm_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
