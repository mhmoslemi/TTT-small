"""
Optimized search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Multi-scale L-BFGS initialization + extended Nesterov GD with 
adaptive log-sum-exp + water-filling peak reduction + direct max gradient.
Key improvements over previous: much more time at n=4096, Nesterov acceleration,
water-filling peak reduction, and better convergence handling.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolution(a):
    return _full_convolve(a, a)


def _score_from_conv(a, c):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _direct_max_grad_z(a, c, top_k=5):
    """Compute gradient of max(c) with respect to z (where a=softmax(z)).
    Uses top-k peak averaging for smoothness."""
    n = len(a)
    s = float(a.sum())
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    
    # Find top-k peaks
    sorted_idx = np.argsort(c)[::-1][:top_k]
    peak_vals = c[sorted_idx]
    # Weight by exponential of peak value (soft weighting)
    weights = np.exp(peak_vals / (cm * 0.1 + 1e-15))
    weights = weights / weights.sum()
    
    # Average gradient over top-k peaks
    grad_a = np.zeros(n)
    for k_idx, wt in zip(sorted_idx, weights):
        k = int(k_idx)
        # dc_k/da_j = 2*a_{k-j} if 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad_a[lo:hi+1] += wt * 2.0 * a[k - (lo:hi+1)]
    
    # Convert to z-gradient via softmax
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _waterfilling_step(a, c, n_reduce=10, alpha=0.01):
    """Water-filling: reduce coefficients that contribute most to peaks."""
    n = len(a)
    cm = float(c.max())
    
    # Find peaks above 0.5 * cm
    peak_threshold = 0.5 * cm
    peak_indices = np.where(c >= peak_threshold)[0]
    
    if len(peak_indices) == 0:
        return a.copy()
    
    # Compute "pressure" at each position
    pressure = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            contributions = a[lo:hi+1] * a[k - (lo:hi+1)]
            pressure[lo:hi+1] += contributions
    
    # Normalize pressure
    p_max = pressure.max()
    if p_max < 1e-30:
        return a.copy()
    pressure = pressure / p_max
    
    # Reduce high-pressure positions
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    z -= alpha * pressure
    z = np.clip(z, -100, 100)
    new_a = _softmax_fn(z)
    return new_a


def _make_profiles(n):
    """Generate various analytical initial profiles."""
    t = np.linspace(0, 1, n)
    profiles = []
    
    # Raised cosine (Hann window)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Hamming window
    a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Various ramps
    for s in [1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential decay from center
    for beta in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(-beta * (t - 0.5) ** 2 * 4)
        profiles.append(a / a.sum())
    
    # Cosine with modulation (chirp-like)
    for freq in [2, 4, 8]:
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) * (1 + 0.3 * np.cos(2 * np.pi * freq * t))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Bell-shaped (Gaussian)
    for sigma in [0.1, 0.2, 0.3]:
        a = np.exp(-((t - 0.5) / sigma) ** 2)
        profiles.append(a / a.sum())
    
    return profiles

def _run_nesterov_gd(a_init, deadline, top_k=10, lr_init=0.5, momentum=0.9):

    n = len(a_init)

    best_a = a_init.copy()

    v = np.zeros(n)  # velocity for Nesterov
    lr = lr_init
    patience = 0
    patience_limit = 5000
    
    c = _full_convolve(a_init, a_init)
    cm = float(c.max())
    beta = 0.1 * cm
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Compute gradient (log-sum-exp for early iterations, direct max for late)
        if it < 50000:
            grad_z = _logsumexp_grad(z, beta, n)
        else:
            a = _softmax_fn(z)
            c = _full_convolve(a, a)

        if not np.all(np.isfinite(grad_z)):
            break
        
        # Nesterov update
        v = momentum * v + grad_z
        z = z - lr * v
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Adaptive beta (decrease over time)
        if it % 1000 == 0:
            a = _softmax_fn(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            beta = max(1e-10 * cm, cm * np.exp(-it / 20000.0))
            
            # Check score
            sc = _score_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1000
            
            # Decay learning rate
            lr = max(0.001, lr * 0.95)
            
            # Restart if stuck
            if patience > patience_limit:
                z = z + 0.05 * np.random.standard_normal(n)
                z = z - z.mean()
                v = np.zeros(n)
                patience = 0
                lr = lr_init * 0.1
    
    return best_a, best_score


def _run_waterfilling(a_init, deadline, alpha_init=0.02):
    """Iterative water-filling peak reduction."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

        c = _full_convolve(a, a)

        cm = float(c.max())
        peak_threshold = 0.7 * cm
        peak_indices = np.where(c >= peak_threshold)[0]

            hi = min(n - 1, k)
            if lo <= hi:

        if p_max < 1e-30:

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy: Multi-scale L-BFGS initialization + extended Nesterov GD with
adaptive log-sum-exp + water-filling peak reduction + direct max gradient.
Key improvements: much more compute at n=4096, Nesterov acceleration,
water-filling, and better convergence handling with annealing restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _direct_max_grad_z(a, c, top_k=5):
    n = len(a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    sorted_idx = np.argsort(c)[::-1]
    k_use = min(top_k, len(sorted_idx))
    idx_sel = sorted_idx[:k_use]
    peak_vals = c[idx_sel]
    temp = max(0.05 * cm, 1e-10)
    weights = np.exp((peak_vals - peak_vals[0]) / temp)
    weights = weights / weights.sum()
    grad_a = np.zeros(n)
    for ki in range(k_use):
        k = int(idx_sel[ki])
        wt = weights[ki]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            grad_a[js] += wt * 2.0 * a[k - js]
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    for s in [0.8, 1.2, 1.5, 2.0, 2.5, 3.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for beta in [1.0, 2.0, 3.0, 4.0, 6.0]:
        a = np.exp(-beta * (t - 0.5) ** 2 * 4)
        profiles.append(a / a.sum())
    for freq in [2, 3, 4, 6, 8]:
        a = 0.5 * (1 + np.cos(2 * np.pi * t)) * (1 + 0.3 * np.cos(2 * np.pi * freq * t))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for sigma in [0.08, 0.15, 0.2, 0.25, 0.35]:
        a = np.exp(-((t - 0.5) / sigma) ** 2)
        profiles.append(a / a.sum())
    for s in [1.5, 2.0, 3.0]:
        a = np.exp(s * t)
        profiles.append(a / a.sum())
    for s in [1.5, 2.0, 3.0]:
        a = np.exp(-s * t)
        profiles.append(a / a.sum())
    return profiles


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _run_lbfgs_at_scale(a_init, deadline, n_betas=10):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_c0 < 1e-30:
        return best_a, best_score
    beta_start = 0.2 * max_c0
    beta_end = max(1e-7 * max_c0, 1e-12)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        level_deadline = time.monotonic() + min(time_budget, 100.0)
        if n <= 256:
            maxiter = 3000
        elif n <= 1024:
            maxiter = 5000
        else:
            maxiter = 8000
        maxfun = maxiter * 8
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': maxfun, 'maxcor': 40}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc = _score_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    return best_a, best_score


def _run_nesterov_gd(a_init, deadline, top_k=10, lr_init=0.3, momentum=0.9, rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_fn(a_init)
    v = np.zeros(n)
    lr = lr_init
    patience = 0
    patience_limit = 8000
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    beta = 0.1 * cm
    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max) if hasattr(float, 'max') else float(c.max())
        if it < 30000:
            grad_z = _logsumexp_grad(z, beta, n)
        else:
            grad_z = _direct_max_grad_z(a, c, top_k)
        if not np.all(np.isfinite(grad_z)):
            break
        v = momentum * v + grad_z
        z = z - lr * v
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        if it % 500 == 0:
            a = _softmax_fn(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            beta = max(1e-10 * cm, cm * np.exp(-it / 15000.0))
            sc = _score_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 500
            if it % 5000 == 0:
                lr = max(0.001, lr * 0.92)
            if patience > patience_limit:
                noise_scale = 0.03 + 0.02 * (patience // patience_limit)
                z = z + noise_scale * rng.standard_normal(n)
                z = np.clip(z, -70, 70)
                z = z - z.mean()
                v = np.zeros(n)
                patience = 0
                lr = min(lr_init, lr * 2.0)
    return best_a, best_score


def _run_waterfilling(a_init, deadline, alpha_init=0.015):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    alpha = alpha_init
    stall = 0
    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break
        pressure = np.zeros(n)
        peak_threshold = 0.6 * cm
        peak_indices = np.where(c >= peak_threshold)[0]
        if len(peak_indices) == 0:
            break
        for k in peak_indices:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                contrib = a[js] * a[k - js]
                pressure[js] += contrib
        p_max = pressure.max()
        if p_max < 1e-30:
            break
        pressure /= p_max
        z_trial = z - alpha * pressure
        z_trial = np.clip(z_trial, -100, 100)
        z_trial = z_trial - z_trial.mean()
        a_trial = _softmax_fn(z_trial)
        sc_trial = _score_fn(a_trial)
        if np.isfinite(sc_trial) and sc_trial < best_score:
            best_score = sc_trial
            best_a = a_trial.copy()
            z = z_trial.copy()
            stall = 0
        else:
            alpha *= 0.85
            stall += 1
            if stall > 200:
                z = z + 0.02 * np.random.standard_normal(n)
                z = z - z.mean()
                alpha = alpha_init * 0.5
                stall = 0
    return best_a, best_score


def _run_direct_max_nesterov(a_init, deadline, lr_init=0.05, momentum=0.93, top_k=3):
