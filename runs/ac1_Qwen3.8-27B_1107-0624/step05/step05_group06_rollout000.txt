LP-based alternating minimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the insight that given one factor f, the optimal other factor g can be
found by solving a linear program. Alternating between factors converges to
the optimal non-negative sequence.

from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Build the convolution matrix A where A[k,i] = f[k-i] (0 if out of bounds)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + i * 0.002)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax_fn(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            else:
                lr *= 0.92

    a_final = _softmax_fn(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    print(f"    Subgrad polish: final score={best_sc:.10f}")
    return best_a, best_sc


def _perturb_fn(a, sigma, rng):
    """Add Gaussian noise in log-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax_fn(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    LP-based alternating minimization with subgradient polishing.
    Uses the LP subproblem structure for fast, principled optimization.
    """
    global height_sequence_1
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # --- Get initial sequence ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
        a0 = parent.copy()
        n_main = len(parent)
    else:
        n_main = 2048
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # --- Phase 1: LP-based alternating optimization (main) ---
    print(f"\nPhase 1: LP alternating (n={n_main}) t={time.monotonic()-t_start:.1f}s")
    lp_deadline = deadline - 400.0
    a_lp, sc_lp = _lp_alternate(a0, t_init=0.5, max_iter=10000,
                                deadline=lp_deadline, verbose=True)
    if _score(a_lp) < best_score:
        best_seq = a_lp.copy()
        best_score = _score(a_lp)
        print(f"  LP improved: {best_score:.10f}")

    # --- Phase 2: Subgradient polishing ---
    print(f"\nPhase 2: Subgradient polish t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 100:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 80,
                                      n_max_steps=5000, lr_init=0.02)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg
            print(f"  SG improved: {best_score:.10f}")

    # --- Phase 3: Another LP round on polished result ---
    print(f"\nPhase 3: LP on polished t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 30:
        a_lp2, sc_lp2 = _lp_alternate(best_seq, t_init=0.3, max_iter=2000,
                                      deadline=deadline - 10, verbose=False)
        if sc_lp2 < best_score:
            best_seq = a_lp2.copy()
            best_score = sc_lp2
            print(f"  LP2 improved: {best_score:.10f}")

        # Final subgradient
        if time.monotonic() < deadline - 3:
            a_sg2, sc_sg2 = _subgrad_polish(best_seq, deadline - 1,
                                            n_max_steps=1000, lr_init=0.01)
            if sc_sg2 < best_score:
                best_seq = a_sg2.copy()
                best_score = sc_sg2
                print(f"  Final SG improved: {best_score:.10f}")

    # --- Phase 4: Try a perturbed restart if time remains ---
    print(f"\nPhase 4: Perturbed restart t={time.monotonic()-t_start:.1f}s")
    if time.monotonic() < deadline - 60:
        a_pert = _perturb_fn(best_seq, 0.02, rng)
        sc_pert = _score(a_pert)
        print(f"  Perturbed start: {sc_pert:.10f}")
        a_pert2, sc_pert2 = _lp_alternate(a_pert, t_init=0.4, max_iter=1000,
                                          deadline=deadline - 15, verbose=False)
        if sc_pert2 < best_score:
            best_seq = a_pert2.copy()
            best_score = sc_pert2
            print(f"  Perturbed LP improved: {best_score:.10f}")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: internal score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
