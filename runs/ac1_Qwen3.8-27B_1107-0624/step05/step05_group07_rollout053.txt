```python
import numpy as np
import time
import cvxpy as cp
from scipy.sparse import csr_matrix
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _build_conv_matrix(n):
    # Build the (2n-1) x n convolution matrix
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.ones(n * n, dtype=np.float64)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def _solve_lp(n, max_time=300):
    """Solve the primal LP: min t s.t. conv(a,a) <= t, sum(a)=1, a>=0"""
    A = _build_conv_matrix(n)
    a = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ a <= t,
        cp.sum(a) == 1,
        a >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if a.value is not None and t.value is not None:
            a_val = np.maximum(a.value, 0)
            a_val = a_val / a_val.sum()
            return a_val, float(t.value)
    except Exception as e:
        print(f"  LP solve failed: {e}")
    return None, None
```

"""
Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines iterative LP, L-BFGS with continuation, peak-shaving heuristics,
and multi-scale multi-restart strategy. Focuses on escaping local optima
via diverse initializations and adaptive peak reduction.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f, max_time=60):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 0:
                g_val = g_val / s
            return g_val, float(t.value)
    except Exception as e:
        print(f"  LP failed: {e}")
    return None, None


def _lp_iterate(f, n_iters=100, deadline=None, step_size=None):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    f = f / f.sum()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(f, f)
        M = float(c.max())
        g, t_val = _lp_minimax(f)
        if g is None:
            break
        if step_size is not None:
            ss = step_size
        else:
            ss = 0.5 / (1 + it * 0.1)
        f_new = (1 - ss) * f + ss * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        if it % 20 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002,
                 0.00001, 0.000005, 0.000002]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(100, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 40, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _peak_shaving(a, n_rounds=100, deadline=None):
    """Iteratively shave the peaks of the convolution."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a = a / a.sum()

    for rnd in range(n_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        k_star = int(np.argmax(c))

        # Find elements contributing most to the peak
        # c[k_star] = sum_i a[i] * a[k_star - i]
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Reduce the top contributors
        top_k = max(1, n // 50)
        top_indices = np.argsort(contrib)[-top_k:]
        reduction = 0.02 * rnd / n_rounds + 0.005

        a[top_indices] *= (1 - reduction)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # Revert if no improvement
            a = best_a.copy()

    return best_a, best_score


def _subgradient_descent(a_init, deadline, lr_init=0.01, n_iters=50000):
    """Subgradient descent with decreasing learning rate."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = lr_init

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        # Subgradient: 2 * a[k_star - i] for valid i
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]

        # Projected gradient step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            lr *= 0.95

        if it % 10000 == 0:
            lr = max(lr_init * 0.1, lr * 0.9)

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=30):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_local_score = _score_norm_fn(a)
            best_delta = 0.0
            for delta in [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 2.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    # Gaussian-like
    a = np.exp(-5 * u**2)
    profiles.append(a / a.sum())
    # Bimodal
    a = np.exp(-20*(u-0.3)**2) + np.exp(-20*(u+0.3)**2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _mutate_sequence(a, rng, strength=0.1):
    """Create a perturbed version of the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + strength * rng.normal(size=n)
    z = np.clip(z, -50, 50)
    z = z - z.mean()
    return _softmax_fn(z)


def _adaptive_gd(a_init, deadline, rng, lr_init=0.15, max_iter=50000):
    """Adaptive gradient descent with momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-8, 0.02 * cm)
        vals = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 5000 == 0:
            lr = max(0.01, lr * 0.85)
        if t % 20000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score, best_n
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = len(p)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = _make_profiles(n_init)[3]
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    print(f"Starting optimization, deadline at t={(deadline-t_start):.0f}s")

    # Phase 1: Multi-n LP iterations (300s)
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\nPhase 1: Multi-n LP iterations")

    n_candidates_lp = sorted(set([2048, n_parent, min(4096, n_parent)]))
    n_candidates_lp = [x for x in n_candidates_lp if x <= 4096]

    for n_try in n_candidates_lp:
        if time.monotonic() > phase1_end:
            break
        if n_try == n_parent:
            a_cur = parent.copy()
        else:
            a_cur = _interp_profile(parent, n_try)

        sc_init = _score_norm_fn(a_cur)
        update_best(a_cur, sc_init, f"lp_init_n{n_try}")

        # Run LP with decreasing step sizes
        for round_idx in range(3):
            if time.monotonic() > phase1_end:
                break
            step = 0.4 / (round_idx + 1)
            a_cur, sc_cur = _lp_iterate(a_cur, n_iters=20, deadline=phase1_end, step_size=step)
            update_best(a_cur, sc_cur, f"lp_r{round_idx}_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.10f} (n={best_n})")

    # Phase 2: L-BFGS continuation at best n (400s)
    phase2_end = min(deadline, t_start + 700.0)
    print(f"\nPhase 2: L-BFGS continuation")

    n_curr = best_n
    a_start = best_seq.copy()

    # Multiple restarts
    n_restarts = 4
    per_restart_time = (phase2_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 20:
            break
        if r == 0:
            a_cur = best_seq.copy()
        elif r == 1:
            a_cur = _mutate_sequence(best_seq, rng, strength=0.05)
        elif r == 2:
            a_cur = _mutate_sequence(best_seq, rng, strength=0.1)
        else:
            a_cur = _mutate_sequence(best_seq, rng, strength=0.15)

        per_dl = min(phase2_end - 10, time.monotonic() + per_restart_time)
        p_best, sc_best = _run_lbfgs_continuation(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}_n{len(a_cur)}")

    print(f"  Phase 2 done: best={best_score:.10f} (n={best_n})")

    # Phase 3: Try different n values with interpolation + optimization (250s)
    phase3_end = min(deadline, t_start + 950.0)
    print(f"\nPhase 3: Multi-n exploration")

    n_explore_list = [x for x in [3072, 4096, 5120, 6144] if x != best_n]
    for n_exp in n_explore_list:
        if time.monotonic() > phase3_end:
            break
        if time.monotonic() > phase3_end - 60:
            break
        p_interp = _interp_profile(best_seq, n_exp)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_exp}")

        if time.monotonic() < phase3_end - 40:
            per_dl = min(phase3_end - 20, time.monotonic() + 80.0)
            p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl)
            update_best(p_best, sc_best, f"lbfgs_n{n_exp}")

    print(f"  Phase 3 done: best={best_score:.10f} (n={best_n})")

    # Phase 4: Peak shaving + adaptive GD (200s)
    phase4_end = min(deadline, t_start + 1150.0)
    print(f"\nPhase 4: Peak shaving and adaptive GD")

    if time.monotonic() < phase4_end and best_seq is not None:
        per_dl = min(phase4_end - 80, time.monotonic() + 80.0)
        p_best, sc_best = _peak_shaving(best_seq, n_rounds=200, deadline=per_dl)
        update_best(p_best, sc_best, "peak_shaving")

        if time.monotonic() < phase4_end - 30:
            per_dl2 = min(phase4_end - 20, time.monotonic() + 100.0)
            p_best2, sc_best2 = _adaptive_gd(best_seq, per_dl2, rng, lr_init=0.12)
            update_best(p_best2, sc_best2, "adaptive_gd")

    print(f"  Phase 4 done: best={best_score:.10f} (n={best_n})")

    # Phase 5: Fine L-BFGS with very small beta (150s)
    phase5_end = min(deadline, t_start + 1300.0)
    print(f"\nPhase 5: Ultra-fine L-BFGS")

    if time.monotonic() < phase5_end and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        for beta_frac in [1e-5, 5e-6, 2e-6, 1e-6]:
            if time.monotonic() > phase5_end:
                break
            beta_tiny = beta_frac * max_c_final
            per_dl = min(phase5_end - 10, time.monotonic() + 40.0)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta_tiny, n_final),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-15,
                             'maxls': 30, 'maxfun': 15000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_seq = a_new.copy()
                        print(f"    Ultra-fine beta={beta_frac}: score={s_new:.10f}")
            except Exception:
                pass

    print(f"  Phase 5 done: best={best_score:.10f} (n={best_n})")

    # Phase 6: Coordinate descent polish + final L-BFGS (remaining)
    phase6_end = deadline
    print(f"\nPhase 6: Final polish")

    if time.monotonic() < phase6_end and best_seq is not None:
        per_dl = min(phase6_end - 20, time.monotonic() + 120.0)
        p_best, sc_best = _coord_descent(best_seq, per_dl, max_rounds=20)
        update_best(p_best, sc_best, "coord_descent")

        # Final subgradient descent
        if time.monotonic() < phase6_end - 15:
            per_dl2 = min(phase6_end - 10, time.monotonic() + 60.0)
            p_best2, sc_best2 = _subgradient_descent(best_seq, per_dl2, lr_init=0.005, n_iters=20000)
            update_best(p_best2, sc_best2, "subgradient")

        # One more L-BFGS pass
        if time.monotonic() < phase6_end - 5:
            per_dl3 = min(phase6_end - 3, time.monotonic() + 30.0)
            p_best3, sc_best3 = _run_lbfgs_continuation(best_seq, per_dl3)
            update_best(p_best3, sc_best3, "final_lbfgs")

    print(f"  Phase 6 done: best={best_score:.10f} (n={best_n})")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
