    W = float(np.dot(w, _full_convolve(a, a)))
    # d/dz_i [sum_k w_k (a*a)_k] = sum_k w_k * (2*a_{k-i} * a_i - 2*a_i * sum_j a_j * a_rev[k-j]... )
    # Actually: a_i = exp(z_i)/sum(exp(z)). da_j/dz_i = a_j*(delta_ji - a_i)
    # d(a*a)_k/dz_i = sum_j [da_j/dz_i * a_{k-j} + a_j * da_{k-j}/dz_i]
    # = sum_j [a_j*(delta_ji - a_i)*a_{k-j} + a_j*a_{k-j}*(delta_{k-j,i} - a_i)]
    # = a_i*a_{k-i} - a_i*sum_j a_j*a_{k-j} + a_{k-i}*a_i - a_i*(a*a)_k
    # = 2*a_i*a_{k-i} - 2*a_i*(a*a)_k
    # So d/dz_i [sum_k w_k (a*a)_k] = sum_k w_k * (2*a_i*a_{k-i} - 2*a_i*(a*a)_k)
    # = 2*a_i * [sum_k w_k*a_{k-i} - (a*a)·w]
    # = 2*a_i * [h_i - W]  where h = conv(w, a_rev)[n-1:2n-1]
    return 2.0 * a * (h - W)

def _fista_backtrack(a_init, beta, deadline, max_iter):
    """FISTA with backtracking line search on the simplex."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    c_curr = _full_convolve(a_curr, a_curr)
    best_score = float(c_curr.max())
    best_a = a_curr.copy()
    
    # Initial step size estimate
    L = 4.0 * n / beta
    step = 1.0 / L
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # FISTA momentum
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute weights (softmax of convolution)
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        e = np.exp(shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        # Gradient
        grad = _compute_gradient(y, w)
        
        # Backtracking
        a_trial = _proj_simplex(y - step * grad)
        # CheckArmijo: f(a_trial) <= f(y) + grad·(a_trial - y) + (1/(2*step))||a_trial - y||²
        # For simplicity, just check that the score doesn't increase too much
        # (non-monotone: allow small increases)
        
        # For efficiency, skip full backtracking and just use the fixed step
        # (the smoothed objective is well-behaved enough)
        
        a_next = a_trial
        
        # Update
        if it % 200 == 0 or it < 5:
            c_next = _full_convolve(a_next, a_next)
            sc = float(c_next.max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
    
    # Final check
    c_final = _full_convolve(a_curr, a_curr)
    sc_final = float(c_final.max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_peak_directed(a_init, deadline, schedule):
    """Adam in log-space with peak-directed gradient weighting.
    schedule is a list of (beta, lr, n_iter) tuples."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = float(_full_convolve(a_init, a_init).max())
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta, lr, n_iter in schedule:
        for _ in range(n_iter):
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax_simplex(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            # Peak-directed: weight the top peaks more heavily
            top_k = max(1, len(c) // 50)
            idx_sorted = np.argsort(c)[::-1][:top_k]
            
            # Smoothed weights
            shifted = (c - cm) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Additional peak emphasis
            peak_boost = 1.0 + 2.0 * (c[idx_sorted] - c.mean()) / (cm + 1e-10)
            # Actually, let's just use the standard softmax weights
            # The peak-directed aspect is already in the softmax
            
            grad = _compute_log_grad(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_norm = 0.5
            if update_norm > max_norm:
                update = update * (max_norm / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_simplex(z)
                sc = float(_full_convolve(a_chk, a_chk).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
    
    a_final = _softmax_simplex(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_simplex(z)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA with adaptive smoothing, peak-directed Adam in log-space, multi-length exploration
(n=4096, n=8192), diverse restarts, and ultra-fine final polishing.
Key improvements over baseline: more aggressive FISTA schedule, cosine-annealed Adam,
n=8192 exploration, and peak-focus weighting.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_simplex(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_log_grad(a, w, n, c):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 2.0 * a * (h - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = float(_full_convolve(a_curr, a_curr).max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        shifted = (c_y - c_max) / beta
        e = np.exp(shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum

        grad = _compute_gradient(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = float(_full_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    sc_final = float(_full_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_fine(a_init, deadline, beta_schedule, lr_init):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = float(_full_convolve(a_init, a_init).max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta, lr in beta_schedule:
        if time.monotonic() > deadline:
            break
        stagnant = 0
        last_best = best_score
        it_phase = 0
        max_iter_phase = 200000

        while it_phase < max_iter_phase:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            a = _softmax_simplex(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            shifted = (c - cm) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_log_grad(a, c, n, c)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            # Cosine-like LR decay within phase
            decay = 0.5 * (1.0 + np.cos(np.pi * it_phase / max_iter_phase))
            eff_lr = lr * decay

            update = eff_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.4:
                update = update * (0.4 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 40 == 0:
                a_chk = _softmax_simplex(z)
                sc = float(_full_convolve(a_chk, a_chk).max())
                if sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
                    stagnant = 0
                else:
                    stagnant += 40

                if stagnant > 30000:
                    break

    a_final = _softmax_simplex(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_simplex(z)


def _perturb_freq(a, sigma, rng):
    n = len(a)
    A = np.fft.rfft(a, n * 2)
    A = np.abs(A)
    A *= (1.0 + sigma * rng.standard_normal(A.shape))
    A = np.maximum(A, 0)
    p = np.fft.irfft(A, n)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        return _softmax_simplex(np.zeros(n))
    return p / s
