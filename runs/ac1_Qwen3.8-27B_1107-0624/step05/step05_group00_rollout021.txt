"""
Multi-phase optimizer combining FISTA, Adam log-space, and exact coordinate-wise
refinement for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovation: exact 1D subproblem solving per coordinate for precise local optimization.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1, so score = 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_conv_grad(a, w):
    """
    Compute gradient of sum_k w_k * (a*a)[k] with respect to a.
    w is a weight vector over the 2n-1 convolution lags.
    Returns gradient of shape (n,).

    # = 2 * sum_l w[j+l] * a[l] (where w is defined for indices 0..2n-2)

    # = (cross_correlation of w and reversed a)[j] ... 
    
    # Simplest: use the fact that this is a convolution-like operation

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    L = 4.0 * n / beta  # Lipschitz constant (conservative)
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _compute_conv_grad(y, w)
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):

    current_lr = lr_init

        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break

        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            e = np.exp(z - z_max)
            s = e.sum()

            else:
                a = e / s

    So dF/dz_j = sum_i a_i*(delta_ij - a_j) * dF/da_i
    = a_j * dF/da_j - a_j * sum_i a_i * dF/da_i
    
    = a_j * grad_a[j] - a_j * (a @ grad_a)
    
    where grad_a = dF/da.
    """
    grad_a = _compute_conv_grad(a, w) * n  # = 2n * sum_k w_k * 2*a[k-i]... 
    
    # Wait, _compute_conv_grad already includes the factor of 2.
    # F = 2n * sum_k w_k * c_k
    # dF/da = 2n * sum_k w_k * dc_k/da = 2n * 2 * cross_corr(w, a) = 4n * cross_corr(w,a)
    # But _compute_conv_grad returns 2 * cross_corr(w, a), so dF/da = 2n * _compute_conv_grad(a, w)
    
    grad_a = 2.0 * n * _compute_conv_grad(a, w)
    
    # dF/dz_j = a_j * grad_a[j] - a_j * (a @ grad_a)
    a_dot_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - a_dot_grad)
    
    # The -a_dot_grad term is a constant offset (doesn't depend on j in a meaningful way
    # for optimization since we're in log-space and the softmax makes it invariant to shifts)
    # So we can just use a * grad_a
    # Actually no, we need the full gradient for Adam to work correctly.
    # But the constant offset doesn't matter since Adam normalizes.
    # Let's keep it for correctness.
    
    return grad_z


def _coord_wise_step(a, j, conv_full, n):
    """
    Given current sequence a (sum=1), full convolution conv_full = conv(a,a),
    compute the optimal value for coordinate j (holding others proportionally scaled).
    
    Returns the new a[j] value.
    """
    aj = a[j]
    S = 1.0 - aj  # sum of other coordinates
    
    if S < 1e-15:
        return aj  # Can't change if all mass is on j
    
    # Compute conv_u = conv(a with a[j]=0, a with a[j]=0)
    # conv_u[k] = conv_full[k] - 2*a[j]*a[k-j]*[0<=k-j<n] + a[j]^2*[k=2j]
    conv_u = conv_full.copy()
    
    # Subtract contributions of a[j]
    # For k where 0 <= k-j < n: conv_full[k] includes a[j]*a[k-j] (from i=j term)
    # and a[k-j]*a[j] (from i=k-j term), unless k-j=j in which case it's just a[j]^2
    k_indices = np.arange(n) + j  # k = j + l for l = 0, ..., n-1
    valid_k = k_indices < (2 * n - 1)  # k must be in [0, 2n-2]
    
    # For k in [j, j+n-1] ∩ [0, 2n-2]: subtract 2*a[j]*a[k-j]
    # But add back a[j]^2 where k-j = j (i.e., k = 2j)
    l_indices = np.arange(n)  # l = 0, ..., n-1 corresponds to k-j
    valid_l = l_indices < n  # always true
    valid_k2 = k_indices < (2 * n - 1)  # k < 2n-1
    
    # conv_u[k] = conv_full[k] - 2*aj*a[k-j] for 0<=k-j<n, k-j!=j
    #            = conv_full[k] - aj^2 for k=2j (since the single term a[j]^2 is removed)
    
    # Simpler: just compute directly
    for l in range(n):
        k = j + l
        if k >= 2 * n - 1:
            break
        # a[k-j] = a[l]
        if l == j:
            conv_u[k] -= aj * aj  # remove the single a[j]*a[j] term
        else:
            conv_u[k] -= 2.0 * aj * a[l]  # remove both a[j]*a[l] and a[l]*a[j]
    
    # Now B_k and D_k:
    # B_k = a[k-j] if 0<=k-j<n and k-j != j, else 0
    # D_k = 1 if k == 2j, else 0
    
    # The quadratic for lag k: (1-t)^2/S^2 * C_k + 2*(1-t)*t/S * B_k + t^2 * D_k
    # where C_k = conv_u[k]
    
    # Simplify: let t be the new value of a[j]
    # a_new[j] = t
    # a_new[i] = (1-t)/S * a[i] for i != j
    # (a_new * a_new)[k] = (1-t)^2/S^2 * conv_u[k] + 2*(1-t)*t/S * a[k-j]*[0<=k-j<n, k-j!=j] + t^2*[k=2j]
    
    # We want to minimize max_k of this over t in [max(0,0), min(1, 1)] = [0, 1]
    # Actually t can be any value in [0, 1] (since a_new[j] = t and others scale to fill)
    
    # Vectorized computation:
    # Let's precompute arrays for all k:
    # C = conv_u (length 2n-1)
    # B[k] = a[k-j] if 0<=k-j<n and k-j!=j, else 0
    # D[k] = 1 if k==2j, else 0
    
    C = conv_u
    B = np.zeros(2 * n - 1)
    D = np.zeros(2 * n - 1)
    
    # Fill B: for k = j + l where 0 <= l < n and l != j
    for l in range(n):
        if l == j:
            continue
        k = j + l
        if k < 2 * n - 1:
            B[k] = a[l]
    
    if 2 * j < 2 * n - 1:
        D[2 * j] = 1.0
    
    # Quadratic: (1-t)^2/S^2 * C + 2*(1-t)*t/S * B + t^2 * D
    # = C/S^2 * (1-2t+t^2) + 2B/S * (t-t^2) + D * t^2
    # = C/S^2 + t*(-2C/S^2 + 2B/S) + t^2*(C/S^2 - 2B/S + D)
    
    A_q = C / (S * S) - 2.0 * B / S + D
    B_q = -2.0 * C / (S * S) + 2.0 * B / S
    C_q = C / (S * S)
    
    # f(t) = max_k (A_q[k] * t^2 + B_q[k] * t + C_q[k])
    # This is convex in t (max of convex functions)
    
    # Find minimum using golden section search on [0, 1]
    # Actually, t should be in [0, 1] since a_new[j] = t and sum = 1
    
    # Golden section search
    gamma = (np.sqrt(5) - 1) / 2  # ~0.618
    
    lo = 0.0
    hi = 1.0
    c1 = hi - gamma * (hi - lo)
    c2 = lo + gamma * (hi - lo)
    
    def _eval_quad(t_val):
        return float(np.max(A_q * t_val * t_val + B_q * t_val + C_q))
    
    f1 = _eval_quad(c1)
    f2 = _eval_quad(c2)
    
    for _ in range(60):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gamma * (hi - lo)
            f1 = _eval_quad(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gamma * (hi - lo)
            f2 = _eval_quad(c2)
    
    t_opt = (lo + hi) / 2.0
    return t_opt


def _coord_wise_pass(a, conv_full, order, max_coords=None):
    """
    Do one pass of coordinate-wise optimization.
    a: sequence with sum=1
    conv_full: full convolution of a
    order: permutation of coordinates to optimize in
    """
    n = len(a)
    a_new = a.copy()
    
    # We'll update a sequentially, recomputing conv as needed
    # For efficiency, we update one coordinate at a time and adjust conv_full
    
    for idx in range(len(order)):
        if max_coords is not None and idx >= max_coords:
            break
        j = order[idx]
        
        aj = a_new[j]
        S = 1.0 - aj
        if S < 1e-12:
            continue
        
        # Compute the optimal t for coordinate j
        # Using the precomputed conv_full (which corresponds to a_new)
        t_opt = _coord_wise_step(a_new, j, conv_full, n)
        
        if abs(t_opt - aj) > 1e-10:
            # Update a_new[j] = t_opt, scale others
            scale = (1.0 - t_opt) / S
            a_new = a_new * 0  # will recompute
            # Actually, let's do it more carefully
            old_j = aj
            a_new[j] = t_opt
            # Scale all other elements
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_new[mask] = a_new[mask] * scale
            
            # Update conv_full
            conv_full = _full_convolve(a_new, a_new)
        
        if idx % 1000 == 0 and idx > 0:
            pass  # Could check score here
    
    return a_new, conv_full


def _coord_wise_optimize(a, deadline, rng, max_passes=100):
    """
    Run coordinate-wise optimization until deadline.
    a: normalized sequence (sum=1)
    """
    n = len(a)
    a_best = a.copy()
    conv_best = _full_convolve(a_best, a_best)
    score_best = float(2.0 * n * conv_best.max())
    
    a_curr = a.copy()
    conv_curr = _full_convolve(a_curr, a_curr)
    
    pass_count = 0
    while time.monotonic() < deadline:
        pass_count += 1
        order = rng.permutation(n)
        
        # Time budget per pass
        time_per_pass = (deadline - time.monotonic()) / max(1, max_passes - pass_count)
        max_coords = int(n * min(1.0, time_per_pass / (n * 0.0001)))  # rough estimate
        max_coords = max(n, min(n, max_coords))  # do at least one full pass
        
        a_curr, conv_curr = _coord_wise_pass(a_curr, conv_curr, order, max_coords)
        
        sc = float(2.0 * n * conv_curr.max())
        if sc < score_best:
            score_best = sc
            a_best = a_curr.copy()
            conv_best = conv_curr.copy()
        
        # Check improvement - if no improvement in a full pass, try smaller steps
        if pass_count % 5 == 0:
            print(f"    Coord-wise pass {pass_count}: score={score_best:.10f}")
        
        # After several passes, try a half-step to avoid overshooting
        if pass_count > 10:
            # Blend with previous best
            a_try = 0.5 * a_curr + 0.5 * a_best
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try > 0:
                a_try = a_try / s_try
                c_try = _full_convolve(a_try, a_try)
                sc_try = float(2.0 * n * c_try.max())
                if sc_try < score_best:
                    score_best = sc_try
                    a_best = a_try.copy()
                    conv_best = c_try.copy()
    
    return a_best, score_best


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
