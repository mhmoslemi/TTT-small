"""
LP best-response + Continuation Adam optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: (1) dual-guided LP direction combining primal solution with shadow price gradient,
(2) continuation method with geometrically decreasing log-sum-exp smoothing,

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -60, 60))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, rows, cols, M, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None, None
        primal = np.maximum(res.x, 0.0)
        dual = None
        try:
            if hasattr(res, 'ineqlin') and res.ineqlin is not None:
                marg = getattr(res.ineqlin, 'marginals', None)
                if marg is not None:
                    dual = np.maximum(marg, 0.0)
        except Exception:
            dual = None
        return primal, dual
    except Exception:
        return None, None


def _compute_gradient_logspace(a, c, weights, n):
    """Compute gradient of smoothed score in log-space."""
    w_rev = weights[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    return grad_z, grad_a


def _line_search(f, g, max_iter=20):
    """Binary search for optimal mixing parameter t."""
    best_t = 0.0
    best_sc = _score_norm(f)
    t_lo, t_hi = 0.0, 1.0
    for _ in range(max_iter):
        if time.monotonic() > 1e9:
            break
        t_mid = (t_lo + t_hi) / 2.0
        if t_mid <= best_t:
            t_mid = (best_t + t_hi) / 2.0
        sc = _score_norm((1.0 - t_mid) * f + t_mid * g)
        if sc < best_sc:
            best_sc = sc
            best_t = t_mid
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 0.01:
            break
    return best_t, best_sc

    g = g / g.sum()

        conv_fd = _full_convolve(f, w_rev)

        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        
        # Gradient clipping
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        # Periodic scoring
        if t % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive learning rate
        if t % 5000 == 0:
            if best_score < prev_best * 0.999:
                current_lr = min(0.05, current_lr * 1.15)
                prev_best = best_score
            elif no_improve > 4:
                current_lr = max(0.001, current_lr * 0.85)
                no_improve = 0
            prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _quick_adam(a_init, deadline, lr=0.005, max_iter=200000):
    """Fast Adam polishing with fixed small smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    c_init = _full_convolve(a_init, a_init)
    beta = 0.001 * float(c_init.max())
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e
        grad_z, _ = _compute_gradient_logspace(a, c, w, n)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 0.5:
            update = update * (0.5 / un)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if t % 300 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _perturb_log(a, sigma, rng):
    """Perturb in log-space and renormalize."""
    z = np.log(np.maximum(a, 1e-300))
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 12.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        a = parent.copy()
        best_seq = a.copy()
        best_score = _score(a)
        print(f"Parent: n={n}, score={best_score:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        best_seq = a.copy()
        best_score = _score(a)
        print(f"No parent, uniform n={n}, score={best_score:.10f}")
    
    # Build LP matrix
    rows, cols = _build_conv_matrix(n)
    print(f"LP matrix: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")
    
    # ===== Phase 1: LP with dual-guided updates (0 - 350s) =====
    phase1_end = min(deadline, t_start + 350.0)
    lp_count = 0
    stall = 0
    prev_sc = _score_norm(a)
    
    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a, t_used = _lp_iteration(a, rows, cols, use_dual=True)
        sc = _score_norm(a)
        lp_count += 1
        dt = time.monotonic() - t0
        
        if sc < prev_sc - 1e-10:
            stall = 0
        else:
            stall += 1
        
        if lp_count % 5 == 0 or stall > 15:
            full_sc = _score(a)
            print(f"  LP1[{lp_count}] score={full_sc:.10f} t_use={t_used:.3f} dt={dt:.1f}s stall={stall} time={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
        
        if stall > 40:
            print(f"  LP1 stalled at iter {lp_count}")
            break
        prev_sc = sc
    
    full_sc = _score(a)
    if full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 done: {lp_count} iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # ===== Phase 2: Continuation Adam (350 - 850s) =====
    phase2_end = min(deadline, t_start + 850.0)
    if time.monotonic() < phase2_end - 30:
        # Run continuation from the LP result
        a_cont, sc_cont = _continuation_adam(best_seq, phase2_end - 10, max_iter=500000)
        full_sc = _score(a_cont)
        print(f"Phase 2 continuation: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_cont.copy()
    
    # ===== Phase 3: LP refinement from best (850 - 1300s) =====
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end - 30:
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp3 = 0
        while time.monotonic() < phase3_end:
            a, t_used = _lp_iteration(a, rows, cols, use_dual=True)
            sc = _score_norm(a)
            lp3 += 1
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if lp3 % 5 == 0:
                full_sc = _score(a)
                if full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
            if stall > 35:
                break
            prev_sc = sc
        print(f"Phase 3: {lp3} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # ===== Phase 4: Multi-restart with diverse strategies (1300 - 1600s) =====
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end - 30:
        strategies = [
            (0.02, "small_perturb"),
            (0.04, "med_perturb"),
            (0.06, "large_perturb"),
            (0.0, "lp_only_from_best"),
            (0.03, "small_perturb2"),
        ]
        for si, (sigma, name) in enumerate(strategies):
            if time.monotonic() > phase4_end - 60:
                break
            if sigma > 0:
                a_r = _perturb_log(best_seq, sigma, rng)
            else:
                a_r = best_seq.copy()
            
            # Quick LP refinement
            prev_sc = _score_norm(a_r)
            stall = 0
            for it in range(20):
                if time.monotonic() > phase4_end - 30:
                    break
                a_r, _ = _lp_iteration(a_r, rows, cols, use_dual=True)
                sc = _score_norm(a_r)
                if sc < prev_sc - 1e-10:
                    stall = 0
                else:
                    stall += 1
                if stall > 10:
                    break
                prev_sc = sc
            
            # Quick gradient polish
            if time.monotonic() < phase4_end - 60:
                deadline_quick = min(phase4_end - 30, time.monotonic() + 60)
                a_r, _ = _quick_adam(a_r, deadline_quick, lr=0.004, max_iter=50000)
            
            full_sc = _score(a_r)
            print(f"  Restart[{si}] {name}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a_r.copy()
    
    # ===== Phase 5: Final fine polishing (1600 - deadline) =====
    if time.monotonic() < deadline - 20:
        deadline5 = deadline - 5
        a_fin, sc_fin = _quick_adam(best_seq, deadline5, lr=0.002, max_iter=300000)
        full_sc = _score(a_fin)
        print(f"Phase 5 final: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_fin.copy()
    
    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
