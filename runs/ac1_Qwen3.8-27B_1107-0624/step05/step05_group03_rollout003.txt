"""
Optimized flat-top window approach for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses parameterized cosine window initialization, adaptive peak-equalization,
and multi-scale L-BFGS continuation.
"""
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _make_flattop_window(n, coeffs):
    """Flat-top window from cosine coefficients.
    w(t) = sum_k coeffs[k] * cos(pi * k * (2t - 1))
    where t = i/(n-1) for i = 0,...,n-1
    """
    t = np.linspace(-1.0, 1.0, n)
    w = np.zeros(n)
    for k, c in enumerate(coeffs):
        w += c * np.cos(np.pi * k * t)
    w = np.maximum(w, 1e-15)
    return w / w.sum()

def _optimize_flattop_coeffs(n, coeffs_init, deadline):
    """Optimize the cosine coefficients of a flat-top window."""
    def obj(params):
        w = _make_flattop_window(n, params)
        return _score_norm_fn(w)
    
    # Use Nelder-Mead for the low-dimensional coefficient optimization
    result = minimize(obj, coeffs_init, method='Nelder-Mead',
                      options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12})
    return result.x, result.fun

def _peak_equalize_coordinate_descent(a_init, deadline, rng, max_iters=50000):
    """
    Coordinate descent that specifically targets peak equalization.
    At each step, identify the top K lags and find the coordinate update
    that most evenly reduces them.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    K = 3  # Number of top lags to target
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        max_c = c.max()
        
        # Find top K lags
        top_lags = np.argsort(c)[-K:]
        
        # Compute the "pressure" on each coordinate from the top lags
        # For coordinate j, its contribution to lag k is a[j] * a[k-j]
        # ∂c_k/∂a_j = a[k-j] + a[j-k] (for valid indices)
        
        # We want to decrease a_j where it contributes most to the peaks
        # and increase where it contributes least
        
        pressure = np.zeros(n)
        for k in top_lags:
            k = int(k)
            for j in range(n):
                partner1 = k - j
                partner2 = j - k
                if 0 <= partner1 < n:
                    pressure[j] += a[partner1]
                if 0 <= partner2 < n and partner2 != partner1:
                    pressure[j] += a[partner2]
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure /= p_max
        
        # Move mass from high-pressure to low-pressure coordinates
        # Use a soft threshold: reduce high-pressure coords, increase low-pressure
        target = np.zeros(n)
        high_mask = pressure > 0.7
        low_mask = pressure < 0.3
        
        if high_mask.any() and low_mask.any():
            # Amount to move
            move_frac = 0.01 * (1.0 + 0.5 * np.sin(it * 0.01))  # Oscillating rate
            total_high = a[high_mask].sum()
            total_low_slots = low_mask.sum()
            
            if total_high > 1e-15 and total_low_slots > 0:
                per_high = total_high * move_frac / high_mask.sum()
                per_low = total_high * move_frac / total_low_slots
                
                a_trial = a.copy()
                a_trial[high_mask] -= per_high
                a_trial[high_mask] = np.maximum(a_trial[high_mask], 0)
                a_trial[low_mask] += per_low
                a_trial /= a_trial.sum()
                
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                else:
                    # Try smaller step
                    a_trial2 = a.copy()
                    a_trial2[high_mask] -= per_high * 0.3
                    a_trial2[high_mask] = np.maximum(a_trial2[high_mask], 0)
                    a_trial2[low_mask] += per_low * 0.3
                    a_trial2 /= a_trial2.sum()
                    sc2 = _score_norm_fn(a_trial2)
                    if sc2 < best_sc:
                        best_sc = sc2
                        best_a = a_trial2.copy()
                        a = a_trial2.copy()
        
        # Periodically increase K
        if it % 500 == 0 and K < 10:
            K += 1
    
    return best_a, best_sc

        # Strategy D: Mirror perturbation (exploit symmetry)
        else:
            # The autocorrelation is symmetric, so we can exploit this
            # Perturb a[i] and a[n-1-i] together
            i = int(rng.integers(n // 2))
            j = n - 1 - i
            a_trial = a.copy()
            delta = rng.uniform(-0.1, 0.1) * a[i]
            a_trial[i] = max(0, a_trial[i] + delta)
            a_trial[j] = max(0, a_trial[j] + delta)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc

def _lp_project(a_init, M, n):
    """
    Solve LP: max sum(g) s.t. convolve(a_init, g) <= M, g >= 0.
    Returns optimal g or None.
    """
    try:
        from scipy.optimize import linprog
        
        # Build constraint matrix
        # For each lag k: sum_j a_init[k-j] * g[j] <= M
        # where k-j is in [0, n-1]
        
        rows = []
        cols = []
        vals = []
        
        for k in range(2 * n - 1):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(a_init[k - j])
        
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        c_obj = np.ones(n)  # Maximize sum(g) => minimize -sum(g)
        
        res = linprog(-c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M), 
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 60})
        
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

def _frank_wolfe_lp(a_init, deadline, rng, max_iters=10):
    """Frank-Wolfe with LP oracle."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    c = _full_convolve(a, a)
    M = float(c.max())
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30:
            break
        
        g = _lp_project(a, M, n)
        if g is None:
            break
        
        s_g = g.sum()
        if s_g < 1e-15:
            break
        g = g / s_g  # Normalize to sum=1
        
        # Frank-Wolfe step size
        t = 2.0 / (it + 2)
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        sc = _score_norm_fn(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        c = _full_convolve(a, a)
        M = float(c.max())
        print(f"    FW iter {it}: M={M:.10f}, score={sc:.10f}, t={t:.3f}")
    
    return best_a, best_sc
