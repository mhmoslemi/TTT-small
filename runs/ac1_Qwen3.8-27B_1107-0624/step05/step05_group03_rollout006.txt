"""
LP-based optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Direct LP minimization of max(f*g) at each iteration,
combined with symmetry exploitation and multi-scale refinement.
"""
import numpy as np
from scipy.sparse import csr_matrix, lil_matrix
import cvxpy as cp
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Score assuming sum=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_parabola(n):
    t = np.linspace(0, 1, n)
    a = 4 * t * (1 - t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_fourth_power(n):
    t = np.linspace(0, 1, n)
    a = (4 * t * (1 - t)) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_sinc_like(n):
    t = np.linspace(-1, 1, n)
    a = np.sinc(0.5 * t) ** 2  # sinc^2 = (sin(pi*x)/(pi*x))^2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_gaussian(n, sigma=0.3):
    t = np.linspace(-1, 1, n)
    a = np.exp(-t**2 / (2 * sigma**2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_bessel(n, order=2):
    from scipy.special import jv
    t = np.linspace(-1, 1, n)
    a = np.abs(jv(order, np.pi * t)) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _make_palindromic(a):
    """Make a palindromic: a[i] = a[n-1-i]."""
    n = len(a)
    return (a + a[::-1]) / 2.0

def _build_conv_matrix_sparse(n, f):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] for full convolution."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A

def _lp_min_conv_peak(f, solver_list=None, timeout=60.0):
    """
    Solve: min t subject to (f*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns optimal g and optimal t.
    """
    n = len(f)
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK, cp.ECOS]
    
    try:
        A = _build_conv_matrix_sparse(n, f)
        g = cp.Variable(n, non_neg=True)
        t = cp.Variable()
        constraints = [A @ g <= t, cp.sum(g) == 1]
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        t0 = time.monotonic()
        for solver in solver_list:
            try:
                if time.monotonic() - t0 > timeout:
                    break
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    if g.value is not None:
                        g_val = np.maximum(g.value, 0.0)
                        g_val = g_val / g_val.sum()
                        t_val = float(t.value)
                        return g_val, t_val
            except Exception:
                continue
        
        # Fallback: use the current f
        return f.copy() / f.sum(), float(_full_convolve(f, f).max())
    except Exception:
        return f.copy() / f.sum(), float(_full_convolve(f, f).max())

def _lp_min_conv_peak_palindromic(f, solver_list=None, timeout=60.0):
    """
    LP restriction to palindromic sequences.
    If n is even, we have n/2 free variables.
    If n is odd, we have (n+1)/2 free variables.
    """
    n = len(f)
    f_pal = _make_palindromic(f)
    n_half = (n + 1) // 2  # number of free variables
    
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK, cp.ECOS]
    
    try:
        # g[i] = g[n-1-i], so g is determined by g[0], ..., g[n_half-1]
        # (f_pal * g_pal)[k] = sum_j f_pal[j] * g_pal[k-j]
        # Since both are palindromic, the convolution is also palindromic.
        # We only need constraints for k = 0, ..., n-1 (half the lags).
        
        # For palindromic g: g[j] = f_pal[n-1-j]... no, g[j] = g[n-1-j]
        # So g[j] for j >= n_half is determined: g[j] = g[n-1-j]
        
        # Let's build the constraint matrix for the half-lags
        # (f_pal * g)[k] for k = 0, ..., n-1
        # = sum_j f_pal[j] * g[k-j]
        # where g[k-j] = g[n-1-(k-j)] if k-j >= n_half
        
        # This gets complicated. Let me just build the full matrix and add symmetry constraints.
        
        A = _build_conv_matrix_sparse(n, f_pal)
        g = cp.Variable(n, non_neg=True)
        t = cp.Variable()
        
        # Symmetry constraints: g[i] == g[n-1-i]
        sym_constraints = []
        for i in range(n // 2):
            sym_constraints.append(g[i] == g[n - 1 - i])
        
        constraints = [A @ g <= t, cp.sum(g) == 1] + sym_constraints
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        t0 = time.monotonic()
        for solver in solver_list:
            try:
                if time.monotonic() - t0 > timeout:
                    break
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    if g.value is not None:
                        g_val = np.maximum(g.value, 0.0)
                        g_val = _make_palindromic(g_val)
                        g_val = g_val / g_val.sum()
                        t_val = float(t.value)
                        return g_val, t_val
            except Exception:
                continue
        
        return f_pal / f_pal.sum(), float(_full_convolve(f_pal, f_pal).max())
    except Exception:
        f_pal = _make_palindromic(f)
        return f_pal / f_pal.sum(), float(_full_convolve(f_pal, f_pal).max())

def _lp_iteration(f, alpha, solver_list, timeout):
    """
    One LP iteration: solve LP, then blend.
    f: current sequence (sum=1)
    alpha: blend factor (1.0 = full update, smaller = conservative)
    """
    g, t_opt = _lp_min_conv_peak(f, solver_list, timeout)
    
    # Blend
    f_new = (1.0 - alpha) * f + alpha * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy(), float(_full_convolve(f, f).max())
    f_new = f_new / s
    
    c = _full_convolve(f_new, f_new)
    M = float(c.max())
    return f_new, M

def _lp_continuation(f_init, deadline, rng, max_iters=100, solver_list=None):
    """
    LP-based iterative method with continuation on alpha.
    Start with small alpha (conservative), increase over time.
    """
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK]
    
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_M = float(_full_convolve(f, f).max())
    best_score = 2.0 * n * best_M
    
    print(f"  LP-Cont start: M={best_M:.10f}, score={best_score:.10f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break
        
        # Alpha schedule: start small, increase
        if it < 10:
            alpha = 0.1 + 0.05 * it
        elif it < 30:
            alpha = 0.6 + 0.01 * (it - 10)
        else:
            alpha = 0.8
        
        t0 = time.monotonic()
        f_new, M_new = _lp_iteration(f, alpha, solver_list, timeout=120.0)
        elapsed = time.monotonic() - t0
        
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            best_score = 2.0 * n * best_M
            print(f"  LP iter {it}: M={M_new:.10f}, score={best_score:.10f}, alpha={alpha:.2f}, time={elapsed:.1f}s")
        
        f = f_new.copy()
    
    return best_f, best_score

def _lp_direct(f_init, deadline, rng, solver_list=None):
    """
    Direct LP: don't blend, just use the LP solution directly.
    """
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK]
    
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_M = float(_full_convolve(f, f).max())
    best_score = 2.0 * n * best_M
    
    print(f"  LP-Direct start: M={best_M:.10f}, score={best_score:.10f}")
    
    for it in range(50):
        if time.monotonic() > deadline - 30.0:
            break
        
        t0 = time.monotonic()
        g, t_opt = _lp_min_conv_peak(f, solver_list, timeout=120.0)
        elapsed = time.monotonic() - t0
        
        M_g = float(_full_convolve(g, g).max())
        
        if M_g < best_M:
            best_M = M_g
            best_f = g.copy()
            best_score = 2.0 * n * best_M
            print(f"  LP-Direct iter {it}: M={M_g:.10f}, score={best_score:.10f}, t_opt={t_opt:.10f}, time={elapsed:.1f}s")
        
        f = g.copy()
    
    return best_f, best_score

def _subgradient_method(f_init, deadline, rng, lr=0.01, n_subgrads=5):
    """
    Subgradient method with averaging for the non-convex problem.
    Use multiple random subgradients (from top-K lags) for variance reduction.
    """
    n = len(f_init)
    a = f_init.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    # Momentum
    v = np.zeros(n)
    beta = 0.9
    lr_cur = lr
    
    max_iters = 200000
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        
        # Get top K lags for subgradient
        K = min(n_subgrads, len(c))
        top_idx = np.argsort(c)[-K:]
        
        # Compute subgradient (average over top K lags)
        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            for j in range(lo, hi + 1):
                partner = k_val - j
                if 0 <= partner < n:
                    grad[j] += a[partner]
        grad *= 2.0 / K
        
        # Gradient step with momentum
        v = beta * v + (1 - beta) * grad
        a_new = a - lr_cur * v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        # Check improvement
        if it % 10 == 0:
            sc = _score_norm_fn(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        a = a_new
        lr_cur *= 0.9999
    
    return best_a, best_sc

def _mirror_descent(f_init, deadline, rng, lr=0.05, K=3):
    """
    Mirror descent on the simplex (multiplicative weights).
    Update: a_i <- a_i * exp(-lr * grad_i) / Z
    """
    n = len(f_init)
    a = f_init.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    max_iters = 100000
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break
        
        c = _full_convolve(a, a)
        K_actual = min(K, len(c))
        top_idx = np.argsort(c)[-K_actual:]
        
        # Compute "gradient" (which direction to increase/decrease)
        # We want to decrease a_i where it contributes to peaks
        contrib = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            for j in range(lo, hi + 1):
                partner = k_val - j
                if 0 <= partner < n:
                    contrib[j] += a[partner]
        contrib /= K_actual
        
        # Multiplicative update: reduce where contrib is high
        # a_i <- a_i * (1 - lr * contrib_i / max_contrib)
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        a = a * (1.0 - lr * contrib / max_c)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        if it % 5 == 0:
            sc = _score_norm_fn(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
        # Decay learning rate
        if it % 1000 == 0:
            lr *= 0.95
    
    return best_a, best_sc

def _logsumexp_optimize(a_init, deadline, rng, beta_start=None, n_levels=15):
    """
    Optimize the smooth approximation: 2n * logsumexp(conv(a,a)/beta)
    using L-BFGS. Continuation on beta.
    """
    from scipy.optimize import minimize
    
    n = len(a_init)
    # Parameterize in log-space with softmax
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_start is None:
        beta_start = max_c0
    
    betas = beta_start * np.exp(np.linspace(0, np.log(0.001), n_levels))
    betas = np.sort(betas)[::-1]  # decreasing
    
    per_level = max(5.0, (deadline - time.monotonic()) / n_levels)
    
    for beta in betas:
        if time.monotonic() > deadline - 10.0:
            break
        
        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            shifted = (c_ - cm) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm
            return 2.0 * n * (cm + beta * np.log(s))
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm = float(c_.max())
            shifted = (c_ - cm) / beta
            e = np.exp(shifted)
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e / s
            W = float(w @ c_)
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        level_deadline = min(deadline - 5, time.monotonic() + per_level)
        
        try:
            t0 = time.monotonic()
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 50, 'maxfun': 20000}
            )
            elapsed = time.monotonic() - t0
            
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
                    print(f"    LSE beta={beta:.6f}: score={best_sc:.10f}, time={elapsed:.1f}s")
        except Exception as ex:
            print(f"    LSE beta={beta:.6f} failed: {ex}")
    
    return best_a, best_sc

def _pairwise_sweep(a_init, deadline, rng, n_passes=50):
    """
    Coordinate descent: sweep over pairs (i, j) and adjust the ratio a[i]/a[j].
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    c = _full_convolve(a, a)
    
    for p in range(n_passes):
        if time.monotonic() > deadline - 5.0:
            break
        
        # Get top lags
        K = 3
        top_lags = np.argsort(c)[-K:]
        
        # For each pair, try to improve
        # Focus on pairs that contribute to top lags
        # Random order for diversity
        indices = rng.permutation(n)
        
        improved = False
        for i in indices:
            if time.monotonic() > deadline - 5.0:
                break
            
            # Find j that, when we move mass from i to j (or vice versa), improves
            # Try a few random j's
            for _ in range(5):
                j = int(rng.integers(n))
                if i == j:
                    continue
                
                # Try moving a small fraction from i to j
                for frac in [0.01, 0.05, 0.1, 0.05, 0.01]:
                    a_trial = a.copy()
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    
                    # Quick score check
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc_trial = 2.0 * n * c_trial.max()
                    if sc_trial < best_sc:
                        best_sc = sc_trial
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        c = c_trial
                        improved = True
                        break
        
        if not improved and p > 5:
            break
    
    return best_a, best_sc

def _multi_start_lp(f_inits, deadline, rng, solver_list=None):
    """Run LP method from multiple starting points."""
    if solver_list is None:
        solver_list = [cp.CBC, cp.GLPK]
    
    best_f = None
    best_sc = np.inf
    
    n = len(f_inits[0])
    n_inits = len(f_inits)
    per_init = max(30.0, (deadline - time.monotonic()) / n_inits)
    
    for i, f_init in enumerate(f_inits):
        if time.monotonic() > deadline - 30.0:
            break
        
        dl = min(deadline - 30, time.monotonic() + per_init)
        sc_init = _score_norm_fn(f_init)
        print(f"  Multi-LP init {i}/{n_inits}: initial score={sc_init:.10f}")
        
        f_best, sc = _lp_direct(f_init, dl, rng, solver_list)
        
        if sc < best_sc:
            best_sc = sc
            best_f = f_best.copy()
    
    return best_f, best_sc
