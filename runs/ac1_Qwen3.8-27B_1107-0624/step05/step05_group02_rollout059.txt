def _next_pow2(n):
    k = 1
    while k < n:

    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    arange = np.arange(1, n + 1, dtype=np.float64)
    cond = 1.0 - u / (cssv / arange) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.argmax(cond)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_smoothed_grad(y, beta, n):
    """Compute gradient of beta*log(sum_k exp(c_k/beta)) at point y."""
    c = _full_convolve(y, y)
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    c_shifted = np.clip(c_shifted, -60.0, 60.0)
    e = np.exp(c_shifted)
    w = e / e.sum()
    a_rev = y[::-1]
    conv_w_arev = _full_convolve(w, a_rev)
    grad = 2.0 * conv_w_arev[n - 1:n - 1 + n]
    return grad, c, c_max


def _compute_subgrad(y, n):
    """Compute exact subgradient at argmax point."""
    c = _full_convolve(y, y)
    k_star = int(c.argmax())
    grad = np.zeros(n)
    lo = max(0, k_star - (n - 1))
    hi = min(n - 1, k_star)
    if lo <= hi:
        j = np.arange(lo, hi + 1)
        grad[j] = 2.0 * y[k_star - j]
    return grad, float(c.max())

def _subgrad_descent(a_init, deadline, verbose=True):

    n = len(a_init)

def _interp_profile(p, n_new):
    n0 = len(p)

    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            beta_end=0.0005 * float(a_init.max()),
            verbose=True
        )
        update_best(a_fista)

    # Subgradient refinement
    if best_seq is not None and time.monotonic() < phase1_end - 10:
        a_init = best_seq / best_seq.sum()
        a_sg, sc_sg = _subgrad_descent(
            a_init, time.monotonic() + (phase1_end - time.monotonic() - 5),
            verbose=True
        )
        update_best(a_sg)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Multi-restart FISTA (500s - 1100s) ===
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 2: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 5
        per_restart = (phase2_end - time.monotonic() - 30) / max(1, n_restarts)
        sigmas = [0.02, 0.04, 0.08, 0.12, 0.20]

        for i in range(n_restarts):
            if time.monotonic() > phase2_end - 20:
                break
            p_start = _perturb(a_best_norm, sigmas[i], rng)
            deadline_i = time.monotonic() + min(per_restart, phase2_end - time.monotonic() - 10)
            a_r, sc_r = _fista_run(
                p_start, deadline_i,
                beta_start=0.02 * float(p_start.max()),
                beta_end=0.0005 * float(p_start.max()),
                verbose=(i == 0)
            )
            update_best(a_r)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam fine-tuning (1100s - 1400s) ===
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 3: Adam fine-tuning ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_init = best_seq / best_seq.sum()
        a_adam, sc_adam = _adam_logspace(
            a_init, time.monotonic() + (phase3_end - time.monotonic() - 20),
            lr=0.005, verbose=True
        )
        update_best(a_adam)

        # Second Adam run with different LR
        if time.monotonic() < phase3_end - 60:
            a_adam2, sc_adam2 = _adam_logspace(
                a_init, time.monotonic() + (phase3_end - time.monotonic() - 10),
                lr=0.002, verbose=False
            )
            update_best(a_adam2)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-length exploration (1400s - 1650s) ===
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Multi-length exploration ===")

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        target_ns = [4096, 16384]
        for target_n in target_ns:
            if time.monotonic() > phase4_end - 80:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            remaining = phase4_end - time.monotonic() - 40
            if remaining > 100:
                a_r, _ = _fista_run(
                    p_interp, time.monotonic() + remaining,
                    beta_start=0.02 * float(p_interp.max()),
                    beta_end=0.0005 * float(p_interp.max()),
                    verbose=False
                )
                update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish (1650s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_final, _ = _adam_logspace(
                a_best_norm, time.monotonic() + remaining,
                lr=0.001, verbose=False
            )
            update_best(a_final)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Final validation and return ===
    if best_seq is None:
        best_seq = _make_uniform(256)

    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = _make_uniform(256)

    sc_final = _score(final)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval check: {ex}")

    return [float(x) for x in final]
