def _spectral_perturb(a, rng, strength=0.1):
    """Perturb a in the frequency domain."""
    n = len(a)
    N = _next_pow2(2*n-1)
    A = np.fft.rfft(a, N)
    # Perturb the magnitude but keep phase
    mag = np.abs(A)
    phase = np.angle(A)
    # Add random modulation to magnitude
    noise = strength * rng.standard_normal(len(mag))
    mag_new = np.maximum(mag * (1.0 + noise), 0.0)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _block_swap(a, rng, n_blocks=2):
    """Swap two random contiguous blocks."""
    n = len(a)
    for _ in range(n_blocks):
        start1 = rng.integers(0, n // 2)
        length1 = rng.integers(1, n // 4)
        end1 = min(start1 + length1, n)
        start2 = rng.integers(n // 2, n - 1)
        length2 = min(length1, n - start2)
        end2 = start2 + length2
        # Swap
        block1 = a[start1:end1].copy()
        block2 = a[start2:end2].copy()
        a[start1:end1] = block2[:end1-start1]
        a[start2:end2] = block1[:end2-start2]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing, active-set subgradient with BB step,
momentum acceleration, and spectral diversification for escaping local minima.

import numpy as np
import time
from scipy.optimize import minimize
from functools import partial

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    h = wc[len(w) - 1: len(w) - 1 + n]
    W = float(np.dot(w, c))
    grad_a = 4.0 * n * a * (h - W)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z

def _smoobj_wrapper(z, beta, n):
    return _smoobj(z, beta, n)

def _smograd_wrapper(z, beta, n):
    return _smograd(z, beta, n)

def _active_set_pressure(a, n, eps):
    """Compute pressure vector h and weight W for active set subgradient."""
    c = _full_convolve(a, a)
    c_max = c.max()
    active = c >= (c_max - eps * c_max)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    m = float(active.sum())
    w = np.zeros(len(c))
    w[active] = 1.0 / m
    w_rev = w[::-1]
    wc = _full_convolve(w_rev, a)
    h = wc[len(w) - 1: len(w) - 1 + n]
    W = float(np.dot(w, c))
    return h, W, c, m

def _subgrad_momentum(a_init, deadline, max_steps=5000, eps_init=0.05, eps_final=0.001, lr_init=0.08, momentum=0.7):
    """Subgradient in z-space with momentum and adaptive eps."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    z_prev = z.copy()
    lr = lr_init
    no_improve_count = 0

            break

        # Adaptive epsilon: start wide, tighten over time
        progress = min(1.0, step / max(1, max_steps * 0.6))
        eps = eps_init + (eps_final - eps_init) * progress

        a = _softmax(z)
        h, W, c, m = _active_set_pressure(a, n, eps)
        grad_z = 4.0 * n * a * (h - W)

        # Momentum step
        z_new = z - lr * grad_z + momentum * (z - z_prev)
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()
        z_prev = z
        z = z_new

            sc = _score(a_check)
            if sc < best_sc - 1e-14:

                best_a = a_check.copy()
                no_improve_count = 0
                lr = min(0.5, lr * 1.02)
            else:

                lr *= 0.9
                if no_improve_count > 20:
                    break

    # Final evaluation
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()

def _subgrad_bb(a_init, deadline, max_steps=3000, eps=0.005):
    """Subgradient in z-space with Barzibil-Borwein step size."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    z_prev = z.copy()
    grad_prev = None
    alpha = 0.1

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        h, W, c, m = _active_set_pressure(a, n, eps)
        grad_z = 4.0 * n * a * (h - W)

        # BB step size
        if grad_prev is not None:
            s = z - z_prev
            y = grad_z - grad_prev
            sy = float(np.dot(s, y))
            yy = float(np.dot(y, y))
            if yy > 1e-20 and sy > 1e-20:
                alpha = min(0.5, max(1e-6, sy / yy))
            elif yy > 1e-20:
                alpha *= 0.5

        z_new = z - alpha * grad_z
        z_new = np.clip(z_new, -80.0, 80.0)
        z_new = z_new - z_new.mean()

        # Backtracking check every 10 steps
        z_prev = z
        z = z_new

            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:

                best_a = a_check.copy()

    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

def _spectral_perturb(a, rng, strength=0.1):

    n = len(a)
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    phase = np.angle(A)
    noise = strength * rng.standard_normal(len(mag))
    mag_new = np.maximum(mag * (1.0 + noise), 0.0)
    # Ensure DC component is positive
    if len(mag_new) > 0:
        mag_new[0] = abs(mag_new[0])
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s

def _fourier_perturb(a, rng, n_modes=3, amp=0.05):
    """Add random Fourier modes in z-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 8))
        a_mode = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + a_mode * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)

def _block_swap(a, rng, n_blocks=2):
    """Swap random contiguous blocks."""
    a = a.copy()
    n = len(a)
    for _ in range(n_blocks):
        start1 = int(rng.integers(0, max(1, n // 3)))
        length1 = int(rng.integers(1, max(2, n // 6)))
        end1 = min(start1 + length1, n)
        start2 = int(rng.integers(n // 3, max(n // 3 + 1, n - 1)))
        length2 = min(length1, n - start2)
        end2 = start2 + length2
        if end1 > start1 and end2 > start2:
            b1 = a[start1:end1].copy()
            b2 = a[start2:end2].copy()
            a[start1:end1] = b2[:end1 - start1]
            a[start2:end2] = b1[:end2 - start2]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def _gaussian_perturb(a, sigma, rng):
    """Gaussian perturbation in z-space."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _run_lbfgs(z_init, n, betas, deadline, min_time_remaining=30.0):
    """Run L-BFGS with a sequence of beta values."""
    z = z_init.copy()
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break
        est_time_per_iter = 0.015 * (n / 1024.0)
        max_iter = min(3000, max(30, int(remaining / (est_time_per_iter * 10))))
        f = partial(_smoobj_wrapper, beta=beta, n=n)
        g = partial(_smograd_wrapper, beta=beta, n=n)
        try:
            result = minimize(
                f, z, jac=g, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-13,
                         'maxcor': 25, 'maxls': 80}
            )
            z = result.x
        except Exception as e:
            print(f"    L-BFGS error beta={beta:.2e}: {e}")
    return z

def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer combining L-BFGS, subgradient, and diversification."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] improve={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        a0 = parent.copy()
        n = len(a0)
        sc0 = _score(a0)
        print(f"Parent loaded: n={n}, score={sc0:.10f}")
        _update_best(a0, sc0, "parent")
    else:
        n = 2048
        a0 = np.ones(n) / n
        sc0 = _score(a0)
        print(f"No parent. Uniform init: n={n}, score={sc0:.10f}")
        _update_best(a0, sc0, "init_uniform")

    # Phase 1: L-BFGS with log-sum-exp continuation
    print(f"\nPhase 1: L-BFGS log-sum-exp (t={time.monotonic()-t_start:.1f}s)")
    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.mean()

    betas_main = max_c0 * np.array([0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001])
    z1 = _run_lbfgs(z0, n, betas_main, deadline, min_time_remaining=400.0)
    a1 = _softmax(z1)
    _update_best(a1, _score(a1), "lbfgs_main")

    # Phase 2: Momentum subgradient polishing
    print(f"\nPhase 2: Momentum subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None:
        a2, sc2 = _subgrad_momentum(best_seq, deadline - 300.0, max_steps=8000, eps_init=0.03, eps_final=0.001, lr_init=0.06, momentum=0.7)
        _update_best(a2, sc2, "subgrad_momentum")

    # Phase 3: BB subgradient
    print(f"\nPhase 3: BB subgradient (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a3, sc3 = _subgrad_bb(best_seq, deadline - 150.0, max_steps=3000, eps=0.003)
        _update_best(a3, sc3, "subgrad_bb")

    # Phase 4: Diversification with re-optimization
    print(f"\nPhase 4: Diversification (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        strategies = [
            ("spec_05", None),
            ("spec_10", None),
            ("fourier", None),
            ("gauss_02", None),
            ("block2", None),
            ("spec_20", None),
        ]
        n_div = min(len(strategies), max(2, int((deadline - 60 - time.monotonic()) / 80)))
        for i in range(n_div):
            if time.monotonic() > deadline - 50:
                break
            if i == 0:
                a_start = _spectral_perturb(best_seq, rng, strength=0.05)
            elif i == 1:
                a_start = _spectral_perturb(best_seq, rng, strength=0.10)
            elif i == 2:
                a_start = _fourier_perturb(best_seq, rng, n_modes=3, amp=0.04)
            elif i == 3:
                a_start = _gaussian_perturb(best_seq, 0.02, rng)
            elif i == 4:
                a_start = _block_swap(best_seq, rng, n_blocks=2)
            else:
                a_start = _spectral_perturb(best_seq, rng, strength=0.20)

            sc_start = _score(a_start)
            label = strategies[i][0]
            print(f"  {label}: start={sc_start:.10f}")

            # Quick optimization: momentum subgradient
            a_opt, sc_opt = _subgrad_momentum(a_start, deadline - 50.0, max_steps=2000, eps_init=0.02, eps_final=0.001, lr_init=0.04, momentum=0.6)
            _update_best(a_opt, sc_opt, label)

            # If good, also try L-BFGS from here
            if sc_opt < best_score + 0.005 and time.monotonic() < deadline - 40:
                z_start = np.log(np.maximum(a_opt, 1e-15))
                z_start = z_start - z_start.mean()
                c_opt = _full_convolve(a_opt, a_opt)
                mc = float(c_opt.max())
                betas_quick = mc * np.array([0.01, 0.005, 0.002, 0.001])
                z_opt = _run_lbfgs(z_start, n, betas_quick, deadline, min_time_remaining=30.0)
                a_lb = _softmax(z_opt)
                _update_best(a_lb, _score(a_lb), label + "_lbfgs")

    # Phase 5: Final fine-tuning with very tight active set
    print(f"\nPhase 5: Final fine-tune (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a5, sc5 = _subgrad_momentum(best_seq, deadline - 2.0, max_steps=3000, eps_init=0.005, eps_final=0.0005, lr_init=0.02, momentum=0.8)
        _update_best(a5, sc5, "final_fine")

        # One more L-BFGS with tiny beta
        if time.monotonic() < deadline - 3:
            z_fin = np.log(np.maximum(best_seq, 1e-15))
            z_fin = z_fin - z_fin.mean()
            c_fin = _full_convolve(best_seq, best_seq)
            mc_fin = float(c_fin.max())
            betas_fin = mc_fin * np.array([0.0001, 0.00005, 0.00002])
            z_fin2 = _run_lbfgs(z_fin, n, betas_fin, deadline, min_time_remaining=2.0)
            a_fin2 = _softmax(z_fin2)
            _update_best(a_fin2, _score(a_fin2), "final_lbfgs")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
