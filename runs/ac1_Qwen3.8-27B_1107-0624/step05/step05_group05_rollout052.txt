L-BFGS with log-sum-exp smoothing for minimizing the autocorrelation peak.
Uses a coarse-to-fine temperature schedule and multiple restarts.

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Linear convolution via FFT."""
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _obj_grad(z, beta, N, n):
    """
    Compute the log-sum-exp smoothed objective and its gradient w.r.t. z.
    a = softmax(z), c = conv(a,a), f = log(sum exp(c/beta))
    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    e = np.exp((c - c_max) / beta)
    s_e = float(e.sum())
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse = c_max / beta
    else:
        w = e / s_e
        lse = c_max / beta + np.log(s_e)
    
    # Gradient w.r.t. a: grad_a[i] = (2/beta) * sum_m w[m+i] * a[m]
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad_a = (2.0 / beta) * corr
    
    # Gradient w.r.t. z (softmax chain rule):
    # grad_z = a * (grad_a - a @ grad_a)
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    
    # Clean up numerical issues
    grad_z = np.clip(grad_z, -1e6, 1e6)
    
    return lse, grad_z

        return lse, grad

                       options={'maxiter': maxiter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': maxiter * 3})
        return res.x, res.nit
    except Exception as e:
        print(f"    L-BFGS exception: {e}")
        return z_init, 0

"""
L-BFGS with log-sum-exp smoothing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a coarse-to-fine temperature (beta) schedule, multiple L-BFGS runs,
and random restarts to escape local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _obj_grad(z, beta, N, n):
    """Compute log-sum-exp(c/beta) objective and gradient w.r.t. z (softmax param)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s_e = float(e.sum())
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse = c_max / beta
    else:
        w = e / s_e
        lse = c_max / beta + np.log(s_e)
    # grad_a[i] = (2/beta) * sum_m w[m+i] * a[m] = (2/beta) * cross_corr(w, a)[i]
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)[:n]
    grad_a = (2.0 / beta) * corr
    # Chain rule through softmax: grad_z = a * (grad_a - a.dot(grad_a))
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)
    return lse, grad_z

    """Run L-BFGS on smoothed objective, return (z, n_iterations)."""
    bounds = [(-60.0, 60.0)] * n

    def packed(z):
        lse, grad = _obj_grad(z,
