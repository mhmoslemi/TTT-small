"""
LP-based peak reduction with adaptive step size for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses direct LP to find optimal descent direction for the max peak,
combined with log-sum-exp continuation for smoothness.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

        pass
    return None


def _lp_phase(a_init, deadline, max_rounds=50):
    """
    Iterative LP-based peak reduction.
    Each round: solve LP to find direction, take step, check improvement.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, _norm_score(a)
    a = a / s
    best_a = a.copy()
    best_sc = _norm_score(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        current_max = float(c.max())
        current_sc = 2.0 * n * current_max

        # Try several target fractions (from conservative to aggressive)
        found_improvement = False
        for tf in [0.999, 0.998, 0.995, 0.99, 0.98, 0.97, 0.95]:
            K = 30 if tf > 0.99 else 50 if tf > 0.97 else 80

            d = _lp_direction(a, tf, K, time_limit=15)
            if d is None:
                continue

            # Line search: find best alpha in [0, 1]
            a_trial_best = None
            sc_best = current_sc
            alphas = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
            for alpha in alphas:
                a_trial = a + alpha * d
                if np.any(a_trial < -1e-10):
                    continue
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_norm = a_trial / s_trial
                sc_trial = _norm_score(a_norm)
                if sc_trial < sc_best - 1e-14:
                    sc_best = sc_trial
                    a_trial_best = a_norm

            if a_trial_best is not None:
                a = a_trial_best
                if _norm_score(a) < best_sc - 1e-14:
                    best_sc = _norm_score(a)
                    best_a = a.copy()
                found_improvement = True
                break

        if not found_improvement:
            # Try with more peaks
            d = _lp_direction(a, 0.99, 150, time_limit=15)
            if d is not None:
                for alpha in [0.5, 0.3, 0.2, 0.1]:
                    a_trial = np.maximum(a + alpha * d, 0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    a_norm = a_trial / s_trial
                    sc_trial = _norm_score(a_norm)
                    if sc_trial < best_sc - 1e-14:
                        best_sc = sc_trial
                        best_a = a_norm.copy()
                        a = a_norm
                        found_improvement = True
                        break

            if not found_improvement:
                break

        if rnd % 5 == 0:
            print(f"    LP round {rnd}: sc={best_sc:.10f}")

    return best_a, best_sc


def _softmax_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmax_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _norm_score(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmax_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmax_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _norm_score(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _subgradient_phase(a_init, deadline, lr_init=0.001, max_iter=100000):
    """
    Projected subgradient method with adaptive step size.
    Projects onto simplex {a >= 0, sum(a) = 1}.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _norm_score(a)
    lr = lr_init
    no_improve = 0
    check_every = 20

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top peaks for smoothed subgradient
        top_k = 5
        peak_vals = np.argsort(c)[::-1][:top_k]
        peak_heights = c[peak_vals]
        # Softmax weights over peaks
        pw = peak_heights - peak_heights.max()
        pw = np.exp(pw * 10)
        pw = pw / pw.sum()

        # Compute smoothed subgradient
        g = np.zeros(n)
        for idx, k in enumerate(peak_vals):
            wi = pw[idx]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                g[i_lo:i_hi + 1] += wi * 2.0 * a[k - np.arange(i_lo, i_hi + 1)]

        # Step
        v = a - lr * g
        # Project onto simplex
        a = _project_simplex(v)

        if (it + 1) % check_every == 0:
            sc = _norm_score(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
                lr = min(lr * 1.1, 0.01)
            else:
                no_improve += 1
                lr *= 0.8
                if no_improve > 50:
                    # Restart with perturbation
                    a = best_a + 0.01 * np.random.randn(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    lr = lr_init * 0.5
                    no_improve = 0

    return best_a, best_sc


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure exactly sum=1 (numerical fix)
    s = w.sum()
    if s > 0:
        w = w / s
    else:
        w = np.ones(n) / n
    return w


def _adam_polish(a_init, deadline, lr=0.02):
    """Adam with log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _norm_score(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.97 ** (t // 50)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % 50 == 0:
            a_cur = _softmax_fn(z)
            sc = _norm_score(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _try_n_value(n, seed, budget, best_tracker):
    """
    Try optimizing with a specific n value.
    best_tracker: [best_seq, best_score_norm, best_score_actual]
    """
    t_start = time.monotonic()
    deadline = t_start + budget

    rng = np.random.default_rng(seed)

    # Try multiple initializations
    inits = []
    greedy = _greedy_init(n)
    inits.append(('greedy', greedy))
    for et in ['asym_exp', 'sqrt', 'cubic']:
        inits.append((et, _parametric_init(n, et)))

    best_local = None
    best_local_sc = np.inf

    for name, a_init in inits:
        if time.monotonic() > deadline:
            break

        sc_init = _norm_score(a_init)
        if best_local is None or sc_init < best_local_sc:
            best_local = a_init.copy()
            best_local_sc = sc_init

        # L-BFGS refinement
        if time.monotonic() < deadline - 60:
            a_b, sc_b = _run_lbfgs(a_init, min(deadline - 30, time.monotonic() + 120),
                                   maxiter_per_beta=3000)
            if sc_b < best_local_sc:
                best_local = a_b.copy()
                best_local_sc = sc_b

    if best_local is None:
        best_local = greedy
        best_local_sc = _norm_score(greedy)

    # LP phase
    if time.monotonic() < deadline - 120:
        a_lp, sc_lp = _lp_phase(best_local, min(deadline - 60, time.monotonic() + 300))
        if sc_lp < best_local_sc:
            best_local = a_lp.copy()
            best_local_sc = sc_lp

    # Adam polish
    if time.monotonic() < deadline - 30:
        a_ad, sc_ad = _adam_polish(best_local, deadline, lr=0.015)
        if sc_ad < best_local_sc:
            best_local = a_ad.copy()
            best_local_sc = sc_ad

    return best_local, best_local_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: LP peak reduction + gradient refinement
    + subgradient method + multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_norm_sc = np.inf
    best_actual_sc = np.inf

    def update_best(p, sc_norm, label=""):
        nonlocal best_seq, best_norm_sc, best_actual_sc
        p = np.asarray(p, dtype=np.float64)
        if p is not None and np.isfinite(sc_norm) and sc_norm < best_norm_sc:
            best_seq = p.copy()
            best_norm_sc = sc_norm
            best_actual_sc = _score_fn(p)
            print(f"  NEW BEST: norm={sc_norm:.10f}, actual={best_actual_sc:.10f} "
                  f"(n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_p = _norm_score(parent)
        print(f"Parent: n={n}, norm_score={sc_p:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n = 4096
        parent = _greedy_init(n)
        update_best(parent, _norm_score(parent), "greedy_default")

    print(f"Starting: n={n}, best_norm={best_norm_sc:.10f}")

    # === Phase 1: LP-based peak reduction on parent (400s) ===
    print("Phase 1: LP peak reduction")
    p1_end = min(deadline, t_start + 400.0)
    if time.monotonic() < p1_end:
        a_lp, sc_lp = _lp_phase(best_seq, p1_end, max_rounds=80)
        update_best(a_lp, sc_lp, "lp_phase1")

    # === Phase 2: L-BFGS continuation (300s) ===
    print("Phase 2: L-BFGS continuation")
    p2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < p2_end:
        a_lb, sc_lb = _run_lbfgs(best_seq, p2_end, maxiter_per_beta=5000)
        update_best(a_lb, sc_lb, "lbfgs_main")

        # Try from greedy init
        if time.monotonic() < p2_end - 60:
            a_g = _greedy_init(n)
            a_gb, sc_gb = _run_lbfgs(a_g, min(p2_end - 10, time.monotonic() + 100),
                                     maxiter_per_beta=3000)
            update_best(a_gb, sc_gb, "lbfgs_greedy")

    # === Phase 3: More LP rounds (300s) ===
    print("Phase 3: LP refinement")
    p3_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < p3_end:
        a_lp2, sc_lp2 = _lp_phase(best_seq, p3_end, max_rounds=80)
        update_best(a_lp2, sc_lp2, "lp_phase2")

    # === Phase 4: Subgradient method (200s) ===
    print("Phase 4: Subgradient")
    p4_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p4_end:
        a_sg, sc_sg = _subgradient_phase(best_seq, p4_end, lr_init=0.0005)
        update_best(a_sg, sc_sg, "subgradient")

    # === Phase 5: Adam polish (200s) ===
    print("Phase 5: Adam polish")
    p5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < p5_end:
        a_ad, sc_ad = _adam_polish(best_seq, p5_end, lr=0.015)
        update_best(a_ad, sc_ad, "adam_main")

        if time.monotonic() < p5_end - 30:
            a_ad2, sc_ad2 = _adam_polish(best_seq, p5_end, lr=0.005)
            update_best(a_ad2, sc_ad2, "adam_slow")

    # === Phase 6: Final LP + tiny beta L-BFGS (200s) ===
    print("Phase 6: Final refinement")
    p6_end = min(deadline - 60, t_start + 1600.0)
    if time.monotonic() < p6_end:
        a_lp3, sc_lp3 = _lp_phase(best_seq, min(p6_end, time.monotonic() + 100), max_rounds=30)
        update_best(a_lp3, sc_lp3, "lp_final")

        if time.monotonic() < p6_end - 20:
            c_b = _full_convolve(best_seq, best_seq)
            m_b = float(c_b.max())
            tiny_betas = [m_b * b for b in [0.0001, 0.00005, 0.00001]]
            a_f, sc_f = _run_lbfgs(best_seq, p6_end, beta_list=tiny_betas,
                                    maxiter_per_beta=10000)
            update_best(a_f, sc_f, "final_tiny_beta")

    # === Phase 7: Try different n if time allows ===
    if time.monotonic() < deadline - 150:
        n_current = len(best_seq)
        # Try smaller n (faster LP)
        n_small = n_current // 2
        if n_small >= 1024:
            print(f"Phase 7a: Trying n={n_small}")
            a_s, sc_s = _try_n_value(n_small, seed + 1, max(0, deadline - time.monotonic() - 60) / 2, (best_seq, best_norm_sc))
            update_best(a_s, sc_s, f"n{n_small}")

        # Try larger n
        n_large = min(n_current * 2, 16384)
        if n_large > n_current and time.monotonic() < deadline - 60:
            print(f"Phase 7b: Trying n={n_large}")
            a_interp = _interp_profile(best_seq, n_large)
            sc_in = _norm_score(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_large}")
            if time.monotonic() < deadline - 30:
                a_inb, sc_inb = _run_lbfgs(a_interp, deadline - 10, maxiter_per_beta=3000)
                update_best(a_inb, sc_inb, f"lbfgs_n{n_large}")

    # === Final validation ===
    print(f"\nFINAL: best_norm={best_norm_sc:.10f}")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(256)
    print(f"FALLBACK: n=256, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
