import time
from scipy.optimize import linprog

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1

    fb = np.fft.rfft(b, N)

Multi-scale L-BFGS optimizer with log-sum-exp smoothing for minimizing
2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.

Strategy:
1. Load parent sequence (n=4096)
2. Coarse-to-fine L-BFGS with decreasing smoothing parameter beta
3. Fine gradient polishing with adaptive step size
4. Multi-restart diversification
5. Final coordinate-level polish
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_grad(z, beta, n):
    """
    Compute smoothed objective and gradient w.r.t. z (log-softmax parameters).
    Objective: 2n * logsumexp(conv(a,a) / beta) where a = softmax(z)
    """
    a = _softmax(z)
    c = _full_convolve(a, a)
    cmax = float(c.max())
    c_shift = (c - cmax) / beta
    e = np.exp(c_shift)
    es = float(e.sum())
    if es < 1e-300:
        return 2.0 * n * cmax, np.zeros_like(z)
    lse = cmax + beta * np.log(es)
    obj = 2.0 * n * lse
    w = e / es
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    grad_a = 2.0 * h
    G = float(np.dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - G)

    """Adam optimizer in log-space for fine polishing."""
    a = _softmax(z)
    best_a = a.copy()
    best_score = float(2.0 * n * _full_convolve(a, a).max())
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > t_deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shift = (c - cm) / beta
        e = np.exp(c_shift)
        s = float(e.sum())
        if s < 1e-300:
            w = np.zeros_like(c)

def _exact_subgradient_step(a, n, eta):
    """One step of exact subgradient descent (on the max)."""
    c = _full_convolve(a, a)
    kstar = int(np.argmax(c))
    # Subgradient: s_i = db_kstar/da_i = 2*a[kstar-i] (with correction)
    s = np.zeros(n)
    for i in range(n):
        j = kstar - i
        if 0 <= j < n:
            s[i] += a[j]
        if i != j and 0 <= j < n:
            s[i] += a[j]
    # Actually: db_k/da_i = a[k-i] + a[k-i] if i != k-i, a[k-i] if i == k-i
    # = 2*a[k-i] if 2i != k, a[k-i] if 2i == k
    # Simplified: s[i] = 2*a[kstar-i] if 0 <= kstar-i < n, with the factor of 2
    # (the correction for i == kstar-i is negligible for large n)
    for i in range(n):
        j = kstar - i
        if 0 <= j < n:
            s[i] = 2.0 * a[j]
        else:
            s[i] = 0.0
    # Update in log-space: z_i -= eta * s[i] * (some scaling)
    # But we need to be careful: the subgradient of max is not the gradient
    # of the softmax parameterization. Let me just do: a_i *= exp(-eta * s[i])
    a_new = a * np.exp(-eta * s / a.mean())
    a_new = np.maximum(a_new, 1e-15)
    a_new = a_new / a_new.sum()
    return a_new
