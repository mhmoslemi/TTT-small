    fa = rfft(a, N)
    fb = rfft(b, N)

    if s < 1e-300:

def _lbfgs_step(m, f_grad_fn, x0, max_iter, deadline):
    """Simple L-BFGS with two-loop recursion.
    
    m: number of correction pairs to store
    f_grad_fn: function that returns (f, grad)
    x0: initial point
    max_iter: maximum iterations
    deadline: time limit
    """
    x = x0.copy()
    f, g = f_grad_fn(x)
    
    # L-BFGS memory
    s_list = []  # position differences
    y_list = []  # gradient differences
    rho_list = []  # 1 / (y^T s)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Two-loop recursion to compute search direction
        q = g.copy()
        alphas = []
        for i in range(len(s_list) - 1, -1, -1):
            alpha = rho_list[i] * np.dot(s_list[i], q)
            alphas.append(alpha)
            q = q - alpha * y_list[i]
        alphas.reverse()
        
        # Scaling: beta = (s^T y) / (s^T s) for the most recent pair
        if s_list:
            s_prev = s_list[-1]
            y_prev = y_list[-1]
            sy = np.dot(s_prev, y_prev)
            ss = np.dot(s_prev, s_prev)
            if sy > 1e-16:
                beta = sy / ss
            else:
                beta = 1.0
        else:
            beta = 1.0
        
        d = -beta * q
        
        for i in range(len(s_list)):
            beta_i = rho_list[i] * np.dot(y_list[i], d)
            d = d + (alphas[i] - beta_i) * s_list[i]
        
        # Line search (backtracking)
        alpha = 1.0
        c1 = 1e-4
        for ls in range(30):
            x_new = x + alpha * d
            f_new, g_new = f_grad_fn(x_new)
            if f_new < f + c1 * alpha * np.dot(g, d):
                break
            alpha *= 0.5
        else:
            break  # Line search failed
        
        # Update memory
        s_new = x_new - x
        y_new = g_new - g
        sy = np.dot(s_new, y_new)
        if sy > 1e-16:  # Curvature condition
            s_list.append(s_new)
            y_list.append(y_new)
            rho_list.append(1.0 / sy)
            if len(s_list) > m:
                s_list.pop(0)
                y_list.pop(0)
                rho_list.pop(0)
        
        x = x_new
        f = f_new
        g = g_new

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s
