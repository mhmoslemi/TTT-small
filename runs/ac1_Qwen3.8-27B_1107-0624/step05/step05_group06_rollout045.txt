"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS with log-sum-exp smoothing, active-set linear equalization,
iterative peeling, and multi-scale exploration for improved bounds.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_optimize(a_init, beta, n, max_iter=5000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'maxcor': 40,
                'maxls': 200,
                'disp': False,
            }
        )
        a_out = _softmax(result.x)
        return a_out, result.success, int(result.nit)
    except Exception:
        return a_init, False, 0


def _active_set_equalize(a, tol=0.002, max_iter=300, step_scale=0.05, deadline=None):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    best_a = a.copy()
    best_cmax = c_max

    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        active = c >= (c_max - tol * c_max)
        m = int(active.sum())

        if m == 0:
            break
        if m > n:
            idx = np.argsort(c)[-n:]
            active = np.zeros_like(c, dtype=bool)
            active[idx] = True
            m = n

        active_idx = np.where(active)[0]
        rhs = (c_max - c[active_idx]) / 2.0 * step_scale

        B = np.zeros((m, n))
        for idx_i, k in enumerate(active_idx):
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            B[idx_i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

        B_aug = np.vstack([B, np.ones(n)])
        rhs_aug = np.concatenate([rhs, [0.0]])

        da, _, _, _ = np.linalg.lstsq(B_aug, rhs_aug, rcond=None)

        a_new = a + da
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new * (a.sum() / s_new)

        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())

        if c_max_new < best_cmax:
            best_cmax = c_max_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            c_max = c_max_new
        else:
            step_scale *= 0.5
            if step_scale < 1e-9:
                break

    return best_a


def _iterative_peel(a, n_peels=50, threshold_frac=0.005, deadline=None):
    """Peel off peaks iteratively by reducing contributions at peak locations."""
    n = len(a)
    a = a.copy()
    c = _full_convolve(a, a)
    c_max = float(c.max())
    target = c_max * (1 - threshold_frac)

    for _ in range(n_peels):
        if deadline is not None and time.monotonic() > deadline:
            break
        if c_max <= target:
            break

        # Find positions where conv exceeds target
        exceed = c > target
        if not exceed.any():
            break

        # For each exceeding position k, we need to reduce conv(a,a)[k]
        # The most efficient way: reduce a[i] where a[k-i] is large
        # Use a simple approach: for the worst position, reduce the product terms
        worst_k = int(np.argmax(c))
        # Determine which pairs (i, k-i) contribute to c[worst_k]
        lo = max(0, worst_k - n + 1)
        hi = min(worst_k, n - 1)
        # Contribution from a[i] * a[worst_k - i] for i in [lo, hi]
        # To reduce, we can scale down a uniform amount
        contrib = a[lo:hi + 1] * a[worst_k - hi:worst_k - lo + 1][::-1]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break
        # We want to reduce c[worst_k] to target
        reduction_needed = c[worst_k] - target
        frac = min(0.5, reduction_needed / (2.0 * total_contrib))
        # Scale down both halves
        a[lo:hi + 1] *= (1 - frac)
        a[worst_k - hi:worst_k - lo + 1] *= (1 - frac)
        a = np.maximum(a, 0)

        c = _full_convolve(a, a)
        c_max = float(c.max())

    return a


def _subgradient_polish(a_init, deadline, n_max_steps=5000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + i * 0.002)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr *= 1.02
            else:
                lr *= 0.9

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 3)))
        amp = 0.01 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_symmetric(a, rng, sigma=0.02):
    """Perturb with symmetric structure to preserve some symmetry."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    # Enforce mild symmetry: average with reversed
    z = 0.5 * (z + z[::-1])
    z = z - z.mean()
    return _softmax(z)


def _upsample(a, new_n):
    """Upsample sequence from len(a) to new_n using interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    t_old = np.linspace(0, 1, old_n, endpoint=False)
    t_new = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(t_new, t_old, a)
    a_new = np.maximum(a_new, 0)
    return a_new / a_new.sum()


def _make_initial(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_schedule(a_start, deadline, min_time_remaining=5.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    current_a = a_start.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        est_iter_time = 0.005 * (n / 2048.0)
        max_iter = min(5000, max(50, int(remaining / (est_iter_time * 6))))

        a_new, success, niter = _lbfgs_optimize(current_a, beta, n, max_iter)
        sc_new = _score(a_new)
        sc_cur = _score(current_a)

        if sc_new < sc_cur:
            current_a = a_new
        else:
            # Try continuing with the same beta but fewer iterations
            if niter > 50:
                break

    return current_a

        a_dn_opt = _lbfgs_schedule(a_dn, deadline, min_time_remaining=10)
        update_best(a_dn_opt, _score(a_dn_opt), "n1024_opt")

        if time.monotonic() < deadline - 10:
            a_dn_act = _active_set_equalize(a_dn_opt, tol=0.002, max_iter=100,
                                            step_scale=0.05, deadline=deadline - 3)
            update_best(a_dn_act, _score(a_dn_act), "n1024_active")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_initial(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
