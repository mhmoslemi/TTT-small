def _np2(n):

    """Optimize shape via L-BFGS with log-sum-exp surrogates."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _sn(a_init)

    c0 = _fc(a_init, a_init)
    m0 = float(c0.max())
    fracs = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    beta_list = [f * m0 for f in fracs]

    iters_per_beta = max(200, max_total_iters // len(beta_list))

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        try:
            res = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': iters_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxls': 80,
                         'maxfun': min(50000, int(remaining * 100))}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc = _sn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
                    print(f"    LBFGS beta={beta:.4g}: score={sc:.8f}")
        except Exception:
            pass
    return best_a, best_score


def _adam_refine(a_init, deadline, lr=0.03, max_iter=200000):
    """Adam with log-sum-exp."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _sn(a_init)

    c0 = _fc(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.05 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_iv = 20

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _fc(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.05 * (0.95 ** (t // 30)))

        sh = (c - cm) / beta
        ms = float(sh.max())
        e = np.exp(sh - ms)
        w = e / e.sum()
        W = float(np.dot(w, c))
        ar = a[::-1]
        wc = _fc(w, ar)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_iv == 0:
            a_cur = _softmax_fn(z)
            sc = _sn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
                if t % (check_iv * 10) == 0:
                    print(f"    Adam it={t}: score={sc:.8f}")
            else:
                no_improve += 1
            if no_improve > 25:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _lp_shave(a, deadline, K=50, max_rounds=30, lp_time=10):
    """LP-based peak shaving for O(n) P-partitioned sequences."""
    n = len(a)
    best_a = a.copy()
    best_score = _sn(a_init) if False else _sn(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _fc(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.98 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]

        Ka = len(active)
        A_mat = np.zeros((Ka, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(Ka, M * 1.001)
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': lp_time})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = float(b_new.sum())
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = _sn(best_a)
        for tt in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - tt) * best_a + tt * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            sc_trial = _sn(a_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break
        if not improved:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale exponential optimization.
    Phase 1: Grid search over (n, c) for exponential sequences.
    Phase 2: Sub-exponential refinement.
    Phase 3: L-BFGS shape optimization.
    Phase 4: Adam refinement.
    Phase 5: LP peak shaving.
    """
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and p.size > 0 and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.8f} (n={len(p)}) [{label}] t={time.monotonic()-t0:.0f}s")

    # Load parent if available
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _sn(parent)
        if np.isfinite(sc_p):
            update_best(parent, sc_p, "parent")
            print(f"  Parent: n={len(parent)}, score={sc_p:.8f}")

    # === Phase 1: Multi-scale grid search (exp sequences) ===
    print("Phase 1: Grid search over (n, c)")
    p1_end = min(deadline, t0 + 300.0)

    n_values = [1024, 2048, 4096, 8192, 16384, 32768, 65536, 131072]
    # For each n, c_opt ≈ ln(n) - 1
    for n in n_values:
        if time.monotonic() > p1_end - 10:
            break
        c_opt = np.log(max(n, 2)) - 1.0
        # Try a range of c values around c_opt
        c_range = np.linspace(max(1.0, c_opt - 3), min(c_opt + 3, np.log(n) + 2), 15)
        for c in c_range:
            if c < 0.5:
                continue
            a = _bexp(n, c)
            sc = _sn(a)
            if np.isfinite(sc):
                update_best(a, sc, f"exp_n{n}_c{c:.1f}")

        # Also try power law
        p_opt = float(np.log(n) - 1.0)
        for p in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0, 16.0, 20.0, 30.0]:
            if time.monotonic() > p1_end - 5:
                break
            a = _bpow(n, p)
            sc = _sn(a)
            if np.isfinite(sc):
                update_best(a, sc, f"pow_n{n}_p{p:.0f}")

    # Also try shifted exponentials and geometric
    for n in [8192, 16384, 32768]:
        if time.monotonic() > p1_end - 20:
            break
        c_opt = np.log(max(n, 2)) - 1.0
        for k in [0.1, 0.5, 1.0, 2.0, 5.0]:
            a = _bexpk(n, c_opt, k)
            sc = _sn(a)
            if np.isfinite(sc):
                update_best(a, sc, f"expk_n{n}_k{k}")

    print(f"  Phase 1 done: best={best_score:.8f} t={time.monotonic()-t0:.0f}s")

    if best_seq is None:
        best_seq = _bexp(8192, 8.0)
        best_score = _sn(best_seq)
        print(f"  Fallback: n=8192, c=8, score={best_score:.8f}")

    # === Phase 2: Sub-exponential refinement ===
    print("Phase 2: Sub-exp refinement")
    p2_end = min(deadline, t0 + 500.0)
    n_best = len(best_seq)
    c_opt = np.log(max(n_best, 2)) - 1.5
    refined_seqs = []

    for c in np.linspace(c_opt - 1.5, c_opt + 1.0, 12):
        if c < 0.5 or time.monotonic() > p2_end - 10:
            continue
        for k in [0.0, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0]:
            a = _bexpk(n_best, c, k)
            sc = _sn(a)
            if np.isfinite(sc):
                refined_seqs.append((a, sc, f"c{c:.2f}_k{k}"))
                update_best(a, sc, f"subexp_{c:.2f}_k{k}")

    # Also try at n_best * 2 and n_best / 2
    for n_try in [n_best // 2, n_best * 2]:
        if n_try < 256 or n_try > 262144 or time.monotonic() > p2_end - 20:
            continue
        c_try = np.log(n_try) - 1.5
        for k in [0.0, 0.5, 2.0, 5.0]:
            a = _bexpk(n_try, c_try, k)
            sc = _sn(a)
            if np.isfinite(sc):
                update_best(a, sc, f"subexp_n{n_try}_k{k}")

    print(f"  Phase 2 done: best={best_score:.8f} t={time.monotonic()-t0:.0f}s")

    # === Phase 3: L-BFGS shape optimization ===
    print("Phase 3: L-BFGS shape optimization")
    p3_end = min(deadline, t0 + 1000.0)

    # Run LBFGS from best_seq
    if time.monotonic() < p3_end and best_seq is not None:
        a_lb, sc_lb = _lbfgs_shape(best_seq, p3_end, max_total_iters=15000)
        update_best(a_lb, sc_lb, "lbfgs_main")

    # If we have a good refined seq, try LBFGS from it too
    if refined_seqs and time.monotonic() < p3_end - 120:
        refined_seqs.sort(key=lambda x: x[1])
        a_ref = refined_seqs[0][0]
        if _sn(a_ref) < best_score * 1.05:
            a_lb2, sc_lb2 = _lbfgs_shape(a_ref, p3_end, max_total_iters=8000)
            update_best(a_lb2, sc_lb2, "lbfgs_refined")

    print(f"  Phase 3 done: best={best_score:.8f} t={time.monotonic()-t0:.0f}s")

    # === Phase 4: Adam refinement ===
    print("Phase 4: Adam refinement")
    p4_end = min(deadline, t0 + 1400.0)

    if time.monotonic() < p4_end and best_seq is not None:
        a_ad, sc_ad = _adam_refine(best_seq, p4_end, lr=0.02, max_iter=150000)
        update_best(a_ad, sc_ad, "adam_main")

        if time.monotonic() < p4_end - 60:
            a_ad2, sc_ad2 = _adam_refine(best_seq, p4_end, lr=0.005, max_iter=100000)
            update_best(a_ad2, sc_ad2, "adam_slow")

    print(f"  Phase 4 done: best={best_score:.8f} t={time.monotonic()-t0:.0f}s")

    # === Phase 5: LP peak shaving ===
    print("Phase 5: LP peak shaving")
    p5_end = deadline - 20.0

    if time.monotonic() < p5_end and best_seq is not None:
        a_lp, sc_lp = _lp_shave(best_seq, p5_end, K=50, max_rounds=20, lp_time=5)
        update_best(a_lp, sc_lp, "lp_shave")

        # Try with different K
        if time.monotonic() < p5_end - 30:
            a_lp2, sc_lp2 = _lp_shave(best_seq, p5_end, K=30, max_rounds=10, lp_time=3)
            update_best(a_lp2, sc_lp2, "lp_shave_K30")

    print(f"  Phase 5 done: best={best_score:.8f} t={time.monotonic()-t0:.0f}s")

    # === Phase 6: Try interpolation to larger n ===
    if time.monotonic() < deadline - 60 and best_seq is not None:
        n_new = min(2 * len(best_seq), 262144)
        if n_new > len(best_seq):
            print(f"Phase 6: Interpolating to n={n_new}")
            x0 = np.linspace(0.0, 1.0, len(best_seq))
            x1 = np.linspace(0.0, 1.0, n_new)
            a_interp = np.interp(x1, x0, best_seq)
            a_interp = np.maximum(a_interp, 0.0)
            s = a_interp.sum()
            if s > 1e-15:
                a_interp = a_interp / s
                sc_in = _sn(a_interp)
                update_best(a_interp, sc_in, f"interp_n{n_new}")

                if time.monotonic() < deadline - 20:
                    a_in_lb, sc_in_lb = _lbfgs_shape(a_interp, deadline - 5, max_total_iters=5000)
                    update_best(a_in_lb, sc_in_lb, f"lbfgs_n{n_new}")

    # === Final preparation ===
    if best_seq is None:
        best_seq = _bexp(4096, 7.0)
        best_score = _sn(best_seq)

    # Ensure valid output
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _bexp(1024, 5.0)
        s = float(final_seq.sum())
    # Scale to be safe (score is homogeneous, so this doesn't matter)
    if s > 0:
        final_seq = final_seq / s

    # Final score check
    sc_final = _sn(final_seq)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

    # Validate with evaluate_sequence if possible
    if len(final_seq) <= 200000:
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator confirmed: {sc_eval:.10f}")
            else:
                print(f"WARNING: Evaluator returned {sc_eval}, using local score {sc_final:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")

    return [float(x) for x in final_seq]
