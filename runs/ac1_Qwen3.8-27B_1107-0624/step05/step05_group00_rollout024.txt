"""
Multi-phase optimizer: FISTA (log-sum-exp + power-mean), Adam log-space,
coordinate local search, and block refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_grad_single(a, k):
    n = len(a)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return np.zeros(n)
    g = np.zeros(n)
    g[lo:hi + 1] = 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _smoothed_grad_lse(a, c, beta, n):
    cm = float(c.max())
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[int(k)]
        if wk < 1e-12:
            continue
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        grad[lo:hi + 1] += wk * 2.0 * a[int(k) - hi:int(k) - lo + 1][::-1]
    return grad


def _smoothed_grad_power(a, c, p, n):
    cp = np.maximum(c, 1e-300) ** p
    s = float(cp.sum())
    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = cp / s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wk = w[int(k)]
        if wk < 1e-12:
            continue
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        grad[lo:hi + 1] += wk * 2.0 * a[int(k) - hi:int(k) - lo + 1][::-1]
    return grad


def _fista_lse(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_max_init = float(_full_convolve(a_curr, a_curr).max())
    L = 16.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        grad = _smoothed_grad_lse(y, c_y, beta, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

def _fista_power(a_init, p, deadline, max_iter):

    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    c_init = _full_convolve(a_curr, a_curr)

    if not np.isfinite(L) or L < 1.0:
        L = 16.0 * n
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        grad = _smoothed_grad_power(y, c_y, p, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 50 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain = max_iter_total - t
        if remain <= 0:
            break
        it_phase = 0
        while it_phase < remain:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            e_z = np.exp(z - z.max())
            a = e_z / e_z.sum()
            c = _full_convolve(a, a)
            cm = float(c.max())
            shifted = (c - cm) / beta
            e = np.exp(shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros(len(c))
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _smoothed_grad_lse(a, c, beta, n)
            grad = 2.0 * n * a * (grad - float(np.dot(w, c) * 2.0) / max(1e-30, a.sum()))
            grad = 2.0 * n * (grad)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.3:
                update = update * (0.3 / un)

            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()

            if t % 20 == 0:
                a_chk = _softmax_z(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_phase % 10000 == 0:
                current_lr = max(0.0005, current_lr * 0.97)

    a_final = _softmax_z(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _softmax_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _coord_local_search(a, deadline, n_passes=5, delta=0.02):
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(a)
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            for d in [-delta, -delta * 0.5, delta * 0.5, delta]:
                a_try = best.copy()
                a_try[i] = max(0.0, a_try[i] * (1.0 + d))
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_norm(a_try)
                if sc < best_sc - 1e-10:
                    best = a_try
                    best_sc = sc
                    improved = True
                    break
        if not improved:
            delta *= 0.5
            if delta < 0.001:
                break
    return best, best_sc


def _block_local_search(a, deadline, block_size=16, n_passes=3):
    n = len(a)
    best = a.copy()
    best_sc = _score_norm(a)
    rng_local = np.random.default_rng(12345)
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        n_blocks = n // block_size
        block_order = rng_local.permutation(n_blocks)
        for bi in block_order:
            if time.monotonic() > deadline:
                break
            start = int(bi) * block_size
            end = start + block_size
            a_try = best.copy()
            noise = rng_local.standard_normal(block_size)
            a_try[start:end] = np.maximum(0.0, a_try[start:end] + 0.01 * noise)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            sc = _score_norm(a_try)
            if sc < best_sc - 1e-10:
                best = a_try
                best_sc = sc
                improved = True
        if not improved:
            break
    return best, best_sc


def _make_profile(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "ramp":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.8 * t
        return a / a.sum()
    elif kind == "ramp2":
        t = np.linspace(0, 1, n)
        a = 0.5 + 1.5 * t ** 2
        return a / a.sum()
    elif kind == "shifted_gauss":
        t = np.linspace(0, 1, n)
        mu = 0.35 + 0.3 * rng.random()
        a = np.exp(-2.0 * ((t - mu) ** 2) / 0.1)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "asym_sine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * t + 0.5)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_z(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score_full(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform", rng)
        sc0 = _score_full(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")

    # === Phase 1: FISTA LSE coarse-to-fine ===
    print("Phase 1: FISTA LSE")
    phase1_end = min(deadline, t_start + 350.0)
    a_curr = a0.copy()
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas = []
    beta = 0.03 * max_cc
    while beta > 0.0005 * max_c0:
        betas.append(beta)
        beta *= 0.6
    for b in betas:
        if time.monotonic() > phase1_end - 10:
            break
        per_t = min(60.0, phase1_end - time.monotonic() - 5)
        max_it = max(2000, int(per_t * 3000))
        a_curr, sc = _fista_lse(a_curr, b, time.monotonic() + per_t, max_it)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score_full(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA beta={b:.6f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: FISTA Power-mean ===
    print("Phase 2: FISTA Power-mean")
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end - 30:
        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())
        for p in [30, 60, 120, 250]:
            if time.monotonic() > phase2_end - 20:
                break
            per_t = min(50.0, phase2_end - time.monotonic() - 5)
            max_it = max(2000, int(per_t * 2000))
            a_curr, sc = _fista_power(a_curr, p, time.monotonic() + per_t, max_it)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score_full(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  Power p={p}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 2: best={best_score:.10f}")

    # === Phase 3: Adam log-space fine ===
    print("Phase 3: Adam log-space")
    phase3_end = min(deadline, t_start + 1100.0)
    a_work = best_seq.copy()
    a_work = np.clip(a_work, 1e-15, 1000.0)
    a_work = a_work / a_work.sum()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    betas_fine = []
    beta = 0.001 * max_cw
    while beta > 0.000001 * max_c0:
        betas_fine.append(beta)
        beta *= 0.5
    if len(betas_fine) > 12:
        betas_fine = betas_fine[:12]

    if time.monotonic() < phase3_end - 20:
        per_t = phase3_end - time.monotonic() - 15
        a_opt, sc_opt = _adam_logspace(a_work, time.monotonic() + per_t, betas_fine, 0.01, 800000)
        sc_opt_full = _score_full(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 3: best={best_score:.10f}")

    # === Phase 4: Coordinate local search ===
    print("Phase 4: Coordinate local search")
    phase4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase4_end - 20:
        a_cs = best_seq.copy()
        a_cs = a_cs / a_cs.sum()
        per_t = phase4_end - time.monotonic() - 10
        a_cs, sc_cs = _coord_local_search(a_cs, time.monotonic() + per_t, n_passes=8, delta=0.015)
        sc_cs_full = _score_full(a_cs)
        if sc_cs_full < best_score:
            best_score = sc_cs_full
            best_seq = a_cs.copy()
            print(f"  Coord search: score={sc_cs_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 4: best={best_score:.10f}")

    # === Phase 5: Multi-start exploration ===
    print("Phase 5: Multi-start")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 60:
        profiles = ["ramp", "asym_sine", "shifted_gauss", "ramp2"]
        per_restart = (phase5_end - time.monotonic() - 30) / len(profiles)
        for pi, prof in enumerate(profiles):
            if time.monotonic() > phase5_end - 20:
                break
            a_ps = _make_profile(n_main, prof, rng)
            # Interpolate from best
            a_ps = 0.5 * a_ps + 0.5 * _interp_profile(best_seq, n_main)
            a_ps = np.maximum(a_ps, 1e-10)
            a_ps = a_ps / a_ps.sum()

            c_ps = _full_convolve(a_ps, a_ps)
            max_cs = float(c_ps.max())
            b_ps = [0.003 * max_cs, 0.001 * max_cs, 0.0003 * max_cs]
            per_t = max(5.0, per_restart - 10)
            a_ps_opt, _ = _adam_logspace(a_ps, time.monotonic() + per_t, b_ps, 0.008, 100000)
            sc_ps = _score_full(a_ps_opt)
            if sc_ps < best_score:
                best_score = sc_ps
                best_seq = a_ps_opt.copy()
                print(f"  Profile '{prof}': score={sc_ps:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 5: best={best_score:.10f}")

    # === Phase 6: Final ultra-fine polish ===
    print("Phase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 1e-15, 1000.0)
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00003 * max_cp, 0.00001 * max_cp, 0.000003 * max_cp]
        per_t = deadline - time.monotonic() - 8
        if per_t > 20:
            a_pol, sc_pol = _adam_logspace(a_pol, time.monotonic() + per_t, betas_ult, 0.003, 300000)
            sc_pol_full = _score_full(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Quick block search
        if time.monotonic() < deadline - 5:
            a_blk = best_seq.copy()
            a_blk = a_blk / a_blk.sum()
            a_blk, sc_blk = _block_local_search(a_blk, time.monotonic() + max(5, deadline - time.monotonic() - 3), block_size=32, n_passes=2)
            sc_blk_full = _score_full(a_blk)
            if sc_blk_full < best_score:
                best_score = sc_blk_full
                best_seq = a_blk.copy()
                print(f"  Block search: score={sc_blk_full:.10f}")

    # Final validation
    print(f"Final best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score_full(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
