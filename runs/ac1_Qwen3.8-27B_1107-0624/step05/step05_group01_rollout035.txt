"""
Multi-phase optimization of 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses LP-based iterative peak balancing, L-BFGS log-sum-exp continuation,
peak-directed Nesterov-accelerated gradient descent, and restart strategies.
Key improvements: tighter LP slack, better line search, momentum-based gradient phase,
peak-weighted convolution for fast gradient computation.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_s1(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat

    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_s1(a_init)
    S = float(best_a.sum())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.92 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 150:
            active = sorted_idx[:150]

        A_mat = _build_lp_matrix(best_a, active, n)
        b_vec = np.full(len(active), M * slack)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        current_score = _score_norm_s1(best_a)
        improved = False
        # Line search over t
        t_vals = np.linspace(0.02, 1.0, 20)
        best_t = 0.0
        best_sc = current_score
        for t in t_vals:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = float(2.0 * n * c_trial.max())
            if sc_trial < best_sc:
                best_sc = sc_trial
                best_t = t

        if best_t > 0 and best_sc < current_score - 1e-14:
            a_trial = (1.0 - best_t) * best_a + best_t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            best_a = a_trial.copy()
            best_score = best_sc
            improved = True

        if not improved and rnd > 10:
            # Try with fewer peaks
            if K > 10:
                K = max(10, K // 2)
            continue
        if not improved and K <= 10:
            # Try slightly larger active set
            K = min(50, K + 10)
            if rnd > 50:
                break

    return best_a, best_score


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=8000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_s1(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002,
                 0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003,
                 0.00002, 0.00001, 0.000005, 0.000003, 0.000002, 0.000001,
                 0.0000005, 0.0000003, 0.0000002, 0.0000001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-25,
                         'gtol': 1e-16, 'maxls': 200, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_s1(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _peak_grad_momentum(a_init, deadline, K=5, lr_init=0.5, decay=0.9999):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_s1(a_init)

    v = np.zeros(n)  # momentum
    lr = lr_init
    no_improve = 0
    check_interval = 20
    a_prev = a_init.copy()

    a_cur = _softmax_fn(z)
    z_prev = z.copy()

    while time.monotonic() < deadline:
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Top-K peaks
        sorted_idx = np.argsort(c)[::-1][:K]
        w = np.zeros(2 * n - 1)
        w[sorted_idx] = 1.0 / K

        # grad_a = 2 * conv(w, rev(a)) truncated to n
        a_rev = a_cur[::-1]
        w_conv = _full_convolve(w, a_rev)
        grad_a = 2.0 * w_conv[n - 1:2 * n - 1]

        # Correct for diagonal terms (where k-i = i, i.e. 2i = k)
        # For each peak k, if k is even, the term at i=k/2 should be a_i not 2*a_i
        # This is a small correction; skip for speed

        # Gradient w.r.t. z (softmax parameterization)
        Wa = float(np.dot(a_cur, grad_a))
        grad_z = a_cur * (grad_a - Wa)

        if not np.all(np.isfinite(grad_z)):
            break

        # Nesterov momentum
        v = 0.9 * v + grad_z
        z = z - lr * v
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        # Decay learning rate
        lr *= decay

        # Adaptive restart if no improvement
        check_interval += 1
        if check_interval >= 50:
            check_interval = 0
            a_check = _softmax_fn(z)
            sc_check = _score_norm_s1(a_check)
            if np.isfinite(sc_check) and sc_check < best_score - 1e-14:
                best_score = sc_check
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 20:
                # Restart momentum, increase lr slightly
                v = np.zeros(n)
                lr = max(lr * 2.0, 0.01)
                no_improve = 0

    return best_a, best_score


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _parametric_init(n, etype='asym_exp'):
    t = np.linspace(0, 1, n)
    if etype == 'asym_exp':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'pow3':
        a = (0.05 + 0.95 * t) ** 3
    elif etype == 'broad':
        a = 1.0 / (1.0 + 5.0 * np.abs(t - 0.5))
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: LP peak balancing, L-BFGS continuation,
    peak-directed momentum gradient descent, and restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_s1(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_norm(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _greedy_init(n)
        update_best(parent, _score_norm_s1(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: LP-based peak balancing (aggressive, ~500s) ===
    print("Phase 1: LP peak balancing (tight)")
    p1_end = min(deadline, t_start + 500.0)

    a_lp, sc_lp = _lp_phase(best_seq, p1_end, K=30, max_rounds=500, slack=1.00005)
    update_best(a_lp, sc_lp, "lp_tight")

    # Second pass with different K
    if time.monotonic() < p1_end - 30:
        a_lp2, sc_lp2 = _lp_phase(best_seq, min(p1_end - 10, time.monotonic() + 120),
                                   K=15, max_rounds=200, slack=1.0001)
        update_best(a_lp2, sc_lp2, "lp_K15")

    # Third pass with broader active set
    if time.monotonic() < p1_end - 20:
        a_lp3, sc_lp3 = _lp_phase(best_seq, min(p1_end - 5, time.monotonic() + 100),
                                   K=60, max_rounds=100, slack=1.0002)
        update_best(a_lp3, sc_lp3, "lp_K60")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: L-BFGS log-sum-exp continuation (~500s) ===
    print("Phase 2: L-BFGS continuation")
    p2_end = min(deadline, t_start + 1000.0)

    a_f, sc_f = _run_lbfgs(best_seq, p2_end, maxiter_per_beta=10000)
    update_best(a_f, sc_f, "lbfgs_main")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak-directed momentum gradient (~350s) ===
    print("Phase 3: Peak-directed momentum")
    p3_end = min(deadline, t_start + 1350.0)

    for K_mom in [3, 5, 8]:
        if time.monotonic() > p3_end - 20:
            break
        a_mom, sc_mom = _peak_grad_momentum(best_seq, min(p3_end - 10, time.monotonic() + 120),
                                            K=K_mom, lr_init=0.3, decay=0.99995)
        update_best(a_mom, sc_mom, f"momentum_K{K_mom}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: LP refinement after momentum (~200s) ===
    print("Phase 4: LP refinement")
    p4_end = min(deadline, t_start + 1550.0)

    if time.monotonic() < p4_end:
        a_lp4, sc_lp4 = _lp_phase(best_seq, p4_end, K=20, max_rounds=100, slack=1.00003)
        update_best(a_lp4, sc_lp4, "lp_refine1")

        if time.monotonic() < p4_end - 30:
            a_lp5, sc_lp5 = _lp_phase(best_seq, p4_end, K=40, max_rounds=50, slack=1.0001)
            update_best(a_lp5, sc_lp5, "lp_refine2")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: LP again after second momentum phase (~200s) ===
    print("Phase 5: Second momentum + LP")
    p5_end = min(deadline, t_start + 1750.0)

    if time.monotonic() < p5_end - 60:
        a_mom2, sc_mom2 = _peak_grad_momentum(best_seq, min(p5_end - 60, time.monotonic() + 100),
                                              K=3, lr_init=0.1, decay=0.99998)
        update_best(a_mom2, sc_mom2, "momentum_fine")

        if time.monotonic() < p5_end - 20:
            a_lp6, sc_lp6 = _lp_phase(best_seq, p5_end, K=25, max_rounds=50, slack=1.00003)
            update_best(a_lp6, sc_lp6, "lp_final")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try different n if time allows ===
    if time.monotonic() < deadline - 120 and best_seq is not None:
        for n_try in [int(n * 1.5), int(n / 1.5)]:
            if time.monotonic() > deadline - 60:
                break
            n_try = max(512, min(n_try, 16384))
            if n_try == n:
                continue
            print(f"Phase 6: Trying n={n_try}")
            a_interp = _interp_profile(best_seq, n_try)
            sc_in = _score_norm_s1(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_try}")
            if time.monotonic() < deadline - 30:
                a_in_b, sc_in_b = _lp_phase(a_interp, deadline - 10, K=20, max_rounds=30, slack=1.0001)
                update_best(a_in_b, sc_in_b, f"lp_n{n_try}")

    # === Phase 7: Final tiny LP polish ===
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_final_lp, sc_final_lp = _lp_phase(best_seq, deadline, K=20, max_rounds=50, slack=1.00001)
        update_best(a_final_lp, sc_final_lp, "lp_final_polish")

    print(f"  All phases done: best={best_score:.10f}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_norm(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_norm(fb):.10f}")
    return [float(x) for x in fb]
