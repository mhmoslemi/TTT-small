```python
"""
Smoothed gradient descent with periodic LP refinement for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Uses softmax-weighted subgradient with
annealing temperature, periodic LP steps for large jumps, and multi-restart.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z, T=1.0):
    z = z / T
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _softmax_weights(c, T):
    """Compute softmax weights over convolution entries with temperature T."""
    c_shifted = (c - c.max()) / max(T, 1e-10)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _compute_gradient(a, w):
    """Compute gradient: g(j) = 2 * sum_k w(k) * a(k-j) for valid indices."""
    n = len(a)
    # w has length 2n-1
    w_rev = w[::-1]
    # convolve a with w_rev, take the middle n entries
    conv_aw = _full_convolve(a, w_rev)
    # The gradient at position j is 2 * (a * w_rev)(n-1+j) ... let me think
    # Actually: g(j) = 2 * sum_k w(k) * a(k-j)
    # This is 2 * (w * a)(j) where the convolution is linear
    # But we need to be careful about the indexing
    # (w * a)(j) = sum_k w(k) a(j-k) ... hmm
    # Let me just compute it directly using the FFT approach
    
    # g(j) = 2 * sum_k w(k) * a(k-j) for 0 <= k-j <= n-1
    # = 2 * sum_{k=j}^{j+n-1} w(k) * a(k-j) (for k in [0, 2n-2])
    # Let i = k-j, then i goes from 0 to n-1 (when k = j+i)
    # g(j) = 2 * sum_{i=0}^{n-1} w(j+i) * a(i)
    # This is 2 * (w_shifted * a)(0) where w_shifted[j:] = w[j:]
    # Actually it's 2 * cross_correlation(w, a) evaluated at lag j
    # = 2 * sum_i w(j+i) * a(i)
    
    # Using FFT: this is the cross-correlation of w and a
    # cross_corr(w, a)[j] = sum_i w(j+i) * conj(a(i))
    # For real signals: = sum_i w(j+i) * a(i)
    
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    # cross[j] = sum_i w(i+j) * a(i) for the cyclic convolution
    # We need the linear version, which is cross[0:n] for j = 0..n-1
    return 2.0 * cross[:n]


def _lp_solve(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)(k) <= M for all k, b >= 0."""
    # Build sparse matrix
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            rows_list.append(np.full(len(j_range), k, dtype=np.int32))
            cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_step(f, M, n):
    """One LP step: find best direction, line search."""
    c = _full_convolve(f, f)
    M_current = float(c.max())
    g_0 = _lp_solve(f, M_current, n)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M_current
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, _score_norm(f)


def _smoothed_grad_step(a, T, eta, n):
    """One step of smoothed gradient descent in probability space."""
    c = _full_convolve(a, a)
    w = _softmax_weights(c, T)
    g = _compute_gradient(a, w)
    # Multiplicative update
    a_new = a * np.exp(-eta * g / max(g.max(), 1e-15))
    s = a_new.sum()
    if s < 1e-30:
        return a
    a_new = a_new / s
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new


def _log_adam_step(z, t, T, lr, beta1, beta2, eps, m_buf, v_buf):
    """Adam step in log-space with smoothed max."""
    a = np.exp(z - np.max(z))
    a = a / a.sum()
    n = len(a)
    c = _full_convolve(a, a)
    w = _softmax_weights(c, T)
    g = _compute_gradient(a, w)
    # Gradient in log-space: d/dz_j score = a_j * (g_j - sum(a*g)/sum(a))
    # Actually for the normalized problem, the gradient of score w.r.t. z is:
    # a_j * g_j - a_j * sum(a_k * g_k)
    W = float(np.dot(a, g))
    grad_z = a * (g - W)
    
    m_buf = beta1 * m_buf + (1 - beta1) * grad_z
    v_buf = beta2 * v_buf + (1 - beta2) * grad_z ** 2
    mh = m_buf / (1 - beta1 ** t)
    vh = v_buf / (1 - beta2 ** t)
    update = lr * mh / (np.sqrt(vh) + eps)
    
    # Clip update
    update_norm = np.linalg.norm(update)
    if update_norm > 1.0:
        update = update * (1.0 / update_norm)
    
    z_new = z - update
    z_new = np.clip(z_new, -30, 30)
```

```python
"""
Smoothed gradient descent with periodic LP refinement for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Uses softmax-weighted subgradient with
annealing temperature in log-space (Adam), periodic LP steps for large jumps,
top-K active set tracking, and multi-restart perturbation.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_weights(c, T):
    """Compute softmax weights over convolution entries with temperature T."""
    c_shifted = (c - c.max()) / max(T, 1e-10)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _topk_weights(c, K):
    """Uniform weights on top K entries of c."""
    K = min(K, len(c))
    idx = np.argpartition(c, -K)[-K:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / K
    return w


def _compute_gradient(a, w):
    """Compute gradient: g(j) = 2 * sum_k w(k) * a(k-j) for valid indices."""
    n = len(a)
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * cross[:n]


def _lp_solve(f, M, n):
    """Solve LP: max sum(b) s.t. (f*b)(k) <= M for all k, b >= 0."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            rows_list.append(np.full(len(j_range), k, dtype=np.int32))
            cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 45})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_step(f, n):
    """One LP step: find best direction, line search."""
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _lp_solve(f, M, n)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, _score_norm(f)


def _log_adam_step(z, t, T, lr, beta1, beta2, eps, m_buf, v_buf, use_topk=False, topk=10):
    """Adam step in log-space with smoothed max objective."""
    a = np.exp(z - np.max(z))
    a = a / a.sum()
    n = len(a)
    c = _full_convolve(a, a)
    if use_topk:
        w = _topk_weights(c, topk)
    else:
        w = _softmax_weights(c, T)
    g = _compute_gradient(a, w)
    W = float(np.dot(a, g))
    grad_z = a * (g - W)

    if not np.all(np.isfinite(grad_z)):
        return z, m_buf, v_buf, float(_score_norm(a))

    m_buf = beta1 * m_buf + (1 - beta1) * grad_z
    v_buf = beta2 * v_buf + (1 - beta2) * grad_z ** 2
    mh = m_buf / (1 - beta1 ** t)
    vh = v_buf / (1 - beta2 ** t)
    update = lr * mh / (np.sqrt(vh) + eps)
    update_norm = np.linalg.norm(update)
    if update_norm > 1.0:
        update = update * (1.0 / update_norm)

    z_new = z - update
    z_new = np.clip(z_new, -25, 25)
    return z_new, m_buf, v_buf, float(_score_norm(a))


def _perturb_log(z, sigma, rng):
    return z + sigma * rng.standard_normal(len(z))


def _mul_weight_step(a, T, eta, n):
    """Multiplicative weight update step."""
    c = _full_convolve(a, a)
    w = _softmax_weights(c, T)
    g = _compute_gradient(a, w)
    g_norm = float(np.dot(a, g))
    a_new = a * np.exp(-eta * (g - g_norm))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-30:
        return a
    a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    c_init = _full_convolve(a, a)
    c_scale = float(c_init.max())
    print(f"Initial convolution max: {c_scale:.8f}")

    # Phase 1: LP warmup (0 - 80s)
    phase1_end = min(deadline, t_start + 80.0)
    sc = _score_norm(a)
    lp_count = 0
    prev_sc = sc
    stall = 0
    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a, sc = _lp_step(a, n)
        lp_count += 1
        dt = time.monotonic() - t0
        if sc < prev_sc - 1e-10:
            stall = 0
        else:
            stall += 1
        prev_sc = sc
        if lp_count % 3 == 0 or stall > 10:
            full_sc = _score(a)
            print(f"  LP[{lp_count}] norm={sc:.10f} full={full_sc:.10f} dt={dt:.1f}s t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
        if stall > 15:
            print(f"  LP stalled at iter {lp_count}")
            break

    full_sc = _score(a)
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Main optimization - smoothed gradient descent with annealing (80s - 1050s)
    phase2_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase2_end:
        a_work = best_seq.copy()
        z = np.log(np.maximum(a_work, 1e-30))
        z = z - z.mean()
        m_buf = np.zeros(n)
        v_buf = np.zeros(n)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        t_adam = 0
        prev_best_p2 = best_score
        no_improve_count = 0
        lr = 0.015
        eval_interval = 500

        # Multi-cycle annealing
        num_cycles = 6
        for cyc in range(num_cycles):
            if time.monotonic() >= phase2_end:
                break
            # Temperature: geometric anneal per cycle
            T_start = c_scale * 0.005 * (0.4 ** cyc)
            T_end = c_scale * 0.0001 * (0.4 ** cyc)
            T = max(T_start, c_scale * 0.000005)
            # Learning rate decreases with cycles
            cyc_lr = lr * (0.9 ** cyc)

            # Alternate between softmax and top-K every 2000 iters
            use_topk_toggle = False

            while time.monotonic() < phase2_end:
                t0 = time.monotonic()
                # Anneal temperature within cycle
                T = max(T * 0.9995, T_end * 0.5)

                use_topk = (t_adam % 4000) < 1000  # 25% of time use top-K
                z, m_buf, v_buf, sc_norm = _log_adam_step(
                    z, t_adam + 1, T, cyc_lr, beta1, beta2, eps, m_buf, v_buf,
                    use_topk=use_topk, topk=8
                )
                t_adam += 1

                # Periodic evaluation
                if t_adam % eval_interval == 0:
                    a_check = np.exp(z - np.max(z))
                    a_check = a_check / a_check.sum()
                    sc = _score(a_check)
                    dt = time.monotonic() - t0
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = a_check.copy()
                        no_improve_count = 0
                        if t_adam % (eval_interval * 5) == 0:
                            print(f"  P2 cyc={cyc} it={t_adam} T={T:.6f} sc={sc:.10f} t={time.monotonic()-t_start:.0f}s")
                    else:
                        no_improve_count += 1

                    # Adaptive learning rate
                    if no_improve_count > 4:
                        cyc_lr = max(0.002, cyc_lr * 0.85)
                        no_improve_count = 0

                    # Periodic LP step every 8000 Adam iterations
                    if t_adam % 8000 == 0 and time.monotonic() < phase2_end - 60:
                        a_lp = np.exp(z - np.max(z))
                        a_lp = a_lp / a_lp.sum()
                        t_lp0 = time.monotonic()
                        a_lp, sc_lp = _lp_step(a_lp, n)
                        full_sc = _score(a_lp)
                        dt_lp = time.monotonic() - t_lp0
                        print(f"    LP in P2: sc={full_sc:.10f} dt={dt_lp:.1f}s t={time.monotonic()-t_start:.0f}s")
                        if np.isfinite(full_sc) and full_sc < best_score:
                            best_score = full_sc
                            best_seq = a_lp.copy()
                            z = np.log(np.maximum(a_lp, 1e-30))
                            z = z - z.mean()
                            m_buf = np.zeros(n)
                            v_buf = np.zeros(n)
                        else:
                            # Revert z to pre-LP state is complex; just continue

    # Final check after phase 2
    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: LP refinement from best (1050s - 1350s)
    phase3_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase3_end:
        a = best_seq.copy()
        sc = _score_norm(a)
        prev_sc = sc
        stall = 0
        lp3 = 0
        while time.monotonic() < phase3_end:
            a, sc = _lp_step(a, n)
            lp3 += 1
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if lp3 % 3 == 0:
                full_sc = _score(a)
                print(f"  LP3[{lp3}] sc={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
            if stall > 20:
                print(f"  LP3 stalled at iter {lp3}")
                break
            prev_sc = sc
        print(f"Phase 3 done: {lp3} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Fine polishing with very low temperature (1350s - 1620s)
    phase4_end = min(deadline, t_start + 1620.0)
    if time.monotonic() < phase4_end:
        a = best_seq.copy()

        m_buf = np.zeros(n)
        v_buf = np.zeros(n)

    phase5_end = min(deadline, t_start + 1770.0)
```
