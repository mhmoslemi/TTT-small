```python

Optimized step function sequence search using log-sum-exp continuation,
active-set equalization, and multi-scale refinement.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

# Helper functions (top-level, no closures)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score_norm(a):
    """Compute 2*n*max(conv(a,a)) - the numerator part."""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def score(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def build_conv_matrix(f):
    """Build sparse matrix A such that (A @ g)_k = (conv(f,g))_k."""
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def lp_minimax(f, timeout=30):
    """Solve the inner LP: max sum(g) s.t. conv(f,g) <= t, sum(g)=1, g>=0. Minimize t."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, timeout=timeout)
        if g.value is not None and t.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None

def lp_iterate(f, n_iters, deadline, step_size=0.3):
    """Iterated LP minimax."""
    n = len(f)
    best_a = f.copy()
    best_score = score_norm(f)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = full_convolve(f, f)
        M = float(c.max())
        g, t_val = lp_minimax(f)
        if g is None:
            break
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = score_norm(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
    
    return best_a, best_score

def lse_objective(z, beta, n):
    """Log-sum-exp smoothed objective."""
    a = softmax(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def lse_gradient(z, beta, n):
    """Gradient of log-sum-exp smoothed objective."""
    a = softmax(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # Gradient of LSE w.r.t. a_j: 2 * (w * a_reversed)_j
    a_rev = a[::-1].copy()
    w_conv = full_convolve(w, a_rev)

    grad = 4.0 * n * a * (h - W)
    return grad

def run_lbfgs_continuation(a_init, deadline, beta_list=None):
    """L-BFGS with beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    
    c0 = full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002]
        beta_list = [b * max_c0 for b in fracs]
    
    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 90.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 30))
        
        try:
            result = minimize(
                lse_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=lse_gradient,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def active_set_equalize(a_init, deadline, n_peaks=10, n_iters=10000):
    """
    Try to equalize the top n_peaks positions of the convolution.
    Uses a projected subgradient method on the "excess" of peak positions.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = score_norm(a)
    
    c = full_convolve(a, a)
    M = float(c.max())
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve(a, a)
        M = float(c.max())
        
        # Find peak positions
        threshold = M * (1 - 1e-4)
        peak_mask = c >= threshold
        n_peaks_actual = peak_mask.sum()
        if n_peaks_actual == 0:
            peak_mask = np.zeros_like(c, dtype=bool)
            peak_mask[np.argmax(c)] = True
            n_peaks_actual = 1
        
        # Compute subgradient: sg_j = 2 * sum_{k in peaks} a_{k-j}
        # Using the formula: sg = 2 * (indicator_peaks * a_rev) convolved appropriately
        peak_indicator = peak_mask.astype(float) / n_peaks_actual
        a_rev = a[::-1]
        sg_conv = full_convolve(peak_indicator, a_rev)
        sg = 2.0 * sg_conv[n-1:2*n-1]
        
        # We want to minimize M = max(c), so descend in direction sg
        # But we also need to maintain sum(a) = 1 and a >= 0
        # Use: a_new = a - alpha * sg, then project and normalize
        alpha = 0.01
        a_new = a - alpha * sg
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            # Keep current a (accept the step)
            a = a_new
        else:
            # Reject and reduce step size
            alpha = max(alpha * 0.5, 1e-6)
            a_new = a - alpha * sg
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                a = a_new
    
    return best_a, best_score

def nesterov_subgradient(a_init, deadline, lr_init=0.1):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score_norm(a)
    lr = lr_init
    
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        c = full_convolve(y, y)
        M = float(c.max())
        peak_mask = c >= M * (1 - 1e-5)
        if not peak_mask.any():
            peak_mask = np.zeros_like(c, dtype=bool)
            peak_mask[np.argmax(c)] = True
        
        # Subgradient of max(c) w.r.t. y
        peak_w = peak_mask.astype(float) / peak_mask.sum()
        y_rev = y[::-1]
        sg_conv = full_convolve(peak_w, y_rev)
        sg = 2.0 * sg_conv[n-1:2*n-1]
        
        # Nesterov momentum
        t_curr = 1.0 + it
        beta = (t_curr - 1) / t_curr
        a_new = np.maximum(a - lr * sg, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        y = (1 - beta) * a_new + beta * y
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        a = a_new
        sc = score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if it % 500 == 0:
            lr = max(lr * 0.95, 1e-5)
    
    return best_a, best_score

def try_n_values(best_seq, deadline, n_values=None):
    """Try different sequence lengths by interpolation and optimization."""
    if n_values is None:
        n_values = [2048, 4096, 8192]
    
    results = {}
    for n_try in n_values:
        if time.monotonic() > deadline:
            break
        if n_try == len(best_seq):
            continue
        
        # Interpolate
        x0 = np.linspace(0, 1, len(best_seq))
        x1 = np.linspace(0, 1, n_try)
        a_try = np.interp(x1, x0, best_seq)
        a_try = np.maximum(a_try, 1e-12)
        a_try = a_try / a_try.sum()
        
        sc_try = score_norm(a_try)
        results[n_try] = (a_try, sc_try)
        
        # Quick L-BFGS optimization
        c0 = full_convolve(a_try, a_try)
        max_c0 = float(c0.max())
        beta = 0.01 * max_c0
        z = np.log(a_try)
        z = z - z.mean()
        
        try:
            result = minimize(
                lse_objective, z, args=(beta, n_try),
                method='L-BFGS-B', jac=lse_gradient,
                options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-12}
            )
            if result.x is not None:
                a_opt = softmax(result.x)
                sc_opt = score_norm(a_opt)
                if sc_opt < sc_try:
                    results[n_try] = (a_opt, sc_opt)
        except Exception:
            pass
    
    return results

def make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    return profiles

def interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def coord_descent(a_init, deadline, max_rounds=30):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = softmax(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_local = score_norm(a)
            best_delta = 0.0
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = softmax(z_trial)
                sc = score_norm(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = softmax(z)
                sc = score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True
        if not improved:
            break
    return best_a, best_score
