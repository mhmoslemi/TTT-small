def _next_pow2(n):

def _full_convolve(a, b):

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _compute_grad_lse(a, beta):
    """Compute gradient of 2n * (max_c + beta * log(sum(exp((c-max_c)/beta)))) w.r.t. a, where sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 4.0 * n * a * (h - W)
    return g, cm, c


def _compute_grad_var(a, beta, lam):
    """Gradient of 2n*(lse(c,beta)) + lam*2n*var(c) w.r.t. a."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g_lse = 4.0 * n * a * (h - W)
    # Variance gradient: var(c) = mean(c^2) - mean(c)^2
    # d(mean(c^2))/da = 2*(c * da/dc) ... 
    # c[k] = sum_j a[j]*a[k-j], dc[k]/da[i] = a[k-i] + a[k+i] (with bounds)
    # d(var(c))/da[i] = 2/(2n-1) * sum_k c[k] * dc[k]/da[i] - 4/(2n-1)^2 * (sum c) * dc[sum]/da[i]
    # sum c = (sum a)^2 = 1, d(sum c)/da[i] = 2
    # So d(var)/da[i] = 2/(2n-1) * sum_k c[k]*dc[k]/da[i] - 8/((2n-1)^2)
    # sum_k c[k]*dc[k]/da[i] = sum_k c[k]*(a[k-i]+a[k+i]) (with bounds)
    # This equals 2 * (c_conv_a)[i] where c_conv_a = conv(c, a_rev)
    c_conv_a = _full_convolve(c, a[::-1])
    h2 = c_conv_a[n - 1:2 * n - 1]
    g_var = (2.0 / (2 * n - 1)) * h2 * 4.0 * n
    return g_lse + lam * g_var, c


def _lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0, min_beta_frac=0.0001, var_lam=0.1):
    """L-BFGS with log-sum-exp continuation + variance penalty."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0007, 0.0003, 0.0001, min_beta_frac]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    per_level = max(5.0, 120.0 / len(betas))

    for li, beta in enumerate(betas):
        if time.monotonic() > deadline - 5.0:
            break
        level_dl = min(deadline - 2.0, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs_ = (c_ - cm_) / beta
            e_ = np.exp(cs_)
            s_ = e_.sum()
            lse = 0.0 if s_ < 1e-300 else float(np.log(s_))
            val = 2.0 * n * (cm_ + beta * lse)
            if var_lam > 0:
                c_mean = c_.mean()
                c_var = float(np.mean((c_ - c_mean) ** 2))
                val += var_lam * 2.0 * n * c_var
            return val

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs_ = (c_ - cm_) / beta
            e_ = np.exp(cs_)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if var_lam > 0:
                c_conv_a = _full_convolve(c_, a_[::-1])
                h2 = c_conv_a[n - 1:2 * n - 1]
                g += var_lam * (8.0 * n / (2 * n - 1)) * h2
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _spectral_equalize(a_init, deadline, rng, n_iters=500):
    """Spectral equalization: iteratively flatten the convolution in frequency domain."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break

        A = np.fft.rfft(a, N)
        C = A * np.conj(A)
        c_full = np.fft.irfft(C, N)[:2 * n - 1]
        c_mean = c_full.mean()
        c_std = c_full.std()
        if c_std < 1e-15:
            break

        # Compute correction: want c_full to be flat
        # In frequency: delta_C = -eta * rfft(c_full - c_mean)
        eta = 0.3 / (it + 10.0) * max(1.0, c_std)
        delta_C = eta * np.fft.rfft(c_full - c_mean, N)
        C_new = C - delta_C
        C_new = np.maximum(C_new, 0.0)
        # Keep DC component fixed
        C_new[0] = C[0]
        # New amplitudes
        amp_new = np.sqrt(C_new)
        # Keep phases
        phases = np.angle(A)
        A_new = amp_new * np.exp(1j * phases)
        # Reconstruct
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

    return best_a, best_score


def _peak_targeted(a_init, deadline, rng, n_trials=30000):
    """Local search targeting highest convolution peaks with smart moves."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break

        c = _full_convolve(a, a)
        K = min(5, len(c))
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])

        if it % 4 == 0:
            # Reduce the pair with largest product at the top lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                idx = pairs[int(np.argmax(products))]

                a_trial = a.copy()
                frac = rng.uniform(0.3, 0.7)
                delta = a_trial[idx] * frac
                a_trial[idx] = max(0.0, a_trial[idx] - delta)

                # Find safe position (low contribution to top lags)
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo = max(0, kl - (n - 1))
                    hi = min(n - 1, kl)
                    for j in range(lo, hi + 1):
                        partner = kl - j
                        if 0 <= partner < n:
                            contrib[j] += a[partner]

                safe = np.argsort(contrib)[:max(1, n // 4)]
                add_idx = int(safe[rng.integers(len(safe))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()

        elif it % 4 == 1:
            # Block multiplicative perturbation
            block_size = int(rng.integers(2, min(32, n // 8)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.6, 1.4, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

        elif it % 4 == 2:
            # Pair swap
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2)
            a_trial[i1] += amt2
            a_trial[i2] += amt1
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

        else:
            # Symmetric perturbation: modify a[i] and a[n-1-i] together
            i = int(rng.integers(n // 2))
            j = n - 1 - i
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3)
            a_trial[i] *= mult
            a_trial[j] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a, best_sc


def _adam_smoothed(a_init, deadline, rng, lr=0.015, k_top=5):

    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for it in range(100000):
        if time.monotonic() > deadline - 2.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]

        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_actual
        grad *= 4.0 * n

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 5000 == 0:
            lr = max(0.002, lr * 0.9)

    return best_a, best_score


def _parametric_search(n, deadline, rng):
    """Search over parametric families for good initial profiles."""
    results = []

    # Bump profiles
    for n_bumps in [2, 3, 4, 5, 6, 8]:
        for sep in [0.3, 0.4, 0.5, 0.6, 0.7]:
            for w in [0.03, 0.05, 0.08, 0.12, 0.15, 0.2]:
                p = _make_bump_profile(n, n_bumps, sep, w)
                sc = _score_norm(p)
                results.append((sc, p.copy()))
                if time.monotonic() > deadline:
                    return results

    # Edge profiles
    for gamma in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        p = _make_edge_profile(n, gamma)
        sc = _score_norm(p)
        results.append((sc, p.copy()))
        if time.monotonic() > deadline:
            return results

    # Parabolic profiles
    for curv in [-2.0, -1.5, -1.0, -0.5, -0.3, -0.1, 0.1, 0.3, 0.5, 1.0, 1.5, 2.0]:
        p = _make_parabola(n, curv)
        sc = _score_norm(p)
        results.append((sc, p.copy()))
        if time.monotonic() > deadline:
            return results

    # Sine modulation
    for freq in [1, 2, 3, 4]:
        for depth in [0.2, 0.4, 0.6, 0.8]:
            p = _make_sine_mod(n, freq, depth)
            sc = _score_norm(p)
            results.append((sc, p.copy()))
            if time.monotonic() > deadline:
                return results

    # W profiles
    for freq in [2, 3, 4, 5, 6]:
        for depth in [0.2, 0.4, 0.6, 0.8]:
            p = _make_w_profile(n, freq, depth)
            sc = _score_norm(p)
            results.append((sc, p.copy()))
            if time.monotonic() > deadline:
                return results

    # Fourier exp profiles
    for coeffs in [
        [-0.5, 0.3], [0.5, -0.3], [-1.0, 0.5], [1.0, -0.5],
        [-0.3, -0.3, 0.2], [0.3, 0.3, -0.2], [-0.8, 0.4, -0.2],
    ]:
        p = _make_fourier_exp_profile(n, coeffs)
        sc = _score_norm(p)
        results.append((sc, p.copy()))
        if time.monotonic() > deadline:
            return results

    results.sort(key=lambda x: x[0])
    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Parametric search for good initial profile
    Phase 2: L-BFGS continuation with variance regularization
    Phase 3: Spectral equalization
    Phase 4: Peak-targeted local search
    Phase 5: Adam smoothing
    Phase 6: Multi-restart
    Phase 7: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_work = 4096
        parent = _make_bump_profile(n_work, 4, 0.5, 0.1)
        sc_parent = _score_norm(parent)
        print(f"No parent, n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    print(f"Working with n={n_work}, initial best={best_score:.10f}")

    # --- Phase 1: Parametric search (100s) ---
    phase1_dl = min(deadline, t_start + 100.0)
    print("Phase 1: Parametric search")
    candidates = _parametric_search(n_work, phase1_dl, rng)
    if candidates:
        update_best(candidates[0][1], candidates[0][0], "parametric_best")
        print(f"Phase 1: top candidates: {[f'{s:.6f}' for s, _ in candidates[:5]]}")

    # Also try multi-scale: optimize at smaller n, then interpolate
    for small_n in [256, 512]:
        if time.monotonic() > phase1_dl - 10:
            break
        if parent is not None:
            p_small = _interp_profile(parent, small_n)
        else:
            p_small = _make_bump_profile(small_n, 4, 0.5, 0.1)
        # Optimize at small n
        dl_small = min(phase1_dl - 5, time.monotonic() + 15.0)
        a_opt, sc_opt = _lbfgs_continuation(p_small, dl_small, rng, beta_scale=1.0, min_beta_frac=0.001, var_lam=0.0)
        # Interpolate back
        p_up = _interp_profile(a_opt, n_work)
        sc_up = _score_norm(p_up)
        update_best(p_up, sc_up, f"multiscale_n{small_n}")

    # --- Phase 2: L-BFGS continuation (450s) ---
    phase2_dl = min(deadline, t_start + 550.0)
    print(f"Phase 2: L-BFGS continuation, starting score={best_score:.10f}")

    # Main run with variance regularization
    a_best, sc_best = _lbfgs_continuation(best_seq, phase2_dl - 60, rng,
                                           beta_scale=1.0, min_beta_frac=0.00005, var_lam=0.15)
    update_best(a_best, sc_best, "lbfgs_var")

    # Second run without variance (pure max reduction)
    if time.monotonic() < phase2_dl - 30:
        a_best2, sc_best2 = _lbfgs_continuation(best_seq, phase2_dl - 10, rng,
                                                 beta_scale=0.8, min_beta_frac=0.00003, var_lam=0.0)
        update_best(a_best2, sc_best2, "lbfgs_pure")

    # Third run with stronger variance
    if time.monotonic() < phase2_dl - 10:
        a_best3, sc_best3 = _lbfgs_continuation(best_seq, phase2_dl - 5, rng,
                                                 beta_scale=1.0, min_beta_frac=0.00005, var_lam=0.4)
        update_best(a_best3, sc_best3, "lbfgs_var2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Spectral equalization (250s) ---
    phase3_dl = min(deadline, t_start + 800.0)
    print(f"Phase 3: Spectral equalization")
    a_best, sc_best = _spectral_equalize(best_seq, phase3_dl, rng, n_iters=2000)
    update_best(a_best, sc_best, "spectral")

    # Combine: spectral then L-BFGS polish
    if time.monotonic() < phase3_dl - 20:
        a_pol, sc_pol = _lbfgs_continuation(a_best, phase3_dl - 5, rng,
                                             beta_scale=0.5, min_beta_frac=0.0001, var_lam=0.1)
        update_best(a_pol, sc_pol, "spectral+lbfgs")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Peak-targeted local search (300s) ---
    phase4_dl = min(deadline, t_start + 1100.0)
    print(f"Phase 4: Peak-targeted local search")
    a_best, sc_best = _peak_targeted(best_seq, phase4_dl, rng, n_trials=50000)
    update_best(a_best, sc_best, "peak_local")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Adam smoothing (200s) ---
    phase5_dl = min(deadline, t_start + 1300.0)
    print(f"Phase 5: Adam smoothing")
    a_best, sc_best = _adam_smoothed(best_seq, min(phase5_dl - 50, t_start + 1400.0), rng, lr=0.012, k_top=3)
    update_best(a_best, sc_best, "adam_k3")

    if time.monotonic() < phase5_dl - 20:
        a_best2, sc_best2 = _adam_smoothed(best_seq, phase5_dl - 5, rng, lr=0.008, k_top=7)
        update_best(a_best2, sc_best2, "adam_k7")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart (250s) ---
    phase6_dl = min(deadline, t_start + 1550.0)
    print(f"Phase 6: Multi-restart")
    n_restarts = 5
    per_restart = max(20.0, (phase6_dl - time.monotonic()) / max(1, n_restarts))

    for i in range(n_restarts):
        if time.monotonic() > phase6_dl - 10:
            break
        noise = 0.01 + 0.01 * i
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + noise * rng.normal(size=len(best_seq))
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        p_start = _softmax_fn(z)

        dl = min(phase6_dl - 5, time.monotonic() + per_restart)
        a_best, sc_best = _lbfgs_continuation(p_start, dl, rng,
                                               beta_scale=0.6, min_beta_frac=0.0001, var_lam=0.1)
        update_best(a_best, sc_best, f"restart{i}")
        # Quick local search after each restart
        if time.monotonic() < phase6_dl - 15:
            a_ls, sc_ls = _peak_targeted(a_best, phase6_dl - 5, rng, n_trials=5000)
            update_best(a_ls, sc_ls, f"restart{i}_ls")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final polish (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Final polish, remaining={remaining:.0f}s")

        # Very fine L-BFGS
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl, rng,
                                                   beta_scale=0.3, min_beta_frac=0.00002, var_lam=0.2)
            update_best(a_best, sc_best, "final_lbfgs")

        # Spectral + local
        if time.monotonic() < deadline - 40:
            a_sp, sc_sp = _spectral_equalize(best_seq, min(deadline - 40, time.monotonic() + remaining * 0.2), rng, n_iters=500)
            update_best(a_sp, sc_sp, "final_spectral")
            if sc_sp < best_score:
                a_ls, sc_ls = _peak_targeted(a_sp, deadline - 20, rng, n_trials=10000)
                update_best(a_ls, sc_ls, "final_spectral_ls")
            else:
                a_ls, sc_ls = _peak_targeted(best_seq, deadline - 20, rng, n_trials=10000)
                update_best(a_ls, sc_ls, "final_local")

        # Adam final
        if time.monotonic() < deadline - 20:
            a_ad, sc_ad = _adam_smoothed(best_seq, deadline - 10, rng, lr=0.003, k_top=2)
            update_best(a_ad, sc_ad, "final_adam")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_bump_profile(4096, 4, 0.5, 0.1)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_bump_profile(4096, 4, 0.5, 0.1)
    sc_fb = _score_full(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
