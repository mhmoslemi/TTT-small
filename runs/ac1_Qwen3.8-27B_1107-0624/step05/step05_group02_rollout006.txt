    Returns gradient of length n.

"""
FFT-based gradient optimization with log-sum-exp annealing, L-BFGS polishing,
and multi-n exploration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time
import math
from scipy.optimize import minimize


def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _full_convolve(a, b, N=None):
    n, m = len(a), len(b)
    out_n = n + m - 1
    if N is None:
        N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind):
    j = np.arange(n, dtype=np.float64)
    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'triangle':
        center = (n - 1) / 2.0
        a = 1.0 - np.abs(j - center) / center
        a = np.maximum(a, 0.0)
    elif kind == 'cosine':
        a = np.cos(np.pi * j / n) ** 2 + 1e-6
    elif kind == 'raised_cos':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * j / n)) + 1e-6
    elif kind == 'gaussian':
        mu = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((j - mu) / sigma) ** 2)
    elif kind == 'u_shape':
        center = (n - 1) / 2.0
        a = (np.abs(j - center) / center) ** 0.5 + 0.1
    elif kind == 'flat_cos':
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * j / n)
        a = np.maximum(a, 1e-6)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _anneal_optimize(a_init, deadline, betas=None, lr=0.01, label=""):
    """
    Adam optimizer in log-space with annealing.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    Nfft = _next_pow2(2 * n - 1)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    if betas is None:
        # Default schedule
        c0 = _full_convolve(a_init, a_init, Nfft)
        cm0 = c0.max()
        betas = [cm0 * s for s in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]

    for beta in betas:
        if time.monotonic() > deadline:
            break

        max_iters = 50000
        iter_count = 0
        best_local = best_score
        patience = 3000
        stall = 0

        while iter_count < max_iters:
            if time.monotonic() > deadline:
                break

            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a, Nfft)
            cm = c.max()

def _lbfgs_optimize(a_init, beta, deadline, label=""):
    """
    L-BFGS optimization in z-space with fixed beta.
    """
    n = len(a_init)
    Nfft = _next_pow2(2 * n - 1)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def func_grad(z):
        a = _softmax(z)
        c = _full_convolve(a, a, Nfft)
        cm = c.max()
        cs = (c - cm) / beta
        cs_max = cs.max()
        e = np.exp(cs - cs_max)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            obj = 2.0 * n * cm
            g_raw = np.zeros(n)
            g_raw[int(np.argmax(c))] = 2.0 * a[int(np.argmax(c))]  # Approximate
        else:
            w = e / s
            obj = 2.0 * n * (cm + beta * (cs_max + np.log(s)))
            Wf = np.fft.rfft(w, Nfft)
            Af = np.fft.rfft(a, Nfft)
            g_raw = np.fft.irfft(np.conj(Wf) * Af, Nfft)[:n]
            g = 2.0 * g_raw
            ga = np.dot(g, a)
            g_raw = a * (g - ga)
        
        if not np.all(np.isfinite(g_raw)):
            g_raw = np.zeros(n)
        return obj, g_raw

    t_start = time.monotonic()
    
    def callback(xk):
        if time.monotonic() > deadline:
            raise _TimeoutError()
        return False

    try:
        res = minimize(
            func_grad, z0, method='L-BFGS-B',
            jac=True,  # func returns (f, g)

                     'maxls': 50}

        a_opt = _softmax(z_opt)

        if np.isfinite(sc):

            return a_opt, sc
    except (ValueError, _TimeoutError) as e:

    except Exception as e:
        if label:
            print(f"  [{label}] L-BFGS error: {e}")

    # Fallback: return initial
    return a_init, _score(a_init)


class _TimeoutError(Exception):
    pass


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
