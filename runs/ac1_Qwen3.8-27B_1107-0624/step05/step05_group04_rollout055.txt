Then (b * w)[k] = sum_j b[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]

Let m = n-1-j, so j = n-1-m:
= sum_m a[m] * w[k-(n-1-m)] = sum_m a[m] * w[m+k-n+1]

So (b*w)[k] = h_{k-n+1} when 0 ≤ k-n+1 < n, i.e., n-1 ≤ k ≤ 2n-2.

So h = (b*w)[n-1 : 2n-1] where b = a[::-1].

The full convolution (b*w) has length n + (2n-1) - 1 = 3n-2.
We want elements at indices n-1 through 2n-2, which is n elements.

We can compute this via FFT:
N = next_pow2(3n-2)
full_bw = ifft(fft(b, N) * fft(w, N))[:3n-2]
h = full_bw[n-1 : 2n-1]

This should work! And it's O(n log n).

So the gradient is:
dF/dz_i = 4n * a_i * (h_i - c̄)
where h = (a_rev * w)[n-1:2n-1] and c̄ = sum_k w_k * conv_k = dot(w, conv)

"""
Optimization of step function sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Key insight: A linear ramp [1,2,...,n] gives asymptotic score 4/3 ≈ 1.333,
far below the 1.5027 target. We start from aggressive ramps and use
gradient-based optimization with log-sum-exp smoothing to potentially
improve beyond the 4/3 barrier. Multi-scale refinement from n=256 to n=4096.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Assumes sum(a)=1, returns 2*n*max_conv."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _objective(z, beta):
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = c.max()
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + np.log(np.sum(np.exp(shifted)))
    return float(beta * logsum)


def _objective_grad(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _fft_convolve(a, a)
    cm = c.max()
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()  # softmax weights over conv positions

    # h_i = sum_m a[m] * w[m+i] for m=0..n-1
    # = (a_rev * w)[n-1 : 2n-1]
    a_rev = a[::-1]
    N = _next_pow2(n + (2 * n - 1) - 1)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    full_bw = np.fft.irfft(fa * fw, N)[:3 * n - 2]
    h = full_bw[n - 1:2 * n - 1]

    c_bar = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - c_bar)
    return grad


def _adam_optimize(z_init, deadline, beta, lr=0.1, max_iter=500000):
    n = len(z_init)
    z = z_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_z = z.copy()
    best_obj = _objective(z, beta)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        g = _objective_grad(z, beta)
        if not np.all(np.isfinite(g)):
            break
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        obj = _objective(z, beta)
        if obj < best_obj:
            best_obj = obj
            best_z = z.copy()

        # Adaptive LR decay
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)

        # Periodic restart with small noise if stuck
        if t % 20000 == 0:
            noise_std = 0.02
            z = z + noise_std * np.random.default_rng(t).normal(size=n)
            z = z - z.mean()

    return best_z, best_obj


def _continuation_optimize(z_init, deadline, max_iters_total=200000):
    """Run Adam with decreasing beta (continuation method)."""
    a = _softmax(z_init)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return z_init

    # Start with large beta, decrease
    beta_start = 0.5 * cm
    beta_end = 1e-4 * cm
    n_steps = 12

    betas = np.geomspace(beta_start, beta_end, n_steps)
    z = z_init.copy()
    best_z = z.copy()
    best_score = _score_normalized(a)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_steps - idx
        time_per_step = (deadline - time.monotonic()) / max(1, remaining)
        step_deadline = time.monotonic() + min(time_per_step, 120.0)

        lr = 0.3 if idx < n_steps // 3 else (0.1 if idx < 2 * n_steps // 3 else 0.05)
        z, obj = _adam_optimize(z, step_deadline, beta=beta, lr=lr)
        a = _softmax(z)
        sc = _score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_z = z.copy()

    return best_z


def _make_ramp(n, p=1.0):
    """Create a power-law ramp: a_i = (i+1)^p, scaled to max=1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    a = a / a.max()
    return a / a.sum()


def _make_exponential(n, rate=2.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    best_a = a.copy()
    best_score = _score_normalized(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax(z)
        c = _fft_convolve(a_cur, a_cur)
        peak_idx = int(np.argmax(c))
        cm = c[peak_idx]

        # Find top contributors to the peak
        valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        if len(valid_i) == 0:
            break
        prod = a_cur[valid_i] * a_cur[peak_idx - valid_i]
        top_order = np.argsort(prod)[::-1]

        improved = False
        for idx_pos in top_order[:6]:
            idx = valid_i[idx_pos]
            for delta in [0.02, 0.05, 0.1, 0.2]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax(z_trial)
                sc = _score_normalized(a_trial)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
        if not improved:
            # Try increasing small elements
            small_idx = np.argsort(a_cur)[:4]
            for idx in small_idx:
                for delta in [0.02, 0.05, 0.1]:
                    z_trial = z.copy()

                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
        if not improved and rnd > 10:
            break

    return best_a
