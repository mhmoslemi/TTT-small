import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat


def _lp_round(a, M, active, slack_factor, deadline):
    """One LP round: solve LP, then line search on blend parameter."""
    n = len(a)
    S = float(a.sum())
    
    A_mat = _build_lp_matrix(a, active, n)
    b_vec = np.full(len(active), M * slack_factor)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 60, 'presolve': True})
        if res.status != 0:
            return a, M, False
        b_new = res.x
        sum_b = b_new.sum()
        if sum_b < 1e-15:
            return a, M, False
        b_scaled = b_new * (S / sum_b)
    except Exception:
        return a, M, False
    
    # Line search on t
    best_t = 0.0
    best_M_trial = M
    
    def _obj(t):
        x = (1.0 - t) * a + t * b_scaled
        x = np.maximum(x, 0.0)
        sx = float(x.sum())
        if sx < 1e-15:
            return np.inf
        x = x / sx
        c = _full_convolve(x, x)
        return float(c.max())
    
    # Coarse search
    for t_trial in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            break

        if m_trial < best_M_trial:
            best_M_trial = m_trial
            best_t = t_trial
    
    # Fine search around best_t

def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    lse = cm + beta * (max_s + float(np.log(np.sum(e))))
    return float(2.0 * n * lse)


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter=5000):

    z = np.log(np.maximum(a_init, 1e-15))

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())

    if beta_list is None:
        fracs = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                 0.00005, 0.00002, 0.00001, 0.000005, 0.000002]
        beta_list = [b * m0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,

                         'gtol': 1e-15, 'maxls': 50, 'maxfun': 20000}

            if result.x is not None and np.all(np.isfinite(result.x)):

The second projection is the hard part. But we can approximate it:

**Another idea: Use the LP dual to get information about which coordinates to adjust.**

The dual of the LP is:

subject to sum_k A[k,j] * y_k >= 1 for all j
y >= 0

where A[k,j] = a[k-j] if valid.

The dual variable y_k tells us the "importance" of constraint k. If y_k is large, constraint k is tight and important. We can use this to weight our gradient.

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Key improvements: aggressive LP with adaptive active sets and line search,
interleaved with ultra-fine L-BFGS log-sum-exp continuation, and Adam polish.
"""
import numpy as np
from scipy.optimize import minimize, minimize_scalar, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_n = a / s
    c = _full_convolve(a_n, a_n)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat


def _lp_solve(a, active, M, slack_factor, deadline):
    n = len(a)
    S = float(a.sum())
    A_mat = _build_lp_matrix(a, active, n)
    b_vec = np.full(len(active), M * slack_factor)
    try:
        res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 45, 'presolve': True})
        if res.status != 0:
            return None
        b_new = res.x
        sum_b = b_new.sum()
        if sum_b < 1e-15:
            return None
        return b_new * (S / sum_b)
    except Exception:
        return None


def _line_search_blend(a, b, deadline):
    n = len(a)
    best_t = 0.0
    best_val = np.inf

    def _eval(t):
        x = (1.0 - t) * a + t * b
        x = np.maximum(x, 0.0)
        sx = float(x.sum())
        if sx < 1e-15:
            return np.inf
        x = x / sx
        c = _full_convolve(x, x)
        return float(c.max())

    coarse_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
    for t_trial in coarse_ts:
        if time.monotonic() > deadline:
            break
        val = _eval(t_trial)
        if val < best_val:
            best_val = val
            best_t = t_trial

    if best_t > 0 and time.monotonic() < deadline - 20:
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        try:
            res_ls = minimize_scalar(_eval, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-9})
            if res_ls.success and res_ls.fun < best_val:
                best_val = float(res_ls.fun)
                best_t = float(res_ls.x)
        except Exception:
            pass

    return best_t, best_val


def _lp_phase(a_init, deadline, max_rounds, K_start, K_max, threshold, slack_factor):
    n = len(a_init)
    best_a = a_init.copy()

    c = _full_convolve(best_a, best_a)
    M = float(c.max())
    S = float(best_a.sum())
    best_score = 2.0 * n * M / (S * S)

    no_improve = 0
    rng_local = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        K = min(K_start + rnd, K_max)
        sorted_idx = np.argsort(c)[::-1]
        top_K = sorted_idx[:K]
        thr_val = M * threshold
        additional = np.where(c >= thr_val)[0]
        active = np.unique(np.concatenate([top_K, additional]))
        if len(active) > K_max:
            active = top_K[:K_max]

        b_sol = _lp_solve(best_a, active, M, slack_factor, deadline)
        if b_sol is None:
            no_improve += 1
            if no_improve >= 15:
                break
            continue

        best_t, best_M_trial = _line_search_blend(best_a, b_sol, deadline)

        if best_t > 0 and best_M_trial < M * (1.0 - 1e-11):
            a_new = (1.0 - best_t) * best_a + best_t * b_sol
            a_new = np.maximum(a_new, 0.0)
            sx = float(a_new.sum())
            if sx > 1e-15:
                a_new = a_new / sx
            c = _full_convolve(a_new, a_new)
            M_new = float(c.max())
            S_new = float(a_new.sum())
            sc_new = 2.0 * n * M_new / (S_new * S_new)
            if sc_new < best_score:
                best_score = sc_new
            best_a = a_new
            M = M_new
            S = S_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                best_a = best_a + 0.002 * rng_local.standard_normal(n)
                best_a = np.maximum(best_a, 0.0)
                best_a = best_a / best_a.sum()
                c = _full_convolve(best_a, best_a)
                M = float(c.max())
                S = float(best_a.sum())
                no_improve = 0
                if no_improve > 40:
                    break

        if rnd % 25 == 0:
            print(f"    LP[{K_start}..{K_max}] round {rnd}: M={M:.12f}, score={2.0*n*M/(S*S):.10f}")

    return best_a, best_score


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    lse = cm + beta * (max_s + float(np.log(np.sum(e))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list, maxiter):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    s0 = float(a_init.sum())
    best_score = 2.0 * n * m0 / (s0 * s0)

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter, 'ftol': 1e-25,
                         'gtol': 1e-16, 'maxls': 50, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                c_new = _full_convolve(a_new, a_new)
                m_new = float(c_new.max())
                s_new = float(a_new.sum())
                sc_new = 2.0 * n * m_new / (s_new * s_new)
                if np.isfinite(sc_new) and sc_new < best_score - 1e-15:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.003 * cm0
    s0 = float(a_init.sum())
    best_score = 2.0 * n * cm0 / (s0 * s0)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.003 * (0.94 ** (t // 200)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % 200 == 0:
            a_cur = _softmax_fn(z)
            c_cur = _full_convolve(a_cur, a_cur)
            m_cur = float(c_cur.max())
            s_cur = float(a_cur.sum())
            sc = 2.0 * n * m_cur / (s_cur * s_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.03 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score
