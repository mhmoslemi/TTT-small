    return _fft_conv(a, a)


def _compute_grad_subgrad(a, beta):
    """Compute subgradient of max(conv(a,a)) using softmax weighting over lags.
    Returns (grad, c) where grad has same shape as a, c is the convolution."""
    n = len(a)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    w = np.exp((c - c_max) / max(beta, 1e-12))
    w_s = w.sum()
    if w_s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_s
    # grad[m] = 2 * sum_j a[j] * w[m+j] for m=0..n-1
    # Using FFT: S = ifft(fft(w_ext) * conj(fft(a_ext)))[:n]
    n_w = len(w)  # 2n-1
    N = _next_pow2(n + n_w - 1)
    w_ext = np.zeros(N)
    w_ext[:n_w] = w
    a_ext = np.zeros(N)
    a_ext[:n] = a
    W = np.fft.rfft(w_ext, N)
    A = np.fft.rfft(a_ext, N)
    S = np.fft.irfft(W * np.conj(A), N)[:n]
    grad = 2.0 * S
    return grad, c


def _compute_multiscale_grad(a, betas):
    """Compute combined gradient using multiple beta values."""
    n = len(a)
    c = _fft_conv(a, a)
    c_max = float(c.max())
    total_grad = np.zeros(n)
    total_weight = 0.0
    for beta in betas:
        w = np.exp((c - c_max) / max(beta, 1e-12))
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= ws
        n_w = len(w)
        N = _next_pow2(n + n_w - 1)
        w_ext = np.zeros(N)
        w_ext[:n_w] = w
        a_ext = np.zeros(N)
        a_ext[:n] = a
        W = np.fft.rfft(w_ext, N)
        A = np.fft.rfft(a_ext, N)
        S = np.fft.irfft(W * np.conj(A), N)[:n]
        weight = 1.0 / max(beta, 1e-12)
        total_grad += weight * 2.0 * S
        total_weight += weight
    if total_weight > 1e-300:
        total_grad /= total_weight
    return total_grad, c


def _build_lp_matrix(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(f, rows, cols, max_time=30):
    n = len(f)
    c = _fft_conv(f, f)
    M = float(c.max())
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'feastol': 1e-9})
        if res.status != 0:
            return None
        g0 = np.maximum(res.x, 0.0)
        if g0.sum() < 1e-15:
            return None
        return g0
    except Exception:
        return None


def _lp_step_with_line_search(f, rows, cols, max_time=30):
    """LP step with line search for the best mixing parameter."""
    n = len(f)
    g0 = _lp_step(f, rows, cols, max_time)
    if g0 is None:
        return f
    g = g0 / g0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.3, 0.5, 0.7, 1.0, 0.15, 0.2, 0.4, 0.6]:
        fc = (1.0 - t) * f + t * g
        sc = _score_norm(fc)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t > 0:
        return (1.0 - best_t) * f + best_t * g
    return f


def _mirror_descent_optimize(a_init, deadline, lr=0.01, momentum=0.9,
                             beta_start=None, beta_end=None, max_iter=1000000):
    """Fast mirror descent with heavy-ball momentum in log-space."""
    n = len(a_init)
    c_init = _fft_conv(a_init, a_init)
    c_max_init = float(c_init.max())
    if beta_start is None:
        beta_start = 0.01 * c_max_init
    if beta_end is None:
        beta_end = 0.0005 * c_max_init

    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    t = 0
    stall = 0
    prev_best = best_sc
    check_interval = max(1, n // 4)

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        # Adaptive beta
        progress = min(1.0, t / max_iter)
        beta = beta_start * (beta_end / beta_start) ** progress
        grad_a, c = _compute_grad_subgrad(a, beta)
        # Center the gradient (project onto tangent space of simplex)
        W = float(np.dot(a, grad_a))
        grad_centered = grad_a - W
        # Heavy ball momentum
        v = momentum * v + grad_centered
        # Step in log-space
        step = lr * a * v
        step_norm = np.linalg.norm(step)
        if step_norm > 1.0:
            step = step * (1.0 / step_norm)
        z = z - step
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % check_interval == 0:
            sc = _score_norm(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            # Adaptive learning rate
            if stall > 20:
                lr = max(0.001, lr * 0.8)
                momentum = min(0.99, momentum * 1.01)
                stall = 0
            if best_sc < prev_best * 0.999:
                lr = min(0.05, lr * 1.1)
                prev_best = best_sc

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _multi_beta_optimize(a_init, deadline, max_iter=500000):
    """Optimize using multi-scale beta for more robust gradients."""
    n = len(a_init)
    c_init = _fft_conv(a_init, a_init)
    c_max_init = float(c_init.max())

    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    v_buf = np.zeros(n)
    m_buf = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.008

    n_betas = 3
    base_beta = 0.005 * c_max_init
    betas_static = [base_beta * (2 ** i) for i in range(n_betas)]

    check_interval = max(1, n // 4)
    stall = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)

        # Adaptive betas
        progress = min(1.0, t / max_iter)
        scale = (1.0 - 0.8 * progress)
        betas = [b * scale for b in betas_static]

        grad_a, c = _compute_multiscale_grad(a, betas)
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 0.8:
            update = update * (0.8 / un)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % check_interval == 0:
            sc = _score_norm(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if stall > 30:
                lr = max(0.002, lr * 0.7)
                stall = 0

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_to_length(a, n_new):
    """Interpolate sequence to new length."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _structured_init(n, kind='sine', rng=None):
    """Create structured initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'sine':
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    elif kind == 'gaussian':
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 / (2 * 0.3 ** 2))
        return a / a.sum()
    elif kind == 'random':
        if rng is None:
            rng = np.random.default_rng(0)
        a = np.abs(rng.standard_normal(n)) + 0.1
        return a / a.sum()
    elif kind == 'triangle':
        x = np.linspace(0, 1, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
        a = np.maximum(a, 1e-6)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n_parent}, score={init_sc:.10f}")
    else:
        n_parent = 2048
        a = np.ones(n_parent) / n_parent
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n_parent}, score={init_sc:.10f}")

    n = n_parent
    rows, cols = _build_lp_indices(n)
    print(f"LP matrix built: {len(rows)} nnz, t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: Fast mirror descent (adaptive beta, heavy ball) ===
    phase1_end = min(deadline, t_start + 350.0)
    print(f"\n--- Phase 1: Mirror descent ({(phase1_end - t_start):.0f}s budget) ---")
    c0 = _fft_conv(a, a)
    cmax0 = float(c0.max())
    a, sc1 = _mirror_descent_optimize(
        a, phase1_end, lr=0.015, momentum=0.92,
        beta_start=0.02 * cmax0, beta_end=0.001 * cmax0
    )
    full_sc1 = _score(a)
    print(f"Phase 1 done: score={full_sc1:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc1) and full_sc1 < best_score:
        best_score = full_sc1
        best_seq = a.copy()

    # === Phase 2: LP-guided steps ===
    phase2_end = min(deadline, t_start + 650.0)
    if time.monotonic() < phase2_end:
        print(f"\n--- Phase 2: LP-guided ({(phase2_end - t_start):.0f}s budget) ---")
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp_count = 0
        while time.monotonic() < phase2_end:
            t0 = time.monotonic()
            a_new = _lp_step_with_line_search(a, rows, cols, max_time=25)
            if np.array_equal(a_new, a):
                # LP didn't help, do a few mirror descent steps
                c_tmp = _fft_conv(a, a)
                a, _ = _mirror_descent_optimize(a, t0 + 5.0, lr=0.01, momentum=0.9,
                                                beta_start=0.005 * c_tmp.max(),
                                                beta_end=0.001 * c_tmp.max(), max_iter=5000)
            else:
                a = a_new
            sc = _score_norm(a)
            lp_count += 1
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if lp_count % 3 == 0 or stall > 15:
                full_sc = _score(a)
                print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
            if stall > 30:
                break
            prev_sc = sc
        print(f"Phase 2 done: {lp_count} iters, best={best_score:.10f}")

    # === Phase 3: Multi-beta Adam optimization ===
    phase3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < phase3_end:
        print(f"\n--- Phase 3: Multi-beta Adam ({(phase3_end - t_start):.0f}s budget) ---")
        a, sc3 = _multi_beta_optimize(best_seq, phase3_end)
        full_sc3 = _score(a)
        print(f"Phase 3 done: score={full_sc3:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc3) and full_sc3 < best_score:
            best_score = full_sc3
            best_seq = a.copy()

    # === Phase 4: Try n=1024 (shorter sequence, lower 2n factor) ===
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        print(f"\n--- Phase 4: n=1024 exploration ({(phase4_end - t_start):.0f}s budget) ---")
        n_small = 1024
        # Initialize from parent interpolated
        a_small = _interp_to_length(best_seq, n_small)
        sc_small_init = _score(a_small)
        print(f"  n=1024 init: score={sc_small_init:.10f}")
        if np.isfinite(sc_small_init) and sc_small_init < best_score:
            best_score = sc_small_init
            best_seq = a_small.copy()

        # Optimize n=1024
        c_s = _fft_conv(a_small, a_small)
        a_small, _ = _mirror_descent_optimize(
            a_small, min(deadline, t_start + 1300.0),

            beta_start=0.02 * c_s.max(), beta_end=0.002 * c_s.max()
        )
        full_sc_small = _score(a_small)
        print(f"  n=1024 optimized: score={full_sc_small:.10f}")
        if np.isfinite(full_sc_small) and full_sc_small < best_score:
            best_score = full_sc_small
            best_seq = a_small.copy()

        # Also try a random init for n=1024
        a_rand = _structured_init(n_small, 'random', rng)
        c_r = _fft_conv(a_rand, a_rand)
        a_rand, _ = _mirror_descent_optimize(
            a_rand, min(deadline, t_start + 1350.0),

        )
        full_sc_rand = _score(a_rand)
        print(f"  n=1024 random init: score={full_sc_rand:.10f}")
        if np.isfinite(full_sc_rand) and full_sc_rand < best_score:
            best_score = full_sc_rand
            best_seq = a_rand.copy()

    # === Phase 5: Multi-restart perturbation ===
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end:
        print(f"\n--- Phase 5: Multi-restart ({(phase5_end - t_start):.0f}s budget) ---")
        n_best = len(best_seq)
        rows_best, cols_best = _build_lp_indices(n_best)
        for attempt in range(5):
            if time.monotonic() > phase5_end - 20:
                break
            sigma = 0.02 + 0.015 * attempt
            p = _perturb_log(best_seq, sigma, rng)
            # Quick optimization
            c_p = _fft_conv(p, p)
            p, _ = _mirror_descent_optimize(
                p, min(deadline, t_start + 1650.0),

            p = _lp_step_with_line_search(p, rows_best, cols_best, max_time=15)
            # Quick mirror descent after LP
            c_p2 = _fft_conv(p, p)
            p, _ = _mirror_descent_optimize(
                p, min(deadline, t_start + 1680.0),
                lr=0.005, momentum=0.9,
                beta_start=0.005 * c_p2.max(), beta_end=0.0005 * c_p2.max(),
                max_iter=50000
            )
            full_sc_p = _score(p)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc_p:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc_p) and full_sc_p < best_score:

    # === Phase 6: Final ultra-fine polish ===

        full_sc_pol = _score(a_polished)
        print(f"Polished: score={full_sc_pol:.10f}")
        if np.isfinite(full_sc_pol) and full_sc_pol < best_score:

            best_seq = a_polished.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 256.0
        # Scale to ensure sum is well above 0.01
        if s < 1.0:
            final = final * (1.0 / s)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
