def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score for sum(a)=1: returns 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_fourier_profile(n, coeffs):
    """a[i] = exp(sum_j c_j * cos(2*pi*j*i/(n-1) + p_j)) normalized."""
    t = np.arange(n, dtype=np.float64)
    denom = max(1, n - 1)
    log_a = np.zeros(n)
    for j, (amp, phase) in enumerate(coeffs):
        log_a += amp * np.cos(2.0 * np.pi * (j + 1) * t / denom + phase)
    a = np.exp(log_a)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cos_profile(n, n_harm, amps, phases):
    """a[i] = 1 + sum_j amps[j]*cos(2*pi*(j+1)*i/(n-1) + phases[j]), clamped positive."""
    t = np.arange(n, dtype=np.float64)
    denom = max(1, n - 1)
    a = np.ones(n)
    for j in range(min(n_harm, len(amps))):
        if j < len(phases):
            a += amps[j] * np.cos(2.0 * np.pi * (j + 1) * t / denom + phases[j])
        else:
            a += amps[j] * np.cos(2.0 * np.pi * (j + 1) * t / denom)
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _make_u_profile(n, power):
    """U-shaped: high at edges, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.abs(u) ** power
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_bimodal(n, sep, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2.0
    c2 = 0.5 + sep / 2.0
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _make_edge_concentrated(n, alpha):
    """High at edges, decaying inward: a[i] = exp(-alpha * min(i, n-1-i) / n)."""
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i + 1, n - i)
    a = np.exp(-alpha * d / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _compute_lse_gradient(a, beta):
    """Compute gradient of LSE_beta(conv(a,a)) with respect to a.
    Returns gradient of 2n * LSE_beta(C) where C = conv(a,a)."""
    n = len(a)
    C = _full_convolve(a, a)
    C_shifted = (C - C.max()) / beta
    e = np.exp(C_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(C)
        w[int(np.argmax(C))] = 1.0
    else:
        w = e / s
    # Gradient: d/d(a[i]) LSE = (1/beta) * sum_k w[k] * dC[k]/da[i]
    # dC[k]/da[i] = a[k-i] if 0<=k-i<n else 0
    # So grad[i] = (1/beta) * sum_k w[k] * a[k-i] (for valid k-i)
    # This is cross-correlation of w with a, extracted at the right part
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    # w_conv has length len(w) + n - 1 = (2n-1) + n - 1 = 3n-2
    # We want w_conv[n-1 : 2n-1] which gives sum_k w[k] * a[k-i] for i=0..n-1
    h = w_conv[n - 1:2 * n - 1]
    grad = 2.0 * n * h / beta
    return grad, C, float(C.max())


def _lbfgs_lse(a_init, deadline, beta_starts, max_wall_per=None):
    """L-BFGS optimization with log-sum-exp continuation on log-parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a_rev_cached = a_init[::-1]

    if max_wall_per is None:
        max_wall_per = max(5.0, 300.0 / len(beta_starts))

    for beta in beta_starts:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + max_wall_per)

        def obj(z_):
            a_ = _softmax_fn(z_)
            C_ = _full_convolve(a_, a_)
            Cm_ = float(C_.max())
            Cs_ = (C_ - Cm_) / beta
            es_ = np.exp(Cs_)
            ss_ = es_.sum()
            if ss_ < 1e-300:
                return 2.0 * n * Cm_
            return 2.0 * n * (Cm_ + beta * np.log(ss_))

        def grad(z_):
            a_ = _softmax_fn(z_)
            C_ = _full_convolve(a_, a_)
            Cm_ = float(C_.max())
            Cs_ = (C_ - Cm_) / beta
            es_ = np.exp(Cs_)
            ss_ = es_.sum()
            if ss_ < 1e-300:
                w_ = np.zeros_like(C_)
                w_[int(np.argmax(C_))] = 1.0
            else:
                w_ = es_ / ss_
            a_rev_ = a_[::-1]
            w_conv_ = _full_convolve(w_, a_rev_)
            h_ = w_conv_[n - 1:2 * n - 1]
            g_ = 2.0 * n * h_ / beta
            if not np.all(np.isfinite(g_)):
                return np.zeros_like(z_)
            return g_

        try:
            result = minimize(
                obj, z.copy(), jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 80, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
                elif time.monotonic() > level_deadline:
                    break
        except Exception:
            pass

    return best_a, best_score


def _peak_equalization(a_init, deadline, n_iter=50000):
    """
    Water-filling peak equalization: iteratively reduce the highest peaks
    by moving mass from positions that contribute to them.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    K = min(10, 2 * n - 1)

    for it in range(n_iter):
        if time.monotonic() > deadline - 1.0:
            break

        C = _full_convolve(a, a)
        top_idx = np.argsort(C)[-K:]
        max_C = C[top_idx[-1]]

        # Compute contribution of each position to top lags
        # contrib[j] = sum_{k in top_idx} a[k-j] * indicator(0<=k-j<n)
        contrib = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                contrib[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]

        # Strategy: reduce positions with high contribution, add to low contribution
        # Pick a "bad" position (high contrib) and a "good" position (low contrib)
        if it % 2 == 0:
            # Move mass from highest-contrib to lowest-contrib
            bad_idx = int(np.argmax(contrib))
            good_candidates = np.argsort(contrib)[:max(1, n // 10)]
            good_idx = int(good_candidates[np.random.randint(len(good_candidates))])
            if bad_idx == good_idx:
                continue
            frac = 0.05 + 0.1 * np.abs(np.sin(it * 0.1))
            delta = a[bad_idx] * frac
            if delta < 1e-15:
                continue
            a[bad_idx] -= delta
            a[good_idx] += delta
            if a[bad_idx] < 0:
                a[good_idx] -= a[bad_idx]
                a[bad_idx] = 0.0
            s = a.sum()
            if s > 1e-10:
                a /= s
                sc = _score_norm_fn(a)
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a.copy()
        else:
            # Random perturbation on a block
            block_size = max(2, min(32, n // 32))
            start = np.random.randint(0, n - block_size)
            a_trial = a.copy()
            mult = np.random.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
            elif sc < best_score * 1.005 and np.random.random() < 0.1:
                # Accept slightly worse (simulated annealing)
                a = a_trial.copy()

    return best_a, best_score


def _dinkelbach_step(a_init, deadline, n_iter=10000):
    """
    Dinkelbach-style: binary search on target T.
    For a given T, try to find a with max(C) <= T using gradient projection.
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    # Get initial peak
    C0 = _full_convolve(a_init, a_init)
    T = float(C0.max())

    for outer in range(20):
        if time.monotonic() > deadline - 2.0:
            break

        # Try to reduce below T
        T_target = T * 0.995
        a = best_a.copy()
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()

        # Gradient descent to push max(C) below T_target
        for inner in range(n_iter // 20):
            if time.monotonic() > deadline - 2.0:
                break
            a_curr = _softmax_fn(z)
            C_curr = _full_convolve(a_curr, a_curr)
            Cm = float(C_curr.max())
            if Cm <= T_target:
                break

            # Gradient of max(C) (using top-K smoothing)
            K = min(5, len(C_curr))
            top_vals = np.sort(C_curr)[-K:]
            thresh = top_vals[0]
            # Soft indicator: how much each C[k] contributes to the max
            alpha_smooth = 0.01 * T_target
            w = np.zeros_like(C_curr)
            above = C_curr > thresh
            w[above] = (C_curr[above] - thresh) / (C_curr[above] - thresh + alpha_smooth)
            w[~above] = 0.0
            if w.sum() < 1e-10:
                w[int(np.argmax(C_curr))] = 1.0
            else:
                w /= w.sum()

            a_rev = a_curr[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            grad = 2.0 * n * h

            lr = 0.01 * T_target / (np.max(np.abs(grad)) + 1e-15)
            lr = min(lr, 0.05)
            z = z - lr * grad
            z = np.clip(z, -80, 80)
            z = z - z.mean()

        a_final = _softmax_fn(z)
        sc = _score_norm_fn(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
            C_new = _full_convolve(best_a, best_a)
            T = float(C_new.max())
        else:
            T = T * 0.99  # Relax target if we can't hit it
            if T * 2 * n < best_score * 0.95:
                break

    return best_a, best_score

            best_a = a_opt.copy()

        # Peak equalization on the result
        if sc_opt < best_score * 1.01 and time.monotonic() < dl:
            dl2 = min(deadline - 3.0, time.monotonic() + per_start * 0.3)
            a_pe, sc_pe = _peak_equalization(a_opt, dl2, n_iter=5000)
            if sc_pe < best_score:
                best_score = sc_pe
                best_a = a_pe.copy()

    return best_a, best_score


def _refine_large_n(a_small, n_large, deadline, rng):
    """Upsample a small-n solution to large n and refine."""
    a_interp = _interp_profile(a_small, n_large)
    sc = _score_norm_fn(a_interp)
    print(f"  Upsampled to n={n_large}: score={sc:.10f}")

    # L-BFGS refinement
    beta_scale = max(sc / (2 * n_large), 1e-6)
    betas = [0.1 * beta_scale, 0.05 * beta_scale, 0.02 * beta_scale,
             0.01 * beta_scale, 0.005 * beta_scale, 0.002 * beta_scale]
    dl = min(deadline, time.monotonic() + 120.0)
    a_ref, sc_ref = _lbfgs_lse(a_interp, dl, betas, max_wall_per=20.0)

    if sc_ref < sc:
        a_interp = a_ref
        sc = sc_ref

    # Peak equalization
    if time.monotonic() < deadline - 30:
        dl2 = min(deadline - 20, time.monotonic() + 60.0)
        a_pe, sc_pe = _peak_equalization(a_interp, dl2, n_iter=10000)
        if sc_pe < sc:
            a_interp = a_pe
            sc = sc_pe

    return a_interp, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer:
    1. Diverse initialization with Fourier-based profiles
    2. L-BFGS log-sum-exp continuation
    3. Peak equalization (water-filling)
    4. Dinkelbach-style ratio optimization
    5. Multi-scale refinement
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_triangular(n_parent)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using triangular n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    # === PHASE 1: Multi-start optimization at parent's size (400s) ===
    phase1_end = min(deadline, t_start + 400.0)
    n_work = n_parent
    print(f"\nPhase 1: Multi-start at n={n_work}, deadline={phase1_end - t_start:.0f}s")

    a_best_p1, sc_best_p1 = _multi_start_optimize(n_work, phase1_end, rng, n_starts=10)
    if a_best_p1 is not None:
        update_best(a_best_p1, sc_best_p1, "phase1_multi")

    # Also optimize the parent directly with L-BFGS
    if parent is not None and time.monotonic() < phase1_end - 30:
        dl = min(phase1_end - 10, time.monotonic() + 120.0)
        beta_scale = max(sc_parent / (2 * n_work), 1e-6)
        betas = [b * beta_scale for b in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]]
        a_par_opt, sc_par = _lbfgs_lse(parent, dl, betas, max_wall_per=15.0)
        update_best(a_par_opt, sc_par, "parent_lbfgs")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === PHASE 2: Peak equalization on best (200s) ===
    phase2_end = min(deadline, t_start + 600.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"\nPhase 2: Peak equalization, score={best_score:.10f}")
        dl = min(phase2_end - 5, time.monotonic() + 180.0)
        a_pe, sc_pe = _peak_equalization(best_seq, dl, n_iter=80000)
        update_best(a_pe, sc_pe, "peak_eq")

    # === PHASE 3: Dinkelbach-style (150s) ===
    phase3_end = min(deadline, t_start + 750.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"\nPhase 3: Dinkelbach-style, score={best_score:.10f}")
        dl = min(phase3_end - 5, time.monotonic() + 140.0)
        a_dink, sc_dink = _dinkelbach_step(best_seq, dl, n_iter=20000)
        update_best(a_dink, sc_dink, "dinkelbach")

    # === PHASE 4: Try different n values (300s) ===
    phase4_end = min(deadline, t_start + 1050.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"\nPhase 4: Multi-scale exploration")
        n_curr = len(best_seq)

        # Try larger n
        n_larger = min(16384, n_curr * 2)
        if n_larger > n_curr and time.monotonic() < phase4_end - 100:
            print(f"  Trying n={n_larger}")
            a_large, sc_large = _refine_large_n(best_seq, n_larger,
                                                 min(phase4_end - 60, time.monotonic() + 150.0), rng)
            update_best(a_large, sc_large, f"n{n_larger}")

        # Try a different n with fresh optimization
        n_alt = 4096 if n_curr != 4096 else 8192
        if n_alt != n_curr and time.monotonic() < phase4_end - 80:
            print(f"  Fresh optimization at n={n_alt}")
            dl = min(phase4_end - 30, time.monotonic() + 120.0)
            a_alt, sc_alt = _multi_start_optimize(n_alt, dl, rng, n_starts=5)
            if a_alt is not None:
                update_best(a_alt, sc_alt, f"fresh_n{n_alt}")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === PHASE 5: Intensive refinement on best (300s) ===
    phase5_end = min(deadline, t_start + 1350.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"\nPhase 5: Intensive refinement, score={best_score:.10f}")

        # 5a: L-BFGS with very fine beta
        n_best = len(best_seq)
        beta_scale = max(best_score / (2 * n_best), 1e-7)
        fine_betas = [b * beta_scale for b in
                      [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
        dl = min(phase5_end - 100, time.monotonic() + 120.0)
        a_fine, sc_fine = _lbfgs_lse(best_seq, dl, fine_betas, max_wall_per=15.0)
        update_best(a_fine, sc_fine, "fine_lbfgs")

        # 5b: Another round of peak equalization
        if time.monotonic() < phase5_end - 60:
            dl2 = min(phase5_end - 40, time.monotonic() + 100.0)
            a_pe2, sc_pe2 = _peak_equalization(best_seq, dl2, n_iter=50000)
            update_best(a_pe2, sc_pe2, "peak_eq2")

        # 5c: Dinkelbach again
        if time.monotonic() < phase5_end - 30:
            dl3 = min(phase5_end - 20, time.monotonic() + 80.0)
            a_dink2, sc_dink2 = _dinkelbach_step(best_seq, dl3, n_iter=15000)
            update_best(a_dink2, sc_dink2, "dinkelbach2")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === PHASE 6: Multi-restart with perturbation (200s) ===
    phase6_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print(f"\nPhase 6: Multi-restart perturbation")
        n_restarts = 5
        per_restart = max(15.0, (phase6_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise_level = 0.01 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_level * rng.normal(size=len(best_seq))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase6_end - 5, time.monotonic() + per_restart)
            beta_scale = max(_score_norm_fn(p_start) / (2 * len(p_start)), 1e-6)
            betas = [b * beta_scale for b in [0.05, 0.02, 0.01, 0.005, 0.002]]
            a_r, sc_r = _lbfgs_lse(p_start, dl, betas, max_wall_per=per_restart * 0.5)
            update_best(a_r, sc_r, f"restart{i}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === PHASE 7: Final polish (remaining) ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(10.0, deadline - time.monotonic())
        print(f"\nPhase 7: Final polish, remaining={remaining:.0f}s")

        # Very fine peak equalization
        if remaining > 40:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.5)
            a_fin, sc_fin = _peak_equalization(best_seq, dl, n_iter=30000)
            update_best(a_fin, sc_fin, "final_peak")

        # Final L-BFGS
        if time.monotonic() < deadline - 20:
            n_best = len(best_seq)
            beta_scale = max(best_score / (2 * n_best), 1e-7)
            betas_final = [b * beta_scale for b in [0.001, 0.0005, 0.0002, 0.0001]]
            dl = min(deadline - 15, time.monotonic() + (deadline - time.monotonic()) * 0.3)
            a_fin2, sc_fin2 = _lbfgs_lse(best_seq, dl, betas_final, max_wall_per=10.0)
            update_best(a_fin2, sc_fin2, "final_lbfgs")

    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={float(final_seq.sum()):.6f}")

        # Validate with evaluator if size permits
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
                else:
                    print(f"Evaluator returned: {sc_eval}")
            except Exception as ex:
                print(f"Evaluate failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
