   import numpy as np
   import time

   def _next_pow2(n):
       k = 1
       while k < n:
           k <<= 1
       return max(2, k)

   def _full_convolve(a, b):

       out_n = n + m - 1
       N = _next_pow2(out_n)
       fa = np.fft.rfft(a, N)
       fb = np.fft.rfft(b, N)
       r = np.fft.irfft(fa * fb, N)[:out_n]
       return r

   def _score(a):

       s = float(a.sum())
       if s < 1e-15:
           return np.inf
       c = _full_convolve(a, a)
       return float(2.0 * n * c.max() / (s * s))

   def _score_and_conv(a):
       n = len(a)
       s = float(a.sum())
       c = _full_convolve(a, a)
       sc = 2.0 * n * c.max() / (s * s)
       return float(sc), c

   def _soft_max_weights(c, beta):
       cm = float(c.max())
       if cm < 1e-30:
           return np.ones(len(c)) / len(c), cm
       shifted = (c - cm) / beta
       e = np.exp(shifted)
       w = e / e.sum()
       soft_max = cm + beta * float(np.log(e.sum()))
       return w, soft_max

"""
Subgradient/soft-gradient Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a=exp(z) parameterization, annealed soft-max smoothing, multi-scale upsampling,
and multiple restarts at the largest scale. Key improvement over previous approach:
direct subgradient tracking of the convolution maximum with adaptive annealing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _cross_corr(w, a):
    """Compute h[j] = sum_k w[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.fft(a, N)
    fw = np.fft.fft(w, N)
    h = np.fft.ifft(fw * np.conj(fa)).real[:n]
    return h


def _compute_grad_soft(z, beta):
    """Compute gradient of 2n*softmax_beta(conv(exp(z),exp(z)))/(sum exp(z))^2 w.r.t. z."""
    z_shift = z - z.max()
    a = np.exp(z_shift)
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n), a, c, cm
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    M_soft = cm + beta * float(np.log(e.sum()))
    # h[j] = sum_k w[k]*a[k-j]
    h = _cross_corr(w, a)
    # Correction for 2j=k term
    correction = np.zeros(n)
    valid_k = 2 * np.arange(n)
    mask = valid_k < len(c)
    correction[mask] = w[valid_k[mask]] * a[mask]
    dM_da = 2.0 * h - correction
    # dF/da = 2n*(dM_da*S - 2*M_soft)/S^3
    dF_da = 2.0 * n * (dM_da * S - 2.0 * M_soft) / (S ** 3)
    dF_dz = dF_da * a
    return dF_dz, a, c, M_soft


def _compute_grad_hard(z):
    """Compute subgradient at the exact argmax of convolution."""
    z_shift = z - z.max()
    a = np.exp(z_shift)
    n = len(a)
    S = float(a.sum())
    c = _full_convolve(a, a)
    kstar = int(np.argmax(c))
    M = float(c[kstar])
    # dc[kstar]/da[j] = 2*a[kstar-j] (with correction)
    j_arr = np.arange(n)
    kj = kstar - j_arr
    valid = (kj >= 0) & (kj < n)
    dM_da = np.zeros(n)
    dM_da[valid] = 2.0 * a[kj[valid]]
    # Correction: if 2j = kstar, subtract a[j]
    if kstar % 2 == 0:
        jhalf = kstar // 2
        if 0 <= jhalf < n:
            dM_da[jhalf] -= a[jhalf]
    dF_da = 2.0 * n * (dM_da * S - 2.0 * M) / (S ** 3)
    dF_dz = dF_da * a
    return dF_dz, a, c, M


def _adam_loop(z_init, deadline, beta_schedule_fn, lr_init, rng,
               patience=20000, kick_interval=10000, kick_size=0.03,
               use_hard_after=None, hard_patience=5000):
    """Adam optimization loop with adaptive beta annealing."""
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    z = z - z.mean()

    best_z = z.copy()
    best_score = _score(np.exp(z))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr = lr_init
    no_improve = 0

    while time.monotonic() < deadline:
        t += 1
        beta = beta_schedule_fn(t)
        use_hard = (use_hard_after is not None and t > use_hard_after and beta < 1e-6)

        if use_hard:
            dF_dz, a, c, M = _compute_grad_hard(z)
        else:
            dF_dz, a, c, M = _compute_grad_soft(z, max(beta, 1e-8))

        if not np.all(np.isfinite(dF_dz)):
            break

        sc = float(2.0 * n * c.max() / (a.sum() ** 2))

        if sc < best_score - 1e-14:
            best_score = sc
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            z = z + kick_size * rng.normal(size=n)
            z = z - z.mean()
            no_improve = 0
            lr = max(0.001, lr * 0.5)
            continue

        m = b1 * m + (1 - b1) * dF_dz
        v = b2 * v + (1 - b2) * (dF_dz * dF_dz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - step

            z = z + 0.01 * rng.normal(size=n)
            z = z - z.mean()

    return best_z, best_score

    if n == 1:

    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, target_n)
    return np.interp(x1, x0, z)

        sc_final = _score(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp_a(256, 2.0)
    print(f"FALLBACK: n=256, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
