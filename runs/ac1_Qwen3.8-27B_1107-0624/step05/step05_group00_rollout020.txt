And the gradient w.r.t. z is:

For softmax: da_j/dz_j = a_j*(1-a_j), da_i/dz_j = -a_i*a_j for i≠j.
So df/dz_j = a_j * (df/da_j - sum_i a_i * df/da_i) = a_j * (df/da_j - mean_weighted_grad)

Then df/dz_j = a_j * (4*n*h[j] - sum_i a_i*4*n*h[i]) = 4*n*a_j*(h[j] - W)

Now, let me think about what improvements to make:

1. **Top-k smoothing**: Instead of softmax over all 2n-1 coefficients, use only the top-k. This gives a tighter approximation to the true max and focuses the gradient on the most important peaks.

2. **Cosine annealing for learning rate**: Instead of step decay, use cosine annealing.

3. **Momentum scheduling**: Use higher momentum early, lower late.

4. **Better initialization**: 
   - Start from parent
   - Also try: raised cosine, Hann window, etc.
   - Try: a_i = 1 + eps*cos(2*pi*i*p/n) for various p (frequency domain shaping)

5. **Structural exploitation**: 
   - Try sequences with approximate periodicity
   - Try sequences that are "anti-convolutional" (minimize peak)
   - Use the fact that the optimal sequence likely has some smooth structure

6. **Multi-n optimization with transfer**:
   - Optimize at n=2048, interpolate to n=4096, refine
   - Or: optimize at n=4096, decimate to n=2048, check if better ratio

7. **Differential evolution or CMA-ES on a low-dimensional parameterization**:
   - Parameterize a as a linear combination of a few basis functions
   - Optimize the coefficients

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses top-k smoothed max gradient descent in log-space with cosine LR annealing,
FISTA warm-start, diverse restarts, and cross-length transfer.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad_logspace(a, w, n, k_top=None, temperature=None):
    """Compute gradient of the smoothed objective w.r.t. log(a) parameters.
    
    Returns: 4*n*a*(h - W) where h[j] = sum_k w[k]*a[k-j]
    """
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    c = _full_convolve(a, a)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _compute_topk_weights(c, k, temperature):
    """Compute weights focused on top-k convolution coefficients."""
    c_max = float(c.max())
    if temperature is None or temperature <= 0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    # Global softmax with temperature
    shifted = (c - c_max) / temperature
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    w = e / s
    return w

def _adam_logspace_optimize(a_init, deadline, beta_schedule, lr_init, k_top, max_iter_total):
    """Adam optimization in log-space with top-k weighted smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    total_iter = max_iter_total
    total_time = max(1, deadline - time.monotonic())
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        while t < total_iter:
            if time.monotonic() > deadline:
                break
            
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())
            
            # Top-k weights
            if k_top is not None and k_top < len(c):
                w = _compute_topk_weights(c, k_top, beta)
            else:
                w = _compute_topk_weights(c, None, beta)
            
            grad = _compute_grad_logspace(a, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            # Cosine annealing learning rate
            progress = t / total_iter
            lr = lr_init * 0.5 * (1 + np.cos(np.pi * progress))
            lr = max(lr, lr_init * 0.01)
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _fista_simplex(a_init, beta, deadline, max_iter):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Lipschitz constant estimate
    L = 8.0 * n / beta  # conservative
    step = 1.0 / L
    
    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(2.0 * n * c_init.max())
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        
        # Softmax weights
        shifted = (c_y - c_max) / beta
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        # Gradient
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        grad = 4.0 * n * h  # gradient of 2n*sum(w*c) w.r.t. a on simplex
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0:
            sc = float(2.0 * n * _full_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _make_hann(n):
    t = np.linspace(0, 1, n, endpoint=False)
    a = 0.5 * (1 - np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _make_cosine_mod(n, freq, amp):
    """a_i = 1 + amp*cos(2*pi*freq*i/n), normalized."""
    t = np.arange(n)
    a = 1.0 + amp * np.cos(2 * np.pi * freq * t / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()

def _perturb_logspace(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses top-k smoothed max, Adam in log-space with cosine LR annealing, FISTA warm-start,
diverse restarts (logspace perturbation, frequency domain, structural), and cross-length transfer.
Key improvement: aggressive top-k focusing on actual peaks for subgradient-like behavior in final phases.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _make_weights_softmax(c, temperature):
    c_max = float(c.max())
    shifted = (c - c_max) / temperature
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _make_weights_topk(c, k):
    """Hard top-k: uniform weight on top-k coefficients."""
    if k >= len(c):
        return np.ones_like(c) / len(c)
    if k <= 1:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    idx = np.argsort(c)[::-1][:k]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    return w


def _make_weights_blended(c, k, temperature):
    """Blend of top-k and softmax for smooth transitions."""
    w_topk = _make_weights_topk(c, k)
    w_softmax = _make_weights_softmax(c, temperature)
    return 0.5 * w_topk + 0.5 * w_softmax


def _compute_grad_logspace(a, w, n):
    """Gradient of 2n*sum(w*c) w.r.t. log-space params z where a=softmax(z).
    Returns 4*n*a*(h - W) where h[j]=sum_k w[k]*a[k-j], W=dot(w,c)."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    c = _full_convolve(a, a)
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, k_mode, max_iter):
    """Adam in log-space. k_mode: (k_top, 'soft'|'hard'|'blend') or None for full softmax."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            c_max = float(c.max())

            if k_mode is not None:
                k_top, mode = k_mode
                if mode == 'hard':
                    w = _make_weights_topk(c, k_top)
                elif mode == 'blend':
                    w = _make_weights_blended(c, k_top, beta)
                else:
                    w = _make_weights_softmax(c, beta)
            else:
                w = _make_weights_softmax(c, beta)

            grad = _compute_grad_logspace(a, w, n)
            if not np.all(np.isfinite(grad)):
                break

            # Cosine annealing
            progress = min(1.0, t / max_iter)
            lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
            lr = max(lr, lr_init * 0.005)

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 30 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_simplex(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 * n / beta
    step = 1.0 / L

    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(2.0 * n * c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = _make_weights_softmax(c_y, beta)

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 4.0 * n * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = float(2.0 * n * _full_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)
