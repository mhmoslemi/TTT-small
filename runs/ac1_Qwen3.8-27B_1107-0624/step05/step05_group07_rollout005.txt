Optimized step function coefficient search using accelerated LP-based iterated minimax
with L-BFGS log-sum-exp polishing and multi-scale exploration.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    # For row k (0-indexed), the non-zero entries are at columns j where 0<=j<n and 0<=k-j<n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js)
            data.extend(f[js].tolist())
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _build_conv_matrix_fast(f):
    """Build convolution matrix using vectorized operations."""
    n = len(f)
    # For each row k, the valid columns are max(0,k-n+1) to min(n-1,k)
    # Total non-zeros = n^2
    all_rows = []
    all_cols = []
    all_data = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        all_rows.append(np.full(len(js), k, dtype=np.int32))
        all_cols.append(js.astype(np.int32))
        all_data.append(f[js])
    
    rows = np.concatenate(all_rows)
    cols = np.concatenate(all_cols)
    data = np.concatenate(all_data)
    
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, max_time=30):
    """Solve LP: min t s.t. A@g <= t, sum(g)=1, g>=0. Returns g and t."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix_fast(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None and np.all(np.isfinite(g.value)):
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s < 1e-15:
                return None, None
            g_val = g_val / s
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters, deadline, step_size_schedule=None):
    """Run LP-based iterated minimax."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        g, t_val = _lp_solve(f)
        if g is None:
            break
        
        # Adaptive step size
        if step_size_schedule is not None:
            step = step_size_schedule(it, n_iters)
        else:
            step = min(1.0, 2.0 / (it + 5))
        
        f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        
        if it % 50 == 0:
            elapsed = time.monotonic()
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    
    return best_a, best_score


def _lse_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # Gradient of max_k c[k] w.r.t. a[i]:
    # c[k] = sum_j a[j]*a[k-j]
    # dc[k]/da[i] = a[k-i] + a[i-k] (with appropriate bounds)
    # For the weighted sum: sum_k w[k]*dc[k]/da[i]
    # = sum_k w[k]*(a[k-i] + a[i-k]) (valid indices)
    # = 2 * sum_k w[k]*a[k-i] (by symmetry of valid ranges)
    # Actually let me be more careful.
    # dc[k]/da[i]: 
    #   c[k] = sum_j a[j]*a[k-j]
    #   term where j=i: a[i]*a[k-i] -> derivative w.r.t. a[i] is a[k-i]
    #   term where k-j=i, i.e., j=k-i: a[k-i]*a[i] -> derivative w.r.t. a[i] is a[k-i]
    #   But if i = k-i (i.e., k=2i), these are the same term, so derivative is a[i]
    #   Otherwise, derivative is 2*a[k-i]
    #   Valid when 0<=i<n and 0<=k-i<n
    #
    # sum_k w[k] * dc[k]/da[i] = sum_k w[k] * (a[k-i] + a[k-i] - a[k-i]*delta(k-2i))
    # = sum_k w[k] * a[k-i] * (1 + 1 - delta(k-2i))
    # = 2 * sum_k w[k] * a[k-i] - sum_k w[k] * a[k-i] * delta(k-2i)
    # = 2 * sum_k w[k] * a[k-i] - w[2i] * a[i] (if 0<=2i<2n-1)
    #
    # The first term 2 * sum_k w[k] * a[k-i] is a convolution!
    # (w_conv_a)[i] = sum_k w[k] * a[k-i] ... hmm, this is a cross-correlation
    # 
    # Actually, let me think of it differently.
    # Let h[i] = sum_k w[k] * dc[k]/da[i]
    # = sum_k w[k] * a[k-i] (for valid k,i) + sum_k w[k] * a[k-i] (for valid)
    #   minus the diagonal correction
    #
    # The sum sum_k w[k]*a[k-i] over valid k (0<=k<2n-1, 0<=k-i<n) 
    # = sum_{j: 0<=j<n} w[i+j] * a[j]  (substituting j=k-i)
    # = cross_correlation(w, a)[i] in some sense
    #
    # This is getting complicated. Let me use the simpler approach:
    # grad_z = 4*n * a * (conv(w, a_reversed)[n-1:2n-1] - W)
    # This is what the previous program uses and it works.
    
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_optimize(a_init, deadline, beta_list=None):
    """Run L-BFGS with beta continuation on log-sum-exp objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001,
                 0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00003, 0.00001]
        beta_list = [b * max_c0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        remaining_betas = beta_list[beta_list.index(beta):]
        time_per = (deadline - time.monotonic()) / max(1, len(remaining_betas))
        time_per = min(time_per, 60.0)
        
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_gradient,
                options={'maxiter': int(time_per * 50), 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxls': 50}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    # Linear
    for slope in [0.5, 1.0, 2.0, -1.0, -2.0]:
        a = np.maximum(1.0 + slope * (t - 0.5), 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [0.5, 1.0, 2.0, 4.0, -1.0, -2.0, -4.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Cosine
    for p in [0.5, 1.0, 2.0]:
        a = np.maximum(np.cos(np.pi * (2*t - 1) / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    # Quadratic
    a = 1.0 + 4*(t-0.5)**2
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + (2*t-1)**2)
    profiles.append(a / a.sum())
    return profiles
