def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex (sum=1, all >= 0)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s


def _subgradient_topk(a, top_k_lags):
    """Compute averaged subgradient of max_j g_j(a) over top_k_lags.
    dg_j/da[k] = 2*a[j-k] if 0 <= j-k < n, else 0.
    """
    n = len(a)
    sg = np.zeros(n)
    for j in top_k_lags:
        j = int(j)
        lo = max(0, j - (n - 1))
        hi = min(n - 1, j)
        # sg[lo:hi+1] += 2 * a[j-hi : j-lo+1][::-1]
        # i.e., for k in [lo, hi]: sg[k] += 2*a[j-k]
        a_seg = a[j - hi : j - lo + 1][::-1]
        sg[lo:hi + 1] += 2.0 * a_seg
    sg /= len(top_k_lags)
    return sg


def _nesterov_subgrad(a_init, deadline, rng, K_list=(1, 3, 5, 8), max_iters=200000):
    """Nesterov-accelerated subgradient method with top-K peak targeting."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    v = np.zeros(n)  # momentum
    alpha = 0.1 / np.sqrt(n)  # initial step size
    alpha_decay = 0.9997
    momentum_beta = 0.7
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Nesterov lookahead
        y = a + momentum_beta * v
        
        # Compute convolution at y
        c = _full_convolve(y, y)
        
        # Pick K based on iteration phase
        if it < 5000:
            K = K_list[0]
        elif it < 20000:
            K = K_list[1]
        elif it < 60000:
            K = K_list[2]
        else:
            K = K_list[3]
        K = min(K, len(c))
        
        # Top K lags
        if K == 1:
            top_lags = np.array([np.argmax(c)])
        else:
            top_lags = np.argsort(c)[-K:]
        
        # Subgradient
        sg = _subgradient_topk(y, top_lags)
        
        # Gradient step
        trial = y - alpha * sg
        
        # Project onto simplex
        a_new = _project_simplex(trial)
        
        # Update momentum
        v = a_new - y
        a = a_new
        
        # Check improvement

            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        alpha = max(alpha, 1e-8 / np.sqrt(n))

        if it % 20000 == 0 and it > 0:
            v *= 0.3
            # Small random perturbation
            a = a + 0.001 * rng.normal(size=n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
    
    # Final check
    sc = _score_norm(a)
    if sc < best_sc:

        best_a = a.copy()
    
    return best_a, best_sc


def _lbfgs_continuation(a_init, deadline, rng, max_wall_time=300.0, beta_scale=1.0):
    """L-BFGS with logsumexp continuation - improved version."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    # More aggressive and finer beta schedule
    beta_fracs = [0.05, 0.03, 0.02, 0.012, 0.007, 0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    betas = [b * max_c0 * beta_scale for b in beta_fracs]
    
    per_level = max(2.0, max_wall_time / len(betas))
    
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        
        # Define objective and gradient
        def obj_fn(z_local):
            a_local = _softmax_fn(z_local)
            c_local = _full_convolve(a_local, a_local)
            cm = float(c_local.max())
            c_shifted = (c_local - cm) / beta
            e_local = np.exp(c_shifted)
            s_local = e_local.sum()
            if s_local < 1e-300:
                return 2.0 * n * cm
            log_sum = float(np.log(s_local))
            return 2.0 * n * (cm + beta * log_sum)
        
        def grad_fn(z_local):
            a_local = _softmax_fn(z_local)
            c_local = _full_convolve(a_local, a_local)
            cm = float(c_local.max())
            c_shifted = (c_local - cm) / beta
            e_local = np.exp(c_shifted)
            s_local = e_local.sum()
            if s_local < 1e-300:
                w = np.zeros_like(c_local)
                w[int(np.argmax(c_local))] = 1.0
            else:
                w = e_local / s_local
            W = float(np.dot(w, c_local))
            a_rev = a_local[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_local * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_local)
            return g
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 20000, 'maxcor': 30}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _palindromic_optimize(a_init, deadline, rng, max_iters=50000):
    """Optimize only the first half (palindromic constraint)."""
    n = len(a_init)
    half = (n + 1) // 2
    # Extract first half
    x = a_init[:half].copy()

    best_sc = _score_norm(a_init)
    
    def make_full(x_local):

            return np.concatenate([x_local, x_local[::-1]])
        else:
            return np.concatenate([x_local, x_local[-2::-1]])
    
    v = np.zeros(half)
    alpha = 0.05 / np.sqrt(half)
    momentum_beta = 0.6
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        a_full = make_full(x)
        a_full = a_full / a_full.sum()
        
        c = _full_convolve(a_full, a_full)
        K = min(5, len(c))
        top_lags = np.argsort(c)[-K:]
        
        # Compute subgradient and map back to half
        sg_full = _subgradient_topk(a_full, top_lags)
        sg_half = np.zeros(half)
        for i in range(half):
            partner = n - 1 - i
            if partner < half:
                sg_half[i] = sg_full[i] + sg_full[partner]
            else:
                sg_half[i] = sg_full[i]
        
        y = x + momentum_beta * v
        trial = y - alpha * sg_half
        x_new = np.maximum(trial, 1e-15)
        
        v = x_new - x
        x = x_new
        
        if it % 500 == 0:
            a_check = make_full(x)
            a_check = a_check / a_check.sum()
            sc = _score_norm(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
        
        alpha *= 0.9998
        alpha = max(alpha, 1e-9 / np.sqrt(half))
        
        if it % 10000 == 0 and it > 0:
            v *= 0.3
            x = x + 0.002 * rng.normal(size=half)
            x = np.maximum(x, 1e-15)
    
    a_final = make_full(x)
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)

def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell(n, width):
    t = np.linspace(0, 1, n)
    a = np.exp(-((t - 0.5) ** 2) / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_tapered(n, power):
    """Tapered edges: a[i] ~ (i+1)^p * (n-i)^p in the interior."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** power * (n - i) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_multi_gauss(n, centers, widths):
    t = np.linspace(0, 1, n)
    a = np.zeros(n)
    for c, w in zip(centers, widths):
        a += np.exp(-(t - c) ** 2 / (2 * w ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _resize(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a / a.sum()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    p = np.interp(x_new, x_old, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    Phase 1: Best initialization from diverse profiles
    Phase 2: Nesterov subgradient (top-K peaks)
    Phase 3: L-BFGS logsumexp continuation
    Phase 4: Palindromic optimization
    Phase 5: Aggressive local search
    Phase 6: Multi-n exploration
    Phase 7: Final intensive polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_triangular(n_parent)
        sc_parent = _score_norm(parent)
        print(f"No parent, using triangular n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    n_work = n_parent
    print(f"Working with n={n_work}")
    
    # --- Phase 1: Diverse initialization (60s) ---
    phase1_end = min(deadline, t_start + 60.0)
    
    profiles = [
        _make_triangular(n_work),
        _make_tapered(n_work, 0.5),
        _make_tapered(n_work, 1.0),
        _make_tapered(n_work, 2.0),
        _make_tapered(n_work, 3.0),
        _make_bell(n_work, 0.2),
        _make_bell(n_work, 0.25),
        _make_bell(n_work, 0.3),
        _make_multi_gauss(n_work, [0.3, 0.7], [0.1, 0.1]),
        _make_multi_gauss(n_work, [0.25, 0.75], [0.12, 0.12]),
        _make_multi_gauss(n_work, [0.2, 0.5, 0.8], [0.08, 0.15, 0.08]),
    ]
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Nesterov subgradient (500s) ---
    phase2_end = min(deadline, t_start + 560.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: Nesterov subgradient")
        dl = phase2_end - 5.0
        a_best, sc_best = _nesterov_subgrad(best_seq, dl, rng, K_list=(1, 3, 5, 8))
        update_best(a_best, sc_best, "nesterov")
        
        # Second run from the best profile
        if time.monotonic() < phase2_end - 30:
            prof_start = candidates[0][1].copy()
            a_best2, sc_best2 = _nesterov_subgrad(prof_start, phase2_end - 5.0, rng, K_list=(1, 2, 3, 5))
            update_best(a_best2, sc_best2, "nesterov_prof")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: L-BFGS continuation (400s) ---
    phase3_end = min(deadline, t_start + 960.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: L-BFGS continuation")
        dl = phase3_end - 10.0
        a_best, sc_best = _lbfgs_continuation(best_seq, dl, rng, max_wall_time=390.0, beta_scale=1.0)
        update_best(a_best, sc_best, "lbfgs_main")
        
        if time.monotonic() < phase3_end - 30:
            a_best2, sc_best2 = _lbfgs_continuation(best_seq, phase3_end - 5.0, rng, max_wall_time=80.0, beta_scale=0.4)
            update_best(a_best2, sc_best2, "lbfgs_fine")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Palindromic optimization (200s) ---
    phase4_end = min(deadline, t_start + 1160.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Palindromic optimization")
        dl = phase4_end - 5.0
        a_best, sc_best = _palindromic_optimize(best_seq, dl, rng)
        update_best(a_best, sc_best, "palindromic")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Aggressive local search (250s) ---
    phase5_end = min(deadline, t_start + 1410.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Aggressive local search")
        dl = phase5_end - 5.0
        a_best, sc_best = _aggressive_local_search(best_seq, dl, rng, n_trials=80000)
        update_best(a_best, sc_best, "local_search")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Multi-n exploration (150s) ---
    phase6_end = min(deadline, t_start + 1560.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        
        # Try upscaling
        for target_n in [n_curr * 2]:
            if target_n > 16384:
                continue
            if time.monotonic() > phase6_end - 50:
                break
            p_up = _resize(best_seq, target_n)
            sc_up = _score_norm(p_up)
            print(f"  Upscale to n={target_n}: score={sc_up:.10f}")
            update_best(p_up, sc_up, f"upscale_{target_n}")
            
            if time.monotonic() < phase6_end - 20:
                dl = min(phase6_end - 10, time.monotonic() + 60.0)
                a_best, sc_best = _lbfgs_continuation(p_up, dl, rng, max_wall_time=55.0, beta_scale=0.5)
                update_best(a_best, sc_best, f"lbfgs_n{target_n}")
        
        # Try downscaling (sometimes helps due to numerical effects)
        n_down = n_curr // 2
        if n_down >= 1024:
            p_down = _resize(best_seq, n_down)
            sc_down = _score_norm(p_down)
            update_best(p_down, sc_down, f"downscale_{n_down}")
        
        # Multi-restart
        if time.monotonic() < phase6_end - 20:
            for i in range(3):
                if time.monotonic() > phase6_end - 15:
                    break
                noise = 0.015 + 0.01 * i
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean() + noise * rng.normal(size=len(best_seq))
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                p_start = _softmax_fn(z)
                dl = min(phase6_end - 5, time.monotonic() + 30.0)
                a_best, sc_best = _lbfgs_continuation(p_start, dl, rng, max_wall_time=25.0, beta_scale=0.6)
                update_best(a_best, sc_best, f"restart{i}")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Final polishing (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")
        
        # 7a: Very fine L-BFGS
        if remaining > 80:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl, rng,
                                                   max_wall_time=remaining * 0.35, beta_scale=0.2)
            update_best(a_best, sc_best, "final_lbfgs_fine")
        
        # 7b: Short Nesterov
        if time.monotonic() < deadline - 40:
            dl = min(deadline - 35, time.monotonic() + remaining * 0.25)
            a_best, sc_best = _nesterov_subgrad(best_seq, dl, rng, K_list=(1, 2, 3, 5))
            update_best(a_best, sc_best, "final_nesterov")
        
        # 7c: Final local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _aggressive_local_search(best_seq, dl, rng, n_trials=30000)
            update_best(a_best, sc_best, "final_local")
        
        # 7d: Another L-BFGS pass
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl, rng,
                                                   max_wall_time=max(5.0, remaining * 0.2), beta_scale=0.1)
            update_best(a_best, sc_best, "final_lbfgs2")
    
    print(f"Phase 7 done: best={best_score:.10f}")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_norm(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular(8192)
    sc_fb = _score_norm(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
