def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    out_n = n + m - 1

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-15:

    c = _full_convolve(a, a)

Multi-phase optimizer using L-BFGS-B with power-sum continuation and
subgradient polishing for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Combines second-order optimization with
direct max-subgradient steps and multi-restart diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cross_corr(w, a):
    """Compute h_j = sum_k w_k * a_{k-j} for j=0..n-1."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n-1:2*n-1]


def _power_sum_obj_and_grad(z, p):
    """
    Compute F(z) = 2n * f_p(softmax(z)) and its gradient.
    f_p(a) = (sum_k c_k^p)^{1/p} where c = conv(a,a).
    """
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    
    c_max = float(c.max())
    if c_max < 1e-300:
        return 2.0 * n, np.zeros_like(z)
    
    u = c / c_max
    sum_u_p = float(np.sum(u ** p))
    if sum_u_p < 1e-300:
        return 2.0 * n * c_max, np.zeros_like(z)
    
    f_p = c_max * sum_u_p ** (1.0 / p)
    obj = 2.0 * n * f_p
    
    # w_k = u_k^{p-1} / sum_u_p
    w = (u ** (p - 1)) / sum_u_p
    h = _cross_corr(w, a)
    
    grad_z = 4.0 * n * f_p * a * (h - 1.0)
    
    return float(obj), grad_z

    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    c_max = float(c[k_star])
    
    # Subgradient of 2n*max(c) in a-space: 4n * a_{k*-j}
    # In log-space: 4n * a_j * (a_{k*-j} - c_max)
    sg_a = 4.0 * n * np.zeros(n)
    for offset in range(n):
        j = k_star - offset
        if 0 <= j < n:
            sg_a[offset] = 4.0 * n * a[j]
    
    mean_sg = float(np.dot(sg_a, a))
    grad_z = a * (sg_a - mean_sg)
