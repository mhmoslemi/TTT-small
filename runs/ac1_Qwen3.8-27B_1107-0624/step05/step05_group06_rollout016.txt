"""
Hybrid LP-direction + L-BFGS optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: Uses LP to find descent directions that provably don't increase
the autocorrelation peak, combined with L-BFGS refinement and subgradient polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _normalize(a):
    s = float(a.sum())
    if s < 1e-30:
        return a
    return a / s

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Subgradient of max(conv(a,a)) w.r.t. a, assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()

    tol = 0.005 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    # Gradient: dc[k]/da[i] = 2*a[k-i] (when 0 <= k-i < n)
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] == 0:
            continue
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * w[k] * a[j]
    return grad

def _solve_lp(a, M, solver=cp.GLPK):
    """Solve the LP: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0."""
    n = len(a)
    g = cp.Variable(n)
    
    # Build constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # conv(a,g)[k] = sum_j A[k,j] * g[j]
    # Use Toeplitz structure
    # For efficiency, build row by row
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # j ranges from max(0, k-n+1) to min(k, n-1)
        j_low = max(0, k - n + 1)
        j_high = min(k, n - 1)
        for j in range(j_low, j_high + 1):
            row[j] = a[k - j]
        rows.append(row)
    A = np.array(rows)
    
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return None
    
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)

def _lp_descent(a, deadline, max_iters=200, t_init=0.1):

            break

        accepted = False
        for trial in range(10):

            c_new = _full_convolve(a_new, a_new)

                c = c_new
                M = M_new
                accepted = True
                if M < best_M:

                    print(f"    LP iter {it}: M={M:.10f}, t={t:.4f}, improve={best_M if best_M != M else 0:.2e}")

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: 4*n*a*(h - W) where h[i] = sum_k w[k]*a[k-i]
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)
