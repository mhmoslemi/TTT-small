"""
Peak-equalization optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses iterative peak balancing, L-BFGS polish, and annealed restarts.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

"""
Peak-equalization optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses top-K log-sum-exp L-BFGS, LP-based peak balancing, peak-directed
coordinate descent, annealed restarts, and multi-n exploration.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _topk_lse_obj(z, beta, K, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if K < len(c):
        idx = np.argpartition(c, -K)[-K:]
        peaks = c[idx]
    else:
        peaks = c
    pm = float(peaks.max())
    if pm < 1e-300:
        return 0.0
    sh = (peaks - pm) / beta
    ms = float(sh.max())
    lse = pm + beta * (ms + float(np.log(np.sum(np.exp(sh - ms)))))
    return float(2.0 * n * lse)


def _topk_lse_grad(z, beta, K, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if K < len(c):
        idx = np.argpartition(c, -K)[-K:]
        peaks = c[idx]
    else:
        idx = np.arange(len(c))
        peaks = c
    pm = float(peaks.max())
    if pm < 1e-300:
        return np.zeros_like(z)
    sh = (peaks - pm) / beta
    ms = float(sh.max())
    e = np.exp(sh - ms)
    w = e / e.sum()

    # Gradient of sum w_i * c[k_i] w.r.t. a
    grad_a = np.zeros(n)
    for ii in range(len(idx)):
        k = idx[ii]
        wi = w[ii]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            grad_a[i_lo:i_hi + 1] += 2.0 * wi * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

    Wa = float(np.dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _run_lbfgs_topk(a_init, deadline, K_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if K_list is None:
        K_list = [3, 5, 10, 20, 50, 100, 200]

    for K in K_list:
        if time.monotonic() > deadline:
            break
        # Adaptive beta schedule
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        beta_list = [m0 * f for f in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]]

        for beta in beta_list:
            if time.monotonic() > deadline:
                break
            try:
                result = minimize(
                    _topk_lse_obj, z, args=(beta, K, n),
                    method='L-BFGS-B', jac=_topk_lse_grad,
                    options={'maxiter': maxiter_per_beta, 'ftol': 1e-18,
                             'gtol': 1e-12, 'maxls': 100, 'maxfun': 15000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score - 1e-14:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
            except Exception:
                pass
    return best_a, best_score


def _lp_peak_balance(a_init, deadline, K=40, max_rounds=20, blend_values=None):
    n = len(a_init)
    if blend_values is None:
        blend_values = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Get active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.90 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]

        # Build LP matrix
        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(len(active), M * 1.0005)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in blend_values:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-13:
                best_a = a_trial.copy()
                best_score = _score_norm(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _peak_directed_cd(a_init, deadline, max_steps=50000, step_scale=0.01):
    """Peak-directed coordinate descent: directly reduce contributing peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    S = float(a.sum())

    no_improve = 0
    step = step_scale

    for it in range(max_steps):
        if time.monotonic() > deadline:
            break
        if it % 200 == 0 and no_improve > 2000:
            break

        c = _full_convolve(a, a)
        M = float(c.max())
        current_score = 2.0 * n * M / (S * S)

        # Find top peaks
        top_k = np.argsort(c)[::-1][:10]

        # For each top peak, find the a[i] that contributes most and reduce it
        improved = False
        for k in top_k:
            if c[k] < M * 0.95:
                break
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo > i_hi:
                continue
            # Contribution of a[i] to c[k] is a[i]*a[k-i]
            contribs = np.array([a[i] * a[k - i] for i in range(i_lo, i_hi + 1)])
            total_contrib = contribs.sum()
            if total_contrib < 1e-30:
                continue
            # Find the index with highest contribution
            max_idx_local = np.argmax(contribs)
            i_star = i_lo + max_idx_local
            j_star = k - i_star

            # Amount to reduce
            # We want to reduce c[k] by some fraction of (c[k] - target)
            target = 0.98 * M  # try to bring peak down to 98% of current max
            reduction_needed = c[k] - target
            if reduction_needed <= 0:
                continue

            # Reduce a[i_star] and a[j_star] (if different)
            if i_star == j_star:
                # a[i]^2 contributes, reduce a[i]
                da = reduction_needed / (2.0 * a[i_star] + 1e-30)
                da = min(da, step * a[i_star])
                if da > 0:
                    a[i_star] -= da
            else:
                # a[i]*a[j] contributes, reduce the smaller one
                if a[i_star] <= a[j_star]:
                    da = reduction_needed / (a[j_star] + 1e-30)
                    da = min(da, step * a[i_star])
                    if da > 0:
                        a[i_star] -= da
                else:
                    da = reduction_needed / (a[i_star] + 1e-30)
                    da = min(da, step * a[j_star])
                    if da > 0:
                        a[j_star] -= da

            a = np.maximum(a, 0.0)
            improved = True

        if not improved:
            # Increase step scale
            step *= 1.5
            if step > 0.1:
                step = 0.01
                break

        # Re-normalize
        S_new = a.sum()
        if S_new < 1e-15:
            break
        a = a / S_new

        # Check improvement
        c_new = _full_convolve(a, a)
        M_new = float(c_new.max())
        sc_new = 2.0 * n * M_new / (a.sum() ** 2)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_a = a.copy()
            no_improve = 0
            step = max(0.001, step * 0.95)
        else:
            no_improve += 1
            a = best_a.copy()  # revert
            if no_improve % 100 == 0:
                step *= 0.9

    return best_a, best_score


def _annealed_search(a_init, deadline, n_iter=200, noise_scale=0.05):
    """Annealed random perturbation search."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    rng = np.random.default_rng(12345)
    noise = noise_scale

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        # Perturb
        z = np.log(np.maximum(best_a, 1e-15))
        z = z + noise * rng.standard_normal(n)
        z = z - z.mean()
        a_trial = _softmax_fn(z)

        sc_trial = _score_norm(a_trial)
        if sc_trial < best_score:
            best_score = sc_trial
            best_a = a_trial.copy()
            noise *= 0.95
        else:
            noise *= 0.99

        if noise < 0.001:
            break

    return best_a, best_score


def _greenkhorn(a_init, deadline, max_rounds=30):
    """Greenhorn-style: alternate between reducing peaks and redistributing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        current_score = 2.0 * n * M / (S * S)

        # Find the single worst peak
        k_worst = int(np.argmax(c))

        # Compute gradient direction to reduce this peak
        # d(c[k])/da[i] = 2*a[k-i] for valid i
        i_lo = max(0, k_worst - n + 1)
        i_hi = min(n - 1, k_worst)
        if i_lo > i_hi:
            break

        # Direction: decrease a[i] proportionally to a[k-i]
        direction = np.zeros(n)
        direction[i_lo:i_hi + 1] = -2.0 * best_a[k_worst - i_lo:k_worst - i_hi + 1]

        # Redistribute: increase a where the autocorrelation is low
        low_idx = np.argsort(c)[:n // 2]
        for k_low in low_idx[:20]:
            i_lo2 = max(0, k_low - n + 1)
            i_hi2 = min(n - 1, k_low)
            if i_lo2 <= i_hi2:
                direction[i_lo2:i_hi2 + 1] += 0.1 * best_a[k_low - i_lo2:k_low - i_hi2 + 1]

        # Normalize direction to have zero sum (preserve normalization)
        direction -= direction.mean()

        # Line search
        best_t = 0.0
        best_sc = current_score
        for t in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            a_trial = best_a + t * direction
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = 2.0 * n * c_trial.max() / (a_trial.sum() ** 2)
            if sc_trial < best_sc:
                best_sc = sc_trial
                best_t = t

        if best_t == 0.0:
            break

        best_a = (best_a + best_t * direction)
        best_a = np.maximum(best_a, 0.0)
        s = best_a.sum()
        if s < 1e-15:
            break
        best_a = best_a / s
        best_score = _score_norm(best_a)

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _parametric_init(n, etype='power'):
    t = np.linspace(0, 1, n)
    if etype == 'power2':
        a = 0.01 + 0.99 * t ** 2
    elif etype == 'power3':
        a = 0.01 + 0.99 * t ** 3
    elif etype == 'exp3':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'exo':
        a = 0.05 + 0.95 * t ** 5
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'power15':
        a = 0.01 + 0.99 * t ** 1.5
    else:
        a = 0.01 + 0.99 * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    print("Phase 2: LP peak balancing")
    p2_end = min(deadline, t_start + 750.0)

    if time.monotonic() < p2_end and best_seq is not None:
        a_lp, sc_lp = _lp_peak_balance(best_seq, p2_end, K=40, max_rounds=20)
        update_best(a_lp, sc_lp, "lp_main")

        if time.monotonic() < p2_end - 60:
            a_lp2, sc_lp2 = _lp_peak_balance(best_seq, p2_end, K=80, max_rounds=15)
            update_best(a_lp2, sc_lp2, "lp_wide")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak-directed CD (200s) ===
    print("Phase 3: Peak-directed CD")
    p3_end = min(deadline, t_start + 950.0)

    if time.monotonic() < p3_end and best_seq is not None:
        a_cd, sc_cd = _peak_directed_cd(best_seq, p3_end, max_steps=30000, step_scale=0.005)
        update_best(a_cd, sc_cd, "peak_cd")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Greenhorn (150s) ===
    print("Phase 4: Greenhorn")
    p4_end = min(deadline, t_start + 1100.0)

    if time.monotonic() < p4_end and best_seq is not None:
        a_gh, sc_gh = _greenkhorn(best_seq, p4_end, max_rounds=50)
        update_best(a_gh, sc_gh, "greenhorn")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Fine L-BFGS with very small beta (250s) ===
    print("Phase 5: Fine L-BFGS")
    p5_end = min(deadline, t_start + 1350.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_f, sc_f = _run_lbfgs_topk(best_seq, p5_end, K_list=[3, 5, 10], maxiter_per_beta=10000)
        update_best(a_f, sc_f, "fine_lbfgs")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Annealed restarts (200s) ===
    print("Phase 6: Annealed restarts")
    p6_end = min(deadline, t_start + 1550.0)

    if time.monotonic() < p6_end and best_seq is not None:
        for noise_scale in [0.03, 0.01, 0.005]:
            if time.monotonic() > p6_end - 10:
                break
            a_an, sc_an = _annealed_search(best_seq, min(p6_end - 5, time.monotonic() + 40),
                                           n_iter=300, noise_scale=noise_scale)
            update_best(a_an, sc_an, f"anneal_{noise_scale}")

            # Re-optimize after anneal
            if time.monotonic() < p6_end - 30:
                a_ar, sc_ar = _run_lbfgs_topk(a_an, min(p6_end - 5, time.monotonic() + 30),
                                               K_list=[5, 10], maxiter_per_beta=3000)
                update_best(a_ar, sc_ar, f"anneal_reopt_{noise_scale}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Try different n (200s) ===
    print("Phase 7: Multi-n exploration")
    p7_end = min(deadline, t_start + 1750.0)

    if time.monotonic() < p7_end and best_seq is not None:
        for n_try in [16384, 4096]:
            if time.monotonic() > p7_end - 30:
                break
            a_interp = _interp_profile(best_seq, n_try)
            sc_in = _score_norm(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_try}")

            if time.monotonic() < p7_end - 20:
                a_in_b, sc_in_b = _run_lbfgs_topk(a_interp, min(p7_end - 5, time.monotonic() + 40),
                                                   K_list=[5, 10], maxiter_per_beta=3000)
                update_best(a_in_b, sc_in_b, f"lbfgs_n{n_try}")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Final LP + L-BFGS polish (remaining) ===
    print("Phase 8: Final polish")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_final_lp, sc_flp = _lp_peak_balance(best_seq, deadline - 10, K=50, max_rounds=5)
        update_best(a_final_lp, sc_flp, "final_lp")

        if time.monotonic() < deadline - 5:
            a_final_lb, sc_flb = _run_lbfgs_topk(best_seq, deadline - 5, K_list=[3, 5], maxiter_per_beta=5000)
            update_best(a_final_lb, sc_flb, "final_lbfgs")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_full(fb):.10f}")
    return [float(x) for x in fb]
