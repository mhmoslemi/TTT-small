def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _make_power_profile(n, p, center=0.5):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(t - center)
    a = (1.0 - 2.0 * u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_skew_triangular(n, skew):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** skew * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric_plateau(n, left_frac, right_frac, height):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    a[:int(left_frac * n)] = height
    a[-int(right_frac * n):] = height
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_ramped_profile(n, ramp_left, ramp_right, plateau_height):
    n_rl = int(ramp_left * n)
    n_rr = int(ramp_right * n)
    a = np.ones(n) * plateau_height
    if n_rl > 0:
        ramp = np.linspace(1e-6, plateau_height, n_rl)
        a[:n_rl] = ramp
    if n_rr > 0:
        ramp = np.linspace(plateau_height, 1e-6, n_rr)
        a[-n_rr:] = ramp
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _compute_conv_weights(a, beta):
    """Compute softmax weights over convolution values (for log-sum-exp gradient)."""
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _lbfgs_logsumexp(a_init, deadline, rng, beta_scale=1.0, max_wall_time=None,
                      min_beta_frac=0.00005, nesterov_momentum=0.0):
    """L-BFGS with log-sum-exp continuation and optional Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078125,
                   0.003906, 0.001953, 0.000977, 0.000488, 0.000244, 0.000122,
                   0.000061, min_beta_frac]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            w_, cm_ = _compute_conv_weights(a_, beta)
            w_rev = w_[::-1]
            conv_wa = _full_convolve(w_rev, a_)
            grad_a = (4.0 * n / beta) * conv_wa[0:n][::-1]
            dot_a_grad = float(np.dot(a_, grad_a))
            grad_z = a_ * (grad_a - dot_a_grad)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_)
            return grad_z

        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _subgradient_method(a_init, deadline, rng, n_iters=50000):
    """Subgradient method with exact max and adaptive step size for simplex."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    # Project onto simplex: a >= 0, sum(a) = 1
    def _project_simplex(v):
        v = v.copy()
        n_v = len(v)
        if v.sum() < 1e-15:
            return np.ones(n_v) / n_v
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, n_v + 1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        proj = np.maximum(v - theta, 0)
        proj /= proj.sum()
        return proj

    step_scale = 0.01
    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        max_c = float(c[k_star])

        # Subgradient of max_k conv_k w.r.t. a
        # g_i = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        g = np.zeros(n)
        if i_min <= i_max:
            partners = k_star - np.arange(i_min, i_max + 1)
            g[i_min:i_max + 1] = 2.0 * a[partners]

        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm

        # Adaptive step size (Polyak-like)
        # Target: reduce max_c to some fraction
        step = step_scale / (np.sqrt(it + 1) + 1.0)
        step = min(step, 0.05)

        a_trial = a - step * g
        a_trial = _project_simplex(a_trial)

        sc_trial = _score_norm_fn(a_trial)
        if sc_trial < best_sc:
            best_sc = sc_trial
            best_a = a_trial.copy()

        # Always accept if it doesn't worsen too much (annealing)
        if sc_trial < best_sc * 1.02 or it < 100:
            a = a_trial.copy()
        else:
            # Try smaller step
            a_trial2 = a - step * 0.1 * g
            a_trial2 = _project_simplex(a_trial2)
            sc2 = _score_norm_fn(a_trial2)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_trial2.copy()
            a = a_trial2.copy()

        # Decay step scale
        if it % 5000 == 0:
            step_scale *= 0.8

    return best_a, best_sc


def _peak_shaving_descent(a_init, deadline, rng, n_trials=50000):
    """Coordinate-descent style peak shaving: reduce elements contributing to top lags."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 0.5:
            break

        c = _full_convolve(a, a)
        # Get top K lags
        K = min(8, len(c))
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])
        max_val = float(c[max_lag])

        # Strategy 1: Reduce the element with highest contribution to max lag
        if it % 4 == 0:
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min > i_max:
                continue
            pairs = np.arange(i_min, i_max + 1)
            partners = max_lag - pairs
            contrib = a[pairs] * a[partners]
            idx = pairs[int(np.argmax(contrib))]

            a_trial = a.copy()
            frac = rng.uniform(0.05, 0.25)
            delta = a_trial[idx] * frac
            if delta < 1e-15:
                continue
            a_trial[idx] -= delta
            a_trial[idx] = max(0.0, a_trial[idx])

            # Redistribute to positions with low contribution to top lags
            contrib_all = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    ps = kl - np.arange(lo, hi + 1)
                    contrib_all[lo:hi + 1] += a[ps]

            # Pick from lowest contribution (excluding idx)
            contrib_all[idx] = np.inf
            n_safe = max(1, n // 10)
            safe = np.argsort(contrib_all)[:n_safe]
            add_idx = int(safe[rng.integers(len(safe))])
            a_trial[add_idx] += delta
            a_trial /= a_trial.sum()

            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        # Strategy 2: Block perturbation
        elif it % 4 == 1:
            block_size = int(rng.integers(3, min(32, n // 16)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.75, 1.25, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        # Strategy 3: Pair swap with mass transfer
        elif it % 4 == 2:
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.05, 0.3)
            amt = a_trial[i1] * frac
            a_trial[i1] -= amt
            a_trial[i1] = max(0.0, a_trial[i1])
            a_trial[i2] += amt
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        # Strategy 4: Simultaneous reduction of top-2 peaks
        else:
            if len(top_lags) < 2:
                continue
            # Find elements that contribute to BOTH top peaks
            lag1, lag2 = int(top_lags[-1]), int(top_lags[-2])
            i1_min = max(0, lag1 - (n - 1))
            i1_max = min(n - 1, lag1)
            i2_min = max(0, lag2 - (n - 1))
            i2_max = min(n - 1, lag2)

            # Elements in both ranges
            lo = max(i1_min, i2_min)
            hi = min(i1_max, i2_max)
            if lo > hi:
                # No overlap, just reduce from lag1
                lo, hi = i1_min, i1_max
                if lo > hi:
                    continue

            candidates_range = np.arange(lo, hi + 1)
            if len(candidates_range) == 0:
                continue
            # Pick the one with highest combined contribution
            p1 = lag1 - candidates_range
            p2 = lag2 - candidates_range
            combined = np.maximum(0, a[candidates_range] * np.maximum(0, a[p1])) + \
                       np.maximum(0, a[candidates_range] * np.maximum(0, a[p2]))
            idx = candidates_range[int(np.argmax(combined))]

            a_trial = a.copy()
            delta = a_trial[idx] * rng.uniform(0.05, 0.2)
            a_trial[idx] = max(0.0, a_trial[idx] - delta)
            # Add to a random safe position
            add_idx = int(rng.integers(n))
            a_trial[add_idx] += delta
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_sc


def _nesterov_smoothed(a_init, deadline, rng, lr=0.005, k_top=3, momentum=0.9):
    """Nesterov-accelerated gradient with top-K peak focusing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    v = np.zeros(n)  # velocity for Nesterov

    def compute_topk_grad(a_curr, k):
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        # Weighted subgradient
        g = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            if lo <= hi:
                partners = k_val - np.arange(lo, hi + 1)
                g[lo:hi + 1] += 2.0 * a_curr[partners]
        g /= k_actual
        g *= 4.0 * n
        return g

    max_iters = 80000
    for it in range(max_iters):
        if time.monotonic() > deadline - 0.5:
            break

        # Nesterov lookahead
        z_lookahead = z - momentum * v
        a_curr = _softmax_fn(z_lookahead)
        grad = compute_topk_grad(a_curr, k_top)

        # Project gradient to z-space
        a_curr2 = _softmax_fn(z)
        dot_val = float(np.dot(a_curr2, grad))
        grad_z = a_curr2 * (grad - dot_val)

        # Nesterov update
        v = momentum * v - lr * grad_z
        z = z - v
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        # Decay learning rate
        if it % 5000 == 0:
            lr *= 0.9
            momentum = min(0.999, momentum + 0.001)

        if it % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    return best_a, best_score


def _multi_beta_lbfgs(a_init, deadline, rng, beta_scales=None):
    """Run L-BFGS with multiple beta scales for robustness."""
    if beta_scales is None:
        beta_scales = [1.0, 0.5, 0.3, 0.15, 0.08]

    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)
    n_scales = len(beta_scales)
    per_scale = max(5.0, (deadline - time.monotonic()) / n_scales)

    for bs in beta_scales:
        if time.monotonic() > deadline - 5.0:
            break
        dl = min(deadline - 2.0, time.monotonic() + per_scale)
        a_best, sc_best = _lbfgs_logsumexp(best_a, dl, rng, beta_scale=bs,
                                            max_wall_time=per_scale, min_beta_frac=0.00002)
        if sc_best < best_sc:
            best_sc = sc_best
            best_a = a_best.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-method optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Combines L-BFGS log-sum-exp, subgradient, peak-shaving coordinate descent,
    Nesterov-accelerated gradient, and multi-restart strategies.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_triangular_weight(n_parent)
        sc_parent = _score_norm_fn(parent)
        update_best(parent, sc_parent, "triangular_init")

    n_work = n_parent
    print(f"Working with n={n_work}")

    # --- Phase 0: Diverse initialization (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    n_try = n_work

    profiles = [
        _make_skew_triangular(n_try, 1.5),
        _make_skew_triangular(n_try, 2.0),
        _make_skew_triangular(n_try, 2.5),
        _make_skew_triangular(n_try, 3.0),
        _make_skew_triangular(n_try, 0.5),
        _make_skew_triangular(n_try, 0.7),
        _make_ramped_profile(n_try, 0.3, 0.1, 1.0),
        _make_ramped_profile(n_try, 0.1, 0.3, 1.0),
        _make_ramped_profile(n_try, 0.4, 0.2, 0.5),
        _make_ramped_profile(n_try, 0.2, 0.4, 0.5),
        _make_bimodal_profile(n_try, 0.3, 0.2),
        _make_bimodal_profile(n_try, 0.4, 0.15),
        _make_bimodal_profile(n_try, 0.5, 0.1),
        _make_w_profile(n_try, 2, 0.4),
        _make_w_profile(n_try, 3, 0.3),
        _make_power_profile(n_try, 2.0, 0.3),
        _make_asymmetric_plateau(n_try, 0.2, 0.1, 2.0),
    ]

    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        update_best(p, sc, f"prof{idx}")

    print(f"Phase 0 done: best={best_score:.10f}")

    # --- Phase 1: Primary L-BFGS optimization (500s) ---
    phase1_end = min(deadline, t_start + 540.0)
    if best_seq is not None:
        print(f"Phase 1: Multi-beta L-BFGS, start score={best_score:.10f}")
        dl = phase1_end - 5.0
        a_best, sc_best = _multi_beta_lbfgs(best_seq, dl, rng,
                                             beta_scales=[1.0, 0.5, 0.25, 0.1])
        update_best(a_best, sc_best, "lbfgs_primary")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Subgradient method (250s) ---
    phase2_end = min(deadline, t_start + 790.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: Subgradient method")
        dl = phase2_end - 5.0
        a_best, sc_best = _subgradient_method(best_seq, dl, rng, n_iters=40000)
        update_best(a_best, sc_best, "subgrad")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Peak-shaving descent (250s) ---
    phase3_end = min(deadline, t_start + 1040.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Peak-shaving descent")
        dl = phase3_end - 5.0
        a_best, sc_best = _peak_shaving_descent(best_seq, dl, rng, n_trials=40000)
        update_best(a_best, sc_best, "peak_shave")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Nesterov accelerated (200s) ---
    phase4_end = min(deadline, t_start + 1240.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Nesterov smoothed")
        dl = phase4_end - 5.0
        a_best, sc_best = _nesterov_smoothed(best_seq, dl, rng, lr=0.004, k_top=3, momentum=0.9)
        update_best(a_best, sc_best, "nesterov_k3")

        if time.monotonic() < phase4_end - 30:
            dl2 = phase4_end - 5.0
            a_best2, sc_best2 = _nesterov_smoothed(best_seq, dl2, rng, lr=0.003, k_top=5, momentum=0.92)
            update_best(a_best2, sc_best2, "nesterov_k5")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multi-restart (250s) ---
    phase5_end = min(deadline, t_start + 1490.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Multi-restart")
        n_restarts = 5
        per_restart = max(15.0, (phase5_end - time.monotonic()) / max(1, n_restarts))

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            noise = [0.01, 0.02, 0.03, 0.05, 0.08][i % 5]
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            p_start = _softmax_fn(z)

            dl = min(phase5_end - 5, time.monotonic() + per_restart)
            a_best, sc_best = _lbfgs_logsumexp(p_start, dl, rng, beta_scale=0.5,
                                                max_wall_time=per_restart, min_beta_frac=0.00005)
            update_best(a_best, sc_best, f"restart{i}_n{noise}")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Coarse-to-fine refinement (150s) ---
    phase6_end = min(deadline, t_start + 1640.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        n_curr = len(best_seq)
        # Try refining at 2x resolution
        if time.monotonic() < phase6_end - 60:
            n_fine = min(n_curr * 2, 16384)
            if n_fine > n_curr:
                p_fine = _interp_profile(best_seq, n_fine)
                sc_fine = _score_norm_fn(p_fine)
                print(f"  Interp to n={n_fine}: score={sc_fine:.10f}")
                dl = min(phase6_end - 10, time.monotonic() + 80.0)
                a_best, sc_best = _lbfgs_logsumexp(p_fine, dl, rng, beta_scale=0.5,
                                                    max_wall_time=70.0, min_beta_frac=0.0001)
                update_best(a_best, sc_best, f"fine_n{n_fine}")

    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Final intensive polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(15.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")

        # 7a: Very fine L-BFGS
        if remaining > 40:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.4)
            a_best, sc_best = _lbfgs_logsumexp(best_seq, dl, rng, beta_scale=0.2,
                                                max_wall_time=remaining * 0.4,
                                                min_beta_frac=0.00001)
            update_best(a_best, sc_best, "final_lbfgs_fine")

        # 7b: Peak shaving
        if time.monotonic() < deadline - 15:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _peak_shaving_descent(best_seq, dl, rng, n_trials=20000)
            update_best(a_best, sc_best, "final_peak")

        # 7c: Another Nesterov pass
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 5, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _nesterov_smoothed(best_seq, dl, rng, lr=0.002, k_top=2, momentum=0.95)
            update_best(a_best, sc_best, "final_nesterov")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular_weight(8192)
    print(f"FALLBACK: n=8192, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
