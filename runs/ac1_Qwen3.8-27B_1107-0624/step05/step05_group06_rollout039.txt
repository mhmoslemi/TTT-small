"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Recursive flat-convolution initialization
2. Nesterov momentum subgradient with adaptive step size
3. L-BFGS with log-sum-exp smoothing (improved beta schedule)
4. Coordinate descent local search for final polishing
5. Multiple sequence lengths (2048, 4096) and diverse restarts
Key differences: direct non-negative space with clipping, momentum-based
subgradient, recursive flat-conv init, and adaptive Polyak-inspired steps.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_grad_nonneg(a, tol_frac=0.001):
    """Subgradient in non-negative space (no normalization constraint)."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return np.zeros(n)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    threshold = c_max * (1.0 - tol_frac)
    active = c >= threshold
    if not active.any():
        idx_max = int(np.argmax(c))
        active = np.zeros_like(c, dtype=bool)
        active[idx_max] = True
    K = float(active.sum())
    w = np.zeros_like(c)
    w[active] = 1.0 / K

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))

    grad = 4.0 * n * (h / (S * S) - W * a / (S * S * S))
    return grad, c_max, S


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_optimize_z(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxcor': 40, 'maxls': 80})
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _flat_conv_init(n):
    """Recursively compute a sequence whose convolution is approximately flat."""
    V = 1.0
    a = np.zeros(n)
    a[0] = np.sqrt(V)
    for k in range(1, n):
        s = 0.0
        for i in range(1, k):
            if i < n and (k - i) < n:
                s += a[i] * a[k - i]
        a[k] = (V - s) / (2.0 * a[0])
        if a[k] < 0:
            a[k] = 0.0
    s_total = a.sum()
    if s_total > 1e-15:
        a = a / s_total
    else:
        a = np.ones(n) / n
    return a


def _flat_conv_init_symmetric(n):
    """Compute flat-conv init then symmetrize."""
    a = _flat_conv_init(n)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    else:
        a_sym = np.ones(n) / n
    return a_sym


def _momentum_subgrad(a_start, deadline, n_max_steps=20000, base_lr=0.01,
                      tol_frac=0.001, momentum_init=0.9):
    """Nesterov momentum subgradient in non-negative space."""
    n = len(a_start)
    a = a_start.copy().astype(np.float64)
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = _score(a)
    lr = base_lr
    momentum = momentum_init
    t = 0

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        t += 1

        grad, c_max, S = _compute_grad_nonneg(a, tol_frac)

        # Nesterov look-ahead
        a_look = a - momentum * v
        a_look = np.maximum(a_look, 0)
        if a_look.sum() < 1e-15:
            a_look = a.copy()

        grad_look, _, _ = _compute_grad_nonneg(a_look, tol_frac)

        # Update velocity
        v = momentum * v + lr * grad_look
        a = a - v
        a = np.maximum(a, 0)

        # Normalize to keep scale reasonable
        s = a.sum()
        if s > 1e-10:
            a = a / s

        # Adaptive learning rate
        if i % 200 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                lr = min(0.05, lr * 1.02)
            else:
                lr *= 0.85
                v *= 0.5

        # Decay momentum slightly
        momentum = min(0.999, momentum_init + 0.099 * i / max(1, i + 100))

    if _score(a) < best_sc:
        best_sc = _score(a)
        best_a = a.copy()
    return best_a, best_sc


def _coord_descent(a_start, deadline, n_sweeps=20):
    """Coordinate descent: try adjusting each coordinate to reduce local peaks."""
    n = len(a_start)
    a = a_start.copy().astype(np.float64)
    best_sc = _score(a)
    best_a = a.copy()

    idx = np.arange(n)
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        rng = np.random.default_rng(42 + sweep)
        perm = rng.permutation(n)
        improved = False
        for j in perm:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            # Try a few multiplicative factors
            for factor in [0.8, 0.9, 1.1, 1.2]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                a[j] = new_val
                s = a.sum()
                if s < 1e-10:
                    a[j] = old_val
                    continue
                sc = _score(a / s)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[j] = old_val
        if not improved:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_conv":
        return _flat_conv_init(n)
    elif kind == "flat_conv_sym":
        return _flat_conv_init_symmetric(n)
    else:
        return np.ones(n) / n

    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

def _perturb_fourier(a, rng, n_modes=3, amp_scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 3))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, n_blocks=4):
    """Perturb by adding random block structure."""
    n = len(a)
    block_size = n // n_blocks
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(n_blocks):
        start = rng.integers(0, max(1, n - block_size))
        length = max(1, block_size // 2)
        amp = 0.03 * rng.uniform(0.5, 2.0) * rng.choice([-1, 1])
        z[start:start + length] += amp
    z = z - z.mean()
    return _softmax(z)
