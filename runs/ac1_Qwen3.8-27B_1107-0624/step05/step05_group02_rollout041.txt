        k <<= 1

"""
Hybrid LP + Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated convolution, adaptive soft-max gradient, and diverse restarts.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_vec(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max(c, beta):
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 0.0
    return float(c.max() + beta * np.log(s))


def _grad_soft_max(c, w, a):
    """Compute gradient of soft_max(conv(a,a), beta) w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j.
    """
    n = len(a)
    # h[j] = sum_k w[k] * a[k-j] * [0 <= k-j < n]
    # This is cross-correlation of w and a
    # w has length 2n-1, a has length n
    # h[j] = sum_{k=j}^{j+n-1} w[k] * a[k-j] (with bounds)
    
    # Use FFT: cross-correlation
    # correlate(w, a)[j] = sum_k w[k] * a[k-j]
    N = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    # Cross-correlation: ifft(conj(fft(w)) * fft(a)) gives sum_k w[k] * a[k-lag]
    # Actually, let me use: correlate(w, a) at lag j = sum_k w[k+j] * a[k] = sum_m w[m] * a[m-j]
    # That's what we want: h[j] = sum_m w[m] * a[m-j]
    # In FFT terms: ifft(fft(w) * conj(fft(a))) gives the cross-correlation
    # Let me verify: ifft(fft(w) * conj(fft(a)))[j] = sum_k w[k] * a[(k-j) mod N]
    # For our case (non-circular), we need padding.
    
    # Alternative: just use the fact that h[j] = (w * a_rev)[j + n - 1] where a_rev = a[::-1]
    # conv(w, a_rev)[j+n-1] = sum_k w[k] * a_rev[j+n-1-k] = sum_k w[k] * a[n-1-(j+n-1-k)] = sum_k w[k] * a[k-j]
    # Yes! So h = conv(w, a[::-1])[n-1 : n-1+n]
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)  # length (2n-1) + n - 1 = 3n-1
    h = wc[n-1 : n-1+n]  # length n
    return 2.0 * h


def _adam_smooth_max(a_init, deadline, beta_schedule=None):
    """Adam optimizer for smooth max of convolution."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_m = float(c_init.max())
    max_c_init = best_m
    
    if beta_schedule is None:
        beta_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta_frac in beta_schedule:
        if time.monotonic() > deadline:
            break
        beta = beta_frac * max_c_init
        lr = 0.01
        max_it = 50000
        
        iters = 0
        while iters < max_it:
            if time.monotonic() > deadline:
                break
            iters += 1
            t_step += 1
            
            a = _softmax_vec(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            # Compute weights for soft-max gradient
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Compute gradient
            h = _grad_soft_max(c, w, a)
            # The gradient of the soft-max objective (unscaled) is 2*h
            # But we want to minimize the soft-max of conv, so grad = 2*h
            grad = 2.0 * h
            
            # Normalize gradient for stability
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad = grad / gnorm * np.sqrt(n)  # Normalize to have unit RMS
            
            # Adam update
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            # Periodic evaluation
            if iters % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_m:
                    best_m = sc
                    best_a = a.copy()
            
            # Adaptive learning rate decay
            if iters % 5000 == 0:
                lr = max(0.001, lr * 0.9)
        
        # End of beta phase: update best
        a_final = _softmax_vec(z)
        sc_final = float(_full_convolve(a_final, a_final).max())
        if np.isfinite(sc_final) and sc_final < best_m:
            best_m = sc_final
            best_a = a_final.copy()
    
    return best_a, best_m

def _lp_alternate(f_init, deadline, time_per_lp=15):
    """LP-based alternating optimization with reduced constraints."""
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    
    best_f = f.copy()
    best_m = float(_full_convolve(f, f).max())
    
    steps = 0
    stuck = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(f, f)
        M = float(c.max())
        
        # Use reduced constraint set: only near-active constraints
        threshold = 0.80 * M
        active_mask = c >= threshold
        k_active = np.where(active_mask)[0]
        
        if len(k_active) < 10:
            k_active = np.where(c >= 0.5 * M)[0]
        
        # Build reduced LP matrix
        n_constraints = len(k_active)
        rows = []
        cols = []
        vals = []
        
        for idx, k in enumerate(k_active):
            # Row idx: sum_j a[j] * f[k-j] for valid j
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_min <= j_max:
                js = np.arange(j_min, j_max + 1)
                vals.extend(f[k - js])  # f[k-j] for j in js
                rows.extend([idx] * len(js))
                cols.extend(js.tolist())
        
        if not rows:
            break
        
        A = csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
        
        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(n_constraints, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': time_per_lp, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            else:
                g = None
        except Exception:
            g = None
        
        if g is None or g.sum() < 1e-15:
            stuck += 1
            if stuck > 10:
                break
            continue
        
        stuck = 0
        g = g / g.sum()
        
        # Try several step sizes
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            f_new_sum = f_new.sum()
            if f_new_sum < 1e-15:
                continue
            f_new = f_new / f_new_sum
            M_new = float(_full_convolve(f_new, f_new).max())
            if M_new < M:
                f = f_new
                if M_new < best_m:
                    best_m = M_new
                    best_f = f.copy()
                    print(f"  LP step {steps}: M={M:.10f} -> {M_new:.10f}")
                steps += 1
                improved = True
                break
            else:
                continue
        
        if not improved:
            stuck += 1
            if stuck > 15:
                break
    
    return best_f, best_m


def _make_hann(n):
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    return w / w.sum()


def _make_exp(n, alpha=1.0):
    a = np.exp(-alpha * np.arange(n) / n)
    return a / a.sum()


def _make_two_sided(n, alpha=2.0):
    x = np.arange(n)
    a = np.exp(-alpha * np.minimum(x, n - 1 - x) / (n / 2))
    return a / a.sum()


def _make_polynomial(n, p=2.0):
    x = np.linspace(0, 1, n)
    a = x ** p + (1 - x) ** p + 0.1
    return a / a.sum()


def _make_cosine_mod(n, f=3.0, depth=0.3):
    x = np.arange(n)
    a = 1.0 + depth * np.cos(2 * np.pi * f * x / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

"""
Hybrid LP + Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-accelerated convolution, reduced-constraint LP alternating,
adaptive soft-max gradient with mirror descent, diverse restarts, and n exploration.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_vec(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_soft_max(c, w, a):
    """Compute 2 * cross_correlate(w, a) efficiently via FFT.
    w has length 2n-1, a has length n.
    Returns array of length n: grad[j] = 2 * sum_k w[k] * a[k-j] (valid k-j).
    """
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1: n - 1 + n]
    return 2.0 * h

    for beta_frac in beta_schedule:

        beta = beta_frac * max_ref
        lr = lr_init

        no_improve = 0
        phase_best = best_m
        last_eval_m = best_m

            iters += 1

            a = _softmax_vec(z)

            cm = float(c.max())

            c_shifted = (c - cm) / beta
            mx = np.max(c_shifted)
            if mx > 50:

            mh = m_buf / (1.0 - b1 ** t_step)

            j_max = min(k, n - 1)
            if j_min <= j_max:
                js = np.arange(j_min, j_max + 1)
                vals_list = f[k - js].tolist()
                vals.extend(vals_list)
                rows.extend([idx] * len(js))
                cols.extend(js.tolist())

        if not rows:
            break

        A = csr_matrix((vals, (rows, cols)), shape=(len(k_active), n))
        c_obj = -np.ones(n)
        b_ub = np.full(len(k_active), M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': time_per_lp, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            else:
                g = None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            stuck += 1
            if stuck > 8:
                break
            continue

        stuck = 0
        g = g / g.sum()

        improved = False
        for t in [t_ls, t_ls * 0.6, t_ls * 0.3, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            fs = f_new.sum()
            if fs < 1e-15:
                continue
            f_new = f_new / fs
            M_new = float(_full_convolve(f_new, f_new).max())
            if M_new < M * (1 - 1e-12):
                f = f_new
                if M_new < best_m:
                    best_m = M_new
                    best_f = f.copy()
                steps += 1
                t_ls = min(0.8, t * 1.3)
                improved = True
                if steps % 5 == 0:
                    print(f"  LP step {steps}: M={M:.10f} -> {M_new:.10f} (t={t:.3f})")
                break
        else:
            t_ls *= 0.7
            stuck += 1
            if stuck > 12:
                break

    return best_f, best_m


def _make_hann(n):
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)))
    return w / w.sum()


def _make_exp(n, alpha=1.0):
    a = np.exp(-alpha * np.arange(n) / n)
    return a / a.sum()


def _make_two_sided(n, alpha=2.0):
    x = np.arange(n)
    a = np.exp(-alpha * np.minimum(x, n - 1 - x) / max(1, n // 2))
    return a / a.sum()


def _make_cosine_mod(n, f=3.0, depth=0.3):
    x = np.arange(n)
    a = 1.0 + depth * np.cos(2 * np.pi * f * x / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _make_blackman(n):
    a = 0.42 + 0.5 * np.cos(2 * np.pi * np.arange(n) / max(1, n - 1)) + \
        0.08 * np.cos(4 * np.pi * np.arange(n) / max(1, n - 1))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_vec(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization: LP alternating (reduced constraints) + Adam soft-max
    + diverse restarts + n exploration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        _update_best(parent)
        print(f"Loaded parent: n={len(parent)}, score={best_score:.10f}")
    else:
        parent = _make_hann(4096)
        _update_best(parent)
        print(f"No parent, using Hann n=4096, score={best_score:.10f}")

    n_main = len(parent)

    # === Phase 1: LP alternating (0-300s) ===
    phase1_end = min(deadline, t_start + 300.0)
    print(f"\n=== Phase 1: LP alternating (n={n_main}, {phase1_end - t_start:.0f}s) ===")

    if time.monotonic() < phase1_end - 20:
        f_res, m_res = _lp_alternate(parent, phase1_end, time_per_lp=8)
        _update_best(f_res)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam soft-max (300s-950s) ===
    phase2_end = min(deadline, t_start + 950.0)
    print(f"\n=== Phase 2: Adam soft-max (n={n_main}, {phase2_end - phase1_end:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase2_end - 20:
        a_init = best_seq / best_seq.sum()
        remaining = phase2_end - time.monotonic() - 10
        a_opt, m_opt = _adam_smooth_max(
            a_init, time.monotonic() + remaining,
            beta_schedule=[0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001],
            lr_init=0.008
        )
        _update_best(a_opt)

    # Second Adam from perturbed start
    if best_seq is not None and time.monotonic() < phase2_end - 50:
        a_pert = _perturb(best_seq / best_seq.sum(), 0.04, rng)
        remaining = phase2_end - time.monotonic() - 10
        a_opt2, _ = _adam_smooth_max(
            a_pert, time.monotonic() + remaining,
            beta_schedule=[0.04, 0.025, 0.015, 0.008, 0.004, 0.002],
            lr_init=0.007
        )
        _update_best(a_opt2)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Diverse restarts (950s-1350s) ===
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 3: Diverse restarts (n={n_main}, {phase3_end - phase2_end:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase3_end - 20:
        a_best = best_seq / best_seq.sum()

        candidates = []
        candidates.append(_perturb(a_best, 0.02, rng))
        candidates.append(_perturb(a_best, 0.08, rng))
        candidates.append(_perturb(a_best, 0.15, rng))
        candidates.append(_make_hann(n_main))
        candidates.append(_make_blackman(n_main))
        candidates.append(_make_two_sided(n_main, 2.5))
        candidates.append(_make_exp(n_main, 1.0))
        candidates.append(_make_cosine_mod(n_main, 2.0, 0.25))
        candidates.append(_make_cosine_mod(n_main, 5.0, 0.2))
        # Blends
        candidates.append(0.6 * a_best + 0.4 * _make_hann(n_main))
        candidates.append(0.7 * a_best + 0.3 * _make_two_sided(n_main, 3.0))
        candidates.append(0.5 * a_best + 0.5 * _make_exp(n_main, 2.0))

        # Normalize
        candidates = [c / c.sum() for c in candidates]

        n_cands = len(candidates)
        per_cand = max(25, (phase3_end - time.monotonic() - 15) / n_cands)

        for i, cand in enumerate(candidates):
            if time.monotonic() > phase3_end - 15:
                break
            remaining = min(per_cand, phase3_end - time.monotonic() - 8)
            if remaining < 15:
                break
            print(f"  Restart {i+1}/{n_cands} ({remaining:.0f}s)")
            a_opt, _ = _adam_smooth_max(
                cand, time.monotonic() + remaining,
                beta_schedule=[0.05, 0.03, 0.015, 0.008, 0.004],
                lr_init=0.007
            )
            _update_best(a_opt)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: n exploration (1350s-1600s) ===
    phase4_end = min(deadline, t_start + 1600.0)
    print(f"\n=== Phase 4: n exploration ({phase4_end - phase3_end:.0f}s) ===")

    if best_seq is not None and time.monotonic() < phase4_end - 20:
        target_ns = [4096, 12288, 16384]
        for target_n in target_ns:
            if time.monotonic() > phase4_end - 40:
                break
            if target_n == n_main:
                continue
            p_interp = _interp_profile(best_seq, target_n)
            _update_best(p_interp)

            remaining = phase4_end - time.monotonic() - 20
            if remaining > 40:
                print(f"  Optimizing n={target_n} ({remaining:.0f}s)")
                a_opt, _ = _adam_smooth_max(
                    p_interp, time.monotonic() + remaining,
                    beta_schedule=[0.05, 0.03, 0.015, 0.008, 0.004],
                    lr_init=0.007
                )
                _update_best(a_opt)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish (1600s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 8:
        a_best = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 5
        if remaining > 15:
            a_final, _ = _adam_smooth_max(
                a_best, time.monotonic() + remaining,
                beta_schedule=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                lr_init=0.003
            )
            _update_best(a_final)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_hann(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_hann(4096)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
