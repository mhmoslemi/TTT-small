from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):

    if s < 1e-300 or not np.isfinite(s):

"""
Multi-scale LP alternating + L-BFGS log-sum-exp + paired coordinate descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses fast sparse LP with precomputed indices, adaptive step size, and multi-restart.

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, identity as sp_identity, hstack, vstack
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _conv_matrix_indices(n):
    """Precompute row/col indices for the convolution matrix."""
    rows = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)[None, :]).ravel()
    cols = np.tile(np.arange(n, dtype=np.int64), n)
    return rows, cols


def _build_conv_matrix(f, rows, cols):
    """Build sparse convolution matrix using precomputed indices."""
    data = np.tile(f, len(f))
    return csr_matrix((data, (rows, cols)), shape=(2 * len(f) - 1, len(f)))


def _lp_minimize(f, rows, cols):
    """
    Solve: min t s.t. P @ b <= t, sum(b)=1, b>=0
    where P is the convolution matrix for f.
    Returns (b_opt, t_opt) or (None, None).
    """
    n = len(f)
    P = _build_conv_matrix(f, rows, cols)
    
    # A_ub = [P, -1; -I, 0], shape (3n-1, n+1)
    # b_ub = zeros(3n-1)
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0  # t doesn't appear in the equality
    b_eq = np.array([1.0])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Build sparse A_ub
    # Top: [P, -1] (2n-1 rows)
    n_top = 2 * n - 1
    col_neg1 = csr_matrix(np.ones((n_top, 1)) * (-1.0))
    top = hstack([P.tocsc(), col_neg1], format='csc')
    
    # Bottom: [-I, 0] (n rows)
    neg_I = -sp_identity(n, format='csc')
    col_zeros = csr_matrix(np.zeros((n, 1)))
    bottom = hstack([neg_I, col_zeros], format='csc')
    
    A_ub = vstack([top, bottom], format='csc')
    b_ub = np.zeros(3 * n - 1)
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b_opt = np.maximum(res.x[:n], 0)
            s = b_opt.sum()
            if s > 1e-15:
                b_opt = b_opt / s
            return b_opt, float(res.x[n])
    except Exception:
        pass
    return None, None

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _paired_coord_descent(a_init, deadline, max_rounds=200):
    """Paired coordinate descent: update pairs of coordinates."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        improved = False
        
        # Random pairs
        for _ in range(2 * n):
            if time.monotonic() > deadline + 5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            best_delta = 0.0
            best_local = sc
            
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[i] += delta
                z_trial[j] -= delta
                a_trial = _softmax_fn(z_trial)
                sc_trial = _score_norm_fn(a_trial)
                if sc_trial < best_local:
                    best_local = sc_trial
                    best_delta = delta
            
            if best_delta != 0.0:
                z[i] += best_delta
                z[j] -= best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                improved = True
        
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Uniform
    profiles.append(np.ones(n) / n)
    
    # Bell curves
    for alpha in [1.0, 2.0, 3.0, 5.0, 8.0]:
        a = np.exp(-alpha * u ** 2)
        profiles.append(a / a.sum())
    
    # U-shapes
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        a = np.abs(u) ** alpha + 0.01
        profiles.append(a / a.sum())
    
    # Sine-based
    for k in [1, 2, 3]:
        a = np.sin(np.pi * t * k) ** 2 + 0.01
        profiles.append(a / a.sum())
    
    # Cosine-based
    for k in [1, 2, 3]:
        a = np.cos(np.pi * u * k / 2) ** 2 + 0.01
        profiles.append(a / a.sum())
    
    # Power laws
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = (0.01 + np.abs(u)) ** (-p)
        a = np.maximum(a, 0)
        profiles.append(a / a.sum())
    
    # Linear ramps
    a = t + 0.01
    profiles.append(a / a.sum())
    a = (1 - t) + 0.01
    profiles.append(a / a.sum())
    
    # Two-peak
    a = np.exp(-20 * (u - 0.5) ** 2) + np.exp(-20 * (u + 0.5) ** 2) + 0.01
    profiles.append(a / a.sum())
    
    # Three-peak
    a = np.exp(-15 * (u - 0.6) ** 2) + np.exp(-15 * u ** 2) + np.exp(-15 * (u + 0.6) ** 2) + 0.01
    profiles.append(a / a.sum())
    
    return profiles


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()
