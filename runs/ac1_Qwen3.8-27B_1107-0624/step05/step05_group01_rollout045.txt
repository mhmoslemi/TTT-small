    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    for _ in range(200):
        u = np.maximum(v, 0.0)
        s = u.sum()
        if s < 1e-300:
            return np.ones(n) / n
        u = u / s
        if np.all(u >= -1e-15):
            return np.maximum(u, 0.0)
        # Bisection for threshold
        break
    # Use the standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _barzilai_borwein_step(x_prev, x_cur, g_prev, g_cur, s_min=1e-20, s_max=1e3):
    """Compute Barzilai-Borwein step size."""
    s = x_cur - x_prev
    y = g_cur - g_prev
    ys = float(np.dot(y, s))
    ss = float(np.dot(s, s))
    yy = float(np.dot(y, y))
    if ss < s_min or abs(ys) < s_min:
        return 0.01
    if ys > 0:
        alpha = ss / ys
    else:
        alpha = ss / yy if yy > s_min else 0.01
    return float(np.clip(alpha, 1e-10, 1e4))


def _pg_topk_phase(a_init, deadline, K=5, max_iter=50000, lr_init=0.005):
    """Projected gradient with top-K softmax for the max of conv peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    c_prev = _full_convolve(a, a)
    g_prev = None
    a_prev = a.copy()
    lr = lr_init
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Top-K peaks with softmax weighting
        sorted_idx = np.argsort(c)[::-1]
        top_k = sorted_idx[:K]
        peak_vals = c[top_k]
        peak_max = float(peak_vals.max())
        
        if peak_max < 1e-300:
            break
        
        # Softmax weights over top-K
        beta_pk = 0.01 * peak_max
        shifted = (peak_vals - peak_max) / beta_pk
        e = np.exp(shifted - shifted.max())
        w = e / e.sum()
        
        # Gradient of weighted top-K peaks
        grad = np.zeros(n)
        for idx, k in enumerate(top_k):
            wi = w[idx]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad[i_lo:i_hi + 1] += 2.0 * wi * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        
        # Project gradient onto constraint sum(a)=1
        grad = grad - grad.mean()
        
        # Barzilai-Borwein step size
        if g_prev is not None:
            lr_new = _barzilai_borwein_step(a_prev, a, g_prev, grad)
            lr = 0.5 * lr + 0.5 * lr_new
            lr = np.clip(lr, 1e-10, 0.1)
        
        # Take step
        a_prev = a.copy()
        g_prev = grad.copy()
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        
        # Check improvement
        c_new = _full_convolve(a_new, a_new)
        cm_new = float(c_new.max())
        
        if cm_new < best_score - 1e-15:
            best_score = cm_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            a_new = a  # Revert
            no_improve += 1
        
        a = a_new
        c_prev = c_new if cm_new < best_score else c
        
        if no_improve > 200:
            # Perturb and restart
            a = a + 0.002 * np.random.randn(n)
            a = _project_simplex(a)
            no_improve = 0
            lr = lr_init
            g_prev = None
    
    return best_a, best_score


def _peak_pull_phase(a_init, deadline, max_rounds=200):
    """Directly pull down the highest peak by adjusting the sequence."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        peak_k = int(np.argmax(c))
        
        # Compute gradient of the peak at position k
        grad = np.zeros(n)
        i_lo = max(0, peak_k - n + 1)
        i_hi = min(n - 1, peak_k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] = a[peak_k - i_lo:peak_k - i_lo + (i_hi - i_lo + 1)]
        
        # Also consider the second-highest peak
        c_no_max = c.copy()
        c_no_max[peak_k] = -np.inf
        peak_k2 = int(np.argmax(c_no_max))
        if c_no_max[peak_k2] > 0.9 * cm:
            i_lo2 = max(0, peak_k2 - n + 1)
            i_hi2 = min(n - 1, peak_k2)
            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2 + 1] += 0.5 * a[peak_k2 - i_lo2:peak_k2 - i_lo2 + (i_hi2 - i_lo2 + 1)]
        
        # Project gradient
        grad = grad - grad.mean()
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-20:
            break
        
        # Try multiple step sizes
        improved = False
        for ls in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a - ls * grad / g_norm
            a_trial = _project_simplex(a_trial)
            c_trial = _full_convolve(a_trial, a_trial)
            cm_trial = float(c_trial.max())
            if cm_trial < best_score - 1e-15:
                best_score = cm_trial
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                break
        
        if not improved:
            # Try a smaller step or perturb
            break
    
    return best_a, best_score


def _lbfgs_lse_phase(a_init, deadline, beta_list=None, maxiter_per_beta=8000):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        # Very fine geometric schedule
        ratio = 0.72
        betas = []
        b = 0.3 * m0
        while b > 0.0001 * m0 and len(betas) < 40:
            betas.append(b)
            b *= ratio
        beta_list = betas
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        def obj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)
        
        def grad_fn(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return np.zeros_like(z_in)
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            e = np.exp(shifted - max_s)
            w = e / e.sum()
            a_rev = a_in[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            W = float(np.dot(w, c_in))
            g = 4.0 * n * a_in * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_in)
            return g
        
        try:
            result = minimize(
                obj, z, method='L-BFGS-B', jac=grad_fn,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-25,
                         'gtol': 1e-16, 'maxls': 50, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _adam_fine(a_init, deadline, lr=0.005, max_iter=500000):
    """Adam with very fine adaptive beta for final polish."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.005 * cm0
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 20
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-10 * cm, cm0 * 0.005 * (0.95 ** (t // 100)))
        
        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -200, 200)
        z = z - z.mean()
        
        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                z = z + 0.01 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0
                lr = max(lr * 0.7, 0.0001)
    
    return best_a, best_score


def _random_coord_search(a_init, deadline, max_iter=200000, n_try=5):
    """Random coordinate perturbation search."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Pick random coordinates to perturb
        idx = rng.integers(0, n, size=min(n_try, n))
        a_trial = a.copy()
        for i in idx:
            factor = 1.0 + rng.uniform(-0.05, 0.05)
            a_trial[i] = max(0.0, a_trial[i] * factor)
        
        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s
        
        c_trial = _full_convolve(a_trial, a_trial)
        cm_trial = float(c_trial.max())
        
        if cm_trial < best_score - 1e-15:
            best_score = cm_trial
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_score


def _symmetry_exploit(a_init, deadline):
    """Try symmetrized and anti-symmetrized versions."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    # Symmetric version
    a_sym = 0.5 * (a + a[::-1])
    a_sym = a_sym / a_sym.sum()
    sc_sym = _score_norm_fn(a_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_a = a_sym.copy()
    
    # Blended versions
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
        a_bl = (1 - alpha) * a + alpha * a[::-1]
        a_bl = np.maximum(a_bl, 1e-15)
        a_bl = a_bl / a_bl.sum()
        sc_bl = _score_norm_fn(a_bl)
        if sc_bl < best_score:
            best_score = sc_bl
            best_a = a_bl.copy()
    
    # Optimize the best from here with PG
    if time.monotonic() < deadline:
        a_pg, sc_pg = _pg_topk_phase(best_a, deadline, K=5, max_iter=20000, lr_init=0.003)
        if sc_pg < best_score:
            best_score = sc_pg
            best_a = a_pg.copy()
    
    return best_a, best_score


def _lp_phase_v2(a_init, deadline, K=50, max_rounds=8):
    """Improved LP peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        
        # Get active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.97 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]
        
        # Build constraint matrix
        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        
        b_vec = np.full(len(active), M * 1.0001)
        
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20, 'presolve': True})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break
        
        # Blend
        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-13:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break
        
        if not improved:
            break
    
    return best_a, best_score


def _coord_descent_log(a_init, deadline, max_rounds=50):
    """Cycle through coordinates optimizing each one (in log space)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15)).copy()
    z = z - z.mean()
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved_any = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Optimize z[i] by golden section (1D)
            z_old = z[i]
            a_cur = _softmax_fn(z)
            c_cur = _full_convolve(a_cur, a_cur)
            cur_max = float(c_cur.max())
            
            # Try a few values
            best_zi = z_old
            best_val = cur_max
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                z[i] = z_old + delta
                a_test = _softmax_fn(z)
                c_test = _full_convolve(a_test, a_test)
                val = float(c_test.max())
                if val < best_val:
                    best_val = val
                    best_zi = z[i]
            z[i] = best_zi
        
        a_new = _softmax_fn(z)
        sc_new = _score_norm_fn(a_new)
        if sc_new < best_score - 1e-15:
            best_score = sc_new
            best_a = a_new.copy()
            improved_any = True
        if not improved_any:
            break
    
    return best_a, best_score


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining L-BFGS continuation, projected gradient
    with top-K softmax, peak-pulling, symmetry exploitation, LP balancing,
    Adam fine-tuning, and random coordinate search.
    """

    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.12f} (n={len(p)}) [{label}] t={elapsed:.0f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.12f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")
    
    print(f"Starting: n={n}, best={best_score:.12f}")
    
    # === Phase 1: L-BFGS with fine beta continuation ===
    print("Phase 1: L-BFGS fine beta continuation")
    p1_end = min(deadline, t_start + 400.0)
    
    a_curr = best_seq.copy()
    a_b1, sc1 = _lbfgs_lse_phase(a_curr, p1_end, maxiter_per_beta=6000)
    update_best(a_b1, sc1, "lbfgs_fine")

    print("Phase 2: PG top-K")
    p2_end = min(deadline, t_start + 700.0)
    
    if time.monotonic() < p2_end - 30 and best_seq is not None:
        a_pg, sc_pg = _pg_topk_phase(best_seq, p2_end, K=5, max_iter=30000, lr_init=0.004)
        update_best(a_pg, sc_pg, "pg_topk5")
        
        if time.monotonic() < p2_end - 30:
            a_pg2, sc_pg2 = _pg_topk_phase(best_seq, p2_end, K=8, max_iter=20000, lr_init=0.003)
            update_best(a_pg2, sc_pg2, "pg_topk8")
    
    print(f"  After phase 2: best={best_score:.12f}")
    
    # === Phase 3: Peak pulling ===
    print("Phase 3: Peak pulling")

        a_pp, sc_pp = _peak_pull_phase(best_seq, p3_end, max_rounds=300)

    # === Phase 4: Symmetry exploitation ===
    print("Phase 4: Symmetry exploitation")
    p4_end = min(deadline, t_start + 1050.0)
    
    if time.monotonic() < p4_end - 30 and best_seq is not None:
        a_sym, sc_sym = _symmetry_exploit(best_seq, p4_end)
        update_best(a_sym, sc_sym, "symmetry")
    
    # === Phase 5: LP peak balancing ===
    print("Phase 5: LP balancing")
    p5_end = min(deadline, t_start + 1200.0)

        a_f, sc_f = _lbfgs_lse_phase(best_seq, p6_end, beta_list=tiny_betas, maxiter_per_beta=10000)
        update_best(a_f, sc_f, "lbfgs_tiny2")
    
    # === Phase 7: Adam ultra-fine ===
    print("Phase 7: Adam ultra-fine")
    p7_end = min(deadline, t_start + 1600.0)
    
    if time.monotonic() < p7_end - 30 and best_seq is not None:
        a_ad, sc_ad = _adam_fine(best_seq, p7_end, lr=0.003, max_iter=400000)
        update_best(a_ad, sc_ad, "adam_fine3")
        
        if time.monotonic() < p7_end - 30:
            a_ad2, sc_ad2 = _adam_fine(best_seq, p7_end, lr=0.001, max_iter=200000)
            update_best(a_ad2, sc_ad2, "adam_fine1")
    
    # === Phase 8: PG again (refined) ===
    print("Phase 8: PG refined")
    p8_end = min(deadline, t_start + 1720.0)
    
    if time.monotonic() < p8_end - 20 and best_seq is not None:
        a_pg3, sc_pg3 = _pg_topk_phase(best_seq, p8_end, K=3, max_iter=30000, lr_init=0.002)
        update_best(a_pg3, sc_pg3, "pg_topk3_refined")
    
    # === Phase 9: Peak pulling again ===
    print("Phase 9: Peak pulling final")
    p9_end = min(deadline, t_start + 1770.0)
    
    if time.monotonic() < p9_end - 10 and best_seq is not None:
        a_pp2, sc_pp2 = _peak_pull_phase(best_seq, p9_end, max_rounds=200)
        update_best(a_pp2, sc_pp2, "peak_pull_final")
    
    # === Phase 10: Random coordinate search ===
    print("Phase 10: Random coord search")
    
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_rc, sc_rc = _random_coord_search(best_seq, deadline - 10, max_iter=50000, n_try=3)
        update_best(a_rc, sc_rc, "random_coord")
    
    # === Final validation ===
    print(f"\nFinal best score: {best_score:.12f}")
    
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.12f}, n={len(final_seq)}")
        
        # Verify with evaluator
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.12f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.12f}")
    return [float(x) for x in fb]
