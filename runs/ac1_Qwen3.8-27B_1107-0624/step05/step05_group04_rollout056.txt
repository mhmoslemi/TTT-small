"""
Adaptive top-k LSE gradient descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses clean subgradient formula: d(c_k)/dz_j = 2*a_j*(a_{k-j} - c_k).
Multi-scale progression from n=512 to n=4096 with parametric initialization.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    """Score assuming sum(a) = 1. Returns 2*n*max_c."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Score for arbitrary non-negative a."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _topk_gradient(z, beta, top_k):
    """Compute gradient of top-k LSE smoothed objective w.r.t. z."""
    n = len(z)
    a = _softmax_fn(z)
    c = _fft_convolve(a, a)
    k = min(top_k, len(c))
    
    if k == 1:
        # Subgradient of max
        k_idx = np.argmax(c)
        c_k = c[k_idx]
        # h_j = a_{k_idx - j}
        h = np.zeros(n)
        for j in range(n):
            idx = k_idx - j
            if 0 <= idx <= n - 1:
                h[j] = a[idx]
        # Actually, h_j = a_{k_idx - j} for valid j
        # Let me use the FFT approach for consistency
        v = np.zeros(len(c))
        v[k_idx] = 1.0
        a_flip = a[::-1]
        conv = _fft_convolve(a_flip, v)
        h = conv[n-1:2*n-1]
        weighted_c = c_k
        grad_z = 2.0 * a * (h - weighted_c)
    else:
        top_idx = np.argpartition(c, -k)[-k:]
        top_c = c[top_idx]
        cm = top_c.max()
        shifted = (top_c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        
        v = np.zeros(len(c))
        v[top_idx] = w
        a_flip = a[::-1]
        conv = _fft_convolve(a_flip, v)
        h = conv[n-1:2*n-1]
        
        weighted_c = float(w @ top_c)
        grad_z = 2.0 * a * (h - weighted_c)
    
    return grad_z


def _topk_objective(z, beta, top_k):
    """Top-k LSE smoothed objective value."""
    n = len(z)
    a = _softmax_fn(z)
    c = _fft_convolve(a, a)
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_c = c[top_idx]
    cm = top_c.max()
    shifted = (top_c - cm) / beta
    return float(cm + beta * np.log(np.exp(shifted).sum()))


def _make_parametric(n, family='ramp', params=None):
    """Generate parametric initial shapes."""
    t = np.linspace(0, 1, n)
    if family == 'ramp':
        a = 1.0 + t
    elif family == 'ramp2':
        a = 1.0 + 2.0 * t
    elif family == 'ramp3':
        a = 1.0 + 3.0 * t
    elif family == 'exp':
        rate = params[0] if params else 2.0
        a = np.exp(rate * t)
    elif family == 'power':
        p = params[0] if params else 2.0
        a = (1.0 + t) ** p
    elif family == 'poly2':
        a = 1.0 + 2.0 * t - 0.5 * t * t
    elif family == 'poly3':
        a = 1.0 + 3.0 * t - 2.0 * t * t + 0.5 * t**3
    elif family == 'sqrt_ramp':

    elif family == 'quad_ramp':
        a = 1.0 + t * t
    else:
        a = 1.0 + t

    return a / a.sum()


def _upsample(a, target_n):
    """Upsample sequence a to length target_n using interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

"""
Adaptive top-k LSE gradient descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses direct subgradient formula: d(c_k)/dz_j = 2*a_j*(a_{k-j} - c_k).
Multi-scale progression with parametric initialization and progressive peak focusing.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return k


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _softmax_fn(z):
    z_arr = np.asarray(z, dtype=np.float64)
    z_arr = z_arr - np.max(z_arr)
    e = np.exp(z_arr)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z_arr) / len(z_arr)
    return e / s


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_topk_gradient(z, beta, top_k):
    n = len(z)
    a = _softmax_fn(z)
    c = _fft_convolve(a, a)
    k = min(top_k, len(c))

    if k <= 1:
        k_idx = np.argmax(c)
        c_k = c[k_idx]
        v = np.zeros(len(c))
        v[k_idx] = 1.0
        a_flip = a[::-1]
        conv = _fft_convolve(a_flip, v)
        h = conv[n - 1:2 * n - 1]
        weighted_c = c_k
    else:
        top_idx = np.argpartition(c, -k)[-k:]
        top_c = c[top_idx]
        cm = top_c.max()
        shifted = (top_c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        v = np.zeros(len(c))
        v[top_idx] = w
        a_flip = a[::-1]
        conv = _fft_convolve(a_flip, v)
        h = conv[n - 1:2 * n - 1]
        weighted_c = float(w @ top_c)

    grad_z = 2.0 * a * (h - weighted_c)
    return grad_z


def _topk_obj_value(z, beta, top_k):
    n = len(z)
    a = _softmax_fn(z)
    c = _fft_convolve(a, a)
    k = min(top_k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_c = c[top_idx]
    cm = top_c.max()
    shifted = (top_c - cm) / beta
    return float(cm + beta * np.log(np.exp(shifted).sum()))


def _make_parametric(n, family, params=None):
    t = np.linspace(0, 1, n)
    if family == 'ramp1':
        a = 1.0 + t
    elif family == 'ramp2':
        a = 1.0 + 2.0 * t
    elif family == 'ramp3':
        a = 1.0 + 3.0 * t
    elif family == 'ramp5':
        a = 1.0 + 5.0 * t
    elif family == 'exp1':
        a = np.exp(1.0 * t)
    elif family == 'exp2':
        a = np.exp(2.0 * t)
    elif family == 'exp3':
        a = np.exp(3.0 * t)
    elif family == 'exp5':
        a = np.exp(5.0 * t)
    elif family == 'pow1':
        a = (1.0 + t) ** 1.5
    elif family == 'pow2':
        a = (1.0 + t) ** 2.0
    elif family == 'pow3':

    elif family == 'poly2':
        a = 1.0 + 2.0 * t - 0.5 * t * t
    elif family == 'poly3':
        a = 1.0 + 3.0 * t - 2.0 * t * t + 0.5 * t**3

    elif family == 'quad_ramp':
        a = 1.0 + t * t
    else:
        a = 1.0 + t
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _adam_optimize(a_init, deadline, rng, lr=0.1, top_k_init=50, beta_init=None,

    best_a = a_init.copy()

    if beta_init is None or beta_init <= 0:

        beta_init = 0.05 * c.max()
    beta = beta_init

    m = np.zeros(n)
    v = np.zeros(n)

    while time.monotonic() < deadline:

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1

            if t % lr_decay_interval == 0:
                lr = max(0.0005, lr * lr_decay_factor)

            if t % beta_decay_interval == 0:
                beta = max(1e-8, beta * beta_decay_factor)

            if no_improve > 20 and t % noise_interval == 0:
                z = z + noise_scale * rng.normal(size=n)
                z = z - z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

    a_final = _softmax_fn(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

    fallback = _make_parametric(256, 'ramp1')
    sc_fb = _score_full(fallback)
    print(f"\nFALLBACK: n=256, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
