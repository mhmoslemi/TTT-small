Optimization of self-convolution peak for non-negative sequences.
Uses L-BFGS with log-sum-exp smoothing, subgradient polishing,
and multi-restart diversification to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_active(z, a, c, n, tol_frac=0.001):
    c_max = c.max()
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lbfgs_phase(a_init, beta, n, max_iter=2000, ftol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)

            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x), result.success
    except Exception:
        return a_init, False

def _beta_continuation(a_start, deadline, min_time=15.0):

    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())

        return a_start

                              0.001953125, 0.0009765625,
                              0.00048828125, 0.000244140625])
    
    current_a = a_start.copy()
    current_sc = _score(current_a)

        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin_taper":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _perturb(a, sigma, rng):

def _perturb_fourier(a, rng, n_modes=3, amp_scale=0.02):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))

    z = z - z.mean()
    return _softmax(z)

Actually, this is exactly the problem of "flat autocorrelation" or "low PSLT" (Peak-to-Sidelobe Ratio) sequences!

In radar/communications, the goal is to design a sequence whose autocorrelation has low sidelobes (small peaks away from the main peak). Here, since we're looking at the convolution (which includes the "main peak" at the center), we want the entire convolution to be as flat as possible.
