"""
Combined L-BFGS + exact coordinate descent optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses active-set tracking for efficient 1D coordinate optimization,
log-sum-exp smoothing for L-BFGS, and multi-restart diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize


def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def smoobj(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def smograd(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def gradient_active(z, a, c, n, tol_frac=0.01):
    """Gradient using active set (positions within tol_frac of max)."""
    c_max = float(c.max())
    if c_max < 1e-30:
        return np.zeros_like(a)
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n - 1: 2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def lbfgs_optimize(a_init, beta, n, max_iter=3000, ftol=1e-14):
    """Run L-BFGS-B for one beta phase."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return smoobj(z, beta, n)

    def jac(z):
        return smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': ftol,
                'gtol': 1e-10,
                'maxcor': 40,
                'maxls': 100,
                'disp': False,
            }
        )
        a_out = softmax(result.x)
        return a_out
    except Exception:
        return a_init


def try_lbfgs_schedule(a_start, deadline, min_time_remaining=100.0):
    """Run L-BFGS beta-continuation schedule."""
    n = len(a_start)
    c_cur = full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    # Adaptive beta schedule
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001])
    current_a = a_start.copy()
    current_sc = score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:
            break
        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 5:
            break

        # Scale iterations with available time
        est_time_per_iter = 0.003 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))

        a_new = lbfgs_optimize(current_a, beta, n, max_iter=max_iter)
        sc = score(a_new)

        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc
            print(f"    beta={beta:.6e}: score={sc:.10f} (iter {i})")

    return current_a

    """Find optimal value for a[j] using active-set 1D optimization.
    Returns (x_opt, new_conv_affected, new_S)."""

    S_prime = S - x_old  # sum of all other elements

    # For position k, the change in conv[k] when a[j] goes from x_old to x:
    #   If k-j in [0,n-1] and k != 2*j: delta = 2*(x - x_old)*a[k-j]
    #   If k == 2*j and 2*j in [0, 2n-2]: delta = x^2 - x_old^2
    #   Otherwise: delta = 0

    # Build candidate x values
    candidates = [0.0, x_old]

    for k in top_idx:
        if k < 0 or k >= 2 * n - 1:
            continue
        d = k - j
        if d < 0 or d >= n:
            continue  # not affected
        c_val = conv[k] - 2.0 * x_old * a[d]  # value at x=0 (for linear case)
        d_coeff = 2.0 * a[d]
        if k == 2 * j:
            # Quadratic case: conv[2j] = base + x^2, where base = conv[2j] - x_old^2
            c_quad = conv[2j] - x_old * x_old
            if c_quad > 0 and S_prime > 0:
                x_q = -S_prime + np.sqrt(S_prime * S_prime + c_quad)
                if x_q > 0:
                    candidates.append(x_q)
        else:
            if d_coeff > 1e-15:
                c_lin = conv[k] - 2.0 * x_old * a[d]
                x_k = S_prime - 2.0 * c_lin / d_coeff
                if x_k > 0:
                    candidates.append(x_k)

    # Also add some heuristic candidates
    candidates.append(x_old * 0.5)
    candidates.append(x_old * 1.5)
    candidates.append(x_old * 0.1)
    candidates.append(x_old * 5.0)

    # Filter valid candidates
    candidates = [x for x in candidates if x >= 0 and x < 10000]

    # Evaluate score at each candidate
    def eval_at_x(x):
        # Compute new conv values at top positions
        max_val = rest_max
        for k in top_idx:
            d = k - j
            if 0 <= d < n:
                if k == 2 * j:
                    new_val = (conv[k] - x_old * x_old) + x * x
                else:
                    new_val = (conv[k] - 2.0 * x_old * a[d]) + 2.0 * x * a[d]
                if new_val > max_val:
                    max_val = new_val
        # Also check: is rest_max still the max, or could an unaffected position be exceeded?
        # rest_max is the max of unaffected positions, which don't change.
        new_S = S_prime + x
        if new_S < 1e-15:
            return np.inf
        return max_val / (new_S * new_S)

    best_x = x_old
    best_val = eval_at_x(x_old)

    for x in candidates:
        v = eval_at_x(x)
        if v < best_val:
            best_val = v
            best_x = x

    # Refine with ternary search around best_x
    lo = max(0.0, best_x * 0.5) if best_x > 0 else 0.0
    hi = best_x * 2.0 + 0.1 if best_x > 0 else 1.0
    # Make sure best_x is in [lo, hi]
    lo = min(lo, best_x)
    hi = max(hi, best_x)

    for _ in range(40):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = eval_at_x(m1)
        f2 = eval_at_x(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1

    best_x = (lo + hi) / 2.0
    best_val = eval_at_x(best_x)

    return best_x


def coord_descent(a, deadline, max_sweeps=5000, K=50):
    """Coordinate descent with exact 1D optimization and active-set tracking."""
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    conv = full_convolve(a, a)
    best_sc = 2.0 * n * float(conv.max()) / (S * S)
    best_a = a.copy()

    def get_top_k(conv, K):
        idx = np.argpartition(-conv, K)[:K]
        top_vals = conv[idx]
        max_top = float(np.max(top_vals))
        rest = np.delete(conv, idx)
        rest_max = float(np.max(rest)) if len(rest) > 0 else 0.0
        return idx, top_vals, rest_max

    top_idx, top_vals, rest_max = get_top_k(conv, K)
    sweeps_without_improvement = 0

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        # Every 10 coordinates, refresh the top set
        for j in range(n):
            if (j % 10) == 9:
                top_idx, top_vals, rest_max = get_top_k(conv, K)

            x_new = coord_descent_1d(a, conv, S, j, top_idx, top_vals, rest_max, n)
            if abs(x_new - a[j]) < 1e-14:
                continue

            # Update convolution
            dx = x_new - a[j]
            a[j] = x_new
            S += dx

            # Update affected conv positions
            for k in range(max(0, j), min(2 * n - 1, j + n)):
                d = k - j
                if 0 <= d < n:
                    if k == 2 * j:
                        conv[k] += x_new * x_new - (x_new - dx) * (x_new - dx)
                    else:
                        conv[k] += 2.0 * dx * a[d]

        # End of sweep: check improvement
        cur_max = float(conv.max())
        cur_sc = 2.0 * n * cur_max / (S * S)
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            sweeps_without_improvement = 0
            if sweep % 50 == 0:
                print(f"    CD sweep {sweep}: score={best_sc:.10f}")
        else:
            sweeps_without_improvement += 1

        # Refresh top set every sweep
        top_idx, top_vals, rest_max = get_top_k(conv, K)

        # Early stopping if no improvement for many sweeps
        if sweeps_without_improvement > 200:
            break
        if sweeps_without_improvement > 50 and sweep % 20 == 0:
            # Check if we've truly converged
            if sweeps_without_improvement > 500:
                break

    return best_a, best_sc


def subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.05):
    """Subgradient polishing in z-space with exact active set."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = score(a_init)
    lr = lr_init

    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break

        a = softmax(z)
        c = full_convolve(a, a)
        grad = gradient_active(z, a, c, n, tol_frac=0.01)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = lr / (1.0 + i * 0.001)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = softmax(z)
            sc = score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            else:
                lr *= 0.9

    a_final = softmax(z)
    sc_final = score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return softmax(z)


def perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.01 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return softmax(z)


def make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
