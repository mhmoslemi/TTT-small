def _next_pow2(n):

def _full_convolve(a, b=None):

    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

"""
Best-response iteration with fast subgradient dual solver + L-BFGS log-sum-exp polish.
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Fast subgradient method on the LP dual (replacing expensive cvxpy LP solves)
2. Time-averaged iterates for better convergence
3. L-BFGS with log-sum-exp continuation for fine polishing
4. Multi-scale exploration (n=2048, 4096, 8192)
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_v(lam, f, N, F_frev):
    n = len(f)
    F_lam = np.fft.rfft(lam, N)
    corr = np.fft.irfft(F_lam * F_frev, N)
    return corr[n - 1:n - 1 + n]


def _best_response_subgrad(f, n_inner=5000, warm_start=None):
    n = len(f)
    K = 2 * n - 1
    N = _next_pow2(K + n - 1)
    F_frev = np.fft.rfft(f[::-1], N)

    if warm_start is not None:
        lam = warm_start.copy()
        s = lam.sum()
        if s < 1e-30:
            lam = np.ones(K) / K
        else:
            lam = lam / s
    else:
        lam = np.ones(K) / K

    lam_avg = np.zeros(K)
    v_final = None

    for it in range(n_inner):
        v = _compute_v(lam, f, N, F_frev)
        j_star = int(np.argmin(v))

        sg = np.zeros(K)
        lo = j_star
        hi = min(j_star + n, K)
        if lo < hi:
            sg[lo:hi] = f[0:hi - lo]

        step = 2.0 / (it + 2)
        lam = lam + step * sg
        lam = _project_simplex(lam)
        lam_avg += lam

    lam_avg /= n_inner
    v_final = _compute_v(lam_avg, f, N, F_frev)

    v_min = v_final.min()
    v_range = v_final.max() - v_min
    if v_range < 1e-12:
        g = np.ones(n) / n
    else:
        alpha = 3.0
        weights = np.exp(-alpha * (v_final - v_min) / v_range)
        g = weights / weights.sum()

    return g, float(v_min), lam_avg


def _best_response_lp(f, n_inner=10000):
    n = len(f)
    K = 2 * n - 1
    N = _next_pow2(K + n - 1)
    F_frev = np.fft.rfft(f[::-1], N)

    lam = np.ones(K) / K
    lam_avg = np.zeros(K)
    v_best = None

    for it in range(n_inner):
        v = _compute_v(lam, f, N, F_frev)
        j_star = int(np.argmin(v))

        sg = np.zeros(K)
        lo = j_star
        hi = min(j_star + n, K)
        if lo < hi:
            sg[lo:hi] = f[0:hi - lo]

        step = 2.0 / (it + 2)
        lam_new = lam + step * sg
        lam_new = _project_simplex(lam_new)
        lam = lam_new
        lam_avg += lam

    lam_avg /= n_inner
    v_best = _compute_v(lam_avg, f, N, F_frev)

    v_min = v_best.min()
    v_range = v_best.max() - v_min
    if v_range < 1e-12:
        g = np.ones(n) / n
    else:
        # Use a sharper activation
        alpha = 5.0
        weights = np.exp(-alpha * (v_best - v_min) / v_range)
        g = weights / weights.sum()

    return g, float(v_min), lam_avg


def _lbfgs_continuation(a_init, deadline, beta_fracs=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_fracs is None:
        beta_fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                      0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                      0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                      0.0002, 0.00015, 0.0001, 0.00008, 0.00005,
                      0.00003, 0.00002, 0.00001, 0.000005]

    n_betas = len(beta_fracs)
    for idx, bf in enumerate(beta_fracs):
        if time.monotonic() > deadline:
            break
        beta = bf * max_c0
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 40))

        def obj(zv):
            av = _softmax_fn(zv)
            cv = _full_convolve(av, av)
            cmv = float(cv.max())
            vals = (cv - cmv) / beta
            vals = np.clip(vals, -50, 50)
            lse = cmv + beta * float(np.log(np.sum(np.exp(vals))))
            return 2.0 * n * lse

        def grad(zv):
            av = _softmax_fn(zv)
            cv = _full_convolve(av, av)
            cmv = float(cv.max())
            vals = (cv - cmv) / beta
            vals = np.clip(vals, -50, 50)
            ev = np.exp(vals)
            wv = ev / ev.sum()
            Wv = float(np.dot(wv, cv))
            a_rev = av[::-1].copy()
            w_conv = _full_convolve(wv, a_rev)
            hv = w_conv[n - 1:2 * n - 1]
            return 4.0 * n * av * (hv - Wv)

        try:
            result = minimize(
                obj, z, method='L-BFGS-B', jac=grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _coord_descent_polish(a_init, deadline, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    deltas = [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            current_score = _score_norm_fn(a)
            best_delta = 0.0
            best_local = current_score
            for d in deltas:
                z[j] += d
                a_trial = _softmax_fn(z)
                sc = _score_norm_fn(a_trial)
                z[j] -= d
                if sc < best_local:
                    best_local = sc
                    best_delta = d
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u ** 2)
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = _make_profiles(n_init)[3]
        update_best(parent, _score_norm_fn(parent), "default")

    # Phase 1: Best response iteration at n=4096 (700s)
    phase1_end = min(deadline, t_start + 700.0)
    print(f"\nPhase 1: Best response iteration (target n=4096)")

    n_lp = 4096
    if len(parent) != n_lp:
        a_cur = _interp_profile(parent, n_lp)
    else:
        a_cur = parent.copy()
    sc_cur = _score_norm_fn(a_cur)
    update_best(a_cur, sc_cur, f"init_n{n_lp}")

    warm_lam = None
    outer_iters = 0
    for outer in range(500):
        if time.monotonic() > phase1_end:
            break
        remaining_iter = (phase1_end - time.monotonic())
        if remaining_iter < 5.0:
            break

        # Adaptive inner iterations
        if outer < 20:
            n_inner = 3000
        elif outer < 80:
            n_inner = 5000
        else:
            n_inner = 8000

        g, v_min, warm_lam = _best_response_subgrad(a_cur, n_inner=n_inner, warm_start=warm_lam)

        # Adaptive step size
        if outer < 10:
            t_step = 0.8
        elif outer < 50:
            t_step = 0.5
        elif outer < 150:
            t_step = 0.3
        else:
            t_step = max(0.1, 0.3 * 0.99 ** (outer - 150))

        a_new = (1 - t_step) * a_cur + t_step * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score_norm_fn(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={n_lp}) [br_iter{outer}] t={elapsed:.0f}s")

        a_cur = a_new
        outer_iters = outer + 1

    print(f"  Phase 1 done: {outer_iters} iterations, best={best_score:.10f}")

    # Phase 2: L-BFGS continuation (450s)
    phase2_end = min(deadline, t_start + 1150.0)
    print(f"\nPhase 2: L-BFGS continuation")

    a_start = best_seq.copy()
    p_best, sc_best = _lbfgs_continuation(a_start, phase2_end)
    update_best(p_best, sc_best, "lbfgs_main")

    # Try a perturbed restart
    if time.monotonic() < phase2_end - 60:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.03 * rng.normal(size=len(z))
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_pert = _softmax_fn(z)
        per_dl = min(phase2_end - 10, time.monotonic() + 120.0)
        p_best2, sc_best2 = _lbfgs_continuation(a_pert, per_dl,
                                                 beta_fracs=[0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005])
        update_best(p_best2, sc_best2, "lbfgs_perturb")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Try different n values (250s)
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"\nPhase 3: Multi-n exploration")

    for n_try in [2048, 8192]:
        if time.monotonic() > phase3_end - 60:
            break
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")

        if time.monotonic() < phase3_end - 30:
            per_dl = min(phase3_end - 10, time.monotonic() + 120.0)
            p_best_n, sc_best_n = _lbfgs_continuation(p_interp, per_dl,
                                                       beta_fracs=[0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005])
            update_best(p_best_n, sc_best_n, f"lbfgs_n{n_try}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Fine-tune best sequence (300s)
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"\nPhase 4: Fine-tuning best sequence (n={len(best_seq)})")

    # Very fine L-BFGS
    if time.monotonic() < phase4_end - 30:
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        max_c_fin = float(c_fin.max())
        fine_betas = [0.0005, 0.0003, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
        per_dl = min(phase4_end - 60, time.monotonic() + 150.0)
        p_best_f, sc_best_f = _lbfgs_continuation(best_seq, per_dl,
                                                   beta_fracs=np.array(fine_betas) / max_c_fin * max_c_fin)
        # Fix: pass as fractions relative to max_c
        p_best_f2, sc_best_f2 = _lbfgs_continuation(best_seq, per_dl,
                                                     beta_fracs=[x / max_c_fin for x in fine_betas])
        update_best(p_best_f2, sc_best_f2, "fine_lbfgs")

    # Coordinate descent polish
    if time.monotonic() < phase4_end - 10:
        per_dl = min(phase4_end - 5, time.monotonic() + 100.0)
        p_best_c, sc_best_c = _coord_descent_polish(best_seq, per_dl, max_rounds=15)
        update_best(p_best_c, sc_best_c, "coord_descent")

    # One more L-BFGS after coord descent
    if time.monotonic() < phase4_end - 5:
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        max_c_fin = float(c_fin.max())
        ultra_betas = [0.00003, 0.00001, 0.000005]
        per_dl = min(phase4_end - 2, time.monotonic() + 30.0)
        p_best_u, sc_best_u = _lbfgs_continuation(best_seq, per_dl,
                                                   beta_fracs=[x / max_c_fin for x in ultra_betas])
        update_best(p_best_u, sc_best_u, "ultra_fine")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Final micro-optimization (remaining time)
    print(f"\nPhase 5: Final micro-optimization")

    if time.monotonic() < deadline and best_seq is not None:
        n_fin = len(best_seq)
        # Try small random perturbations
        for attempt in range(10):
            if time.monotonic() > deadline - 5:
                break
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_scale = 0.01 * (1 - attempt / 10.0)
            z = z + noise_scale * rng.normal(size=n_fin)
            z = np.clip(z, -70, 70)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            sc_pert = _score_norm_fn(a_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_seq = a_pert.copy()
                print(f"  Perturb {attempt}: {best_score:.10f}")
            per_dl = min(deadline - 5, time.monotonic() + 10.0)
            p_best_m, sc_best_m = _lbfgs_continuation(a_pert, per_dl,
                                                       beta_fracs=[0.0001, 0.00005, 0.00002])
            update_best(p_best_m, sc_best_m, f"micro_{attempt}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 50000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed score: {sc_eval:.10f}")
                    return [float(x) for x in final_seq]
            except Exception as ex:
                print(f"evaluate_sequence error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
