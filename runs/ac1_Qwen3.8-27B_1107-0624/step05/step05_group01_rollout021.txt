"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated gradient descent with peak-focused log-sum-exp,
LP-based peak balancing, L-BFGS continuation, and multi-scale strategy (n=8192, 16384).
Key improvements: aggressive beta cooling, peak-focused gradients, Nesterov momentum,
randomized restarts, and larger sequence length.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_actual(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n, top_k=0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        peaks = c[idx]
    else:
        peaks = c
    pm = float(peaks.max())
    if pm < 1e-300:
        return 0.0
    sh = (peaks - pm) / beta
    mx = float(sh.max())
    lse = pm + beta * (mx + float(np.log(np.sum(np.exp(sh - mx)))))
    return float(2.0 * n * lse)


def _lse_grad(z, beta, n, top_k=0):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
        peaks = c[idx]
        w_full = np.zeros(len(c))
        pm = float(peaks.max())
        sh = (peaks - pm) / beta
        mx = float(sh.max())
        e = np.exp(sh - mx)
        w = e / e.sum()
        w_full[idx] = w
    else:
        pm = float(c.max())
        sh = (c - pm) / beta
        mx = float(sh.max())
        e = np.exp(sh - mx)
        w_full = e / e.sum()
    W = float(np.dot(w_full, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w_full, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_phase(a_init, deadline, max_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())

    fracs = [0.5, 0.3, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
             0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        # Decrease max iterations as beta gets smaller
        if beta > 0.01 * m0:
            maxiter = 5000
            maxfun = 20000
        elif beta > 0.001 * m0:
            maxiter = 8000
            maxfun = 30000
        else:
            maxiter = 12000
            maxfun = 50000

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n, 0),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxls': 100, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _nesterov_phase(a_init, deadline, lr=0.05, momentum=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    beta = 0.05 * m0

    v = np.zeros(n)
    no_improve = 0
    check_interval = 20
    top_k = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax(z)
        beta = max(1e-14 * m0, m0 * 0.05 * (0.96 ** (it // 30)))

        if it > 2000:
            top_k = 50
        elif it > 1000:
            top_k = 20
        elif it > 500:
            top_k = 10
        else:
            top_k = 0

        g = _lse_grad(z, beta, n, top_k)
        if not np.all(np.isfinite(g)):
            break

        v = momentum * v + g
        z = z - lr * v
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        # Adaptive learning rate

            lr = lr * 0.995

        if it % check_interval == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)

def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat


def _lp_phase(a_init, deadline, K=40, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.97 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = sorted_idx[:300]

        A_mat = _build_lp_matrix(best_a, active, n)
        b_vec = np.full(len(active), M * 1.002)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0 or res.x is None:
                break
            b_new = res.x
            sum_b = float(b_new.sum())
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = _score_norm(best_a)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_norm = a_trial / s_trial
            sc_trial = _score_norm(a_norm)
            if sc_trial < current_score - 1e-12:
                best_a = a_norm
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    # === Final: try one more L-BFGS on the overall best ===
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Final L-BFGS polish ===")
        af, scf = _lbfgs_phase(best_seq, deadline - 15)
        update_best(af, scf, "final_lbfgs")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256) / 256.0

        sc_final = _score_actual(final_seq)
        print(f"\nFINAL: actual score={sc_final:.10f}, n={len(final_seq)}")

        # Verify with evaluator if possible
        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using safe fallback")
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Absolute fallback
    fb = np.ones(256) / 256.0
    print(f"FALLBACK: n=256, score={_score_actual(fb):.10f}")
    return [float(x) for x in fb]
