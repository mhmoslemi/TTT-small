```python
"""
Hybrid Nesterov + LP + a-space simplex projection optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses smoothed-max gradients with Nesterov momentum, periodic LP kicks for
guaranteed progress, and direct a-space optimization with Euclidean projection
onto the probability simplex for fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, M, n, time_limit=25):
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _nesterov_phase(a_init, deadline, n_betas=8):
    """Nesterov momentum descent in z-space with decreasing beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    momentum = 0.9
    v = np.zeros(n)

    beta_fracs = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.004, 0.003, 0.002, 0.001, 0.0008, 0.0006, 0.0004, 0.0003, 0.0002]
    max_it_per_beta = 5000

    for beta_frac in beta_fracs:
        if time.monotonic() > deadline:
            break
        beta = beta_frac * max_c_init
        lr = 0.05 * beta_frac ** 0.5
        iters = 0
        while iters < max_it_per_beta:
            if time.monotonic() > deadline:
                break
            iters += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad_a = 4.0 * n * a * (h - W)

            # Softmax Jacobian: grad_z[i] = a[i] * (grad_a[i] - dot(a, grad_a))
            dot_ag = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - dot_ag)

            if not np.all(np.isfinite(grad_z)):
                break

            v = momentum * v + grad_z
            step = lr * v
            step_norm = np.linalg.norm(step)
            if step_norm > 1.0:
                step = step * (1.0 / step_norm)
            z = z - step
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if iters % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if iters % 500 == 0:
                lr *= 0.97

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _a_space_phase(a_init, deadline, lr_init=0.1):

    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    c_init = _full_convolve(a, a)
    best_score = float(c_init.max())
    max_c_init = best_score

    lr = lr_init
    momentum = 0.85
    v = np.zeros(n)

    beta_fracs = [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.004, 0.003, 0.002, 0.001, 0.0008, 0.0005, 0.0003, 0.0002]
    max_it_per_beta = 4000

    for beta_frac in beta_fracs:
        if time.monotonic() > deadline:
            break
        beta = beta_frac * max_c_init
        iters = 0
        while iters < max_it_per_beta:
            if time.monotonic() > deadline:
                break
            iters += 1

            c = _full_convolve(a, a)
            cm = float(c.max())

            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            grad_a = 2.0 * h

            # Project gradient onto tangent space of simplex
            grad_mean = float(np.mean(grad_a))
            grad_tangent = grad_a - grad_mean

            v = momentum * v + grad_tangent
            a_new = a - lr * v
            a_new = _project_simplex(a_new)

            sc = float(_full_convolve(a_new, a_new).max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

            if sc < cm * (1 + 1e-9):
                a = a_new
            else:
                lr *= 0.7
                v = v * 0.5

            if iters % 200 == 0:
                lr = min(lr_init * 0.5, lr * 1.1)

            if iters % 1000 == 0:
                lr *= 0.95

    return best_a, best_score

    beta = beta_frac * max_c_init

        for _ in range(lp_interval):

            total_steps += 1

            cm = float(c.max())

            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e

            a_rev = a[::-1]

            h = wc[n - 1:2 * n - 1]
            W_w = float(np.dot(w, c))
            grad_a = 4.0 * n * a * (h - W_w)
            dot_ag = float(np.dot(a, grad_a))
            grad_z = a * (grad_a - dot_ag)

            if not np.all(np.isfinite(grad_z)):

            v = momentum * v + grad_z
            step = lr * v
            step_norm = np.linalg.norm(step)
            if step_norm > 0.8:
                step = step * (0.8 / step_norm)

            # Apply in z-space
            z = np.log(np.maximum(a, 1e-15))

            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()
            f_new = _softmax(z)

            sc = float(_full_convolve(f, f).max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = f.copy()
                stuck = 0
            else:
                stuck += 1

            if total_steps % 100 == 0:
                lr *= 0.98

        # LP phase
        c = _full_convolve(f, f)
        M = float(c.max())

        if A_cache is None or np.max(np.abs(f - f_for_matrix)) > 1e-8:
            A_cache = _build_lp_matrix(f, n)
            f_for_matrix = f.copy()

        g = _lp_solve(f, M, n, time_limit=20)
        lp_calls += 1
        steps_since_lp = 0

        if g is not None:

            f_new = 0.5 * f + 0.5 * g
            f_new = f_new / f_new.sum()
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())

                sc = _score(f)

                    best_a = f.copy()
                max_c_init = max_c_init * 0.99
                beta = beta_frac * max_c_init

        else:

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_uniform(n):
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid optimization: Nesterov in z-space, LP kicks, a-space simplex
    projection, multi-restart, and larger-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}", flush=True)
        update_best(a0)
    else:
        n_main = 8192
        a0 = _make_uniform(n_main)
        print(f"No parent, using uniform n={n_main}", flush=True)
        update_best(a0)

    # === Phase 1: Nesterov descent in z-space (0 - 500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: Nesterov z-space (n={n_main}) ===", flush=True)

    a1, sc1 = _nesterov_phase(a0, phase1_end)
    update_best(a1)
    print(f"Phase 1 done: score={_score(a1):.10f}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 2: LP + Gradient hybrid (500s - 950s) ===
    phase2_end = min(deadline, t_start + 950.0)
    print(f"\n=== Phase 2: LP + Gradient hybrid ===", flush=True)

    if best_seq is not None:
        a2, sc2 = _lp_gradient_hybrid(best_seq / best_seq.sum(), phase2_end, max_lp_calls=8)
        update_best(a2)
    print(f"Phase 2 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 3: a-space simplex projection (950s - 1250s) ===
    phase3_end = min(deadline, t_start + 1250.0)
    print(f"\n=== Phase 3: a-space simplex projection ===", flush=True)

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a3, sc3 = _a_space_phase(best_seq / best_seq.sum(), phase3_end, lr_init=0.05)
        update_best(a3)
    print(f"Phase 3 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 4: Multi-restart + Nesterov (1250s - 1600s) ===
    phase4_end = min(deadline, t_start + 1600.0)
    print(f"\n=== Phase 4: Multi-restart ===", flush=True)

    if best_seq is not None and time.monotonic() < phase4_end - 60:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 4
        per_restart = (phase4_end - time.monotonic() - 20) / max(1, n_restarts)
        sigmas = [0.02, 0.04, 0.08, 0.15]

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            p_start = _perturb(a_best_norm, sigmas[i], rng)
            deadline_i = min(time.monotonic() + per_restart, phase4_end - time.monotonic() - 10)
            if deadline_i - time.monotonic() < 15:
                break
            a_r, sc_r = _nesterov_phase(p_start, deadline_i, n_betas=5)
            update_best(a_r)

            # Also try a-space from this restart
            if time.monotonic() < phase4_end - 30:
                a_r2, _ = _a_space_phase(a_r, deadline_i, lr_init=0.03)
                update_best(a_r2)

    print(f"Phase 4 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 5: Larger n exploration (1600s - 1720s) ===
    phase5_end = min(deadline, t_start + 1720.0)
    print(f"\n=== Phase 5: Larger n exploration ===", flush=True)

    if best_seq is not None and len(best_seq) < 16384 and time.monotonic() < phase5_end - 60:
        for target_n in [n_main * 2]:
            if time.monotonic() > phase5_end - 80:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)
            remaining = phase5_end - time.monotonic() - 30
            if remaining > 60:
                a_r, _ = _a_space_phase(p_interp, time.monotonic() + remaining, lr_init=0.03)
                update_best(a_r)

    # Also try slightly smaller n
    if best_seq is not None and len(best_seq) > 2048 and time.monotonic() < phase5_end - 30:
        target_n = len(best_seq) // 2
        p_interp = _interp_profile(best_seq, target_n)
        update_best(p_interp)
        remaining = phase5_end - time.monotonic() - 15
        if remaining > 30:
            a_r, _ = _a_space_phase(p_interp, time.monotonic() + remaining, lr_init=0.03)
            update_best(a_r)

    print(f"Phase 5 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 6: Final fine polish (1720s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===", flush=True)
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_final, _ = _a_space_phase(a_best_norm, time.monotonic() + remaining, lr_init=0.01)
            update_best(a_final)

    print(f"Phase 6 done: best={best_score:.10f}, time={time.monotonic()-t_start:.1f}s", flush=True)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"Eval check: {ex}", flush=True)
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
```
