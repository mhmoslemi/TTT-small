"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set projected gradient, spectral methods, and multi-scale refinement.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_normalized(a):
    """Score assuming sum(a)=1: 2n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _full_score(a):
    """Full score: 2n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Binary search on tau
    lo = v.min() - target_sum / n
    hi = v.max()
    for _ in range(64):
        tau = (lo + hi) / 2
        proj = np.maximum(v - tau, 0)
        s = proj.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2
    proj = np.maximum(v - tau, 0)
    # Adjust for numerical precision
    diff = target_sum - proj.sum()
    if diff != 0:
        proj += diff / n
        proj = np.maximum(proj, 0)
    return proj


def _subgradient(c, a, n, top_k=10, temperature=0.01):
    """Compute smoothed subgradient of max(c) with respect to a."""
    # Weight the top lags by their proximity to max
    c_max = c.max()
    # Softmax-like weighting
    shifted = (c - c_max) / max(temperature, 1e-15)
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient: d(c[k])/d(a[i]) = a[k-i] if 0 <= k-i < n
    # For each lag k with weight w[k], add w[k] * a[k-i] to grad[i]
    grad = np.zeros(n)
    # Only consider lags with significant weight
    thresh = w.max() * 1e-3
    active = np.where(w > thresh)[0]
    
    for k in active:
        wk = w[k]
        # d c[k] / d a[i] = a[k-i] if 0 <= k-i < n
        # So for i in [max(0,k-n+1), min(n-1,k)]:
        i_lo = max(0, int(k) - n + 1)
        i_hi = min(n - 1, int(k))
        if i_lo <= i_hi:
            # a[k-i] for i in [i_lo, i_hi] is a[k-i_hi] ... a[k-i_lo]
            indices = np.arange(i_lo, i_hi + 1)
            a_indices = k - indices  # these are the indices into a
            grad[i_lo:i_hi+1] += wk * a[a_indices]
    
    return grad * 2 * n


def _projected_gradient_step(a, lr, c, top_k=20, temperature=None):
    """One step of projected gradient descent on simplex."""
    n = len(a)
    c_max = c.max()
    if temperature is None:
        # Adaptive temperature
        sorted_c = np.sort(c)[::-1]
        # Temperature based on gap between top values
        gaps = np.diff(sorted_c[:min(top_k, len(sorted_c))])
        temperature = max(1e-6, float(np.median(gaps[gaps > 0])) if np.any(gaps > 0) else c_max * 0.01)
    
    grad = _subgradient(c, a, n, top_k=top_k, temperature=temperature)
    a_new = a - lr * grad
    return _project_simplex(a_new, 1.0)

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    """Add multiplicative log-normal noise."""
    noise = np.exp(sigma * rng.standard_normal(len(a)))
    a_new = a * noise
    return a_new / a_new.sum()


def _structured_init(n, rng):
    """Create a structured initial guess with low peak autocorrelation."""
    # Start with a "banded" structure: alternate high and low
    a = np.ones(n)
    # Add some structure
    for _ in range(5):
        center = rng.integers(n // 4, 3 * n // 4)
        width = rng.integers(n // 8, n // 3)
        sigma = width / 2.0
        x = np.arange(n)
        gaussian = np.exp(-(x - center) ** 2 / (2 * sigma ** 2))
        a = a + 0.5 * gaussian
    a = np.maximum(a, 0.01)
    return a / a.sum()

"""
FFT-accelerated active-set optimization with annealing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key innovations: FFT-based gradient, exact simplex projection,
temperature annealing, Nesterov momentum, and multi-scale search.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _full_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v, target=1.0):
    """Euclidean projection onto {x>=0, sum(x)=target}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    proj = np.maximum(v - theta, 0)
    # Fix numerical issues
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        pos = proj > 0
        if pos.any():
            proj[pos] += diff / pos.sum()
            proj = np.maximum(proj, 0)
    return proj


def _compute_grad(a, w, n):
    """Compute 4n * cross_corr(w, a) using FFT. w has length 2n-1, a has length n."""
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(W * np.conj(A), N)[:n]
    return 4.0 * n * g


def _weight_lags(c, temperature):
    """Compute softmax weights over convolution lags."""
    c_max = c.max()
    shifted = (c - c_max) / max(temperature, 1e-15)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    return w


def _anneal_optimize(a_init, deadline, lr=0.03, temp_start=0.05, temp_end=0.001,
                     momentum=0.9, max_iter=100000):
    """Nesterov-accelerated projected gradient with temperature annealing."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_simplex(a, 1.0)
    
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = _score_norm(a)
    
    t_step = 0
    total_iters_est = 50000  # rough estimate for annealing schedule
    
    for t_step in range(max_iter):
        if time.monotonic() > deadline:
            break
        if t_step % 2 == 1:  # Skip half the iterations for speed
            continue
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Anneal temperature
        frac = min(1.0, t_step / total_iters_est)
        temp = temp_start * (temp_end / temp_start) ** frac
        temp = max(temp, c_max * 1e-4)
        
        w = _weight_lags(c, temp)
        grad = _compute_grad(a, w, n)
        
        # Nesterov step
        v = momentum * v + (1 - momentum) * grad
        a_new = _project_simplex(a - lr * v, 1.0)
        
        # Adaptive learning rate
        c_new = _full_convolve(a_new, a_new)
        score_new = 2.0 * n * float(c_new.max())
        
        if score_new > score * (1 + 0.1):
            lr *= 0.95
        elif t_step % 1000 == 0:
            lr *= 0.995
        
        a = a_new
    
    # Final check
    final_score = _score_norm(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def _adam_refine(a_init, deadline, lr=0.015):
    """Adam optimizer on simplex for fine-tuning."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_simplex(a, 1.0)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = _score_norm(a)
    current_lr = lr
    temp_scale = 0.02
    
    while time.monotonic() < deadline:
        t += 1
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        temp = temp_scale * c_max
        w = _weight_lags(c, temp)
        grad = _compute_grad(a, w, n)
        
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        unorm = np.linalg.norm(update)
        if unorm > 0.5:
            update *= 0.5 / unorm
        
        a = _project_simplex(a - update, 1.0)
        
        if t % 2000 == 0:
            current_lr *= 0.93
            temp_scale *= 0.95
    
    final_score = _score_norm(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    noise = np.exp(sigma * rng.standard_normal(len(a)))
    a_new = a * noise
    return a_new / a_new.sum()

Instead of minimizing max(c), minimize (sum(c^p))^{1/p} for large p (which approaches max as p→∞). For finite p, this is a smooth approximation that might be easier to optimize.

The gradient of (sum(c^p))^{1/p} with respect to a[i] is:
= (1/p) * (sum(c^p))^{1/p - 1} * p * sum_k (c[k])^{p-1} * ∂c[k]/∂a[i]
= (sum(c^p))^{1/p - 1} * sum_k c[k]^{p-1} * 2*a[k-i]

For large p, this is dominated by the c[k] values near the max, similar to the softmax weighting but with a sharper transition.

Let me use this as an alternative to the softmax weighting. It might help escape local minima because it's smoother.

"""
FFT-accelerated active-set optimization with temperature annealing, power-mean
smooth surrogates, and multi-scale search for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: FFT-based subgradient, exact simplex projection, adaptive
annealing, power-mean smoothing for escaping local minima, and multi-n exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _full_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v, target=1.0):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - target)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, target / n)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target) / (rho + 1)
    proj = np.maximum(v - theta, 0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        pos = proj > 0
        if pos.any():
            proj[pos] += diff / pos.sum()
            proj = np.maximum(proj, 0)
    return proj


def _compute_grad_fft(a, w, n):
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    g = np.fft.irfft(W * np.conj(A), N)[:n]
    return 4.0 * n * g


def _softmax_weights(c, temperature):
    c_max = c.max()
    shifted = (c - c_max) / max(temperature, 1e-15)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    return w


def _power_mean_weights(c, power):
    c_pos = np.maximum(c, 1e-30)
    w = c_pos ** power
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    return w


def _anneal_optimize(a_init, deadline, lr=0.03, temp_start=0.04, temp_end=0.002,
                     momentum=0.85, use_power_mean=False, power=20):
    n = len(a_init)
    a = _project_simplex(a_init.copy(), 1.0)
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    current_lr = lr
    t = 0

    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score:
            best_score = score
            best_a = a.copy()

        frac = min(1.0, t / 8000.0)
        if use_power_mean:
            w = _power_mean_weights(c, power)
        else:
            temp = temp_start * (temp_end / temp_start) ** frac
            temp = max(temp, c_max * 1e-4)
            w = _softmax_weights(c, temp)

        grad = _compute_grad_fft(a, w, n)
        v = momentum * v + (1.0 - momentum) * grad
        a_new = _project_simplex(a - current_lr * v, 1.0)

        if t % 100 == 0:
            c_new = _full_convolve(a_new, a_new)
            score_new = 2.0 * n * float(c_new.max())
            if score_new > score * 1.05:
                current_lr *= 0.9
                v *= 0.5
            elif t % 2000 == 0:
                current_lr *= 0.98
        a = a_new

    final_score = _score_norm(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def _adam_refine(a_init, deadline, lr=0.012, use_power_mean=False, power=15):
    n = len(a_init)
    a = _project_simplex(a_init.copy(), 1.0)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_score = _score_norm(a)
    current_lr = lr
    temp_scale = 0.015

    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max

        if score < best_score:
            best_score = score
            best_a = a.copy()

        if use_power_mean:
            w = _power_mean_weights(c, power)
        else:
            temp = temp_scale * c_max
            w = _softmax_weights(c, temp)

        grad = _compute_grad_fft(a, w, n)
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        unorm = np.linalg.norm(update)
        if unorm > 0.3:
            update *= 0.3 / unorm
        a = _project_simplex(a - update, 1.0)

        if t % 3000 == 0:
            current_lr *= 0.92
            temp_scale *= 0.93

    final_score = _score_norm(a)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    noise = np.exp(sigma * rng.standard_normal(len(a)))
    a_new = a * noise
    return a_new / a_new.sum()
