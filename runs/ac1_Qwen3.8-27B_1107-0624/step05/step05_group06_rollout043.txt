    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    eta = 0.1  # learning rate

        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Active set: constraints within 0.1% of max
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        w = active / active.sum()

        # grad[i] = sum_k w[k] * 2*a[k-i] (valid indices)
        # This is 2 * (w * a_reversed) shifted
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        grad = wc[n-1:2*n-1] * 2

        # But we need to handle the constraint sum(a) = 1
        # Use multiplicative update: a_new[i] = a[i] * exp(-eta * grad[i])
        # Then normalize
        
        z = -eta * grad
        z = z - z.mean()  # recentering for numerical stability
        a = a * np.exp(z)
        a = a / a.sum()
        
        # Decay learning rate
        eta *= 0.999
        
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

Optimized L-BFGS + subgradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements: adaptive beta, larger n, multi-start, line search.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _subgrad_active(a, tol_rel=0.001):
    """Compute exact subgradient using active set."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    active = c >= (c_max - tol_rel * c_max)
    if active.sum() == 0:
        active = np.zeros(len(c), dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros(len(c))
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W), c_max

def _lbfgs_round(a_init, beta, n, max_iter=2000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:

                         options={'maxiter': max_iter, 'ftol': 1e-15,
                                  'gtol': 1e-12, 'maxcor': 40, 'maxls': 50,
                                  'disp': False})
        return _softmax(result.x), result.nit
    except:
        return a_init, 0

"""
Improved L-BFGS with log-sum-exp smoothing, adaptive beta-continuation,
and subgradient polishing with backtracking line search for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements: adaptive active set, proper line search, multi-start,
Fourier parameterization for shape search, and KKT-based convergence.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _exact_subgrad_z(a, n, tol_rel=None):
    """Compute exact subgradient in z-space using active set.
    Returns (grad_z, c, c_max, active_count, h_spread)."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    if tol_rel is None:
        tol_rel = 0.0005
    tol = tol_rel * c_max
    active = c >= (c_max - tol)
    n_active = int(active.sum())
    if n_active == 0:
        active = np.zeros(len(c), dtype=bool)
        active[int(np.argmax(c))] = True
        n_active = 1
    w = np.zeros(len(c))
    w[active] = 1.0 / n_active
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    h_spread = float(h.max() - h.min())
    grad_z = 4.0 * n * a * (h - W)
    return grad_z, c, c_max, n_active, h_spread


def _lbfgs_round(a_init, beta, n, max_iter=2000, deadline=None):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        if deadline is not None and time.monotonic() > deadline:
            raise TimeoutError()
        return _smoobj(z, beta, n)

    def jac(z):
        if deadline is not None and time.monotonic() > deadline:
            raise TimeoutError()
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxcor': 50, 'maxls': 30,
                                   'disp': False})
        a_out = _softmax(result.x)
        return a_out, int(result.nit)
    except (TimeoutError, Exception) as e:
        return a_init, 0


def _subgrad_polish(a_init, deadline, n_steps=5000, lr_init=0.1):
    """Subgradient method with backtracking line search in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    consecutive_fails = 0

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        grad_z, c, c_max, n_active, h_spread = _exact_subgrad_z(a, n, tol_rel=0.0005)

        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-14:
            break

        # Check KKT-like condition: h should be constant on support
        a_supp = a > 1e-8
        if a_supp.sum() > 10:
            h_on_supp = None  # h_spread is a quick proxy
            if h_spread < 1e-8 * c_max:
                break

        # Backtracking line search with normalized direction
        direction = -grad_z / grad_norm
        step = lr
        improved = False
        best_trial_sc = best_sc
        best_trial_z = z.copy()

        for backtrack in range(15):
            z_trial = z + step * direction
            z_trial = np.clip(z_trial, -50.0, 50.0)
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax(z_trial)
            sc_trial = _score(a_trial)

            if sc_trial < best_sc - 1e-12:
                best_trial_sc = sc_trial
                best_trial_z = z_trial
                improved = True
                break
            step *= 0.5
            if step < 1e-12:
                break

        if improved:
            z = best_trial_z.copy()
            best_a = _softmax(z).copy()
            best_sc = best_trial_sc
            consecutive_fails = 0
            # Slightly increase lr if we made good progress
            if i % 100 == 0 and i > 0:
                lr = min(lr * 1.05, 0.5)
        else:
            consecutive_fails += 1
            # Take a very small step anyway to escape stagnation
            tiny_step = lr * 0.01 / (1 + consecutive_fails)
            z = z + tiny_step * direction
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            if consecutive_fails > 50:
                break

        # Anneal learning rate
        lr *= 0.9995
        if consecutive_fails > 10:
            lr *= 0.95

    # Final evaluation
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_tapered(n, power=0.7):
    t = np.linspace(0, 1, n)
    a = np.sin(np.pi * t) ** power
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_taper(n, depth=0.2):
    t = np.linspace(0, 1, n)
    a = 1.0 - depth * (2.0 * t - 1.0) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_taper(n):
    t = np.linspace(0, 1, n)
    a = 1.0 - (2.0 * t - 1.0) ** 4
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_decrease(n):
    t = np.linspace(0, 1, n)
    a = 1.0 - 0.5 * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _fourier_log_shape(n, coeffs):
    """Create a sequence from Fourier coefficients in log-space."""
    t = np.arange(n)
    z = np.zeros(n)
    for k, (real, imag) in enumerate(coeffs):
        if real != 0:
            z += real * np.cos(2.0 * np.pi * k * t / n)
        if imag != 0:
            z += imag * np.sin(2.0 * np.pi * k * t / n)
    a = _softmax(z)
    return a


def _fourier_optimize(n, n_modes, deadline, a_init=None, rng=None):
    """Optimize Fourier coefficients to find good shape."""
    if rng is None:
        rng = np.random.default_rng(42)

    # Initialize coefficients
    params = np.zeros(2 * n_modes)
    if a_init is not None:
        # Get initial log-space
        z_init = np.log(np.maximum(a_init, 1e-15))
        z_init = z_init - z_init.mean()
        # Fit Fourier coefficients via least squares
        t = np.arange(n)
        A = np.zeros((n, 2 * n_modes))
        for k in range(n_modes):
            A[:, 2 * k] = np.cos(2.0 * np.pi * k * t / n)
            A[:, 2 * k + 1] = np.sin(2.0 * np.pi * k * t / n)
        try:
            params = np.linalg.lstsq(A, z_init, rcond=None)[0]
        except:
            params = np.zeros(2 * n_modes)

    def _fourier_score(params):
        a = _fourier_log_shape(n, list(zip(params[0::2], params[1::2])))
        return _score(a)

    def _fourier_grad(params):
        a = _fourier_log_shape(n, list(zip(params[0::2], params[1::2])))
        n_local = len(a)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros(len(c), dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros(len(c))
        w[active] = 1.0 / active.sum()
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n_local - 1:2 * n_local - 1]
        W = float(np.dot(w, c))
        grad_a = 4.0 * n_local * a * (h - W)
        # Map from a-space to param-space via chain rule
        t = np.arange(n)
        g = np.zeros(2 * n_modes)
        for k in range(n_modes):
            g[2 * k] = np.sum(grad_a * a * np.cos(2.0 * np.pi * k * t / n))
            g[2 * k + 1] = np.sum(grad_a * a * np.sin(2.0 * np.pi * k * t / n))
        return g

    try:
        result = minimize(_fourier_score, params, jac=_fourier_grad,
                          method='L-BFGS-B',
                          options={'maxiter': 300, 'ftol': 1e-14,
                                   'gtol': 1e-10, 'maxcor': 30, 'disp': False})
        a_out = _fourier_log_shape(n, list(zip(result.x[0::2], result.x[1::2])))
        return a_out
    except:
        return a_init


def _try_optimization(a_init, deadline, n, label=""):
    """Run full L-BFGS beta-continuation + subgradient polish."""
    a_cur = a_init.copy()
    sc_cur = _score(a_cur)
    c = _full_convolve(a_cur, a_cur)
    max_c = float(c.max())

    # Adaptive beta schedule - more values, geometric
    n_betas = 25
    betas = max_c * np.exp(np.linspace(np.log(0.5), np.log(0.0001), n_betas))

    best_in_phase = a_cur.copy()
    best_sc_in_phase = sc_cur

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        # Estimate remaining betas and allocate time
        remaining_betas = n_betas - i
        time_per_beta = max(2.0, (deadline - time.monotonic()) / remaining_betas)
        # Estimate iterations: ~10ms per iter for n=2048
        est_iter_time = 0.012 * max(1.0, n / 2048.0)
        max_iter = max(30, min(5000, int(time_per_beta / (est_iter_time * 12))))

        a_new, niter = _lbfgs_round(a_cur, beta, n, max_iter, deadline)
        sc_new = _score(a_new)

        if sc_new < sc_cur - 1e-13:
            a_cur = a_new
            sc_cur = sc_new
            if sc_new < best_sc_in_phase:
                best_sc_in_phase = sc_new
                best_in_phase = a_cur.copy()
        # If no improvement, skip to next beta

    # Subgradient polishing
    if time.monotonic() < deadline - 5:
        polish_deadline = min(deadline - 2, time.monotonic() + max(10, (deadline - time.monotonic()) * 0.4))
        a_pg, sc_pg = _subgrad_polish(a_cur, polish_deadline, n_steps=5000, lr_init=0.08)
        if sc_pg < best_sc_in_phase:
            best_in_phase = a_pg
            best_sc_in_phase = sc_pg

    return best_in_phase, best_sc_in_phase
