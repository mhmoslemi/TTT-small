import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max_conv."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Full score: 2*n*max_conv/sum^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.05 * cm:
        return max(1e-10, 0.05 * cm)
    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:
        return max(1e-10, 0.3 * gap)


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta_start = 0.3 * max_c0
    beta_end = 1e-6 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 180.0)
        level_deadline = time.monotonic() + time_budget

        if beta > 0.1 * max_c0:
            maxiter = 500
        elif beta > 0.01 * max_c0:
            maxiter = 2000
        else:
            maxiter = 5000
        maxfun = maxiter * 5

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': maxfun}

            if result.x is not None and np.all(np.isfinite(result.x)):

                a_new = _softmax_fn(z_new)

        for k in active:

            hi = min(n - 1, k)

def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n + 1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
