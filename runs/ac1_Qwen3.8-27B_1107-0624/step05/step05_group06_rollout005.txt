    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    e = np.exp(z)
    s = e.sum()

    return e / s

    # So (w ⊛ a)_j = sum_k w_k * a_{k-j}

    # And (a ⊛ u1)_k = sum_l a_l * u1_{l-k} = sum_l a_{l} * u1_{l-k}
    # Let's define: corr(x, y)_k = sum_l x_l * y_{l-k} = (reverse(x) conv y)_{...}
    
    # Let me use: (x ⊛ y)_k = sum_l x_l * y_{l+k} (this is one definition of cross-correlation)
    # (x ⊛ y)_k = (reverse(x) * y)_{k + len(x) - 1} ... this is getting confusing
    
    # Let me just use the formula from the gradient:
    # h_j = sum_k w_k * a_{k-j} = full_convolve(w, a_rev)[n-1+j]
    # So (w ⊛ a)_j where (w ⊛ a)_j = sum_k w_k * a_{k-j}
    
    # For s_k = sum_l a_k * u1_l ... wait, what do I actually need?
    # I need: M_{jl} = sum_k w_k * a_{k-j} * a_{k-l}
    # And (H_a @ v)_j = (8n/beta) * (sum_l M_{jl} * v_l - h_j * sum_l h_l * v_l) + 4n * sum_l w_{j+l} * v_l
    
    # sum_l M_{jl} * v_l = sum_l sum_k w_k * a_{k-j} * a_{k-l} * v_l
    # = sum_k w_k * a_{k-j} * (sum_l a_{k-l} * v_l)
    # = sum_k w_k * a_{k-j} * s_k
    # where s_k = sum_l a_{k-l} * v_l = (a ⊛ v)_k ... let me define this properly
    # s_k = sum_l a_{k-l} * v_l. Let m = k-l, l = k-m: s_k = sum_m a_m * v_{k-m}
    # This is the convolution (a * v)_k!
    
    # So s = full_convolve(a, v) (standard convolution)
    # Then sum_l M_{jl} * v_l = sum_k w_k * a_{k-j} * s_k
    # = sum_k (w * s)_k * a_{k-j}
    # Let u_k = w_k * s_k
    # = sum_k u_k * a_{k-j}
    # This is the cross-correlation (u ⊛ a)_j = sum_k u_k * a_{k-j}
    # = full_convolve(u, a_rev)[n-1+j]
    
    # And sum_l h_l * v_l = dot(h, v)
    
    # And sum_l w_{j+l} * v_l: let m = j+l, l = m-j: sum_m w_m * v_{m-j}
    # This is (w ⊛ v)_j = sum_m w_m * v_{m-j}
    # = full_convolve(w, v_rev)[n-1+j] ... wait
    # full_convolve(w, v_rev)[i] = sum_k w_k * v_rev[i-k] = sum_k w_k * v[len(v)-1-(i-k)]
    # If v has length n: = sum_k w_k * v[n-1-i+k]
    # We want sum_m w_m * v_{m-j}. Set m = k, so sum_k w_k * v_{k-j}.
    # Set n-1-i+k = k-j → n-1-i = -j → i = n-1+j
    # So (w ⊛ v)_j = full_convolve(w, v_rev)[n-1+j]
    
    # OK so the full HVP with H_a is:
    # (H_a @ v)_j = (8n/beta) * ((u ⊛ a)_j - h_j * (h·v)) + 4n * (w ⊛ v)_j
    # where u = w * conv(a, v) (elementwise product)
    
    # Now the full H_z @ v = J_a^T @ H_a @ (J_a @ v)
    # J_a @ v = a * (v - (a·v))  [done as u1 above]
    # H_a @ u1 = (8n/beta) * ((u ⊛ a)_j - h_j * (h·u1)) + 4n * (w ⊛ u1)_j
    #   where u = w * conv(a, u1)
    # J_a^T @ result = result - a * (a·result)
    
    a_rev = a[::-1]
    
    # Compute conv(a, u1) -- this is standard convolution of a and u1
    s = _full_convolve(a, u1)  # s_k = sum_l a_l * u1_{k-l}
    # s has length 2n-1
    
    # u_k = w_k * s_k
    u = w * s
    
    # (u ⊛ a)_j = full_convolve(u, a_rev)[n-1+j] for j=0,...,n-1
    uc = _full_convolve(u, a_rev)
    corr_ua = uc[n-1:2*n-1]  # length n
    
    # h · u1
    hu1 = float(np.dot(h, u1))
    
    # (w ⊛ u1)_j = full_convolve(w, u1_rev)[n-1+j] for j=0,...,n-1
    u1_rev = u1[::-1]
    wc = _full_convolve(w, u1_rev)
    corr_wu1 = wc[n-1:2*n-1]  # length n
    
    # (H_a @ u1)_j = (8n/beta) * (corr_ua_j - h_j * hu1) + 4n * corr_wu1_j
    Ha_u1 = (8.0 * n / beta) * (corr_ua - h * hu1) + 4.0 * n * corr_wu1
    
    # J_a^T @ Ha_u1 = Ha_u1 - a * (a · Ha_u1)
    aHa = float(np.dot(a, Ha_u1))
    result = Ha_u1 - a * aHa
    
    return result

def _cg_solve_hessian(grad, a, w, h, n, beta, max_iter=100, tol=1e-12):
    """Solve H_z @ x = grad using Conjugate Gradient."""
    r = grad.copy()  # r = grad - H @ x, initially x=0 so r=grad
    p = r.copy()
    rs = float(np.dot(r, r))
    x = np.zeros_like(grad)
    
    for i in range(max_iter):
        Hp = _hvp_smoothed(p, a, w, h, n, beta)
        pHp = float(np.dot(p, Hp))
        if abs(pHp) < 1e-30:
            break
        alpha = rs / pHp
        x += alpha * p
        r -= alpha * Hp
        rs_new = float(np.dot(r, r))
        if np.sqrt(rs_new) < tol * (np.linalg.norm(grad) + 1e-30):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    
    return x
