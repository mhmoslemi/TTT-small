So `a[m] = c + C * (-lambda_1/lambda_2)^m`.

For the solution to be non-negative and bounded, we need `|lambda_1/lambda_2| <= 1`, i.e., `lambda_1 <= lambda_2` (assuming both positive). And `c + C * r^m >= 0` for all `m`, where `r = -lambda_1/lambda_2`.

If `|r| < 1`, the solution is bounded and converges to `c`. For non-negativity, we need `c >= |C|` (roughly).

If `r = -1` (i.e., `lambda_1 = lambda_2 = 0.5`), the solution is `a[m] = c + C * (-1)^m`, which alternates between `c + C` and `c - C`. For non-negativity, we need `C <= c`.

This is very interesting! For `K = 2` with adjacent peaks, the optimal sequence is a geometric sequence (or a constant plus geometric). This gives us an explicit formula for the optimal sequence!

Let me verify: if `a[m] = c + C * r^m` where `r = -lambda_1/lambda_2`, then:
`lambda_1 * a[m] + lambda_2 * a[m+1] = lambda_1 * (c + C*r^m) + lambda_2 * (c + C*r^{m+1})`
`= c*(lambda_1 + lambda_2) + C*r^m*(lambda_1 + lambda_2*r)`
`= c + C*r^m*(lambda_1 + lambda_2*r)`

For this to equal `c` for all `m`, we need `lambda_1 + lambda_2 * r = 0`, i.e., `r = -lambda_1/lambda_2`. ✓

`a[m] = c + C * (-lambda_1/lambda_2)^m`

"""
Optimized step function search using:
1. L-BFGS with log-sum-exp continuation
2. KKT-based refinement exploiting sparse Hankel structure
3. Fine-tuning with ultra-small beta
4. Length interpolation for additional gains
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csc_matrix
from scipy.sparse.linalg import splu
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _build_hankel_sparse(peak_indices, lambda_vec, n):
    """Build sparse Hankel matrix H where H[j,m] = sum_i lambda_i * 1[j+m = k_i]."""
    H = lil_matrix((n, n))
    for i, k_i in enumerate(peak_indices):
        lam = lambda_vec[i]
        if lam == 0:
            continue
        j_min = max(0, k_i - n + 1)
        j_max = min(n - 1, k_i)
        for j in range(j_min, j_max + 1):
            m = k_i - j
            H[j, m] += lam
    return H


def _solve_kkt(peak_indices, n, max_rounds=50):
    """Solve KKT system: find a >= 0, sum(a) = 1 such that
    sum_i lambda_i * a[k_i - j] = const for all j,
    and (a*a)[k_i] equal for all i.
    Uses iterative lambda update + sparse linear solve."""
    K = len(peak_indices)
    lambda_vec = np.ones(K) / K
    best_a = None
    best_score = np.inf
    
    for rnd in range(max_rounds):
        # Build and solve the linear system
        H = _build_hankel_sparse(peak_indices, lambda_vec, n).tocsr()
        
        try:
            lu = splu(H.tocsc())
            a_sol = lu.solve(np.ones(n))
        except Exception:
            # Try with a small regularization
            H_reg = H + 1e-10 * lil_matrix.eye(n).tocsr()
            try:
                a_sol = H_reg.tocsc().solve(np.ones(n))
            except Exception:
                return None, np.inf
        
        if a_sol is None or not np.all(np.isfinite(a_sol)):
            return None, np.inf
        
        s = a_sol.sum()
        if abs(s) < 1e-15:
            return None, np.inf
        a_sol = a_sol / s
        
        # Check non-negativity (allow small negative)
        min_val = float(a_sol.min())
        if min_val < -0.05:
            # Try clipping and see if it works
            a_clip = np.maximum(a_sol, 0)
            sc = a_clip.sum()
            if sc < 1e-10:
                return None, np.inf
            a_clip = a_clip / sc
            # Only accept if the clipping was mild
            if min_val > -0.05:
                a_sol = a_clip
            else:
                # Try different lambda
                pass
        
        a_sol = np.maximum(a_sol, 0)
        s = a_sol.sum()
        if s < 1e-10:
            return None, np.inf
        a_sol = a_sol / s
        
        # Compute peaks
        c = _full_convolve(a_sol, a_sol)
        peak_vals = c[peak_indices]
        score = _score_norm_fn(a_sol)
        
        if score < best_score:
            best_score = score
            best_a = a_sol.copy()
        
        # Check if peaks are equal (convergence)
        pk_max = float(peak_vals.max())
        pk_min = float(peak_vals.min())
        if pk_max < 1e-15:
            break
        rel_spread = (pk_max - pk_min) / pk_max
        if rel_spread < 1e-8:
            break
        
        # Update lambda: weights inversely proportional to peak values
        # Peaks that are larger should get less weight (to push them down less)
        # Actually, we want to equalize: give more weight to smaller peaks
        new_lambda = 1.0 / np.maximum(peak_vals, 1e-15)
        new_lambda = new_lambda / new_lambda.sum()
        
        # Damp the update
        lambda_vec = 0.5 * lambda_vec + 0.5 * new_lambda
        lambda_vec = lambda_vec / lambda_vec.sum()
    
    return best_a, best_score


def _kkt_phase(a_init, deadline):
    """Try KKT refinement with different active sets."""
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Try different K values and peak selections
    for K in [2, 3, 4, 5]:
        if time.monotonic() > deadline:
            break
        
        # Get top K peaks
        sorted_idx = np.argsort(c)[::-1]
        
        # Also try peaks within 95% of max
        M = float(c.max())
        near_peaks = np.where(c >= 0.9 * M)[0]
        
        for peak_set in [sorted_idx[:K], sorted_idx[:K+2], near_peaks[:min(len(near_peaks), 2*K)]]:
            if len(peak_set) < 2:
                continue
            if len(peak_set) > 8:
                peak_set = peak_set[:8]
            
            a_kkt, sc_kkt = _solve_kkt(peak_set, n, max_rounds=30)
            if a_kkt is not None and sc_kkt < best_score:
                best_score = sc_kkt
                best_a = a_kkt.copy()
                print(f"    KKT K={len(peak_set)}: score={sc_kkt:.10f} (improved)")
            
            # Also try the KKT solution as a starting point for further optimization
            if a_kkt is not None:
                # Re-identify peaks and do another round
                c_new = _full_convolve(a_kkt, a_kkt)
                sorted_new = np.argsort(c_new)[::-1]
                for K2 in [2, 3]:
                    if time.monotonic() > deadline:
                        break
                    a_kkt2, sc_kkt2 = _solve_kkt(sorted_new[:K2], n, max_rounds=30)
                    if a_kkt2 is not None and sc_kkt2 < best_score:
                        best_score = sc_kkt2
                        best_a = a_kkt2.copy()
                        print(f"    KKT2 K={K2}: score={sc_kkt2:.10f} (improved)")
    
    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.02, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 50
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.97 ** (t // 50)))
        
        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if
