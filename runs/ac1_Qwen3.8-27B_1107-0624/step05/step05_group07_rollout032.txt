def _full_convolve(a, b=None):
    """Compute full convolution using FFT."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-300:

    sorted_c = np.sort(c)[::-1]
    top = sorted_c[:min(k, len(c))]

    top_shifted = top - top[0]
    e = np.exp(top_shifted / beta)
    lse = top[0] + beta * np.log(np.sum(e))

"""
Multi-scale LP minimax + top-k subgradient + log-sum-exp L-BFGS + coordinate descent
for minimizing 2*n*max(conv(a,a))/sum(a)^2. Key improvement over precedenti:
- Multi-scale LP escalation (512→1024→2048→4096) with warm starts
- Top-k subgradient mirror descent targeting the peak lags directly
- Adaptive beta continuation with non-uniform schedule
- Peak-lag targeted coordinate descent for final polish

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _normalize(a):
    s = float(np.sum(a))
    if s < 1e-15:
        return a
    return a / s


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return _normalize(p0)
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return _normalize(p)


def _build_conv_matrix_single(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_direction(f, solver=None):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix_single(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        if solver is not None:
            prob.solve(solver=solver)
        else:
            prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_val = g_val / g_val.sum()
            return g_val, float(t.value)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.ECOS)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_val = g_val / g_val.sum()
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters=50, deadline=None, step_init=0.5, decay=0.95):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm(f)
    step = step_init

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g, t_val = _lp_direction(f)
        if g is None:
            break
        f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0)
        f_new = _normalize(f_new)
        sc = _score_norm(f_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        step = max(0.05, step * decay)
        if it % 10 == 0:
            elapsed = time.monotonic() - (deadline - 1000) if deadline else 0
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}, step={step:.4f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 120.0)
        time_per = max(time_per, 5.0)
        maxiter = max(200, int(time_per * 20))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _topk_mirror_descent(a_init, deadline, k=10, lr_init=0.1, decay=0.9995):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    lr = lr_init
    iteration = 0

    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Find top-k lags
        if k >= len(c):
            top_indices = np.arange(len(c))
        else:
            top_indices = np.argsort(c)[::-1][:k]

        # Compute subgradient: d/da_j sum_{i in topk} conv(a,a)[i] / k
        # conv(a,a)[i] = sum_j a[j]*a[i-j]
        # d/da_j conv(a,a)[i] = a[i-j] + a[i-j+n-1] (with bounds checking)
        grad = np.zeros(n)
        for i in top_indices:
            for j in range(n):
                # a[i-j] if 0 <= i-j < n
                ij = i - j
                if 0 <= ij < n:
                    grad[j] += a[ij]
                # a[i-j] where we consider the other term: a[j-(i-(n-1))] = a[j+n-1-i]
                ij2 = j + n - 1 - i
                if 0 <= ij2 < n:
                    grad[j] += a[ij2]
        grad /= k

        # Scale gradient
        grad = 2.0 * n * grad

        # Mirror descent update
        z = z - lr * grad
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            iteration += 1

        lr = max(0.001, lr * decay)
        if iteration % 100 == 0 and iteration > 0:
            elapsed_info = time.monotonic()
            print(f"      TopK-MD: score={sc:.10f}, best={best_score:.10f}, lr={lr:.6f}, iter={iteration}")

        # Periodic warm restart
        if iteration > 0 and iteration % 500 == 0:
            z = z + 0.02 * np.random.randn(n)
            z = z - z.mean()

    return best_a, best_score


def _topk_mirror_descent_fast(a_init, deadline, k=5, lr_init=0.05):
    """Faster version using FFT-based gradient computation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    lr = lr_init
    iteration = 0

    while time.monotonic() < deadline:
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        if k >= len(c):
            top_mask = np.ones(len(c), dtype=bool)
        else:
            top_mask = np.zeros(len(c), dtype=bool)
            top_indices = np.argsort(c)[::-1][:k]
            top_mask[top_indices] = True

        # Weight vector w on convolution lags
        w = top_mask.astype(np.float64) / k

        # conv(w, a_reversed)[n-1:2n-1] gives the gradient contribution
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = float(np.dot(w, c))

        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        z = z - lr * grad
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        a_new = _softmax_fn(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            iteration += 1

        lr = max(0.001, lr * 0.9997)
        if iteration > 0 and iteration % 200 == 0:
            print(f"      TopK-MD-fast: score={sc:.10f}, best={best_score:.10f}, lr={lr:.6f}, iter={iteration}")

        if iteration > 0 and iteration % 1000 == 0:
            z = z + 0.01 * np.random.randn(n)
            z = z - z.mean()

    return best_a, best_score


def _coord_descent_targeted(a_init, deadline, n_rounds=20):
    """Coordinate descent targeting the peak lag."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        peak_lag = int(np.argmax(c))
        improved = False

        # Focus on elements that most affect the peak
        # conv(a,a)[peak_lag] involves pairs (i, peak_lag - i)
        important = []
        for i in range(n):
            j = peak_lag - i
            if 0 <= j < n:
                important.append(i)
        important = list(set(important))
        # Sort by contribution
        important.sort(key=lambda i: -a[i] * (a[peak_lag - i] if 0 <= peak_lag - i < n else 0))
        important = important[:min(50, len(important))]

        # Also include some random elements for exploration
        random_idx = np.random.choice(n, size=min(50, n), replace=False).tolist()
        all_idx = list(set(important + random_idx))

        for j in all_idx:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm(a)
            for delta in [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Linear decreasing
    a = np.maximum(1.0 - 0.5 * t, 1e-12)
    profiles.append(a / a.sum())
    # Linear increasing
    a = np.maximum(0.5 + 0.5 * t, 1e-12)
    profiles.append(a / a.sum())
    # Exponential decay
    a = np.exp(-t * 3)
    profiles.append(a / a.sum())
    # U-shaped
    a = 1.0 + u * u
    profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    # Bimodal
    a = np.exp(-10 * (t - 0.2)**2) + np.exp(-10 * (t - 0.8)**2)
    profiles.append(a / a.sum())
    # Cosine bump
    a = np.maximum(np.cos(np.pi * (t - 0.5)), 1e-12)**2
    profiles.append(a / a.sum())
    # Power law
    a = (t + 0.01)**(-0.5)
    profiles.append(a / a.sum())
    return profiles


def _single_beta_lbfgs(a_init, beta, deadline, maxiter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    try:
        result = minimize(
            _logsumexp_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_logsumexp_grad,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': maxiter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        parent = _make_profiles(n_init)[0]  # linear decreasing
        update_best(parent, _score_norm(parent), "default")

    # Phase 1: LP at n=512 (fast, many iterations)
    phase1_end = min(deadline, t_start + 150.0)
    print(f"\nPhase 1: LP at n=512")

    n1 = 512
    a1 = _interp_profile(parent, n1)
    update_best(a1, _score_norm(a1), f"lp_init_n{n1}")

    a1, sc1 = _lp_iterate(a1, n_iters=80, deadline=phase1_end, step_init=0.6, decay=0.97)
    update_best(a1, sc1, f"lp_n{n1}")

    # Phase 2: LP at n=1024
    phase2_end = min(deadline, t_start + 400.0)
    print(f"\nPhase 2: LP at n=1024")

    n2 = 1024
    a2 = _interp_profile(a1, n2)
    update_best(a2, _score_norm(a2), f"interp_n{n2}")

    a2, sc2 = _lp_iterate(a2, n_iters=50, deadline=phase2_end, step_init=0.4, decay=0.96)
    update_best(a2, sc2, f"lp_n{n2}")

    # Phase 3: LP at n=2048
    phase3_end = min(deadline, t_start + 700.0)
    print(f"\nPhase 3: LP at n=2048")

    n3 = 2048
    a3 = _interp_profile(a2, n3)
    update_best(a3, _score_norm(a3), f"interp_n{n3}")

    a3, sc3 = _lp_iterate(a3, n_iters=30, deadline=phase3_end, step_init=0.3, decay=0.95)
    update_best(a3, sc3, f"lp_n{n3}")

    # Phase 4: LP at n=4096 (fewer iterations, more expensive)
    phase4_end = min(deadline, t_start + 1000.0)
    print(f"\nPhase 4: LP at n=4096")

    n4 = 4096
    a4 = _interp_profile(a3, n4)
    update_best(a4, _score_norm(a4), f"interp_n{n4}")

    # Check if parent is at n=4096 and better
    if parent is not None and len(parent) == 4096:
        sc_p = _score_norm(parent)
        if sc_p < _score_norm(a4):
            a4 = parent.copy()
            update_best(a4, sc_p, "parent_n4096")

    a4, sc4 = _lp_iterate(a4, n_iters=15, deadline=phase4_end, step_init=0.2, decay=0.93)
    update_best(a4, sc4, f"lp_n{n4}")

    print(f"  Phases 1-4 done: best={best_score:.10f}")

    # Phase 5: Top-k mirror descent
    phase5_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 5: Top-k mirror descent")

    a_cur = best_seq.copy()
    n_cur = len(a_cur)

    # Try different k values
    for k in [3, 5, 10]:
        if time.monotonic() > phase5_end - 30:
            break
        per_dl = min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic()) / 3)
        a_k, sc_k = _topk_mirror_descent_fast(a_cur, per_dl, k=k, lr_init=0.03)
        update_best(a_k, sc_k, f"topk_{k}_n{n_cur}")
        a_cur = best_seq.copy()

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Phase 6: L-BFGS smooth optimization
    phase6_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 6: L-BFGS continuation")

    a_cur = best_seq.copy()
    # Use a moderate beta list for the remaining time
    c_cur = _full_convolve(a_cur, a_cur)
    max_c = float(c_cur.max())
    beta_list_custom = [b * max_c for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]
    a6, sc6 = _run_lbfgs_continuation(a_cur, phase6_end, beta_list=beta_list_custom)
    update_best(a6, sc6, f"lbfgs_n{len(a_cur)}")

    # Try a restart with noise
    if time.monotonic() < phase6_end - 30:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.03 * rng.normal(size=len(z))
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_restart = _softmax_fn(z)
        per_dl = min(phase6_end - 5, time.monotonic() + 60.0)
        a6b, sc6b = _run_lbfgs_continuation(a_restart, per_dl, beta_list=beta_list_custom[:4])
        update_best(a6b, sc6b, "lbfgs_restart")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # Phase 7: Very fine L-BFGS + coordinate descent
    phase7_end = deadline
    print(f"\nPhase 7: Fine polish")

    # Tiny beta L-BFGS
    if time.monotonic() < phase7_end - 60 and best_seq is not None:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_tiny = 5e-5 * max_cf
        per_dl = min(phase7_end - 30, time.monotonic() + 80.0)
        a7, sc7 = _single_beta_lbfgs(best_seq, beta_tiny, per_dl, maxiter=10000)
        update_best(a7, sc7, "tiny_beta")

    # Coordinate descent
    if time.monotonic() < phase7_end - 20 and best_seq is not None:
        per_dl = min(phase7_end - 5, time.monotonic() + (phase7_end - time.monotonic()) * 0.8)
        a8, sc8 = _coord_descent_targeted(best_seq, per_dl, n_rounds=10)
        update_best(a8, sc8, "coord_descent")

    # One more tiny-beta L-BFGS after coordinate descent
    if time.monotonic() < phase7_end - 10 and best_seq is not None:
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_ultra = 2e-5 * max_cf
        per_dl = min(phase7_end - 2, time.monotonic() + 30.0)
        a9, sc9 = _single_beta_lbfgs(best_seq, beta_ultra, per_dl, maxiter=5000)
        update_best(a9, sc9, "ultra_tiny_beta")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # Phase 8: Try n=8192 if time permits
    if time.monotonic() < deadline - 60 and best_seq is not None:
        print(f"\nPhase 8: Try n=8192")
        n8 = 8192
        a8 = _interp_profile(best_seq, n8)
        sc8 = _score_norm(a8)
        update_best(a8, sc8, f"interp_n{n8}")
        if time.monotonic() < deadline - 30:
            c8 = _full_convolve(a8, a8)
            max_c8 = float(c8.max())
            beta_list_8 = [b * max_c8 for b in [0.01, 0.005, 0.002, 0.001]]
            per_dl = min(deadline - 10, time.monotonic() + 50.0)
            a8b, sc8b = _run_lbfgs_continuation(a8, per_dl, beta_list=beta_list_8)
            update_best(a8b, sc8b, f"lbfgs_n{n8}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[0]
    print(f"FALLBACK: n={n_fb}, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
