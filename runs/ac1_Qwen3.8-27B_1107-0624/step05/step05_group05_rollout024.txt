import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        if len(j_range) > 0:
            rows_list.append(np.full(len(j_range), k, dtype=np.int32))
            cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    # A[k, i] = f[|k - i|] if valid, representing (f * g)[k] = sum_i A[k,i] * g[i]
    # Actually for conv(f, g)[k] = sum_i f[max(0, k-n+1+... )]... 
    # Let's use: (f*g)[k] = sum_{i} f[i] * g[k-i] where both indices in [0,n-1]
    # So the coefficient of g[j] in (f*g)[k] is f[k-j] if 0<=k-j<n, else 0
    # So A[k, j] = f[k-j] if 0 <= k-j < n else 0
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    if M < 1e-300:
        return f, M
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M

        grad, w, c, c_max = _compute_gradient_smoothed(a, beta)
        
        # Center the gradient (projection onto tangent space of simplex)
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % 500 == 0 or t == 1:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        
        if t % 5000 == 0:
            if no_improve_count < 2:
                current_lr = min(0.05, current_lr * 1.05)
            elif no_improve_count > 10:
                current_lr = max(0.0005, current_lr * 0.9)
            no_improve_count = 0

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _peak_shaving_coordinate_descent(a, n_steps=50, rng=None):
    """
    Coordinate descent that targets the top convolution peaks.
    For each peak, tries to reduce it by adjusting coordinates.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for step in range(n_steps):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top K peaks
        K = min(10, 2 * n - 1)
        top_k = np.argsort(c)[::-1][:K]
        
        # For each peak, compute the shaving direction
        # Shaving: reduce a[i] proportional to a[k-i] for the peak positions
        shave_direction = np.zeros(n)
        for k in top_k:
            weight = c[k] - c_max * 0.9  # Only shave peaks that are significantly at max
            if weight <= 0:
                weight = 0.1  # Still shave the top ones slightly
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    shave_direction[i] += weight * a[j]
        
        # Normalize the shaving direction
        sd_norm = np.linalg.norm(shave_direction)
        if sd_norm < 1e-15:
            break
        shave_direction /= sd_norm
        
        # Try several step sizes in the shaving direction
        # New a = a - eta * shave_direction, then project to simplex
        best_eta = 0.0
        best_local_sc = best_sc
        for eta in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_trial = a - eta * shave_direction
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_local_sc:
                best_local_sc = sc
                best_eta = eta
        
        if best_eta > 0:
            a_new = a - best_eta * shave_direction
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            if _score_norm(a_new) < best_sc:
                best_sc = _score_norm(a_new)
                best_a = a_new.copy()
                a = a_new.copy()
        else:
            # Try the opposite: "fill" direction (increase where partners are small)
            fill_direction = np.zeros(n)
            for k in top_k:
                for i in range(n):
                    j = k - i
                    if 0 <= j < n:
                        fill_direction[i] -= a[j]  # Negative of shave
            fd_norm = np.linalg.norm(fill_direction)
            if fd_norm < 1e-15:
                break
            fill_direction /= fd_norm
            # fill_direction is -shave_direction normalized, already tried above
            # So break
            break
    
    return best_a, best_sc


def _structured_perturb(a, strength, rng):
    """
    Perturb the sequence in a structured way that targets convolution peaks.
    Instead of pure Gaussian noise, use noise shaped by the current peak structure.
    """
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Create a "peak sensitivity" vector: how much would increasing a[i] increase the max conv?
    sensitivity = np.zeros(n)
    top_k = np.argsort(c)[::-1][:5]
    for k in top_k:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                sensitivity[i] += a[j]
    
    # Normalize
    s_norm = np.linalg.norm(sensitivity)
    if s_norm > 1e-15:
        sensitivity /= s_norm
    else:
        sensitivity = np.ones(n) / n
    
    # Mix: structured noise (antithetic to peaks) + random noise
    structured = -strength * sensitivity
    random = 0.5 * strength * rng.standard_normal(n)
    
    z = np.log(np.maximum(a, 1e-15))
    z = z + structured + random
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()

    best_seq = None
    best_score = np.inf

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc

    else:
        n = 2048

        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    rows, cols = _build_lp_matrix(n)
    print(f"LP matrix built: {len(rows)} nonzeros, t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 1: LP warmup (0 - 120s) ===
    phase1_end = min(deadline, t_start + 120.0)
    if time.monotonic() < phase1_end:
        lp_count = 0
        prev_sc = _score_norm(a)
        stall = 0
        while time.monotonic() < phase1_end:
            t0 = time.monotonic()
            a, M = _lp_step(a, rows, cols, time_limit=20)

            lp_count += 1
            dt = time.monotonic() - t0

            else:
                stall += 1
            if lp_count % 3 == 0:

                print(f"  LP[{lp_count}] score={full_sc:.10f} dt={dt:.1f}s stall={stall} t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:

            if stall > 20:

        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc

        print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
