def _project_simplex(v, S=1.0):
    """Project v onto {a >= 0, sum(a) = S}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if S <= 0:
        return np.zeros(n)
    # Project onto {a >= 0, sum(a) = S}
    # This is equivalent to S * project(v/S onto standard simplex)
    u = v / S
    # Project onto standard simplex {a >= 0, sum(a) = 1}
    n = len(u)
    if n == 0:
        return np.array([0.0])
    mu = 0.0
    sv = np.sort(u)[::-1]
    cssv = np.cumsum(sv)
    rho = np.nonzero(1 + sv * np.arange(1, n + 1) > cssv)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    proj = np.maximum(u - theta, 0)
    return S * proj

```python
def _project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(1 + u * np.arange(1, n+1) > cssv)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

"""

Key improvements over previous: direct simplex projection (not log-params),
Nesterov acceleration, adaptive step backtracking, direct subgradient on max,

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([0.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = (u * idx) + 1 > cssv
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_lse_c(a, beta):
    """Gradient of beta*log(sum(exp(c/beta))) w.r.t. a, where c=conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_shifted = (c - c.max()) / beta
    max_s = float(c_shifted.max())
    e = np.exp(c_shifted - max_s)
    w = e / e.sum()
    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices
    # = (a * w)[j] + (a * w_rev)[2n-2-j]
    w_rev = w[::-1]
    q_full = _full_convolve(a, w)  # length 3n-2
    p_full = _full_convolve(a, w_rev)  # length 3n-2
    p = p_full[n - 1:2 * n - 1][::-1]
    q = q_full[:n]
    return p + q


def _grad_max_subgrad(a, peak_idx):
    """Subgradient of c[peak_idx] = sum_j a[j]*a[peak_idx-j] w.r.t. a."""
    n = len(a)
    k = peak_idx
    grad = np.zeros(n)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo <= i_hi:
        grad[i_lo:i_hi + 1] += a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    j_lo = max(0, k)
    j_hi = min(n - 1, 2 * n - 2 - k)
    if j_lo <= j_hi:
        grad[j_lo:j_hi + 1] += a[k - j_lo + (n - 1 - (j_hi - j_lo)):][::-1][:0] if False else np.zeros(j_hi - j_lo + 1)
    # Second term: a[j-k] for j in [k, k+n-1] intersect [0, n-1]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        grad[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]
    return grad


def _nesterov_lse_phase(a_init, deadline, n_beta_decreases=30, max_iter_total=500000):
    """Nesterov accelerated projected gradient on LSE surrogate with decreasing beta."""
    n = len(a_init)
    a = a_init.copy()
    c0 = _full_convolve(a, a)
    cm0 = float(c0.max())
    best_a = a.copy()
    best_score = _score_norm(a)

    beta = 0.1 * cm0
    beta_min = 1e-6 * cm0
    momentum = 0.9
    step = 1.0 / (float(n) * cm0)
    max_iter = max_iter_total

    t = 0
    a_prev = a.copy()
    no_improve = 0

    while t < max_iter and time.monotonic() < deadline:
        # Nesterov extrapolation
        y = a + momentum * (a - a_prev)
        a_prev = a.copy()

        # Gradient at y
        g = _grad_lse_c(y, beta)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        # Adaptive step with backtracking
        alpha = step
        for _ in range(20):
            a_new = _project_simplex(y - alpha * g)
            f_new = float(beta * np.log(np.sum(np.exp((_full_convolve(a_new, a_new) - _full_convolve(a_new, a_new).max()) / beta))))
            f_y = float(beta * np.log(np.sum(np.exp((_full_convolve(y, y) - _full_convolve(y, y).max()) / beta))))
            if f_new <= f_y + 1e-4 * alpha * g_norm * float(np.dot(g, a_new - y)):
                break
            alpha *= 0.5

        a = a_new
        t += 1

        # Check progress
        if t % 50 == 0:
            sc = _score_norm(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Decrease beta periodically
            if no_improve > 10:
                beta *= 0.7
                momentum = min(0.95, momentum + 0.01)
                no_improve = 0

            if beta < beta_min:
                break

    return best_a, best_score


def _subgrad_momentum_phase(a_init, deadline, max_iter=200000):
    """Direct subgradient on max with momentum and adaptive step."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)

    momentum_vec = np.zeros(n)
    momentum = 0.7
    step = 1.0 / (float(n) * float(_full_convolve(a, a).max()))
    t = 0
    no_improve = 0

    while t < max_iter and time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())
        peak = int(np.argmax(c))

        # Subgradient of max at the argmax
        k = peak
        grad = np.zeros(n)
        # dc[k]/da[j] = a[k-j] (if valid) + a[j-k] (if valid)
        # First term: a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] = a[k - lo1:k - lo1 + (hi1 - lo1 + 1)]
        # Second term: a[j-k] for j in [max(0,k), min(n-1, 2n-2-k)]
        lo2 = max(0, k)
        hi2 = min(n - 1, 2 * n - 2 - k)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += a[lo2 - k:hi2 - k + 1]

        # Project gradient onto tangent space of simplex (remove mean of free components)
        grad = grad - grad.mean()

        # Momentum update
        momentum_vec = momentum * momentum_vec + grad
        g_norm = float(np.linalg.norm(momentum_vec))
        if g_norm < 1e-15:
            break

        # Take step with line search
        alpha = step
        for _ in range(15):
            a_trial = _project_simplex(a - alpha * momentum_vec)
            c_trial = _full_convolve(a_trial, a_trial)
            if c_trial.max() < cm + 1e-12:
                a = a_trial
                break
            alpha *= 0.5
        else:
            a = _project_simplex(a - alpha * momentum_vec)

        t += 1
        if t % 100 == 0:
            sc = _score_norm(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 20:
                step *= 0.5
                momentum = min(0.9, momentum + 0.02)
                no_improve = 0
            if momentum > 0.95:
                momentum = 0.95

    return best_a, best_score


def _lp_refine_phase(a_init, deadline, max_rounds=20):
    """LP-based peak balancing with adaptive active set."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Get active peaks: top K and those within 90% of max
        sorted_idx = np.argsort(c)[::-1]
        K = min(50, len(sorted_idx))
        active = sorted_idx[:K]
        threshold = 0.92 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]

        # Build LP: max sum(b) s.t. (a*b)[k] <= M for k in active, b >= 0
        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = a_init[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=np.full(len(active), M * 1.0001),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        # Try blending
        improved = False
        current_score = _score_norm(best_a)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc_trial = _score_norm(a_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _peak_shave_phase(a_init, deadline, max_iter=100000):
    """Iteratively identify and shave peaks using direct gradient steps."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    step = 0.01 / float(_full_convolve(a, a).max()) * float(a.sum())
    no_improve = 0

    it = 0
    while it < max_iter and time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Top 3 peaks
        top_idx = np.argsort(c)[::-1][:3]
        top_vals = c[top_idx]
        # Weight by how close to max
        weights = np.exp(-5.0 * (top_vals - top_vals.min()) / max(1e-15, cm - top_vals.min() + 1e-30))
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            lo1 = max(0, k - n + 1)
            hi1 = min(n - 1, k)
            if lo1 <= hi1:
                grad[lo1:hi1 + 1] += wi * a[k - lo1:k - lo1 + (hi1 - lo1 + 1)]
            lo2 = max(0, k)
            hi2 = min(n - 1, 2 * n - 2 - k)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] += wi * a[lo2 - k:hi2 - k + 1]

        grad = grad - grad.mean()
        # Normalize
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        grad = grad / gn

        a_new = _project_simplex(a - step * grad)
        c_new = _full_convolve(a_new, a_new)
        if c_new.max() < cm - 1e-15:
            a = a_new
            no_improve = 0
        else:
            step *= 0.7
            no_improve += 1

        if no_improve > 20:
            break
        it += 1

        if it % 200 == 0:
            sc = _score_norm(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _lbfgs_lse_phase(a_init, deadline, beta_list=None):
    """L-BFGS on log-sum-exp with log-parameters (proven effective for mid-range beta)."""
    n = len(a_init)

    def _softmax(z):
        z = z - z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            return np.ones_like(z) / len(z)
        return e / s

    def _obj(z):
        a_s = _softmax(z)
        c_s = _full_convolve(a_s, a_s)
        cm = float(c_s.max())
        if cm < 1e-300:
            return 0.0
        sh = (c_s - cm) / beta_cur[0]
        ms = float(sh.max())
        lse = cm + beta_cur[0] * (ms + float(np.log(np.sum(np.exp(sh - ms)))))
        return float(2.0 * n * lse)

    def _grad(z):
        a_s = _softmax(z)
        c_s = _full_convolve(a_s, a_s)
        cm = float(c_s.max())
        if cm < 1e-300:
            return np.zeros_like(z)
        sh = (c_s - cm) / beta_cur[0]
        ms = float(sh.max())
        e = np.exp(sh - ms)
        w = e / e.sum()
        w_rev = w[::-1]
        q_full = _full_convolve(a_s, w)
        p_full = _full_convolve(a_s, w_rev)
        p = p_full[n - 1:2 * n - 1][::-1]
        q = q_full[:n]
        grad_a = p + q
        Wa = float(np.dot(w, c_s))
        grad_z = 2.0 * n * a_s * (grad_a - Wa)
        if not np.all(np.isfinite(grad_z)):
            return np.zeros_like(z)
        return grad_z

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        beta_list = [m0 * f for f in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        beta_cur = [beta]

        def obj_cur(z_in):
            return _obj(z_in)

        def grad_cur(z_in):
            return _grad(z_in)

        try:
            result = minimize(obj_cur, z, method='L-BFGS-B', jac=grad_cur,
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxls': 50, 'maxfun': 10000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _greedy_init(n):
    """Binomial-based initialization."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _spectral_init(n, shift=0.0):
    """Initialize using flat-spectral design: make |FFT(a)|^2 as flat as possible.
    Start with a whose FFT has magnitude ~ 1/sqrt(n) everywhere (rectangular in freq)."""
    # Ideal: |A[k]|^2 = constant for all k → a is a delta (not helpful)
    # Instead: use a window that's "maximally spread" in frequency
    # Use raised-cosine or Tukey window as starting point
    t = np.linspace(0, 1, n)
    # Tukey window with alpha = 0.5
    alpha = 0.5 + shift
    w = np.ones(n)
    m = int(alpha * n / 2)
    for i in range(min(m, n)):
        w[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * n) - 1)))
        w[n - 1 - i] = w[i]
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def _bimodal_init(n):
    """Bimodal initialization: high at edges, low in middle."""
    t = np.linspace(0, 1, n)
    a = 1.0 - 0.8 * np.sin(np.pi * t) ** 2
    a = np.maximum(a, 0.05)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: Nesterov gradient, subgradient momentum,
    L-BFGS continuation, LP refinement, peak shaving, multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_actual = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score, best_actual
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            actual = _score_fn(p)
            best_actual = actual
            print(f"  NEW BEST: norm={sc:.10f}, actual={actual:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        update_best(parent, _score_norm(parent), "parent")
        print(f"Parent loaded: n={n_parent}")
    else:
        n_parent = 8192
        parent = _greedy_init(n_parent)
        update_best(parent, _score_norm(parent), "greedy_init")

    # === Phase 1: L-BFGS continuation from parent (400s) ===
    print("=" * 60)
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 400.0)

    a_b, sc_b = _lbfgs_lse_phase(best_seq, p1_end)
    update_best(a_b, sc_b, "lbfgs_main")

    # Try from different inits
    if time.monotonic() < p1_end - 60:
        for init_fn, name in [(_greedy_init, "greedy"), (_spectral_init, "spectral"), (_bimodal_init, "bimodal")]:
            if time.monotonic() > p1_end - 30:
                break
            a_try = init_fn(n_parent)
            sc_try = _score_norm(a_try)
            if sc_try < best_score * 1.05:
                a_t2, sc_t2 = _lbfgs_lse_phase(a_try, min(p1_end - 5, time.monotonic() + 50),
                                               beta_list=[_full_convolve(a_try, a_try).max() * f
                                                         for f in [0.1, 0.05, 0.02, 0.01]])
                update_best(a_t2, sc_t2, f"lbfgs_{name}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov accelerated gradient (300s) ===
    print("=" * 60)
    print("Phase 2: Nesterov gradient")
    p2_end = min(deadline, t_start + 700.0)

    if best_seq is not None:
        a_n, sc_n = _nesterov_lse_phase(best_seq, p2_end)
        update_best(a_n, sc_n, "nesterov")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP refinement (300s) ===
    print("=" * 60)
    print("Phase 3: LP refinement")
    p3_end = min(deadline, t_start + 1000.0)

    if best_seq is not None and time.monotonic() < p3_end:
        a_lp, sc_lp = _lp_refine_phase(best_seq, p3_end, max_rounds=25)
        update_best(a_lp, sc_lp, "lp_refine")

        if time.monotonic() < p3_end - 60:
            a_lp2, sc_lp2 = _lp_refine_phase(best_seq, p3_end, max_rounds=15)
            update_best(a_lp2, sc_lp2, "lp_refine2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Peak shaving (200s) ===
    print("=" * 60)
    print("Phase 4: Peak shaving")
    p4_end = min(deadline, t_start + 1200.0)

    if best_seq is not None and time.monotonic() < p4_end:
        a_ps, sc_ps = _peak_shave_phase(best_seq, p4_end)
        update_best(a_ps, sc_ps, "peak_shave")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Subgradient momentum (200s) ===
    print("=" * 60)
    print("Phase 5: Subgradient momentum")
    p5_end = min(deadline, t_start + 1400.0)

    if best_seq is not None and time.monotonic() < p5_end:
        a_sg, sc_sg = _subgrad_momentum_phase(best_seq, p5_end, max_iter=100000)
        update_best(a_sg, sc_sg, "subgrad")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: L-BFGS with very small beta (150s) ===
    print("=" * 60)
    print("Phase 6: Fine L-BFGS")
    p6_end = min(deadline, t_start + 1550.0)

    if best_seq is not None and time.monotonic() < p6_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * f for f in [0.001, 0.0005, 0.0002, 0.0001]]
        a_f, sc_f = _lbfgs_lse_phase(best_seq, p6_end, beta_list=tiny_betas)
        update_best(a_f, sc_f, "fine_lbfgs")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Try nearby n values (120s) ===
    print("=" * 60)
    print("Phase 7: Multi-n exploration")
    p7_end = min(deadline, t_start + 1670.0)

    if best_seq is not None and time.monotonic() < p7_end:
        n_cur = len(best_seq)
        for n_try in [n_cur - 512, n_cur + 512, n_cur + 1024]:
            if time.monotonic() > p7_end - 20:
                break
            if n_try < 100 or n_try > 20000:
                continue
            a_interp = _interp_profile(best_seq, n_try)
            sc_interp = _score_norm(a_interp)
            update_best(a_interp, sc_interp, f"interp_n{n_try}")
            if sc_interp < best_score * 1.01 and time.monotonic() < p7_end - 30:
                a_ib, sc_ib = _lbfgs_lse_phase(a_interp, min(p7_end - 5, time.monotonic() + 30),
                                               beta_list=[_full_convolve(a_interp, a_interp).max() * f
                                                         for f in [0.01, 0.005, 0.002]])
                update_best(a_ib, sc_ib, f"lbfgs_n{n_try}")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Final combined polish (remaining ~100s) ===
    print("=" * 60)
    print("Phase 8: Final polish")

    if best_seq is not None and time.monotonic() < deadline - 30:
        # Quick Nesterov
        a_fn, sc_fn = _nesterov_lse_phase(best_seq, min(deadline - 20, time.monotonic() + 40),
                                          n_beta_decreases=10, max_iter_total=50000)
        update_best(a_fn, sc_fn, "final_nesterov")

        # Quick peak shave
        if time.monotonic() < deadline - 20:
            a_fp, sc_fp = _peak_shave_phase(best_seq, min(deadline - 10, time.monotonic() + 30), max_iter=30000)
            update_best(a_fp, sc_fp, "final_shave")

    print(f"  Phase 8 done: best={best_score:.10f}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
