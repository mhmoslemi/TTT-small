```python
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse
import time

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def score_norm(a):
    """Returns 2*n*max(conv) for normalized a (sum=1)."""
    c = full_convolve(a, a)
    return 2.0 * len(a) * c.max()

def make_lp_matrix(a, k_indices):
    """
    Build the constraint matrix for the LP.
    A[k_idx, i] = a[i] if 0 <= k_idx - i < n, else 0
    Only for k in k_indices.
    Returns a sparse matrix of shape (len(k_indices), n).
    """
    n = len(a)
    rows = []
    cols = []
    vals = []
    for ki, k in enumerate(k_indices):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(ki)
            cols.append(i)
            vals.append(a[i])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    return A

def lp_iteration(a, M, k_indices, solver=' highs'):
    """
    Solve: max sum(g) s.t. A @ g <= M, g >= 0
    where A[k_i, i] = a[i] if valid.
    """
    n = len(a)
    A = make_lp_matrix(a, k_indices)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b_ub = np.full(len(k_indices), M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method=solver)
    if res.status == 0:
        return res.x, -res.fun
    return None, None
```

```python
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        M = c[k_star]

        # Subgradient: g[i] = 2 * a[k_star - i] (with a[j]=0 outside [0,n-1])
        g = np.zeros(n)
        valid = np.arange(max(0, k_star - n + 1), min(n, k_star + 1))
        g[valid] = 2.0 * a[k_star - valid]

        # Mirror descent update: a[i] *= exp(-eta * g[i])
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * g
        a = _softmax_fn(log_a)

        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Decay eta
        if step % 1000 == 0 and step > 0:
            eta *= 0.8

    return best_a, best_sc


def _peak_local_search(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(123)

    for iteration in range(2000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        top_peaks = np.argsort(c)[::-1][:5]

        improved = False
        for pk in top_peaks:
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = best_a[valid_i] * best_a[pk - valid_i]
            top_contrib = np.argsort(prod)[::-1][:10]

            for idx_pos in top_contrib:
                idx = valid_i[idx_pos]
                for frac in [0.95, 0.9, 0.85, 0.8, 0.7]:
                    a_try = best_a.copy()
                    a_try[idx] *= frac
                    # Redistribute to neighbors
                    redist = (best_a[idx] - a_try[idx])
                    neighbors = [max(0, idx - 1), min(n - 1, idx + 1),
                                 max(0, idx - 2), min(n - 1, idx + 2)]
                    for nb in neighbors:
                        a_try[nb] += redist / len(neighbors)
                    a_try = np.maximum(a_try, 0)
                    a_try = _normalize(a_try)
                    sc_try = _score_norm(a_try)
                    if sc_try < best_sc - 1e-12:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try adding to small elements
            small_idx = np.argsort(best_a)[:10]
            for idx in small_idx:
                for mult in [1.1, 1.2, 1.5]:
                    a_try = best_a.copy()
                    a_try[idx] *= mult
                    a_try = np.maximum(a_try, 0)
                    a_try = _normalize(a_try)
                    sc_try = _score_norm(a_try)
                    if sc_try < best_sc - 1e-12:
                        best_sc = sc_try
                        best_a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and iteration > 200:
            break

    return best_a, best_sc


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline_final = t_start + total_budget - 30.0

    print(f"Total budget: {total_budget}s, deadline at t={total_budget - 30:.0f}s")

    # Get initial sequence
    a_init = _get_initial_sequence()
    n = len(a_init)
    print(f"Initial: n={n}, score={_score(a_init):.10f}")

    best_seq = a_init.copy()
    best_score = _score(a_init)

    def update_best(a_new, sc_new, label=""):
        nonlocal best_seq, best_score
        if a_new is not None and np.isfinite(sc_new) and sc_new < best_score:
            best_seq = np.asarray(a_new, dtype=np.float64).copy()
            best_score = float(sc_new)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(best_seq)}) [{label}] t={elapsed:.0f}s")

    # Phase 0: Multi-scale warmup (100s)
    phase0_end = t_start + 100.0
    if n > 512:
        # Downsample for quick optimization, then upsample back
        a_small = a_init[:64] if n >= 64 else a_init
        a_small = _normalize(a_small)
        a_small_opt, sc_small = _lbfgs_optimize(a_small, min(phase0_end, t_start + 20), n_betas=6)
        # Upsample back
        a_warm = _upsample(a_small_opt, n)
        sc_warm = _score(a_warm)
        update_best(a_warm, sc_warm, "upsampled_warmup")

    # Quick L-BFGS on full size
    if time.monotonic() < phase0_end - 5:
        a_p0, sc_p0 = _lbfgs_optimize(best_seq, phase0_end, n_betas=8)
        update_best(a_p0, sc_p0, "phase0_lbfgs")

    # Phase 1: Mirror descent for quick improvement (100s)
    phase1_end = t_start + 200.0
    if time.monotonic() < phase1_end - 5:
        a_md, sc_md = _mirror_descent(best_seq, n_steps=5000, deadline=phase1_end)
        update_best(a_md, sc_md, "mirror_descent")

    # Phase 2: LP-based optimization (main, 1300s)
    phase2_end = t_start + total_budget - 400.0
    print(f"\nPhase 2: LP optimization (deadline t={phase2_end - t_start:.0f}s)")
    if time.monotonic() < phase2_end - 10:
        a_lp, sc_lp = _lp_optimize(best_seq, phase2_end)
        update_best(a_lp, sc_lp, "lp_optimize")

        # Second LP round with different parameters if time allows
        if time.monotonic() < phase2_end - 60:
            # Perturb slightly and run again
            rng = np.random.default_rng(int(seed) + 1)
            a_pert = best_seq + 0.001 * rng.normal(size=len(best_seq))
            a_pert = np.maximum(a_pert, 0)
            a_pert = _normalize(a_pert)
            a_lp2, sc_lp2 = _lp_optimize(a_pert, phase2_end)
            update_best(a_lp2, sc_lp2, "lp_optimize_perturbed")

    # Phase 3: L-BFGS polishing (250s)
    phase3_end = t_start + total_budget - 130.0
    print(f"\nPhase 3: L-BFGS polishing (deadline t={phase3_end - t_start:.0f}s)")
    if time.monotonic() < phase3_end - 10:
        a_fine, sc_fine = _lbfgs_optimize(best_seq, phase3_end, n_betas=20)
        update_best(a_fine, sc_fine, "phase3_lbfgs")

    # Phase 4: Peak local search (80s)
    phase4_end = t_start + total_budget - 40.0
    print(f"\nPhase 4: Peak local search (deadline t={phase4_end - t_start:.0f}s)")
    if time.monotonic() < phase4_end - 5:
        a_ls, sc_ls = _peak_local_search(best_seq, phase4_end)
        update_best(a_ls, sc_ls, "peak_search")

    # Phase 5: Final tiny-beta L-BFGS (30s)
    if time.monotonic() < deadline_final - 5:
        n_final = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-30))
        z = z - z.mean()
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_tiny = 1e-6 * max_c_final

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny, n_final),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 50000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 100, 'maxfun': 150000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score(a_new)
                update_best(a_new, sc_new, "final_tiny_beta")
        except Exception:
            pass

    # Final validation
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    s = float(final_seq.sum())
    if s < 0.01:
        final_seq = _make_ramp(256, 1.5)
    else:
        # Normalize to have a "nice" sum
        final_seq = final_seq / s

    sc_final = _score(final_seq)
    print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in final_seq]
```
