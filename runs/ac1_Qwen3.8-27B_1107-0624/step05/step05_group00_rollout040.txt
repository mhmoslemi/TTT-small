import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses improved Adam log-space with very low beta smoothing, LP-based direction finding,
structured restarts with power-law/exponential initializations, and multi-n exploration.
Key improvements over previous: lower beta schedule, structured inits, LP step, n=8192.
"""
import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _projection_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total,
                   lr_decay=0.97, clip_norm=0.4, z_clip=40.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_global = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for phase_idx, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        while t_global < max_iter_total:
            if time.monotonic() > deadline:
                break
            t_global += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_global)
            vh = v_buf / (1.0 - b2 ** t_global)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > clip_norm:
                update = update * (clip_norm / update_norm)

            z = z - update
            z = np.clip(z, -z_clip, z_clip)
            z = z - z.mean()

            if t_global % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t_global % 8000 == 0:
                current_lr = max(0.0005, current_lr * lr_decay)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _projection_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _lp_step(a, max_lp_time=60.0):
    """
    LP-based direction: find b maximizing sum(b) s.t. (a*b)(k) <= M, b >= 0.
    Only works for n <= 2048 due to memory.
    Returns normalized b or None.
    """
    n = len(a)
    if n > 2048:
        return None

    m = 2 * n - 1
    M = float(_full_convolve(a, a).max())
    if M < 1e-15:
        return None

    # Build constraint matrix: A[k,j] = a[k-j] if 0 <= k-j < n
    # Use vectorized construction
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        # a[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(j + n, m)
        A[k_start:k_end, j] = a[:k_end - k_start]

    # Max sum(b) s.t. A @ b <= M, b >= 0
    # linprog minimizes, so negate objective
    t_lp_start = time.monotonic()
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(m),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_lp_time - 5})
    except Exception:
        return None

    if res.success and res.x is not None:
        b = res.x
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None


def _make_power_law(n, p, reverse=False):
    """Create a power-law initialization."""
    if reverse:
        idx = np.arange(n - 1, -1, -1, dtype=np.float64)
    else:
        idx = np.arange(n, dtype=np.float64)
    a = (idx + 1.0) ** p
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_exponential(n, lam):
    """Create an exponential initialization."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx / n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_v_shape(n, p):
    """Create a V-shaped (anti-concentrated) initialization."""
    t = np.linspace(0, 1, n)
    a = np.abs(t - 0.5) ** p + 1e-10
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _combine_init(a, b, alpha):
    """Mix two distributions."""
    c = alpha * a + (1.0 - alpha) * b
    c = np.maximum(c, 1e-15)
    return c / c.sum()

        a_pol = best_seq.copy()
        a_pol = np.maximum(a_pol, 0)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        # Very low beta for ultra-fine
        betas_ult = [0.00005 * max_cp, 0.00002 * max_cp, 0.00001 * max_cp,
                     0.000005 * max_cp, 0.000002 * max_cp]
        per_t = max(10.0, deadline - time.monotonic() - 5)
        a_pol, sc_pol = _adam_logspace(
            a_pol, time.monotonic() + per_t,
            betas_ult, lr_init=0.004, max_iter_total=500000,
            lr_decay=0.98, clip_norm=0.2
        )
        sc_pol_full = _score(a_pol)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol.copy()
            print(f"  Final polish: score={sc_pol_full:.10f}")
        else:
            print(f"  Final polish: score={sc_pol_full:.10f} (no improve)")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        print(f"Score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_power_law(1024, 1.5)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
