def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):

    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

    return e / s

def make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5)**2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def fourier_param_score(coeffs, n, n_modes):
    """Evaluate score for a sequence parameterized by Fourier coefficients."""
    i = np.arange(n)
    log_a = np.zeros(n)
    for k in range(n_modes + 1):
        c_k = coeffs[2*k]
        if k > 0:
            s_k = coeffs[2*k + 1]
            log_a += c_k * np.cos(2.0 * np.pi * k * i / n) + s_k * np.sin(2.0 * np.pi * k * i / n)
        else:
            log_a += c_k
    
    a = np.exp(log_a - np.max(log_a))
    a = a / a.sum()
    return score(a), a

def perturb(a, sigma, rng):

def perturb_fourier(a, rng, n_modes=3):

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines PRIME-like multiplicative updates, aggressive L-BFGS beta-continuation,
active-set subgradient with backtracking, CVXPY QP polishing, and multi-restart.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1].copy()
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _grad_active(a, c, n, tol_frac=0.002):
    c_max = c.max()
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1].copy()
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _prime_update(a, n_iter=300, alpha=0.05, tol_frac=0.005):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = _score(a)
    for it in range(n_iter):
        c = _full_convolve(a, a)
        c_max = c.max()
        threshold = c_max * (1.0 - tol_frac)
        bad = c > threshold
        if bad.sum() == 0:
            break
        bad_lags = np.where(bad)[0]
        bad_weights = (c[bad_lags] - threshold)
        if bad_weights.max() < 1e-15:
            break
        bad_weights = bad_weights / bad_weights.max()
        bw_full = np.zeros(2 * n - 1)
        bw_full[bad_lags] = bad_weights
        b_rev = bw_full[::-1].copy()
        conv_ab = _full_convolve(a, b_rev)
        # bad_contrib[i] = conv_ab[2n-2-i] for i=0..n-1
        # Need indices: 2n-2, 2n-3, ..., n-1
        lo = n - 1
        hi = 2 * n - 1
        bad_contrib = conv_ab[lo:hi][::-1]
        bc_max = bad_contrib.max()
        if bc_max < 1e-15:
            break
        bad_contrib = bad_contrib / bc_max
        a = a * (1.0 - alpha * bad_contrib)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.04):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    no_improve = 0
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _grad_active(a, c, n, tol_frac=0.002)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        # Backtracking line search
        step = lr / (1.0 + i * 0.0005)
        improved = False
        for attempt in range(15):
            z_trial = z - step * grad
            z_trial = np.clip(z_trial, -50.0, 50.0)
            z_trial = z_trial - z_trial.mean()
            a_trial = _softmax(z_trial)
            sc_trial = _score(a_trial)

                a = a_trial
                best_sc = sc_trial
                best_a = a.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            no_improve += 1
            lr *= 0.8
            if no_improve > 200:
                break
        else:
            no_improve = 0
            lr = min(lr * 1.02, 0.1)
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc

class _LbfgsProb:
    def __init__(self, beta, n):
        self.beta = beta
        self.n = n
    def fun(self, z):
        return _smoobj(z, self.beta, self.n)
    def jac(self, z):

def _lbfgs_one(a_init, beta, n, max_iter=3000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    prob = _LbfgsProb(beta, n)
    try:
        result = minimize(
            prob.fun, z0, jac=prob.jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-15,
                'gtol': 1e-13,
                'maxcor': 50,
                'maxls': 200,
                'disp': False,
            }
        )
        return _softmax(result.x)
    except Exception:
        return a_init

def _lbfgs_schedule(a_start, deadline, min_time_remaining=100.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start
    betas = max_c * np.array([0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006,

                              0.0001, 0.00005, 0.00002])

    current_sc = _score(current_a)
    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

            current_sc = sc
        else:
            # Try one more with slightly different beta
            pass
    return current_a

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        # Flat in middle, smooth taper at edges
        t = np.linspace(0, 1, n)
        edge = 0.15
        a = np.ones(n)
        for i in range(n):
            if t[i] < edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * t[i] / edge))
            elif t[i] > 1 - edge:
                a[i] = 0.5 * (1 + np.cos(np.pi * (t[i] - (1 - edge)) / edge))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

def _active_set_descent(a_init, deadline, max_steps=2000):

    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = 0.02

    for step_i in range(max_steps):

            break

        c = _full_convolve(a, a)
        c_max = c.max()

        tol_frac = max(0.0005, 0.01 - step_i * 0.000005)

        if active.sum() == 0:

        w = np.zeros_like(c)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        
        # Normalized gradient direction
        direction = grad / grad_norm
        
        # Backtracking line search
        step_size = lr
        improved = False
        for attempt in range(20):
            z_trial = z - step_size * direction
            z_trial = np.clip(z_trial, -50.0, 50.0)

            a_trial = _softmax(z_trial)

            if sc_trial < best_sc - 1e-12:

    sc_final = _score(a_final)
    if sc_final < best_sc:

    return best_a, best_sc

        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, uniform n={n_main}, score={_score(a0):.10f}")
        _upd(a0, _score(a0), "init")
    
    # Phase 1: PRIME updates for quick peak reduction
    print(f"\nPhase 1: PRIME (t={time.monotonic() - t_start:.1f}s)")
    a_p1, sc_p1 = _prime_update(a0, n_iter=400, alpha=0.04, tol_frac=0.003)
    _upd(a_p1, sc_p1, "prime")
    
    # Phase 2: Aggressive L-BFGS schedule
    print(f"\nPhase 2: L-BFGS schedule (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 250:
        a_p2 = _lbfgs_schedule(a_p1, deadline, min_time_remaining=250.0)
        _upd(a_p2, _score(a_p2), "lbfgs_main")
    
    # Phase 3: Active-set descent
    print(f"\nPhase 3: Active-set descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 200:
        a_p3, sc_p3 = _active_set_descent(best_seq, deadline - 200, max_steps=3000)
        _upd(a_p3, sc_p3, "active_set")
    
    # Phase 4: Subgradient polish
    print(f"\nPhase 4: Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_p4, sc_p4 = _subgrad_polish(best_seq, deadline - 150, n_max_steps=4000, lr_init=0.03)
        _upd(a_p4, sc_p4, "subgrad")
    
    # Phase 5: Second round - PRIME + L-BFGS from current best
    print(f"\nPhase 5: Second round (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 120:
        a_p5a, _ = _prime_update(best_seq, n_iter=200, alpha=0.02, tol_frac=0.002)
        _upd(a_p5a, _score(a_p5a), "prime2")
        
        if time.monotonic() < deadline - 80:
            a_p5b = _lbfgs_schedule(a_p5a, deadline, min_time_remaining=80.0)
            _upd(a_p5b, _score(a_p5b), "lbfgs2")
    
    # Phase 6: Multi-restart with diverse perturbations
    print(f"\nPhase 6: Multi-restart (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        n_restarts = min(4, max(1, int(remaining / 100)))
        
        strategies = [
            (0.03, 'gauss'),
            ('fourier3', None),
            (0.01, 'gauss'),
            ('fourier5', None),
        ]
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break
            
            if i == 0:
                p_start = _perturb(best_seq, 0.03, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3)
            elif i == 2:
                p_start = _perturb(best_seq, 0.01, rng)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            
            # Quick optimization from perturbed start
            a_r, _ = _prime_update(p_start, n_iter=100, alpha=0.03, tol_frac=0.003)
            _upd(a_r, _score(a_r), f"restart_{i}_prime")
            
            if time.monotonic() < deadline - 20:
                a_r2 = _lbfgs_schedule(a_r, deadline, min_time_remaining=20.0)
                _upd(a_r2, _score(a_r2), f"restart_{i}_lbfgs")
    
    # Phase 7: Try different length
    print(f"\nPhase 7: Different n (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 80:
        for n_try in [4096, 3072]:
            if time.monotonic() > deadline - 40:
                break
            for kind in ["flat_taper", "tapered"]:
                if time.monotonic() > deadline - 20:
                    break
                a_try = _make_profile(n_try, kind)
                a_try2, _ = _prime_update(a_try, n_iter=150, alpha=0.04, tol_frac=0.003)
                _upd(a_try2, _score(a_try2), f"n={n_try}_{kind}")
                
                if time.monotonic() < deadline - 20:
                    a_try3 = _lbfgs_schedule(a_try2, deadline, min_time_remaining=20.0)
                    _upd(a_try3, _score(a_try3), f"n={n_try}_{kind}_lbfgs")
    
    # Phase 8: Final fine optimization
    print(f"\nPhase 8: Final (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        
        # Very fine L-BFGS
        if time.monotonic() < deadline - 3:
            for beta_frac in [0.00003, 0.00001]:
                if time.monotonic() > deadline - 2:
                    break
                beta = mb * beta_frac
                a_fin = _lbfgs_one(best_seq, beta, n, max_iter=2000)
                _upd(a_fin, _score(a_fin), f"final_b{beta_frac}")
        
        # Final active-set descent
        if time.monotonic() < deadline - 1:
            a_fin2, sc_fin2 = _active_set_descent(best_seq, deadline - 1, max_steps=1000)
            _upd(a_fin2, sc_fin2, "final_descent")
    
    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
