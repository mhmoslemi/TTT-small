def _lse_momentum_descent(a_init, deadline, rng, max_wall_time=600.0,
                          beta_start=0.5, beta_end=50.0, momentum=0.9):
    """
    Mirror descent with momentum on the log-sum-exp smoothed objective.
    Works in the 'log' parameterization (z = log(a)).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))

    # Learning rate schedule
    base_lr = 0.8 / np.sqrt(n)

    for it in range(total_iters):
        if time.monotonic() > deadline - 2.0:
            break

        beta = beta_schedule[min(it, total_iters - 1)]

        a_curr = np.exp(z - z.max())
        a_curr /= a_curr.sum()

        c = _convolve_a(a_curr)
        c_shifted = beta * c - beta * c.max()

        eta = min(eta, 2.0 / (it + 1.0))

        z = z + m
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        # Check score periodically

            a_new /= a_new.sum()
            sc = _score_norm(a_new)

def _subgradient_waterfill(a_init, deadline, rng, max_wall_time=300.0):
    """
    Alternating subgradient method with water-filling:
    Targets the top-K lags simultaneously, adjusting step size to equalize them.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _convolve_a(a_init)
    max_c0 = float(c0.max())

    K = 5  # Number of top lags to consider
    total_iters = 2000

    for it in range(total_iters):
        if time.monotonic() > deadline - 2.0:
            break

        a_curr = np.exp(z - z.max())
        a_curr /= a_curr.sum()
        c = _convolve_a(a_curr)

        # Find top K lags
        top_idx = np.argsort(c)[-K:]
        top_vals = c[top_idx]
        max_val = top_vals[-1]
        min_top = top_vals[0]

        # Weight lags by how much they exceed the minimum top-K value
        # This encourages equalization
        spread = max_val - min_top + 1e-15
        w = np.zeros(len(c))
        for idx in top_idx:
            w[idx] = (c[idx] - min_top) / spread + 0.1

        w_sum = w.sum()
        w /= w_sum

        # Gradient
        a_rev = a_curr[::-1]
        corr = _fft_convolve(a_rev, w)[n - 1:2 * n - 1]
        grad = 4.0 * n * corr

        # Step size: larger when spread is large
        eta = 0.5 * spread / (np.linalg.norm(a_curr) + 1e-10)
        eta = min(eta, 0.3)

        # Update with projection
        z = z - eta * grad / a_curr
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if it % 5 == 0:
            a_new = np.exp(z - z.max())
            a_new /= a_new.sum()
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

    return best_a, best_score


def _local_search_hotspot(a_init, deadline, rng, n_trials=50000):
    """
    Aggressive local search: identify hotspot elements (those contributing
    most to the peak convolution) and redistribute mass to cold spots.
    Uses simulated annealing acceptance criterion.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)

    # Temperature schedule
    T_start = 0.01
    T_end = 0.0001
    total = n_trials

    for it in range(total):
        if time.monotonic() > deadline - 1.0:
            break

        frac = it / total
        T = T_start * (T_end / T_start) ** frac

        c = _convolve_a(a)
        max_lag = int(np.argmax(c))
        max_val = c[max_lag]

        # Compute "heat" for each element: contribution to top lags
        K = min(3, len(c))
        top_lags = np.argsort(c)[-K:]

        heat = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            lo = max(0, kl - (n - 1))
            hi = min(n - 1, kl)
            for j in range(lo, hi + 1):
                partner = kl - j
                if 0 <= partner < n:
                    heat[j] += c[kl] * a[partner]

        # Normalize heat
        heat_max = heat.max()
        if heat_max < 1e-15:
            heat = np.ones(n) / n
        else:
            heat /= heat_max

        # Strategy selection
        strat = it % 4

        if strat == 0:
            # Move mass from hottest to coldest element
            hot_idx = int(np.argmax(heat))
            cold_candidates = np.argsort(heat)[:max(1, n // 10)]
            cold_idx = int(cold_candidates[rng.integers(len(cold_candidates))])
            if hot_idx == cold_idx:
                continue
            frac_move = rng.uniform(0.05, 0.3)
            amt = a[hot_idx] * frac_move
            a_trial = a.copy()
            a_trial[hot_idx] -= amt
            a_trial[cold_idx] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm(a_trial)
            # Simulated annealing acceptance
            if sc < best_sc or rng.random() < np.exp(-(sc - best_sc) / max(T, 1e-10)):
                a = a_trial.copy()
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()

        elif strat == 1:
            # Spread mass from hot block to cold block
            block_size = int(rng.integers(4, min(32, n // 16)))
            hot_start = int(rng.integers(0, n - block_size))
            # Weight by heat
            hot_weights = heat[hot_start:hot_start + block_size]
            if hot_weights.sum() < 1e-10:
                continue
            cold_start = int(rng.integers(0, n - block_size))
            if cold_start == hot_start:
                continue
            amt = a[hot_start] * rng.uniform(0.1, 0.3)
            a_trial = a.copy()
            a_trial[hot_start] -= amt
            a_trial[cold_start] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm(a_trial)
            if sc < best_sc or rng.random() < np.exp(-(sc - best_sc) / max(T, 1e-10)):
                a = a_trial.copy()
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()

        elif strat == 2:
            # Multiplicative perturbation of a random block
            block_size = int(rng.integers(3, min(20, n // 10)))
            start = int(rng.integers(0, n - block_size))
            mult = rng.uniform(0.8, 1.2, size=block_size)
            a_trial = a.copy()
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc or rng.random() < np.exp(-(sc - best_sc) / max(T, 1e-10)):
                a = a_trial.copy()
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()

        else:
            # Pair swap: exchange mass between two positions
            i1 = int(np.argmax(heat))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            frac = rng.uniform(0.1, 0.4)
            amt1 = a[i1] * frac
            amt2 = a[i2] * frac
            a_trial = a.copy()
            a_trial[i1] = max(0.0, a_trial[i1] - amt1 + amt2)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2 + amt1)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm(a_trial)
            if sc < best_sc or rng.random() < np.exp(-(sc - best_sc) / max(T, 1e-10)):
                a = a_trial.copy()
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()

    return best_a, best_sc


def _make_ramp_profile(n, p=1.0, reverse=False):
    """Power-law ramp: a[i] ~ (i+1)^p or (n-i)^p."""
    i = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        i = n + 1 - i
    a = i ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asym_triangle(n, alpha=1.0, beta=1.0):
    """Asymmetric triangle: a[i] ~ (i+1)^alpha * (n-i)^beta."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** alpha * (n - i) ** beta
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp_profile(n, rate=0.5, reverse=False):
    """Exponential profile."""
    i = np.arange(n, dtype=np.float64)
    if reverse:
        i = n - 1 - i
    a = np.exp(rate * i / n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell_profile(n, center=0.3, width=0.2):
    """Gaussian bell at non-center position."""
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-(t - center) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabola_profile(n, curv=1.0, shift=0.0):
    """Parabolic profile with shift."""
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 - curv * (t - shift) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_cosine_asym(n, freq=1.0, amp=0.3, phase=0.0):
    """Asymmetric cosine modulation."""
    t = np.linspace(0.0, 1.0, n)
    env = t  # Linear envelope
    a = env * (1.0 + amp * np.cos(2.0 * np.pi * freq * t + phase))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sawtooth(n, periods=2):
    """Sawtooth wave."""
    t = np.linspace(0.0, 1.0, n)
    a = (t * periods) % 1.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _dual_gap(a, tol=1e-6):
    """
    Compute a lower bound on the optimal max_b using dual variables.
    If we find w >= 0, sum(w) = 1 such that sum_k w[k] * b[k] >= LB,
    then max_b >= LB.
    Simple approach: use the softmax weights as dual variables.
    """
    n = len(a)
    c = _convolve_a(a)
    S = a.sum()
    # The dual bound: for any probability vector w on lags,
    # max_b >= sum_k w[k] * b[k]
    # Best "simple" bound: w[k] = 1/(2n-1) for all k gives average = S^2/(2n-1)
    avg_bound = S * S / (2 * n - 1)
    # Score lower bound: 2n * avg_bound / S^2 = 2n / (2n-1)
    return 2.0 * n / (2 * n - 1)

    a2, sc2 = _lse_momentum_descent(a1, dl2, rng,
                                     max_wall_time=phase2_time,
                                     beta_start=5.0, beta_end=80.0,
                                     momentum=0.85)

    return a2, sc2


def _try_multiple_n(a_best, score_best, deadline, rng):
    """
    Try interpolating the best sequence to different lengths and optimizing.
    """
    results = []
    n_curr = len(a_best)

    # Try smaller n (might have better discreteness effects)
    for target_n in [n_curr // 2, n_curr // 4]:
        if target_n < 256:
            continue
        if time.monotonic() > deadline - 30:
            break
        p = _interp_profile(a_best, target_n)
        sc = _score_norm(p)
        results.append((sc, p, target_n))
        if sc < score_best:
            print(f"  Better at n={target_n}: {sc:.10f}")

    # Try larger n
    for target_n in [n_curr * 2]:
        if target_n > 32768:
            continue
        if time.monotonic() > deadline - 30:
            break
        p = _interp_profile(a_best, target_n)
        sc = _score_norm(p)
        results.append((sc, p, target_n))
        if sc < score_best:
            print(f"  Better at n={target_n}: {sc:.10f}")

    return results


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer: momentum mirror descent, water-filling subgradient,
    hotspot local search with simulated annealing, and diverse asymmetric
    initializations. Targets improvement over 1.505927.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent sequence ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_asym_triangle(n_parent, 1.0, 1.0)
        sc_parent = _score_norm(parent)
        print(f"Default triangular: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "default")

    n_work = n_parent
    print(f"Working n={n_work}, target upper bound: 1.5027")

    # --- Phase 0: Diverse asymmetric initializations (60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    print("Phase 0: Diverse initialization")

    profiles = []
    # Ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_make_ramp_profile(n_work, p, reverse=False))
        profiles.append(_make_ramp_profile(n_work, p, reverse=True))
    # Asymmetric triangles
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        for beta in [0.5, 1.0, 1.5, 2.0]:
            if alpha != beta:  # Only truly asymmetric
                profiles.append(_make_asym_triangle(n_work, alpha, beta))
    # Exponentials
    for rate in [0.3, 0.5, 1.0, 2.0]:
        profiles.append(_make_exp_profile(n_work, rate, reverse=False))
        profiles.append(_make_exp_profile(n_work, rate, reverse=True))
    # Off-center bells
    for center in [0.2, 0.3, 0.4, 0.6, 0.7, 0.8]:
        for width in [0.1, 0.15, 0.2, 0.3]:
            profiles.append(_make_bell_profile(n_work, center, width))
    # Parabolic
    for curv in [0.5, 1.0, 1.5, 2.0]:
        for shift in [0.3, 0.5, 0.7]:
            profiles.append(_make_parabola_profile(n_work, curv, shift))
    # Sawtooth
    for periods in [1, 2, 3, 5]:
        profiles.append(_make_sawtooth(n_work, periods))
    # Asymmetric cosine
    for amp in [0.2, 0.4, 0.6]:
        profiles.append(_make_cosine_asym(n_work, 1.0, amp, 0.0))
        profiles.append(_make_cosine_asym(n_work, 2.0, amp, np.pi / 2))

    # Also include parent
    profiles.append(parent.copy())

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        if sc < best_score:
            update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    top_candidates = [c[1] for c in candidates[:5]]
