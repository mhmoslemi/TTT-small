"""
Multi-scale FISTA optimization with adaptive log-sum-exp smoothing,
LP-based peak shaving, and diverse restarts for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max(a):
    c = _full_convolve(a, a)
    return float(c.max()), c


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _soft_max_and_weights(c, beta):
    """Compute soft max of c with temperature beta, returns (soft_max_value, weights)."""
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c.max(), w
    w = e / s
    sm = c.max() + beta * np.log(s)
    return float(sm), w


def _gradient_smoothed(a, n, beta, c=None):
    """
    Gradient of (1/beta)*log(sum(exp(beta*c_k))) with respect to a.
    c_k = conv(a,a)_k
    grad = sum_k w_k * 2*a[k-m] (reversed and shifted)
    """
    if c is None:
        c = _full_convolve(a, a)
    _, w = _soft_max_and_weights(c, beta)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # The gradient of sum_k w_k * c_k with respect to a is:
    # For each k, grad_k[m] = 2*a[k-m] if valid
    # So sum_k w_k * grad_k = 2 * conv(w, a_rev)[n-1:2n-1]
    h = wc[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - (w @ c))  # This is for the full objective with normalization
    # Actually let's just return the raw gradient of soft_max(c(a))
    # d(soft_max)/da = sum_k w_k * dc_k/da = sum_k w_k * 2 * a[k-m]
    # = 2 * (w conv a_rev)[n-1 : 2n-1]
    raw_grad = 2.0 * (wc[n - 1:2 * n - 1])
    return raw_grad, w, c


def _fista_optimize(a_init, deadline, beta_start=5.0, beta_end=200.0,
                    lr=0.05, max_betas=8):
    """
    FISTA optimization for minimizing soft_max(conv(a,a)) with log-sum-exp smoothing.
    Uses variable temperature (beta) annealing.
    """
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s  # normalize to sum=1

    best_a = a.copy()
    best_score = _conv_max(a)[0]

    # Compute initial score for reference
    c_init = _full_convolve(a, a)
    M_init = float(c_init.max())

    # FISTA variables
    y = a.copy()
    t = 1.0

    # Beta schedule
    n_betas = max_betas
    betas = np.geomspace(beta_start, beta_end, n_betas)

    total_iter = 0
    lr_decay = 0.9999

    for beta in betas:
        if time.monotonic() > deadline:
            break
        # Reset FISTA momentum at each beta change
        y = a.copy()
        t = 1.0

        iter_count = 0
        max_iter_per_beta = 5000
        while iter_count < max_iter_per_beta:
            if time.monotonic() > deadline:
                break
            iter_count += 1
            total_iter += 1

            c = _full_convolve(y, y)
            sm, w = _soft_max_and_weights(c, beta)

            # Gradient of soft_max with respect to y
            a_rev = y[::-1]
            wc = _full_convolve(w, a_rev)
            grad = 2.0 * wc[n - 1:2 * n - 1]

            # Proximal step (project onto sum=1, non-negative)
            step = lr * grad
            y_new = y - step
            # Project onto simplex-like constraint: y >= 0, sum(y) = 1
            # Simple: clip and renormalize
            y_new = np.maximum(y_new, 0.0)
            sn = y_new.sum()
            if sn < 1e-15:
                y_new = y.copy()
            else:
                y_new = y_new / sn

            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            a_new = y_new + ((t - 1.0) / t_new) * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            sa = a_new.sum()
            if sa < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / sa

            y = y_new
            a = a_new
            t = t_new

            # Track best
            if iter_count % 100 == 0:
                cm = float(_full_convolve(a, a).max())
                if cm < best_score:
                    best_score = cm
                    best_a = a.copy()

            # Adaptive learning rate
            if iter_count % 2000 == 0:
                lr *= lr_decay

    # Final check
    cm = float(_full_convolve(a, a).max())
    if cm < best_score:
        best_score = cm
        best_a = a.copy()

    return best_a, best_score

            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t_step % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()

            if iters % 10000 == 0:
                current_lr = max(0.001, current_lr * 0.93)

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _build_lp_matrix(f, n):
    """Build sparse constraint matrix A where A[k,j] = f[k-j] if valid."""
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_peak_shave(f, deadline, max_steps=30):
    """LP-based peak shaving: find direction that reduces max(conv) while maintaining feasibility."""
    n = len(f)
    f = f / f.sum()
    best_f = f.copy()
    best_M = _conv_max(f)[0]

    A_cache = None
    f_for_matrix = None
    stuck = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(f, f)
        M = float(c.max())

        if A_cache is None or f_for_matrix is None or np.max(np.abs(f - f_for_matrix)) > 1e-8:
            A_cache = _build_lp_matrix(f, n)
            f_for_matrix = f.copy()

        # LP: max sum(g) s.t. conv(f, g) <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds_lp = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A_cache, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'time_limit': 20, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            else:
                g = None
        except Exception:
            g = None

        if g is None or g.sum() < 1e-15:
            stuck += 1
            if stuck > 10:
                break
            continue

        stuck = 0
        g_norm = g / g.sum()  # normalize to sum=1

        # Line search
        improved = False
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g_norm
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            M_new = _conv_max(f_new)[0]
            if M_new < M * (1 - 1e-10):
                f = f_new
                if M_new < best_M:
                    best_M = M_new
                    best_f = f.copy()
                improved = True
                break

        if not improved:
            stuck += 1
            if stuck > 10:
                break

    return best_f, best_M


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_block(a, rng, block_size=64):
    """Perturb by randomly swapping blocks of elements."""
    a = a.copy()
    n = len(a)
    n_blocks = n // block_size
    if n_blocks < 2:
        block_size = max(1, n // 16)
        n_blocks = n // block_size
    if n_blocks < 2:
        return a
    idx = rng.permutation(n_blocks)
    for i in range(min(4, n_blocks)):
        b1 = idx[i]
        b2 = idx[(i + 1) % n_blocks]
        s1, e1 = b1 * block_size, (b1 + 1) * block_size
        s2, e2 = b2 * block_size, (b2 + 1) * block_size
        a[s1:e1], a[s2:e2] = a[s2:e2].copy(), a[s1:e1].copy()
    return a / a.sum()


def _perturb_spike(a, rng, n_spikes=10):
    """Add random spikes to the sequence."""
    a = a.copy()
    n = len(a)
    for _ in range(n_spikes):
        pos = rng.integers(0, n)
        a[pos] += 0.1 * rng.random()
    return a / a.sum()


def _make_raised_cosine(n):
    """Raised cosine window."""
    x = np.linspace(0, np.pi, n)
    w = 0.5 * (1 + np.cos(x))
    return w / w.sum()


def _make_exponential(n):
    """Exponential decay window."""
    w = np.exp(-np.linspace(0, 3, n))
    return w / w.sum()


def _make_uniform(n):
    return np.ones(n) / n


def _subgradient_descent(a_init, deadline, lr=0.01, momentum=0.9):
    """Subgradient descent with the exact max (not smoothed) and momentum."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_M = _conv_max(a)[0]

    v = np.zeros(n)
    t = 0
    step_decay = 0.99995

    while time.monotonic() < deadline:
        t += 1
        c = _full_convolve(a, a)
        M = float(c.max())

        # Find positions near the max (within 2% of max)
        threshold = 0.98 * M
        mask = c >= threshold
        w = c[mask]
        w = w / w.sum()
        idx = np.where(mask)[0]

        # Subgradient: weighted average of gradients at peak positions
        grad = np.zeros(n)
        for wi, ki in zip(w, idx):
            # grad_k[m] = 2*a[k-m] if valid
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            for m in range(lo, hi):
                km = ki - m
                if 0 <= km < n:
                    grad[m] += 2.0 * wi * a[km]

        # Use FFT for efficiency
        a_rev = a[::-1]
        # We need weighted sum of shifted a
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a
        else:
            a_new = a_new / sn

        a = a_new
        current_lr = lr * (step_decay ** (t // 1000))

        if t % 50 == 0:
            cm = float(_full_convolve(a, a).max())
            if cm < best_M:
                best_M = cm
                best_a = a.copy()

    return best_a, best_M


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale FISTA + Adam + LP hybrid optimization for minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(a0)
    else:
        n_main = 8192
        a0 = _make_uniform(n_main)
        print(f"No parent, using uniform n={n_main}")
        update_best(a0)

    # === Phase 1: FISTA optimization from parent (0 - 500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: FISTA (n={n_main}) ===")

    if time.monotonic() < phase1_end - 30:
        a_fista, _ = _fista_optimize(
            a0, phase1_end - 20,
            beta_start=3.0, beta_end=100.0,
            lr=0.03, max_betas=6
        )
        update_best(a_fista)

        # Also try from a structured initialization
        a_rc = _make_raised_cosine(n_main)
        if _conv_max(a_rc)[0] < _conv_max(a0)[0]:
            print("  Raised cosine is better initial!")
            update_best(a_rc)
            a_fista2, _ = _fista_optimize(
                a_rc, min(deadline, phase1_end + 100.0),
                beta_start=3.0, beta_end=100.0,
                lr=0.03, max_betas=6
            )
            update_best(a_fista2)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam fine-tuning (500s - 900s) ===
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: Adam fine-tuning ===")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        remaining = phase2_end - time.monotonic() - 20
        if remaining > 60:
            # Split time between two Adam runs
            a_adam1, _ = _adam_optimize(
                a_best_norm, time.monotonic() + remaining * 0.5, lr=0.008
            )
            update_best(a_adam1)

            if time.monotonic() < phase2_end - 30:
                remaining2 = phase2_end - time.monotonic() - 10
                if remaining2 > 30:
                    a_adam2, _ = _adam_optimize(
                        a_adam1, time.monotonic() + remaining2, lr=0.004
                    )
                    update_best(a_adam2)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: LP-based peak shaving (900s - 1200s) ===
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\n=== Phase 3: LP peak shaving ===")

    if best_seq is not None and time.monotonic() < phase3_end - 60:
        a_lp = best_seq / best_seq.sum()
        a_lp_shaved, M_shaved = _lp_peak_shave(

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse perturbations (1200s - 1500s) ===
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 4: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        perturbations = [

            (None, _perturb_block),

        n_restarts = len(perturbations)

            if deadline_i <= time.monotonic():
                break

            # Alternate between Adam and FISTA
            if i % 2 == 0:
                a_r, _ = _adam_optimize(p_start, deadline_i, lr=0.006)
            else:

    # === Phase 5: Multi-n exploration (1500s - 1700s) ===
    phase5_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 5: Multi-n exploration ===")

    if best_seq is not None and time.monotonic() < phase5_end - 60:
        # Try different n values
        current_n = len(best_seq)
        target_ns = sorted(set([4096, 8192, 16384]) - {current_n})
        target_ns = [tn for tn in target_ns if tn <= 32768]

        for target_n in target_ns[:2]:
            if time.monotonic() > phase5_end - 80:

            p_interp = _interp_profile(best_seq, target_n)
