"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA (coarse-to-fine beta), L-BFGS in log-space (local refinement),
and peak-equalization (active set balancing) for final polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_conv(a, w, n):
    """Compute gradient of weighted sum of convolution entries.
    g[i] = (2/beta) * sum_k w[k] * a[k-i] (for valid k-i)
    Here w is already normalized (sums to 1) and beta is assumed to be 1.
    So we compute: g[i] = 2 * sum_k w[k] * a[k-i]
    """
    # Cross-correlation of a with w
    N = _next_pow2(n + len(w) - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # ifft(conj(A) * W) gives sum_m a[m] * w[m+i mod N]
    # which is what we want: g[i] = sum_m a[m] * w[m+i]
    g = np.fft.irfft(np.conj(A) * W, N)[:n]
    return 2.0 * g

def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Lipschitz constant: L = (2/beta) * ||H_max||_2
    # H_max has eigenvalues bounded by 2 (it's 2*permutation matrix)
    # So L = (2/beta) * 2 = 4/beta
    L = 4.0 / beta
    step = 1.0 / L
    
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        
        # Softmax weights
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _grad_conv(y, w, n)
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0 or it < 3:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()

Improved optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _grad_conv(a, w):
    """g[i] = 2 * sum_k w[k] * a[k-i] for valid k-i in [0, n).
    Computed via FFT cross-correlation.
    """
    n = len(a)
    N = _next_pow2(n + len(w) - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # ifft(conj(A) * W)[i] = sum_m a[m] * w[(m+i) mod N]
    # For N >= n + len(w) - 1, no wrap-around for m+i < n + len(w) - 1
    g = np.fft.irfft(np.conj(A) * W, N)[:n]
    return 2.0 * g
