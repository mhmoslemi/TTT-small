def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _solve_lp(f, solver=cp.CLARABEL):
    """
    Solve: min t s.t. conv(f, g) <= t, sum(g)=1, g>=0
    Returns (g_opt, t_opt) or (None, None) on failure.
    """
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    
    # Build convolution matrix A where A @ g = conv(f, g)
    # A is (2n-1) x n, A[k, i] = f[i] if 0 <= k-i < n else 0
    # Equivalently: row k has f at positions max(0,k-n+1) to min(n-1,k)
    
    # Build sparse matrix
    row_idx = []
    col_idx = []
    data = []
    
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            row_idx.extend([k] * (i_hi - i_lo + 1))
            col_idx.extend(range(i_lo, i_hi + 1))
            data.extend(f[i_lo:i_hi+1])
    
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver)
        if g.value is not None and t.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None

def _line_search_step(a, g, n_steps=10):
    """Find best step size t in (0,1] for a_new = (1-t)*a + t*g"""
    n = len(a)
    best_t = 0.3
    best_score = _score_norm(a)
    for t in np.linspace(0.05, 1.0, n_steps):
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t, best_score

def _lp_alternate(a, max_time, verbose=True):

    # A is m x n

    # Row 1: A[1, j] = f[1-j]. j=0 gives f[1], j=1 gives f[0]. So row 1 = [f[1], f[0], 0, ...]

    # ...

"""
LP-based iterated minimax with adaptive line search for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses full-resolution LP at the parent's n,
line search for step adaptation, periodic exploration, and coordinate
descent polish. Optionally explores larger n with gradient methods.
"""
import numpy as np
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _build_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    j_vals = np.repeat(np.arange(n, dtype=np.int32), n)
    i_vals = np.tile(np.arange(n, dtype=np.int32), n)
    row_idx = (j_vals + i_vals).astype(np.int32)
    col_idx = j_vals
    data = f.astype(np.float64)[i_vals]
    return csr_matrix((data, (row_idx, col_idx)), shape=(m, n))


def _solve_lp(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None and t.value is not None:
            g_val = np.maximum(g.value, 0.0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _line_search(a, g):
    n = len(a)
    best_t = 0.3
    best_score = _score_norm(a)
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t, best_score


def _lp_alternate(a, max_time, verbose=True):
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < max_time:
        it += 1
        g, t_opt = _solve_lp(a)
        if g is None:
            break
        
        t_step, sc_new = _line_search(a, g)
        
        if sc_new < best_score - 1e-12:
            a_new = (1.0 - t_step) * a + t_step * g
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new / a_new.sum()
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
        
        if verbose and it % 5 == 0:

            print(f"  LP iter {it}: score={_score_norm(a):.10f}, "
                  f"best={best_score:.10f}, t_opt={t_opt:.8f}, "
                  f"step={t_step:.3f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n, f_arr):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_optimize(a_init, deadline, beta_list=None):
    from scipy.optimize import minimize
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        beta_list = [b * max_c0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        time_per = min(remaining / 3.0, 30.0)
        maxiter = max(100, int(time_per * 20))
        
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n, None),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 30, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _coord_descent_polish(a, max_time):
    t0 = time.monotonic()
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    c = _full_convolve(a, a)
    max_c = c.max()
    # Find active lags
    threshold = 0.9 * max_c
    active = np.where(c >= threshold)[0]
    
    # Compute impact of each coordinate
    impact = np.zeros(n)
    for k in active:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        impact[j_lo:j_hi+1] += c[k]
    
    order = np.argsort(-impact)[:500]
    
    for rnd in range(200):
        if time.monotonic() - t0 > max_time:
            break
        improved = False
        for j in order:
            if time.monotonic() - t0 > max_time:
                break
            cur_val = a[j]
            for frac in [0.5, 0.7, 0.8, 1.2, 1.5, 2.0]:
                a[j] = cur_val * frac
                s = a.sum()
                if s < 1e-15:
                    a[j] = cur_val
                    continue
                a_norm = a / s
                sc = _score_norm(a_norm)
                if sc < best_score:
                    best_score = sc
                    best_a = a_norm.copy()
                    a = a_norm
                    improved = True
                    break
                else:
                    a[j] = cur_val
        if not improved:
            break
    
    return best_a, best_score
