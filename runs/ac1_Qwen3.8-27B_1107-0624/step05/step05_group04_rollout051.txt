def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):

def _subgradient_step(a, deadline, max_iter=5000, top_k=5, init_alpha=None):

    n = len(a)

    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    if init_alpha is None:
        init_alpha = 0.5 / best_score
    
    alpha = init_alpha
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        g, c = _subgradient(a, top_k)
        cm = float(c.max())
        # Normalize a to sum=1
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Step
        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        sc_new = _score_norm_fn(a_new)
        if np.isfinite(sc_new) and sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new.copy()
        
        # Polyak step size
        gap = cm / n - lower_bound / (2.0 * n)  # approximate gap

def _coordinate_descent(a, deadline, max_rounds=2000):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a = a / a.sum()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Cycle through a subset of coordinates
        step = max(1, n // 200)
        for j in range(0, n, step):
            if time.monotonic() > deadline:
                break
            # Optimize a[j] in 1D
            # c[k] depends on a[j] linearly (or quadratically if k=2j)
            old_val = a[j]
            # Try a few values around current
            for trial_frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:

            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                g_a[j_lo:j_hi + 1] += 2.0 * w_k * a[k - j_lo:j_hi + 1]
        
        offset = float(np.dot(a, g_a))
        grad_z = a * (g_a - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        a = _softmax_fn(z)
        
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)
        
        if t % 20000 == 0:
            z = z + 0.05 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    profiles = []
    # Ramps
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Quadratic
    for s, c in [(1.0, 0.5), (2.0, -0.3), (1.5, 1.0), (2.5, 0.3), (3.0, -0.5)]:
        a = 1.0 + s * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0, 8.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Power law
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = (t + 0.01) ** alpha
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def _diversified_init(n, rng, num_trials=10):
    """Try many initializations and return the best."""
    best_a = None
    best_sc = np.inf
    profiles = _make_profiles(n)
    rng_profiles = []
    for _ in range(num_trials):
        a = np.abs(rng.normal(size=n)) + 0.1
        rng_profiles.append(a / a.sum())
    
    all_profiles = profiles[:6] + rng_profiles
    for a in all_profiles:
        sc = _score_norm_fn(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s", flush=True)
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}", flush=True)
        update_best(parent, sc_p, "parent")
    
    # === Phase 1: Multi-scale optimization (500s) ===
    phase1_end = t_start + 500.0
    print(f"\n=== Phase 1: Multi-scale (deadline t+500s) ===", flush=True)
    
    # Start at n=64 with diverse initialization
    n0 = 64
    a_cur, sc_cur = _diversified_init(n0, rng, num_trials=8)
    print(f"  n={n0} init: score={sc_cur:.6f}", flush=True)
    
    # Quick L-BFGS at n=64
    dl = min(phase1_end, t_start + 15.0)
    a_cur, sc_cur = _run_lbfgs_continuation(a_cur, dl, rng, n_betas=6)
    print(f"  n={n0} after L-BFGS: score={sc_cur:.6f}", flush=True)
    
    # Try subgradient at n=64
    dl = min(phase1_end, t_start + 30.0)
    a_sg, sc_sg = _subgradient_step(a_cur, dl, max_iter=2000, top_k=3)
    if sc_sg < sc_cur:
        a_cur, sc_cur = a_sg, sc_sg
        print(f"  n={n0} after subgrad: score={sc_cur:.6f}", flush=True)
    
    # Multi-scale up
    scales = [128, 256, 512, 1024, 2048, 4096]
    time_alloc = {128: 20, 256: 30, 512: 50, 1024: 80, 2048: 120, 4096: 150}
    
    for target_n in scales:
        if time.monotonic() > phase1_end - 10:
            break
        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)
        print(f"  Upsample to n={target_n}: score={sc_target:.6f}", flush=True)
        
        alloc = time_alloc.get(target_n, 40)
        opt_dl = min(phase1_end - 5, time.monotonic() + alloc)
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_dl, rng, n_betas=10)
        print(f"  n={target_n} after L-BFGS: score={sc_opt:.6f}", flush=True)
        
        # Subgradient refinement
        if time.monotonic() < phase1_end - 15:
            sg_dl = min(phase1_end - 5, time.monotonic() + 15.0)
            a_sg2, sc_sg2 = _subgradient_step(a_opt, sg_dl, max_iter=1000, top_k=5)
            if sc_sg2 < sc_opt:
                a_opt, sc_opt = a_sg2, sc_sg2
                print(f"  n={target_n} after subgrad: score={sc_opt:.6f}", flush=True)
        
        if sc_opt < sc_cur:
            a_cur, sc_cur = a_opt, sc_opt
    
    if best_seq is not None and best_score < sc_cur:
        a_cur = best_seq.copy()
        sc_cur = best_score
    else:
        update_best(a_cur, sc_cur, "multi_scale_p1")
    
    print(f"  Phase 1 done: best={best_score:.10f}", flush=True)
    
    # === Phase 2: Refine parent (300s) ===
    phase2_end = min(t_start + 800.0, deadline)
    print(f"\n=== Phase 2: Refine parent (deadline t+800s) ===", flush=True)
    
    if parent is not None and time.monotonic() < phase2_end - 20:
        n_p = len(parent)
        print(f"  Parent n={n_p}, score={_score_norm_fn(parent):.10f}", flush=True)
        
        # L-BFGS on parent
        dl = min(phase2_end - 60, time.monotonic() + 120.0)
        a_p2, sc_p2 = _run_lbfgs_continuation(parent, dl, rng, n_betas=15)
        update_best(a_p2, sc_p2, "parent_lbfgs")
        
        # Subgradient on parent
        if time.monotonic() < phase2_end - 30:
            sg_dl = min(phase2_end - 10, time.monotonic() + 60.0)
            a_sg_p, sc_sg_p = _subgradient_step(parent, sg_dl, max_iter=3000, top_k=7)
            update_best(a_sg_p, sc_sg_p, "parent_subgrad")
        
        # Adam GD
        if time.monotonic() < phase2_end - 20:
            gd_dl = min(phase2_end - 5, time.monotonic() + 80.0)
            a_gd_p, sc_gd_p = _adam_gd(parent, gd_dl, rng, lr_init=0.03)
            update_best(a_gd_p, sc_gd_p, "parent_adam")
    
    # If best from Phase 1 is better, refine that instead
    if best_seq is not None and len(best_seq) != len(parent if parent is not None else []):
        if time.monotonic() < phase2_end - 20:
            n_b = len(best_seq)
            dl = min(phase2_end - 5, time.monotonic() + 100.0)
            a_r, sc_r = _run_lbfgs_continuation(best_seq, dl, rng, n_betas=12)
            update_best(a_r, sc_r, "phase1_refined")
    
    print(f"  Phase 2 done: best={best_score:.10f}", flush=True)
    
    # === Phase 3: Intensive final polish (remaining time) ===
    print(f"\n=== Phase 3: Final polish (deadline t+{budget_s - 30:.0f}s) ===", flush=True)
    
    if best_seq is None:
        # Fallback
        a = _make_profiles(256)[2]
        return [float(x) for x in a]
    
    n_best = len(best_seq)
    print(f"  Working on n={n_best}, current best={best_score:.10f}", flush=True)
    
    # Alternating refinement loop
    remaining_budget = deadline - time.monotonic()
    n_cycles = max(2, int(remaining_budget / 60))
    cycle_budget = remaining_budget / n_cycles
    
    for cycle in range(n_cycles):
        if time.monotonic() > deadline - 10:
            break
        
        print(f"\n  Cycle {cycle+1}/{n_cycles} (budget {cycle_budget:.0f}s)", flush=True)
        dl = time.monotonic() + cycle_budget * 0.9
        
        # L-BFGS with small beta
        if time.monotonic() < dl * 0.6:
            dl1 = min(time.monotonic() + cycle_budget * 0.4, deadline - 5)
            a_l, sc_l = _run_lbfgs_continuation(best_seq, dl1, rng, n_betas=12)
            update_best(a_l, sc_l, f"cycle{cycle}_lbfgs")
        
        # Subgradient
        if time.monotonic() < dl * 0.8:
            dl2 = min(time.monotonic() + cycle_budget * 0.25, deadline - 5)
            a_s, sc_s = _subgradient_step(best_seq, dl2, max_iter=2000, top_k=7)
            update_best(a_s, sc_s, f"cycle{cycle}_subgrad")
        
        # Adam GD
        if time.monotonic() < dl:
            dl3 = min(time.monotonic() + cycle_budget * 0.2, deadline - 5)
            a_g, sc_g = _adam_gd(best_seq, dl3, rng, lr_init=0.02)
            update_best(a_g, sc_g, f"cycle{cycle}_adam")
        
        # Peak surgery
        if time.monotonic() < deadline - 15:
            dl4 = min(time.monotonic() + cycle_budget * 0.15, deadline - 5)
            a_p, sc_p = _peak_surgery(best_seq, dl4, max_rounds=200)
            update_best(a_p, sc_p, f"cycle{cycle}_surgery")
        
        # Random restart from best
        if cycle > 0 and rng.random() < 0.3:
            z = np.log(np.maximum(best_seq, 1e-15))
            noise_scale = 0.05 + 0.03 * cycle
            z = z + noise_scale * rng.normal(size=n_best)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            if time.monotonic() < deadline - 30:
                dl5 = min(time.monotonic() + 30.0, deadline - 5)
                a_r, sc_r = _run_lbfgs_continuation(a_pert, dl5, rng, n_betas=8)
                update_best(a_r, sc_r, f"cycle{cycle}_restart")
        
        print(f"  End cycle {cycle+1}: best={best_score:.10f}", flush=True)
    
    # === Phase 4: Tiny-beta final L-BFGS ===
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final tiny-beta L-BFGS ===", flush=True)
        n_f = len(best_seq)
        c_f = _full_convolve(best_seq, best_seq)
        max_c_f = float(c_f.max())
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = 1e-6 * max_c_f
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny, n_f),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 50000, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxls': 50, 'maxfun': 100000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Tiny-beta: {best_score:.10f}", flush=True)
        except Exception as ex:
            print(f"  Tiny-beta failed: {ex}", flush=True)
    
    # === Final quick peak surgery ===
    if time.monotonic() < deadline - 5:
        dl_final = time.monotonic() + min(20.0, max(1.0, deadline - time.monotonic()))
        a_fs, sc_fs = _peak_surgery(best_seq, dl_final, max_rounds=100)
        update_best(a_fs, sc_fs, "final_surgery")
    
    # === Return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s * np.sqrt(2.0 * len(final_seq))
        # Scale to ensure sum > 0.01
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        s2 = float(final_seq.sum())
        if s2 < 0.01:
            final_seq = final_seq * (1.0 / s2)
        
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}", flush=True)
        return [float(x) for x in final_seq]
    
    # Fallback
    t = np.linspace(0, 1, 256)
    fallback = (1.0 + 1.5 * t)
    fallback = fallback / fallback.sum() * np.sqrt(512.0)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
