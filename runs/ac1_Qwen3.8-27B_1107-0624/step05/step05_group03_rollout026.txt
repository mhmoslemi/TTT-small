"""
Optimizes 2*n*max(conv(a,a))/sum(a)^2 using Nesterov-accelerated gradient descent
with Lp-norm continuation, FFT-based gradient computation, and direct peak shaving.
Key innovations: O(n log n) gradient via FFT convolution, adaptive Lp schedule,
and active-set peak manipulation for final polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return np.maximum(r, 0.0)


def _score_norm(a):
    """Score assuming sum(a)=1: returns 2n*max_conv."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _lp_norm(c, p):
    """Compute Lp norm of c safely."""
    if p == np.inf:
        return float(c.max())
    if p < 100:
        return float((np.sum(c ** p)) ** (1.0 / p))
    else:
        log_c = np.log(np.maximum(c, 1e-300))
        lc_max = log_c.max()
        log_sum = lc_max + np.log(np.sum(np.exp(p * (log_c - lc_max))))
        return float(np.exp(log_sum / p))


def _lp_grad_a(c, a, p):
    """
    Compute gradient of 2n * Lp(conv(a,a), p) with respect to a.
    Uses FFT for efficiency: grad_a[i] = 4n * Lp^(1/p - 1) * (w * a_rev)[n-1-i]
    where w[k] = c[k]^(p-1).
    """
    n = len(a)
    lp_val = _lp_norm(c, p)
    if lp_val < 1e-300:
        return np.zeros(n)
    
    # Factor: (sum c^p)^(1/p - 1) = lp_val^(1 - p)
    if p < 100:
        factor = lp_val ** (1.0 - p)
        w = c ** (p - 1)
    else:
        log_c = np.log(np.maximum(c, 1e-300))
        lc_max = log_c.max()
        log_sum_exp = lc_max + np.log(np.sum(np.exp(p * (log_c - lc_max))))
        # factor = exp((1-p) * log_sum_exp)
        log_factor = (1.0 - p) * log_sum_exp
        # w[k] = c[k]^(p-1) = exp((p-1)*log_c[k])
        log_w = (p - 1.0) * log_c
        log_w_max = log_w.max()
        w = np.exp(log_w - log_w_max)
        factor = np.exp(log_factor + log_w_max)
    
    # Compute convolution w * a[::-1]
    a_rev = a[::-1]
    conv_len = len(w) + len(a_rev) - 1
    N = _next_pow2(conv_len)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a_rev, N)
    conv_result = np.fft.irfft(W * A, N)[:conv_len]
    
    # grad_a[i] = 4n * factor * conv_result[n-1-i]
    # conv_result[n-1-i] for i=0..n-1 corresponds to indices n-1, n-2, ..., 0
    grad_a = 4.0 * n * factor * conv_result[n - 1 :: -1][:n]
    
    if not np.all(np.isfinite(grad_a)):
        return np.zeros(n)
    return grad_a


def _nesterov_lp_optimize(a_init, deadline, p_schedule, iters_per_level=3000, verbose=True):
    """
    Nesterov-accelerated gradient descent with Lp continuation.
    Works in z-space: a = softmax(z), sum(a)=1 automatically.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    z = np.log(a)
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    if verbose:
        print(f"    Nesterov Lp start: score={best_score:.10f}")
    
    for p in p_schedule:
        if time.monotonic() > deadline - 5.0:
            break
        
        # Initialize Nesterov state
        y = z.copy()
        z_prev = z.copy()
        
        if p < 100:
            max_iters = iters_per_level
        else:
            max_iters = max(500, iters_per_level // 2)
        
        lr = 1.0 / np.sqrt(p) * 0.5
        for it in range(max_iters):
            if time.monotonic() > deadline - 5.0:
                    break
            
            # Compute a from y (Nesterov lookahead point)
            y_shifted = y - y.max()
            exp_y = np.exp(np.clip(y_shifted, -70, 70))
            a_y = exp_y / exp_y.sum()
            
            # Compute gradient
            c = _full_convolve(a_y, a_y)
            grad_a = _lp_grad_a(c, a_y, p)
            
            # Convert to z-gradient: grad_z[i] = a[i] * (grad_a[i] - dot(a, grad_a))
            mean_grad = float(np.dot(a_y, grad_a))
            grad_z = a_y * (grad_a - mean_grad)
            
            if not np.all(np.isfinite(grad_z)):
                break
            
            # Nesterov update
            t = it + 1
            momentum = (t - 1) / (t + 2.0) if t > 1 else 0.0
            
            y_new = z + momentum * (z - z_prev)
            
            # Adaptive step size: use gradient norm
            gnorm = float(np.linalg.norm(grad_z))
            if gnorm < 1e-15:
                break
            step = lr / (gnorm + 1e-10) * 0.1
            step = min(step, 0.5)
            
            z_new = y_new - step * grad_z
            z_prev = z.copy()
            z = z_new
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            
            # Periodic evaluation
            if (it + 1) % 100 == 0 or it == max_iters - 1:
                a_curr = _softmax_from_z(z)
                sc = _score_norm(a_curr)
                if sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()
                    if verbose and sc < 1.51:
                        print(f"    p={p}, iter={it}: score={sc:.10f}")
        
        # Update z for next level
        a_curr = _softmax_from_z(z)
        z = np.log(np.maximum(a_curr, 1e-15))
        z = z - z.mean()
    
    return best_a, best_score


def _softmax_from_z(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -70, 70))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _peak_shave(a_init, deadline, rng, n_trials=50000):

                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    return best_a, best_sc


def _fine_gradient_polish(a_init, deadline, rng, lr_init=0.001, n_iters=200000):
    """
    Very fine gradient polishing with adaptive step size.
    Uses the true max (not smoothed) via subgradient.
    """
    n = len(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    # Use large-p Lp as proxy for max
    p = 5000
    
    lr = lr_init
    t = 0
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        t += 1
        a_curr = _softmax_from_z(z)
        c = _full_convolve(a_curr, a_curr)
        
        # Gradient using Lp (large p approximates max)
        grad_a = _lp_grad_a(c, a_curr, p)
        mean_grad = float(np.dot(a_curr, grad_a))
        grad_z = a_curr * (grad_a - mean_grad)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z ** 2)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        # Decay learning rate
        if t % 5000 == 0:
            lr *= 0.95
        
        # Evaluate periodically
        if (t + 1) % 200 == 0:
            a_eval = _softmax_from_z(z)
            sc = _score_norm(a_eval)
            if sc < best_sc:
                best_sc = sc
                best_a = a_eval.copy()
    
    return best_a, best_sc

def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_asymmetric(n, shift=0.3):
    """Asymmetric profile: shifted raised cosine."""
    t = np.linspace(0.0, 1.0, n)
    center = 0.5 + shift
    width = 0.3
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_multi_bump(n, n_bumps=3, spacing=None):
    """Multiple Gaussian bumps at different positions."""
    t = np.linspace(0.0, 1.0, n)
    if spacing is None:
        centers = np.linspace(0.2, 0.8, n_bumps)
    else:
        centers = spacing
    a = np.zeros(n)
    width = 0.25 / n_bumps
    for c in centers:
        a += np.exp(-0.5 * ((t - c) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_sinusoidal_mod(n, freq=2, depth=0.3):
    """Sinusoidally modulated profile (asymmetric)."""
    t = np.linspace(0.0, 1.0, n)
    base = 1.0 + 0.5 * np.sin(2 * np.pi * t)  # Asymmetric base
    mod = 1.0 + depth * np.cos(2 * np.pi * freq * t)
    a = base * mod
    a = np.maximum(a, 1e-12)
    return a / a.sum()

        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.25)
            a_f, sc_f = _fine_gradient_polish(best_seq, dl, rng,
                                               lr_init=0.0005, n_iters=200000)
            update_best(a_f, sc_f, "final_grad")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = float(2.0 * len(final_seq) * _full_convolve(final_seq, final_seq).max() / (s * s))
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence error: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular(8192)
    print(f"FALLBACK: n=8192")
    return [float(x) for x in fallback]
