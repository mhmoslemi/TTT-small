def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Return 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    """Return the full score 2*n*max(conv(a,a))/sum(a)^2."""

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _logsumexp(x, beta):
    """Compute max(x) + beta * log(sum(exp((x - max(x))/beta)))"""
    m = float(x.max())
    if m < 1e-300:
        return 0.0
    shifted = (x - m) / beta
    max_s = float(shifted.max())
    return m + beta * float(np.log(np.sum(np.exp(shifted - max_s))))

def _run_lbfgs_continuation(a_init, deadline, beta_fracs=None, maxiter=5000):

    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    
    if beta_fracs is None:
        beta_fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                      0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                      0.00005, 0.00003, 0.00001, 0.000005, 0.000003, 0.000001]
    
    for beta_frac in beta_fracs:
        if time.monotonic() > deadline:
            break
        beta = beta_frac * m0
        try:
            res = minimize(_lse_obj, z, args=(beta, n),
                          method='L-BFGS-B', jac=_lse_grad,
                          options={'maxiter': maxiter, 'ftol': 1e-25,
                                   'gtol': 1e-15, 'maxls': 80, 'maxfun': 30000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax(res.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _topk_gradient_descent(a_init, deadline, lr=0.01, top_k=5,

    beta = 0.001 * m0

    no_improve = 0
    check_interval = 100

            break
        
        # Nesterov: evaluate gradient at y = z + momentum * v
        y = z + nesterov_momentum * v
        a_y = _softmax(y)
        c = _full_convolve(a_y, a_y)
        cm = float(c.max())
        beta = max(1e-10 * cm, 0.001 * m0 * (0.9 ** (it // 100)))

        c_use = c[idx]

        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()

        w_full[idx] = w
        a_rev = a_y[::-1]
        w_conv = _full_convolve(w_full, a_rev)
        h = w_conv[n-1:2*n-1]
        diag = 2*np.arange(n) < (2*n-1)
        h[diag] -= w_full[2*np.arange(n)[diag]] * a_y[diag]

        Wa = float(np.dot(h, a_y))
        grad = 2.0 * n * a_y * (h - Wa)

        z = z + v
        z = np.clip(z, -100, 100)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses aggressive LSE continuation, peak-directed Nesterov GD,
coordinate descent, and multi-resolution refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j-1] * (2.0*j - 1.0) / (2.0*j)
    a = b[::-1].copy()
    return a / a.sum()


def _param_init(n, etype):
    t = np.linspace(0, 1, n)
    if etype == 'gaussian':
        a = np.exp(-3.0 * (t - 0.35)**2)
    elif etype == 'exp_ramp':
        a = np.exp(2.5 * t)
    elif etype == 'sqrt_ramp':
        a = np.sqrt(t + 0.01)
    elif etype == 'power2':
        a = (t + 0.01)**2
    elif etype == 'power3':
        a = (t + 0.01)**3
    elif etype == 'cosine':
        a = np.cos(np.pi * t / 2.0)**0.7 + 0.01
    elif etype == 'log':
        a = np.log(t + 0.05) + 1.0
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _lse_obj(z, beta, n, top_k):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_use = c[idx]
    else:
        c_use = c
    cm = float(c_use.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c_use - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * float(np.log(np.sum(np.exp(shifted - max_s))))
    return float(2.0 * n * lse)

    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    a = _softmax_fn(z)
