            quad_pos = 2 * i - k_min

            # Unaffected max
            left_max = prefix_max[k_min - 1] if k_min > 0 else 0.0
            right_max = suffix_max[k_max + 1] if k_max + 1 < total_len else 0.0
            c_unaffected = max(left_max, right_max)

            # Current objective
            old_max = max(c_unaffected, float(c_slice.max()))
            old_h = old_max / (S * S)

            # Candidate deltas
            candidates = np.array([
                -0.95 * v, -0.7 * v, -0.4 * v, -0.15 * v, -0.05 * v,
                0.0,
                0.05 * v, 0.15 * v, 0.4 * v, 0.7 * v
            ])

            best_h = old_h
            best_d = 0.0

            for d in candidates:
                new_S = S + d
                if new_S <= 1e-12:
                    continue
                new_S2 = new_S * new_S
                new_c_slice = c_slice + d_vals * d
                if has_quad:
                    new_c_slice[quad_pos] += d * d
                new_max = max(c_unaffected, float(new_c_slice.max()))
                h = new_max / new_S2
                if h < best_h:
                    best_h = h
                    best_d = d

            if best_h < old_h * (1 - 1e-12) and best_d != 0.0:
                a[i] = max(0.0, v + best_d)
                actual_d = a[i] - v
                if abs(actual_d) > 1e-18:
                    c[k_min:k_max + 1] += d_vals * actual_d
                    if has_quad:
                        c[2 * i] += actual_d * actual_d
                    S = S + actual_d
                    improved += 1
                new_score = _score_from_c(c, n, S)
                if new_score < best_score:
                    best_score = new_score

        if (pass_idx + 1) % verbose_every == 0 or pass_idx == 0:
            print(f"      CD pass {pass_idx + 1}/{max_passes}: score={best_score:.10f}, improved_coords={improved}")
        if improved == 0:
            break

    return best_score, improved


def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=500000, lr=0.01, nesterov=True):
    """Adam with optional Nesterov acceleration on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    v_ = np.zeros(n)  # Nesterov velocity

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_start * (max(beta_end, 1e-15) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break

        if nesterov:
            v_ = 0.9 * v_ - current_lr * grad_z
            z_lookahead = z + v_
            a_look = _softmax(z_lookahead)
            c_look = _full_convolve(a_look, a_look)
            c_max_look = float(c_look.max())
            beta_look = beta_start * (max(beta_end, 1e-15) / beta_start) ** (t / max_iter)
            e_look = np.exp((c_look - c_max_look) / max(beta_look, 1e-10))
            s_look = float(e_look.sum())
            if s_look < 1e-300:
                w_look = np.zeros_like(c_look)
                w_look[int(np.argmax(c_look))] = 1.0
            else:
                w_look = e_look / s_look
            w_rev_look = w_look[::-1]
            conv_look = _full_convolve(a_look, w_rev_look)
            grad_look = 2.0 * conv_look[n - 1:2 * n - 1][::-1]
            W_look = float(np.dot(a_look, grad_look))
            grad_z = a_look * (grad_look - W_look)

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

        if t % 5000 == 0:
            if best_score < prev_best * 0.9995:
                current_lr = min(0.05, current_lr * 1.05)
            else:
                current_lr = max(0.001, current_lr * 0.92)
            prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    # Build LP sparse indices
    rows, cols = _build_lp_sparse(n)
    print(f"LP indices built: {len(rows)} nonzeros, t={time.monotonic()-t_start:.1f}s")

    # ========== Phase 1: LP iterations (0 - 400s) ==========
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: LP (target end {phase1_end - t_start:.0f}s)")
    a_c = _full_convolve(a, a)
    S = float(a.sum())
    M = float(a_c.max())
    lp_count = 0
    prev_sc = _score_from_c(a_c, n, S)
    stall = 0

    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a, M = _lp_step(a, M, rows, cols, n, time_limit=20)
        a_c = _full_convolve(a, a)
        S = float(a.sum())
        sc = _score_from_c(a_c, n, S)
        lp_count += 1
        dt_lp = time.monotonic() - t0

        if sc < prev_sc - 1e-12:
            stall = 0
        else:
            stall += 1

        if lp_count % 3 == 0 or stall > 10:
            full_sc = _score(a)
            print(f"  LP[{lp_count}] score={full_sc:.10f} M={M:.8f} dt={dt_lp:.1f}s stall={stall} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()

        if stall > 25:
            print(f"  LP stalled at iter {lp_count}")
            break
        if time.monotonic() > phase1_end - 5:
            break
        prev_sc = sc

    full_sc = _score(a)
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ========== Phase 2: Coordinate Descent (400 - 800s) ==========
    phase2_end = min(deadline, t_start + 800.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Coordinate Descent (target end {phase2_end - t_start:.0f}s)")
        a_c = _full_convolve(best_seq, best_seq)
        S = float(best_seq.sum())
        # Repeated CD passes with decreasing step sizes (implicit via smaller improvements)
        cd_start = time.monotonic()
        # First block: many passes
        remaining = phase2_end - cd_start
        est_pass_time = 0.5  # rough estimate per pass for n=2048
        n_passes = max(10, int(remaining * 0.6 / max(est_pass_time, 0.1)))
        print(f"  Planning {n_passes} CD passes")
        _coord_descent_pass(best_seq, a_c, S, n, rng, max_passes=n_passes, verbose_every=20)
        sc_cd = _score(best_seq)
        print(f"  After CD block 1: score={sc_cd:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc_cd) and sc_cd < best_score:
            best_score = sc_cd
            best_seq = best_seq.copy()

        # Second block from the improved point
        if time.monotonic() < phase2_end - 30:
            a_c = _full_convolve(best_seq, best_seq)
            S = float(best_seq.sum())
            n_passes2 = max(5, int((phase2_end - time.monotonic()) / max(est_pass_time, 0.1)))
            _coord_descent_pass(best_seq, a_c, S, n, rng, max_passes=n_passes2, verbose_every=20)
            sc_cd2 = _score(best_seq)
            print(f"  After CD block 2: score={sc_cd2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc_cd2) and sc_cd2 < best_score:
                best_score = sc_cd2
                best_seq = best_seq.copy()

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ========== Phase 3: Adam smoothing (800 - 1200s) ==========
    phase3_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: Adam (target end {phase3_end - t_start:.0f}s)")
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        # First Adam run with moderate smoothing
        deadline3a = min(phase3_end - 5, t_start + 1000.0)
        a_pol, sc_pol = _adam_smooth(
            best_seq, deadline3a,
            beta_start=0.002 * max_cb, beta_end=0.0001 * max_cb,
            max_iter=300000, lr=0.008
        )
        full_sc = _score(a_pol)
        print(f"  Adam A: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_pol.copy()

        # Second Adam run with finer smoothing
        if time.monotonic() < phase3_end - 10:
            c_best2 = _full_convolve(best_seq, best_seq)
            max_cb2 = float(c_best2.max())
            deadline3b = phase3_end - 3
            a_pol2, sc_pol2 = _adam_smooth(
                best_seq, deadline3b,
                beta_start=0.0005 * max_cb2, beta_end=0.00002 * max_cb2,
                max_iter=200000, lr=0.004
            )
            full_sc2 = _score(a_pol2)
            print(f"  Adam B: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc2) and full_sc2 < best_score:
                best_score = full_sc2
                best_seq = a_pol2.copy()

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ========== Phase 4: LP + CD from improved point (1200 - 1550s) ==========
    phase4_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: LP+CD (target end {phase4_end - t_start:.0f}s)")
        a = best_seq.copy()
        a_c = _full_convolve(a, a)
        S = float(a.sum())
        M = float(a_c.max())
        prev_sc = _score_from_c(a_c, n, S)
        stall = 0
        lp4_count = 0

        while time.monotonic() < phase4_end - 120:
            a, M = _lp_step(a, M, rows, cols, n, time_limit=20)
            a_c = _full_convolve(a, a)
            S = float(a.sum())
            sc = _score_from_c(a_c, n, S)
            lp4_count += 1
            if sc < prev_sc - 1e-12:
                stall = 0
            else:
                stall += 1
            if lp4_count % 3 == 0:
                full_sc = _score(a)
                print(f"  LP4[{lp4_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
            if stall > 20:
                break
            prev_sc = sc

        print(f"  LP4 done: {lp4_count} iters, best={best_score:.10f}")

        # CD polish after LP
        if time.monotonic() < phase4_end - 5:
            a_c = _full_convolve(best_seq, best_seq)
            S = float(best_seq.sum())
            n_passes4 = max(5, int((phase4_end - time.monotonic()) / 0.5))
            _coord_descent_pass(best_seq, a_c, S, n, rng, max_passes=min(n_passes4, 30), verbose_every=10)
            sc_cd4 = _score(best_seq)
            print(f"  CD after LP4: score={sc_cd4:.10f}")
            if np.isfinite(sc_cd4) and sc_cd4 < best_score:
                best_score = sc_cd4
                best_seq = best_seq.copy()

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ========== Phase 5: Perturb + LP restart (1550 - 1700s) ==========
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end:
        print(f"Phase 5: Perturb+LP (target end {phase5_end - t_start:.0f}s)")
        for attempt in range(3):
            if time.monotonic() > phase5_end - 40:
                break
            sigma = 0.02 + 0.015 * attempt
            p_start = _perturb(best_seq, sigma, rng)
            p_c = _full_convolve(p_start, p_start)
            p_S = float(p_start.sum())
            p_M = float(p_c.max())
            prev_p_sc = _score_from_c(p_c, n, p_S)
            stall_p = 0
            for it in range(15):
                if time.monotonic() > phase5_end - 20:
                    break
                p_start, _ = _lp_step(p_start, p_M, rows, cols, n, time_limit=15)
                p_c = _full_convolve(p_start, p_start)
                p_S = float(p_start.sum())
                sc_p = _score_from_c(p_c, n, p_S)
                if sc_p < prev_p_sc - 1e-12:
                    stall_p = 0
                else:
                    stall_p += 1
                if stall_p > 8:
                    break
                prev_p_sc = sc_p
            # Quick CD polish
            _coord_descent_pass(p_start, p_c, p_S, n, rng, max_passes=10, verbose_every=5)
            full_sc = _score(p_start)
            print(f"  Restart[{attempt}] (sigma={sigma:.3f}): score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_start.copy()

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # ========== Phase 6: Final ultra-fine CD + Adam (1700 - deadline) ==========
    if time.monotonic() < deadline - 10:
        print(f"Phase 6: Final polish (until deadline {deadline - t_start:.0f}s)")
        # Final CD
        a_c = _full_convolve(best_seq, best_seq)
        S = float(best_seq.sum())
        n_final = max(5, int((deadline - time.monotonic() - 60) / 0.5))
        _coord_descent_pass(best_seq, a_c, S, n, rng, max_passes=min(n_final, 50), verbose_every=10)
        sc_fin = _score(best_seq)
        if np.isfinite(sc_fin) and sc_fin < best_score:
            best_score = sc_fin
            best_seq = best_seq.copy()

        # Final Adam with very fine smoothing
        if time.monotonic() < deadline - 15:
            c_fin = _full_convolve(best_seq, best_seq)
            max_cf = float(c_fin.max())
            a_fin, sc_af = _adam_smooth(
                best_seq, deadline - 5,
                beta_start=0.0001 * max_cf, beta_end=0.000005 * max_cf,
                max_iter=100000, lr=0.002
            )
            full_sc = _score(a_fin)
            print(f"  Final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a_fin.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
