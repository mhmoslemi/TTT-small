def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_profile(n, ptype, params):
    t = np.linspace(0.0, 1.0, n)
    if ptype == 'linear':
        slope = params
        a = 1.0 + slope * (t - 0.5)
    elif ptype == 'quad':
        a1, a2 = params
        u = 2.0 * t - 1.0
        a = 1.0 + a1 * u + a2 * u ** 2
    elif ptype == 'bimodal':
        sep, width = params
        c1 = 0.5 - sep / 2.0
        c2 = 0.5 + sep / 2.0
        a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    elif ptype == 'wshape':
        freq, depth = params
        a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    elif ptype == 'power':
        p = params
        u = np.abs(2.0 * t - 1.0)
        a = (1.0 - u) ** p
    elif ptype == 'triangular':
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) * (n - i)
    elif ptype == 'one_sided_linear':
        a = 1.0 - t
    elif ptype == 'one_sided_power':
        p = params
        a = (1.0 - t) ** p
    elif ptype == 'sin_half':
        a = np.sin(np.pi * t)
    elif ptype == 'cos_bell':
        a = np.cos(np.pi * (t - 0.5)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _compute_smooth_grad(z, beta, n):

    cm = float(c.max())
    c_shifted = (c - cm) / beta

def _lbfgs_continuation(a_init, deadline, max_wall, beta_scale=1.0, min_beta_frac=0.00005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_starts = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.00075,
                   0.0004, 0.0002, 0.0001, 0.00005, min_beta_frac]
    betas = [b * max_c0 * beta_scale for b in beta_starts]
    per_level = max(2.0, max_wall / len(betas))
    
    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_dl = min(deadline, time.monotonic() + per_level)
        
        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs_ = (c_ - cm_) / beta
            e_ = np.exp(cs_)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s_))
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs_ = (c_ - cm_) / beta
            e_ = np.exp(cs_)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = result.x.copy()
                a_new = _softmax_fn(z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _gradient_local_search(a_init, deadline, rng, n_trials=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Compute gradient info
        c = _full_convolve(a, a)
        K = min(10, len(c))
        top_lags = np.array(np.argsort(c)[-K:], dtype=int)
        max_c = float(c.max())
        
        if it % 4 == 0:
            # Gradient-informed pair move: reduce high-contribution, increase low-contribution
            # Compute contribution of each element to top peaks
            contrib = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    partners = kl - np.arange(lo, hi + 1)
                    contrib[lo:hi + 1] += a[partners]
            contrib /= K
            
            # Reduce from highest contribution, add to lowest
            idx_reduce = int(np.argmax(contrib))
            safe_pool = np.argsort(contrib)[:max(1, n // 4)]
            idx_add = int(safe_pool[rng.integers(len(safe_pool))])
            
            if idx_reduce == idx_add:
                continue
            
            frac = rng.uniform(0.05, 0.25)
            delta = a[idx_reduce] * frac
            if delta < 1e-15:
                continue
            
            a_trial = a.copy()
            a_trial[idx_reduce] -= delta
            a_trial[idx_add] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        elif it % 4 == 1:
            # Block multiplicative perturbation (small)
            block_size = int(rng.integers(2, min(8, n // 16)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.85, 1.15, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        elif it % 4 == 2:
            # Pair swap (exchange mass)
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.05, 0.2)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1 + amt2)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2 + amt1)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        else:
            # Smoothed gradient step (top-2 peaks only)
            a_curr = _softmax_fn(z)
            c_curr = _full_convolve(a_curr, a_curr)
            top2 = np.array(np.argsort(c_curr)[-2:], dtype=int)
            
            grad = np.zeros(n)
            for kl in top2:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    grad[lo:hi + 1] += 2.0 * a_curr[kl - hi:kl - lo + 1][::-1]
            
            # Normalize gradient
            gn = np.linalg.norm(grad)
            if gn > 1e-15:
                grad /= gn
            
            lr = 0.005 + 0.003 * (1.0 - it / max(1, n_trials))
            z = z - lr * grad
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
    
    return best_a, best_sc


def _adam_optimizer(a_init, deadline, rng, lr=0.01, k_top=5, max_iters=30000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        
        grad = np.zeros(n)
        for kl in top_idx:
            kl = int(kl)
            lo = max(0, kl - (n - 1))
            hi = min(n - 1, kl)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * a_curr[kl - hi:kl - lo + 1][::-1]
        grad /= k_actual
        grad *= 4.0 * n
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.8)
    
    return best_a, best_score


def _fine_tune_single_elem(a_init, deadline, rng, n_rounds=20):
    """For each element, try small perturbations to see if score improves."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    indices = rng.permutation(n)
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 2.0:
            break
        for idx in indices:
            if time.monotonic() > deadline - 2.0:
                break
            # Try small multiplicative changes
            for mult in [0.95, 0.98, 1.02, 1.05]:
                a_trial = a.copy()
                a_trial[idx] *= mult
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-10:
                    continue
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_profile(n_parent, 'sin_half', 0)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using sin_half n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    n_work = n_parent
    print(f"Working with n={n_work}")
    
    # Phase 1: Quick profile sweep (30s)
    phase1_end = min(deadline, t_start + 30.0)
    print("Phase 1: Profile sweep")
    
    profiles_to_try = [
        ('sin_half', 0),
        ('cos_bell', 0),
        ('triangular', 0),
        ('one_sided_linear', 0),
        ('one_sided_power', 1.5),
        ('one_sided_power', 2.0),
        ('one_sided_power', 3.0),
        ('power', 1.5),
        ('power', 2.0),
        ('power', 3.0),
        ('power', 5.0),
        ('linear', 1.0),
        ('linear', 2.0),
        ('linear', 3.0),
        ('quad', 1.0, 0.3),
        ('quad', 1.5, -0.2),
        ('quad', 0.5, 0.5),
        ('bimodal', 0.4, 0.15),
        ('bimodal', 0.6, 0.2),
        ('wshape', 2, 0.3),
        ('wshape', 3, 0.4),
    ]
    
    best_start = parent.copy()
    best_start_sc = sc_parent
    
    for ptype, *params in profiles_to_try:
        if time.monotonic() > phase1_end:
            break
        if ptype in ('linear', 'power', 'one_sided_power', 'sin_half', 'cos_bell', 'triangular', 'one_sided_linear'):
            p = _make_profile(n_work, ptype, params[0] if params else 0)
        elif ptype == 'quad':
            p = _make_profile(n_work, ptype, (params[0], params[1]))
        elif ptype == 'bimodal':
            p = _make_profile(n_work, ptype, (params[0], params[1]))
        elif ptype == 'wshape':
            p = _make_profile(n_work, ptype, (params[0], params[1]))
        else:
            continue
        sc = _score_norm_fn(p)
        if sc < best_start_sc:
            best_start_sc = sc
            best_start = p.copy()
        update_best(p, sc, f"prof_{ptype}")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # Use the better of parent or best profile as starting point
    if best_start_sc < sc_parent:
        work_start = best_start.copy()
        print(f"Using profile start: {best_start_sc:.10f}")
    else:
        work_start = parent.copy()
        print(f"Using parent start: {sc_parent:.10f}")
    
    # Phase 2: L-BFGS continuation (500s) - primary optimizer
    phase2_end = min(deadline, t_start + 530.0)
    print("Phase 2: L-BFGS continuation (main)")
    dl2 = phase2_end - 5.0
    a_best, sc_best = _lbfgs_continuation(work_start, dl2, 520.0, beta_scale=1.0)
    update_best(a_best, sc_best, "lbfgs_main")
    
    # Phase 3: Informed local search (300s)
    phase3_end = min(deadline, t_start + 830.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Gradient-informed local search")
        dl3 = phase3_end - 5.0
        a_best, sc_best = _gradient_local_search(best_seq, dl3, rng, n_trials=80000)
        update_best(a_best, sc_best, "local_search")
    
    # Phase 4: Second L-BFGS round with finer beta (300s)
    phase4_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: L-BFGS round 2 (finer beta)")
        dl4 = phase4_end - 5.0
        a_best, sc_best = _lbfgs_continuation(best_seq, dl4, 290.0, beta_scale=0.4, min_beta_frac=0.00003)
        update_best(a_best, sc_best, "lbfgs_fine")
        
        # Quick local search after
        if time.monotonic() < phase4_end - 20:
            dl4b = phase4_end - 5.0
            a_best2, sc_best2 = _gradient_local_search(best_seq, dl4b, rng, n_trials=20000)
            update_best(a_best2, sc_best2, "local_after_lbfgs2")
    
    # Phase 5: Adam optimizer (200s)
    phase5_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Adam optimizer")
        dl5 = phase5_end - 5.0
        a_best, sc_best = _adam_optimizer(best_seq, dl5, rng, lr=0.008, k_top=3, max_iters=20000)
        update_best(a_best, sc_best, "adam_k3")
        
        if time.monotonic() < phase5_end - 30:
            dl5b = phase5_end - 5.0
            a_best2, sc_best2 = _adam_optimizer(best_seq, dl5b, rng, lr=0.004, k_top=7, max_iters=10000)
            update_best(a_best2, sc_best2, "adam_k7")
    
    # Phase 6: Multi-n exploration (200s)
    phase6_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Multi-n exploration")
        n_curr = len(best_seq)
        
        target_ns = [4096, 8192, 16384]
        target_ns = [tn for tn in target_ns if tn != n_curr]
        
        remaining = phase6_end - time.monotonic()
        per_n = max(20.0, remaining / max(1, len(target_ns)))
        
        for tn in target_ns:
            if time.monotonic() > phase6_end - 30:
                break
            p_interp = _interp_profile(best_seq, tn)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  n={tn}: interp score={sc_interp:.10f}")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{tn}")
            
            if time.monotonic() < phase6_end - 30:
                dl = min(phase6_end - 20, time.monotonic() + per_n * 0.7)
                a_best, sc_best = _lbfgs_continuation(p_interp, dl, per_n * 0.7, beta_scale=0.5)
                update_best(a_best, sc_best, f"lbfgs_n{tn}")
    
    # Phase 7: Final intensive polishing (remaining time)
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(10.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")
        
        # 7a: Very fine L-BFGS
        if remaining > 50:
            dl7a = min(deadline - 20, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl7a, remaining * 0.35,
                                                   beta_scale=0.2, min_beta_frac=0.00002)
            update_best(a_best, sc_best, "final_lbfgs")
        
        # 7b: More local search
        if time.monotonic() < deadline - 20:
            dl7b = min(deadline - 15, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _gradient_local_search(best_seq, dl7b, rng, n_trials=30000)
            update_best(a_best, sc_best, "final_local")
        
        # 7c: Fine-tune individual elements
        if time.monotonic() < deadline - 15 and remaining > 40:
            dl7c = min(deadline - 10, time.monotonic() + remaining * 0.25)
            a_best, sc_best = _fine_tune_single_elem(best_seq, dl7c, rng, n_rounds=10)
            update_best(a_best, sc_best, "final_tune")
        
        # 7d: One more quick L-BFGS
        if time.monotonic() < deadline - 10:
            dl7d = min(deadline - 8, time.monotonic() + 60.0)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl7d, 50.0, beta_scale=0.15, min_beta_frac=0.00002)
            update_best(a_best, sc_best, "final_lbfgs2")
    
    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_profile(8192, 'sin_half', 0)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_profile(8192, 'sin_half', 0)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
