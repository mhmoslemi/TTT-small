"""
Multi-phase optimizer: FISTA (softmax-smoothed) -> Subgradient (exact max) -> 
Coordinate descent -> Multi-length exploration. Minimizes 2n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    n_full = n_a + n_b - 1
    N = _next_pow2(n_full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_full]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Assumes a is on the simplex (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _compute_grad(a, w):
    """Gradient of sum_k w[k]*b[k] w.r.t. a, where b=conv(a,a)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h

def _fista(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max of convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Lipschitz constant estimate
    L = 4.0 * n / beta
    step = 1.0 / L
    
    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(c_init.max())
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        
        # Softmax weights
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)
        
        if it % 50 == 0:
            sc = float(_full_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = float(_full_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parametrization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while t < max_iter_total and time.monotonic() < deadline:
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            # Gradient in z-space: grad_z[i] = 4*n*a[i]*(h[i] - W) where W = w^T c
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)
            
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            
            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if t % 10000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _subgradient_step(a, deadline, iters, lr):
    """Proximal gradient with exact subgradient (indicator of argmax set)."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for it in range(iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = c.max()
        
        # Find near-argmax lags (within 1% of max)
        threshold = cm * 0.999
        active = np.where(c >= threshold)[0]
        
        # Average the subgradients at active lags
        a_rev = a[::-1]
        total_grad = np.zeros(n)
        for k in active:
            # grad of b[k] w.r.t. a
            # b[k] = sum_i a[i]*a[k-i], db[k]/da[j] = a[k-j] + a[j-k] (with 0 for invalid)
            # This equals 2 * (w_k conv a_rev)[n-1:2n-1] where w_k is delta at k
            # = 2 * a_rev convolved with delta_k, taken at positions [n-1:2n-1]
            # = 2 * a[::-1][k-(n-1) : k-(n-1)+n] (with zero-padding)
            # More simply: grad[k] = a[k-j] + a[j-k] for valid indices
            # Which is the same as: 2 * (delta_k * a_rev)[n-1:2n-1]
            lo = k - (n - 1)
            hi = lo + n
            # Extract from a_rev with zero-padding
            vals = np.zeros(n)
            a_rev_len = n
            for idx in range(n):
                src = lo + idx
                if 0 <= src < a_rev_len:
                    vals[idx] = a_rev[src]
            total_grad += 2.0 * vals
        
        total_grad /= len(active)
        
        # Proximal gradient step
        a_next = _proj_simplex(a - lr * total_grad)
        
        sc = _score_norm(a_next)
        if sc < best_score:
            best_score = sc
            best_a = a_next.copy()
        else:
            lr *= 0.5
        # Reset lr if we improved
        if sc < best_score * 1.001:
            lr = min(lr * 1.5, 0.01)
    
    return best_a, best_score

def _interp_profile(p, n_new):

Wait, I need to review this more carefully. The `_subgradient_step` function is O(n * |active|) per iteration which could be slow for n=4096. Let me optimize it.

Actually, the subgradient at a specific lag k is:
`grad_k[j] = a[k-j] + a[j-k]` (with a[x]=0 for invalid x)

This can be computed as: `2 * (delta_k conv a_rev)[n-1:2n-1]` where `delta_k` is 1 at position k and 0 elsewhere.

But `delta_k conv a_rev = a_rev shifted by k` (or something like that). Let me think...

Actually, `conv([0,...,0,1,0,...,0], a_rev)` where the 1 is at position k, is just `a_rev` shifted by k positions. So `wc = a_rev` shifted, and we take `wc[n-1:2n-1]`.

This means for a single lag k, the gradient is:
`2 * a_rev[k-(n-1) : k-(n-1)+n]` (with zero-padding for out-of-bounds).

So for multiple active lags, the total gradient is:
`2 * sum_k a_rev[k-(n-1) : k-(n-1)+n]` (with zero-padding).

This can be computed efficiently using the convolution of the weight vector `w` (indicator of active set) with `a_rev`:

"""

Uses: FISTA (softmax-smoothed max) -> Adam log-space -> Subgradient (exact active set) ->

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    n_full = n_a + n_b - 1
    N = _next_pow2(n_full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_full]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 20 == 0:
            sc = float(_full_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = float(_full_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score

def _subgradient_bt(a_init, deadline, max_iter, lr_init):
    """Proximal gradient with backtracking using exact active set subgradient."""
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())

        # Active set: lags within a small tolerance of max
        # Use adaptive tolerance: start tight, relax if too few active
        for tol_frac in [1e-5, 1e-4, 5e-4, 1e-3, 5e-3]:
            threshold = cm * (1.0 - tol_frac)
            w = (c >= threshold).astype(np.float64)
            w_sum = w.sum()
            if w_sum >= 2:
                break
        w /= w_sum

        grad = _compute_grad(a_curr, w)

        # Backtracking line search
        improved = False
        lr_try = lr
        for _ in range(15):
            a_next = _proj_simplex(a_curr - lr_try * grad)
            sc = _score_norm(a_next)
            if sc <= best_score + 1e-15:
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()
                    improved = True
                a_curr = a_next.copy()
                break
            lr_try *= 0.5
        else:
            # Try smaller scale
            a_curr = a_curr.copy()

        # Adapt learning rate
        if improved:
            lr = min(lr * 1.05, 0.005)
        else:
            lr *= 0.7

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

    per_time = phase2_end - time.monotonic() - 10
    if per_time > 20:
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.008, max_iter_total=500000
        )
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = best_seq.copy() / best_seq.sum()

    # === Phase 3: Subgradient with backtracking ===
    print("Phase 3: Subgradient with backtracking")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end - 20:
        a_sg = a_work.copy()
        # Learning rate: small for stability
        lr_init = 0.3 / n_main
        a_sg, sc_sg = _subgradient_bt(a_sg, time.monotonic() + (phase3_end - time.monotonic() - 5),
                                      max_iter=20000, lr_init=lr_init)
        sc_sg_full = _score(a_sg)
        if sc_sg_full < best_score:
            best_score = sc_sg_full
            best_seq = a_sg.copy()
            print(f"  Subgradient: score={sc_sg_full:.10f} t={time.monotonic()-t_start:.1f}s")
        a_work = best_seq.copy() / best_seq.sum()

        # Second round of subgradient with even smaller step
        if time.monotonic() < phase3_end - 10:
            a_sg2 = a_work.copy()
            lr2 = 0.1 / n_main
            a_sg2, sc_sg2 = _subgradient_bt(a_sg2, time.monotonic() + (phase3_end - time.monotonic() - 3),
                                            max_iter=10000, lr_init=lr2)
            sc_sg2_full = _score(a_sg2)
            if sc_sg2_full < best_score:
                best_score = sc_sg2_full
                best_seq = a_sg2.copy()
                print(f"  Subgradient2: score={sc_sg2_full:.10f} t={time.monotonic()-t_start:.1f}s")
            a_work = best_seq.copy() / best_seq.sum()

    # === Phase 4: n=8192 exploration ===
    print("Phase 4: n=8192 exploration")
    phase4_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase4_end - 40 and best_seq is not None:
        n_8192 = 8192
        a_8192 = _interp_profile(best_seq, n_8192)
        sc_8192 = _score(a_8192)
        print(f"  Interpolated n=8192: score={sc_8192:.10f}")
        if sc_8192 < best_score:
            best_score = sc_8192
            best_seq = a_8192.copy()

        # Optimize the 8192 version
        if time.monotonic() < phase4_end - 30:
            c_8192 = _full_convolve(a_8192, a_8192)
            max_c8 = float(c_8192.max())

            # FISTA on 8192
            b8 = 0.003 * max_c8
            t_fista = min(40.0, phase4_end - time.monotonic() - 20)
            if t_fista > 5:
                a_8192_f, _ = _fista(a_8192, b8, time.monotonic() + t_fista, 30000)
                a_8192_f = a_8192_f / a_8192_f.sum()
                sc_8192_f = _score(a_8192_f)
                if sc_8192_f < best_score:
                    best_score = sc_8192_f
                    best_seq = a_8192_f.copy()
                    print(f"  n=8192 FISTA: score={sc_8192_f:.10f}")
                a_8192_cur = best_seq.copy() / best_seq.sum()
            else:
                a_8192_cur = a_8192

            # Adam on 8192
            if time.monotonic() < phase4_end - 20:
                c_8192_cur = _full_convolve(a_8192_cur, a_8192_cur)
                max_c8_cur = float(c_8192_cur.max())
                betas_8 = [0.0005 * max_c8_cur, 0.0002 * max_c8_cur, 0.0001 * max_c8_cur,
                           0.00005 * max_c8_cur]
                per_t = phase4_end - time.monotonic() - 10
                if per_t > 15:
                    a_8192_a, _ = _adam_logspace(a_8192_cur, time.monotonic() + per_t,
                                                  betas_8, lr_init=0.006, max_iter_total=100000)
                    sc_8192_a = _score(a_8192_a)
                    if sc_8192_a < best_score:
                        best_score = sc_8192_a
                        best_seq = a_8192_a.copy()
                        print(f"  n=8192 Adam: score={sc_8192_a:.10f}")

    # === Phase 5: Multi-restart with diverse perturbations ===
    print("Phase 5: Multi-restart diversification")
    phase5_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase5_end - 30 and best_seq is not None:
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic() - 15) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break

            # Diverse perturbation strategies
            if i == 0:
                p_start = _perturb(best_seq / best_seq.sum(), 0.015, rng)
            elif i == 1:
                p_start = _spectral_perturb(best_seq / best_seq.sum(), rng, n_modes=8)
            elif i == 2:
                # Circular shift
                sz = len(best_seq) // 128
                shift = rng.integers(1, max(2, sz))
                p_start = np.roll(best_seq / best_seq.sum(), shift).copy()
                p_start = p_start / p_start.sum()
            elif i == 3:
                # Mix with a different profile (raised cosine)
                n_cur = len(best_seq)
                t_lin = np.linspace(0, 1, n_cur)
                profile = 1.0 + 0.3 * np.cos(2.0 * np.pi * t_lin)
                profile = np.maximum(profile, 1e-10)
                profile = profile / profile.sum()
                p_start = 0.7 * (best_seq / best_seq.sum()) + 0.3 * profile
                p_start = p_start / p_start.sum()
            else:
                p_start = _perturb(best_seq / best_seq.sum(), 0.04, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs, 0.00004 * max_cs]

            t_restart = max(5.0, per_restart)
            p_best, _ = _adam_logspace(p_start, time.monotonic() + t_restart,
                                       bs, lr_init=0.007, max_iter_total=60000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final ultra-fine polish ===
    print("Phase 6: Final ultra-fine polish")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        # Very fine beta schedule
        betas_ult = [3e-5 * max_cp, 1.5e-5 * max_cp, 8e-6 * max_cp, 4e-6 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, _ = _adam_logspace(a_pol, time.monotonic() + per_t,
                                       betas_ult, lr_init=0.002, max_iter_total=150000)
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f}")

        # Final subgradient micro-steps
        if time.monotonic() < deadline - 3:
            a_final_sg = best_seq.copy() / best_seq.sum()
            a_final_sg, _ = _subgradient_bt(a_final_sg, deadline - 0.5,
                                            max_iter=5000, lr_init=0.05 / len(a_final_sg))
            sc_final_sg = _score(a_final_sg)
            if sc_final_sg < best_score:
                best_score = sc_final_sg
                best_seq = a_final_sg.copy()
                print(f"  Final subgrad: score={sc_final_sg:.10f}")

    # Final validation
    print(f"All phases complete: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
