"""
Multi-phase optimizer: LP direction finding at moderate sizes, then gradient
refinement (FISTA + Adam) at the target size of 4096.
"""
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_peak(a):
    """Return (score, peak_value, convolution)."""
    n = len(a)
    c = _full_convolve(a, a)
    peak = float(c.max())
    s = float(a.sum())
    sc = 2.0 * n * peak / (s * s) if s > 1e-15 else np.inf
    return sc, peak, c


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_lp_matrix(f, n):
    """Build sparse matrix A (2n-1 x n) where A[k,j] = f[k-j] if 0<=k-j<n."""
    n_rows = 2 * n - 1
    diags_data = []
    diags_offsets = []
    for d in range(n):
        length = min(n, n_rows - d)
        data = np.full(length, f[d])
        diags_data.append(data)
        diags_offsets.append(-d)
    A = sp.diags(diags_data, diags_offsets, shape=(n_rows, n), format='csr')
    return A


def _lp_solve(f, M, n):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0. Return g and success flag."""
    A = _build_lp_matrix(f, n)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x, True
        else:
            return None, False
    except Exception:
        return None, False


def _lp_phase(f_init, deadline, max_iters, verbose=True):
    """LP-based optimization phase. Returns best f (on simplex) and score."""
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_score = _score(f)
    
    if verbose:
        print(f"  LP start: n={n}, score={best_score:.8f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        M = float(_full_convolve(f, f).max())
        g, success = _lp_solve(f, M, n)
        
        if not success or g is None:
            break
        
        sum_g = float(g.sum())
        if sum_g <= 1.0 + 1e-10:
            # No improvement possible with this LP
            break
        
        # Try blending
        improved = False
        for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            f_new = (1 - t) * f + t * g / sum_g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            sc = _score(f_new)
            if sc < best_score - 1e-12:
                f = f_new
                best_score = sc
                best_f = f.copy()
                improved = True
                if verbose and it % 10 == 0:
                    print(f"    iter {it}: t={t}, score={sc:.8f}")
                break
        
        if not improved:
            # Try gradient step as fallback
            break
    
    return best_f, best_score

    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_lp_matrix(f, n):
    n_rows = 2 * n - 1
    diags_data = []
    diags_offsets = []
    for d in range(n):
        length = min(n, n_rows - d)
        data = np.full(length, f[d])
        diags_data.append(data)
        diags_offsets.append(-d)
    return sp.diags(diags_data, diags_offsets, shape=(n_rows, n), format='csr')


def _lp_solve(f, M, n):
    A = _build_lp_matrix(f, n)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return np.maximum(res.x, 0), True
        return None, False
    except Exception:
        return None, False


def _lp_phase(f_init, deadline, max_iters=200):
    n = len(f_init)
    f = f_init.copy() / f_init.sum()
    best_f = f.copy()
    best_score = _score(f)
    print(f"  LP phase start: n={n}, score={best_score:.8f}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t0 = time.monotonic()
        c = _full_convolve(f, f)
        M = float(c.max())
        
        g, ok = _lp_solve(f, M, n)
        if not ok or g is None:
            print(f"  LP failed at iter {it}")
            break
        
        sum_g = float(g.sum())
        if sum_g < 1.0 + 1e-12:
            print(f"  LP saturated at iter {it}, sum_g={sum_g:.6f}")
            break
        
        best_t = None
        best_sc = best_score
        for t in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05, 0.01]:
            f_new = (1.0 - t) * f + t * g / sum_g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                continue
            f_new = f_new / s
            sc = _score(f_new)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t is None:
            print(f"  No improvement at iter {it} (sum_g={sum_g:.4f})")
            break
        
        f = (1.0 - best_t) * f + best_t * g / sum_g
        f = np.maximum(f, 0.0) / f.sum()
        best_f = f.copy()
        best_score = best_sc
        
        if it % 20 == 0 or it < 5:
            dt = time.monotonic() - t0
            print(f"    iter {it}: t={best_t}, score={best_score:.8f}, sum_g={sum_g:.4f}, {dt:.1f}s")
    
    return best_f, best_score
