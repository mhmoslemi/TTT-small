def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 2.0 * n * a * (h - W)
    return grad


def _interp_profile(p, n_new):
    p = np.asarray(p, dtype=np.float64).ravel()
    n0 = len(p)
    if n0 == n_new:
        return p / p.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * t))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _cosine_power(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = np.maximum(np.cos(np.pi * u / 2.0), 1e-12) ** p
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    profiles.append(_raised_cosine(n))
    profiles.append(_cosine_power(n, 1.0))
    profiles.append(_cosine_power(n, 2.0))
    profiles.append(_cosine_power(n, 3.0))
    profiles.append(_cosine_power(n, 4.0))
    for p in [1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(1.0 - (u / 1.1) ** 2, 1e-12) ** p
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0]:
        a = np.exp(-rate * u * u)
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    return profiles


def _lbfgs_single(a_init, beta, deadline, maxiter=5000, m=25):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            _lse_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_lse_grad,
            options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                     'maxls': 50, 'maxfun': maxiter * 5, 'm': m, 'maxcor': 15}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, beta_fracs=None, m=25):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_fracs is None:
        beta_fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                      0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                      0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                      0.0002, 0.00015, 0.0001, 0.00008, 0.00006,
                      0.00005, 0.00004, 0.00003, 0.00002, 0.000015, 0.00001]

    n_betas = len(beta_fracs)
    for idx, bf in enumerate(beta_fracs):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(300, int(time_per * 40))
        beta = bf * max_c0

        a_cur, sc_cur = _lbfgs_single(best_a, beta, level_dl, maxiter=maxiter, m=m)
        if sc_cur < best_score:
            best_score = sc_cur
            best_a = a_cur.copy()
            max_c0 = float(_full_convolve(best_a, best_a).max())

    return best_a, best_score


def _anneal_optimize(a_init, deadline, rng, n_restarts=None, m=25):
    """Simulated annealing: perturb + re-optimize with decreasing noise."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if n_restarts is None:
        n_restarts = max(5, int((deadline - time.monotonic()) / 60))

    if n_restarts < 1:
        return best_a, best_score

    per_restart_time = (deadline - time.monotonic()) / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        # Noise level decreases with restart index
        noise_scale = 0.03 * (1.0 + r * 0.5) * (1.0 / (1.0 + r * 0.1))
        noise_scale = max(0.01, min(0.15, noise_scale))

        z = np.log(np.maximum(best_a, 1e-15))
        z = z - z.mean()
        z = z + noise_scale * rng.normal(size=n)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_cur = _softmax_fn(z)

        # Optimize this perturbed candidate
        level_dl = min(deadline - 5, time.monotonic() + per_restart_time * 0.8)
        beta_fracs_short = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        a_opt, sc_opt = _lbfgs_continuation(a_cur, level_dl, beta_fracs=beta_fracs_short, m=m)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            elapsed = time.monotonic()

    return best_a, best_score


def _fw_quick(a_init, n_steps=200, deadline=None):
    """Quick Frank-Wolfe iterations to reduce peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        # Subgradient at argmax
        a_rev = a[::-1].copy()
        e_k = np.zeros(2 * n - 1)
        e_k[k_star] = 1.0
        g = _full_convolve(e_k, a_rev)[n - 1:2 * n - 1]
        j_star = int(np.argmin(g))
        # FW step with diminishing step size
        t = 2.0 / (step + 2.0)
        a_new = (1.0 - t) * a + t * np.zeros(n)
        a_new[j_star] += t
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 60.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n_def = 4096
        parent = _raised_cosine(n_def)
        update_best(parent, _score_norm_fn(parent), "default")

    # ========== Try n=4096 (if parent is larger, downsample) ==========
    print("\n" + "=" * 60)
    print("Phase A: n=4096 optimization")
    print("=" * 60)

    n_A = 4096
    phase_a_end = min(deadline, t_start + 500.0)

    if len(parent) == n_A:
        a_A = parent.copy()
    elif len(parent) > n_A:
        a_A = _interp_profile(parent, n_A)
    else:
        a_A = _interp_profile(parent, n_A)

    sc_A = _score_norm_fn(a_A)
    update_best(a_A, sc_A, "init_n4096")

    # Quick FW
    fw_dl = min(phase_a_end - 300, t_start + 30)
    a_A, sc_A = _fw_quick(a_A, n_steps=100, deadline=fw_dl)
    update_best(a_A, sc_A, "fw_n4096")

    # L-BFGS continuation
    lbfgs_dl = min(phase_a_end - 150, t_start + 350)
    a_A, sc_A = _lbfgs_continuation(a_A, lbfgs_dl, m=25)
    update_best(a_A, sc_A, "lbfgs_n4096")

    # Annealing
    ann_dl = min(phase_a_end - 30, t_start + 470)
    a_A, sc_A = _anneal_optimize(a_A, ann_dl, rng, n_restarts=5, m=25)
    update_best(a_A, sc_A, "anneal_n4096")

    print(f"  Phase A done: best={best_score:.10f}")

    # ========== Try n=8192 ==========
    print("\n" + "=" * 60)
    print("Phase B: n=8192 optimization")
    print("=" * 60)

    n_B = 8192
    phase_b_end = deadline

    b_ref = best_seq if best_seq is not None else a_A
    a_B = _interp_profile(b_ref, n_B)
    sc_B = _score_norm_fn(a_B)
    update_best(a_B, sc_B, "interp_n8192")

    # Also try diverse initializations at n=8192
    profiles_8192 = _make_profiles(n_B)
    for idx, prof in enumerate(profiles_8192[:4]):
        sc_prof = _score_norm_fn(prof)
        if sc_prof < _score_norm_fn(a_B):
            a_B = prof.copy()
            update_best(a_B, sc_prof, f"profile{idx}_n8192")
        if time.monotonic() > phase_b_end - 120:
            break

    # FW quick
    fw_dl_B = min(phase_b_end - 600, t_start + 540)
    a_B, sc_B = _fw_quick(a_B, n_steps=100, deadline=fw_dl_B)
    update_best(a_B, sc_B, "fw_n8192")

    # L-BFGS continuation (longer time)
    lbfgs_dl_B = min(phase_b_end - 300, t_start + 1200)
    a_B, sc_B = _lbfgs_continuation(a_B, lbfgs_dl_B, m=30)
    update_best(a_B, sc_B, "lbfgs_n8192")

    # Annealing multi-restart
    ann_dl_B = min(phase_b_end - 60, t_start + budget_s - 80)
    a_B, sc_B = _anneal_optimize(a_B, ann_dl_B, rng, n_restarts=8, m=30)
    update_best(a_B, sc_B, "anneal_n8192")

    print(f"  Phase B done: best={best_score:.10f}")

    # ========== Final polish ==========
    print("\n" + "=" * 60)
    print("Phase C: Final polish")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 10:
        n_final = len(best_seq)
        polish_dl = deadline - 5
        # Very fine L-BFGS
        c_fin = _full_convolve(best_seq, best_seq)
        max_c_fin = float(c_fin.max())
        beta_ultra = 5e-6 * max_c_fin
        a_fin, sc_fin = _lbfgs_single(best_seq, beta_ultra, min(polish_dl, time.monotonic() + 30), maxiter=2000, m=30)
        update_best(a_fin, sc_fin, "ultra_polish")

        # One more anneal cycle with tiny noise
        if time.monotonic() < deadline - 20:
            small_restarts = max(2, int((deadline - time.monotonic() - 10) / 30))
            a_fin, sc_fin = _anneal_optimize(best_seq, deadline - 10, rng,
                                             n_restarts=small_restarts, m=30)
            update_best(a_fin, sc_fin, "final_anneal")

    # ========== Return ==========
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _raised_cosine(n_fb)
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
