    """Assume sum(a)=1, return 2*n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _cross_correlation(a, w, N):
    """Compute linear cross-correlation (a star w)[i] = sum_k a[k]*w[k+i] for i=0..len(a)-1."""
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    return np.fft.irfft(np.conj(A) * W, N)[:len(a)]


def _compute_grad(a, c, w, N):
    """Compute gradient in log-space: 2*a[i]*(g1[i] - g2) where g1 = cross-corr, g2 = dot(w,c)."""
    g1 = _cross_correlation(a, w, N)
    g2 = float(np.dot(w, c))
    return 2.0 * a * (g1 - g2)


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            j_range = np.arange(j_start, j_end + 1)
            rows_list.append(np.full(len(j_range), k, dtype=np.int32))
            cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _adam_optimize(a_init, deadline, N, beta_start_frac, beta_end_frac,
                   lr_start, lr_end, max_iter, patience):
    """
    Adam optimizer in log-space for the normalized problem.
    Returns (best_a, best_score_norm).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    
    t = 0
    no_improve = 0
    cur_lr = lr_start
    prev_sc = best_sc
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Compute beta (temperature) with exponential schedule
        frac = t / max_iter
        beta = 0.5 * (beta_start_frac + beta_end_frac * (beta_end_frac / beta_start_frac) ** frac)
        
        # Current point
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive beta in absolute terms
        beta_abs = beta * c_max
        
        # Softmax weights over convolution positions
        c_shifted = (c - c_max) / max(beta_abs, 1e-12)
        e = np.exp(c_shifted)
        w = e / e.sum()
        
        # Gradient
        grad = _compute_grad(a, c, w, N)
        
        # Adam update
        cur_lr = lr_start * (lr_end / lr_start) ** frac
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = cur_lr * mh / (np.sqrt(vh) + eps)
        
        # Clamp update
        update_norm = np.linalg.norm(update)
        if update_norm > 3.0:
            update = update * (3.0 / update_norm)
        
        z = z - update
        z = z - z.mean()
        
        # Periodic score check
        if t % 500 == 0:
            sc = 2.0 * n * c_max
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > patience:
                # Reduce learning rate
                cur_lr *= 0.7
                no_improve = 0
                
                # If stuck for too long, try a small perturbation
                if no_improve > patience * 3:
                    z = z + 0.05 * np.random.standard_normal(n)
                    z = z - z.mean()
                    no_improve = 0
    
    # Final check
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc


def _lp_phase(a, rows, cols, deadline, max_iters=50):
    """Run LP-based refinement."""
    n = len(a)
    prev_sc = _score_norm(a)
    best_a = a.copy()
    best_sc = prev_sc
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        M = float(_full_convolve(a, a).max())
        g_0 = _solve_lp(a, M, rows, cols, time_limit=20)
        if g_0 is None or g_0.sum() < 1e-15:
            break
        g = g_0 / g_0.sum()
        
        best_t = 0.0
        best_sc_local = _score_norm(a)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - t) * a + t * g
            sc = _score_norm(cand)
            if sc < best_sc_local:
                best_sc_local = sc
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        
        if stall > 15:
            break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        a = parent.copy()
        n = len(a)
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent loaded: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")
    
    N_fft = _next_pow2(2 * n - 1)
    
    # Try a few alternative initializations and pick best
    init_candidates = [a.copy()]
    
    # Raised cosine
    x = np.linspace(0, np.pi, n)
    rc = (0.5 + 0.5 * np.cos(x)) ** 1.5
    rc = np.maximum(rc, 1e-10)
    rc = rc / rc.sum()
    init_candidates.append(rc)
    
    # Gaussian centered
    xc = np.linspace(-1, 1, n)
    gauss = np.exp(-3.0 * xc ** 2)
    gauss = gauss / gauss.sum()
    init_candidates.append(gauss)
    
    # Triangular
    tri = np.maximum(0, 1 - np.abs(np.linspace(-1, 1, n)))
    tri = np.maximum(tri, 1e-10)
    tri = tri / tri.sum()
    init_candidates.append(tri)
    
    # Broad Gaussian
    gauss2 = np.exp(-1.5 * xc ** 2)
    gauss2 = gauss2 / gauss2.sum()
    init_candidates.append(gauss2)
    
    for ic in init_candidates:
        sc = _score(ic)
        if sc < best_score:
            best_score = sc
            best_seq = ic.copy()
        print(f"  Init candidate: score={sc:.10f}")
    
    a = best_seq.copy()
    print(f"Best init: score={best_score:.10f}")
    
    # Build LP indices (for later use)
    rows, cols = _build_lp_indices(n)
    print(f"LP indices built: {len(rows)} nonzeros, t={time.monotonic()-t_start:.1f}s")
    
    # === Phase 1: Main Adam optimization (0 - 700s) ===
    phase1_deadline = min(deadline, t_start + 700.0)
    if time.monotonic() < phase1_deadline:
        print(f"Phase 1: Adam optimization, target {phase1_deadline - time.monotonic():.0f}s")
        
        # Multiple passes with decreasing temperature
        passes = [
            (0.05, 0.005, 5.0, 2.0, 500000, 5),
            (0.02, 0.002, 3.0, 1.0, 500000, 5),
            (0.008, 0.001, 2.0, 0.5, 500000, 4),
            (0.003, 0.0003, 1.5, 0.3, 500000, 4),
            (0.001, 0.0001, 1.0, 0.1, 500000, 3),
        ]
        
        a = best_seq.copy()
        for pi, (b_s, b_e, lr_s, lr_e, m_iter, pat) in enumerate(passes):
            if time.monotonic() > phase1_deadline:
                break
            a, sc = _adam_optimize(a, phase1_deadline, N_fft, b_s, b_e,
                                    lr_s, lr_e, m_iter, pat)
            full_sc = _score(a)
            print(f"  Pass {pi}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
    
    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # === Phase 2: LP refinement (700 - 1100s) ===
    phase2_deadline = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase2_deadline:
        print(f"Phase 2: LP refinement, target {phase2_deadline - time.monotonic():.0f}s")
        a = best_seq.copy()
        a, sc = _lp_phase(a, rows, cols, phase2_deadline, max_iters=80)
        full_sc = _score(a)
        print(f"  LP result: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()
    
    # === Phase 3: Adam polish after LP (1100 - 1400s) ===
    phase3_deadline = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase3_deadline:
        print(f"Phase 3: Fine Adam polish, target {phase3_deadline - time.monotonic():.0f}s")
        a = best_seq.copy()
        
        fine_passes = [
            (0.001, 0.0001, 0.8, 0.2, 500000, 3),
            (0.0003, 0.00003, 0.5, 0.1, 500000, 3),
            (0.0001, 0.00001, 0.3, 0.05, 500000, 2),
        ]
        
        for pi, (b_s, b_e, lr_s, lr_e, m_iter, pat) in enumerate(fine_passes):
            if time.monotonic() > phase3_deadline:
                break
            a, sc = _adam_optimize(a, phase3_deadline, N_fft, b_s, b_e,
                                    lr_s, lr_e, m_iter, pat)
            full_sc = _score(a)
            print(f"  Fine pass {pi}: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()
    
    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # === Phase 4: Multi-restart exploration (1400 - 1650s) ===
    phase4_deadline = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase4_deadline:
        print(f"Phase 4: Multi-restart, target {phase4_deadline - time.monotonic():.0f}s")
        
        for attempt in range(5):
            if time.monotonic() > phase4_deadline - 30:
                break
            
            # Perturb in log-space
            z = np.log(np.maximum(best_seq, 1e-300))
            sigma = 0.02 + 0.01 * attempt
            z = z + sigma * rng.standard_normal(n)
            z = z - z.mean()
            a_restarts = _softmax(z)
            
            # Quick Adam from perturbed start
            a_r, sc_r = _adam_optimize(a_restarts, phase4_deadline - 10, N_fft,
                                        0.005, 0.0005, 2.0, 0.5, 200000, 4)
            full_sc = _score(a_r)
            print(f"  Restart[{attempt}] (sig={sigma:.3f}): score={full_sc:.10f}, "
                  f"t={time.monotonic()-t_start:.0f}s")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a_r.copy()
    
    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # === Phase 5: Final ultra-fine polish (1650 - deadline) ===
    if time.monotonic() < deadline - 10:
        print(f"Phase 5: Final polish, target {deadline - time.monotonic():.0f}s")
        a = best_seq.copy()
        a, sc = _adam_optimize(a, deadline - 5, N_fft,
                                0.0001, 0.00001, 0.2, 0.02, 500000, 2)
        full_sc = _score(a)
        print(f"  Final polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()
    
    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
