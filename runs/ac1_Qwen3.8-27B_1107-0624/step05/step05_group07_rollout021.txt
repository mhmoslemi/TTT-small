"""
Optimized hybrid approach for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based minimax, L-BFGS with ultra-fine log-sum-exp continuation,
active-set subgradient descent, and coordinate descent polish.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c))


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    row_idx = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    data = np.repeat(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f):
    """Solve: min t s.t. conv(f,g) <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_sum = g_val.sum()
            if g_sum > 1e-15:
                g_val = g_val / g_sum
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters, deadline, step_size):
    """Run LP-based iterated minimax."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(f, f)
        M = float(c.max())
        g, t_val = _lp_minimax(f)
        if g is None:
            break
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
        if it % 20 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    
    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # Gradient of LSE w.r.t. a: 2 * sum_k w_k * a_{k-j}
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h  # gradient of LSE w.r.t. a
    # Chain rule: dLSE/dz_j = a_j * (grad_a_j - sum_i a_i * grad_a_i)
    # Actually, since a = softmax(z), da_j/dz_k = a_j*(delta_jk - a_k)
    # So dLSE/dz_k = sum_j (dLSE/da_j) * (da_j/dz_k)
    # = sum_j grad_a_j * a_j * (delta_jk - a_k)
    # = a_k * (grad_a_k - sum_j a_j * grad_a_j)
    W2 = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W2)
    return 2.0 * n * grad_z


def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002,
                 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00002, 0.00001, 0.000005]
        beta_list = [b * max_c0 for b in fracs]
    
    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 90.0)
        maxiter = max(300, int(time_per * 40))
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,

                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:

def _active_set_subgrad(a_init, deadline, max_iter=50000):
    """Direct subgradient method on the active set with Nesterov acceleration."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Nesterov momentum
    z_prev = z.copy()
    momentum = 0.9
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        score = 2.0 * n * cm
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active lags (within 1% of max)
        active = np.where(c >= 0.99 * cm)[0]
        
        # Subgradient: average of gradients at active lags
        # grad c_k / da_j = 2 * a_{k-j} if 0 <= k-j < n
        grad_a = np.zeros(n)
        for k in active:
            for j in range(n):
                i = k - j
                if 0 <= i < n:
                    grad_a[j] += 2.0 * a[i] / len(active)
        
        # Chain rule through softmax
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        
        # Nesterov: extrapolate
        z = z + momentum * (z - z_prev)
        z_prev = z.copy()
        
        # Step
        # Adaptive step size: use Polyak-like
        # We need a lower bound. Since max(c) >= sum(c)/(2n-1) = sum(a^2)/(2n-1) >= 1/(n*(2n-1))
        # with sum(a)=1, sum(a^2) >= 1/n, so lower bound on 2n*max(c) is 2n/(n*(2n-1)) = 2/(2n-1) ≈ 1/n
        # Actually that's very loose. Let's use the uniform value 2 as an upper bound
        # and 0 as lower bound (trivially).
        # Better: use the current best as upper bound, and estimate lower bound.
        # For simplicity, use a decreasing step size.
        eta = 0.5 / ((it + 1) ** 0.5)
        # Actually, for mirror descent, the step size should be ~ 1/sqrt(t)
        # But since we're working in z-space (log space), the geometry is different.
        # Let's use a fixed small step size with occasional restarts.
        eta = 0.1
        z = z - eta * grad_z
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if it % 5000 == 0:
            print(f"    AS iter {it}: score={score:.10f}, best={best_score:.10f}, |active|={len(active)}")
        
        # Regular noise injection to escape local minima
        if it % 20000 == 10000:
            z = z + 0.01 * np.random.randn(n)
            z = z - z.mean()
    
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=100):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_local_score = _score_norm_fn(a)
            best_delta = 0.0
            for delta in [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local_score - 1e-12:
                    best_local_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 2.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
