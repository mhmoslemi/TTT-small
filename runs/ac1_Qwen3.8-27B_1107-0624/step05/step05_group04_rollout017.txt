def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

        return np.zeros(len(z))
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    n = len(z)
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _fista(z_init, beta, deadline, lr_init=None, max_iter=30000):
    n = len(z_init)
    z = z_init.copy()
    z = z - z.mean()
    z_prev = z.copy()
    t = 1.0
    best_z = z.copy()
    best_f = _lse_obj(z, beta)

    g0 = _lse_grad(z, beta)
    ng = np.linalg.norm(g0)
    if ng < 1e-15:
        lr = 50.0
    else:
        lr = min(50.0, 0.5 / (ng + 1e-12))
    if lr_init is not None:
        lr = lr_init

    f_val = best_f
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = z + momentum * (z - z_prev)

        g = _lse_grad(y, beta)
        if not np.all(np.isfinite(g)):
            break

        step = lr
        f_y = _lse_obj(y, beta)
        z_new = y - step * g
        f_new = _lse_obj(z_new, beta)

        max_bt = 8
        for _ in range(max_bt):
            if np.isfinite(f_new) and f_new <= f_y + 0.5 * step * step * np.dot(g, g):
                break
            step *= 0.5
            z_new = y - step * g
            f_new = _lse_obj(z_new, beta)
            if step < 1e-12:
                break

        if np.isfinite(f_new) and f_new < best_f:
            best_f = f_new
            best_z = z_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                break

        z_prev = z
        z = z_new
        z = z - z.mean()
        t = t_new

        if it % 500 == 0:
            lr = min(lr * 1.05, 200.0)

    return best_z, _softmax_fn(best_z)


def _lbfgs_refine(z_init, beta, deadline, maxiter=10000):
    n = len(z_init)
    try:
        result = minimize(
            _lse_obj, z_init, args=(beta,),
            method='L-BFGS-B', jac=_lse_grad,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': maxiter * 8}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            z_out = result.x
            z_out = np.clip(z_out, -80, 80)
            z_out = z_out - z_out.mean()
            a_out = _softmax_fn(z_out)
            return z_out, a_out
    except Exception:
        pass
    return z_init, _softmax_fn(z_init)


def _multi_scale_optimize(a_start, deadline, rng):
    n = len(a_start)
    best_a = a_start.copy()
    best_sc = _score_norm_fn(a_start)
    z = np.log(np.maximum(a_start, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_start, a_start)
    max_c0 = float(c0.max())
    if max_c0 < 1e-15:
        return best_a, best_sc

    # FISTA phase with decreasing beta
    betas_fista = [0.2 * max_c0, 0.05 * max_c0, 0.01 * max_c0,
                   0.003 * max_c0, 0.001 * max_c0]

    time_per_beta = (deadline - time.monotonic()) / (len(betas_fista) + len(betas_fista))
    for i, beta in enumerate(betas_fista):
        if time.monotonic() > deadline - 30:
            break
        bdl = min(deadline - 20, time.monotonic() + time_per_beta * 1.5)
        z, a_f = _fista(z, beta, bdl, max_iter=10000)
        sc = _score_norm_fn(a_f)
        if sc < best_sc:
            best_sc = sc
            best_a = a_f.copy()

    # L-BFGS phase with finer betas
    betas_lbfgs = [0.0005 * max_c0, 0.0001 * max_c0, 0.00003 * max_c0,
                   0.00001 * max_c0, 0.000003 * max_c0, 0.000001 * max_c0,
                   0.0000003 * max_c0, 0.0000001 * max_c0]

    remaining = deadline - time.monotonic()
    time_per_lbfgs = remaining / len(betas_lbfgs)

    for i, beta in enumerate(betas_lbfgs):
        if time.monotonic() > deadline - 5:
            break
        bdl = min(deadline - 3, time.monotonic() + time_per_lbfgs * 1.2)
        maxit = min(15000, max(2000, int(time_per_lbfgs * 200)))
        z, a_l = _lbfgs_refine(z, beta, bdl, maxiter=maxit)
        sc = _score_norm_fn(a_l)
        if sc < best_sc:
            best_sc = sc
            best_a = a_l.copy()

    return best_a, best_sc


def _upsample_cubic(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    # Use cubic interpolation via scipy
    from scipy.interpolate import CubicSpline
    try:
        cs = CubicSpline(x0, a, bc_type='not-a-knot')
        new_a = cs(x1)
        new_a = np.maximum(new_a, 1e-15)
    except Exception:
        new_a = np.interp(x1, x0, a)
        new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _coordinate_descent(a, deadline, max_rounds=500):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Try adjusting each coordinate (sample if n is large)
        if n <= 2048:
            indices = list(range(n))
            np.random.shuffle(indices)
        else:
            indices = list(np.random.choice(n, size=2048, replace=False))

        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break

            # Compute effect of changing a_j
            # db_k/da_j = a_{k-j} (if valid) + a_{j-k} (if valid)
            # We want to find the direction that reduces the max

            # Try small perturbations in log-space
            for delta in [0.01, 0.005, 0.02, 0.002, -0.01, -0.005, -0.02, -0.002]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                c_trial = _full_convolve(a_trial, a_trial)
                cm_trial = float(c_trial.max())
                sc_trial = 2.0 * n * cm_trial
                if sc_trial < best_sc - 1e-14:
                    best_sc = sc_trial
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break

        if not improved and rnd > 50:
            break

    return best_a, best_sc


def _peak_equalize(a, deadline, max_rounds=200):
    """Directly try to equalize the top peaks by adjusting coordinates."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Get top peaks
        peak_idx = np.argsort(c)[::-1][:5]

        # For each peak, find the coordinate with the largest marginal contribution
        # and try to reduce it
        improved = False
        for pk in peak_idx[:3]:
            if time.monotonic() > deadline:
                break

            # For lag pk, contribution from a[j] is a[j] * a[pk-j] (if valid)
            valid_j = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_j) == 0:
                continue

            # Marginal effect: da[j] changes c[pk] by a[pk-j] + a[j-pk] (approx)
            # But more precisely, c[pk] = sum_i a[i]*a[pk-i]
            # dc[pk]/da[j] = a[pk-j] (if 0<=pk-j<n) + a[j-pk] (if 0<=j-pk<n)
            # which equals 2*a[pk-j] when pk >= j (by symmetry of autocorrelation)
            # Actually for autocorrelation: dc[pk]/da[j] = a[pk-j] + a[j-pk] if both valid
            # But for the full convolution c = conv(a,a): c[k] = sum_i a[i]*a[k-i]
            # dc[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if n-k<=j<n+k, carefully)

            # Simplified: just find the j that maximizes a[j] * a[pk-j] and reduce it
            prods = np.zeros(n)
            for j in valid_j:
                other = pk - j
                if 0 <= other < n:
                    prods[j] = a_cur[j] * a_cur[other]

            top_contribs = np.argsort(prods)[::-1][:10]
            for j in top_contribs[:5]:
                if prods[j] < 1e-15:
                    continue
                for delta in [0.01, 0.005, 0.02, 0.003, 0.03]:
                    z_trial = z.copy()
                    z_trial[j] -= delta
                    a_trial = _softmax_fn(z_trial)
                    c_trial = _full_convolve(a_trial, a_trial)
                    cm_trial = float(c_trial.max())
                    sc_trial = 2.0 * n * cm_trial
                    if sc_trial < best_sc - 1e-14:
                        best_sc = sc_trial
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing small coordinates (to spread energy)
            small_idx = np.argsort(a_cur)[:10]
            for j in small_idx:
                for delta in [0.01, 0.005, 0.02]:
                    z_trial = z.copy()
                    z_trial[j] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc_trial = _score_norm_fn(a_trial)
                    if sc_trial < best_sc - 1e-14:
                        best_sc = sc_trial
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rnd > 30:
            break

    return best_a, best_sc


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0]:
        profiles.append(_make_ramp(n, s))
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # S-curve
    for sharp in [5, 10, 20]:
        a = 1.0 / (1.0 + np.exp(-sharp * (t - 0.5)))
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Quadratic
    for c in [0.5, -0.5, 1.0, -1.0]:
        a = 1.0 + 1.5 * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(512, 1.5)
        update_best(a0, _score_norm_fn(a0), "default")
        parent = a0

    n_parent = len(parent)

    # Phase 1: Direct optimization from parent (if large enough)
    if n_parent >= 1024:
        print(f"\nPhase 1: Direct optimization from n={n_parent}")
        phase1_deadline = t_start + (budget_s - 20.0) * 0.55
        a_opt, sc_opt = _multi_scale_optimize(parent, phase1_deadline, rng)
        update_best(a_opt, sc_opt, "phase1_direct")
    else:
        # Multi-scale from smaller
        print(f"\nPhase 1: Multi-scale from n={n_parent}")
        phase1_deadline = t_start + (budget_s - 20.0) * 0.50
        a_cur = parent.copy()
        sc_cur = _score_norm_fn(a_cur)

        scales = [256, 512, 1024, 2048, 4096, 8192]
        time_allocs = [20, 25, 40, 80, 150, 200]
        total_time = sum(time_allocs)

        for i, target_n in enumerate(scales):
            if target_n <= n_parent:
                continue
            if time.monotonic() > phase1_deadline - 10:
                break

            a_up = _upsample_cubic(a_cur, target_n)
            sc_up = _score_norm_fn(a_up)
            print(f"  Upsampled to n={target_n}: score={sc_up:.10f}")

            alloc = time_allocs[i] * (phase1_deadline - time.monotonic()) / (phase1_deadline - t_start - 10) if phase1_deadline > time.monotonic() else 10
            alloc = min(alloc, 300.0)
            opt_dl = min(phase1_deadline - 5, time.monotonic() + alloc)

            a_opt, sc_opt = _multi_scale_optimize(a_up, opt_dl, rng)
            print(f"  After n={target_n} opt: score={sc_opt:.10f}")

            if sc_opt < sc_cur:
                a_cur = a_opt
                sc_cur = sc_opt

        update_best(a_cur, sc_cur, "phase1_multiscale")

    print(f"\n  Phase 1 done: best={best_score:.10f}")

    # Phase 2: More aggressive L-BFGS from best
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\nPhase 2: Aggressive L-BFGS refinement")
        phase2_dl = t_start + (budget_s - 20.0) * 0.75
        n_best = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        mc_best = float(c_best.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        # Very fine beta schedule
        fine_betas = [1e-4 * mc_best, 1e-5 * mc_best, 1e-6 * mc_best,
                      1e-7 * mc_best, 1e-8 * mc_best]
        remaining = phase2_dl - time.monotonic()
        per_beta = remaining / len(fine_betas)

        for beta in fine_betas:
            if time.monotonic() > phase2_dl - 5:
                break
            bdl = min(phase2_dl - 3, time.monotonic() + per_beta * 1.1)
            maxit = min(20000, max(5000, int(per_beta * 100)))
            z, a_l = _lbfgs_refine(z, beta, bdl, maxiter=maxit)
            sc = _score_norm_fn(a_l)
            if sc < best_score:
                best_score = sc
                best_seq = a_l.copy()
                print(f"    L-BFGS beta={beta:.2e}: {best_score:.10f}")

    print(f"\n  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Coordinate descent + peak equalization
    if time.monotonic() < deadline - 15 and best_seq is not None:
        print(f"\nPhase 3: Coordinate descent + peak equalization")
        phase3_dl = t_start + (budget_s - 20.0) * 0.92

        # Peak equalization
        pe_dl = min(phase3_dl - 30, time.monotonic() + (phase3_dl - time.monotonic()) * 0.4)
        a_pe, sc_pe = _peak_equalize(best_seq, pe_dl, max_rounds=500)
        update_best(a_pe, sc_pe, "peak_equalize")

        # Coordinate descent
        if time.monotonic() < phase3_dl - 5:
            cd_dl = min(phase3_dl - 3, time.monotonic() + (phase3_dl - time.monotonic()) * 0.5)
            a_cd, sc_cd = _coordinate_descent(best_seq, cd_dl, max_rounds=300)
            update_best(a_cd, sc_cd, "coord_descent")

    print(f"\n  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Final L-BFGS polish with very tight settings
    if time.monotonic() < deadline - 5 and best_seq is not None:
        print(f"\nPhase 4: Final polish")
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        mc_final = float(c_final.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        # Final ultra-fine L-BFGS
        beta_final = 1e-9 * mc_final
        bdl = min(deadline - 2, time.monotonic() + 60.0)
        z, a_fin = _lbfgs_refine(z, beta_final, bdl, maxiter=20000)
        sc_fin = _score_norm_fn(a_fin)
        update_best(a_fin, sc_fin, "final_lbfgs")

        # One more FISTA pass
        if time.monotonic() < deadline - 3:
            z2 = np.log(np.maximum(best_seq, 1e-15))
            z2 = z2 - z2.mean()
            bdl2 = min(deadline - 1, time.monotonic() + 30.0)
            z2, a_f2 = _fista(z2, beta_final, bdl2, max_iter=20000)
            sc_f2 = _score_norm_fn(a_f2)
            update_best(a_f2, sc_f2, "final_fista")

    print(f"\n  Phase 4 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(512, 1.5)

        # Verify with evaluate_sequence
        sc_final = evaluate_sequence([float(x) for x in final_seq])
        print(f"\nFINAL: eval_score={sc_final:.10f}, n={len(final_seq)}")

        if not np.isfinite(sc_final):
            # Fallback
            fallback = _make_ramp(512, 1.5)
            print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
            return [float(x) for x in fallback]

        return [float(x) for x in final_seq]

    # Absolute fallback
    fallback = _make_ramp(512, 1.5)
    print(f"FALLBACK: n=512, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
