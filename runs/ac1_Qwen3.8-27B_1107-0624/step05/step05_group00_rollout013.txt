    while beta >= beta_end and total_iters < max_iter:
        if time.monotonic() > deadline:

            a_next = y - lr * grad
            a_next = _proj_nonneg_sum1(a_next)

            if total_iters % 50 == 0:

def _adam_optimize(a_init, deadline, beta_schedule, lr_init=0.01, max_iter=200000):
    """
    Adam optimizer in log-space with softmax parameterization.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        while t < max_iter:
            if time.monotonic() > deadline:
                break
            
            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            c_max, w = _smoothed_max_and_weights(c, beta)
            grad = _compute_grad_z(a, c, w, n)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 100 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """Gradient of 2n*smoothed_max(conv(a,a)) w.r.t. z in softmax(a) = exp(z)/sum(exp(z))."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    # Gradient w.r.t. a: 2n * 2 * h = 4n * h (from the smoothed max approximation)
    # Chain rule through softmax: grad_z = a * (grad_a - sum(a * grad_a))
    grad_a = 4.0 * n * h
    grad_z = a * (grad_a - np.dot(a, grad_a))
    # Divide by W (the normalizing constant of the softmax in the smoothed max)
    # Actually, the smoothed max is log(sum(exp(c/beta))) = c_max + log(sum(exp((c-c_max)/beta)))
    # Its gradient w.r.t. a is sum_k w[k] * dc[k]/da where w[k] = exp((c[k]-c_max)/beta) / sum(exp(...))
    # So grad_a = 2n * sum_k w[k] * dc[k]/da = 2n * 2 * h = 4n * h
    # This is already correct since w is normalized.
    return grad_z


def _make_uniform(n):
    return np.ones(n) / n


def _make_cosine(n):
    t = np.linspace(0, np.pi, n)
    a = 0.5 + 0.5 * np.cos(t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_raised_cosine(n):
    t = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_sinc_like(n):
    t = np.linspace(-1, 1, n)
    a = np.sinc(t) ** 2  # sinc^2 is always non-negative
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_gaussian(n):
    t = np.linspace(-1, 1, n)
    a = np.exp(-t**2 / (2 * 0.3**2))
    return a / a.sum()


def _make_bump(n):
    """A bump function that's concentrated in the middle."""
    t = np.linspace(-1, 1, n)
    a = np.exp(-t**2 / (2 * 0.5**2))
    # Add a small uniform component to ensure it's not too peaked
    a = 0.7 * a + 0.3 / n
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _peak_reduction(a, n_peaks=10, n_steps=100, lr=0.01):

    best_a = a.copy()
    best_score = _score_norm(a)
    
    for step in range(n_steps):
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top peaks
        # Use a small beta focused on the peaks
        beta = 0.001 * c_max
        c_shifted = (c - c_max) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h
        
        # Gradient step with projection
        a_new = a - lr * grad
        a_new = _proj_nonneg_sum1(a_new)
        
        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        else:
            # Try smaller step
            a_new = a - 0.5 * lr * grad
            a_new = _proj_nonneg_sum1(a_new)
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = best_a.copy()
            lr *= 0.9  # Decay learning rate
        
        a = best_a.copy()
    
    return best_a, best_score
