        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxls': 40, 'disp': False}
            )
            if time.monotonic() > level_dl:
                break
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -70, 70)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _peak_shave(a, deadline, max_rounds=500):
    """Systematic single-element peak shaving."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_pos = int(np.argmax(c))

        # Find top contributors to the peak
        valid_i = np.arange(max(0, peak_pos - n + 1), min(n, peak_pos + 1))
        if len(valid_i) == 0:
            break
        contrib = a_cur[valid_i] * a_cur[peak_pos - valid_i]
        order = np.argsort(contrib)[::-1][:10]

        improved = False
        for idx_pos in order:
            idx = valid_i[idx_pos]
            for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements
            small_idx = np.argsort(a_cur)[:5]
            for idx in small_idx:
                for delta in [0.002, 0.005, 0.01, 0.03]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rnd > 50:
            break

    return best_a, best_score


def _spectral_balance(a, deadline, max_iter=2000):
    """Use FFT structure to identify and reduce spectral peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top 5 peaks and compute combined gradient
        top_idx = np.argsort(c)[::-1][:5]
        # Simple average of peak gradients
        grad = np.zeros(n)
        for ki in top_idx:
            i_min = max(0, ki - n + 1)
            i_max = min(n - 1, ki)
            for i in range(i_min, i_max + 1):
                grad[i] += a_cur[ki - i]

        grad /= len(top_idx)
        grad_centered = grad - np.mean(grad)

        # Adaptive step
        a_trial = a_cur - 0.01 * grad_centered
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        sc = _score_norm_fn(a_trial)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()
        else:
            if it % 100 == 0:
                # Add small noise to escape
                noise = 0.02 * np.random.default_rng(it + 1000).normal(size=n)
                z = z + noise
                z = z - z.mean()
            else:
                if it > 200:
                    break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Main search function combining LP, gradient, and spectral methods."""
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(4096, 1.5)
        update_best(a0, _score_norm_fn(a0), "default_ramp")

    n_best = len(best_seq)
    print(f"Starting with n={n_best}, score={best_score:.10f}")

    # Phase 1: LP-based iterations (30% of time)
    phase1_end = t_start + (budget_s - 30.0) * 0.30
    print(f"\nPhase 1: LP iterations (until t={phase1_end - t_start:.0f}s)")
    if time.monotonic() < phase1_end - 5:
        a_lp, sc_lp = _run_lp_iterations(best_seq, phase1_end, num_active=25, max_iter=300)
        update_best(a_lp, sc_lp, "phase1_lp")

    # Phase 2: L-BFGS continuation (25% of time)
    phase2_end = t_start + (budget_s - 30.0) * 0.55
    print(f"\nPhase 2: L-BFGS continuation (until t={phase2_end - t_start:.0f}s)")
    if time.monotonic() < phase2_end - 5:
        a_lb, sc_lb = _run_lbfgs_continuation(best_seq, phase2_end, n_betas=15)
        update_best(a_lb, sc_lb, "phase2_lbfgs")

    # Phase 3: Peak gradient descent (20% of time)
    phase3_end = t_start + (budget_s - 30.0) * 0.75
    print(f"\nPhase 3: Peak gradient descent (until t={phase3_end - t_start:.0f}s)")
    if time.monotonic() < phase3_end - 5:
        a_pg, sc_pg = _run_peak_gradient_descent(best_seq, phase3_end, num_peaks=10, lr_init=0.01)
        update_best(a_pg, sc_pg, "phase3_pg")

    # Phase 4: Peak shaving (15% of time)
    phase4_end = t_start + (budget_s - 30.0) * 0.90
    print(f"\nPhase 4: Peak shaving (until t={phase4_end - t_start:.0f}s)")
    if time.monotonic() < phase4_end - 5:
        a_ps, sc_ps = _peak_shave(best_seq, phase4_end, max_rounds=500)
        update_best(a_ps, sc_ps, "phase4_shave")

    # Phase 5: Spectral balance + restarts (10% of time)
    print(f"\nPhase 5: Spectral balance + restarts")
    if time.monotonic() < deadline - 5:
        # Spectral balance
        sb_end = min(deadline - 30, time.monotonic() + 30)
        a_sb, sc_sb = _spectral_balance(best_seq, sb_end, max_iter=2000)
        update_best(a_sb, sc_sb, "phase5_spectral")

        # Restart with perturbation
        if time.monotonic() < deadline - 10:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            for restart in range(3):
                if time.monotonic() > deadline - 5:
                    break
                noise_scale = 0.05 * (restart + 1)
                z_pert = z + noise_scale * rng.normal(size=n_best)
                z_pert = np.clip(z_pert, -60, 60)
                z_pert = z_pert - z_pert.mean()
                a_pert = _softmax_fn(z_pert)
                a_opt, sc_opt = _run_lbfgs_continuation(a_pert, min(deadline - 3, time.monotonic() + 20), n_betas=8)
                update_best(a_opt, sc_opt, f"restart_{restart}")
                if time.monotonic() < deadline - 20:
                    a_opt2, sc_opt2 = _peak_shave(a_opt, min(deadline - 3, time.monotonic() + 15), max_rounds=100)
                    update_best(a_opt2, sc_opt2, f"restart_{restart}_shave")

    # Final polish
    print(f"\nFinal polish")
    if time.monotonic() < deadline - 3 and best_seq is not None:
        a_final, sc_final = _peak_shave(best_seq, deadline, max_rounds=200)
        update_best(a_final, sc_final, "final_poly")

    # Return best
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
