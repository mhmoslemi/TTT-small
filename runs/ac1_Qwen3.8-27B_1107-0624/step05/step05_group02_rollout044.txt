with AdamW optimizer and soft-max gradient. Uses multi-resolution approach
and exploits the analytical structure of the gradient.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    # db_j/dz_m = b_j * (delta_jm - b_m)
    # 
    # grad_z_m = sum_j df/db_j * db_j/dz_m
    # = sum_j 4n * h_j * b_j * (delta_jm - b_m)
    # = 4n * h_m * b_m - 4n * b_m * sum_j h_j * b_j
    # = 4n * b_m * (h_m - dot(h, b))
    #
    # Note: dot(h, b) = sum_j h_j * b_j = sum_j b_j * sum_k w_k * b(k-j)
    # = sum_k w_k * sum_j b_j * b(k-j) = sum_k w_k * c(k) = W_val
    #
    # So: grad_z_m = 4n * b_m * (h_m - W_val)
    
    grad = 4.0 * n * b * (h - W_val)
    
    return grad, M, W_val

def _adamw_optimize(a_init, deadline, lr=0.005, beta_start=0.1, beta_end=0.001,
                    beta_decay=0.999, weight_decay=0.01, max_iter=None):
    """
    AdamW optimizer in log-space for minimizing 2n*max(conv(b,b)).
    b is kept normalized (sum=1) throughout.
    """
    n = len(a_init)
    # Work in log-space: z = log(b)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()  # Center for numerical stability
    
    b = _softmax(z)
    c = _full_convolve(b, b)
    best_score = float(c.max())
    best_b = b.copy()
    
    # AdamW state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    beta = beta_start
    current_lr = lr
    stagnant_count = 0
    patience = 500
    
    if max_iter is None:
        max_iter = 10**9
    
    while t_step < max_iter:
        if time.monotonic() > deadline:
            break
        
        t_step += 1
        beta *= beta_decay
        beta = max(beta, beta_end)
        
        b = _softmax(z)
        c = _full_convolve(b, b)
        M = float(c.max())
        
        if np.isfinite(M) and M < best_score:
            best_score = M
            best_b = b.copy()
            stagnant_count = 0
            if t_step % 200 == 0:
                print(f"    Improved: M={M:.10f}, score={2*n*M:.10f}, t={t_step}")
        else:
            stagnant_count += 1
        
        if stagnant_count > patience:
            # Decay learning rate
            current_lr *= 0.5
            stagnant_count = 0
            if current_lr < 1e-6:
                break
        
        # Compute gradient
        grad, M, W_val = _compute_grad(c, b, beta)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # AdamW update
        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t_step)
        vh = v_buf / (1 - b2 ** t_step)
        
        # Weight decay (applied to z, which is log(b))
        update = current_lr * (mh / (np.sqrt(vh) + eps) + weight_decay * z)
        
        # Clip update for stability
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -30.0, 30.0)
        z = z - z.mean()  # Renormalize
    
    b_final = _softmax(z)
    c_final = _full_convolve(b_final, b_final)
    sc_final = float(c_final.max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_b = b_final.copy()
    
    return best_b, 2.0 * n * best_score

def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _make_hann(n):
    """Create a Hann window normalized to sum 1."""
    k = np.arange(n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * k / (n - 1)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_kaiser(n, alpha=5.0):
    """Create a Kaiser window normalized to sum 1."""
    from scipy.special import i0
    k = np.arange(n)
    norm = (n - 1) / 2.0
    x = 1.0 - (k - norm) / norm
    x = np.clip(x, -1.0, 1.0)
    a = np.ones(n)
    for i in range(n):
        a[i] = i0(alpha * np.sqrt(max(0, 1 - x[i]**2))) / i0(alpha)
    a = np.maximum(a, 1e-12)
    return a / a.sum()
