def _next_pow2(n):

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _run_lbfgs(a_init, beta, deadline, maxiter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            _logsumexp_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_logsumexp_grad,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                     'maxls': 50, 'maxfun': maxiter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm_fn(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score

def _pair_coord_descent(a_init, deadline, max_rounds=20):
    """Coordinate descent where we move pairs of coordinates (mass transfer)."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        improved = False
        # Try random pairs
        n_pairs = max(50, n // 4)
        for _ in range(n_pairs):
            if time.monotonic() > deadline:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Try transferring mass from i to j and vice versa
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[i] += delta
                z_trial[j] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[i] += best_delta
                z[j] -= best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
        if rnd % 5 == 0:
            print(f"    Pair CD round {rnd}: best={best_score:.10f}")
    
    return best_a, best_score

def _simulated_annealing(a_init, deadline, rng, n_steps=5000):
    """Simulated annealing with structured perturbations."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    T = 0.1  # Initial temperature
    T_min = 0.001
    cooling = 0.9995
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        T = max(T_min, T * cooling)
        
        # Structured perturbation: add a small smooth perturbation
        # Use a random low-frequency modulation
        freq = rng.integers(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        amp = T * 0.5
        perturbation = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        
        # Also add some random high-frequency noise
        noise = T * 0.1 * rng.normal(size=n)
        
        z_trial = z + perturbation + noise
        z_trial = np.clip(z_trial, -80, 80)
        z_trial = z_trial - z_trial.mean()
        a_trial = _softmax_fn(z_trial)
        sc_trial = _score_norm_fn(a_trial)
        sc_current = _score_norm_fn(a_init)  # Use current, not best
        
        # Metropolis acceptance
        if sc_trial < best_score or rng.random() < np.exp(-(sc_trial - best_score) / max(T, 1e-10)):
            z = z_trial.copy()
            a_new = _softmax_fn(z)
            sc_new = _score_norm_fn(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
            a_init = a_new.copy()
    
    return best_a, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    return profiles

def _lbfgs_continuation(a_init, deadline):
    """Run L-BFGS with decreasing beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    fracs = [0.05, 0.03, 0.02, 0.01, 0.008, 0.005, 0.003, 0.002,
             0.001, 0.0008, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]
    
    n_betas = len(fracs)
    for idx, frac in enumerate(fracs):
        if time.monotonic() > deadline:
            break
        beta = frac * max_c0
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 40))
        
        a_cur = _softmax_fn(z)
        sc_cur = _score_norm_fn(a_cur)
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score
