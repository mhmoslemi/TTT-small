"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses L-BFGS with log-sum-exp smoothing, Nesterov subgradient polishing, LP-based peak shaving,
and harmonic parameterization for diverse restarts.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(z, a, c, n, tol_frac=0.005):
    c_max = float(c.max())
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, n, max_iter=2000, ftol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10,
                     'maxcor': 40, 'maxls': 200, 'disp': False}
        )
        a_out = _softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False


def _nesterov_polish(a_init, deadline, max_steps=5000, lr0=0.04, tol_frac=0.003):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    v = np.zeros(n)
    best_a = a_init.copy()
    best_sc = _score(a_init)
    mu = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(z, a, c, n, tol_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        lr = lr0 / (1.0 + 0.002 * i)
        v = mu * v - lr * grad
        z_new = z + v
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        z = z_new
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    return best_a, best_sc


def _lp_peak_shave(a_init, deadline, M=None):
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    c_max = float(c.max())
    if M is None:
        M = c_max
    S = float(a_init.sum())

    A_ub_rows = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        for i in range(lo, hi + 1):
            row[i] = a_init[k - i]
        A_ub_rows.append(row)
        b_ub.append(M)

    A_ub = np.array(A_ub_rows)
    c_obj = -np.ones(n)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs')
        if res.success:
            g = np.maximum(res.x, 0)
            g_sum = float(g.sum())
            if g_sum > 1e-10:
                g_scaled = g * (S / g_sum)
                return g_scaled
    except Exception:
        pass
    return a_init


def _harmonic_init(n, freqs=None, amps=None, rng=None):
    if freqs is None:
        freqs = [1, 2]
    if amps is None:
        amps = [0.3, 0.1]
    if rng is None:
        rng = np.random.default_rng(0)
    t = np.arange(n)
    a = np.ones(n)
    for f, amp in zip(freqs, amps):
        phase = rng.uniform(0, 2 * np.pi)
        a = a + amp * np.cos(2.0 * np.pi * f * t / n + phase)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _perturb_harmonic(a, rng, n_modes=3, amp_scale=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 8)))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _lbfgs_schedule(a_start, deadline, min_remain=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.25, 0.12, 0.06, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remain > deadline:
            break
        remaining = deadline - min_remain - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.004 * (n / 2048.0) / (n / 2048.0)
        max_iter = min(3000, max(50, int(remaining / (est_time * 10))))
        a_new, success = _lbfgs_run(current_a, beta, n, max_iter=max_iter)
        sc = _score(a_new)
        if sc < current_sc - 1e-14:
            current_a = a_new
            current_sc = sc
    return current_a
