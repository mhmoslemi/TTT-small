```python
"""
Simulated-annealing Adam optimizer in log-space for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses diverse initializations, beta-reheating to escape local optima,
multi-restart diversification, and multi-length exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    n_full = n_a + n_b - 1
    N = _next_pow2(n_full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_full]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(a, c, w, n):
    """Gradient of dot(w, conv(a,a)) w.r.t. a, computed via FFT."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_uniform(n):
    return np.ones(n) / n


def _make_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.01
    return a / a.sum()


def _make_sinusoidal(n, freq=1.0, amp=0.3):
    t = np.linspace(0, 1, n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_block(n, nblocks=16):
    block_size = n // nblocks
    a = np.zeros(n)
    for i in range(nblocks):
        a[i * block_size:(i + 1) * block_size] = 1.0 + 0.5 * np.cos(np.pi * i / nblocks)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_logspace(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    beta = beta_init_frac * max_c_init
    beta_min = beta_min_frac * max_c_init

    it_no_improve = 0
    improve_threshold = 500

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** (t + 1))
        vh = v_buf / (1.0 - b2 ** (t + 1))

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        t += 1

        # Anneal beta
        if it % 100 == 0:
            beta = max(beta_min, beta * 0.985)

        # Reheat with some probability
        if it % 200 == 0 and rng.random() < reheat_prob:
            beta = min(beta_init_frac * max_c_init, beta * reheat_factor * rng.uniform(0.5, 1.5))

        # Decay learning rate
        if it % 2000 == 0:
            current_lr = max(0.0005, current_lr * 0.97)

        # Check score periodically
        if it % 50 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
                it_no_improve = 0
            else:
                it_no_improve += 50

            # If stuck for a long time, do a small reheat
            if it_no_improve > 1000:
                beta = max(beta_min, beta * 2.0)
                it_no_improve = 0

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_simplex(a_init, beta, deadline, max_iter=50000):
    """FISTA on the simplex with smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_curr = a_curr / a_curr.sum()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_curr = _full_convolve(a_curr, a_curr)
    max_c = float(c_curr.max())

    L = 8.0 * n / beta
    step = 1.0 / L

    best_a = a_curr.copy()
    best_score = 2.0 * n * max_c

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        grad = 2.0 * wc[n - 1:2 * n - 1]

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_steps=20000):
    """Exact subgradient steps on the simplex for final polishing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    lr = 0.01
    momentum = 0.0
    v = np.zeros(n)

    for it in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        # Subgradient: g[i] = 2*a[k_star - i] if 0 <= k_star - i <= n-1
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        g = np.zeros(n)
        if hi >= lo:
            g[lo:hi + 1] = 2.0 * a[k_star - hi:k_star - lo + 1][::-1]

        # Momentum update
        momentum = 0.9 * momentum + 0.1 * lr
        v = 0.9 * v + momentum * g
        a_new = _proj_simplex(a - v)

        if it % 200 == 0:
            sc = _score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        a = a_new

        # Decay
        if it % 5000 == 0:
            lr = max(0.001, lr * 0.7)

    return best_a, best_score


def _load_parent():
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                return parent_raw / s
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def _try_update(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 0.01:
            return
        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  New best: score={sc:.10f}, n={len(a)}, t={time.monotonic()-t0:.1f}s")

    # Load parent
    parent = _load_parent()
    n_main = 4096

    if parent is not None:
        n_main = len(parent)
        print(f"Parent: n={n_main}, score={_score(parent):.10f}")
        _try_update(parent)
    else:
        parent = _make_uniform(4096)
        print(f"No parent, using uniform n=4096")

    # === Phase 1: FISTA coarse-to-fine from parent ===
    print(f"\nPhase 1: FISTA coarse-to-fine (n={n_main})")
    phase1_deadline = min(deadline, t0 + 400.0)
    a_work = parent.copy()

    c_work = _full_convolve(a_work, a_work)
    max_c = float(c_work.max())

    betas = []
    beta = 0.05 * max_c
    while beta > 0.00001 * max_c:
        betas.append(beta)
        beta *= 0.5
    if len(betas) > 8:
        betas = betas[:8]

    for i, b in enumerate(betas):
        phase_dl = min(phase1_deadline, time.monotonic() + 50.0)
        a_work, sc = _fista_simplex(a_work, b, phase_dl, max_iter=30000)
        a_work = a_work / a_work.sum()
        _try_update(a_work)
        c_work = _full_convolve(a_work, a_work)
        max_c = float(c_work.max())

    print(f"  End Phase 1: score={best_score:.10f}")

    # === Phase 2: Adam annealed (main refinement) ===
    print(f"\nPhase 2: Adam annealed (n={n_main})")
    phase2_deadline = min(deadline, t0 + 900.0)
    a_work, sc = _adam_annealed(
        a_work, phase2_deadline, rng,
        lr_init=0.01, beta_init_frac=0.005,
        beta_min_frac=1e-5, reheat_prob=0.015,
        reheat_factor=2.5, max_iter=800000
    )
    _try_update(a_work)
    print(f"  End Phase 2: score={best_score:.10f}")

    # === Phase 3: Subgradient polish ===
    print(f"\nPhase 3: Subgradient polish")
    phase3_deadline = min(deadline, t0 + 1100.0)
    if best_seq is not None:
        a_pol, sc_pol = _subgradient_polish(best_seq, phase3_deadline, max_steps=50000)
        _try_update(a_pol)
    print(f"  End Phase 3: score={best_score:.10f}")

    # === Phase 4: Diversified restarts ===
    print(f"\nPhase 4: Diversified restarts")
    phase4_deadline = min(deadline, t0 + 1500.0)
    if time.monotonic() < phase4_deadline - 30 and best_seq is not None:
        n_restarts = 8
        per_restart = (phase4_deadline - time.monotonic() - 10) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase4_deadline - 5:
                break

            if i % 4 == 0:
                a_start = _perturb_logspace(best_seq, 0.04 + 0.01 * i, rng)
            elif i % 4 == 1:
                # Shift perturbation
                sz = len(best_seq) // 256
                shift = rng.integers(1, max(2, sz))
                a_start = np.roll(best_seq, int(shift)).copy()
                a_start = a_start / a_start.sum()
            elif i % 4 == 2:
                # Envelope perturbation
                t = np.linspace(0, 1, len(best_seq))
                env = 1.0 + 0.2 * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 2 * np.pi))
                a_start = np.maximum(best_seq * env, 1e-12)
                a_start = a_start / a_start.sum()
            else:
                # Geometric mean with random init
                a_rand = _make_random(len(best_seq), rng)
                a_start = np.sqrt(np.maximum(best_seq, 1e-15) * np.maximum(a_rand, 1e-15))
                a_start = a_start / a_start.sum()

            dl = min(phase4_deadline, time.monotonic() + max(5.0, per_restart))
            a_opt, sc_opt = _adam_annealed(
                a_start, dl, rng,
                lr_init=0.008, beta_init_frac=0.003,
                beta_min_frac=1e-5, reheat_prob=0.03,
                reheat_factor=3.0, max_iter=100000
            )
            _try_update(a_opt)

    print(f"  End Phase 4: score={best_score:.10f}")

    # === Phase 5: Try different lengths ===
    print(f"\nPhase 5: Multi-length exploration")
    phase5_deadline = min(deadline, t0 + 1700.0)
    if time.monotonic() < phase5_deadline - 30 and best_seq is not None:
        for n_try in [n_main * 2, max(1024, n_main // 2)]:
            if time.monotonic() > phase5_deadline - 10:
                break
            if n_try < 64 or n_try > 16384:
                continue
            a_interp = _interp_profile(best_seq, n_try)
            sc_interp = _score(a_interp)
            print(f"  n={n_try}: interp score={sc_interp:.10f}")
            _try_update(a_interp)

            if sc_interp < best_score * 1.01:
                dl = min(phase5_deadline, time.monotonic() + 60.0)
                a_opt, sc_opt = _adam_annealed(
                    a_interp, dl, rng,
                    lr_init=0.008, beta_init_frac=0.003,
                    beta_min_frac=1e-5, reheat_prob=0.02,
                    reheat_factor=2.5, max_iter=100000
                )
                _try_update(a_opt)

    print(f"  End Phase 5: score={best_score:.10f}")

    # === Phase 6: Final ultra-fine polish ===
    print(f"\nPhase 6: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_final = best_seq.copy()
        a_final = np.clip(a_final, 0, 1000)
        a_final = a_final / a_final.sum()

        c_final = _full_convolve(a_final, a_final)
        max_cf = float(c_final.max())

        # Very fine Adam
        betas_ult = [0.00003 * max_cf, 0.00001 * max_cf, 0.000003 * max_cf]
        for b_ult in betas_ult:
            if time.monotonic() > deadline - 3:
                break
            a_final, sc = _adam_annealed(
                a_final, deadline, rng,
                lr_init=0.002, beta_init_frac=b_ult / max_cf,
                beta_min_frac=b_ult / (100 * max_cf),
                reheat_prob=0.0, max_iter=200000
            )
            _try_update(a_final)

        # Subgradient micro-polish
        if time.monotonic() < deadline - 2:
            a_final, sc = _subgradient_polish(a_final, deadline, max_steps=5000)
            _try_update(a_final)

    # === Final validation ===
    print(f"\nFinal: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        else:
            # Ensure all values are valid
            final = [float(max(0.0, min(1000.0, x))) for x in final]
        sc_final = _score(np.array(final))
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t0:.1f}s")
        try:
            sc_eval = float(evaluate_sequence(list(final)))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return list(final)

    # Fallback
    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
