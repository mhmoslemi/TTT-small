"""
Hybrid LP + Adam + Coordinate Descent optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2.

Uses:
1. LP-based direction finding with linearized self-convolution
2. Adam with log-sum-exp smoothed gradient
3. Coordinate descent for final polishing
4. Multiple restarts to escape local minima
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp
import scipy.optimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_direct(a):
    """Score assuming sum(a)=1, so return 2*n*max(conv)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_matrix(f, rows, cols):
    """Build sparse matrix for the LP constraint (f*d)[k]."""
    n = len(f)
    # A[k, j] = f[k-j] if 0 <= k-j < n else 0
    # We need 2*(f*d)[k] as the LHS
    vals = []
    r = []
    c = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        vals.extend(2.0 * f[j_range])
        r.extend(np.full(len(j_range), k, dtype=np.int32))
        c.extend(j_range.astype(np.int32))
    return sp.csc_matrix((vals, (r, c)), shape=(2 * n - 1, n))
