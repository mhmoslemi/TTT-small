from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    """Build sparse matrix for conv(f, g) constraints."""
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

                g = g / g_sum
                return g, t / g_sum  # rescale t
    except Exception as e:

def _smoothed_obj_grad(z, beta):
    """Compute objective and gradient of smoothed function in z-space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    n = len(a)
    
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    
    return obj, grad


def _lbfgs_smooth(z_init, beta, time_limit, maxiter=300):
    """Run L-BFGS on the smoothed objective."""
    t_start = time.monotonic()
    
    # Wrap for scipy
    def fun(z):
        if time.monotonic() - t_start > time_limit:
            return np.inf
        obj, grad = _smoothed_obj_grad(z, beta)
        return obj
    
    def fun_and_grad(z):
        if time.monotonic() - t_start > time_limit:
            obj, grad = _smoothed_obj_grad(z, beta)
            return obj, grad
        obj, grad = _smoothed_obj_grad(z, beta)
        return obj, grad
    
    # Bound z to avoid numerical issues
    bounds = [(-50.0, 50.0)] * len(z_init)
    
    try:
        res = minimize(
            fun_and_grad, z_init, jac=True, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'maxfun': 5000, 'maxls': 50,
                     'ftol': 1e-12, 'gtol': 1e-8, 'maxcor': 20}
        )
        if res.x is not None:
            return res.x
    except Exception:
        pass
    return z_init


def _adam_optimize(a_init, deadline, beta_list, lr=0.01):
    """Adam optimizer with multi-beta scaling."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        # Determine time budget for this beta
        remaining_betas = len(beta_list) - beta_list.index(beta)
        time_per_beta = (deadline - time.monotonic()) / max(1, remaining_betas)
        beta_deadline = time.monotonic() + time_per_beta
        
        stall = 0
        last_best = best_score
        
        while time.monotonic() < beta_deadline:
            if time.monotonic() > deadline:
                break
            t_step += 1
            
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n-1:2*n-1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1**t_step)
            vh = v_buf / (1.0 - b2**t_step)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 1.0:
                update = update * (1.0 / un)
            
            z = z - update
            z = np.clip(z, -100.0, 100.0)
            z = z - z.mean()
            
            if t_step % 10 == 0:
                sc = float(_full_convolve(_softmax(z), _softmax(z)).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z).copy()
                    stall = 0
                else:
                    stall += 1
                
                if stall > 200:
                    m_buf *= 0.1
                    v_buf *= 0.1
                    stall = 0
                    current_lr = max(0.001, current_lr * 0.8)
    
    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding, L-BFGS on log-sum-exp smoothed objective,
Adam fine-tuning, and multi-restart exploration.
Key innovation: L-BFGS captures local curvature of the smoothed max,
enabling faster convergence in the fine-tuning regime than first-order methods.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, hstack


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_structure(n):
    """Precompute sparsity pattern for convolution constraint matrix."""
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    idx = diff[rows, cols]
    return rows, cols, idx


def _make_conv_matrix(f, structure):
    """Build sparse convolution matrix from precomputed structure."""
    rows, cols, idx = structure
    vals = f[idx]
    return csr_matrix((vals, (rows, cols)), shape=(2 * len(f) - 1, len(f)))


def _proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_obj_grad(z, beta):
    """Combined objective and gradient for L-BFGS (takes z, beta as args)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    n = len(a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * c_max
    else:
        w = e / s
        obj = 2.0 * n * (c_max + beta * np.log(s))
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _lbfgs_run(z_init, beta, maxiter, maxfun):
    """Run L-BFGS-B on the smoothed objective."""
    n = len(z_init)
