"""
Hybrid optimizer: LP + Adaptive Smoothed Gradient + Coordinate Descent + Multi-n exploration.
Targets minimizing 2n * max(autoconv) / sum^2 for non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _simplex_proj(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    i = np.where(cond)[0][-1] if len(np.where(cond)[0]) > 0 else 0
    theta = cssv[i] / (i + 1)
    return np.maximum(v - theta, 0)

def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    # Adaptive line search
    best_t = 0.0
    best_sc = _score_norm(f)
    # Coarse search
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    # Fine search around best
    if best_t > 0:
        for t in np.linspace(max(0, best_t - 0.1), min(1, best_t + 0.1), 20):
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M

def _smoothed_adam(a_init, deadline, beta_start, beta_end, max_iter=5000000, lr=0.01):
    """Adam optimizer with log-sum-exp smoothing of the max objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    no_improve = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        # Adaptive beta: start large (smooth), decrease (sharpen)
        progress = min(1.0, t / max_iter)
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** progress
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive learning rate
        if t % 500 == 0:
            if no_improve < 5:
                current_lr = min(0.05, current_lr * 1.05)
            elif no_improve > 20:
                current_lr = max(0.0005, current_lr * 0.9)
            no_improve = 0

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

def _coord_descent(a, deadline, max_rounds=100):
    """Coordinate descent: try small perturbations of individual elements."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        # Random order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            # Try a few perturbation sizes
            for delta in [0.01, 0.005, 0.001]:
                a_try = a.copy()
                a_try[j] += delta
                a_try[j + 1 if j + 1 < n else j - 1] -= delta
                if a_try[j] < 0 or a_try[j + 1 if j + 1 < n else j - 1] < 0:
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    a = a_try
                    best_score = sc
                    improved = True
                    break
            # Also try decreasing
            for delta in [0.01, 0.005, 0.001]:
                a_try = a.copy()
                j2 = (j + n // 2) % n  # Move mass to opposite end
                a_try[j] -= delta
                a_try[j2] += delta
                if a_try[j] < 0:
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    a = a_try
                    best_score = sc
                    improved = True
                    break
        if not improved and rnd > 10:
            break
        if rnd % 10 == 0:
            print(f"    CoordDescent round {rnd}: score={best_score:.10f}")
    
    return a, best_score

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _deflation_step(a, deadline, num_steps=50):
    """Targeted deflation: identify hottest lag and try to reduce it specifically."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for step in range(num_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        c_max = c.max()
        hot_lags = np.where(c > 0.95 * c_max)[0]
        
        # For each hot lag, try to reduce it by adjusting a
        # The gradient of conv[i] w.r.t. a[j] is 2*a[i-j] (if valid)
        # So to reduce conv[i], we should decrease a[j] where a[i-j] is large
        # and increase a[j] where a[i-j] is small
        
        for i in hot_lags[:5]:  # Only top 5 hot lags
            if time.monotonic() > deadline:
                break
            # Gradient direction for this lag
            grad_i = np.zeros(n)
            for j in range(n):
                if 0 <= i - j < n:
                    grad_i[j] = 2.0 * a[i - j]
            # Normalize
            grad_norm = np.linalg.norm(grad_i)
            if grad_norm < 1e-15:
                continue
            # Move in negative gradient direction (projected to simplex)
            for eta in [0.01, 0.005, 0.001]:
                a_try = a - eta * grad_i / grad_norm
                a_try = _simplex_proj(a_try)
                sc = _score_norm(a_try)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_try.copy()
                    a = a_try
                    break
    
    return best_a, best_score
