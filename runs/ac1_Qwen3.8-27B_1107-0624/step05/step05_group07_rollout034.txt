The problem is: minimize `max_k c_k(a)` subject to `a >= 0, sum(a) = 1`.

By minimax theorem (if the problem were convex), the optimum is achieved at a saddle point. For the non-convex case, we can still use a "no-regret" approach:

1. Solve: minimize `sum_k p_k * c_k(a)` subject to `a >= 0, sum(a) = 1`. (This is a QP!)
2. Update `p` using exponential weights: `p_k ∝ p_k * exp(-eta * c_k(a_new))`.

Step 1 is a QP: minimize `a^T Q a` where `Q = sum_k p_k * Q_k` and `Q_k` is the quadratic form for `c_k`. The constraint is `a >= 0, 1^T a = 1`.

2. The linearized objective is: `sum_k p_k * [c_k(a) + 2 * sum_j a_{k-j} * a'_j]` (dropping constants).
3. = `sum_j a'_j * [2 * sum_k p_k * a_{k-j}]` + const.
4. = `a'^T * [2 * (p * a_reversed)]` + const.

Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_best_response(a, max_time=30):
    n = len(a)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(a)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _mirror_descent_step(a, eta, active_frac=0.01):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return a.copy()
    threshold = cm * (1.0 - active_frac)
    active = c >= threshold
    if not active.any():
        active[int(np.argmax(c))] = True
    p = active.astype(np.float64)
    p = p / p.sum()
    a_rev = a[::-1].copy()
    v = _full_convolve(p, a_rev)
    v = v[n - 1:2 * n - 1]
    log_w = np.log(np.maximum(a, 1e-30)) - eta * 2.0 * v
    log_w = log_w - np.max(log_w)
    w = np.exp(log_w)
    s = w.sum()
    if s < 1e-300 or not np.isfinite(s):
        return a.copy()
    return w / s

            a = a_new

def _lse_obj_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    lse = cm + beta * float(np.log(e.sum()))
    obj = 2.0 * n * lse
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W) * beta
    return obj, grad


def _nesterov_lse(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002,
                 0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]
        beta_list = [b * max_c0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        z_cur = z.copy()
        z_mom = z.copy()
        t_nest = 1.0
        lr = 1.0 / (4.0 * n * beta + 1e-10)

        for it in range(30000):
            if time.monotonic() > deadline:
                break
            obj, grad = _lse_obj_grad(z_mom, beta, n)
            if not np.all(np.isfinite(grad)):
                break
            g_norm = np.linalg.norm(grad)
            if g_norm < 1e-15:
                break
            z_prev = z_cur.copy()
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            z_mom = z_cur + (t_nest - 1.0) / t_new * (z_cur - z_prev)
            z_mom = np.clip(z_mom, -80, 80)
            z_mom = z_mom - z_mom.mean()
            z_cur = z_mom - lr * grad
            z_cur = np.clip(z_cur, -80, 80)
            z_cur = z_cur - z_cur.mean()
            t_nest = t_new

            if it % 500 == 0 or it == 0:
                a_check = _softmax_fn(z_cur)
                sc_check = _score_norm_fn(a_check)
                if np.isfinite(sc_check) and sc_check < best_score:
                    best_score = sc_check
                    best_a = a_check.copy()
        z = z_cur.copy()
        a_end = _softmax_fn(z)
        sc_end = _score_norm_fn(a_end)
        if np.isfinite(sc_end) and sc_end < best_score:
            best_score = sc_end
            best_a = a_end.copy()

    return best_a, best_score


def _lbfgs_lse(a_init, beta, deadline, maxiter=5000):
    n = len(a_init)

    def obj_fn(z):
        o, _ = _lse_obj_grad(z, beta, n)
        return o

    def grad_fn(z):
        _, g = _lse_obj_grad(z, beta, n)
        return g

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            obj_fn, z, jac=grad_fn,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 40, 'maxfun': maxiter * 4}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm_fn(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=100):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        improved = False
        order = np.random.default_rng(rnd).permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_power_law_profiles(n):
    profiles = []
    for alpha in [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        t = np.arange(n, dtype=np.float64)
        a = (t + 1.0) ** (-alpha)
        a = a / a.sum()
        profiles.append(a)
        a2 = a[::-1].copy()
        profiles.append(a2)
    for alpha in [0.3, 0.5, 1.0, 1.5]:
        t = np.linspace(0, 1, n)
        a = (t + 0.01) ** (-alpha)
        a = a / a.sum()
        profiles.append(a)
        a2 = a[::-1].copy()
        profiles.append(a2)
    t = np.linspace(0, 1, n)
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.maximum(t ** p, 1e-12)
        a = a / a.sum()
        profiles.append(a)
        a2 = a[::-1].copy()
        profiles.append(a2)
    a = np.ones(n)
    profiles.append(a / n)
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 45.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    best_n = 0

    def _update(p, sc, label=""):
        nonlocal best_seq, best_score, best_n
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            best_n = len(best_seq)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        _update(parent, sc_parent, "parent")
    else:
        n_init = 4096
        profiles = _make_power_law_profiles(n_init)
        parent = profiles[0].copy()
        _update(parent, _score_norm_fn(parent), "default_profile")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, score={best_score:.10f}")

    # Phase 1: Mirror descent + LP at parent n (or 4096 if parent is different)
    n_work = min(n_parent, 4096) if n_parent <= 8192 else 4096
    if n_parent == n_work:
        a_work = parent.copy()
    else:
        a_work = _interp_profile(parent, n_work)
    sc_work = _score_norm_fn(a_work)
    _update(a_work, sc_work, f"init_n{n_work}")

    phase1_end = min(deadline, t_start + 500.0)
    print(f"\nPhase 1: Mirror descent + LP (n={n_work}, target {phase1_end - t_start:.0f}s)")

    # Run mirror descent with periodic LP
    a_md, sc_md = _mirror_descent_loop(a_work, phase1_end, max_iters=500000, lp_every=2000, eta_init=0.3, eta_decay=0.99995)
    _update(a_md, sc_md, "mirror_descent")

    # If parent was larger, interpolate back
    if n_parent > n_work:
        a_back = _interp_profile(a_md, n_parent)
        sc_back = _score_norm_fn(a_back)
        _update(a_back, sc_back, f"interp_back_n{n_parent}")
        a_work = a_back
    else:
        a_work = a_md.copy()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Nesterov LSE continuation
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\nPhase 2: Nesterov LSE (n={len(best_seq)}, target {phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end - 30:
        a_start = best_seq.copy()
        per_restarts = 2
        time_per_restart = (phase2_end - time.monotonic() - 10) / per_restarts

        for r in range(per_restarts):
            if time.monotonic() > phase2_end - 20:
                break
            if r == 0:
                a_cur = a_start.copy()
            else:
                z = np.log(np.maximum(a_start, 1e-15))
                z = z - z.mean()
                noise = 0.03
                z = z + noise * rng.normal(size=len(z))
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                a_cur = _softmax_fn(z)

            per_dl = min(phase2_end - 10, time.monotonic() + time_per_restart)
            p_best, sc_best = _nesterov_lse(a_cur, per_dl)
            _update(p_best, sc_best, f"nesterov_r{r}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: L-BFGS on fine beta
    phase3_end = min(deadline, t_start + 1200.0)
    print(f"\nPhase 3: L-BFGS fine beta (n={len(best_seq)}, target {phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        n_curr = len(best_seq)
        c_curr = _full_convolve(best_seq, best_seq)
        max_c_curr = float(c_curr.max())

        for beta_frac in [0.001, 0.0005, 0.0002, 0.0001]:
            if time.monotonic() > phase3_end - 20:
                break
            beta = beta_frac * max_c_curr
            per_dl = min(phase3_end - 10, time.monotonic() + 60.0)
            p_best, sc_best = _lbfgs_lse(best_seq, beta, per_dl, maxiter=3000)
            _update(p_best, sc_best, f"lbfgs_beta{beta_frac}")
            max_c_curr = _full_convolve(best_seq, best_seq).max()

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Try different n values
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 4: Multi-n exploration (target {phase4_end - t_start:.0f}s)")

    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        n_try_list = [2048, 3072, 4096, 5120, 6144, 8192]
        n_try_list = [x for x in n_try_list if x != len(best_seq)]
        n_try_list = [x for x in n_try_list if 1024 <= x <= 8192]

        for n_try in n_try_list:
            if time.monotonic() > phase4_end - 30:
                break
            p_interp = _interp_profile(best_seq, n_try)
            sc_interp = _score_norm_fn(p_interp)
            if np.isfinite(sc_interp):
                _update(p_interp, sc_interp, f"interp_n{n_try}")
                # Quick mirror descent at this n
                if time.monotonic() < phase4_end - 60 and sc_interp < best_score * 1.005:
                    per_dl = min(phase4_end - 30, time.monotonic() + 80.0)
                    p_md, sc_md = _mirror_descent_loop(p_interp, per_dl, max_iters=50000, lp_every=10000, eta_init=0.2, eta_decay=0.9999)
                    _update(p_md, sc_md, f"md_n{n_try}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Polishing - coordinate descent + ultra-fine L-BFGS
    phase5_end = deadline
    print(f"\nPhase 5: Polishing (n={len(best_seq)}, target {phase5_end - t_start:.0f}s)")

    if time.monotonic() < phase5_end - 20 and best_seq is not None:
        # Coordinate descent
        per_dl1 = min(phase5_end - 60, time.monotonic() + 200.0)
        p_cd, sc_cd = _coord_descent(best_seq, per_dl1, max_rounds=50)
        _update(p_cd, sc_cd, "coord_descent")

        # Ultra-fine L-BFGS
        if time.monotonic() < phase5_end - 30:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            for beta_frac in [5e-5, 2e-5, 1e-5]:
                if time.monotonic() > phase5_end - 15:
                    break
                beta = beta_frac * max_c_final
                per_dl2 = min(phase5_end - 10, time.monotonic() + 40.0)
                p_best, sc_best = _lbfgs_lse(best_seq, beta, per_dl2, maxiter=5000)
                _update(p_best, sc_best, f"ultra_lbfgs_{beta_frac}")

        # Final mirror descent with tiny eta
        if time.monotonic() < phase5_end - 10:
            per_dl3 = min(phase5_end - 5, time.monotonic() + 30.0)
            p_md, sc_md = _mirror_descent_loop(best_seq, per_dl3, max_iters=100000, lp_every=0, eta_init=0.05, eta_decay=0.9999)
            _update(p_md, sc_md, "final_md")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(len(final_seq)) / len(final_seq)
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence check: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    profiles = _make_power_law_profiles(n_fb)
    fallback = profiles[0]
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n={n_fb}, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
