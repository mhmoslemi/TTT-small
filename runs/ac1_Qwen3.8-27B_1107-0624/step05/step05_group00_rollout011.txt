```python
"""
Multi-phase optimizer: FISTA warm-start + adaptive-beta Adam + 
active-set equalization + n=8192 exploration for minimizing 
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a) ≈ 1, returns 2n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    """
    Gradient of 2*n*dot(w, c) w.r.t. z where a = softmax(z).
    c = conv(a, a).
    Returns gradient of size n.
    
    dc[k]/da[j] = 2*a[k-j] if 0 <= k-j < n else 0
    da[i]/dz[j] = a[i]*(delta_ij - a[j])
    
    d/dz[j] [2n * sum_k w[k] c[k]]
    = 2n * sum_k w[k] * sum_i 2*a[k-i] * a[i]*(delta_{ij} - a[j])
    = 2n * sum_k w[k] * [2*a[k-j] ... ]
    
    Let me define g_a[j] = 2 * sum_k w[k] * a[k-j] (gradient w.r.t. a at position j)
    Then d/dz[j] = sum_i g_a[i] * da[i]/dz[j] = sum_i g_a[i] * a[i]*(delta_{ij} - a[j])
    = g_a[j]*a[j] - a[j]*sum_i g_a[i]*a[i]
    = a[j] * (g_a[j] - dot(g_a, a))
    
    So the full gradient w.r.t. z is:
    2n * a * (g_a - dot(g_a, a))
    where g_a[j] = 2 * sum_k w[k] * a[k-j]
    """
    # Compute g_a = 2 * cross_corr(w, a)
    # cross_corr(w, a)[j] = sum_i w[i+j] * a[i]
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    # cross_corr[j] = sum_i w[i+j] * a[i]
    # This is IFFT(conj(W) * A)[j] for the appropriate range
    cross = np.fft.irfft(np.conj(W) * A, N)[:n]
    g_a = 2.0 * cross
    
    # Full gradient w.r.t. z
    dot_ga_a = float(np.dot(g_a, a))
    grad_z = 4.0 * n * a * (g_a - dot_ga_a)
    return grad_z


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _compute_grad(y, w)
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score


def _compute_grad(a, w):
    """Gradient of dot(w, conv(a,a)) w.r.t. a (on the simplex)."""
    n = len(a)
    N = _next_pow2(2 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A, N)[:n]
    return 2.0 * cross


def _adam_adaptive(a_init, deadline, beta_start, beta_min, lr_init, max_iter_total, angular=None):
    """
    Adam optimizer in log-space with adaptive beta.
    adaptive: if True, adapt beta based on the gap between top-2 conv values.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    while t < max_iter_total:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Adaptive beta
        if angular:
            # Sort top values
            sorted_c = np.sort(c)[::-1]
            gap = sorted_c[0] - sorted_c[1] if len(sorted_c) > 1 else sorted_c[0] * 0.001
            beta = max(gap / 3.0, beta_min)
        else:
            beta = max(beta_start * (beta_min / beta_start) ** (t / max(1, max_iter_total // 4)), beta_min)
        
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        grad = _compute_grad_z(a, c, w, n)
        if not np.all(np.isfinite(grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % 20 == 0:
            a_chk = _softmax_compute(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()
        
        if t % 10000 == 0:
            current_lr = max(0.0005, current_lr * 0.95)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _adam_fixed_beta(a_init, deadline, betas, lr_init, max_iter_per_beta):
    """
    Adam optimizer in log-space with fixed beta schedule.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        for _ in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            t += 1
            
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s
            
            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.8:
                update = update * (0.8 / update_norm)
            
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()
            
            if t % 30 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            if t % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.97)
    
    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)
