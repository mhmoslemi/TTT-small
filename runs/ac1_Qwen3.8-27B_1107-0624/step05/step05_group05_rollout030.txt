```python
def _deflate(a, top_k=5, strength=0.5):
    """
    Identify top-k peaks in convolution and create a bias vector
    that, when added to the gradient, pushes those peaks down.
    """
    n = len(a)
    c = _full_convolve(a, a)
    # Find top-k positions
    idx = np.argpartition(c, -top_k)[-top_k:]
    
    # Create bias: for each peak position k, the "direction" to reduce it
    # is to decrease a[i] where a[i] * a[k-i] is large.
    # Simple approach: create a penalty based on the peak locations.
    # bias[i] = sum over peaks k of a[k-i] (this is the gradient direction for that peak)
    bias = np.zeros(n)
    for k in idx:
        for i in range(max(0, k - n + 1), min(n, k + 1)):
            bias[i] += a[k - i] if 0 <= k - i < n else 0
    
    return bias * strength, c
```

```python
"""
FISTA + LP + Deflation + Coordinate Descent hybrid optimizer for minimizing
2*n*max(convolve(a,a))/sum(a)^2.
Uses Nesterov-accelerated mirror descent (FISTA) on the log-sum-exp smoothed
objective, strategic LP steps for large principled moves, deflation to escape
local minima, and coordinate descent for final polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(c, beta, a):
    """Compute grad[i] = 2*beta * sum_j a[j] * w[j+i] where w = softmax(beta*c)."""
    n = len(a)
    bc = beta * c
    bc_shift = bc - bc.max()
    w = np.exp(np.clip(bc_shift, -700, 700))
    ws = w.sum()
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws
    a_flip = a[::-1]
    conv = _full_convolve(a_flip, w)
    grad = 2.0 * beta * conv[n - 1:2 * n - 1]
    return grad


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(f, rows, cols, time_limit=30):
    """One LP step: maximize sum(b) s.t. conv(f,b)_k <= M, b >= 0."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    if M < 1e-15:
        return f
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return f
        g0 = np.maximum(res.x, 0.0)
        s = g0.sum()
        if s < 1e-15:
            return f
        g = g0 / s
        # Line search
        best_t = 0.0
        best_sc = _score_norm(f)
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            fc = (1.0 - t) * f + t * g
            sc = _score_norm(fc)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        if best_t == 0.0:
            for t in [0.02, 0.05]:
                fc = (1.0 - t) * f + t * g
                sc = _score_norm(fc)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
        return f
    except Exception:
        return f


def _fista_run(a_init, deadline, beta_start, beta_end, eta, max_iter=10000000):
    """FISTA with mirror map on the simplex. Anneals beta from beta_start to beta_end."""
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_obj = _score_norm(a)
    total_iters = max(1, max_iter)

    for it in range(max_iter):
        if time.monotonic() > deadline - 2:
            break
        if it % 100000 == 0 and time.monotonic() > deadline - 30:
            break

        frac = min(1.0, it / total_iters)
        beta = beta_start * (beta_end / beta_start) ** frac

        c_y = _full_convolve(y, y)
        grad = _compute_grad(c_y, beta, y)

        if not np.all(np.isfinite(grad)):
            break

        # Mirror step in log-space
        z = np.log(np.maximum(y, 1e-300)) - eta * grad
        z = z - z.max()
        a_new = _softmax(z)

        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = a_new + momentum * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-300:
            y = a_new.copy()
        else:
            y = y / ys

        a_prev = a
        a = a_new
        t = t_new

        if it % 500 == 0:
            obj = _score_norm(a)
            if np.isfinite(obj) and obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    obj_final = _score_norm(a)
    if np.isfinite(obj_final) and obj_final < best_obj:
        best_obj = obj_final
        best_a = a.copy()

    return best_a, best_obj


def _deflation_step(a, top_k=5, strength=1.0, deadline_check=None):
    """
    Identify top-k peaks in convolution, compute a bias that penalizes them,
    do a few FISTA steps with the bias added to gradient.
    """
    n = len(a)
    c = _full_convolve(a, a)
    idx = np.argpartition(c, -top_k)[-top_k:]

    # Build bias vector: for each peak at position k, the gradient contribution
    # is proportional to a[k-i] for valid i. We use this as a "virtual peak"
    # to push mass away from positions that feed into the peaks.
    bias = np.zeros(n)
    for k in idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for i in range(j_start, j_end + 1):
            if 0 <= k - i < n:
                bias[i] += a[k - i]

    # Normalize bias
    bmax = bias.max()
    if bmax > 1e-300:
        bias = bias / bmax * strength
    else:
        return a

    # Do a few gradient steps with the bias
    c_max = float(c.max())
    beta = 50.0 / max(c_max, 1e-15)
    eta = 1.0 / beta

    for step in range(100):
        if deadline_check is not None and time.monotonic() > deadline_check:
            break
        c_y = _full_convolve(a, a)
        grad = _compute_grad(c_y, beta, a)
        z = np.log(np.maximum(a, 1e-300)) - eta * (grad + bias)
        z = z - z.max()
        a = _softmax(z)

    return a


def _coordinate_descent(a, deadline, max_passes=20):
    """Greedy coordinate descent: for each coord, try several multipliers."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    multipliers = [0.0, 0.2, 0.5, 0.8, 1.2, 1.5, 2.0, 4.0]

    for passage in range(max_passes):
        if time.monotonic() > deadline - 2:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline - 2:
                break
            old_val = best_a[i]
            best_sc_local = best_score
            best_mult = 1.0

            for mult in multipliers:
                if mult == 1.0:
                    continue
                new_val = old_val * mult
                if new_val < 0:
                    continue
                best_a[i] = new_val
                s = best_a.sum()
                if s < 1e-15:
                    best_a[i] = old_val
                    continue
                best_a = best_a / s
                sc = _score_norm(best_a)
                if sc < best_sc_local:
                    best_sc_local = sc
                    best_mult = mult
                best_a[i] = old_val  # revert for next trial
                best_a = best_a / best_a.sum()

            if best_mult != 1.0:
                best_a[i] = old_val * best_mult
                s = best_a.sum()
                best_a = best_a / s
                best_score = best_sc_local
                improved = True

        if not improved:
            break

    return best_a, best_score

            b_e_r = 80.0 / max_cr
            eta_r = 1.0 / b_s_r
            r_deadline = time.monotonic() + max(10.0, time_per_restart - 5)

            a_r, _ = _fista_run(a_r, r_deadline, b_s_r, b_e_r, eta_r, max_iter=300000)
            sc_r = _score(a_r)
            print(f"  Restart[{r}] (sigma={sigma:.3f}): init={sc_init_r:.10f} -> final={sc_r:.10f}")
            if np.isfinite(sc_r) and sc_r < best_score:
                best_score = sc_r
                best_a = a_r.copy()

    # === Phase 6: Fine FISTA with high beta (1200 - 1550s) ===
    print(f"\n=== Phase 6: Fine FISTA ===")
    phase6_end = min(t_start + 1550.0, deadline)
    if time.monotonic() < phase6_end - 10:
        a6 = best_a.copy()
        c6 = _full_convolve(a6, a6)
        max_c6 = float(c6.max())
        b_s6 = 50.0 / max_c6
        b_e6 = 300.0 / max_c6
        eta6 = 0.5 / b_s6

        a6, obj6 = _fista_run(a6, phase6_end, b_s6, b_e6, eta6, max_iter=8000000)
        sc6 = _score(a6)
        print(f"Phase 6 done: score={sc6:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc6) and sc6 < best_score:
            best_score = sc6
            best_a = a6.copy()

    # === Phase 7: Coordinate descent polish (1550 - deadline) ===
    print(f"\n=== Phase 7: Coordinate Descent ===")
    if time.monotonic() < deadline - 5:
        a7, sc7 = _coordinate_descent(best_a, deadline - 5, max_passes=15)
        sc7_full = _score(a7)
        print(f"Phase 7 done: score={sc7_full:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc7_full) and sc7_full < best_score:
            best_score = sc7_full
            best_a = a7.copy()

    # === Final: one more LP and coordinate sweep if time allows ===
    if time.monotonic() < deadline - 10:
        a_final = _lp_step(best_a, rows, cols, time_limit=10)
        sc_fl = _score(a_final)
        if np.isfinite(sc_fl) and sc_fl < best_score:
            best_score = sc_fl
            best_a = a_final.copy()
            print(f"Final LP improved: {best_score:.10f}")

        if time.monotonic() < deadline - 3:
            a_cd, _ = _coordinate_descent(best_a, deadline - 2, max_passes=5)
            sc_cd = _score(a_cd)
            if np.isfinite(sc_cd) and sc_cd < best_score:
                best_score = sc_cd
                best_a = a_cd.copy()
                print(f"Final CD improved: {best_score:.10f}")

    # Final validation
    final = np.clip(best_a, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
        s = float(final.sum())
    final = final / s * np.sqrt(2 * len(final))  # rescale for good measure

    # Actually, let's just return as-is since score is scale invariant
    final = np.clip(best_a, 0.0, 1000.0)

    final_score = _score(final)
    print(f"\nFINAL: score={final_score:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")

    return [float(x) for x in final]
```
