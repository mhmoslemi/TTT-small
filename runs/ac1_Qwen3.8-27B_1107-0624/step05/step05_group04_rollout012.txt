def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    """Score with sum(a)^2 in denominator removed (scale-invariant part)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _downsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(len(a))
    n = len(a)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # h_i = sum_k w[k] * a[k-i] for valid k-i in [0, n-1]
    # This is the cross-correlation of a with w, evaluated at lags 0 to n-1
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adaptive_beta(c):
    """Adaptively choose beta based on peak distribution."""
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    # Look at the gap between top peaks
    for k in [3, 5, 10, 20, 50]:
        if k < len(sorted_c):
            gap = cm - sorted_c[k]
            if gap > 0.01 * cm:
                return max(1e-10, 0.3 * gap)
    return max(1e-10, 0.05 * cm)


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=15):
    """Run L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.5 * max_c0
    beta_end = max(1e-8, 1e-6 * max_c0)

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget

        if beta > 0.2 * max_c0:

        elif beta > 0.01 * max_c0:
            maxiter = 2000
        else:
            maxiter = 5000

        maxfun = maxiter * 3

        try:
            result = minimize(

                method='L-BFGS-B', jac=_logsumexp_grad,

def _peak_shaving(a, deadline, rng, max_rounds=500):
    """Directly shave the peak of the convolution by adjusting contributing elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))

        if time.monotonic() > deadline:

        c = _full_convolve(a_cur, a_cur)

        valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
        if len(valid_i) == 0:
            break
        prod = a_cur[valid_i] * a_cur[pk - valid_i]
        
        # Find the top contributors
        top_k = min(10, len(valid_i))
        top_idx = np.argsort(prod)[::-1][:top_k]
        
        improved = False
        for idx_pos in top_idx:
            idx = valid_i[idx_pos]
            partner = pk - idx
            # Try reducing this element
            for delta in [0.02, 0.05, 0.1, 0.2, 0.5]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()

            elif rnd > 50:

def _adam_gd(a_init, deadline, rng, lr_init=0.05, max_iter=50000):
    """Adam gradient descent on log-softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        if it % 50 == 0:
            beta = _adaptive_beta(c)

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)

        # Random kick if stuck
        if t % 10000 == 0 and t > 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _make_ramp(n, slope=1.5):

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    n = len(a)
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    for k in [2, 3, 5, 10, 20, 50, 100]:
        if k < len(sorted_c):
            gap = cm - sorted_c[k]
            if gap > 0.005 * cm:
                return max(1e-10, 0.25 * gap)
    return max(1e-10, 0.02 * cm)


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=15, aggressive=False):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if aggressive:
        beta_start = 0.3 * max_c0
        beta_end = max(1e-10, 1e-7 * max_c0)
    else:
        beta_start = 0.5 * max_c0
        beta_end = max(1e-10, 1e-6 * max_c0)

    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = max(1, n_betas - idx)
        time_budget = (deadline - time.monotonic()) / remaining
        time_budget = min(time_budget, 150.0)
        if time_budget < 2.0:
            break
        level_deadline = time.monotonic() + time_budget

        if beta > 0.15 * max_c0:
            maxiter = 150
        elif beta > 0.05 * max_c0:
            maxiter = 400
        elif beta > 0.01 * max_c0:
            maxiter = 1500
        else:
            maxiter = 4000

        maxfun = maxiter * 3

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                         'maxls': 40, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -65, 65)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _peak_shaving(a, deadline, rng, max_rounds=300):
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        pk = int(np.argmax(c))

        valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
        if len(valid_i) == 0:
            break
        prod = a_cur[valid_i] * a_cur[pk - valid_i]
        top_k = min(8, len(valid_i))
        top_idx = np.argsort(prod)[::-1][:top_k]

        improved = False
        for idx_pos in top_idx:
            idx = valid_i[idx_pos]
            for delta in [0.03, 0.08, 0.15, 0.3, 0.6]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Also try reducing the second peak
            c2 = _full_convolve(_softmax_fn(z), _softmax_fn(z))
            # Mask the first peak
            c2_masked = c2.copy()
            c2_masked[pk] = -1
            pk2 = int(np.argmax(c2_masked))
            valid_i2 = np.arange(max(0, pk2 - n + 1), min(n, pk2 + 1))
            if len(valid_i2) > 0:
                prod2 = a_cur[valid_i2] * a_cur[pk2 - valid_i2]
                top_idx2 = np.argsort(prod2)[::-1][:5]
                for idx_pos in top_idx2:
                    idx = valid_i2[idx_pos]
                    for delta in [0.03, 0.08, 0.15]:
                        z_trial = z.copy()
                        z_trial[idx] -= delta
                        a_trial = _softmax_fn(z_trial)
                        sc = _score_norm_fn(a_trial)
                        if sc < best_score - 1e-12:
                            best_score = sc
                            best_a = a_trial.copy()
                            z = z_trial.copy()
                            improved = True
                            break
                    if improved:
                        break

            if not improved:
                if rnd % 30 == 0 and rnd > 0:
                    z = z + 0.04 * rng.normal(size=n)
                    z = z - z.mean()
                elif rnd > 40:
                    break

    return best_a, best_score


def _adam_gd(a_init, deadline, rng, lr_init=0.05, max_iter=30000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)

        if it % 30 == 0:
            beta = _adaptive_beta(c)

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -65, 65)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 3000 == 0:
            lr = max(0.003, lr * 0.85)

        if t % 8000 == 0 and t > 0:
            z = z + 0.015 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp(n, rate=2.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_power(n, p=1.0):
    t = np.linspace(0.001, 1.0, n)
    a = t ** p
    return a / a.sum()


def _make_profiles(n):
    profiles = []
    for s in [1.0, 2.0, 3.0, 5.0]:
        profiles.append(_make_ramp(n, s))
    for r in [2.0, 4.0, 6.0, 10.0]:
        profiles.append(_make_exp(n, r))
    for p in [0.5, 1.0, 2.0, 3.0]:
        profiles.append(_make_power(n, p))
    return profiles


def _try_multistart(a_init, deadline, rng, n_betas=10):
    """Try several perturbed versions and return the best."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    attempts = 3
    per_attempt_time = (deadline - time.monotonic()) / max(1, attempts)

    for _ in range(attempts):
        if time.monotonic() > deadline - 5:
            break
        z_pert = z + 0.08 * rng.normal(size=n)
        z_pert = np.clip(z_pert, -60, 60)
        z_pert = z_pert - z_pert.mean()
        a_pert = _softmax_fn(z_pert)
        sub_deadline = min(deadline - 5, time.monotonic() + per_attempt_time - 2)
        a_opt, sc_opt = _run_lbfgs_continuation(a_pert, sub_deadline, rng, n_betas=n_betas)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            z = np.log(np.maximum(best_a, 1e-15))
            z = z - z.mean()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")
            return True
        return False

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(256, 2.0)
        update_best(a0, _score_norm_fn(a0), "default_ramp")

    # --- Phase 1: Multi-scale optimization (65% of time) ---
    phase1_end = t_start + (budget_s - 30.0) * 0.65
    print(f"\nPhase 1: Multi-scale optimization (until t={phase1_end - t_start:.0f}s)")

    # Start from best profile at n=64
    n_start = 64
    profiles_64 = _make_profiles(n_start)
    a_cur = None
    sc_cur = np.inf
    for prof in profiles_64:
        sc = _score_norm_fn(prof)
        if sc < sc_cur:
            sc_cur = sc
            a_cur = prof.copy()

    # Quick L-BFGS at n=64
    quick_deadline = min(phase1_end, t_start + 15.0)
    a_cur, sc_cur = _run_lbfgs_continuation(a_cur, quick_deadline, rng, n_betas=8)
    print(f"  n=64: score_norm={sc_cur:.10f}")

    # Multi-scale up
    scales = [128, 256, 512, 1024, 2048, 4096, 8192]
    # Time allocation proportional to problem size
    total_alloc = 15 + 25 + 40 + 70 + 120 + 200 + 250
    time_alloc = {128: 15, 256: 25, 512: 40, 1024: 70, 2048: 120, 4096: 200, 8192: 250}

    for target_n in scales:
        if time.monotonic() > phase1_end - 10:
            break
        if time.monotonic() + time_alloc[target_n] * 0.3 > phase1_end:
            # Not enough time for this scale, skip or shorten
            alloc = max(5, int(phase1_end - time.monotonic() - 5))
            if alloc < 10:
                break
        else:
            alloc = time_alloc[target_n]

        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)

        opt_deadline = min(phase1_end - 3, time.monotonic() + alloc)
        n_betas = 10 if target_n <= 512 else (14 if target_n <= 2048 else 18)
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_deadline, rng, n_betas=n_betas,
                                                 aggressive=(target_n >= 2048))
        print(f"  n={target_n}: score_norm={sc_opt:.10f} (upsample was {sc_target:.10f})")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")
    else:
        a_cur = best_seq.copy()
        sc_cur = best_score

    print(f"\n  Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Fine optimization at best scale (20% of time) ---
    phase2_end = t_start + (budget_s - 30.0) * 0.85
    print(f"\nPhase 2: Fine optimization (until t={phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end - 15 and best_seq is not None:
        # Fine L-BFGS with very small beta
        n_best = len(best_seq)
        c_best = _full_convolve(best_seq, best_seq)
        max_c_best = float(c_best.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        beta_final = max(1e-9, 1e-6 * max_c_best)
        sub_deadline = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_final,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 20000, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(np.clip(result.x, -65, 65))
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    update_best(a_new, sc_new, "phase2_fine_lbfgs")
        except Exception as ex:
            print(f"  Fine L-BFGS error: {ex}")

        # Multistart from best
        if time.monotonic() < phase2_end - 10:
            ms_deadline = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.7)
            a_ms, sc_ms = _try_multistart(best_seq, ms_deadline, rng, n_betas=8)
            update_best(a_ms, sc_ms, "phase2_multistart")

    print(f"\n  Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Peak shaving + Adam GD (10% of time) ---
    phase3_end = t_start + (budget_s - 30.0) * 0.95
    print(f"\nPhase 3: Peak shaving + Adam (until t={phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        # Peak shaving
        ps_deadline = min(phase3_end - 20, time.monotonic() + 60.0)
        a_ps, sc_ps = _peak_shaving(best_seq, ps_deadline, rng, max_rounds=200)
        update_best(a_ps, sc_ps, "peak_shaving")

        # Adam GD
        if time.monotonic() < phase3_end - 10:
            gd_deadline = min(phase3_end - 5, time.monotonic() + 60.0)
            a_gd, sc_gd = _adam_gd(best_seq, gd_deadline, rng, lr_init=0.03)
            update_best(a_gd, sc_gd, "adam_gd")

    print(f"\n  Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Final polish ---
    print(f"\nPhase 4: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        # One more tiny-beta L-BFGS
        n_final = len(best_seq)
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        c_final = _full_convolve(best_seq, best_seq)
        beta_tiny = max(1e-10, 5e-7 * float(c_final.max()))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 15000, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxls': 40, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(np.clip(result.x, -65, 65))
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    update_best(a_new, sc_new, "final_polish")
        except Exception:
            pass

        # Final peak shave
        if time.monotonic() < deadline - 3:
            a_ps2, sc_ps2 = _peak_shaving(best_seq, deadline - 1, rng, max_rounds=50)
            update_best(a_ps2, sc_ps2, "final_shave")

    # --- Return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
