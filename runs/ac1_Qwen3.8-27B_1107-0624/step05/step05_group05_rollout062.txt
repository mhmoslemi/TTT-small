                best_sc = sc
                best_t = t

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _conv_variance_grad(a):
    """
    Gradient of sum_k ((a*a)_k - target)^2 with respect to a,
    where target = 1/(2n-1) (assuming sum(a)=1).
    Efficiently computed via FFT.
    """
    n = len(a)
    c = _full_convolve(a, a)
    target = 1.0 / (2 * n - 1)
    c_shifted = c - target
    a_rev = a[::-1]
    conv_ac = _full_convolve(a_rev, c_shifted)
    # dV/da[i] = 4 * sum_j a[j] * (c[i+j] - target) = 4 * (a_rev * c_shifted)[n-1+i]
    grad = 4.0 * conv_ac[n - 1:2 * n - 1]
    return grad


def _max_subgrad(a):
    """
    Subgradient of max_k (a*a)_k w.r.t. a.
    Returns gradient vector.
    """
    n = len(a)
    c = _full_convolve(a, a)
    k_max = int(np.argmax(c))
    grad = np.zeros(n)
    i_start = max(0, k_max - n + 1)
    i_end = min(k_max, n - 1)
    indices = np.arange(i_start, i_end + 1)
    grad[indices] = 2.0 * a[k_max - indices]
    return grad, k_max


def _adam_var(a_init, deadline, max_iter=5000000, lr=0.01, var_weight=1.0, max_weight=0.1):
    """
    Adam optimizer using variance gradient + max subgradient.
    Works in log-domain with softmax parameterization.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    patience = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Variance gradient (smooth)
        grad_var = _conv_variance_grad(a)

        # Max subgradient (non-smooth, exact)
        grad_max, _ = _max_subgrad(a)

        # Combined gradient
        grad = var_weight * grad_var + max_weight * grad_max

        # Project gradient to tangent space of simplex (remove mean direction)
        W = float(np.dot(a, grad))
        grad_tangent = grad - W * a

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad_tangent
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_tangent * grad_tangent)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Periodic scoring
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                if best_score - sc > 1e-8:
                    patience = 0
                else:
                    patience += 1
                best_score = sc
                best_a = a.copy()
            else:
                patience += 1

        # Adaptive learning rate
        if t % 5000 == 0:
            if patience == 0:
                current_lr = min(0.05, current_lr * 1.05)
            elif patience > 5:
                current_lr = max(0.0005, current_lr * 0.9)
            prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _adam_smooth_max(a_init, deadline, max_iter=3000000, lr=0.005, beta_start=None, beta_end=None):
    """
    Adam with log-sum-exp smoothing of the max, transitioning to exact max.
    This provides a smooth approximation that becomes exact over time.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    if beta_start is None:
        c0 = _full_convolve(a_init, a_init)
        beta_start = 0.01 * c0.max()
    if beta_end is None:
        beta_end = beta_start * 0.001

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Log-sum-exp smoothing parameter (decreases over time)
        progress = min(1.0, t / max_iter)
        beta = beta_start * (beta_end / beta_start) ** progress
        c_shifted = (c - c_max) / max(beta, 1e-12)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        # Gradient via weighted convolution
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

        # Project to tangent space
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if t % 10000 == 0:
            sc_check = _score_norm(_softmax(z))
            if sc_check < best_score * 0.999:
                current_lr = min(0.03, current_lr * 1.1)
            elif t > 50000:
                current_lr = max(0.001, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _coordinate_descent(a, max_sweeps=50, deadline=None):
    """
    Coordinate descent: for each coordinate i, find optimal value
    that minimizes the max convolution.
    """
    n = len(a)
    a = a.copy()
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            old_val = a[i]
            # Binary search for optimal a[i]
            lo, hi = 0.0, old_val * 3 + 0.1
            # Quick check: try a few values
            best_val = old_val
            best_max = _full_convolve(a, a).max()
            for frac in [0.5, 0.8, 1.2, 1.5, 2.0]:
                a[i] = old_val * frac
                rem = 1.0 - a[i]  # need to redistribute to others
                if rem < 0:
                    a[i] = old_val
                    continue
                # Redistribute remaining to others proportionally
                others = np.arange(n) != i
                if others.sum() == 0:
                    a[i] = old_val
                    continue
                a[others] = a[others] * (rem / a[others].sum()) if a[others].sum() > 0 else rem / others.sum()
                new_max = _full_convolve(a, a).max()
                if new_max < best_max:
                    best_max = new_max
                    best_val = a.copy()
            a = best_val if best_max < _full_convolve(a, a).max() else a
            a[i] = best_val[i]
            # Renormalize
            s = a.sum()
            if s > 0:
                a = a / s
        # Full recheck
        if _score_norm(a) < _score_norm(a) - 1e-10:
            improved = True
    return a
