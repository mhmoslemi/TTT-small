def _active_set_solve(S, support, n):
    """
    Solve the KKT system for a given active set S and support.
    S: set of lag indices where conv is at maximum
    support: set of indices where a_i > 0
    Returns a or None if no valid solution.
    """
    # Build the linear system:
    # For each i in support: sum_{k in N(i)} a[k] = c (constant)
    # Plus: sum(a) = 1
    # 
    # This gives |support| + 1 equations in |support| + 1 unknowns (a[i] for i in support, plus c).
    
    import scipy.sparse as sp
    import scipy.sparse.linalg as spla
    
    idx = list(support)
    n_idx = len(idx)
    
    # Build matrix A and vector b for: A * x = b
    # x = [a[i_0], a[i_1], ..., a[i_{n_idx-1}], c]
    # Row for each i in support: sum_{j: idx[j] in N(i)} x[j] - c = 0
    # Last row: sum x[j] = 1
    
    rows = []
    cols = []
    vals = []
    
    for i_val in idx:
        row_idx = idx.index(i_val)
        # N(i_val) = {k in [0,n-1] : k + i_val in S}
        # But we only care about k in support
        for j_val, k_val in enumerate(idx):
            if (k_val + i_val) in S:
                rows.append(row_idx)
                cols.append(j_val)
                vals.append(1.0)
        # -c term
        rows.append(row_idx)
        cols.append(n_idx)  # index of c
        vals.append(-1.0)
    
    # Sum constraint
    for j in range(n_idx):
        rows.append(n_idx)
        cols.append(j)
        vals.append(1.0)
    # c doesn't appear in sum constraint
    
    A = sp.csr_matrix((vals, (rows, cols)), shape=(n_idx + 1, n_idx + 1))
    b = np.zeros(n_idx + 1)
    b[n_idx] = 1.0
    
    try:
        x = spla.spsolve(A, b)
    except Exception:
        return None
    
    a = np.zeros(n)
    for j, i_val in enumerate(idx):
        a[i_val] = x[j]
    
    c_const = x[n_idx]
    
    # Check: all a[i] >= 0 for i in support
    if np.any(x[:n_idx] < -1e-10):
        return None
    
    # Check: c_k < max for k not in S (with some tolerance)
    # Compute actual conv
    aa = np.maximum(a, 0)
    s = aa.sum()
    if s < 1e-15:
        return None
    aa = aa / s  # normalize
    conv = _full_convolve(aa, aa)
    max_c = conv.max()
    
    # Check that non-active lags are strictly below max
    for k in range(len(conv)):
        if k not in S and conv[k] > max_c * 1.001:
            return None
    
    # Check that active lags are at max
    for k in S:
        if k < len(conv) and conv[k] < max_c * 0.99:
            return None
    
    return aa

"""
Adam-based optimizer with log-sum-exp smoothing, active-set polishing,
and palindromic variant exploration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: per-coordinate Adam adaptation, conservative beta schedule,
exact active-set subgradient refinement, and symmetry-constrained search.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _to_log(a):
    return np.log(np.maximum(a, 1e-15))


def _smoothed_obj_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        obj = 2.0 * n * c_max
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        obj = 2.0 * n * (c_max + beta * np.log(s))
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return obj, grad


def _exact_active_grad(z, a, n, frac=0.005):
    c = _full_convolve(a, a)
    c_max = c.max()
    thresh = c_max * (1.0 - frac)
    active = c >= thresh
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad, c_max

                   b1=0.9, b2=0.999, patience=300, use_exact=False, exact_frac=0.005):
    z = z_init.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_obj = np.inf
    best_z = z.copy()
    best_score = _score(_softmax(z))
    best_a = _softmax(z).copy()
    no_improve = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        if use_exact and i > 100:
            a = _softmax(z)
            grad, c_max = _exact_active_grad(z, a, n, exact_frac)
            obj = 2.0 * n * c_max / (a.sum() ** 2)
        else:
            obj, grad = _smoothed_obj_grad(z, beta, n)

        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        progress = i / max_steps
        lr = base_lr * (0.5 + 0.5 * np.cos(np.pi * progress))
        lr = max(lr, base_lr * 0.005)

        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_z = z.copy()
            no_improve = 0
        else:

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)

                best_a = a_check.copy()

    sc_final = _score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    else:
        a_final = best_a

def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.05, frac=0.002):

    z = _to_log(a_init)
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

        a = _softmax(z)

        c_max = c.max()
        if c_max < 1e-15:

        active = c >= thresh
        if active.sum() == 0:

        grad = 4.0 * n * a * (h - W)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:

        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

    sc_final = _score(a_final)
    if sc_final < best_sc:

        best_a = a_final.copy()

    return best_a, best_sc


def _palindromic_score(a_half, n_full):
    """Compute score for palindromic sequence from its first half."""
    if n_full % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, a_half[-2::-1]])
    return _score(a)


def _palindromic_optimize(a_init, deadline, max_steps=3000, base_lr=0.002):
    """Optimize palindromic sequence: a[i] = a[n-1-i]."""
    n = len(a_init)
    half_len = n // 2
    if n % 2 == 0:
        a_half = a_init[:half_len].copy()
    else:
        a_half = a_init[:half_len + 1].copy()

    z = _to_log(a_half)
    z = z - z.mean()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    t = 0
    best_sc = _score(a_init)
    best_a = a_init.copy()
    no_improve = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        if n % 2 == 0:
            a = np.concatenate([_softmax(z), _softmax(z)[::-1]])
        else:
            a = np.concatenate([_softmax(z), _softmax(z)[-2::-1]])

        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / (c_max * 0.001)
        e = np.exp(shifted)
        s = e.sum()
        w = e / s

        # Gradient w.r.t. a_half
        # d/d a_half[j] of sum_k w_k * c_k
        # = sum_k w_k * dc_k/d a_half[j]
        # For palindromic, dc_k/d a_half[j] = 2 * dc_k/d a[j] (since a[n-1-j] = a_half[j] too)
        # Actually: a[j] = a_half[j] and a[n-1-j] = a_half[j] (for j < n/2)
        # dc_k/d a[j] contributes to da_half[j], and dc_k/d a[n-1-j] also contributes to da_half[j]
        # So: dc_k/d a_half[j] = dc_k/d a[j] + dc_k/d a[n-1-j]

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad_full = 4.0 * n * a * (h - W)

        if n % 2 == 0:
            grad_half = grad_full[:half_len] + grad_full[half_len:][:][:half_len][::-1]
            grad_half = grad_full[:half_len] + grad_full[n - 1 - np.arange(half_len)]
        else:
            first_half = half_len
            grad_half = grad_full[:first_half] + grad_full[n - 1 - np.arange(first_half)]
            # Middle element
            grad_half = np.concatenate([grad_half, [grad_full[n // 2]]])

        t += 1
        m = 0.9 * m + 0.1 * grad_half
        v = 0.999 * v + 0.001 * grad_half * grad_half
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        progress = i / max_steps
        lr = base_lr * (0.5 + 0.5 * np.cos(np.pi * progress))
        lr = max(lr, base_lr * 0.005)

        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 200 == 0:
            if n % 2 == 0:
                a_check = np.concatenate([_softmax(z), _softmax(z)[::-1]])
            else:
                a_check = np.concatenate([_softmax(z), _softmax(z)[-2::-1]])
            sc = _score(a_check)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.15 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _perturb(a, sigma, rng):
    z = _to_log(a)
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return _softmax(z)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase Adam optimizer with active-set polishing and palindromic search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "tapered")
        sc0 = _score(a0)
        print(f"No parent, using tapered n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "init_tapered")

    n = n_main

    # --- Phase 1: Adam with moderate beta ---
    print(f"\nPhase 1: Adam moderate beta (t={time.monotonic() - t_start:.1f}s)")
    z0 = _to_log(a0)
    z0 = z0 - z0.mean()

    c_init = _full_convolve(a0, a0)
    max_c_init = float(c_init.max())

    betas_p1 = [max_c_init * f for f in [0.05, 0.02, 0.01, 0.005, 0.002]]
    current_a = a0.copy()

    for bi, beta in enumerate(betas_p1):
        phase_deadline = min(deadline - 300.0, time.monotonic() + 120.0)
        if time.monotonic() > phase_deadline - 5:
            break
        z_cur = _to_log(current_a)
        z_cur = z_cur - z_cur.mean()
        a_new, sc_new = _adam_optimize(z_cur, beta, n, phase_deadline,
                                       max_steps=4000, base_lr=0.004)
        if sc_new < _score(current_a) - 1e-12:
            current_a = a_new
        update_best(current_a, _score(current_a), f"p1_beta{bi}")

    # --- Phase 2: Adam with fine beta ---
    print(f"\nPhase 2: Adam fine beta (t={time.monotonic() - t_start:.1f}s)")
    c_cur = _full_convolve(current_a, current_a)
    max_c_cur = float(c_cur.max())
    betas_p2 = [max_c_cur * f for f in [0.001, 0.0005, 0.0002, 0.0001]]
    current_a = best_seq.copy() if best_seq is not None else current_a

    for bi, beta in enumerate(betas_p2):
        if time.monotonic() > deadline - 200.0:
            break
        phase_deadline = min(deadline - 150.0, time.monotonic() + 100.0)
        if time.monotonic() > phase_deadline - 5:
            break
        z_cur = _to_log(current_a)
        z_cur = z_cur - z_cur.mean()
        a_new, sc_new = _adam_optimize(z_cur, beta, n, phase_deadline,
                                       max_steps=4000, base_lr=0.002,
                                       use_exact=True, exact_frac=0.005)
        if sc_new < _score(current_a) - 1e-12:
            current_a = a_new
        update_best(current_a, _score(current_a), f"p2_beta{bi}")

    # --- Phase 3: Subgradient polishing with very tight active set ---
    print(f"\nPhase 3: Subgradient polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        for frac in [0.005, 0.002, 0.001, 0.0005]:
            if time.monotonic() > deadline - 60:
                break
            phase_deadline = min(deadline - 50.0, time.monotonic() + 60.0)
            a_new, sc_new = _subgrad_polish(best_seq, phase_deadline,
                                            n_steps=2000, lr_init=0.03, frac=frac)
            update_best(a_new, sc_new, f"sg_frac{frac}")

    # --- Phase 4: Palindromic optimization ---
    print(f"\nPhase 4: Palindromic (t={time.monotonic() - t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        # Start from current best, project to palindromic
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = np.maximum(a_sym, 1e-10)
        a_sym = a_sym / a_sym.sum()
        sc_sym = _score(a_sym)
        print(f"  Palindromic init: score={sc_sym:.10f}")
        update_best(a_sym, sc_sym, "pali_init")

        pali_deadline = min(deadline - 30.0, time.monotonic() + 80.0)
        a_pali, sc_pali = _palindromic_optimize(a_sym, pali_deadline,
                                                max_steps=3000, base_lr=0.003)
        update_best(a_pali, sc_pali, "pali_opt")

        # Polish the palindromic result
        if time.monotonic() < deadline - 20:
            a_pali2, sc_pali2 = _subgrad_polish(a_pali, deadline - 15,
                                                n_steps=1000, lr_init=0.02, frac=0.001)
            update_best(a_pali2, sc_pali2, "pali_polish")

    # --- Phase 5: Multi-restart perturbation ---
    print(f"\nPhase 5: Multi-restart (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 25:
        n_restarts = min(3, max(1, int((deadline - 25 - time.monotonic()) / 60)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if i == 0:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb(best_seq, 0.005, rng)
            else:
                # Fourier perturbation
                z = _to_log(best_seq)
                t_arr = np.arange(n)
                freq = rng.integers(1, max(2, n // 8))
                amp = 0.01 * rng.uniform(0.5, 1.5)
                phase = rng.uniform(0, 2 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
                z = z - z.mean()
                p_start = _softmax(z)

            sc_start = _score(p_start)
            print(f"  Restart {i}: start_score={sc_start:.10f}")

            restart_deadline = min(deadline - 15.0, time.monotonic() + 50.0)
            z_r = _to_log(p_start)
            z_r = z_r - z_r.mean()
            c_r = _full_convolve(p_start, p_start)
            beta_r = float(c_r.max()) * 0.001
            a_r, sc_r = _adam_optimize(z_r, beta_r, n, restart_deadline,
                                       max_steps=2000, base_lr=0.002,
                                       use_exact=True, exact_frac=0.003)
            update_best(a_r, sc_r, f"restart_{i}")

    # --- Phase 6: Final fine polish ---
    print(f"\nPhase 6: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # Very tight subgradient
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 3,
                                        n_steps=1000, lr_init=0.01, frac=0.0003)
        update_best(a_fin, sc_fin, "final_sg")

        # One more Adam with tiny beta
        if time.monotonic() < deadline - 3:
            z_f = _to_log(best_seq)
            z_f = z_f - z_f.mean()
            c_f = _full_convolve(best_seq, best_seq)
            beta_f = float(c_f.max()) * 0.00005
            a_f, sc_f = _adam_optimize(z_f, beta_f, n, deadline - 2,
                                       max_steps=500, base_lr=0.001,
                                       use_exact=True, exact_frac=0.001)
            update_best(a_f, sc_f, "final_adam")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
