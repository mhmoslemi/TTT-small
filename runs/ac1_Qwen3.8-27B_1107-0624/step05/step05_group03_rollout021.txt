```python
def kkt_solve(a_init, m_top=5):
    n = len(a_init)
    c = convolve(a_init, a_init)
    C = c.max()
    # Find top m lags
    top_lags = np.argsort(c)[-m_top:]
    # Equal weights (simplest choice)
    lam = np.ones(m_top) / m_top
    k_list = top_lags.tolist()
    
    # Build matrix M: M[i, j] = sum_r lam[r] * 1[j == k_r - i]
    # M is sparse (banded)
    from scipy.sparse import lil_matrix, csr_matrix
    M = lil_matrix((n, n))
    for i in range(n):
        for r, k_r in enumerate(k_list):
            j = k_r - i
            if 0 <= j < n:
                M[i, j] += lam[r]
    M = M.tocsr()
    
    # Solve M @ a = mu * 1, sum(a) = 1, a >= 0
    # First ignore a >= 0
    # M @ a = mu * 1 => a = mu * M^{-1} @ 1
    # sum(a) = mu * 1^T @ M^{-1} @ 1 = 1 => mu = 1 / (1^T @ M^{-1} @ 1)
    
    try:
        from scipy.sparse.linalg import spsolve
        rhs = np.ones(n)
        a_sol = spsolve(M, rhs)
        if not np.all(np.isfinite(a_sol)):
            return None
        mu = 1.0 / a_sol.sum()
        a_sol = mu * a_sol
        if np.all(a_sol >= 0):
            return a_sol
        else:
            # Active set: set negative entries to 0
            support = a_sol > 0
            if support.sum() < 2:
                return None
            M_sub = M[np.ix_(support, support)]
            rhs_sub = np.ones(support.sum())
            a_sub = spsolve(M_sub.tocsr(), rhs_sub)
            if not np.all(np.isfinite(a_sub)):
                return None
            mu_sub = 1.0 / a_sub.sum()
            a_sub = mu_sub * a_sub
            if np.all(a_sub >= 0):
                a_full = np.zeros(n)
                a_full[support] = a_sub
                return a_full
            else:
                return None
    except:
        return None
```

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import lil_matrix, csr_matrix
from scipy.sparse.linalg import spsolve
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _project_sum1_nonneg(a):
    """Project onto {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s

def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_uniform(n):

def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-15)
    return a / a.sum()

def _make_bimodal(n, sep, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2.0
    c2 = 0.5 + sep / 2.0
    a = np.exp(-(t - c1)**2 / (2*width**2)) + np.exp(-(t - c2)**2 / (2*width**2))
    a = np.maximum(a, 1e-15)
    return a / a.sum()

    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)

def _interp_to_n(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = len(a)
    if n_old == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

    Solve KKT conditions: find a that satisfies
    sum_r lambda_r * a[k_r - i] = mu for all i
    with sum(a) = 1, a >= 0.
    Returns (a_solution, score) or (None, inf).
    """
    n = len(a_init)
    c = _full_convolve(a_init, a_init)
    C = float(c.max())
    
    # Find top m lags
    top_idx = np.argsort(c)[-m_top:]
    k_list = [int(top_idx[r]) for r in range(m_top)]
    
    # Weights: use softmax of the top c values for better conditioning
    c_top = c[k_list]
    c_shifted = c_top - c_top.max()
    lam = np.exp(20.0 * c_shifted) / np.exp(20.0 * c_shifted).sum()
    
    # Build matrix M: M[i, j] = sum_r lam[r] * 1[j == k_r - i]
    M = lil_matrix((n, n))
    for i in range(n):
        for r in range(m_top):
            j = k_list[r] - i
            if 0 <= j < n:
                M[i, j] += lam[r]
    M_csr = M.tocsr()
    
    # Solve M @ a = mu * 1, sum(a) = 1
    try:
        rhs = np.ones(n)
        a_sol = spsolve(M_csr, rhs)
        if not np.all(np.isfinite(a_sol)):
            return None, np.inf
        s = a_sol.sum()
        if abs(s) < 1e-15:
            return None, np.inf
        a_sol = a_sol / s
        
        # Check non-negativity
        if np.all(a_sol >= -1e-10):
            a_sol = np.maximum(a_sol, 0.0)
            a_sol = a_sol / a_sol.sum()
            sc = _score_norm(a_sol)
            return a_sol, sc
        else:
            # Active set: iteratively remove negative entries
            support = np.ones(n, dtype=bool)
            for _ in range(10):
                n_sub = support.sum()
                if n_sub < 2:
                    return None, np.inf
                idx = np.where(support)[0]
                M_sub = M_csr[np.ix_(idx, idx)]
                rhs_sub = np.ones(n_sub)
                a_sub = spsolve(M_sub.tocsr(), rhs_sub)
                if not np.all(np.isfinite(a_sub)):
                    return None, np.inf
                s_sub = a_sub.sum()
                if abs(s_sub) < 1e-15:
                    return None, np.inf
                a_sub = a_sub / s_sub
                if np.all(a_sub >= -1e-10):
                    a_full = np.zeros(n)
                    a_full[idx] = np.maximum(a_sub, 0.0)
                    a_full = a_full / a_full.sum()
                    sc = _score_norm(a_full)
                    return a_full, sc
                else:
                    # Remove the most negative
                    worst = idx[int(np.argmin(a_sub))]
                    support[worst] = False
            return None, np.inf
    except Exception:
        return None, np.inf

def _kkt_iterate(a_init, max_iters=50, deadline=None):
    """Iteratively solve KKT with updated active sets."""
    a = a_init.copy()
    a = _project_sum1_nonneg(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Try different m_top values
        for m in [3, 5, 7, 10, 15]:
            a_kkt, sc_kkt = _kkt_solve(a, m_top=m)
            if a_kkt is not None and sc_kkt < best_sc:
                best_sc = sc_kkt
                best_a = a_kkt.copy()
                a = a_kkt.copy()
                print(f"    KKT iter {it}, m={m}: score={sc_kkt:.10f}")
        
        # Check convergence
        c = _full_convolve(a, a)
        C = c.max()
        c_mean = c.mean()
        ratio = C / c_mean
        if ratio < 1.001:
            break
    
    return best_a, best_sc

def _lbfgs_continuation(a_init, deadline, beta_starts=None, max_wall_time=None):
    """L-BFGS with logsumexp continuation, very fine betas."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_starts is None:
        beta_starts = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00001, 0.000005, 0.000001]
    
    betas = [b * max_c0 for b in beta_starts]
    
    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 20.0
    
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 2.0:
            break
        level_end = min(deadline, time.monotonic() + per_level)
        
        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300 or not np.isfinite(s_):
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300 or not np.isfinite(s_):
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n-1:2*n-1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        # Time-box this L-BFGS call
        t0 = time.monotonic()
        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 100, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
                    print(f"    L-BFGS beta={beta:.2e}: score={s_new:.10f}")
        except Exception:
            pass
        
        # Check time
        if time.monotonic() - t0 > per_level:
            pass  # Just continue to next beta
    
    return best_a, best_score

def _subgradient_method(a_init, deadline, lr_init=0.01, momentum=0.9):
    """Direct subgradient method with projection."""
    n = len(a_init)
    a = a_init.copy()
    a = _project_sum1_nonneg(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    v = np.zeros(n)  # momentum
    lr = lr_init
    stalled = 0
    
    max_iters = 100000
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        C = c.max()
        max_k = int(np.argmax(c))
        
        # Subgradient: g_i = a[max_k - i] (if valid)
        g = np.zeros(n)
        for i in range(n):
            j = max_k - i
            if 0 <= j < n:
                g[i] = a[j]
        
        # Average with a few top lags for stability
        top3 = np.argsort(c)[-3:]
        g_avg = np.zeros(n)
        for k_val in top3:
            k_val = int(k_val)
            for i in range(n):
                j = k_val - i
                if 0 <= j < n:
                    g_avg[i] += a[j]
        g_avg /= 3.0
        
        # Momentum update
        v = momentum * v + (1 - momentum) * g_avg
        a_new = a - lr * v
        a_new = _project_sum1_nonneg(a_new)
        
        sc_new = _score_norm(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            stalled = 0
            if it % 1000 == 0:
                print(f"    Subgrad iter {it}: score={sc_new:.10f}, lr={lr:.6f}")
        else:
            stalled += 1
        
        a = a_new
        
        # Adaptive learning rate
        if stalled > 200:
            lr *= 0.95
            stalled = 0
        elif it % 5000 == 0 and it > 0:
            lr *= 0.9
        
        if lr < 1e-8:
            break
    
    return best_a, best_sc
