```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _construct_lp_matrix(a, active_lags):
    """
    Construct the constraint matrix for the LP.
    Row for lag k: coefficients of b in (a*b)[k].
    """
    n = len(a)
    m = len(active_lags)
    A = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        # (a*b)[k] = sum_i a[i] * b[k-i] for valid i
        # j = k-i, so i = k-j
        # Valid: 0 <= i < n and 0 <= j < n
        # i.e., 0 <= k-j < n and 0 <= j < n
        # j ranges: max(0, k-n+1) <= j <= min(k, n-1)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            A[idx, j] = a[i]
    return A

def _solve_lp(a, M, active_lags):
    """
    Solve the LP: max sum(b) s.t. (a*b)[k] <= M for k in active_lags, b >= 0.
    Returns b (unnormalized).
    """
    n = len(a)
    A = _construct_lp_matrix(a, active_lags)
    b_obj = -np.ones(n)  # minimize -sum(b)
    res = linprog(b_obj, A_ub=A, b_ub=np.full(len(active_lags), M), bounds=(0, None), method='highs')
    if res.success:
        return res.x
    return None

def _fw_step(a, deadline, max_lp_iter=500, eps_active=0.99):
    """
    Frank-Wolfe step using LP direction finding.
    """
    n = len(a)
    a = a / a.sum()  # normalize to sum=1
    c = _full_convolve(a, a)
    M = float(c.max())
    
    for it in range(max_lp_iter):
        if time.monotonic() > deadline:
            break
        
        # Determine active set
        threshold = M * eps_active
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(c))])
        
        # Solve LP
        b = _solve_lp(a, M, active_lags)
        if b is None:
            break
        
        b_sum = float(b.sum())
        if b_sum < 1e-15:
            break
        
        b = b / b_sum  # normalize to sum=1
        
        # Line search for step size
        best_t = 0.0
        best_val = M
        for t_candidate in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = (1 - t_candidate) * a + t_candidate * b
            c_try = _full_convolve(a_try, a_try)
            val = float(c_try.max())
            if val < best_val:
                best_val = val
                best_t = t_candidate
        
        if best_t == 0.0:
            # Try smaller steps
            for t_candidate in [0.01, 0.05, 0.1]:
                a_try = (1 - t_candidate) * a + t_candidate * b
                c_try = _full_convolve(a_try, a_try)
                val = float(c_try.max())
                if val < best_val:
                    best_val = val
                    best_t = t_candidate
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * b
        a = np.maximum(a, 0)  # ensure non-negativity
        a = a / a.sum()
        
        c = _full_convolve(a, a)
        new_M = float(c.max())
        if new_M >= M * (1 - 1e-10):
            # No improvement, try smaller active set
            eps_active *= 0.9
            if eps_active < 0.95:
                break
        M = new_M
    
    return a
```

```python
"""
Combined LP-guided Frank-Wolfe and L-BFGS optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences of length n.
Uses log-sum-exp smoothing for the non-smooth max, subgradient polishing,
and LP-based direction finding for large-scale updates.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _subgrad_step(a, c, n, active_frac=0.001):
    """Compute subgradient of max(c) w.r.t. a using active set."""
    c_max = c.max()
    tol = active_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    # grad[j] = 2 * sum_k w[k] * a[k-j] (if 0 <= k-j < n)
    # = 2 * (w * a_rev)[j + n - 1] ... let me think
    # (w * a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)]
    # We want: sum_k w[k] * a[k-j] for 0 <= k-j < n
    # Let i = k-j, so k = i+j: sum_i w[i+j] * a[i]
    # This is (w * a)[j + n - 1] ... no
    # Actually: sum_i w[i+j] * a[i] = (flip_w * a)[n-1-j] where flip_w = w[::-1]
    # Or more simply: it's the cross-correlation
    # Let me just use FFT
    a_rev = a[::-1]
    wa = _full_convolve(w, a_rev)
    # wa[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-m+k]
    # We want sum_k w[k] * a[k-j] = sum_k w[k] * a_rev[n-1-k+j]
    # = wa[n-1+j] ... let me check
    # wa[m] = sum_k w[k] * a_rev[m-k]
    # Set m = n-1+j: wa[n-1+j] = sum_k w[k] * a_rev[n-1+j-k]
    # a_rev[n-1+j-k] = a[(n-1)-(n-1+j-k)] = a[k-j]
    # So wa[n-1+j] = sum_k w[k] * a[k-j] ✓
    h = wa[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 2.0 * h  # gradient of max_k c[k] w.r.t. a


def _subgrad_polish(a_init, deadline, n_max_steps=5000, lr_init=0.01):
    """Subgradient polishing in the primal with projection to simplex + nonneg."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s  # normalize
    best_a = a.copy()
    best_sc = _score(a)
    lr = lr_init
    momentum = np.zeros(n)
    mu = 0.9
    
    for i in range(n_max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        grad = _subgrad_step(a, c, n)
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        a = a - lr * momentum
        
        # Project to {x >= 0, sum(x) = 1}
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        if i % 100 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                lr *= 0.95
            else:
                lr *= 0.85
                momentum *= 0.5
        
        if lr < 1e-8:
            break
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_a = a.copy()
        best_sc = sc_final
    
    return best_a, best_sc


def _lp_direction(a, M, active_lags, max_constraints=100):
    """
    Solve LP: max sum(b) s.t. (a*b)[k] <= M for k in active_lags, b >= 0.
    Returns normalized b (sum=1) or None.
    """
    n = len(a)
    # Limit number of constraints
    if len(active_lags) > max_constraints:
        # Take the most "active" ones
        c_vals = np.array([float(a.sum() * a[active_lags[0]])] * len(active_lags))  # placeholder
        # Actually, just take the first max_constraints
        active_lags = active_lags[:max_constraints]
    
    m = len(active_lags)
    if m == 0:
        return None
    
    # Build constraint matrix A_ub (m x n)
    A_ub = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            i = k - j
            A_ub[idx, j] = a[i]
    
    b_ub = np.full(m, M)
    c_obj = -np.ones(n)  # minimize -sum(b)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if res.success and res.x is not None:
            b = res.x
            b_sum = float(b.sum())
            if b_sum > 1e-10:
                return b / b_sum
    except Exception:
        pass
    
    return None


def _fw_loop(a_init, deadline, max_iter=3000, max_active=50):
    """LP-guided Frank-Wolfe loop."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    
    c = _full_convolve(a, a)
    M = float(c.max())
    best_a = a.copy()
    best_M = M
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Determine active set
        threshold = M * 0.999
        active_mask = c >= threshold
        active_lags = np.where(active_mask)[0]
        if len(active_lags) > max_active:
            # Take the top max_active
            idx = np.argsort(c[active_lags])[::-1][:max_active]
            active_lags = active_lags[idx]
        
        # Solve LP
        b = _lp_direction(a, M, active_lags)
        if b is None:
            # Fall back to subgradient
            grad = _subgrad_step(a, c, n)
            b = a - 0.01 * grad
            b = np.maximum(b, 0)
            bs = b.sum()
            if bs < 1e-15:
                break
            b = b / bs
        else:
            # Check if b is "close" to a (to avoid oscillation)
            sim = float(np.dot(a, b))
            if sim < 0.5:
                b = 0.5 * a + 0.5 * b
                b = b / b.sum()
        
        # Try a few step sizes
        best_t = 0
        best_new_M = M
        for t in [0.05, 0.1, 0.2, 0.3]:
            a_try = (1 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            c_try = _full_convolve(a_try, a_try)
            new_M_val = float(c_try.max())
            if new_M_val < best_new_M:
                best_new_M = new_M_val
                best_t = t
        
        if best_t == 0:
            no_improve += 1
            if no_improve > 200:
                break
            # Try subgradient step
            grad = _subgrad_step(a, c, n)
            a = a - 0.005 * grad
            a = np.maximum(a, 0)
            st = a.sum()
            a = a / st
            c = _full_convolve(a, a)
            M = float(c.max())
        else:
            no_improve = 0
            a = (1 - best_t) * a + best_t * b
            a = np.maximum(a, 0)
            st = a.sum()
            a = a / st
            c = _full_convolve(a, a)
            M = float(c.max())
        
        if M < best_M:
            best_M = M
            best_a = a.copy()
    
    return best_a


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return float(2.0 * n * c_max)
    return float(2.0 * n * (c_max + beta * np.log(s)))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_run(a_init, beta, deadline, max_iter=2000):
    """Run L-BFGS-B on the smoothed objective."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        res = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 50, 'disp': False}
        )
        return _softmax(res.x)
    except Exception:
        return a_init


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

            a_new = _lbfgs_run(current_a, beta, deadline - 60, max_iter)
            sc = _score(a_new)
            if sc < best_score:
                update_best(a_new, sc)
                current_a = a_new.copy()
            else:
                # Didn't improve, try anyway
                if sc < _score(current_a):
                    current_a = a_new.copy()
    
    # Phase 4: More subgradient on the best
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 4: Final subgradient (t={time.monotonic()-t_start:.1f}s)")
        a_sg2, sc_sg2 = _subgrad_polish(best_a, deadline - 10, n_max_steps=3000, lr_init=0.003)
        update_best(a_sg2)
    
    # Phase 5: Restarts
    if time.monotonic() < deadline - 30:
        print(f"\nPhase 5: Restarts (t={time.monotonic()-t_start:.1f}s)")
        remaining = deadline - 30 - time.monotonic()
        n_restarts = min(3, max(1, int(remaining / 100)))
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            p_start = _perturb(best_a, 0.02 * (i + 1), rng)
            sc_s = _score(p_start)
            if sc_s < best_score:
                update_best(p_start, sc_s)
            
            # Quick FW from perturbed start
            fw_d = deadline - 20 - time.monotonic()
            if fw_d > 30:
                a_r = _fw_loop(p_start, time.monotonic() + min(60, fw_d), max_iter=500, max_active=30)
                update_best(a_r)
            
            # Quick subgrad
            if time.monotonic() < deadline - 15:
                a_r2, _ = _subgrad_polish(best_a, deadline - 10, n_max_steps=1000, lr_init=0.003)
                update_best(a_r2)
    
    # Final
    print(f"\n{'='*50}")
    if best_a is not None:
        final = np.clip(best_a, 0.0, 1000.0)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]
    
    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
