"""
Optimizing non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses gradient-based optimization with adaptive soft-max temperature,
LP-based descent directions, and multi-restart exploration.
"""
import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_z(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_a(a, w):
    """
    Compute gradient of sum_k w_k * c_k w.r.t. a, where c = conv(a,a).
    g_a[i] = 2 * sum_k w[k] * a[k-i]
    """
    n = len(a)
    N = _next_pow2(3 * n)  # Need N >= (2n-1) + n - 1
    W = np.fft.fft(w, N)
    A = np.fft.fft(a, N)
    corr = np.fft.ifft(W * np.conj(A)).real
    return 2.0 * corr[:n]


def _grad_to_z(grad_a, a):
    """Map gradient from a-space to z-space (where a = softmax(z))."""
    # dL/dz_k = a_k * (g_a_k - sum_i g_a_i * a_i)
    dot = np.dot(grad_a, a)
    return a * (grad_a - dot)


def _soft_weights(c, beta):
    """Compute soft-max weights."""
    c_max = c.max()
    x = (c - c_max) / beta
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return w


def _make_uniform(n):
    return np.ones(n) / n


def _make_hann(n):
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    return w / w.sum()


def _make_raised_cos(n):
    x = np.arange(n) / (n - 1)
    w = np.cos(np.pi * x) ** 2
    w = w / w.sum()
    return w


def _make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2
    w = np.exp(-x**2 / (2 * (sigma_frac * n)**2))
    return w / w.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_z(z)


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, M, n, time_limit=20):
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _gradient_optimize(a_init, deadline, lr=0.005, beta_start=0.5, beta_end=0.01,
                       max_iters=500000, print_interval=10000):
    """
    Main gradient-based optimization in log-space.
    Adaptive temperature from beta_start to beta_end.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c = _full_convolve(a_init, a_init)
    best_m = float(c.max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t_step += 1

        # Adaptive beta
        frac = min(1.0, it / max(1000, max_iters // 3))
        beta = beta_start * (beta_end / beta_start) ** frac

        a = _softmax_z(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        if cm < best_m:
            best_m = cm
            best_a = a.copy()

        w = _soft_weights(c, beta)
        grad_a = _compute_grad_a(a, w)
        grad_z = _grad_to_z(grad_a, a)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1 - b1) * grad_z
        v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
        mh = m_buf / (1 - b1 ** t_step)
        vh = v_buf / (1 - b2 ** t_step)

        update = lr * mh / (np.sqrt(vh) + eps)
        u_norm = np.linalg.norm(update)
        if u_norm > 1.0:
            update = update * (1.0 / u_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if (it + 1) % print_interval == 0:
            sc = 2.0 * n * cm
            print(f"    iter {it+1}: beta={beta:.4f}, max_c={cm:.6e}, "
                  f"score={sc:.8f}, best_score={2.0*n*best_m:.8f}, "
                  f"time={time.monotonic():.1f}s")

    # Final check
    a_final = _softmax_z(z)
    c_final = _full_convolve(a_final, a_final)
    m_final = float(c_final.max())
    if m_final < best_m:
        best_m = m_final
        best_a = a_final.copy()

    return best_a

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    s = float(a.sum())

    c = _full_convolve(a, a)

def _softmax_z(z):

    e = np.exp(z)
    s = e.sum()

def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_final = _full_convolve(a_final, a_final)
    m_final = float(c_final.max())
    if m_final < best_m:
        best_m = m_final
        best_a = a_final.copy()

    return best_a, 2.0 * n * best_m


def _lp_phase(f_init, deadline, max_steps=30, print_interval=3):
    """LP-based optimization with line search."""
    n = len(f_init)
    f = f_init.copy()
    s = float(f.sum())
    if s < 1e-15:
        return f
    f = f / s

    c = _full_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M

    A_cache = _build_lp_matrix(f, n)
    f_for_matrix = f.copy()
    stuck = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        if np.max(np.abs(f - f_for_matrix)) > 5e-9:
            A_cache = _build_lp_matrix(f, n)
            f_for_matrix = f.copy()

        g = _lp_solve(f, M, n, time_limit=15)
        if g is None:
            stuck += 1
            if stuck > 5:
                break
            continue
        stuck = 0

        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_norm = g / g_sum

        best_t = 0.0
        best_M_new = M
        for t_try in [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.01]:
            f_new = (1.0 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            sn = float(f_new.sum())
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())
            if M_new < best_M_new:
                best_M_new = M_new
                best_t = t_try

        if best_t == 0.0:
            stuck += 1
            if stuck > 8:
                break
            continue

        f = (1.0 - best_t) * f + best_t * g_norm
        f = np.maximum(f, 0)
        f = f / f.sum()
        M = best_M_new

        if M < best_M:
            best_M = M
            best_f = f.copy()

        if step % print_interval == 0:
            print(f"      LP step {step+1}: M={M:.8e}, score={2.0*n*M:.8f}")

    return best_f


def _peak_peel(a, num_peels=5, lam=0.1):
    """
    Explicitly reduce peaks by redistributing mass.
    For each peak in the convolution, move mass from contributors to non-contributors.
    """
    n = len(a)
    a = a.copy()
    for _ in range(num_peels):
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        M = c[k_star]

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        top_idx = np.argsort(contrib)[-max(1, n // 10):]
        bottom_idx = np.argsort(contrib)[:max(1, n // 10)]

        mass_to_move = lam * M / max(1, len(top_idx))
        for i in top_idx:
            move = min(mass_to_move, a[i] * 0.3)
            if move > 1e-15:
                a[i] -= move
                redistribute = move / max(1, len(bottom_idx))
                for j in bottom_idx:
                    if abs(j - i) > n // 5:
                        a[j] += redistribute

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

    return a


def _multi_scale_optimize(a_init, n, deadline, lr=0.005, print_interval=20000):
    """
    Multi-scale optimization: start with coarse (high beta) and refine.
    """
    stages = [
        (0.2, 0.05, 30),
        (0.05, 0.01, 30),
        (0.01, 0.003, 30),
        (0.003, 0.001, 10),
    ]

    current = a_init.copy()
    time_per_stage = (deadline - time.monotonic()) / len(stages)

    for i, (b_start, b_end, extra_spin) in enumerate(stages):
        if time.monotonic() > deadline - 5:
            break
        stage_deadline = min(deadline - 5 * (len(stages) - i - 1),
                             time.monotonic() + time_per_stage)
        current, sc = _gradient_optimize(
            current, stage_deadline, lr=lr * (0.7 ** i),
            beta_start=b_start, beta_end=b_end,
            min_iters=2000, max_iters=200000,
            print_interval=print_interval
        )
        print(f"    Stage {i+1}/{len(stages)}: score={sc:.8f}")

    return current, sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: initialization, gradient-based refinement,
    LP improvements, peak peeling, multi-restart, and multi-length exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  ** NEW BEST: {best_score:.10f} (n={len(p)}) "
                  f"t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_main = len(parent) if parent is not None else 8192
    print(f"Parent: n={n_main}" if parent is not None else "No parent, using n=8192")
    if parent is not None:
        update_best(parent)
        print(f"  Parent score: {best_score:.10f}")

    # === Phase 1: Best initialization (0-80s) ===
    p1_end = min(deadline, t_start + 80.0)
    print(f"\n=== Phase 1: Initialization (n={n_main}) ===")

    initializations = []
    if parent is not None:
        initializations.append(("parent", parent.copy()))
    initializations.append(("uniform", _make_uniform(n_main)))

    # Quick gradient optimization from each init
    inits_time = (p1_end - time.monotonic()) / len(initializations)
    best_init = None
    best_init_score = np.inf

    for name, init in initializations:
        if inits_time < 10:
            break
        a_test, sc = _gradient_optimize(init, time.monotonic() + inits_time,
                                        lr=0.005, beta_start=0.1, beta_end=0.01,
                                        min_iters=1000, max_iters=100000,
                                        print_interval=1000000)
        update_best(a_test)
        if sc < best_init_score:
            best_init_score = sc
            best_init = a_test.copy()
        print(f"  Init '{name}': score={sc:.8f}")

    if best_init is None:
        best_init = parent if parent is not None else _make_uniform(n_main)

    # === Phase 2: Multi-scale gradient optimization (80s-700s) ===
    p2_end = min(deadline, t_start + 700.0)
    print(f"\n=== Phase 2: Multi-scale gradient optimization ===")

    if time.monotonic() < p2_end - 30:
        a_opt, sc_opt = _multi_scale_optimize(best_init, n_main, p2_end - 10,
                                              lr=0.005, print_interval=20000)
        update_best(a_opt)

    # === Phase 3: LP-based refinement (700s-1000s) ===
    p3_end = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 3: LP-based refinement ===")

    if best_seq is not None and time.monotonic() < p3_end - 30:
        f_lp = best_seq / best_seq.sum()
        f_lp = _lp_phase(f_lp, p3_end - 10, max_steps=20, print_interval=3)
        update_best(f_lp)

        # Gradient polish after LP
        if time.monotonic() < p3_end - 20:
            a_polish, _ = _gradient_optimize(f_lp, p3_end - 5, lr=0.003,
                                             beta_start=0.01, beta_end=0.002,
                                             min_iters=1000, max_iters=100000,
                                             print_interval=10000)
            update_best(a_polish)

    # === Phase 4: Peak peeling + refinement (1000s-1250s) ===
    p4_end = min(deadline, t_start + 1250.0)
    print(f"\n=== Phase 4: Peak peeling ===")

    if best_seq is not None and time.monotonic() < p4_end - 20:
        a_peeled = _peak_peel(best_seq / best_seq.sum(), num_peels=10, lam=0.05)
        update_best(a_peeled)

        if time.monotonic() < p4_end - 30:
            a_peeled2, _ = _gradient_optimize(a_peeled, p4_end - 10, lr=0.003,
                                              beta_start=0.005, beta_end=0.001,
                                              min_iters=2000, max_iters=200000,
                                              print_interval=20000)
            update_best(a_peeled2)

    # === Phase 5: Multi-restart (1250s-1500s) ===
    p5_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 5: Multi-restart ===")

    if best_seq is not None and time.monotonic() < p5_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 5
        per_restart = max(30, (p5_end - time.monotonic() - 20) / n_restarts)
        sigmas = [0.02, 0.04, 0.06, 0.1, 0.15]

        for i in range(n_restarts):
            if time.monotonic() > p5_end - 20:
                break
            p_start = _perturb(a_best_norm, sigmas[i], rng)
            dl_i = time.monotonic() + min(per_restart, p5_end - time.monotonic() - 10)
            a_r, sc_r = _gradient_optimize(p_start, dl_i, lr=0.004,
                                           beta_start=0.01, beta_end=0.002,
                                           min_iters=1000, max_iters=100000,
                                           print_interval=200000)
            update_best(a_r)

    # === Phase 6: Asymmetric exploration (1500s-1700s) ===
    p6_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 6: Asymmetric exploration ===")

    if best_seq is not None and time.monotonic() < p6_end - 30:
        # Try asymmetric profiles
        asym_profiles = [
            _make_asymmetric(n_main, shift_frac=0.1),
            _make_asymmetric(n_main, shift_frac=0.2),
            _make_asymmetric(n_main, shift_frac=-0.1),
        ]

        per_try = max(20, (p6_end - time.monotonic() - 20) / len(asym_profiles))
        for i, prof in enumerate(asym_profiles):
            if time.monotonic() > p6_end - 20:
                break
            # Blend with best
            for alpha in [0.3, 0.5]:
                blended = (1 - alpha) * a_best_norm + alpha * prof
                blended = np.maximum(blended, 0)
                blended = blended / blended.sum()
                dl_i = time.monotonic() + min(15, per_try)
                a_b, _ = _gradient_optimize(blended, dl_i, lr=0.004,
                                            beta_start=0.01, beta_end=0.002,
                                            min_iters=500, max_iters=50000,
                                            print_interval=200000)
                update_best(a_b)

    # === Phase 7: Different length exploration (1700s-1850s) ===
    p7_end = min(deadline, t_start + 1850.0)
    print(f"\n=== Phase 7: Multi-length exploration ===")

    if best_seq is not None and time.monotonic() < p7_end - 30:
        target_ns = [4096, 16384]
        for target_n in target_ns:
            if time.monotonic() > p7_end - 40:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            remaining = p7_end - time.monotonic() - 20
            if remaining > 60 and target_n <= 16384:
                a_r, _ = _gradient_optimize(p_interp, time.monotonic() + remaining * 0.8,
                                            lr=0.004, beta_start=0.005, beta_end=0.001,
                                            min_iters=1000, max_iters=100000,
                                            print_interval=50000)
                update_best(a_r)

    # === Phase 8: Final polish (remaining time) ===
    print(f"\n=== Phase 8: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_final_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            a_fin, _ = _gradient_optimize(a_final_norm, time.monotonic() + remaining,
                                          lr=0.002, beta_start=0.003, beta_end=0.0005,
                                          min_iters=1000, max_iters=200000,
                                          print_interval=50000)
            update_best(a_fin)

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"Final score: {sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.8f}")
    return [float(x) for x in fallback]
