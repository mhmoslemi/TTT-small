    return float(2.0 * n * c.max() / (s * s))

    """Solve one SCA subproblem using CVXPY."""
    n = len(a_curr)
    a = cp.Variable(n)
    t = cp.Variable()
    
    c_curr = _full_convolve(a_curr, a_curr)
    
    constraints = [a >= 0, cp.sum(a) == 1, t >= 0]
    
    for k in active_lags:
        # Linearization of (a*a)[k] around a_curr:
        # (a_curr*a_curr)[k] + sum_i (a_curr[k-i] + a_curr[i-k]) * (a[i] - a_curr[i])

        # coeff[i] = a_curr[k-i] + a_curr[i-k] (with a_curr[m]=0 for m outside [0,n-1])
        coeff = np.zeros(n)
        for i in range(n):

            ik = i - k
            if 0 <= ki < n:
                coeff[i] += a_curr[ki]
            if 0 <= ik < n:
                coeff[i] += a_curr[ik]

def _build_coeff(a_curr, k):
    n = len(a_curr)
    coeff = np.zeros(n)
    # First term: a_curr[k-i] for valid i
    # i ranges from max(0, k-(n-1)) to min(n-1, k)
    i_start = max(0, k - (n-1))
    i_end = min(n-1, k)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end+1)
        coeff[idx] += a_curr[k - idx]
    # Second term: a_curr[i-k] for valid i
    i_start2 = max(0, k)
    i_end2 = min(n-1, k + (n-1))
    if i_start2 <= i_end2:
        idx2 = np.arange(i_start2, i_end2+1)
        coeff[idx2] += a_curr[idx2 - k]
    return coeff

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses SCA (Successive Convex Approximation) with LP subproblems for fast local convergence,
FISTA with softmax smoothing, Adam in log-space, multi-restart diversification, and multi-n exploration.
Key improvements over previous: SCA-LP for exact local optimization, aggressive beta schedules,
n=8192 exploration, and structured frequency-domain restarts.
"""
import numpy as np
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except Exception:
    HAS_CVXPY = False


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _build_coeff_matrix(a_curr, active_lags, n):
    """Build the coefficient matrix for SCA linearization. Returns (|active|, n) matrix."""
    M = np.zeros((len(active_lags), n))
    for idx, k in enumerate(active_lags):
        # coeff[i] = a_curr[k-i] + a_curr[i-k] (with out-of-range = 0)
        # First term: a_curr[k-i] valid when 0 <= k-i < n, i.e. k-n+1 <= i <= k
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx_arr = np.arange(i_start, i_end + 1)
            M[idx, idx_arr] += a_curr[k - idx_arr]
        # Second term: a_curr[i-k] valid when 0 <= i-k < n, i.e. k <= i <= k+n-1
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + (n - 1))
        if i_start2 <= i_end2:
            idx_arr2 = np.arange(i_start2, i_end2 + 1)
            M[idx, idx_arr2] += a_curr[idx_arr2 - k]
    return M


def _scga_lp_solve(a_curr, active_frac=0.05, max_iters=50, deadline=None):
    """SCA with LP subproblems using CVXPY."""
    if not HAS_CVXPY:
        return a_curr, _score_norm(a_curr)

    n = len(a_curr)
    a_curr = a_curr.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    num_active = max(10, int(active_frac * (2 * n - 1)))

    for it in range(max_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        if score < best_score:
            best_score = score
            best_a = a_curr.copy()

        # Find active lags (top num_active peaks)
        top_idx = np.argsort(c)[::-1][:num_active]
        threshold = c[top_idx[-1]]

        # Additional constraint: center region
        center = n - 1
        if center not in top_idx:
            top_idx = np.append(top_idx, center)
            num_active += 1

        # Build coefficient matrix
        M = _build_coeff_matrix(a_curr, top_idx, n)
        c_active = c[top_idx]

        # Solve LP: min t s.t. M @ a <= c_active + t, sum(a)=1, a>=0, t>=0
        a_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            a_var >= 0,
            cp.sum(a_var) == 1,
            t_var >= 0,
            M @ a_var <= c_active + t_var
        ]
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)

        t_start_lp = time.monotonic()
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except Exception:
                break

        if time.monotonic() - t_start_lp > 10:
            break

        if a_var.value is None:
            break

        a_new = np.clip(a_var.value, 0, None)
        s = a_new.sum()
        if s < 1e-10 or not np.isfinite(s):
            break
        a_new = a_new / s

        # Check actual improvement (since LP solves linearized problem)
        c_new = _full_convolve(a_new, a_new)
        c_new_max = float(c_new.max())
        score_new = 2.0 * n * c_new_max

        if score_new < c_max * (1 + 1e-6):
            a_curr = a_new
        else:
            # Step back: blend
            blend = 0.5
            a_curr = blend * a_new + (1 - blend) * a_curr
            a_curr = np.clip(a_curr, 0, None)
            a_curr = a_curr / a_curr.sum()

        if it % 10 == 0:
            print(f"    SCA iter {it}: score={_score_norm(a_curr):.10f}, t={t_var.value:.10f}")

    return best_a, best_score


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while time.monotonic() < deadline:
            t += 1
            if t > max_iter_total:
                break

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 5000 == 0:
                current_lr = max(0.001, current_lr * 0.97)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _freq_perturb(a, rng, n_modes=3):
    """Perturb in frequency domain to target specific convolution lags."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for _ in range(n_modes):
        # Random frequency
        freq = rng.integers(1, max(2, n // 2))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        t = np.arange(n)
        z += amp * np.cos(2 * np.pi * freq * t / n + phase)
    
    z = z - z.mean()
    return _softmax_compute(z)
