def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)

    n = len(a_init)

    c_curr = _full_convolve(a_curr, a_curr)
    obj_curr = float(c_curr.max())
    best_a = a_curr.copy()
    best_obj = obj_curr
    step0 = beta / (4.0 * n)  # 1/L where L = 4n/beta
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        
        # Softmax weights
        shifted = (c_y - c_max) / beta
        e = np.exp(shifted)
        w_sum = float(e.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = e / w_sum
        
        grad = _compute_grad(y, w)
        
        # Adaptive step: try the nominal step, backtrack if needed
        step = step0
        a_next_test = _proj_simplex(y - step * grad)
        
        # Simple backtracking (1 step)
        c_test = _full_convolve(a_next_test, a_next_test)
        if c_test.max() > obj_curr + 0.1:
            step *= 0.5
            a_next_test = _proj_simplex(y - step * grad)

        obj_next = float(c_next.max())
        if obj_next < best_obj:
            best_obj = obj_next
            best_a = a_next.copy()

            a_prev = a_curr.copy()
            t_val = 1.0
        
        a_prev = a_curr.copy()
        a_curr = a_next

        obj_curr = obj_next

                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: score={sc_full:.10f} (no improvement) t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: n=8192 exploration (350s) ===
    print("Phase 4: n=8192 exploration")
    phase4_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        n_8192 = 8192
        # Use FFT-based upsampling for better quality
        a_8192 = _upsample_fft(best_seq, n_8192)
        sc_8192 = _score(a_8192)
        print(f"  Upsampled n=8192: score={sc_8192:.10f}")

        if sc_8192 < best_score:
            best_score = sc_8192
            best_seq = a_8192.copy()

        # Optimize the 8192 version
        if time.monotonic() < phase4_end - 20:
            c_8192 = _full_convolve(a_8192, a_8192)
            max_c8 = float(c_8192.max())

            # Quick FISTA
            beta8 = 0.003 * max_c8
            per_fista = min(40.0, phase4_end - time.monotonic() - 30)
            if per_fista > 10:
                a_8192, sc8 = _fista_restart(a_8192, beta8, time.monotonic() + per_fista, 20000)
                a_8192 = a_8192 / a_8192.sum()
                sc8_full = _score(a_8192)
                if sc8_full < best_score:
                    best_score = sc8_full
                    best_seq = a_8192.copy()
                    print(f"  FISTA n=8192: score={sc8_full:.10f}")

            # Adam fine
            c_8192 = _full_convolve(a_8192, a_8192)
            max_c8 = float(c_8192.max())
            betas_8192 = [0.001 * max_c8, 0.0005 * max_c8, 0.0002 * max_c8,
                          0.0001 * max_c8, 0.00005 * max_c8]
            per_t = phase4_end - time.monotonic() - 10
            if per_t > 15:
                a_8192_opt, sc_8192_opt = _adam_logspace(
                    a_8192, time.monotonic() + per_t,
                    betas_8192, lr_init=0.01, max_iter_total=200000
                )
                sc_8192_full = _score(a_8192_opt)
                if sc_8192_full < best_score:
                    best_score = sc_8192_full
                    best_seq = a_8192_opt.copy()
                    print(f"  Adam n=8192: score={sc_8192_full:.10f}")

    # === Phase 5: Final ultra-fine polish (remaining time) ===
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        n_pol = len(a_pol)

        # Use very small betas for ultra-fine
        betas_ult = [0.00003 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 10
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.004, max_iter_total=300000
            )
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"All phases done: best_score={best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(512, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
