"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: tilted-binomial init family, full-constraint LP bisection,
iterated peak shaving with exact line search, Nesterov Adam with cyclical restarts.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm_fn(a) / (s * s)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _binomial_seq(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    return b


def _tilted_binomial_init(n, tilt=0.0):
    b = _binomial_seq(n)
    j = np.arange(n, dtype=np.float64)
    a = b[::-1].copy()
    if tilt != 0.0:
        a = a * np.exp(tilt * j / max(n - 1, 1))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_init(n, etype='exp', param=1.0):
    t = np.linspace(0, 1, n)
    if etype == 'exp':
        a = np.exp(param * (t - 0.5))
    elif etype == 'pow':
        a = np.maximum(0.01 + 0.99 * t, 0.01) ** param
    elif etype == 'sigmoid':
        a = 1.0 / (1.0 + np.exp(-param * (t - 0.5)))
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'step':
        a = np.where(t < 0.5, 0.1, 1.0)
    elif etype == 'ramp':
        a = 0.05 + 0.95 * t
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxls': 200, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _build_lp_matrix_full(a, n, max_constraints=500):
    """Build constraint matrix for top max_constraints peaks."""
    c = _full_convolve(a, a)
    M = float(c.max())
    # Get indices sorted by value
    sorted_idx = np.argsort(c)[::-1]
    # Include all peaks within 90% of max
    threshold = 0.90 * M
    active = sorted_idx[c[sorted_idx] >= threshold]
    if len(active) > max_constraints:
        active = sorted_idx[:max_constraints]
    active = active[:max_constraints]
    A_mat = np.zeros((len(active), n))
    for row, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat, active, M


def _lp_phase(a_init, deadline, K=200, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        A_mat, active, M_val = _build_lp_matrix_full(best_a, n, K)
        if len(active) == 0:
            break
        b_vec = np.full(len(active), M_val * 1.001)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _peak_shaving(a_init, deadline, m_peaks=5, max_iters=5000):
    """Iterated peak shaving: steepest descent on max with line search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Get active peaks (top m or within 95%)
        sorted_idx = np.argsort(c)[::-1]
        threshold = 0.95 * cm
        active = sorted_idx[c[sorted_idx] >= threshold]
        if len(active) > m_peaks:
            active = sorted_idx[:m_peaks]

        # Compute gradient of max at active peaks (equal weight)
        grad_a = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                grad_a[i_lo:i_hi + 1] += 2.0 * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        grad_a /= len(active)

        # Convert to z-gradient: grad_z = a * (grad_a - dot(a, grad_a)) * 2n
        Wa = float(np.dot(a, grad_a))
        grad_z = 2.0 * n * a * (grad_a - Wa)

        if not np.all(np.isfinite(grad_z)):
            break

        # Line search: try several step sizes
        best_t = 0.0
        best_sc = best_score
        for t_val in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            z_trial = z - t_val * grad_z
            z_trial = np.clip(z_trial, -100, 100)
            a_trial = _softmax_fn(z_trial)
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = t_val

        if best_t > 0:
            z = z - best_t * grad_z
            z = z - z.mean()
            if best_sc < best_score - 1e-10:
                best_score = best_sc
                best_a = _softmax_fn(z).copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            # Restart with noise
            z = z + 0.05 * np.random.randn(n)
            z = z - z.mean()
            no_improve = 0
            if it > max_iters * 0.8:
                break

    return best_a, best_score


def _nesterov_adam(a_init, deadline, lr=0.03, max_iter=500000):
    """Nesterov-accelerated Adam with cyclical LR and-beta schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30
    cycle_len = 10000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        # Cyclical learning rate
        phase = (t % cycle_len) / cycle_len
        lr_cur = lr * (1.0 + 2.0 * (1.0 - phase))

        # Adaptive beta
        beta = cm0 * 0.01 * (0.96 ** (t // 30))
        beta = max(beta, 1e-14 * cm0)

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        # Nesterov: look-ahead
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Nesterov step: use momentum look-ahead for gradient computation
        z = z - lr_cur * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.1 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0
            if no_improve > 200:
                break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _tilted_binomial_init(n, 1.0)
        update_best(parent, _score_norm_fn(parent), "default_init")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Multi-init exploration (300s) ===
    print("Phase 1: Multi-init exploration")
    p1_end = min(deadline, t_start + 300.0)

    if time.monotonic() < p1_end:
        a_multi, sc_multi = _try_multiple_inits(n, p1_end, n_inits=12)
        update_best(a_multi, sc_multi, "multi_init")

    # Also try more aggressively from parent with different tilts
    if time.monotonic() < p1_end - 30:
        for tilt in [0.5, 1.5, 2.5]:
            a_t = parent * np.exp(tilt * np.arange(n) / max(n - 1, 1))
            a_t = np.maximum(a_t, 1e-15)
            a_t = a_t / a_t.sum()
            sc_t = _score_norm_fn(a_t)
            if sc_t < best_score * 1.02:
                t_end = min(p1_end - 10, time.monotonic() + 40)
                a_to, sc_to = _run_lbfgs(a_t, t_end, maxiter_per_beta=3000)
                update_best(a_to, sc_to, f"tilt_{tilt}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Main L-BFGS continuation (400s) ===
    print("Phase 2: Main L-BFGS continuation")
    p2_end = min(deadline, t_start + 700.0)

    a_b1, sc1 = _run_lbfgs(best_seq, p2_end, maxiter_per_beta=8000)
    update_best(a_b1, sc1, "lbfgs_main")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP peak balancing (300s) ===
    print("Phase 3: LP peak balancing")
    p3_end = min(deadline, t_start + 1000.0)

    if time.monotonic() < p3_end:
        a_lp, sc_lp = _lp_phase(best_seq, p3_end, K=200, max_rounds=25)
        update_best(a_lp, sc_lp, "lp_full")

        # Second LP with different K
        if time.monotonic() < p3_end - 60 and best_seq is not None:
            a_lp2, sc_lp2 = _lp_phase(best_seq, p3_end, K=100, max_rounds=15)
            update_best(a_lp2, sc_lp2, "lp_K100")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Peak shaving (200s) ===
    print("Phase 4: Peak shaving")
    p4_end = min(deadline, t_start + 1200.0)

    if time.monotonic() < p4_end and best_seq is not None:
        for m_p in [3, 5, 8]:
            if time.monotonic() > p4_end - 30:
                break
            a_ps, sc_ps = _peak_shaving(best_seq, min(p4_end, time.monotonic() + 60), m_peaks=m_p)
            update_best(a_ps, sc_ps, f"peak_shave_m{m_p}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Nesterov Adam with schedule (250s) ===
    print("Phase 5: Nesterov Adam")
    p5_end = min(deadline, t_start + 1450.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_ad, sc_ad = _nesterov_adam(best_seq, min(p5_end, time.monotonic() + 120), lr=0.02)
        update_best(a_ad, sc_ad, "nadam_lr02")

        if time.monotonic() < p5_end - 30:
            a_ad2, sc_ad2 = _nesterov_adam(best_seq, p5_end, lr=0.008)
            update_best(a_ad2, sc_ad2, "nadam_slow")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final tight L-BFGS (150s) ===
    print("Phase 6: Final tight L-BFGS")
    p6_end = min(deadline, t_start + 1600.0)

    if time.monotonic() < p6_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.0003, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]]
        a_f, sc_f = _run_lbfgs(best_seq, p6_end, beta_list=tiny_betas, maxiter_per_beta=10000)
        update_best(a_f, sc_f, "final_tiny_beta")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Interpolate to n*2 and optimize ===
    if time.monotonic() < deadline - 80 and best_seq is not None:
        n_new = min(2 * len(best_seq), 8192)
        if n_new > len(best_seq):
            print(f"Phase 7: Interpolating to n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm_fn(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")
            if time.monotonic() < deadline - 20:
                a_in_b, sc_in_b = _run_lbfgs(a_interp, deadline - 10, maxiter_per_beta=5000)
                update_best(a_in_b, sc_in_b, f"lbfgs_n{n_new}")

    # === Final: One more peak shaving sweep ===
    if time.monotonic() < deadline - 15 and best_seq is not None:
        a_final_ps, sc_final_ps = _peak_shaving(best_seq, deadline - 5, m_peaks=3)
        update_best(a_final_ps, sc_final_ps, "final_peak_shave")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _tilted_binomial_init(256, 1.0)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _tilted_binomial_init(512, 1.0)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _tilted_binomial_init(256, 1.0)
    print(f"FALLBACK: n=256, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
