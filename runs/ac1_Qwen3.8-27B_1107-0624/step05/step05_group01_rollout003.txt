import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_actual(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype='binom'):
    t = np.linspace(0, 1, n)
    if etype == 'binom':
        return _greedy_init(n)
    elif etype == 'gaussian':
        a = np.exp(-3.0 * (2.0 * t - 1.0) ** 2)

    else:

    a = np.maximum(a, 1e-12)
    return a / a.sum()

    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)

    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=3000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _build_lp_matrix(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat


def _lp_phase(a_init, deadline, K=50, max_rounds=20, tightness=0.995):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K].copy()
        threshold = 0.9 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = sorted_idx[:300]

        A_mat = _build_lp_matrix(best_a, active, n)
        b_vec = np.full(len(active), M * tightness)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                # Try with slightly looser constraint
                b_vec = np.full(len(active), M * 1.0)
                res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                              bounds=[(0, None)] * n, method='highs',
                              options={'time_limit': 20})
                if res.status != 0:
                    break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-14:
                best_a = a_trial.copy()
                best_score = _score_norm(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            # Try tighter constraint next round
            tightness = max(0.95, tightness * 0.99)

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.02, max_iter=500000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.96 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.1 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _active_set_phase(a_init, deadline, K=5, max_outer=50):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for outer in range(max_outer):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]
        pk = peak_indices.copy()
        nn = n

        def ao(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[pk]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            bb = 1e-8 * pm
            sh = (peaks - pm) / bb
            lse = pm + bb * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * nn * lse)

        def ag(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[pk]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            bb = 1e-8 * pm
            sh = (peaks - pm) / bb
            e = np.exp(sh - sh.max())
            w = e / e.sum()

            grad_a = np.zeros(nn)
            for idx_k in range(len(pk)):
                k = pk[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - nn + 1)
                i_hi = min(nn - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * nn * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            result = minimize(ao, z, method='L-BFGS-B', jac=ag,
                              options={'maxiter': 2000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 50, 'maxfun': 8000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score - 1e-14:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _palindromic_fold(a):
    n = len(a)
    half = (n + 1) // 2
    a_half = a[:half].copy()
    if n % 2 == 0:
        a_folded = np.concatenate([a_half, a_half[::-1]])
    else:
        a_folded = np.concatenate([a_half, a_half[-2::-1]])
    return a_folded / a_folded.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_main = 8192
    if parent is not None:
        n_main = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_main}, norm_score={sc_parent:.10f}, actual={_score_actual(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _greedy_init(n_main)
        update_best(parent, _score_norm(parent), "greedy_default")

    print(f"Starting n={n_main}, initial best={best_score:.10f}")

    # === Phase 1: Quick L-BFGS on parent (200s) ===
    print("Phase 1: Quick L-BFGS on parent")
    p1_end = min(deadline, t_start + 200.0)
    a1, sc1 = _run_lbfgs(parent, p1_end, maxiter_per_beta=2000)
    update_best(a1, sc1, "lbfgs_quick")

    # === Phase 2: Aggressive LP peak balancing (300s) ===
    print("Phase 2: LP peak balancing (tight)")
    p2_end = min(deadline, t_start + 500.0)
    if best_seq is not None:
        for tight in [0.99, 0.98, 0.97]:
            if time.monotonic() > p2_end - 20:
                break
            a_lp, sc_lp = _lp_phase(best_seq, min(p2_end, time.monotonic() + 100),
                                     K=50, max_rounds=8, tightness=tight)
            update_best(a_lp, sc_lp, f"lp_tight{tight}")

    # === Phase 3: Deep L-BFGS continuation (250s) ===
    print("Phase 3: Deep L-BFGS continuation")
    p3_end = min(deadline, t_start + 750.0)
    if best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.001, 0.0005, 0.0003, 0.0002,
                                         0.0001, 0.00005, 0.00002]]
        a3, sc3 = _run_lbfgs(best_seq, p3_end, beta_list=tiny_betas, maxiter_per_beta=5000)
        update_best(a3, sc3, "lbfgs_deep")

    # === Phase 4: Active set refinement (150s) ===
    print("Phase 4: Active set refinement")
    p4_end = min(deadline, t_start + 900.0)
    if best_seq is not None:
        for K in [4, 6, 8]:
            if time.monotonic() > p4_end - 20:
                break
            a_act, sc_act = _active_set_phase(best_seq, min(p4_end, time.monotonic() + 50), K=K)
            update_best(a_act, sc_act, f"active_K{K}")

    # === Phase 5: Try n=4096 (200s) ===
    print("Phase 5: Try n=4096")
    p5_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < p5_end:
        n4k = 4096
        for et in ['binom', 'gaussian', 'cubic_trap']:
            if time.monotonic() > p5_end - 30:
                break
            a_p = _parametric_init(n4k, et)
            sc_p = _score_norm(a_p)
            a_pb, sc_pb = _run_lbfgs(a_p, min(p5_end, time.monotonic() + 60), maxiter_per_beta=2000)
            update_best(a_pb, sc_pb, f"n4k_{et}")
            # Quick LP
            if time.monotonic() < p5_end - 20:
                a_lp4, sc_lp4 = _lp_phase(a_pb, min(p5_end, time.monotonic() + 30),
                                           K=30, max_rounds=5, tightness=0.98)
                update_best(a_lp4, sc_lp4, f"n4k_{et}_lp")

    # === Phase 6: Try n=16384 via interpolation (150s) ===
    print("Phase 6: Try n=16384")
    p6_end = min(deadline, t_start + 1250.0)
    if best_seq is not None and time.monotonic() < p6_end:
        n16k = 16384
        a_interp = _interp_profile(best_seq, n16k)
        sc_in = _score_norm(a_interp)
        update_best(a_interp, sc_in, "interp_16k")
        if time.monotonic() < p6_end - 30:
            a_i6, sc_i6 = _run_lbfgs(a_interp, p6_end, maxiter_per_beta=1000)
            update_best(a_i6, sc_i6, "lbfgs_16k")

    # === Phase 7: Adam polish on best (250s) ===
    print("Phase 7: Adam polish")
    p7_end = min(deadline, t_start + 1500.0)
    if best_seq is not None:
        a_ad, sc_ad = _adam_polish(best_seq, min(p7_end, time.monotonic() + 120), lr=0.015)
        update_best(a_ad, sc_ad, "adam_main")
        if time.monotonic() < p7_end - 30:
            a_ad2, sc_ad2 = _adam_polish(best_seq, p7_end, lr=0.005)
            update_best(a_ad2, sc_ad2, "adam_slow")

    # === Phase 8: Multi-start with perturbations (150s) ===
    print("Phase 8: Multi-start perturbations")
    p8_end = min(deadline, t_start + 1650.0)
    if best_seq is not None and time.monotonic() < p8_end:
        n_ps = len(best_seq)
        for trial in range(8):
            if time.monotonic() > p8_end - 10:
                break
            perturb_scale = [0.05, 0.1, 0.03, 0.08, 0.06, 0.12, 0.04, 0.07][trial]
            a_pert = best_seq + perturb_scale * rng.standard_normal(n_ps)
            a_pert = np.maximum(a_pert, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-15:
                continue
            a_pert = a_pert / s_pert
            sc_pert = _score_norm(a_pert)
            if sc_pert < best_score * 1.05:
                # Quick L-BFGS polish
                a_pp, sc_pp = _run_lbfgs(a_pert, min(p8_end, time.monotonic() + 15),
                                          maxiter_per_beta=500)
                update_best(a_pp, sc_pp, f"perturb_{trial}")
            # Also try palindromic version
            a_pal = _palindromic_fold(a_pert)
            sc_pal = _score_norm(a_pal)
            if sc_pal < best_score:
                update_best(a_pal, sc_pal, f"palindromic_{trial}")

    # === Phase 9: Final LP + L-BFGS on best (100s) ===
    print("Phase 9: Final refinement")
    p9_end = deadline - 20
    if best_seq is not None and time.monotonic() < p9_end:
        a_f1, sc_f1 = _lp_phase(best_seq, min(p9_end, time.monotonic() + 40),
                                 K=40, max_rounds=5, tightness=0.97)
        update_best(a_f1, sc_f1, "final_lp")
        if time.monotonic() < p9_end:
            c_b2 = _full_convolve(best_seq, best_seq)
            m_b2 = float(c_b2.max())
            final_betas = [m_b2 * b for b in [0.00001, 0.000005, 0.000002]]
            a_f2, sc_f2 = _run_lbfgs(best_seq, p9_end, beta_list=final_betas, maxiter_per_beta=3000)
            update_best(a_f2, sc_f2, "final_lbfgs")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_actual(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_actual(fb):.10f}")
    return [float(x) for x in fb]
