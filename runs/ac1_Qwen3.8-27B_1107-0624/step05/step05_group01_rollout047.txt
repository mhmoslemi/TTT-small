def _next_pow2(n):

Given:
- `a` of length n, parameterized as `a = softmax(z)`
- `c = a * a` (linear convolution, length 2n-1)
- `f = 2n * LSE(c, beta)` where `LSE(c, beta) = cm + beta * log(sum(exp((c-cm)/beta)))`

Step 1: `df/dc[k] = 2n * w[k]` where `w[k] = exp((c[k]-cm)/beta) / sum(exp((c'[k]-cm)/beta))`

Step 2: `dc[k]/da[i] = 2*a[k-i]` if `0 <= k-i < n`, else 0.

Step 3: `df/da[i] = sum_k df/dc[k] * dc[k]/da[i] = 2n * sum_k w[k] * 2*a[k-i] = 4n * sum_k w[k] * a[k-i]`

Now, `sum_k w[k] * a[k-i]` is the cross-correlation of w and a at lag i.

This is the convolution of w (length 2n-1) with reversed a (length n), evaluated at position i + (n-1):
`conv(w, a_rev)[i + n - 1] = sum_{j=0}^{n-1} w[i + n - 1 - j] * a_rev[j] = sum_{j=0}^{n-1} w[i + n - 1 - j] * a[n-1-j]`

Let k = n-1-j, so j = n-1-k:
`= sum_{k=0}^{n-1} w[i+k] * a[k]`

Yes! So `corr[i] = conv(w, a_rev)[i + n - 1]`.

Therefore: `df/da[i] = 4n * corr[i] = 4n * conv(w, a_rev)[i + n - 1]`

Step 4: Chain rule through softmax:
`da[i]/dz[j] = a[i] * (1[i==j] - a[j])`

`df/dz[j] = sum_i df/da[i] * da[i]/dz[j]`
`= sum_i g[i] * a[i] * (1[i==j] - a[j])` where `g[i] = df/da[i]`

`= a[j] * (g[j] - sum_i g[i]*a[i])`

So: `df/dz[j] = a[j] * (g[j] - a·g)` where `a·g = sum_i a[i]*g[i]`.

- `corr = conv_result[n-1:2*n-1]` (length n)
- `g = 4*n*corr`

This is cleaner. Let me use this.

Now let me think about the peak-directed gradient phase more carefully.

For the peak-directed phase, I want to directly minimize `max(c)` (not a smooth surrogate). I'll use:
1. Compute c = a*a
2. Find peaks (top K or those above threshold)
3. Compute the subgradient: `sg[i] = 2 * sum_{k in peaks} w_k * a[k-i]` where w_k are weights (uniform for hard subgradient, or proportional for soft)
4. Project to zero-sum: `sg = sg - sg.mean()`
5. Take a step: `a_new = a - lr * sg`, clip, renormalize
6. Use backtracking line search to ensure the max doesn't increase

For the weights, I'll use a "focus" approach:
- Weight = 1 for peaks within epsilon of the max
- Or weight = (c[k]/c_max)^alpha for peaks above some threshold

The key innovation: **adaptive step size with backtracking**. After computing the direction, try a large step and reduce until we actually decrease the max.

Let me also add **momentum** (Nesterov or classical) to accelerate convergence.

One more idea: **peak migration handling**. After a step, the peak might have moved. I should re-identify peaks at each iteration.

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _peak_shave(a_init, deadline, lr=0.005, momentum=0.9, focus_alpha=5.0):
    """Momentum-based peak shaving with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = _score_norm(a)
    no_improve = 0
    max_no_improve = 200

    it = 0
    while time.monotonic() < deadline:
        c = _full_convolve(a, a)
        cm = float(c.max())
        score = 2.0 * n * cm

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            # Random perturbation
            a = a + 0.01 * np.random.randn(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0
            continue

        # Focus on peaks close to max
        threshold = cm / (1.0 + 0.1 * (1.0 + no_improve / 50.0))
        w = np.zeros(2 * n - 1)
        peak_idx = np.where(c >= threshold)[0]
        if len(peak_idx) == 0:
            w[int(c.argmax())] = 1.0
        else:
            # Soft weights based on proximity to max
            w[peak_idx] = ((c[peak_idx] / cm) ** focus_alpha)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(2 * n - 1)
                w[int(c.argmax())] = 1.0
            else:
                w /= w_sum

        # Compute gradient
        corr = _corr_grad(a, w, n)
        g = 4.0 * n * corr
        g = g - g.mean()  # project to zero-sum

        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            a = np.ones(n) / n
            v = np.zeros(n)
            continue
        a_new = a_new / a_sum

        # Check improvement; if not, backtrack
        c_new = _full_convolve(a_new, a_new)
        cm_new = float(c_new.max())
        if cm_new > cm * 1.001:
            # Backtrack
            v *= 0.5
            a_new = a + v
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()

        a = a_new
        it += 1

        if it % 500 == 0:
            print(f"    peak_shave iter={it}, score={score:.10f}, best={best_score:.10f}")

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:

    x0 = np.linspace(0.0, 1.0, n0)

    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        update_best(parent, _score_norm(parent), "greedy")

    print(f"Starting: n={n}, best={best_score:.10f}")

    # Phase 1: L-BFGS (300s)
    print("Phase 1: L-BFGS")
    p1_end = min(deadline, t_start + 300.0)
    a1, sc1 = _lbfgs_phase(best_seq, p1_end, maxiter=3000)
    update_best(a1, sc1, "lbfgs")

    # Also try from greedy
    if time.monotonic() < p1_end - 60:
        a_g = _greedy_init(n)
        sc_g = _score_norm(a_g)
        if sc_g < best_score * 1.05:
            a_gb, sc_gb = _lbfgs_phase(a_g, min(p1_end - 10, time.monotonic() + 80), maxiter=2000)
            update_best(a_gb, sc_gb, "lbfgs_greedy")

    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: Peak shaving with momentum (500s)
    print("Phase 2: Peak shaving")
    p2_end = min(deadline, t_start + 800.0)
    a2, sc2 = _peak_shave(best_seq, p2_end, lr=0.003, momentum=0.9, focus_alpha=5.0)
    update_best(a2, sc2, "peak_shave_1")

    if time.monotonic() < p2_end - 100:
        a2b, sc2b = _peak_shave(best_seq, p2_end, lr=0.001, momentum=0.95, focus_alpha=10.0)
        update_best(a2b, sc2b, "peak_shave_2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # Phase 3: Subgradient with backtracking (300s)
    print("Phase 3: Subgradient")
    p3_end = min(deadline, t_start + 1100.0)
    for K in [5, 10, 20]:
        if time.monotonic() > p3_end - 30:
            break
        a3, sc3 = _subgradient_phase(best_seq, min(p3_end, time.monotonic() + 100), K=K)
        update_best(a3, sc3, f"subgrad_K{K}")

    print(f"Phase 3 done: best={best_score:.10f}")

    # Phase 4: Fine L-BFGS with tiny beta (200s)
    print("Phase 4: Fine L-BFGS")
    p4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < p4_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        for beta in [0.0001 * m_b, 0.00005 * m_b, 0.00001 * m_b]:
            if time.monotonic() > p4_end:
                break
            # Inline L-BFGS with this beta
            def obj4(z_in):
                a_in = _softmax_fn(z_in)
                c_in = _full_convolve(a_in, a_in)
                cm = float(c_in.max())
                if cm < 1e-300:
                    return 0.0
                shifted = (c_in - cm) / beta
                max_s = float(shifted.max())
                lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
                return float(2.0 * n * lse)

            def grad4(z_in):
                a_in = _softmax_fn(z_in)
                c_in = _full_convolve(a_in, a_in)
                cm = float(c_in.max())
                if cm < 1e-300:
                    return np.zeros_like(z_in)
                shifted = (c_in - cm) / beta
                max_s = float(shifted.max())
                e = np.exp(shifted - max_s)
                w = e / e.sum()
                corr = _corr_grad(a_in, w, n)
                g = 4.0 * n * corr
                ag = float(np.dot(a_in, g))
                gr = a_in * (g - ag)
                if not np.all(np.isfinite(gr)):
                    return np.zeros_like(z_in)
                return gr

            try:
                res = minimize(obj4, z, method='L-BFGS-B', jac=grad4,
                               options={'maxiter': 5000, 'ftol': 1e-18,
                                        'gtol': 1e-14, 'maxls': 50})
                if res.x is not None and np.all(np.isfinite(res.x)):
                    a_new = _softmax_fn(res.x)
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_seq = a_new.copy()
                        z = res.x.copy()
                        print(f"  Fine L-BFGS beta={beta:.2e}: score={s_new:.10f}")
            except Exception:
                pass

    print(f"Phase 4 done: best={best_score:.10f}")

    # Phase 5: Final peak shaving (remaining time)
    print("Phase 5: Final peak shaving")
    if time.monotonic() < deadline - 20:
        a5, sc5 = _peak_shave(best_seq, deadline - 10, lr=0.0005, momentum=0.98, focus_alpha=20.0)
        update_best(a5, sc5, "final_shave")

    print(f"Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator confirmed: {sc_eval:.10f}")
            else:
                print("Evaluator returned inf, using fallback")
                fb = _greedy_init(512)
                return [float(x) for x in fb]
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    fb = _greedy_init(256)
    return [float(x) for x in fb]
