from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    vals = a_old[rows - cols]  # This gives a_old[k-j] for each (k,j) pair

            if a_new.sum() > 1e-15:
                a_new /= a_new.sum()
            return a_new, res.x[-1]
        return None, None

Actually, the cleanest way to compute the gradient of `sum_k w_k * (a*a)_k` with respect to `a` is:

The "autocorrelation matrix" (or "Gram matrix") G is defined by:
`G[i,j] = sum_k w_k * [k-i == j or k-j == i]` ... this is getting too complicated.

Let me just use the fact that for the objective `f(a) = sum_k w_k * (a*a)_k`, the gradient is:
`grad_i = sum_k w_k * d(a*a)_k/da_i`

And `d(a*a)_k/da_i` is:
- `a[k-i]` if `0 <= k-i <= n-1` (from the j=i term)
- `+ a[k-i]` if `0 <= k-i <= n-1` and `k-i != i` (from the j=k-i term)

So `d(a*a)_k/da_i = a[k-i] * (1 + [0 <= k-i <= n-1 and k-i != i])`

For the "standard" case where `0 <= k-i <= n-1`:
- If `k != 2i`: `d(a*a)_k/da_i = 2*a[k-i]`
- If `k == 2i`: `d(a*a)_k/da_i = a[i]`

So `grad_i = 2*sum_{k: 0<=k-i<=n-1, k!=2i} w_k * a[k-i] + sum_{k: k=2i, 0<=i<=n-1} w_{2i} * a[i]`

The second sum is just `w[2i]*a[i]` if `2i < 2n-1` (which is always true for i < n).

For the first sum: `2*sum_{j: 0<=j<=n-1, j!=i} w_{i+j} * a[j]` (substituting j=k-i)
= `2*(sum_j w_{i+j}*a[j] - w[2i]*a[i])`
= `2*R[i] - 2*w[2i]*a[i]`

where `R[i] = sum_j w[i+j]*a[j]` (the cross-correlation).

So `grad_i = 2*R[i] - 2*w[2i]*a[i] + w[2i]*a[i] = 2*R[i] - w[2i]*a[i]`

For most practical purposes, the correction term `w[2i]*a[i]` is small (since w is spread over 2n-1 values), so `grad ≈ 2*R`.

Let me just use `grad = 2*R` for simplicity (the error is O(1/n) per component).

Now let me also think about the "projected gradient" step. We're optimizing over the simplex `{a >= 0, sum(a) = 1}`. The natural parameterization is `a = softmax(z)` and we optimize over z. The gradient in z-space is:
`dz = a * (grad - dot(a, grad))`

This is the standard mirror descent gradient for the entropy-regularized problem.

"""
Direct LP minimization + Adam log-sum-exp optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses direct sequential LP (minimize new M
given current a), smoothed Adam in log-space, and multi-phase strategy
with perturbations for escaping local minima.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_matrix(n, a):
    """Build sparse matrix A where A[k,j] = a[k-j] for valid indices."""
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(js), k, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        vals_list.append(a[ks] if False else a[k - js])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return rows, cols, vals


def _solve_direct_lp(a_old, time_limit=60):
    """
    Solve: minimize M
    s.t. (a_new * a_old)[k] <= M for all k=0..2n-2
         sum(a_new) = 1
         a_new >= 0
    Returns (a_new, M_opt) or (None, None).
    """
    n = len(a_old)
    num_rows = 2 * n - 1

    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(js), k, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        vals_list.append(a_old[k - js])
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)

    A = sp.csc_matrix((vals, (rows, cols)), shape=(num_rows, n))

    # Variables: [a_new (n), M (1)]
    c = np.zeros(n + 1)
    c[n] = 1.0

    # A_ub: [A, -I] -> (A @ a_new) - M <= 0
    neg_I = sp.diags(-np.ones(num_rows), 0, format='csc')
    A_ub = sp.hstack([A, neg_I], format='csc')
    b_ub = np.zeros(num_rows)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            a_new = np.maximum(res.x[:n], 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
            return a_new, float(res.x[n])
    except Exception:
        pass
    return None, None


def _compute_grad(a, w):
    """Compute gradient of sum_k w_k * (a*a)_k w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    Returns grad of shape (n,)."""
    n = len(a)
    N = 2 * n
    a_f = np.fft.rfft(a, N)
    w_f = np.fft.rfft(w, N)
    # R[i] = sum_j w[j+i] * a[j] (cross-correlation)
    R = np.fft.irfft(np.conj(a_f) * w_f, N)[:n]
    # Full gradient: grad_i = 2*R[i] - w[2i]*a[i] (correction for k=2i)
    # For simplicity and speed, use grad = 2*R (error is O(1/n) per component)
    # But include correction for better accuracy:
    grad = 2.0 * R
    # Correction: subtract w[2i]*a[i] for i where 2i < 2n-1
    indices = np.arange(n)
    valid = 2 * indices < 2 * n - 1
    grad[valid] -= w[2 * indices[valid]] * a[indices[valid]]
    return grad


def _adam_smooth(a_init, deadline, beta_start, beta_end, lr=0.01, max_iter=500000, log_interval=1000):
    """Adam optimizer with log-sum-exp smoothing in log-space (softmax parameterization)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z -= z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())

        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_e

        grad = _compute_grad(a, w)
        # Project onto tangent space of simplex
        grad -= np.dot(a, grad) * a

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update *= 0.5 / update_norm

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z -= z.mean()

        if t % log_interval == 0:
            a_new = _softmax(z)
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            print(f"    Adam[{t}]: score={sc:.10f}, beta={beta:.4e}, dt={time.monotonic()-t0:.0f}s")

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z -= z.mean()
    return _softmax(z)
