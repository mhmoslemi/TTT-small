```python
from scipy.optimize import linprog

def solve_lp(a, M, n):
    # Maximize sum(b) subject to conv(a,b) <= M, b >= 0
    # c[k] = sum_j a[j] * b[k-j] <= M for k = 0, ..., 2n-2
    
    # Build constraint matrix A where A[k,j] = a[j] if 0 <= k-j < n
    # A has shape (2n-1, n)
    A = np.zeros((2n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - (n-1))
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[j]
    
    # minimize -sum(b) subject to A@b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n)
    
    if res.success:
        return res.x
    return None
```

Hankel QP + LP hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set QP (exploiting Hankel structure) with projected gradient,
LP-based direction finding for escaping local optima, and subgradient polishing.

import numpy as np
from scipy.optimize import linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _hankel_matvec(a, w, n):
    """Compute Q @ a where Q[i,j] = w[i+j]. w has length 2n-1."""
    # (Q*a)[i] = sum_j w[i+j] a[j]
    # = conv(reverse(w), a)[2n-2-i]
    v = w[::-1]
    conv = _full_convolve(v, a)  # length (2n-1) + n - 1 = 3n-2
    # result[i] = conv[2n-2-i] for i=0,...,n-1
    return conv[n-1:2*n-1][::-1]

def _hankel_quadform(a, w):
    """Compute a^T Q a = sum_k w[k] * conv(a,a)[k]."""
    c = _full_convolve(a, a)
    return float(np.dot(w, c))

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _solve_hankel_qp(w, a_init, n, max_iter=20000, lr=None, tol=1e-12):
    """Solve min a^T Q a s.t. sum(a)=1, a>=0 where Q[i,j]=w[i+j]."""
    a = a_init.copy()

        a_new = _project_simplex(a - lr * grad)
        obj_new = _hankel_quadform(a_new, w)
        
        if obj_new < obj_prev - 1e-15:
            a = a_new
            obj_prev = obj_new
            if i > 0 and i % 1000 == 0:
                # Adapt step size
                pass

    return a

def _build_conv_matrix(a):
    """Build matrix A where A[k,j] = a[j] if 0<=k-j<n else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_lo = max(0, k - n + 1)
        j_hi = min(n, k + 1)
        if j_hi > j_lo:
            A[k, j_lo:j_hi] = a[j_lo:j_hi]
    return A

def _lp_direction(a, M):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return b/sum(b)."""
    n = len(a)
    A = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                      bounds=[(0, None)]*n, method='highs',
                      options={'time_limit': 15, 'presolve': True})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            return res.x / res.x.sum()
    except Exception:
        pass
    return None

def _subgrad_step(a, lr):
    """One subgradient step for max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    tol = 0.005 * c_max
    active = c >= (c_max - tol)
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    
    # Subgradient: s[i] = 2 * sum_k w[k] * a[k-i] for 0<=k-i<n
    # Using FFT: s = 2 * cross_correlation(w, a)
    N = _next_pow2(3 * n)
    w_ext = np.zeros(N)
    w_ext[:len(w)] = w
    a_ext = np.zeros(N)
    a_ext[:n] = a
    cross = np.fft.irfft(np.conj(np.fft.rfft(w_ext)) * np.fft.rfft(a_ext), N)
    grad = 2.0 * cross[:n]
    
    a_new = a - lr * grad
    a_new = _project_simplex(a_new)
    return a_new
