```python
import numpy as np
import time

def _next_pow2(n):
    return 1 << (n - 1).bit_length() if n > 1 else 2

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_max_obj(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_weights(c, beta):
    """Compute softmax weights over convolution coefficients."""
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return w

def _grad_smoothed_max(a, beta):
    """Gradient of (1/beta)*log(sum(exp(beta*conv(a,a)))) wrt a."""
    n = len(a)
    c = _full_convolve(a, a)
    w = _softmax_weights(c, beta)
    a_rev = a[::-1]
    # conv(w, a_rev) has length (2n-1) + n - 1 = 3n-2
    # We want elements [n-1, 2n-2] (0-indexed)
    N = _next_pow2(3 * n - 1)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv_res = np.fft.irfft(W * Ar, N)[:3*n-2]
    h = conv_res[n-1:2*n-1]
    return 2.0 * h

def _grad_in_z(a, beta):
    """Gradient in z-space where a = softmax(z)."""
    G = _grad_smoothed_max(a, beta)
    Ga = float(np.dot(G, a))
    return a * (G - Ga)
```

```python
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()  # Center
    
    best_a = a_init.copy()
    best_score = _conv_max_obj(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Beta schedule: start small, increase
    n_total_est = 100000  # estimate total steps
    current_beta = beta_start
    
    while time.monotonic() < deadline:
        t_step += 1
        # Update beta
        progress = min(1.0, t_step / n_total_est)
        current_beta = beta_start + (beta_end - beta_start) * progress
        
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        # Compute gradient
        grad_z = _grad_in_z(a, current_beta)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m_buf = b1 * m_buf + (1 - b1) * grad_z
        v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
        mh = m_buf / (1 - b1**t_step)
        vh = v_buf / (1 - b2**t_step)
        
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        # Periodic evaluation
```

```python
"""
Gradient-based optimization in log-space for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses both smoothed (log-sum-exp) and exact (subgradient) objectives with Adam optimizer.
Explores multiple sequence lengths, initializations, and adaptive temperature schedules.
Key innovation: efficient exact-max subgradient in z-space (grad_z = 2*a*(a[k*-j] - c_max))
combined with multi-scale beta annealing and periodic shape exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    if n <= 1:
        return 2
    return 1 << (int(n) - 1).bit_length()


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_max_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_weights(c, beta):
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return w


def _grad_smoothed(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    w = _softmax_weights(c, beta)
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    N = _next_pow2(conv_len)
    W = np.fft.rfft(w, N)
    Ar = np.fft.rfft(a_rev, N)
    conv_res = np.fft.irfft(W * Ar, N)[:conv_len]
    h = conv_res[n - 1:2 * n - 1]
    return 2.0 * h


def _grad_exact_z(a):
    """Exact subgradient in z-space for max of conv(a,a). Returns (grad_z, c, k_star)."""
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    c_max = c[k_star]
    # grad_z_j = 2 * a_j * (a[k_star - j] - c_max)
    # a[k_star - j] is nonzero only when 0 <= k_star - j <= n-1
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    a_shifted = np.zeros(n)
    a_shifted[valid] = a[idx[valid]]
    grad_z = 2.0 * a * (a_shifted - c_max)
    return grad_z, c, k_star


def _grad_smoothed_z(a, beta):
    """Smoothed gradient in z-space."""
    G = _grad_smoothed(a, beta)
    n = len(a)
    c = _full_convolve(a, a)
    w = _softmax_weights(c, beta)
    W = float(np.dot(w, c))
    return a * (2.0 * (G / 2.0) - 2.0 * W)


def _make_cosine(n):
    j = np.arange(n)
    a = np.cos(np.pi * j / n) ** 2
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def _make_gaussian(n, sigma_frac=0.4):
    j = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    a = np.exp(-j ** 2 / (2.0 * sigma ** 2))
    a = a / a.sum()
    return a


def _make_triangular(n):
    j = np.arange(n)
    mid = (n - 1) / 2.0
    a = np.maximum(0, 1.0 - np.abs(j - mid) / mid)
    a = a / a.sum()
    return a


def _interp_to_n(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _adam_optimize(a_init, deadline, lr=0.005, beta_start=0.0005, beta_end=0.005,
                   use_exact_after_frac=0.5, max_update_norm=0.5, eval_every=50):
    """Adam optimizer in log-space with adaptive beta and transition to exact gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _conv_max_normalized(a_init)
    max_c_ref = float(_full_convolve(a_init, a_init).max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    total_time_est = max(1.0, deadline - time.monotonic())
    est_steps = int(total_time_est / 0.002)  # rough estimate of ~2ms per step
    use_exact_after = int(est_steps * use_exact_after_frac)

    while time.monotonic() < deadline:
        t_step += 1
        progress = min(1.0, t_step / max(1, est_steps))

        # Beta schedule: cosine interpolation
        if t_step < use_exact_after:
            beta = beta_start + (beta_end - beta_start) * (1 - np.cos(np.pi * progress / use_exact_after_frac * (1.0 if progress < use_exact_after_frac else 1.0)))
            beta = beta_start + (beta_end - beta_start) * min(1.0, t_step / max(1, use_exact_after))
            use_exact = False
        else:
            use_exact = True

        a = np.exp(z - z.max())
        a = a / a.sum()

        if use_exact:
            grad_z, c, k_star = _grad_exact_z(a)
        else:
            G = _grad_smoothed(a, beta)
            W_val = float(np.dot(_softmax_weights(_full_convolve(a, a), beta), _full_convolve(a, a)))
            grad_z = a * (G - 2.0 * W_val)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > max_update_norm:
            update = update * (max_update_norm / update_norm)

        z = z - update
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        # Learning rate decay
        if t_step % 5000 == 0:
            current_lr = max(0.0005, current_lr * 0.9)

        if t_step % eval_every == 0:
            sc = _conv_max_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _conv_max_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _simplified_lp_descent(f, n, max_iters=5, time_per_lp=10):
    """
    Simplified LP-based descent: find g maximizing sum(g) subject to conv(f,g) <= max(conv(f,f)), g>=0.
    Use as a direction for gradient step.
    """
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    c = _full_convolve(f, f)
    M = float(c.max())

    # Build constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0
    k_range = np.arange(2 * n - 1)
    j_range = np.arange(n)
    diff = k_range[:, np.newaxis] - j_range[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_per_lp, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s", flush=True)

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        print(f"Parent loaded: n={len(parent)}, score={_score(parent):.10f}", flush=True)
        update_best(parent)

    # === Phase 1: Exploration (0 - 500s) ===
    phase1_end = t_start + 500.0
    print(f"\n=== Phase 1: Exploration (cutoff={phase1_end - t_start:.0f}s) ===", flush=True)

    # Try multiple n values and initializations
    n_values = [4096, 8192, 16384]
    inits = {}

    for n in n_values:
        inits[(n, 'cosine')] = _make_cosine(n)
        inits[(n, 'gaussian')] = _make_gaussian(n, 0.35)
        inits[(n, 'triangular')] = _make_triangular(n)
        inits[(n, 'uniform')] = np.ones(n) / n
        if parent is not None:
            inits[(n, 'parent')] = _interp_to_n(parent, n)

    # Sort by initial score, try best ones first
    scored_inits = []
    for key, a in inits.items():
        sc = _conv_max_normalized(a)
        scored_inits.append((sc, key, a))
    scored_inits.sort(key=lambda x: x[0])

    per_init_time = 35.0
    tried = 0
    max_tries = int(500.0 / per_init_time)

    for sc_init, key, a_init in scored_inits[:max_tries]:
        if time.monotonic() > phase1_end - 20:
            break
        tried += 1
        n_key, name = key
        print(f"  Phase 1 try {tried}: n={n_key}, init={name}, init_score={sc_init:.6f}", flush=True)
        deadline_i = min(time.monotonic() + per_init_time, phase1_end - 10)
        a_res, sc_res = _adam_optimize(
            a_init, deadline_i, lr=0.008,
            beta_start=0.001, beta_end=0.008,
            use_exact_after_frac=0.6, max_update_norm=0.4
        )
        update_best(a_res)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 2: Fine-tuning with smoothed gradient (500s - 1100s) ===
    phase2_end = t_start + 1100.0
    print(f"\n=== Phase 2: Fine-tuning (cutoff={phase2_end - t_start:.0f}s) ===", flush=True)

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_best = best_seq / best_seq.sum()
        n_best = len(a_best)

        # Run longer optimization on the best
        remaining = phase2_end - time.monotonic() - 20
        # Split into 2 runs with different betas
        half = remaining / 2.0

        a_res, sc_res = _adam_optimize(
            a_best, time.monotonic() + half * 0.9, lr=0.004,
            beta_start=0.0003, beta_end=0.003,
            use_exact_after_frac=0.5, max_update_norm=0.3
        )
        update_best(a_res)

        if time.monotonic() < phase2_end - 10:
            a_best2 = best_seq / best_seq.sum()
            a_res2, sc_res2 = _adam_optimize(
                a_best2, min(time.monotonic() + half * 0.8, phase2_end - 5), lr=0.003,
                beta_start=0.0002, beta_end=0.002,
                use_exact_after_frac=0.4, max_update_norm=0.25
            )
            update_best(a_res2)

    # Try different n from best shape
    if best_seq is not None and time.monotonic() < phase2_end - 60:
        for target_n in [n_best // 2, n_best * 2]:
            if target_n < 1024 or target_n > 32768:
                continue
            if time.monotonic() > phase2_end - 40:
                break
            a_interp = _interp_to_n(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"  Interpolated to n={target_n}: score={sc_interp:.10f}", flush=True)
            update_best(a_interp)

            if time.monotonic() < phase2_end - 30:
                a_res, sc_res = _adam_optimize(
                    a_interp, min(time.monotonic() + 40, phase2_end - 5),
                    lr=0.004, beta_start=0.0005, beta_end=0.005,
                    use_exact_after_frac=0.5, max_update_norm=0.3
                )
                update_best(a_res)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 3: Polish with exact gradient (1100s - 1650s) ===
    phase3_end = t_start + 1650.0
    print(f"\n=== Phase 3: Exact gradient polish (cutoff={phase3_end - t_start:.0f}s) ===", flush=True)

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        a_best = best_seq / best_seq.sum()
        remaining = phase3_end - time.monotonic() - 20

        # Multiple passes with decreasing LR
        n_passes = 3
        lrs = [0.002, 0.001, 0.0005]
        per_pass = remaining / n_passes

        for i in range(n_passes):
            if time.monotonic() > phase3_end - 10:
                break
            a_start = best_seq / best_seq.sum()
            a_res, sc_res = _adam_optimize_exact_only(
                a_start, time.monotonic() + per_pass * 0.9,
                lr=lrs[i], max_update_norm=0.2 * (0.7 ** i),
                eval_every=20
            )
            update_best(a_res)

    # Try a few perturbed restarts
    if best_seq is not None and time.monotonic() < phase3_end - 30:
        remaining = phase3_end - time.monotonic() - 20
        n_restarts = 3
        per_restart = remaining / n_restarts
        sigmas = [0.01, 0.02, 0.04]

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 10:
                break
            a_base = best_seq / best_seq.sum()
            z = np.log(np.maximum(a_base, 1e-15))
            z = z + sigmas[i] * rng.standard_normal(len(a_base))
            z = z - z.mean()
            a_pert = np.exp(z - z.max())
            a_pert = a_pert / a_pert.sum()

            a_res, sc_res = _adam_optimize_exact_only(
                a_pert, min(time.monotonic() + per_restart * 0.8, phase3_end - 5),
                lr=0.001, max_update_norm=0.15, eval_every=20
            )
            update_best(a_res)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s", flush=True)

    # === Phase 4: Final ultra-fine polish (1650s - deadline) ===
    print(f"\n=== Phase 4: Final polish ===", flush=True)

    if best_seq is not None and time.monotonic() < deadline - 10:
        a_best = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 8
        if remaining > 15:
            a_final, sc_final = _adam_optimize_exact_only(
                a_best, time.monotonic() + remaining,
                lr=0.0003, max_update_norm=0.1, eval_every=10
            )
            update_best(a_final)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_cosine(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s", flush=True)
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}", flush=True)
        except Exception as ex:
            print(f"Eval check: {ex}", flush=True)
        return [float(x) for x in final]

    # Fallback
    fallback = _make_cosine(2048)
    print(f"FALLBACK: score={_score(fallback):.10f}", flush=True)
    return [float(x) for x in fallback]
```
