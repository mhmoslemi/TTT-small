def _next_pow2(n):
    k = 1

"""
Advanced multi-phase Adam optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines log-softmax parametrization, adaptive temperature cycling, aggressive
learning rate schedules, multi-restart diversity, and length exploration.
Skips LP in favor of faster gradient-based methods for more total iterations.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_lse(a, c, beta):
    """Compute gradient using log-sum-exp approximation to max(c)."""
    n = len(a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad


def _compute_grad_exact(a, c):
    """Compute exact subgradient at argmax of c."""
    n = len(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return 4.0 * n * a * grad


def _adam_run(a_init, deadline, lr_init=0.03, lr_decay=0.93, lr_decay_period=3000,
              beta_init_scale=0.05, beta_min_scale=0.0003, max_beta_iters=15000,
              clip_norm=1.0, b1=0.9, b2=0.999, eps=1e-8, use_exact_max=False,
              exact_max_after=0.8, name=""):
    """
    Adam optimizer with log-sum-exp (or exact max) smoothed objective.
    Operates on a_init normalized to sum=1. Returns (best_a_normalized, best_score).
    """
    n = len(a_init)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    max_c_init = float(c_init.max())
    best_score = float(2.0 * n * max_c_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    current_lr = lr_init

    n_beta_steps = max(3, int(np.ceil(np.log(max(1e-8, beta_init_scale / beta_min_scale)) / np.log(2))))
    betas = [beta_init_scale * max_c_init / (2.0 ** i) for i in range(n_beta_steps)]
    betas[-1] = beta_min_scale * max_c_init

    beta_idx = 0
    iters_in_beta = 0
    total_iters = 0
    start_time = time.monotonic()

    while time.monotonic() < deadline:
        if beta_idx >= len(betas):
            beta_idx = len(betas) - 1
        beta = betas[beta_idx]

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Decide whether to use exact max gradient
        frac_done = (time.monotonic() - start_time) / max(1e-8, deadline - start_time)
        if use_exact_max and frac_done > exact_max_after:
            grad = _compute_grad_exact(a, c)
        else:
            grad = _compute_grad_lse(a, c, beta)

        if not np.all(np.isfinite(grad)):
            break

        t_step += 1
        total_iters += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        iters_in_beta += 1

        if total_iters % 200 == 0:
            a_check = _softmax(z)
            c_check = _full_convolve(a_check, a_check)
            sc = float(2.0 * n * c_check.max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

        if total_iters % lr_decay_period == 0:
            current_lr *= lr_decay

        if iters_in_beta >= max_beta_iters:
            beta_idx += 1
            iters_in_beta = 0

    a_final = _softmax(z)
    c_final = _full_convolve(a_final, a_final)
    sc_final = float(2.0 * n * c_final.max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    elapsed = time.monotonic() - start_time
    if name:
        print(f"    [{name}] iters={total_iters}, elapsed={elapsed:.1f}s, score={best_score:.10f}")

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _fista_run(a_init, deadline, lr=0.01, beta_scale=0.01, name=""):
    """
    FISTA-style optimization with log-sum-exp smoothing.
    Uses proximal gradient with projection onto {a>=0, sum(a)=1}.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    y = a.copy()
    t_k = 1.0
    beta = beta_scale * float(_full_convolve(a, a).max())
    best_a = a.copy()
    best_score = float(2.0 * n * _full_convolve(a, a).max())

    while time.monotonic() < deadline:
        c = _full_convolve(y, y)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s_exp = float(e.sum())
        if s_exp < 1e-300 or not np.isfinite(s_exp):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s_exp

        y_rev = y[::-1]
        wc = _full_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * y * (h - W)

        # Gradient step
        z = y - lr * grad
        # Projection onto {a >= 0, sum(a) = 1}
        # Use bisection for the threshold
        lo, hi = -1e6, z.max() + 1
        for _ in range(60):
            mid = (lo + hi) / 2
            proj = np.maximum(z - mid, 0)
            s_proj = float(proj.sum())
            if s_proj > 1.0:
                lo = mid
            else:
                hi = mid
        proj = np.maximum(z - (lo + hi) / 2, 0)
        s_proj = float(proj.sum())
        if s_proj > 1e-15:
            proj = proj / s_proj
        a_new = proj

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a_new + ((t_k - 1.0) / t_new) * (a_new - a)
        y = np.maximum(y, 0)
        s_y = float(y.sum())
        if s_y > 1e-15:
            y = y / s_y
        a = a_new
        t_k = t_new

        sc = float(2.0 * n * _full_convolve(a, a).max())
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        beta = max(0.0001 * float(_full_convolve(a, a).max()), beta * 0.999)

    return best_a, best_score
