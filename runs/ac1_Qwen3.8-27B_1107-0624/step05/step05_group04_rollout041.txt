import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm(a):
    """Score assuming sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    return beta * (np.log(np.sum(np.exp(shifted))) + cm / beta)

def _lse_grad(z, beta):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(len(z))
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # h[i] = sum_k w[k] * a[k-i] (cyclic-like)
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n-1:2*n-1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    return a * (grad_a - offset)

def _nesterov_aad(z, beta, deadline, rng, lr_init=0.1, max_iter=500000):
    """Nesterov-accelerated Adam on the log-sum-exp objective."""
    n = len(z)
    best_a = _softmax(z).copy()
    best_score = _score_norm(best_a)
    
    v = np.zeros(n)  # momentum (Nesterov velocity)
    m = np.zeros(n)  # Adam first moment
    vhat = np.zeros(n)  # Adam second moment
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Nesterov lookahead
        z_look = z + v
        grad = _lse_grad(z_look, beta)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        vhat = b2 * vhat + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = vhat / (1 - b2**t)
        
        v_new = -lr * mh / (np.sqrt(vh) + eps)
        z = z + v_new
        v = v_new  # Store for Nesterov
        
        # Clip and renormalize
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Adaptive learning rate decay
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.95)
        
        # Random perturbation to escape local minima
        if t % 20000 == 0:
            z = z + 0.05 * rng.normal(size=n)

def _softmax(z):
    z = z - z.max()

    c = _full_conv(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _lse_grad(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _full_conv(a, a)
    cm = float(c.max())

def _upsample(a, target_n):

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS continuation, momentum gradient descent on log-sum-exp,
peak-directed refinement, and random restarts.
Key improvements over previous: more time on final fine-tuning, peak-directed
gradients for exact max reduction, adaptive peak targeting, and aggressive
restarts to escape local minima.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_conv(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta):
    a = _softmax(z)
    c = _full_conv(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _lse_grad(z, beta):
    n = len(z)
    a = _softmax(z)
    c = _full_conv(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_conv(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    return a * (grad_a - offset)


def _peak_grad_from_a(a, peak_idx):
    n = len(a)
    h = np.zeros(n)
    lo = max(0, peak_idx - n + 1)
    hi = min(n, peak_idx + 1)
    if lo >= hi:
        return np.zeros(n)
    valid = np.arange(lo, hi)
    h[valid] = 2.0 * a[peak_idx - valid]
    offset = float(np.dot(a, h))
    return a * (h - offset)


def _run_lbfgs(z_init, beta, deadline):
    n = len(z_init)
    maxiter = 30000
    maxfun = 80000
    try:
        result = minimize(
            _lse_obj, z_init.copy(), args=(beta,),
            method='L-BFGS-B', jac=_lse_grad,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                     'maxls': 50, 'maxfun': maxfun}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            z_new = np.clip(result.x, -80, 80)
            z_new = z_new - z_new.mean()
            return z_new
    except Exception:
        pass
    return z_init.copy()


def _run_momentum_gd(z_init, beta, deadline, lr=0.03, momentum=0.9, noise_period=15000, noise_scale=0.03):
    n = len(z_init)
    z = z_init.copy()
    v = np.zeros(n)
    best_a = _softmax(z).copy()
    best_score = _score_norm(best_a)
    rng_local = np.random.default_rng(12345)

    t = 0
    no_improve = 0
    while time.monotonic() < deadline:
        t += 1
        grad = _lse_grad(z, beta)
        if not np.all(np.isfinite(grad)):
            break
        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if t % 8000 == 0:
            lr = max(0.003, lr * 0.93)

        if noise_period > 0 and t % noise_period == 0:
            z = z + noise_scale * rng_local.normal(size=n)
            z = z - z.mean()
            v *= 0.5

        if no_improve > 50000:
            z = z + 0.1 * rng_local.normal(size=n)
            z = z - z.mean()
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_score


def _run_peak_gd(z_init, deadline, top_k=3, lr_init=0.015, momentum=0.85):
    n = len(z_init)
    z = z_init.copy()
    best_a = _softmax(z).copy()
    best_score = _score_norm(best_a)
    v = np.zeros(n)
    lr = lr_init
    rng_local = np.random.default_rng(999)

    t = 0
    while time.monotonic() < deadline:
        t += 1
        a = _softmax(z)
        c = _full_conv(a, a)

        peak_indices = np.argsort(c)[::-1][:top_k]
        cm = float(c[peak_indices[0]])

        # Weight peaks by how close they are to the max
        peak_weights = np.array([max(0.0, (c[pk] - 0.5 * cm) / cm + 0.1) for pk in peak_indices])
        peak_weights = peak_weights / peak_weights.sum()

        grad = np.zeros(n)
        for i, pk in enumerate(peak_indices):
            g = _peak_grad_from_a(a, pk)
            grad += peak_weights[i] * g

        if not np.all(np.isfinite(grad)):
            break

        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a_new = _softmax(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()

        if t % 5000 == 0:
            lr = max(0.001, lr * 0.94)

        if t % 30000 == 0:
            z = z + 0.04 * rng_local.normal(size=n)
            z = z - z.mean()
            v *= 0.3

    return best_a, best_score

        cm = float(c.max())
        peak_idx = int(np.argmax(c))

        # Adaptive top_k: include peaks within 95% of max
        threshold = 0.95 * cm
        active_peaks = np.where(c >= threshold)[0]
        if len(active_peaks) == 0:
            active_peaks = np.array([peak_idx])
        if len(active_peaks) > 10:

        grad = np.zeros(n)
        for pk in active_peaks:
            g = _peak_grad_from_a(a, pk)
            grad += g
        grad /= len(active_peaks)

        if not np.all(np.isfinite(grad)):
            break

        v = momentum * v - lr * grad
        z = z + v
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        sc = _score_norm(a_new)

            z = z + 0.03 * rng_local.normal(size=n)
            z = z - z.mean()
            v *= 0.4

    return best_a, best_score


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

            z_pert = z_base + noise_scale * rng.normal(size=n_best)
            z_pert = np.clip(z_pert, -80, 80)
            z_pert = z_pert - z_pert.mean()

            t_alloc = min(remaining / (15 - attempt), 90.0)
            dl = time.monotonic() + t_alloc

            c_p = _full_conv(_softmax(z_pert), _softmax(z_pert))
            max_c_p = float(c_p.max())
            beta_p = 1e-3 * max_c_p

            a_r, sc_r = _run_momentum_gd(z_pert, beta_p, dl, lr=0.015, momentum=0.9,
                                          noise_period=8000, noise_scale=0.015)
            update_best(a_r, sc_r, f"restart_{attempt}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    n = 4096
    t = np.linspace(0, 1, n)
    a = (1.0 + 1.5 * t)
    a = a / a.sum()
    print(f"FALLBACK: score={_score_full(a):.10f}")
    return [float(x) for x in a]
