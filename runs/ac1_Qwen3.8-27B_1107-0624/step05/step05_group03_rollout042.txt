```python
def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)

    if s < 1e-300 or not np.isfinite(s):

    if s < 1e-15:
```

```python
def coordinate_step(a, conv, i, max_steps=10):
    n = len(a)
    for _ in range(max_steps):
        # Compute slopes
        # s_k = da[i]/dz[i] * 2 * a[k-i] for valid k
        # da[i]/dz[i] = a[i]*(1-a[i])
        slope_factor = a[i] * (1 - a[i]) * 2
        
        # Find the current max lag
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]
        
        # Slope at the max lag
        k_offset = k_max - i
        if k_offset < 0 or k_offset >= n:
            slope_at_max = 0
        else:
            slope_at_max = slope_factor * a[k_offset]
        
        if abs(slope_at_max) < 1e-15:
            break
        
        # Direction: decrease dz if slope > 0, increase if slope < 0
        direction = -1 if slope_at_max > 0 else 1
        
        # Find the maximum step before another lag becomes the max
        # conv[k] + slope_factor * a[k-i] * dz <= max_val + slope_at_max * dz
        # For the current max: new_max = max_val + slope_at_max * dz
        # For another lag k: conv[k] + slope_factor * a[k-i] * dz
        # The limiting step is where conv[k] + s_k * dz = max_val + slope_at_max * dz for some k
        # i.e., (s_k - slope_at_max) * dz = max_val - conv[k]
        # dz = (max_val - conv[k]) / (s_k - slope_at_max)
        
        best_dz = direction * 1.0  # Default: full step
        
        for k in range(2*n-1):
            k_off = k - i
            if k_off < 0 or k_off >= n:
                s_k = 0
            else:
                s_k = slope_factor * a[k_off]
            
            diff_s = s_k - slope_at_max
            if abs(diff_s) < 1e-15:
                continue
            
            dz_limit = (max_val - conv[k]) / diff_s
            
            if direction > 0 and dz_limit > 0:
                best_dz = min(best_dz, dz_limit)
            elif direction < 0 and dz_limit < 0:
                best_dz = max(best_dz, dz_limit)
        
        # Also limit by non-negativity: a[i] + da[i]/dz * dz >= 0
        # a[i] * (1 + (1-a[i]) * dz) >= 0 (approximately, for small dz)
        # So dz >= -1/(1-a[i]) (if a[i] < 1)
        
        # Apply the step (with a damping factor for safety)
        dz = 0.5 * best_dz  # Damp by half
        
        # Update z[i] (which changes a)
        # a[i] = exp(z[i]) / sum(exp(z))
        # New a[i] = a[i] * exp(dz) / (1 - a[i] + a[i]*exp(dz))
        # New a[j] = a[j] / (1 - a[i] + a[i]*exp(dz)) for j != i
        
        t = np.exp(dz)
        D = 1 + a[i] * (t - 1)
        a[i] = a[i] * t / D
        if D != 1:
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            a[mask] = a[mask] / D
        
        # Update conv (full recompute for safety, or incremental)
        conv = _full_convolve(a, a)
    
    return a, conv
```

Asymmetry-exploiting optimizer using logsumexp + power mean continuation,

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _autocorr(a):
    return _full_convolve(a, a)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s
