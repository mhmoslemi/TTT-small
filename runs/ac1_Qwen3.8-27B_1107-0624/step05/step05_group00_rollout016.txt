"""
Variance-based gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses the key insight that minimizing the variance of the convolution is a smoother
proxy for minimizing the peak, enabling L-BFGS-B in log-space. Combines multiple
initialization strategies, a variance-flattening phase, and a peak-polishing phase.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < max(2, n):
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _variance_obj_and_grad(x):
    """
    Objective: minimize variance of convolution (smooth proxy for max).
    x: log-space parameters, a = exp(x - max(x)) normalized to sum=1.
    Returns (objective, gradient).
    """
    a = np.exp(x - x.max())
    a = a / a.sum()
    n = len(a)
    c = _full_convolve(a, a)
    n_conv = len(c)
    target = 1.0 / n_conv  # since sum(a)=1, sum(conv) = 1
    D = c - target
    V = float(np.sum(D * D))

    # Gradient: dV/da[m] = 4 * sum_j D[j+m] * a[j] (cross-correlation)
    # Compute via FFT
    N = _next_pow2(2 * n_conv - 1)
    D_pad = np.zeros(N)
    D_pad[:n_conv] = D
    a_pad = np.zeros(N)
    a_pad[:n] = a
    FFT_D = np.fft.rfft(D_pad)
    FFT_a = np.fft.rfft(a_pad)
    # Cross-correlation: ifft(FFT(D) * conj(FFT(a)))
    corr = np.fft.irfft(FFT_D * np.conj(FFT_a), N)[:n]
    grad_a = 4.0 * corr

    # Chain rule: dV/dx[m] = grad_a[m] * a[m] - a[m] * sum(grad_a * a)
    # (Jacobian of normalization a -> a/sum(a))
    # dV/dx[m] = (grad_a[m] - sum(grad_a * a)/sum(a)) * a[m]
    # Since sum(a) = 1:
    # dV/dx[m] = (grad_a[m] - sum(grad_a * a)) * a[m]
    correction = float(np.dot(grad_a, a))
    grad_x = (grad_a - correction) * a

    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros_like(x)

    return V, grad_x


def _max_obj_and_grad(x):
    """
    Objective: minimize 2n * max(conv) with sum(a)=1.
    x: log-space, a = exp(x-max(x))/sum(exp(x-max(x))).
    """
    a = np.exp(x - x.max())
    s = a.sum()
    a = a / s
    n = len(a)
    c = _full_convolve(a, a)
    k_star = int(np.argmax(c))
    peak = float(c[k_star])
    obj = 2.0 * n * peak

    # Gradient of peak w.r.t. a: d(conv[k*])/da[m] = 2*a[k*-m] if valid
    grad_a = np.zeros(n)
    for m in range(n):
        j = k_star - m
        if 0 <= j < n:
            grad_a[m] = 2.0 * a[j]
        else:
            grad_a[m] = 0.0

    # Chain rule for normalization
    correction = float(np.dot(grad_a, a))
    grad_x = (grad_a - correction) * a

    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros_like(x)

    return obj, grad_x


def _smoothed_max_obj_and_grad(x, beta):
    """
    Smoothed max objective using softmax approximation.
    """
    a = np.exp(x - x.max())
    s = a.sum()
    a = a / s
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    w = np.exp((c - c_max) / beta)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum

    smoothed_peak = c_max + beta * np.log(w_sum)
    obj = 2.0 * n * smoothed_peak

    # Gradient
    # d(smoothed_peak)/da[m] = sum_k w[k] * d(conv[k])/da[m]
    grad_a = np.zeros(n)
    for m in range(n):
        total = 0.0
        for k in range(len(c)):

    correction = float(np.dot(grad_a, a))
    grad_x = (grad_a - correction) * a

    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros_like(x)

    return obj, grad_x


def _lbfgs_optimize(obj_func, x0, max_iter, max_time, deadline):
    """Run L-BFGS-B optimization with time limit."""
    t_start = time.monotonic()
    best_x = x0.copy()
    best_f = obj_func(x0)[0]

    def func_wrapper(x):
        f, g = obj_func(x)
        if not np.isfinite(f):
            return 1e10, np.zeros_like(x)
        return f, g

    try:
        result = minimize(
            func_wrapper, x0, method='L-BFGS-B',
            options={'maxiter': max_iter, 'maxfun': max_iter * 2,
                     'ftol': 1e-14, 'gtol': 1e-10}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            f_res = result.fun
            if np.isfinite(f_res) and f_res < best_f:
                best_f = f_res
                best_x = result.x.copy()
    except Exception:
        pass

    return best_x, best_f


def _gradient_descent_momentum(obj_func, x0, lr, momentum, max_iter, deadline):

    x = x0.copy()
    n = len(x)
    v = np.zeros(n)
    best_x = x0.copy()
    best_f = obj_func(x0)[0]
    beta1 = momentum

    for it in range(max_iter):
        if time.monotonic() > deadline:

        if not np.isfinite(f):
            break
        if f < best_f:
            best_f = f
            best_x = x.copy()

        v = beta1 * v - lr * g
        x = x + v
        x = x - x.mean()  # re-center for gauge invariance
        x = np.clip(x, -50, 50)

        if it % 1000 == 0:
            lr = lr * 0.95

    return best_x, best_f


def _make_init(n, kind, rng):
    """Create various initial profiles."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sin1":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * t)
        return np.maximum(a, 1e-10) / n
    elif kind == "sin2":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * t)
        return np.maximum(a, 1e-10) / n
    elif kind == "sin3":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.3 * np.cos(6 * np.pi * t)
        return np.maximum(a, 1e-10) / n
    elif kind == "bimodal":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-4.0 * (t - 0.25) ** 2) + np.exp(-4.0 * (t - 0.75) ** 2) + 0.1
        return a / a.sum()
    elif kind == "trilateral":
        t = np.linspace(0, 1, n, endpoint=False)
        a = np.exp(-3.0 * (t - 0.0) ** 2) + np.exp(-3.0 * (t - 0.5) ** 2) + np.exp(-3.0 * (t - 1.0) ** 2) + 0.1
        return a / a.sum()
    elif kind == "block4":
        block_sizes = [n // 4] * 4
        if n % 4 != 0:
            block_sizes[0] += n % 4
        a = np.zeros(n)
        vals = [1.5, 0.8, 1.5, 0.8]
        start = 0
        for i, bs in enumerate(block_sizes):
            a[start:start + bs] = vals[i]
            start += bs
        return a / a.sum()
    elif kind == "block8":
        block_sizes = [n // 8] * 8
        if n % 8 != 0:
            block_sizes[0] += n % 8
        a = np.zeros(n)
        vals = [1.5, 0.7, 1.3, 0.6, 1.5, 0.7, 1.3, 0.6]
        start = 0
        for i, bs in enumerate(block_sizes):
            a[start:start + bs] = vals[i]
            start += bs
        return a / a.sum()
    elif kind == "smooth_noise":
        t = np.linspace(0, 1, n, endpoint=False)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * t) + 0.1 * np.sin(2 * np.pi * 7 * t)
        a = a + 0.05 * rng.standard_normal(n)
        return np.maximum(a, 1e-10) / n
    else:
        return np.ones(n) / n


def _interp_to_n(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        print(f"Parent loaded: n={n_main}, score={_score(parent):.10f}")
        best_seq = parent.copy()
        best_score = _score(parent)
    else:
        n_main = 2048
        parent = np.ones(n_main) / n_main
        best_seq = parent.copy()
        best_score = _score(parent)
        print(f"No parent: n={n_main}, score={best_score:.10f}")

    # Determine working length
    n_work = n_main
    if n_work > 4096:
        n_work = 4096

    a_start = _interp_to_n(parent, n_work)
    sc_start = _score(a_start)
    print(f"Working: n={n_work}, score={sc_start:.10f}")

    initializations = []

    # Add parent-based init
    initializations.append(("parent", a_start.copy()))

    # Add structured inits at n_work
    for kind in ["sin1", "sin2", "bimodal", "block4", "smooth_noise"]:
        a_init = _make_init(n_work, kind, rng)
        initializations.append((kind, a_init))

    # Phase 1: Variance-based optimization (L-BFGS-B in log-space)
    print("\n=== Phase 1: Variance-based L-BFGS-B ===")
    phase1_end = min(deadline, t_start + 600.0)

    for name, a_init in initializations:
        if time.monotonic() > phase1_end - 30:
            break
        x0 = np.log(np.maximum(a_init, 1e-15))
        x0 = x0 - x0.mean()

        per_time = phase1_end - time.monotonic() - 10
        max_iter = max(100, int(per_time * 200))

        x_opt, v_opt = _lbfgs_optimize(
            _variance_obj_and_grad, x0,
            max_iter=max_iter, max_time=per_time, deadline=phase1_end
        )

        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        sc = _score(a_opt)
        print(f"  {name}: var={v_opt:.10f}, score={sc:.10f}, t={time.monotonic()-t_start:.1f}s")

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()

    # Phase 2: Polished max-based optimization from best variance result
    print("\n=== Phase 2: Peak polishing (smoothed max L-BFGS-B) ===")
    phase2_end = min(deadline, t_start + 1000.0)

    a_work = best_seq.copy() / best_seq.sum()
    c_work = _full_convolve(a_work, a_work)
    max_c = float(c_work.max())

    # Decreasing beta schedule
    betas = [0.01 * max_c, 0.005 * max_c, 0.002 * max_c, 0.001 * max_c, 0.0005 * max_c,
             0.0002 * max_c, 0.0001 * max_c]

    x_work = np.log(np.maximum(a_work, 1e-15))
    x_work = x_work - x_work.mean()

    for beta in betas:
        if time.monotonic() > phase2_end - 30:
            break
        per_time = phase2_end - time.monotonic() - 10
        max_iter = max(50, int(per_time * 150))

        def beta_func(x, b=beta):
            return _smoothed_max_obj_and_grad(x, b)

        x_opt, f_opt = _lbfgs_optimize(
            beta_func, x_work,
            max_iter=max_iter, max_time=per_time, deadline=phase2_end
        )

        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        sc = _score(a_opt)
        print(f"  beta={beta:.6f}: score={sc:.10f}, t={time.monotonic()-t_start:.1f}s")

        if sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            x_work = x_opt.copy()
        else:
            x_work = x_opt.copy()

    # Phase 3: Multi-restart with perturbations
    print("\n=== Phase 3: Multi-restart polishing ===")
    phase3_end = min(deadline, t_start + 1400.0)

    a_best = best_seq.copy() / best_seq.sum()
    c_best = _full_convolve(a_best, a_best)
    max_cb = float(c_best.max())
    betas_fine = [0.0005 * max_cb, 0.0002 * max_cb, 0.0001 * max_cb, 0.00005 * max_cb]

    n_restarts = 5
    per_restart = (phase3_end - time.monotonic() - 20) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase3_end - 15:
            break

        # Different perturbation strategies
        if r == 0:
            a_r = _make_init(n_work, "sin3", rng)
            a_r = 0.7 * a_best + 0.3 * a_r
            a_r = np.maximum(a_r, 1e-12) / np.maximum(a_r, 1e-12).sum()
        elif r == 1:
            # Cyclic shift
            shift = n_work // 7
            a_r = np.roll(a_best, shift).copy()
            a_r = a_r / a_r.sum()
        elif r == 2:
            # Gaussian perturbation in log-space
            z = np.log(np.maximum(a_best, 1e-15))
            z += 0.05 * rng.standard_normal(n_work)
            z = z - z.mean()
            a_r = np.exp(z)
            a_r = a_r / a_r.sum()
        elif r == 3:
            # Mirror + blend
            a_mirror = a_best[::-1].copy()
            a_r = 0.5 * a_best + 0.5 * a_mirror
            a_r = a_r / a_r.sum()
        else:
            # Block structure perturbation
            a_r = a_best.copy()
            n_blocks = 16
            for b in range(n_blocks):
                start = b * n_work // n_blocks
                end = (b + 1) * n_work // n_blocks
                scale = 1.0 + 0.1 * rng.standard_normal()
                a_r[start:end] *= scale
            a_r = np.maximum(a_r, 1e-12)
            a_r = a_r / a_r.sum()

        x_r = np.log(np.maximum(a_r, 1e-15))
        x_r = x_r - x_r.mean()

        sc_r = _score(a_r)
        print(f"  Restart {r}: init score={sc_r:.10f}, t={time.monotonic()-t_start:.1f}s")

        # Run smoothed max optimization
        for beta in betas_fine[:3]:
            if time.monotonic() > phase3_end - 10:
                break
            per_t = min(per_restart / 3, phase3_end - time.monotonic() - 5)
            if per_t < 5:
                break
            max_iter = max(20, int(per_t * 150))

            def beta_func2(x, b=beta):
                return _smoothed_max_obj_and_grad(x, b)

            x_opt, f_opt = _lbfgs_optimize(
                beta_func2, x_r,
                max_iter=max_iter, max_time=per_t, deadline=phase3_end
            )
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt = a_opt / a_opt.sum()
            sc = _score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                x_r = x_opt.copy()
                print(f"    beta={beta:.7f}: NEW BEST score={sc:.10f}")
            else:
                x_r = x_opt.copy()

    # Phase 4: Try n=4096 if not already
    print("\n=== Phase 4: n=4096 exploration ===")
    phase4_end = min(deadline, t_start + 1650.0)

    if n_work != 4096 and time.monotonic() < phase4_end - 60:
        n_4096 = 4096
        a_4096 = _interp_to_n(best_seq, n_4096)
        sc_4096 = _score(a_4096)
        print(f"  Interpolated n=4096: score={sc_4096:.10f}")

        if sc_4096 < best_score or True:  # Try optimizing even if not yet better
            # Variance optimization on 4096
            x_4096 = np.log(np.maximum(a_4096, 1e-15))
            x_4096 = x_4096 - x_4096.mean()
            per_t = min(120.0, phase4_end - time.monotonic() - 30)
            if per_t > 30:
                x_4096, v_4096 = _lbfgs_optimize(
                    _variance_obj_and_grad, x_4096,
                    max_iter=int(per_t * 100), max_time=per_t, deadline=phase4_end
                )
                a_4096 = np.exp(x_4096 - x_4096.max())
                a_4096 = a_4096 / a_4096.sum()
                sc_4096 = _score(a_4096)
                print(f"  After variance opt n=4096: score={sc_4096:.10f}")

                if sc_4096 < best_score:
                    best_score = sc_4096
                    best_seq = a_4096.copy()

    # Phase 5: Final ultra-fine polish on the best sequence
    print("\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 20 and best_seq is not None:
        a_final = best_seq.copy() / best_seq.sum()
        n_final = len(a_final)
        c_final = _full_convolve(a_final, a_final)
        max_cf = float(c_final.max())

        x_final = np.log(np.maximum(a_final, 1e-15))
        x_final = x_final - x_final.mean()

        # Very fine beta
        betas_ult = [0.0001 * max_cf, 0.00005 * max_cf, 0.00002 * max_cf,
                     0.00001 * max_cf, 0.000005 * max_cf]

        for beta in betas_ult:
            if time.monotonic() > deadline - 10:
                break
            per_t = deadline - time.monotonic() - 5
            if per_t < 5:
                break
            max_iter = max(10, int(per_t * 100))

            def beta_func3(x, b=beta):
                return _smoothed_max_obj_and_grad(x, b)

            x_opt, f_opt = _lbfgs_optimize(
                beta_func3, x_final,
                max_iter=max_iter, max_time=per_t, deadline=deadline
            )
            a_opt = np.exp(x_opt - x_opt.max())
            a_opt = a_opt / a_opt.sum()
            sc = _score(a_opt)
            if sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                x_final = x_opt.copy()
            else:
                x_final = x_opt.copy()

            print(f"  beta={beta:.8f}: score={sc:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (1.0 / 256)
        sc_final = _score(final)
        print(f"Best score: {sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
