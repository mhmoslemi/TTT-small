        # = 2 * sum_k w_full[k] * a[k-j] (with a[i]=0 for i outside [0,n-1])
        # This is the cross-correlation of w_full with a, evaluated at lag j
        # = 2 * (a_full * w_full_flipped)[j + n - 1] ... 
        # Actually, let's just compute it directly for clarity (n is at most 8192, K is small)
        
        g = np.zeros(n)
        for i, k in enumerate(top_k_idx):
            # g[j] += 2 * w[i] * a[k-j] for valid j (0 <= k-j < n => max(0,k-n+1) <= j <= min(n-1,k))
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            j_range = np.arange(j_lo, j_hi + 1)
            g[j_range] += 2.0 * w[i] * a[k - j_range]
        
        # Nesterov: project to simplex (renormalize)
        g = g - np.dot(a, g) / np.dot(a, a) * a  # project gradient to tangent of simplex
        # Actually for softmax parameterization, we don't need projection.
        # Let's work in the probability simplex directly.
        
        # Step: update y using Nesterov
        # a_new = y - lr * g, then project to simplex
        a_new = y - lr * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        # Nesterov momentum
        theta = (1 + np.sqrt(1 + 4 * momentum * momentum * t)) / 2
        y_new = a_new + (momentum * (t - 1) / (t + theta)) * (a_new - a)
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / s
        
        a = a_new
        y = y_new
        
        # Decay learning rate
        lr = base_lr / (1 + 0.01 * t)
        
        # Evaluate a (not y)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Occasionally evaluate y too
        if t % 50 == 0:
            sc_y = _score_norm_fn(y)
            if np.isfinite(sc_y) and sc_y < best_score:
                best_score = sc_y
                best_a = y.copy()
                a = y.copy()
    
    return best_a, best_score


def _peak_direction_opt(a_init, deadline, rng, max_rounds=10000):
    """For the dominant peak, find direction that best reduces it."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    round_num = 0
    while time.monotonic() < deadline and round_num < max_rounds:
        round_num += 1
        c = _autocorr(a)
        k_peak = int(np.argmax(c))
        
        # Gradient of c[k_peak] w.r.t. a
        j_lo = max(0, k_peak - n + 1)
        j_hi = min(n - 1, k_peak)
        j_range = np.arange(j_lo, j_hi + 1)
        g = np.zeros(n)
        g[j_range] = 2.0 * a[k_peak - j_range]
        
        # Try moving in -g direction (reduces the peak)
        # Use line search with a few trial steps
        best_step = None
        best_trial_sc = best_score
        for step_size in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            a_trial = a - step_size * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc_trial = _score_norm_fn(a_trial)
            if sc_trial < best_trial_sc - 1e-12:
                best_trial_sc = sc_trial
                best_step = a_trial.copy()
        
        if best_step is not None:
            a = best_step
            if _score_norm_fn(a) < best_score:
                best_score = _score_norm_fn(a)
                best_a = a.copy()
        else:
            # Try adding to low elements (anti-peak direction)
            low_idx = np.argsort(a)[:10]
            improved = False
            for idx in low_idx:
                for delta in [0.001, 0.002, 0.005]:
                    a_trial = a.copy()
                    a_trial[idx] += delta
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc_trial = _score_norm_fn(a_trial)
                    if sc_trial < best_score - 1e-12:
                        best_score = sc_trial
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved and round_num > 100:
                break
    
    return best_a, best_score


def _lbfgs_adaptive(a_init, deadline, rng, n_betas=15):
    """L-BFGS with adaptive log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _autocorr(a_init)
    max_c0 = float(c0.max())
    
    beta_start = 0.3 * max_c0
    beta_end = max(1e-8 * max_c0, 1e-20)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 3000)
        maxfun = maxiter * 5
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                         'maxls': 50, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -70, 70)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _coord_descent_polish(a_init, deadline, rng, max_rounds=5000):
    """Coordinate descent: try small perturbations of individual elements."""
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    # Focus on elements that contribute to top peaks
    c = _autocorr(a)
    top_peaks = np.argsort(c)[::-1][:5]
    
    # Which elements contribute most to top peaks?
    important = np.zeros(n)
    for pk in top_peaks:
        j_lo = max(0, pk - n + 1)
        j_hi = min(n - 1, pk)
        j_range = np.arange(j_lo, j_hi + 1)
        contrib = a[j_range] * a[pk - j_range]
        for j in j_range:
            important[j] += contrib[j - j_lo]
    
    # Sort by importance
    order = np.argsort(important)[::-1]
    
    rnd = 0
    while time.monotonic() < deadline and rnd < max_rounds:
        rnd += 1
        improved = False
        # Cycle through important elements
        start = (rnd * 37) % len(order)
        for ii in range(min(50, len(order))):
            idx = order[(start + ii) % len(order)]
            for delta in [0.002, -0.002, 0.005, -0.005, 0.001, -0.001]:
                z_trial = z.copy()
                z_trial[idx] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved and rnd > 50:
            # Add small noise to escape
            z = z + 0.01 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _adam_gd(a_init, deadline, rng, lr_init=0.05):
    """Adam gradient descent on softmax parameterization with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _autocorr(a)
        beta = _adaptive_beta(c)
        cm = float(c.max())
        if cm < 1e-30:
            break
        
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        
        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = _softmax_fn(z)
        
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if t % 2000 == 0:
            lr = max(0.01, lr * 0.85)
        
        if t % 10000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        # Fallback: asymmetric ramp
        n_fb = 256
        t = np.linspace(0, 1, n_fb)
        a0 = 1.0 + 1.5 * t
        a0 = a0 / a0.sum()
        update_best(a0, _score_norm_fn(a0), "fallback_ramp")
    
    n_best = len(best_seq)
    
    # === Phase 1: Nesterov subgradient (25% time) ===
    p1_end = t_start + (budget_s - 20.0) * 0.25
    print(f"\n=== Phase 1: Nesterov subgradient (budget ~{p1_end - t_start:.0f}s) ===")
    if time.monotonic() < p1_end - 5:
        a_nest, sc_nest = _nesterov_subgrad(best_seq, p1_end, rng, base_lr=0.5)
        update_best(a_nest, sc_nest, "nesterov")
    
    # === Phase 2: Peak direction optimization (20% time) ===
    p2_end = t_start + (budget_s - 20.0) * 0.45
    print(f"\n=== Phase 2: Peak direction opt (budget ~{p2_end - t_start:.0f}s) ===")
    if time.monotonic() < p2_end - 5 and best_seq is not None:
        a_pk, sc_pk = _peak_direction_opt(best_seq, p2_end, rng)
        update_best(a_pk, sc_pk, "peak_dir")
    
    # === Phase 3: L-BFGS with adaptive beta (25% time) ===
    p3_end = t_start + (budget_s - 20.0) * 0.70
    print(f"\n=== Phase 3: L-BFGS adaptive (budget ~{p3_end - t_start:.0f}s) ===")
    if time.monotonic() < p3_end - 10 and best_seq is not None:
        a_lb, sc_lb = _lbfgs_adaptive(best_seq, p3_end, rng, n_betas=20)
        update_best(a_lb, sc_lb, "lbfgs_adaptive")
        
        # Try from a slightly perturbed version too
        if time.monotonic() < p3_end - 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.05 * rng.normal(size=n_best)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            dl2 = min(p3_end - 5, time.monotonic() + 60.0)
            a_pert2, sc_pert2 = _lbfgs_adaptive(a_pert, dl2, rng, n_betas=10)
            update_best(a_pert2, sc_pert2, "lbfgs_perturbed")
    
    # === Phase 4: Adam GD (15% time) ===
    p4_end = t_start + (budget_s - 20.0) * 0.85
    print(f"\n=== Phase 4: Adam GD (budget ~{p4_end - t_start:.0f}s) ===")
    if time.monotonic() < p4_end - 5 and best_seq is not None:
        a_adam, sc_adam = _adam_gd(best_seq, p4_end, rng, lr_init=0.03)
        update_best(a_adam, sc_adam, "adam_gd")
    
    # === Phase 5: Coordinate polish (10% of remaining) ===
    p5_end = min(deadline - 10, t_start + (budget_s - 20.0) * 0.95)
    print(f"\n=== Phase 5: Coordinate polish (budget ~{p5_end - t_start:.0f}s) ===")
    if time.monotonic() < p5_end - 5 and best_seq is not None:
        a_cd, sc_cd = _coord_descent_polish(best_seq, p5_end, rng)
        update_best(a_cd, sc_cd, "coord_polish")

        a_up = _upsample(best_seq, 8192)
        sc_up = _score_norm_fn(a_up)

        a_8k, sc_8k = _lbfgs_adaptive(a_up, dl6, rng, n_betas=8)

        try:

            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny, n_final),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 10000, 'ftol': 1e-16, 'gtol': 1e-11,
                         'maxls': 40, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Final L-BFGS: {best_score:.10f}")
        except Exception as ex:
            print(f"  Final L-BFGS failed: {ex}")
    
    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:

        else:

            final_seq = final_seq / final_seq.sum()
        
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]
    
    # Fallback
    t = np.linspace(0, 1, 256)
    fallback = (1.0 + 1.5 * t)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
