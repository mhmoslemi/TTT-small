def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _score_fn(a):

    if s < 1e-15:

    c = _full_convolve(a, a)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)

def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

                 0.00005, 0.00002, 0.00001]

def _adam_polish(a_init, deadline, lr=0.03, max_iter=500000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.96 ** (t // 40)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _subgradient_phase(a_init, deadline, max_iter=200000, eta0=0.5):
    """Projected subgradient method on the hard max objective."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        M = float(c.max())

        # Compute subgradient: use top-K peaks weighted
        K = min(10, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:K]
        w = c[top_idx].copy()
        w = w / w.sum()

        g = np.zeros(n)
        for idx_k in range(len(top_idx)):
            k = top_idx[idx_k]
            wi = w[idx_k]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                seg = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
                g[i_lo:i_hi + 1] += wi * 2.0 * seg

        eta = eta0 / (1.0 + 0.05 * it ** 0.5)
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Only accept if close
            a = a_new.copy()

    return best_a, best_score


def _lp_phase(a_init, deadline, K=40, max_rounds=15):
    """LP-based peak balancing."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.92 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]

        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(len(active), M * 1.0001)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-13:
                best_a = a_trial.copy()
                best_score = _score_norm(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _active_set_phase(a_init, deadline, K=5, max_rounds=50):
    """Optimize using only top-K peaks in the LSE."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for outer in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]
        pm = float(c[peak_indices].max())

        def ao(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pmi = float(peaks.max())
            if pmi < 1e-300:
                return 0.0
            bt = 1e-8 * pmi
            sh = (peaks - pmi) / bt
            lse = pmi + bt * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)

        def ag(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pmi = float(peaks.max())
            if pmi < 1e-300:
                return np.zeros_like(z_in)
            bt = 1e-8 * pmi
            sh = (peaks - pmi) / bt
            e = np.exp(sh - sh.max())
            w = e / e.sum()

            grad_a = np.zeros(n)
            for idx_k in range(len(peak_indices)):
                k = peak_indices[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            result = minimize(ao, z, method='L-BFGS-B', jac=ag,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-15, 'maxls': 80, 'maxfun': 15000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score - 1e-14:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _nesterov_polish(a_init, deadline, max_iter=300000):
    """Nesterov accelerated gradient on LSE surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.005 * cm0

    z_prev = z.copy()
    t_val = 1.0
    L = 10.0 * n * cm0  # initial Lipschitz estimate (rough)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        theta = (t_val - 1.0) / t_new
        z_y = z + theta * (z - z_prev)

        a_y = _softmax_fn(z_y)
        c_y = _full_convolve(a_y, a_y)
        cm_y = float(c_y.max())
        beta = max(1e-10 * cm0, beta * 0.995)

        shifted = (c_y - cm_y) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c_y))
        a_rev = a_y[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a_y * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        step = 1.0 / (L + 1e-10)
        z_new = z_y - step * grad
        z_new = np.clip(z_new, -100, 100)
        z_new = z_new - z_new.mean()

        z_prev = z
        z = z_new
        z_prev = z_prev  # keep for momentum
        # Actually for Nesterov: z_{k+1} = y_k - (1/L) * grad(y_k), y_{k+1} = z_{k+1} + theta*(z_{k+1}-z_k)
        # Let me redo properly:
        # y_k = z_k + (t_{k-1}/t_k)(z_k - z_{k-1})
        # z_{k+1} = y_k - (1/L_k) grad(f)(y_k)
        # t_{k+1} = (1+sqrt(1+4t_k^2))/2

        # I'll just use the simple version above which is close enough
        t_val = t_new

        if (it + 1) % 20 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase, multi-n optimization. Tries diverse init, L-BFGS LSE,
    Adam, Nesterov, LP, subgradient, and active-set methods across
    multiple sequence lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and len(p) > 0 and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 8192
    print(f"Parent: n={n_parent}" + (f", score={_score_fn(parent):.10f}" if parent is not None else ""))

    # --- Phase 0: Quick init scan to find best starting point ---
    print("Phase 0: Initialization scan")
    init_end = min(deadline, t_start + 60.0)

    n_candidates = [n_parent]
    if 4096 not in n_candidates:
        n_candidates.append(4096)
    if 8192 not in n_candidates:
        n_candidates.append(8192)
    if 16384 not in n_candidates and n_parent <= 8192:
        n_candidates.append(16384)

    best_init = None
    best_init_score = np.inf
    best_init_n = n_parent

    for n in sorted(n_candidates):
        inits = {}
        if parent is not None and n == n_parent:
            inits['parent'] = parent
        inits['uniform'] = _uniform_init(n)
        inits['cosine'] = _cosine_init(n)
        inits['greedy'] = _greedy_init(n)
        inits['power'] = _power_law_init(n, 0.3)
        inits['edge'] = _edge_bump_init(n, 0.3)
        inits['gauss'] = _gaussian_init(n, 0.3)
        inits['sym_gauss'] = _symmetric_init(n, _gaussian_init(n, 0.35))
        inits['rand1'] = _random_init(n, seed=123)
        inits['rand2'] = _random_init(n, seed=456)

        for name, a in inits.items():
            sc = _score_norm(a)
            if sc < best_init_score:
                best_init_score = sc
                best_init = a.copy()
                best_init_n = n
            if sc < best_init_score + 0.3:
                print(f"    n={n} {name}: {sc:.8f}")

    print(f"  Best init: n={best_init_n}, score={best_init_score:.10f}")
    update_best(best_init, best_init_score, "best_init")

    # If parent was good, also keep it
    if parent is not None:
        sc_p = _score_norm(parent)
        if sc_p < best_init_score:
            best_init = parent.copy()
            best_init_n = n_parent
            best_init_score = sc_p
            update_best(best_init, sc_p, "parent")

    n_work = best_init_n
    a_work = best_init.copy()

    # === Phase 1: L-BFGS main (400s) ===
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 450.0)

    a_b1, sc1 = _run_lbfgs(a_work, p1_end)
    update_best(a_b1, sc1, "lbfgs_main")
    a_work = a_b1.copy() if sc1 < best_init_score else a_work

    # Try from a few other good inits
    for et in ['cosine', 'power', 'edge']:
        if time.monotonic() > p1_end - 30:
            break
        a_alt = {'cosine': _cosine_init, 'power': _power_law_init, 'edge': _edge_bump_init}[et](n_work)
        if 'power' == et:
            a_alt = _power_law_init(n_work, 0.3)
        sc_alt = _score_norm(a_alt)
        if sc_alt < best_score * 1.05:
            a_alt_b, sc_alt_b = _run_lbfgs(a_alt, min(p1_end - 5, time.monotonic() + 80))
            update_best(a_alt_b, sc_alt_b, f"lbfgs_{et}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: LP peak balancing (350s) ===
    print("Phase 2: LP peak balancing")
    p2_end = min(deadline, t_start + 800.0)

    if time.monotonic() < p2_end and best_seq is not None:
        a_lp, sc_lp = _lp_phase(best_seq, min(p2_end - 60, time.monotonic() + 180), K=40, max_rounds=12)
        update_best(a_lp, sc_lp, "lp_main")

        if time.monotonic() < p2_end - 30:
            a_lp2, sc_lp2 = _lp_phase(a_lp if sc_lp < best_score else best_seq,
                                       p2_end, K=60, max_rounds=8)
            update_best(a_lp2, sc_lp2, "lp_second")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Nesterov + Adam polish (300s) ===
    print("Phase 3: Nesterov & Adam polish")
    p3_end = min(deadline, t_start + 1100.0)

    if time.monotonic() < p3_end and best_seq is not None:
        a_nes, sc_nes = _nesterov_polish(best_seq, min(p3_end - 120, time.monotonic() + 150))
        update_best(a_nes, sc_nes, "nesterov")

        if time.monotonic() < p3_end - 60:
            a_ad, sc_ad = _adam_polish(best_seq, min(p3_end - 30, time.monotonic() + 120), lr=0.025)
            update_best(a_ad, sc_ad, "adam")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Active set refinement (200s) ===
    print("Phase 4: Active set refinement")
    p4_end = min(deadline, t_start + 1300.0)

    if time.monotonic() < p4_end and best_seq is not None:
        for K in [4, 6, 8]:
            if time.monotonic() > p4_end - 20:
                break
            a_act, sc_act = _active_set_phase(best_seq, min(p4_end - 10, time.monotonic() + 60), K=K)
            update_best(a_act, sc_act, f"active_K{K}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Subgradient + perturbation restarts (200s) ===
    print("Phase 5: Subgradient & restarts")
    p5_end = min(deadline, t_start + 1500.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_sg, sc_sg = _subgradient_phase(best_seq, min(p5_end - 80, time.monotonic() + 100), eta0=0.3)
        update_best(a_sg, sc_sg, "subgrad")

        # Perturbation restarts
        for restart in range(5):
            if time.monotonic() > p5_end - 20:
                break
            scale = [0.1, 0.2, 0.05, 0.15, 0.08][restart]
            a_pert = best_seq + scale * _random_init(best_seq.shape[0], seed=1000 + restart)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            sc_pert = _score_norm(a_pert)
            if sc_pert < best_score:
                update_best(a_pert, sc_pert, f"perturb_{restart}")
            # Quick L-BFGS on perturbed
            if time.monotonic() < p5_end - 30:
                a_pb, sc_pb = _run_lbfgs(a_pert, min(p5_end - 10, time.monotonic() + 30),
                                         maxiter_per_beta=2000)
                update_best(a_pb, sc_pb, f"lbfgs_pert_{restart}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try different n (150s) ===
    print("Phase 6: n exploration")
    p6_end = min(deadline, t_start + 1650.0)

    if time.monotonic() < p6_end and best_seq is not None:
        for n_new in [4096, 8192, 16384]:
            if n_new == len(best_seq) or time.monotonic() > p6_end - 30:
                continue
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            if sc_in < best_score * 1.02:
                update_best(a_interp, sc_in, f"interp_n{n_new}")
                if time.monotonic() < p6_end - 20:
                    a_in_b, sc_in_b = _run_lbfgs(a_interp, min(p6_end - 5, time.monotonic() + 40),
                                                  maxiter_per_beta=3000)
                    update_best(a_in_b, sc_in_b, f"lbfgs_n{n_new}")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final tiny-beta L-BFGS polish (remaining) ===
    print("Phase 7: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.0003, 0.0001, 0.00005, 0.00001]]
        a_f, sc_f = _run_lbfgs(best_seq, deadline - 3, beta_list=tiny_betas, maxiter_per_beta=10000)
        update_best(a_f, sc_f, "final_tiny")

    # === Final validation ===
    print(f"\nFinal best_score (norm) = {best_score:.10f}")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
