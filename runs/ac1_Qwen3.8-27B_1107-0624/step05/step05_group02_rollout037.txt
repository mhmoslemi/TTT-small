"""
Optimized Adam-based search for minimizing 2*n*max(conv(a,a))/sum(a)^2
using log-sum-exp smoothing, adaptive temperature scheduling, and
multi-restart strategy with diverse initializations.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w):
    """Compute gradient in log-space: grad_z = a * (G - dot(G, a))
    where G_j = 2 * sum_m w_m * a_{m-j}
    """
    n = len(a)
    w_rev = w[::-1]
    N2 = _next_pow2(n + len(w_rev) - 1)
    fa = np.fft.rfft(a, N2)
    fw = np.fft.rfft(w_rev, N2)
    conv_full = np.fft.irfft(fa * fw, N2)[:n + len(w_rev) - 1]
    G = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    W = float(np.dot(G, a))
    grad_z = a * (G - W)
    return grad_z

                max_iters, m_buf, v_buf, t_start, lr_decay_iters=5000,
                lr_decay_rate=0.85):

    Returns (best_a, best_max_c, m_buf, v_buf, t_step).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    c = _full_convolve(a_init, a_init)
    best_max_c = float(c.max())
    best_a = a_init.copy()

    m_buf = m_buf.copy() if m_buf is not None else np.zeros(n)
    v_buf = v_buf.copy() if v_buf is not None else np.zeros(n)
    t_step = t_start

    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    beta = max(beta_init, beta_min)

    iters_since_improve = 0
    patience = max(2000, max_iters // 5)

    while time.monotonic() < deadline:
        max_per_loop = min(200, max(10, max_iters // 100))
        for _ in range(max_per_loop):
            if time.monotonic() > deadline:
                break
            t_step += 1
            lr = max(0.0005, lr * (lr_decay_rate ** (1.0 / lr_decay_iters)))

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            if cm < best_max_c:
                best_max_c = cm
                best_a = a.copy()
                iters_since_improve = 0
            else:
                iters_since_improve += 1

            # Compute smoothed weights
            c_shifted = (c - cm) / beta
            c_shifted = np.maximum(c_shifted, -50.0)
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)

            z = np.clip(z, -40.0, 40.0)

            v_buf *= 0.5

            z = z - z.mean()
            iters_since_improve = 0

    return best_a, best_max_c, m_buf, v_buf, t_step

    x = np.linspace(0, 1, n)
    a = x ** alpha
    # Add a reverse component to make it non-palindromic
    a += 0.3 * (1 - x) ** (alpha * 0.7)

    return a / a.sum()


def _make_bump(n, center_frac=0.3, width_frac=0.15):
    """Create an off-center Gaussian bump."""
    x = np.linspace(0, 1, n)
    center = center_frac
    width = width_frac
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _make_two_bumps(n, c1=0.2, c2=0.8, w1=0.08, w2=0.08):

    a = np.exp(-0.5 * ((x - c1) / w1) ** 2) + 0.5 * np.exp(-0.5 * ((x - c2) / w2) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)

    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            decay_factor=0.8, max_iters=10**9,
            m_buf=np.zeros(n_small), v_buf=np.zeros(n_small), t_start=0
        )
        update_best(a_small)

        # Second pass at n=4096 with lower temp
        max_c_s2 = float(_full_convolve(a_small, a_small).max())
        a_small, max_c_s2, _, _, _ = _adam_phase(
            a_small, min(phase3_end, time.monotonic() + 150), lr_init=0.005,
            beta_init=0.03 * max_c_s2, beta_min=0.001 * max_c_s2,
            decay_factor=0.75, max_iters=10**9,
            m_buf=np.zeros(n_small), v_buf=np.zeros(n_small), t_start=0
        )
        update_best(a_small)

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Upscale back to n=8192 and refine (1200 - 1500s) ===
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 4: Upscale to n=8192 ===")

    if best_seq is not None and len(best_seq) < n_main:
        a_up = _interp_profile(best_seq, n_main)
        update_best(a_up)

        max_c_up = float(_full_convolve(a_up, a_up).max())
        a_up, max_c_up, _, _, _ = _adam_phase(
            a_up, min(phase4_end, time.monotonic() + 250), lr_init=0.008,
            beta_init=0.08 * max_c_up, beta_min=0.003 * max_c_up,
            decay_factor=0.8, max_iters=10**9,
            m_buf=np.zeros(n_main), v_buf=np.zeros(n_main), t_start=0
        )
        update_best(a_up)

    # Also refine the n=8192 best directly
    if best_seq is not None and len(best_seq) == n_main:
        a_ref = best_seq / best_seq.sum()
        max_c_ref = float(_full_convolve(a_ref, a_ref).max())
        a_ref, max_c_ref, _, _, _ = _adam_phase(
            a_ref, min(phase4_end, time.monotonic() + 250), lr_init=0.005,
            beta_init=0.05 * max_c_ref, beta_min=0.002 * max_c_ref,
            decay_factor=0.75, max_iters=10**9,
            m_buf=np.zeros(n_main), v_buf=np.zeros(n_main), t_start=0
        )
        update_best(a_ref)

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Final polish (1500s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")

    if best_seq is not None and time.monotonic() < deadline - 20:
        a_fin = best_seq / best_seq.sum()
        n_fin = len(a_fin)
        max_c_fin = float(_full_convolve(a_fin, a_fin).max())
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            a_fin, max_c_fin, _, _, _ = _adam_phase(
                a_fin, time.monotonic() + remaining, lr_init=0.003,
                beta_init=0.02 * max_c_fin, beta_min=0.0005 * max_c_fin,
                decay_factor=0.7, max_iters=10**9,
                m_buf=np.zeros(n_fin), v_buf=np.zeros(n_fin), t_start=0
            )
            update_best(a_fin)

    # Additional: try a few more restarts if time remains
    if best_seq is not None and time.monotonic() < deadline - 40:
        a_best_n = best_seq / best_seq.sum()
        n_r = len(a_best_n)
        for sigma in [0.01, 0.03]:
            if time.monotonic() > deadline - 30:
                break
            a_p = _perturb(a_best_n, sigma, rng)
            max_c_p = float(_full_convolve(a_p, a_p).max())
            remaining = deadline - time.monotonic() - 15
            if remaining < 20:
                break
            a_p, _, _, _, _ = _adam_phase(
                a_p, time.monotonic() + remaining * 0.6, lr_init=0.005,
                beta_init=0.05 * max_c_p, beta_min=0.002 * max_c_p,
                decay_factor=0.75, max_iters=10**9,
                m_buf=np.zeros(n_r), v_buf=np.zeros(n_r), t_start=0
            )
            update_best(a_p)

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        sc_final = _score(final)
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={elapsed:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
