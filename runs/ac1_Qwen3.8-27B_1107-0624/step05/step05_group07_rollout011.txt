"""
Multi-scale optimization with L2-projected gradient spreading, adaptive log-sum-exp L-BFGS,
iterated local search, and peak-flatness-adaptive smoothing for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations: L2 projective gradient for peak spreading, momentum-based updates,
iterated local search with diverse perturbations, and multi-scale coarse-to-fine approach.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _l2_spread_grad(a):
    """Gradient of sum(c - mean(c))^2 w.r.t. a, where c = conv(a,a)."""
    n = len(a)
    c = _full_convolve(a, a)
    mean_c = c.sum() / len(c)
    d = c - mean_c
    a_rev = a[::-1].copy()
    d_conv = _full_convolve(d, a_rev)
    grad = 2.0 * d_conv[n - 1:2 * n - 1]
    return grad, c, float(c.max())


def _weighted_l2_grad(a, top_frac=0.1, power=2.0):
    """Weighted L2 gradient focusing on top convolution values."""
    n = len(a)
    c = _full_convolve(a, a)
    mean_c = c.sum() / len(c)
    d = c - mean_c
    # Weight by current values (focus on peaks)
    weights = (c / max(c.max(), 1e-30)) ** power
    wd = weights * d
    a_rev = a[::-1].copy()
    d_conv = _full_convolve(wd, a_rev)
    grad = 2.0 * d_conv[n - 1:2 * n - 1]
    return grad, c, float(c.max())


def _run_l2_projected_grad(a_init, deadline, rng, lr_init=0.01, use_momentum=True,
                           use_weighted=False, max_iter=100000):
    """L2 projected gradient with optional momentum and weighted peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        return a_init.copy(), _score_norm_fn(a_init)
    a = a / s
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.99
    lr = lr_init
    patience = 0
    patience_limit = 5000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        if use_weighted:

        else:
            grad, c, cm = _l2_spread_grad(a)

        if not np.all(np.isfinite(grad)):
            break

        if use_momentum:
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            a_new = a - lr * mh / (np.sqrt(vh) + 1e-8)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_profiles(n, rng):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Cosine bumps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    # Linear
    for slope in [0.5, 1.0, 2.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Cosine modulated
    for freq in [2, 3, 4, 6]:
        a = 1.0 + 0.3 * np.cos(freq * np.pi * u)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    # Bimodal
    a = np.exp(-20 * (u - 0.4)**2) + np.exp(-20 * (u + 0.4)**2)
    profiles.append(a / a.sum())
    # Random smooth
    for _ in range(5):
        z = rng.normal(size=n) * 0.5
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        z_s = gaussian_filter1d(z, sigma=5)
        a = np.exp(z_s)
        profiles.append(a / a.sum())
    return profiles


def _coordinate_descent(a_init, deadline, max_rounds=20):
    """Coordinate descent on log-scale."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    deltas = [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        order = np.arange(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in deltas:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local - 1e-12:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -60, 60)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _spectral_perturb(a, rng, strength=0.05):

    n = len(a)
    N = _next_pow2(n)
    A = np.fft.rfft(a, N)
    # Perturb magnitudes
    noise = strength * rng.normal(size=A.shape)
    A_nb = A + 1j * (noise * np.abs(A))
    # Reconstruct: a_new = ifft(A_nb * A / |A|^2) to preserve phase
    phase = np.angle(A)
    mag = np.abs(A) * np.exp(noise * 0.1)
    A_new = mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        profiles = _make_profiles(n_init, rng)
        parent = profiles[0]
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    print(f"\nStarting optimization with n={n_parent}, score={best_score:.10f}")

    # === Phase 1: L2 Projected Gradient (spreading) ===
    phase1_end = min(deadline, t_start + 200.0)
    print(f"\n=== Phase 1: L2 Projected Gradient (spread) ===")

    # Run at parent size
    a_cur = parent.copy()
    a_best, sc_best = _run_l2_projected_grad(a_cur, min(phase1_end, t_start + 100.0),
                                             rng, lr_init=0.02, use_momentum=True,
                                             use_weighted=False)
    update_best(a_best, sc_best, "l2_pg_n2048")

    # Also run weighted version
    if time.monotonic() < phase1_end - 30:
        a_dw = best_seq.copy() if best_seq is not None else parent.copy()
        a_best, sc_best = _run_l2_projected_grad(a_dw, phase1_end,
                                                 rng, lr_init=0.01, use_momentum=True,
                                                 use_weighted=True)
        update_best(a_best, sc_best, "l2_wpg")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Adaptive beta L-BFGS ===
    phase2_end = min(deadline, t_start + 600.0)
    print(f"\n=== Phase 2: Adaptive Beta L-BFGS ===")

    # Run from current best
    a_cur = best_seq.copy()
    a_best, sc_best = _run_adaptive_beta_lbfgs(a_cur, min(phase2_end, time.monotonic() + 250.0),
                                               max_total_iter=20000)
    update_best(a_best, sc_best, "adapt_lbfgs_1")

    # Run with noise perturbation
    if time.monotonic() < phase2_end - 50:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.03 * rng.normal(size=len(z))
        z = z - z.mean()
        a_pert = _softmax_fn(z)
        a_best, sc_best = _run_adaptive_beta_lbfgs(a_pert, phase2_end,
                                                   max_total_iter=15000)
        update_best(a_best, sc_best, "adapt_lbfgs_pert")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Iterated Local Search ===
    phase3_end = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 3: Iterated Local Search ===")

    n_ils = 6
    ills_time_per = (phase3_end - time.monotonic()) / max(1, n_ils)

    for ills_i in range(n_ils):
        if time.monotonic() > phase3_end - 20:
            break
        per_dl = min(phase3_end - 10, time.monotonic() + ills_time_per)

        # Create diverse perturbation
        strength = 0.02 + 0.01 * ills_i
        if ills_i % 3 == 0:
            # Frequency domain perturbation
            a_start = _spectral_perturb(best_seq, rng, strength)
        elif ills_i % 3 == 1:
            # Additive noise
            a_start = best_seq + strength * rng.normal(size=len(best_seq))
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        else:
            # Multiply by random smooth function
            n = len(best_seq)
            t = np.linspace(0, 1, n)
            mult = 1.0 + strength * np.cos(2 * np.pi * rng.integers(1, 8) * t + rng.uniform(0, 2 * np.pi))
            a_start = best_seq * mult
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()

        # Quick L-BFGS from perturbed start
        beta_list = [b * float(_full_convolve(a_start, a_start).max())
                     for b in [0.15, 0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]]
        a_best, sc_best = _run_lse_lbfgs(a_start, per_dl, beta_list=beta_list, max_iter_total=8000)
        update_best(a_best, sc_best, f"ils_{ills_i}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Multi-scale approach ===
    phase4_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 4: Multi-scale refinement ===")

    # Try optimizing at smaller n, then upsample
    if time.monotonic() < phase4_end - 60:
        for n_small in [1024, 512]:
            if time.monotonic() > phase4_end - 30:
                break
            a_small = _interp_profile(best_seq, n_small)
            sc_small = _score_norm_fn(a_small)
            print(f"  Small n={n_small}: score={sc_small:.10f}")

            # Optimize at small n
            per_dl = min(phase4_end - 30, time.monotonic() + 60)
            a_small_opt, sc_small_opt = _run_lse_lbfgs(a_small, per_dl, max_iter_total=10000)

            # Upsample back
            a_up = _interp_profile(a_small_opt, n_parent)
            sc_up = _score_norm_fn(a_up)
            update_best(a_up, sc_up, f"multiscale_n{n_small}_to_n{n_parent}")

    # Try larger n
    if time.monotonic() < phase4_end - 30:
        n_large = max(n_parent, 8192)
        if n_large > len(best_seq):
            a_large = _interp_profile(best_seq, n_large)
            sc_large = _score_norm_fn(a_large)
            print(f"  Large n={n_large}: interp score={sc_large:.10f}")
            update_best(a_large, sc_large, f"interp_n{n_large}")

            # Quick optimize at large n
            per_dl = min(phase4_end - 10, time.monotonic() + 80)
            a_large_opt, sc_large_opt = _run_lse_lbfgs(a_large, per_dl, max_iter_total=8000)
            update_best(a_large_opt, sc_large_opt, f"opt_n{n_large}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Fine-tuning ===
    phase5_end = min(deadline, t_start + 1600.0)
    print(f"\n=== Phase 5: Fine-tuning ===")

    if time.monotonic() < phase5_end and best_seq is not None:
        # Very fine L-BFGS
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        beta_tiny = 2e-5 * max_cf

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        per_dl = min(phase5_end - 60, time.monotonic() + 100)
        try:
            result = minimize(
                _lse_obj, z, args=(beta_tiny, len(best_seq)),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': 10000, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 30, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_seq = a_new.copy()
                    print(f"  Fine L-BFGS: {best_score:.10f}")
        except Exception:
            pass

        # Coordinate descent polish
        if time.monotonic() < phase5_end - 20:
            per_dl = min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic()) * 0.6)
            a_cd, sc_cd = _coordinate_descent(best_seq, per_dl, max_rounds=15)
            update_best(a_cd, sc_cd, "coord_descent")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final attempts with remaining time ===
    phase6_end = deadline
    print(f"\n=== Phase 6: Final attempts ===")

    if time.monotonic() < phase6_end - 10 and best_seq is not None:
        # One more L2-grad pass
        per_dl = min(phase6_end - 20, time.monotonic() + 60)
        a_l2, sc_l2 = _run_l2_projected_grad(best_seq, per_dl, rng,
                                             lr_init=0.005, use_momentum=True,
                                             use_weighted=True)
        update_best(a_l2, sc_l2, "final_l2")

        # Tiny beta L-BFGS
        if time.monotonic() < phase6_end - 10:
            c_f = _full_convolve(best_seq, best_seq)
            beta_ultra = 1e-5 * float(c_f.max())
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            per_dl2 = min(phase6_end - 5, time.monotonic() + 40)
            try:
                result = minimize(
                    _lse_obj, z, args=(beta_ultra, len(best_seq)),
                    method='L-BFGS-B', jac=_lse_grad,
                    options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-14,
                             'maxls': 20, 'maxfun': 10000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_seq = a_new.copy()
                        print(f"  Ultra-β L-BFGS: {best_score:.10f}")
            except Exception:
                pass

    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
