High-performance optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based convolution, adaptive top-K gradient descent with momentum,

from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    n = len(a)

    return float(2.0 * n * c.max() / (s * s))

    v = np.asarray(v, dtype=np.float64).copy()
    n = len(v)
    if n == 0:
        return v
    total = v.sum()
    if total < 1e-300:

    v = v * (target_sum / total)

    cond = u * arange > (cssv - 1.0)
    rho = np.nonzero(cond)[0]

    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)

    return a / a.sum()


def _make_power_shape(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u * 0.99) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_resample(a, n_new):
    a = np.asarray(a, dtype=np.float64)
    n_old = len(a)
    if n_old == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n_old)
    x1 = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _topk_gradient(a, conv, K, beta):
    n = len(a)
    K = min(K, len(conv))
    top_idx = np.argsort(conv)[-K:]
    top_vals = conv[top_idx]
    shifted = (top_vals - top_vals.max()) / beta
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(K)
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(K):
        k = int(top_idx[idx])
        wi = w[idx]
        if wi < 1e-15:
            continue
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * wi * a[k - hi:k - lo + 1][::-1]
    return grad


def _full_gradient_fft(a, conv, beta):
    n = len(a)
    N_conv = len(conv)
    shifted = (conv - conv.max()) / beta
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(N_conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    N = _next_pow2(N_conv + n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(W * np.conj(A), N)
    grad = 2.0 * cross[:n]
    grad = np.maximum(grad, 0.0)
    return grad


def _adaptive_topk_gd(a_init, deadline, K_schedule=None, beta_start=None, beta_end=None,
                      lr_start=0.01, momentum=0.9, seed=42):
    """Adaptive top-K gradient descent with momentum and simplex projection."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    if beta_start is None:
        c0 = _full_convolve(a, a)
        beta_start = 0.05 * c0.max()
    if beta_end is None:
        beta_end = 0.0001 * beta_start
    if K_schedule is None:
        K_schedule = [3, 5, 10, 20, 40, 80]

    beta = beta_start
    lr = lr_start
    velocity = np.zeros(n)
    total_iters = 0

    phase_count = len(K_schedule)
    per_phase_time = max(1.0, (deadline - time.monotonic()) / phase_count * 0.8)

    for phase_idx, K in enumerate(K_schedule):
        phase_deadline = min(deadline - 2.0, time.monotonic() + per_phase_time * 2)
        beta_ratio = phase_idx / max(1, phase_count - 1)
        target_beta = beta_start * (beta_end / beta_start) ** beta_ratio

        while time.monotonic() < phase_deadline:
            batch = _full_convolve(a, a)
            grad = _topk_gradient(a, batch, K, beta)

            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-300:
                grad = grad / gnorm

            velocity = momentum * velocity - lr * grad
            a_new = a + velocity
            a_new = _simplex_project(a_new, 1.0)

            a = a_new
            total_iters += 1

            if total_iters % 50 == 0:
                sc = _score_norm_fn(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

            beta = max(target_beta, beta * 0.995)
            lr = max(0.0001, lr * 0.9995)

    sc_final = _score_norm_fn(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()

    return best_a, best_sc


def _lbfgs_logspace(a_init, deadline, beta_scale=1.0, max_wall_time=None):
    """L-BFGS in log-space with logsumexp continuation."""
    n = len(a_init)
    a0 = np.maximum(a_init, 1e-15)
    z = np.log(a0)
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    betas = [b * max_c0 * beta_scale for b in
             [0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]

    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 20.0

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 3.0:
            break
        level_dl = min(deadline - 2.0, time.monotonic() + per_level)

        def obj(z_):
            z_shift = z_ - z_.max()
            e = np.exp(z_shift)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * max_c0
            a_ = e / s
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            es = np.exp(cs)
            ss = es.sum()
            if ss < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(ss))

        def grad_fn(z_):
            z_shift = z_ - z_.max()
            e = np.exp(z_shift)
            s = e.sum()
            if s < 1e-300:
                return np.zeros_like(z_)
            a_ = e / s
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            es = np.exp(cs)
            ss = es.sum()
            if ss < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = es / ss
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            g = g / s * n
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(obj, z.copy(), jac=grad_fn, method='L-BFGS-B',
                              options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-14,
                                       'maxls': 50, 'maxfun': 20000,
                                       'iprint': -1})
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_shift = z_new - z_new.max()
                e_new = np.exp(z_shift)
                a_new = e_new / e_new.sum()
                a_new = np.maximum(a_new, 0.0)
                s_val = a_new.sum()
                if s_val > 1e-10:
                    a_new = a_new / s_val
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_sc:
                        best_sc = sc_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_sc


def _coordinate_descent(a_init, deadline, n_trials=50000, seed=42):
    """Coordinate descent targeting the maximum convolution lag."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        k_max = int(np.argmax(c))
        max_val = c[k_max]

        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        if i_min > i_max:
            i_min, i_max = 0, n - 1

        # Find the pair with largest product at k_max
        pairs = np.arange(i_min, i_max + 1)
        partners = k_max - pairs
        products = a[pairs] * a[partners]
        idx = pairs[int(np.argmax(products))]

        # Perturb: reduce idx, add to a random "safe" position
        a_trial = a.copy()
        frac = rng.uniform(0.05, 0.3)
        delta = a_trial[idx] * frac
        a_trial[idx] -= delta

        # Choose safe position (low contribution to top lags)
        top3 = np.argsort(c)[-3:]
        contrib = np.zeros(n)
        for kl in top3:
            kl = int(kl)
            lo = max(0, kl - (n - 1))
            hi = min(n - 1, kl)
            for j in range(lo, hi + 1):
                partner = kl - j
                if 0 <= partner < n:
                    contrib[j] += a[partner]
        safe_idx = np.argsort(contrib)[:max(1, n // 10)]
        add_idx = int(safe_idx[rng.integers(len(safe_idx))])
        a_trial[add_idx] += delta
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s > 1e-10:
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        # Also try pure random perturbation every 5 steps
        if it % 5 == 0:
            a_trial = a.copy()
            block = int(rng.integers(1, min(8, n // 100)))
            start = int(rng.integers(0, n - block))
            a_trial[start:start + block] *= rng.uniform(0.8, 1.2, size=block)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_sc


def _stochastic_subgrad(a_init, deadline, seed=42, lr_start=0.005, n_trials=200000):
    """Stochastic subgradient with top-K and learning rate decay."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    lr = lr_start
    momentum = 0.95
    velocity = np.zeros(n)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        K = min(5, n)
        top_idx = np.argsort(c)[-K:]
        top_vals = c[top_idx]
        shifted = (top_vals - top_vals.max()) / (0.01 * c.max())
        w = np.exp(shifted)
        w /= w.sum()

        grad = np.zeros(n)
        for idx_i in range(K):
            k = int(top_idx[idx_i])
            wi = w[idx_i]
            if wi < 1e-10:
                continue
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi + 1] += 2.0 * wi * a[k - hi:k - lo + 1][::-1]

        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-300:
            grad /= gnorm

        velocity = momentum * velocity - lr * grad
        a_new = a + velocity
        a_new = _simplex_project(a_new, 1.0)
        a = a_new

        if it % 20 == 0:
            sc = _score_norm_fn(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

        lr = max(0.00001, lr_start / (1.0 + it * 0.0001))
        momentum = min(0.99, 0.9 + 0.09 * it / n_trials)

    return best_a, best_sc


def _multi_n_search(a_best, deadline, seed=42):
    """Try different sequence lengths."""
    n_curr = len(a_best)
    best_a = a_best.copy()
    best_sc = _score_norm_fn(a_best)

    target_ns = []
    for ns in [2048, 4096, 8192, 16384, 32768]:
        if ns != n_curr and (ns > 512):
            target_ns.append(ns)

    target_ns.sort(key=lambda x: abs(x - n_curr))

    for target_n in target_ns[:3]:
        if time.monotonic() > deadline - 30.0:
            break
        p_interp = _interp_resample(a_best, target_n)
        sc_interp = _score_norm_fn(p_interp)
        if sc_interp < best_sc:
            best_sc = sc_interp
            best_a = p_interp.copy()
            print(f"  multi_n: n={target_n} interp score={sc_interp:.10f}")

        # Quick optimize at this n
        if time.monotonic() < deadline - 40.0:
            dl = min(deadline - 30.0, time.monotonic() + 40.0)
            a_opt, sc_opt = _adaptive_topk_gd(p_interp, dl, seed=seed + target_n)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt.copy()
                print(f"  multi_n: n={target_n} opt score={sc_opt:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse init -> adaptive GD -> L-BFGS polish ->
    coordinate descent -> multi-n search -> final polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(10.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1) if int(seed) < 2**63 - 1 else 12345)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_work = len(parent)
        print(f"Parent: n={n_work}, score={_score_fn(parent):.10f}")
    else:
        n_work = 8192
        parent = _make_triangular(n_work)
        print(f"No parent, using triangular n={n_work}")

    # Phase 1: Diverse initialization (30s)
    phase1_end = min(deadline, t_start + 30.0)
    candidates = []

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        candidates.append((sc_p, parent.copy(), "parent"))
        update_best(parent, sc_p, "parent")

    profiles = [
        _make_triangular(n_work),
        _make_b_spline(n_work, 2),
        _make_b_spline(n_work, 3),
        _make_b_spline(n_work, 4),
        _make_raised_cosine(n_work),
        _make_power_shape(n_work, 1.5),
        _make_power_shape(n_work, 2.0),
        _make_power_shape(n_work, 3.0),
        _make_power_shape(n_work, 5.0),
    ]

    for i, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p, f"prof{i}"))
        update_best(p, sc, f"prof{i}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}")

    # Phase 2: Adaptive top-K gradient descent (600s)
    phase2_end = min(deadline, t_start + 630.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: Adaptive top-K GD")
        start_a = candidates[0][1].copy()
        a_opt, sc_opt = _adaptive_topk_gd(start_a, phase2_end,
                                          K_schedule=[3, 5, 10, 20, 50, 100],
                                          seed=seed)
        update_best(a_opt, sc_opt, "topk_gd")

        # Second run with different params
        if time.monotonic() < phase2_end - 60.0:
            a_opt2, sc_opt2 = _adaptive_topk_gd(start_a, phase2_end - 30.0,
                                                K_schedule=[5, 10, 30, 50],
                                                lr_start=0.005, momentum=0.97,
                                                seed=seed + 100)
            update_best(a_opt2, sc_opt2, "topk_gd_v2")

    # Phase 3: L-BFGS polishing (400s)
    phase3_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: L-BFGS polish")
        dl = phase3_end - 5.0
        a_opt, sc_opt = _lbfgs_logspace(best_seq, dl, beta_scale=1.0,
                                        max_wall_time=min(300.0, dl - time.monotonic()))
        update_best(a_opt, sc_opt, "lbfgs_main")

        # More aggressive
        if time.monotonic() < phase3_end - 30.0:
            dl2 = phase3_end - 5.0
            a_opt2, sc_opt2 = _lbfgs_logspace(best_seq, dl2, beta_scale=0.5,
                                              max_wall_time=min(120.0, dl2 - time.monotonic()))
            update_best(a_opt2, sc_opt2, "lbfgs_fine")

    # Phase 4: Stochastic subgradient (200s)
    phase4_end = min(deadline, t_start + 1230.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Stochastic subgradient")
        dl = phase4_end - 5.0
        a_opt, sc_opt = _stochastic_subgrad(best_seq, dl, seed=seed + 200,
                                            lr_start=0.003, n_trials=300000)
        update_best(a_opt, sc_opt, "stoch_sg")

    # Phase 5: Coordinate descent (150s)
    phase5_end = min(deadline, t_start + 1380.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Coordinate descent")
        dl = phase5_end - 5.0
        a_opt, sc_opt = _coordinate_descent(best_seq, dl, n_trials=30000, seed=seed + 300)
        update_best(a_opt, sc_opt, "coord_desc")

    # Phase 6: Multi-n search (120s)
    phase6_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Multi-n search")
        a_opt, sc_opt = _multi_n_search(best_seq, phase6_end, seed=seed + 400)
        update_best(a_opt, sc_opt, "multi_n")

    # Phase 7: Final intensive polish (remaining)
    if best_seq is not None and time.monotonic() < deadline - 20.0:
        remaining = max(10.0, deadline - time.monotonic() - 20.0)
        print(f"Phase 7: Final polish, remaining={remaining:.0f}s")

        # 7a: Very fine L-BFGS
        if remaining > 40.0:
            dl = min(deadline - 20.0, time.monotonic() + remaining * 0.4)
            a_opt, sc_opt = _lbfgs_logspace(best_seq, dl, beta_scale=0.3,
                                            max_wall_time=dl - time.monotonic())
            update_best(a_opt, sc_opt, "final_lbfgs")

        # 7b: More coordinate descent
        if time.monotonic() < deadline - 25.0:
            dl = min(deadline - 25.0, time.monotonic() + remaining * 0.3)
            a_opt, sc_opt = _coordinate_descent(best_seq, dl, n_trials=20000,
                                                seed=seed + 500)
            update_best(a_opt, sc_opt, "final_coord")

        # 7c: Tiny-step stochastic
        if time.monotonic() < deadline - 20.0:
            dl = min(deadline - 20.0, time.monotonic() + remaining * 0.2)
            a_opt, sc_opt = _stochastic_subgrad(best_seq, dl, seed=seed + 600,
                                                lr_start=0.001, n_trials=100000)
            update_best(a_opt, sc_opt, "final_stoch")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator
        if len(final_seq) <= 32768:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular(8192)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
