def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Compute h[i] = sum_j a[j] * w[i+j] for i = 0..n-1"""
    n = len(a)
    L = len(w)

    W = np.fft.rfft(w, N)

    return C[:n]

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over
non-negative sequences. Combines LP-based direction finding with Adam
gradient descent using log-sum-exp smoothing, multi-scale beta annealing,
multi-restart, and coordinate perturbation polishing.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _mc(a):
    """Maximum of self-convolution (for normalized sequences with sum=1)."""
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_direction(f, M, time_limit=20):
    n = len(f)
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _lp_refine(a, deadline):
    n = len(a)
    f = a.copy()
    f = f / f.sum()
    
    steps = 0
    t_step = 0.5
    stuck = 0
    
    while time.monotonic() < deadline:
        c = _full_convolve(f, f)
        M = float(c.max())
        
        g = _lp_direction(f, M, time_limit=15)
        if g is None:
            g = f.copy()
            stuck += 1
            if stuck > 8:
                break
        else:
            stuck = 0
        
        g = g / g.sum()
        
        # Line search
        improved = False
        t_try = t_step
        for _ in range(8):
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / s
            M_new = _mc(f_new)
            if M_new < M * (1 - 1e-10):
                f = f_new
                t_step = min(1.0, t_try * 1.3)
                improved = True
                steps += 1
                if steps % 2 == 0:
                    print(f"  LP step {steps}: M={M:.10f} -> {M_new:.10f} "
                          f"score={_score(f):.10f} t={time.monotonic()-t_start_global:.1f}s")
                break
            else:
                t_try *= 0.5
        
        if not improved:
            t_step *= 0.5
            stuck += 1
            if stuck > 15:
                break
    
    return f


def _adam_phase(a_init, deadline, beta_start, beta_end, n_levels, lr_start,
                lr_decay_per_1000=0.97, max_update_norm=0.5, 
                patience=5000, verbose=True):

    n = len(a_init)
    betas = np.geomspace(beta_start, beta_end, n_levels)
    
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_mc = _mc(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    total_time = deadline - time.monotonic()
    time_per_level = total_time / n_levels
    
    for level, beta in enumerate(betas):
        level_deadline = time.monotonic() + time_per_level
        current_lr = lr_start
        no_improve_count = 0
        last_best_mc = best_mc
        level_start_mc = best_mc
        
        while time.monotonic() < level_deadline:
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            
            # Cross-correlation: h[i] = sum_j a[j] * w[i+j]
            a_rev = a[::-1]
            wc = _full_convolve(a_rev, w)
            h = wc[n-1:2*n-1]
            W = float(np.dot(a, h))
            
            grad = 4.0 * n * a * (h - W)
            
            if not np.all(np.isfinite(grad)):
                break
            
            # Gradient clipping
            gn = np.linalg.norm(grad)
            if gn > 100.0:
                grad = grad * (100.0 / gn)
            
            t_step += 1
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > max_update_norm:
                update = update * (max_update_norm / un)
            
            z = z - update
            z = np.clip(z, -80.0, 80.0)
            z = z - z.mean()
            
            if t_step % 200 == 0:
                a_check = _softmax(z)
                mc_check = _mc(a_check)
                if mc_check < best_mc:
                    best_mc = mc_check
                    best_a = a_check.copy()
                    no_improve_count = 0
                    if verbose and (mc_check < level_start_mc * (1 - 1e-5)):
                        print(f"    Level {level} (beta={beta:.6f}): "
                              f"MC={best_mc:.12f} score={_score(best_a):.10f} "
                              f"t={time.monotonic()-t_start_global:.1f}s")
                else:
                    no_improve_count += 200
                
                if t_step % 3000 == 0:
                    current_lr *= lr_decay_per_1000
                    current_lr = max(current_lr, 0.001)
        
        # End of level
        if verbose:
            print(f"  Beta level {level} done (beta={beta:.6e}): "
                  f"best_mc={best_mc:.12f}")
    
    # Final evaluation
    a_final = _softmax(z)
    mc_final = _mc(a_final)
    if mc_final < best_mc:
        best_mc = mc_final
        best_a = a_final.copy()
    
    return best_a, best_mc


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _coord_polish(a, deadline, rng):
    """Perturb random blocks of coordinates and keep improvements."""
    n = len(a)
    best_a = a.copy()
    best_mc = _mc(a)
    
    block_sizes = [4, 8, 16, 32, 64, 128, 256]
    
    while time.monotonic() < deadline:
        # Pick a random block size
        bs = block_sizes[rng.integers(len(block_sizes))]
        
        if bs > n:
            continue
        
        # Pick a random start position
        start = rng.integers(0, n - bs + 1)
        end = start + bs
        
        # Try a few perturbations
        for _ in range(5):
            a_try = best_a.copy()
            # Log-normal perturbation
            noise = rng.standard_normal(bs) * 0.1
            a_try[start:end] = a_try[start:end] * np.exp(noise)

            mc_try = _mc(a_try)
            if mc_try < best_mc:

                best_a = a_try.copy()

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
