    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Compute subgradient of max_{k in active_lags} conv(a,a)_k with respect to a."""
    n = len(a_full)
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # d(conv_k)/d(a[i]) = 2 * a[k-i] for i in [lo, hi]

        idx = np.arange(lo, hi + 1)
        partners = k - idx
        grad[lo:hi+1] += 2.0 * a_full[partners]
    # Normalize: average over active lags
    grad /= len(active_lags)
    return grad
