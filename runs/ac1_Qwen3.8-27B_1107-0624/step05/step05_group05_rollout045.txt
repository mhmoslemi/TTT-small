def _next_pow2(n):
    k = 1

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score without the sum^2 denominator (assumes sum=1)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _normalize(a):
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    return a / s


def _softmax_stable(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(n):
    rows_list = []
    cols_list = []
    vals_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
        vals_list.append(np.ones(len(j_range), dtype=np.float64))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = np.concatenate(vals_list)
    return sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(A_full, M, time_limit=30):
    """Solve LP: max sum(b) s.t. A_full * b <= M, b >= 0."""
    n = A_full.shape[1]
    c_obj = -np.ones(n)
    b_ub = np.full(A_full.shape[0], M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_full, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _mirror_descent_step(a, beta, eta, top_k=None):
    """Mirror descent with optional top-k peak targeting."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    if top_k is not None and top_k < len(c):
        # Focus on top-k peaks
        idx = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros_like(c)
        c_top = c[idx]
        w[idx] = np.exp((c_top - c_top.max()) / beta)
        w /= w.sum()
    else:
        c_shifted = (c - c_max) / max(beta, 1e-12)
        w = np.exp(c_shifted)
        s = w.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / s
    
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    
    # Multiplicative (mirror) update
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    z -= z.mean()
    return _softmax_stable(z), c_max


def _peak_shaving(a, num_peaks=5, eta=0.1, iterations=50):
    """Target the top peaks specifically and reduce contributions."""
    n = len(a)
    for _ in range(iterations):
        c = _full_convolve(a, a)
        # Find top peaks
        c_sorted_idx = np.argsort(c)[::-1][:num_peaks]
        c_max = c[c_sorted_idx[0]]
        
        # Compute gradient w.r.t. a for the top peaks only
        grad = np.zeros(n)
        for k in c_sorted_idx:
            # c[k] = sum_j a_j * a[k-j]
            # d(c[k])/d(a_i) = 2*a[k-i] if k-i in [0,n-1] and i != k-i
            #                  = a[i] if 2*i == k
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                partner = k - j
                if 0 <= partner < n:
                    if j == partner:
                        grad[j] += a[j]
                    else:
                        grad[j] += 2.0 * a[partner]
        
        # Scale gradient by peak value
        grad *= c_max
        
        # Soft update: reduce elements that contribute to peaks
        z = np.log(np.maximum(a, 1e-300))
        z -= eta * grad / (grad.max() + 1e-15)
        z -= z.mean()
        a = _softmax_stable(z)
    
    return a


def _adam_refine(a_init, deadline, lr_start=0.01, lr_end=0.002, beta_smoothing=0.005):
    """Adam optimizer on log-space with adaptive smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    prev_best = best_sc
    cooldown = 0
    
    while time.monotonic() < deadline:
        t += 1
        a = _softmax_stable(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive smoothing: tighter as we get closer to optimum
        progress = min(1.0, t / 50000.0)
        beta = beta_smoothing * c_max * (1.0 - 0.8 * progress)
        beta = max(beta, 1e-6 * c_max)
        
        c_shifted = (c - c_max) / beta
        e = np.exp(c_shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        
        # Center the gradient (to maintain sum=1 constraint in log space)
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        lr = lr_start + (lr_end - lr_start) * min(1.0, t / 50000.0)
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Gradient clipping
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        
        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()
        
        # Track best
        if t % 100 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                cooldown = 0
            else:
                cooldown += 1
            
            # Learning rate adjustment
            if cooldown > 200:
                lr = max(lr_end, lr * 0.9)
                cooldown = 0
    
    a_final = _softmax_stable(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _coordinate_descent(a, max_sweeps=5, deadline=None):
    """Coordinate descent: try small adjustments to each element."""
    n = len(a)
    current_sc = _score_norm(a)
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Shuffle order for diversity
        order = np.random.permutation(n)
        for i in order:
            if deadline is not None and time.monotonic() > deadline and np.random.random() < 0.5:
                    break
            # Try multiplying a[i] by factors
            best_mult = 1.0
            best_local_sc = current_sc
            for mult in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_trial = a.copy()
                a_trial[i] *= mult
                sc = _score_norm(a_trial)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_mult = mult
            if best_mult != 1.0:
                a[i] *= best_mult
                current_sc = best_local_sc
                improved = True
    
    return a, current_sc


def propose_candidate(seed=42, budget_s=1800):

    Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses mirror descent, LP, peak shaving, Adam refinement, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()

            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent loaded: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")
    
    # Pre-build LP matrix (this is the static constraint matrix)
    print(f"Building LP matrix...")
    t0 = time.monotonic()
    A_lp = _build_lp_matrix(n)
    print(f"  Done in {time.monotonic()-t0:.1f}s, nnz={A_lp.nnz}")
    
    # === Phase 1: Aggressive Mirror Descent (0 - 200s) ===
    phase1_end = min(deadline, t_start + 200.0)
    print(f"\n=== Phase 1: Aggressive Mirror Descent ===")
    
    # Use progressively smaller beta (from wide to narrow focus)
    c_init = _full_convolve(a, a)
    c_max_init = float(c_init.max())
    
    beta_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    eta_schedule = [0.15, 0.12, 0.10, 0.08, 0.06, 0.05]
    
    for bi, (beta_frac, eta) in enumerate(zip(beta_schedule, eta_schedule)):
        phase_time = (phase1_end - t_start) / (len(beta_schedule) - bi)
        target_time = t_start + (200.0 * (bi + 1) / len(beta_schedule))
        
        max_iters = 2000
        for it in range(max_iters):
            if time.monotonic() > target_time or time.monotonic() > deadline:
                break
            beta = beta_frac * c_max_init
            a, c_max = _mirror_descent_step(a, beta, eta)
            c_max_init = c_max  # Adapt to current scale
        
        sc = _score(a)
        print(f"  MD phase {bi+1}/6 (beta={beta_frac}): score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
    
    # === Phase 2: LP-Based Steps (200 - 500s) ===
    phase2_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 2: LP Steps ===")
    
    a = best_seq.copy()
    lp_count = 0
    stall = 0

    while time.monotonic() < phase2_end:

        g_0 = _lp_solve(A_lp, M, time_limit=lp_time_limit)
