def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):

    s = float(a.sum())

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

"""
LP-iterated minimax with line search, running average, truncated LP,
and L-BFGS polish. Targets sequences of length ~2048-8192.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_structure(n):
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        count = i_hi - i_lo + 1
        rows_parts.append(np.full(count, k, dtype=np.int32))
        cols_parts.append(np.arange(i_lo, i_hi + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    return rows, cols


def _make_conv_matrix(f, rows, cols):
    n = len(f)
    vals = f[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve_full(f, rows, cols, max_time=30):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _make_conv_matrix(f, rows, cols)
    constraints = [A @ g <= t, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None
    if time.monotonic() - t0 > max_time:
        return None, None
    if g.value is not None and t.value is not None:
        g_val = np.maximum(g.value, 0)
        s = g_val.sum()
        if s > 1e-15:
            g_val = g_val / s
            return g_val, float(t.value)
    return None, None


def _lp_solve_truncated(f, top_k, max_time=15):
    """Solve LP with only the top_k most critical constraints."""
    n = len(f)
    c = _full_convolve(f, f)
    # Find the top_k lags with highest convolution values
    top_lags = np.argsort(c)[::-1][:top_k]
    
    # Build reduced matrix
    rows_r = []
    cols_r = []
    for k in top_lags:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        count = i_hi - i_lo + 1
        rows_r.append(np.full(count, k, dtype=np.int32))
        cols_r.append(np.arange(i_lo, i_hi + 1, dtype=np.int32))
    rows_rf = np.concatenate(rows_r)
    cols_rf = np.concatenate(cols_r)
    
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    A_r = _make_conv_matrix(f, rows_rf, cols_rf)
    constraints = [A_r @ g_var <= t_var, cp.sum(g_var) == 1, g_var >= 0]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    t0 = time.monotonic()
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        return None, None
    if time.monotonic() - t0 > max_time:
        return None, None
    if g_var.value is not None and t_var.value is not None:
        g_val = np.maximum(g_var.value, 0)
        s = g_val.sum()
        if s > 1e-15:
            g_val = g_val / s
            return g_val, float(t_var.value)
    return None, None


def _line_search(f, g, n_trials=15):
    n = len(f)
    best_t = 0.3
    best_score = _score_norm_fn(f)
    for t in np.linspace(0.05, 1.0, n_trials):
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _interp_profile(p0, n_new):

"""
LP-iterated minimax with line search, running average, truncated LP for speed,
spectral IFTA-inspired refinement, and L-BFGS/coordinate-descent polish.
Key differences from previous: proper line search per iteration, running average
tracking, truncated LP after initial phase, and a spectral reshaping step.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_structure(n):
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        count = i_hi - i_lo + 1
        rows_parts.append(np.full(count, k, dtype=np.int32))
        cols_parts.append(np.arange(i_lo, i_hi + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    return rows, cols


def _make_conv_matrix_from_structure(f, rows, cols):
    n = len(f)
    vals = f[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _make_conv_matrix_truncated(f, top_lags):
    n = len(f)
    rows_parts = []
    cols_parts = []
    for k in top_lags:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        count = i_hi - i_lo + 1
        rows_parts.append(np.full(count, k, dtype=np.int32))
        cols_parts.append(np.arange(i_lo, i_hi + 1, dtype=np.int32))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    vals = f[rows - cols]
    return csr_matrix((vals, (rows, cols)), shape=(len(top_lags), n))


def _lp_solve_full(f, rows, cols):
    n = len(f)
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    A = _make_conv_matrix_from_structure(f, rows, cols)
    constraints = [A @ g_var <= t_var, cp.sum(g_var) == 1, g_var >= 0]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None, None
    if g_var.value is not None and t_var.value is not None:
        g_val = np.maximum(g_var.value, 0)
        s = g_val.sum()
        if s > 1e-15:
            g_val = g_val / s
            return g_val, float(t_var.value)
    return None, None


def _lp_solve_truncated(f, top_k):
    n = len(f)
    c = _full_convolve(f, f)
    top_lags = np.argsort(c)[::-1][:top_k]
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    A_r = _make_conv_matrix_truncated(f, top_lags)
    constraints = [A_r @ g_var <= t_var, cp.sum(g_var) == 1, g_var >= 0]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except Exception:
            return None, None
    if g_var.value is not None and t_var.value is not None:
        g_val = np.maximum(g_var.value, 0)
        s = g_val.sum()
        if s > 1e-15:
            g_val = g_val / s
            return g_val, float(t_var.value)
    return None, None


def _line_search(f, g, n_trials=12):
    best_t = 0.3
    best_score = _score_norm_fn(f)
    for t in np.linspace(0.05, 1.0, n_trials):
        f_new = (1 - t) * f + t * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_t = t
    return best_t

def _spectral_reshaping(a, n_iters=200, deadline=None):

        cm = c.max()
        peak_idx = np.argmax(c)

        c_mod = c.copy()
        reduction = 0.9
        c_mod[peak_idx] *= reduction

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_refine(a_init, deadline, max_betas=12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    fracs = [0.04, 0.025, 0.015, 0.01, 0.008, 0.006, 0.004,
             0.003, 0.002, 0.001, 0.0006, 0.0003, 0.0002, 0.0001]
    beta_list = [b * max_c0 for b in fracs[:max_betas]]

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = max(1, max_betas - idx)
        time_per = min((deadline - time.monotonic()) / remaining, 50.0)
        level_dl = time.monotonic() + max(5.0, time_per)
        maxiter = max(50, int(time_per * 15))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxls': 40, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                    print(f"      L-BFGS beta_idx={idx} beta={beta:.2e}: "
                          f"score={s_new:.10f}")
        except Exception:
            pass

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=12):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            cur_score = _score_norm_fn(a)
            for delta in [-0.04, -0.02, -0.01, 0.01, 0.02, 0.04]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < cur_score:
                    cur_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                if _score_norm_fn(a) < best_score:
                    best_score = _score_norm_fn(a)
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.3 + u ** 2)
    profiles.append(a / a.sum())
    a = 1.0 / (1.0 + 20.0 * u ** 2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 80.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score_val = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score_val
        if p is not None and np.isfinite(sc) and sc < best_score_val:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score_val = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score_val:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
                sc_p = _score_norm_fn(parent)
                print(f"Parent: n={len(parent)}, norm_score={sc_p:.10f}")
                update_best(parent, sc_p, "parent")
    except Exception:
        parent = None

    if parent is None:
        n_init = 2048
        parent = _make_profiles(n_init)[3]
        sc_p = _score_norm_fn(parent)
        print(f"Default: n={len(parent)}, norm_score={sc_p:.10f}")
        update_best(parent, sc_p, "default")

    n_parent = len(parent)

    # Determine which n values to try
    n_options = []
    if 1024 <= n_parent <= 8192:
        n_options.append(n_parent)
    for n_try in [2048, 4096]:
        if n_try not in n_options and 1024 <= n_try <= 8192:
            n_options.append(n_try)
    n_options = sorted(set(n_options))

    print(f"\nWill try n values: {n_options}")
    time_per_n = (deadline - time.monotonic()) / max(1, len(n_options))
    time_per_n = min(time_per_n, 700.0)

    for n_idx, n_try in enumerate(n_options):
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        alloc = min(time_per_n, remaining)
        if n_try == len(parent):
            a_start = parent.copy()
        else:
            a_start = _interp_profile(parent, n_try)

        sc_start = _score_norm_fn(a_start)
        update_best(a_start, sc_start, f"start_n{n_try}")
        print(f"\n=== n={n_try}: initial score={sc_start:.10f}, alloc={alloc:.0f}s ===")

        # Phase A: Full LP iterations
        full_lp_budget = alloc * 0.4
        full_lp_end = time.monotonic() + full_lp_budget
        print(f"  Phase A: Full LP ({full_lp_budget:.0f}s)")
        a_lp, a_avg, sc_lp, n_it = _lp_iteration_loop(
            a_start, full_lp_end, full_lp_iters=15, top_k=200)
        update_best(a_lp, sc_lp, f"lp_full_n{n_try}")
        if a_avg is not None and n_it > 10:
            sc_avg = _score_norm_fn(a_avg)
            update_best(a_avg, sc_avg, f"lp_avg_n{n_try}")

        # Phase B: Truncated LP
        trunc_budget = alloc * 0.3
        trunc_end = time.monotonic() + trunc_budget
        if time.monotonic() < deadline - 60 and a_lp is not None:
            print(f"  Phase B: Truncated LP ({trunc_budget:.0f}s)")
            a_tr, a_tr_avg, sc_tr, n_it2 = _lp_iteration_loop(
                a_lp, trunc_end, full_lp_iters=0, top_k=150)
            update_best(a_tr, sc_tr, f"lp_trunc_n{n_try}")

        # Phase C: Spectral reshaping
        spec_end = min(time.monotonic() + alloc * 0.1, deadline - 40)
        if time.monotonic() < spec_end and a_tr is not None:
            print(f"  Phase C: Spectral reshaping")
            a_spec, sc_spec = _spectral_reshaping(a_tr, n_iters=300, deadline=spec_end)
            update_best(a_spec, sc_spec, f"spectral_n{n_try}")

        # Phase D: L-BFGS refinement
        lbfgs_end = min(time.monotonic() + alloc * 0.2, deadline - 30)
        if time.monotonic() < lbfgs_end and a_tr is not None:
            print(f"  Phase D: L-BFGS refinement")
            a_ref, sc_ref = _lbfgs_refine(a_tr, lbfgs_end, max_betas=10)
            update_best(a_ref, sc_ref, f"lbfgs_n{n_try}")

    # Phase E: Coordinate descent polish on overall best
    cd_end = min(time.monotonic() + 150.0, deadline - 30)
    if time.monotonic() < cd_end and best_seq is not None:
        print(f"\nPhase E: Coordinate descent (n={len(best_seq)})")
        a_cd, sc_cd = _coord_descent(best_seq, cd_end, max_rounds=15)
        update_best(a_cd, sc_cd, "coord_descent")

    # Phase F: Final ultra-fine L-BFGS
    final_end = deadline
    if time.monotonic() < final_end - 10 and best_seq is not None:
        print(f"\nPhase F: Final L-BFGS polish")
        a_fin, sc_fin = _lbfgs_refine(best_seq, final_end, max_betas=6)
        update_best(a_fin, sc_fin, "final_lbfgs")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirms: {sc_eval:.10f}")
            except Exception:
                pass
        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
