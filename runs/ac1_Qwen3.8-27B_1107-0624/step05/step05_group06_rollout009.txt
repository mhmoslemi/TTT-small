LP-based active set optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions that respect
the current max-convolution constraint, combined with L-BFGS polishing.
"""
import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _conv_matrix(f, n):
    """Build the (2n-1) x n convolution matrix C where (C @ b)[k] = (f*b)[k]."""
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f[j]
    return C

def _lp_direction(f, M, n):
    """Solve LP: max sum(b) s.t. b >= 0, f*b <= M."""
    b_var = cp.Variable(n, nonneg=True)
    C = _conv_matrix(f, n)
    constraints = [C @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            prob.solve(solver=cp.GLPK, verbose=False)

        return f.copy()  # fallback
    b_val = np.asarray(b_var.value).ravel()
    b_val = np.clip(b_val, 0, None)
    if b_val.sum() < 1e-15:

    return b_val

    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):

    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):

    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta

    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        return _smoobj(z, beta, n)

        return _smograd(z, beta, n)

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    """Build the (2n-1) x n matrix C where (C @ b)[k] = (f*b)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f[j]
    return C

def _lp_solve_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, (f*b)(k) <= M for all k."""
    n = len(f)
    b_var = cp.Variable(n, nonneg=True)

    return 2.0 * n * (c_max + beta * np.log(s))

    c = _full_convolve(a, a)
    c_max = c.max()

    Returns the optimal b vector."""
    n = len(f)
    C = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.clip(res.x, 0, None)
    except Exception as e:
        print(f"    LP error: {e}")
    return f.copy()


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_phase(a_init, beta_list, deadline, base_max_iter=2000):
    """Run L-BFGS with a list of decreasing betas (continuation)."""
    n = len(a_init)
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for beta in beta_list:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - 10 - time.monotonic()
        if remaining < 5:
            break

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        # Scale max_iter based on remaining time (~5ms per iter for n=2048)
        est_time_per_iter = 0.005 * (n / 2048.0)
        max_iter = min(base_max_iter, max(50, int(remaining / (est_time_per_iter * 10))))

        _beta = float(beta)
        _n = int(n)

        def fun(z):
            return _smoobj(z, _beta, _n)

        def jac(z):
            return _smograd(z, _beta, _n)

        try:
            result = minimize(
                fun, z0, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'ftol': 1e-15,
                    'gtol': 1e-12,
                    'maxcor': 40,
                    'maxls': 50,
                    'disp': False,
                }
            )
            a_new = _softmax(result.x)
            sc = _score(a_new)
            if sc < current_sc - 1e-14:
                current_a = a_new
                current_sc = sc
        except Exception as e:
            print(f"    L-BFGS error with beta={beta:.2e}: {e}")

    return current_a, current_sc


def _subgrad_polish(a_init, deadline, n_steps=3000, lr_init=0.04):
    """Subgradient method with exact active set in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set: lags within 0.05% of max
        tol = 0.0005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc = _full_convove(w, a_rev) if False else _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        step = lr / (1.0 + i * 0.001)
        z = z - step * grad
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if i % 200 == 0:
            sc = _score(_softmax(z))
            if sc < best_sc:
                best_sc = sc
                best_a = _softmax(z).copy()
            else:
                lr *= 0.92

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _lp_optimize(f_init, deadline, max_iters=100):
    """LP-based optimization with adaptive line search."""
    n = len(f_init)
    f = f_init.copy()
    best_f = f_init.copy()
    best_sc = _score(f_init)
    stagnant = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        t0 = time.monotonic()
        c_ff = _full_convolve(f, f)
        M = float(c_ff.max())

        b = _lp_direction(f, M)
        s_b = float(b.sum())
        s_f = float(f.sum())
        if s_b < 1e-10:
            break
        g = b * (s_f / s_b)

        lp_time = time.monotonic() - t0

        # Line search: try multiple step sizes
        best_t = 0.0
        best_sc_ls = best_sc

        candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.7, 0.8, 0.4, 0.15, 0.08]
        for t in candidates:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            if f_new.sum() < 1e-10:
                continue
            sc = _score(f_new)
            if sc < best_sc_ls:
                best_sc_ls = sc
                best_t = t

        if best_t > 0:
            f = (1.0 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            sc = _score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                stagnant = 0
                if it % 5 == 0 or sc < 1.503:
                    print(f"  LP iter {it}: M={M:.8f}, t={best_t}, score={sc:.10f}, time={lp_time:.1f}s")
            else:
                stagnant += 1
        else:
            stagnant += 1
            if stagnant >= 5:
                break

        if stagnant >= 15:
            break

    return best_f, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = 0.015 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _stochastic_search(a_init, deadline, rng, n_steps=500):
    """Simulated-annealing-style stochastic local search."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    current_sc = best_sc
    T = 0.001  # initial "temperature"

    for i in range(n_steps):
        if time.monotonic() > deadline:
            break
        # Cool down
        T *= 0.995

        a = _softmax(z)
        # Random perturbation in log-space
        sigma = 0.01 * (1.0 + T * 100)
        z_new = z + sigma * rng.standard_normal(n)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)
        sc_new = _score(a_new)

        # Metropolis acceptance
        if sc_new <= current_sc or rng.random() < np.exp(-(sc_new - current_sc) / max(T, 1e-8)):
            z = z_new
            current_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                if best_sc < 1.503:
                    print(f"    Stochastic iter {i}: new best {best_sc:.10f}")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Hybrid LP + L-BFGS + stochastic optimizer for the step function problem.
    Phase 1: LP-based active set optimization (principled direction finding)
    Phase 2: L-BFGS polishing with log-sum-exp continuation
    Phase 3: Subgradient refinement with exact active set
    Phase 4: Stochastic search for escape from local minima
    Phase 5: Multi-restart diversification
    Phase 6: Final fine-grained polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  ** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={len(a0)}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "init")

    n = len(a0)

    # Phase 1: LP-based optimization (main contribution)
    print(f"\n=== Phase 1: LP-based optimization ===")
    lp_deadline = t_start + 500
    a_lp, sc_lp = _lp_optimize(a0, lp_deadline, max_iters=100)
    update_best(a_lp, sc_lp, "lp_opt")

    # Phase 2: L-BFGS polishing with continuation
    print(f"\n=== Phase 2: L-BFGS polishing ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        betas = mb * np.array([0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
        lbfgs_deadline = min(deadline - 200, time.monotonic() + 350)
        a_lb, sc_lb = _lbfgs_phase(best_seq, betas, lbfgs_deadline, base_max_iter=2000)
        update_best(a_lb, sc_lb, "lbfgs_polish")

    # Phase 3: Subgradient refinement
    print(f"\n=== Phase 3: Subgradient refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 100:
        sg_deadline = min(deadline - 80, time.monotonic() + 120)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, n_steps=3000, lr_init=0.03)
        update_best(a_sg, sc_sg, "subgrad")

    # Phase 4: Stochastic search
    print(f"\n=== Phase 4: Stochastic search ===")
    if best_seq is not None and time.monotonic() < deadline - 80:
        st_deadline = min(deadline - 60, time.monotonic() + 200)
        a_st, sc_st = _stochastic_search(best_seq, st_deadline, rng, n_steps=800)
        update_best(a_st, sc_st, "stochastic")

    # Phase 5: Multi-restart with diverse perturbations
    print(f"\n=== Phase 5: Multi-restart ===")
    if best_seq is not None and time.monotonic() < deadline - 60:
        restart_deadline = deadline - 40
        n_restarts = min(4, max(1, int((restart_deadline - time.monotonic()) / 120)))

        for i in range(n_restarts):
            if time.monotonic() > restart_deadline:
                break

            if i == 0:
                p_start = _perturb(best_seq, 0.04, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3)
            elif i == 2:
                p_start = _perturb(best_seq, 0.015, rng)
            else:
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)

            print(f"  Restart {i}: start_score={_score(p_start):.10f}")

            # Quick LP
            a_r, sc_r = _lp_optimize(p_start, restart_deadline, max_iters=25)
            update_best(a_r, sc_r, f"restart_lp_{i}")

            # L-BFGS polish
            c_r = _full_convolve(a_r, a_r)
            mr = float(c_r.max())
            betas_r = mr * np.array([0.005, 0.002, 0.001, 0.0005, 0.0002])
            a_r2, sc_r2 = _lbfgs_phase(a_r, betas_r, restart_deadline, base_max_iter=1000)
            update_best(a_r2, sc_r2, f"restart_lbfgs_{i}")

            # Quick subgradient
            if time.monotonic() < restart_deadline - 10:
                a_r3, sc_r3 = _subgrad_polish(a_r2, restart_deadline - 5, n_steps=500, lr_init=0.02)
                update_best(a_r3, sc_r3, f"restart_sg_{i}")

    # Phase 6: Final fine-grained polish
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_fin = _full_convolve(best_seq, best_seq)
        mf = float(c_fin.max())
        betas_fin = mf * np.array([0.00005, 0.00002, 0.00001])
        a_fin, sc_fin = _lbfgs_phase(best_seq, betas_fin, deadline - 3, base_max_iter=1500)
        update_best(a_fin, sc_fin, "final_lbfgs")

        if time.monotonic() < deadline - 3:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, n_steps=300, lr_init=0.008)
            update_best(a_fin2, sc_fin2, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
