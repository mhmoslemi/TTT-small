"""
Improved multi-scale optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements:
1. Parametric initialization with beta-distribution and monomial forms
2. Adaptive log-sum-exp continuation with more levels at small beta
3. Direct subgradient phase with Nesterov momentum at the end
4. Multiple restarts at final scale
5. Option to explore n=8192
6. Peak-targeted coordinate refinement

import numpy as np
from scipy.optimize import minimize, brent
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _make_beta_dist(n, alpha, beta_p):
    t = np.linspace(0.01, 0.99, n)
    a = t ** (alpha - 1) * (1 - t) ** (beta_p - 1)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_monomial(n, p):
    t = np.linspace(0, 1, n)
    a = t ** p
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_sigmoid(n, sharpness, center):
    t = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-sharpness * (t - center)))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _make_profiles(n):
    profiles = []
    # Monomials
    for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0]:
        profiles.append(_make_monomial(n, p))
    # Beta distributions
    for a1, b1 in [(0.5, 2.0), (0.5, 3.0), (1.0, 2.0), (1.0, 3.0), (1.5, 2.0), (2.0, 3.0), (0.5, 1.5), (1.0, 1.5)]:
        profiles.append(_make_beta_dist(n, a1, b1))
    # Sigmoid
    for s, c in [(5, 0.5), (10, 0.5), (15, 0.5), (8, 0.4), (12, 0.6)]:
        profiles.append(_make_sigmoid(n, s, c))
    return profiles


def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=20):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_c0 < 1e-20:
        return best_a, best_score

    # Adaptive continuation: more levels at small beta
    beta_start = 0.3 * max_c0
    beta_end = 1e-7 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break

        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget

        # More iterations for smaller beta
        if beta > 0.1 * max_c0:
            maxiter = 300
        elif beta > 0.01 * max_c0:
            maxiter = 1000
        else:
            maxiter = 3000

        maxfun = maxiter * 5

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -100, 100)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_adam_gd(a_init, deadline, rng, lr_init=0.05, momentum=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    v_elem = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())
    # Start with moderate beta, decay to near-zero
    beta = 0.1 * max_c0
    beta_min = 1e-7 * max_c0

    # Nesterov momentum for subgradient phase
    grad_prev = None
    momentum_vec = np.zeros(n)

    for it in range(500000):
        if time.monotonic() > deadline:
            break

        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break

        # Decay beta
        beta = max(beta_min, beta * 0.995)

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Nesterov: use grad at current + momentum position
        if grad_prev is not None and beta < 0.01 * max_c0:
            # Switch to subgradient + Nesterov
            momentum_vec = momentum * momentum_vec - lr * grad_z
            z = z + momentum_vec
        else:
            z = z - lr * mh / (np.sqrt(vh) + eps)

        z = np.clip(z, -100, 100)
        z = z - z.mean()

        a = _softmax_fn(z)

        # Evaluate every 10 steps for speed
        if it % 10 == 0 or it < 100:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Decay learning rate
        if t % 2000 == 0:
            lr = max(0.001, lr * 0.85)

        # Kick if stuck (check every 5000 steps)
        if t % 5000 == 0 and t > 5000:
            last_best_t = getattr(_run_adam_gd, '_last_best_t', 0)
            if t - last_best_t > 5000:
                z = z + 0.02 * rng.normal(size=n)
                z = z - z.mean()
                _run_adam_gd._last_best_t = t
            else:
                _run_adam_gd._last_best_t = t

    return best_a, best_score


def _subgradient_nesterov(a_init, deadline, rng, n_steps=50000, lr_init=0.01):
    """Direct subgradient descent with Nesterov momentum on the true max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    momentum_vec = np.zeros(n)
    lr = lr_init
    c = _full_convolve(a_init, a_init)
    max_c0 = float(c.max())

    # Use a very small effective beta (essentially true max with small smoothing)
    beta = 1e-4 * max_c0

    for it in range(n_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        if cm < 1e-30:
            break

        # Compute subgradient (with tiny smoothing for numerical stability)
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        # Nesterov momentum
        momentum_vec = 0.9 * momentum_vec - lr * grad_z
        z = z + momentum_vec
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        # Evaluate every 5 steps
        if it % 5 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        # Decay LR
        if (it + 1) % 1000 == 0:
            lr = max(0.0001, lr * 0.9)

    return best_a, best_score


def _peak_coordinate_descent(a_init, deadline, rng, max_rounds=200):
    """Targeted coordinate descent on the top peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    no_improve = 0

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top 5 peaks
        top_k = min(5, 2 * n - 1)
        peak_indices = np.argsort(c)[::-1][:top_k]

        improved = False
        for pk in peak_indices[:3]:
            # Find top contributors to this peak
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:6]

            for idx_pos in order[:4]:
                idx = valid_i[idx_pos]
                for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try increasing small values
            small_idx = np.argsort(a_cur)[:4]
            for idx in small_idx:
                for delta in [0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.12f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.12f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_monomial(256, 2.0)
        update_best(a0, _score_norm_fn(a0), "default")
        best_seq = a0
        best_score = _score_norm_fn(a0)

    # --- Phase 0: Parametric initialization (quick) ---
    print("\nPhase 0: Parametric initialization")
    phase0_end = t_start + 10.0
    n_param = 512
    best_param_a = None
    best_param_sc = np.inf

    profiles = _make_profiles(n_param)
    for idx, prof in enumerate(profiles):
        sc = _score_norm_fn(prof)
        if sc < best_param_sc:
            best_param_sc = sc
            best_param_a = prof.copy()
        if time.monotonic() > phase0_end:
            break

    if best_param_a is not None:
        print(f"  Best parametric: score={best_param_sc:.12f}")

        # If parametric is better than parent at this scale, use it for multi-scale
        # But we need to compare at the same scale
        if parent is not None:
            parent_512 = _upsample(parent, n_param)
            sc_parent_512 = _score_norm_fn(parent_512)
            if best_param_sc < sc_parent_512:
                print(f"  Parametric better than parent at n={n_param}")
            else:
                best_param_a = parent_512
                best_param_sc = sc_parent_512
    else:
        best_param_a = _make_monomial(n_param, 2.0)
        best_param_sc = _score_norm_fn(best_param_a)

    # --- Phase 1: Multi-scale optimization (55% of time) ---
    phase1_end = t_start + (deadline - t_start) * 0.55
    print(f"\nPhase 1: Multi-scale optimization (deadline t={phase1_end - t_start:.0f}s)")

    a_cur = best_param_a.copy()
    sc_cur = best_param_sc

    # Quick L-BFGS at n=512
    quick_deadline = min(phase1_end, t_start + 15.0)
    a_cur, sc_cur = _run_lbfgs_continuation(a_cur, quick_deadline, rng, n_betas=10)
    print(f"  After n=512: score={sc_cur:.12f}")

    # Multi-scale up
    if parent is not None and len(parent) >= 4096:
        # Skip intermediate scales, go straight to 4096
        scales = [1024, 2048, 4096]
        time_alloc = {1024: 30, 2048: 80, 4096: 250}
    else:
        scales = [256, 512, 1024, 2048, 4096]
        time_alloc = {256: 20, 512: 30, 1024: 60, 2048: 120, 4096: 300}

    for target_n in scales:
        if time.monotonic() > phase1_end - 15:
            break
        if target_n <= len(a_cur):
            continue

        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)
        print(f"  Upsampled to n={target_n}: score={sc_target:.12f}")

        alloc = time_alloc.get(target_n, 60)
        # Cap total phase1 time
        remaining = phase1_end - 15 - time.monotonic()
        alloc = min(alloc, remaining)
        if alloc < 10:
            break

        opt_deadline = time.monotonic() + alloc
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_deadline, rng, n_betas=15)
        print(f"  After n={target_n}: score={sc_opt:.12f}")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")

    # Also try from parent if we have one
    if parent is not None and len(parent) == 4096:
        if time.monotonic() < phase1_end - 20:
            alloc = min(60.0, phase1_end - 20 - time.monotonic())
            opt_dl = time.monotonic() + alloc
            a_p, sc_p = _run_lbfgs_continuation(parent, opt_dl, rng, n_betas=15)
            update_best(a_p, sc_p, "parent_lbfgs")

    print(f"\n  Phase 1 done: best={best_score:.12f}")

    # --- Phase 2: Intensive L-BFGS at final scale (20% of time) ---
    phase2_end = t_start + (deadline - t_start) * 0.75
    print(f"\nPhase 2: Intensive L-BFGS at final scale")

    if time.monotonic() < phase2_end - 10 and best_seq is not None:
        n_best = len(best_seq)

        # Run with many beta levels
        alloc = min(phase2_end - 10 - time.monotonic(), 200.0)
        opt_dl = time.monotonic() + alloc
        a_p2, sc_p2 = _run_lbfgs_continuation(best_seq, opt_dl, rng, n_betas=30)
        update_best(a_p2, sc_p2, "phase2_fine")

        # Try a perturbed restart
        if time.monotonic() < phase2_end - 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise = 0.05 * rng.normal(size=n_best)
            z = z + noise
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a_pert = _softmax_fn(z)
            alloc2 = min(60.0, phase2_end - 10 - time.monotonic())
            opt_dl2 = time.monotonic() + alloc2
            a_p2b, sc_p2b = _run_lbfgs_continuation(a_pert, opt_dl2, rng, n_betas=15)
            update_best(a_p2b, sc_p2b, "phase2_restart")

    print(f"\n  Phase 2 done: best={best_score:.12f}")

    # --- Phase 3: Adam GD + Subgradient (15% of time) ---
    phase3_end = t_start + (deadline - t_start) * 0.90
    print(f"\nPhase 3: Adam + Subgradient")

    if time.monotonic() < phase3_end - 15 and best_seq is not None:
        # Adam
        alloc = min(60.0, phase3_end - 30 - time.monotonic())
        if alloc > 10:
            gd_dl = time.monotonic() + alloc
            a_gd, sc_gd = _run_adam_gd(best_seq, gd_dl, rng, lr_init=0.03)
            update_best(a_gd, sc_gd, "adam")

        # Subgradient with Nesterov
        if time.monotonic() < phase3_end - 10:
            alloc2 = min(40.0, phase3_end - 10 - time.monotonic())
            if alloc2 > 5:
                sg_dl = time.monotonic() + alloc2
                a_sg, sc_sg = _subgradient_nesterov(best_seq, sg_dl, rng, n_steps=30000, lr_init=0.005)
                update_best(a_sg, sc_sg, "subgrad")

    print(f"\n  Phase 3 done: best={best_score:.12f}")

    # --- Phase 4: Peak coordinate descent (5% of time) ---
    phase4_end = t_start + (deadline - t_start) * 0.95
    print(f"\nPhase 4: Peak coordinate descent")

    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        ps_dl = phase4_end - 10
        a_ps, sc_ps = _peak_coordinate_descent(best_seq, ps_dl, rng, max_rounds=500)
        update_best(a_ps, sc_ps, "peak_cd")

    print(f"\n  Phase 4 done: best={best_score:.12f}")

    # --- Phase 5: Try n=8192 if time permits ---
    print(f"\nPhase 5: Try n=8192")
    if time.monotonic() < deadline - 60 and best_seq is not None:
        n_curr = len(best_seq)
        if n_curr < 8192:
            a_8192 = _upsample(best_seq, 8192)
            sc_8192 = _score_norm_fn(a_8192)
            print(f"  Upsampled to 8192: score={sc_8192:.12f}")

            alloc = min(80.0, deadline - 60 - time.monotonic())
            if alloc > 20:
                opt_dl = time.monotonic() + alloc
                a_8192_opt, sc_8192_opt = _run_lbfgs_continuation(a_8192, opt_dl, rng, n_betas=15)
                update_best(a_8192_opt, sc_8192_opt, "n8192")

    print(f"\n  Phase 5 done: best={best_score:.12f}")

    # --- Phase 6: Final polish ---
    print(f"\nPhase 6: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        # Tiny beta L-BFGS
        beta_tiny = 1e-6 * max_c_final
        alloc = min(30.0, deadline - 5 - time.monotonic())
        if alloc > 5:
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta_tiny, n_final),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': 10000, 'ftol': 1e-20, 'gtol': 1e-14,
                             'maxls': 100, 'maxfun': 30000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_seq = a_new.copy()
                        print(f"  Final L-BFGS: {best_score:.12f}")
            except Exception as ex:
                print(f"  Final L-BFGS failed: {ex}")

        # Quick peak CD
        if time.monotonic() < deadline - 3:
            a_ps, sc_ps = _peak_coordinate_descent(best_seq, deadline - 3, rng, max_rounds=100)
            update_best(a_ps, sc_ps, "final_peak_cd")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_monomial(256, 2.0)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.12f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_monomial(256, 2.0)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.12f}")
    return [float(x) for x in fallback]
