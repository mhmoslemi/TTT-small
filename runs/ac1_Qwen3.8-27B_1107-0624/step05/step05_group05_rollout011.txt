"""
Proximal gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient in log-space with multi-peak handling, adaptive step sizes,
and multi-length exploration.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_subgrad_logspace(a, c, top_k=5):
    """
    Compute the subgradient of the score in log-space.
    Returns grad_z where z = log(a), so a = exp(z).
    
    The score (with S=sum(a)) is: 2*n*max(c) / S^2
    The subgradient w.r.t. z_i is:
      (2*n / S^2) * sum_j w_j * [2*a_i*a_{k_j - i}] - (4*n*max_c / S^3) * a_i * S
    = (4*n*a_i / S^2) * [sum_j w_j * a_{k_j - i} - max_c/S]
    
    where the sum is over top_k peaks with weights w_j.
    """
    n = len(a)
    S = float(a.sum())
    c_max = float(c.max())
    
    # Find top peaks
    # c has length 2*n-1
    idx = np.argsort(c)[::-1][:top_k]
    # Get weights (proportional to how close to max)
    peak_vals = c[idx]
    # Use exponential weighting based on proximity to max
    weights = np.exp((peak_vals - c_max) / (0.01 * c_max + 1e-15))
    weights /= weights.sum()
    
    # grad_z[i] = 4*n*a_i/S^2 * [sum_j w_j * a_{k_j - i} - c_max/S]
    # The term a_{k_j - i} means: for peak at position k_j, the "partner" of index i is k_j - i
    # (with a_j = 0 for j < 0 or j >= n)
    
    weighted_partner = np.zeros(n)
    for j in range(len(idx)):
        k = idx[j]
        w = weights[j]
        # For each i in [0, n-1], partner is k - i
        # Valid when 0 <= k - i < n, i.e., max(0, k-n+1) <= i <= min(k, n-1)
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            partners = np.arange(i_start, i_end + 1)
            vals = a[k - partners]  # a[k-i_start], a[k-i_start-1], ..., a[k-i_end]
            weighted_partner[partners] += w * vals
    
    avg_partner = float(c_max / S)
    grad_z = (4.0 * n / (S * S)) * a * (weighted_partner - avg_partner)
    
    return grad_z

def _lp_step_enhanced(f, top_k_budget=10):

        rows_list.append(np.full(len(j_range), k, dtype=np.int32))

    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30})
        if res.status != 0:
            return f, M
        g_0 = np.maximum(res.x, 0.0)
    except Exception:
        return f, M
    
    if g_0.sum() < 1e-15:
        return f, M
    
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    
    if best_t > 0.0:

def _cosine_init(n, rng):
    """Initialize with a cosine-modulated sequence."""
    x = np.arange(n)
    # Sum of cosines at different frequencies
    a = np.ones(n)
    freqs = [0.1, 0.23, 0.37]
    amps = [0.3, 0.2, 0.1]
    for f, amp in zip(freqs, amps):
        a += amp * np.cos(2 * np.pi * f * x / n + rng.uniform(0, 2*np.pi))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _geometric_init(n, rng):
    """Initialize with geometric-like decay modulated by noise."""
    x = np.arange(n)
    a = np.exp(-0.1 * x / n) * (1 + 0.1 * rng.standard_normal(n))
    a = np.maximum(a, 0.01)
    return a / a.sum()
