import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Score with sum(a)=1: 2n * max(convolve(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _compute_topk_gradient(a, K):
    """Compute subgradient of max_k c(k) (approximated by top-K average).
    Returns gradient w.r.t. a (not z)."""
    n = len(a)
    c = _full_convolve(a, a)
    K = min(K, len(c))
    top_idx = np.argsort(c)[-K:]
    
    grad = np.zeros(n)
    for k_val in top_idx:
        k_val = int(k_val)
        # dc(k)/da[i] = 2*a[k-i] if 0 <= k-i < n
        # (the factor of 2 accounts for both a[i]*a[k-i] and a[k-i]*a[i])
        # But when k-i == i, i.e., k = 2i, it's just a[i]
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        for i in range(lo, hi + 1):
            partner = k_val - i
            if 0 <= partner < n:
                if i == partner:
                    grad[i] += a[i]
                else:
                    grad[i] += 2.0 * a[partner]
    grad /= K
    return grad, c.max()

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 using:
1. Top-K subgradient descent with backtracking line search and momentum in log-softmax space
2. Adaptive K schedule (start smooth, end sharp)
3. Peak-shaving direct manipulation
4. Local search with informed moves
5. Multi-restart for escaping local minima
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_subgrad_a(a, K):
    """Compute subgradient of max_k (a*a)(k) approximated by top-K average.
    Returns (gradient w.r.t. a, current max value)."""
    n = len(a)
    c = _full_convolve(a, a)
    K = min(K, len(c))
    top_idx = np.argsort(c)[-K:]
    grad = np.zeros(n)
    for k_val in top_idx:
        k_val = int(k_val)
        lo = max(0, k_val - (n - 1))
        hi = min(n - 1, k_val)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            partners = k_val - indices
            grad[indices] += 2.0 * a[partners]
            if k_val % 2 == 0:
                mid = k_val // 2
                if lo <= mid <= hi:
                    grad[mid] -= a[mid]
    grad /= K
    return grad, float(c.max())


def _softmax_grad(z, K):
    """Compute softmax-space gradient of top-K smoothed max."""
    a = _softmax_fn(z)
    grad_a, max_c = _topk_subgrad_a(a, K)
    # Convert to z-space: h[i] = a[i] * (grad_a[i] - mean)
    mean_grad = float(np.dot(a, grad_a))
    h = a * (grad_a - mean_grad)
    if not np.all(np.isfinite(h)):
        h = np.zeros_like(z)
    return h, a, max_c


def _backtrack_step(z, h, a_curr, score_curr, max_lr=2.0, min_lr=1e-10, max_iter=30):
    """Backtracking line search in z-space. Returns (new_z, new_score, accepted)."""
    n = len(z)
    h_norm = np.linalg.norm(h)
    if h_norm < 1e-15:
        return z, score_curr, False
    h = h / h_norm
    direction = -h
    lr = max_lr
    for _ in range(max_iter):
        z_new = z + lr * direction
        z_new = np.clip(z_new, -80, 80)
        a_new = _softmax_fn(z_new)
        sc_new = _score_norm_fn(a_new)
        if np.isfinite(sc_new) and sc_new < score_curr:
            return z_new, sc_new, True
        lr *= 0.5
        if lr < min_lr:
            break
    return z, score_curr, False


def _gradient_optimize(a_init, deadline, rng, max_lr=1.0, K_start=20, K_end=1,
                       momentum=0.9, phase_name="grad", min_improve=1e-12):
    """Main gradient optimization loop with adaptive K and backtracking."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z = np.clip(z, -60, 60)
    
    a_curr = _softmax_fn(z)
    score_curr = _score_norm_fn(a_curr)
    best_a = a_curr.copy()
    best_score = score_curr
    v = np.zeros(n)  # momentum
    
    total_iters = 0
    stagnant = 0
    epoch = 0
    # Epoch-based: each epoch has a target number of iterations
    epoch_len = 2000
    
    while time.monotonic() < deadline - 5.0:
        for _ in range(epoch_len):
            if time.monotonic() > deadline - 5.0:
                break
            
            total_iters += 1
            epoch += 1
            
            # Adaptive K
            progress = min(1.0, total_iters / 200000.0)
            K = max(K_end, int(K_start * (1.0 - progress) + K_end * progress))
            
            h, a_curr, max_c = _softmax_grad(z, K)
            if np.linalg.norm(h) < 1e-15:
                break
            
            # Momentum
            v = momentum * v + (1.0 - momentum) * h
            v_norm = np.linalg.norm(v)
            if v_norm > 0:
                v = v / v_norm
            
            # Backtracking
            z_new, sc_new, accepted = _backtrack_step(z, v, a_curr, score_curr,
                                                       max_lr=max_lr, min_lr=1e-10, max_iter=20)
            if accepted:
                # Check if improvement is significant
                if score_curr - sc_new < min_improve:
                    stagnant += 1
                else:
                    stagnant = 0
                
                z = z_new
                score_curr = sc_new
                
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = _softmax_fn(z).copy()
                    print(f"  [{phase_name}] iter={total_iters} K={K} score={best_score:.10f}")
                
                # Periodic centering
                if total_iters % 500 == 0:
                    z = z - z.mean()
                    z = np.clip(z, -60, 60)
            else:
                stagnant += 1
                # Reset momentum occasionally
                if stagnant % 10 == 0:
                    v = np.zeros(n)
                
                # Try smaller max_lr
                if stagnant > 50:
                    max_lr = max(0.01, max_lr * 0.8)
                    stagnant = 0
            
            # Convergence check
            if stagnant > 200:
                break
        
        # Between epochs: decay momentum, reduce lr
        momentum = max(0.5, momentum * 0.995)
        if best_score - score_curr < 1e-10 and total_iters > 10000:
            # Very close to convergence, try K=1 for a bit
            K = 1
            h, a_curr, _ = _softmax_grad(z, 1)
            if np.linalg.norm(h) < 1e-12:
                break
    
    return best_a, best_score


def _peak_shave(a_init, deadline, rng, n_trials=5000):
    """Directly reduce elements contributing to the top peak."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break
        
        c = _full_convolve(a, a)
        max_k = int(np.argmax(c))
        max_val = c[max_k]
        
        # Find the pair with largest product at max_k
        lo = max(0, max_k - (n - 1))
        hi = min(n - 1, max_k)
        if lo > hi:
            continue
        
        pairs = np.arange(lo, hi + 1)
        partners = max_k - pairs
        products = a[pairs] * a[partners]
        
        # Find the element with largest contribution across all top peaks
        top_lags = np.argsort(c)[-3:]
        contrib = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            lo2 = max(0, kl - (n - 1))
            hi2 = min(n - 1, kl)
            if lo2 <= hi2:
                p2 = np.arange(lo2, hi2 + 1)
                part2 = kl - p2
                contrib[p2] += a[part2]
        
        # Strategy 1: Reduce the most contributing element, add to least contributing
        if it % 2 == 0:
            idx_reduce = int(np.argmax(contrib))
            # Find safe position (low contribution)
            safe_candidates = np.argsort(contrib)[:max(1, n // 4)]
            idx_add = int(safe_candidates[rng.integers(len(safe_candidates))])
            
            if idx_reduce == idx_add:
                continue
            
            a_trial = a.copy()
            frac = rng.uniform(0.3, 0.7)
            delta = a_trial[idx_reduce] * frac
            a_trial[idx_reduce] -= delta
            a_trial[idx_add] += delta
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        # Strategy 2: Reduce top product pair
        else:
            idx = pairs[int(np.argmax(products))]
            a_trial = a.copy()
            frac = rng.uniform(0.2, 0.5)
            delta = a_trial[idx] * frac
            a_trial[idx] -= delta
            
            # Distribute to random low-contribution positions
            safe = np.argsort(contrib)[:max(1, n // 4)]
            add_idx = int(safe[rng.integers(len(safe))])
            a_trial[add_idx] += delta
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    return best_a, best_sc


def _local_search(a_init, deadline, rng, n_trials=20000):
    """Combined local search: block perturbations, pair moves, peak shaving."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break
        
        strategy = it % 4
        
        if strategy == 0:
            # Block multiplicative perturbation
            block_size = int(rng.integers(3, min(32, n // 4) + 1))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.6, 1.4, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
        
        elif strategy == 1:
            # Pair swap
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt = min(a_trial[i1], a_trial[i2]) * frac
            a_trial[i1] -= amt
            a_trial[i2] += amt
            if a_trial[i1] < 0:
                continue
            a_trial /= a_trial.sum()
        
        elif strategy == 2:
            # Gaussian perturbation on a block
            block_size = int(rng.integers(10, min(100, n // 8) + 1))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            noise = rng.normal(0, 0.1, size=block_size)
            a_trial[start:start + block_size] *= np.exp(noise)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
        
        else:
            # Single element move (peak-directed)
            c = _full_convolve(a, a)
            max_k = int(np.argmax(c))
            lo = max(0, max_k - (n - 1))
            hi = min(n - 1, max_k)
            if lo > hi:
                continue
            pairs = np.arange(lo, hi + 1)
            partners = max_k - pairs
            products = a[pairs] * a[partners]
            idx = pairs[int(np.argmax(products))]
            
            a_trial = a.copy()
            delta = a_trial[idx] * rng.uniform(0.3, 0.6)
            a_trial[idx] -= delta
            # Add to position far from peak
            dist = np.abs(np.arange(n) - idx)
            candidates = np.where(dist > n // 4)[0]
            if len(candidates) == 0:
                candidates = np.arange(n)
            add_idx = int(candidates[rng.integers(len(candidates))])
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
        
        sc = _score_norm_fn(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_sc


def _make_profiles(n):
    """Generate diverse analytical profiles."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    profiles = []
    
    # Linear
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + slope * u
        profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # Quadratic
    for a1, a2 in [(0.5, 0.3), (1.0, -0.3), (0.8, 0.5), (1.5, -0.5), (0.3, 1.0)]:
        a = 1.0 + a1 * u + a2 * u ** 2
        profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # Power
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0]:
        a = (1.0 - np.abs(u)) ** p
        profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # Raised cosine
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        a = 0.5 * (1.0 + np.cos(np.pi * alpha * u))
        profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # B-spline-like
    for order in [2, 3, 4, 6, 8]:
        x = np.clip(t, 0.01, 0.99)
        a = x ** (order - 1) * (1.0 - x) ** (order - 1)
        profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # Triangular weight
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # Bimodal
    for sep in [0.3, 0.5, 0.7]:
        for w in [0.1, 0.15, 0.2]:
            c1 = 0.5 - sep / 2
            c2 = 0.5 + sep / 2
            a = np.exp(-(t - c1)**2 / (2*w**2)) + np.exp(-(t - c2)**2 / (2*w**2))
            profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    # W-shaped
    for freq in [2, 3, 4, 6]:
        for depth in [0.2, 0.4, 0.6]:
            a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
            profiles.append(np.maximum(a, 1e-12) / np.maximum(a, 1e-12).sum())
    
    return profiles


def _interp_to_n(a, n_new):
    """Interpolate a sequence to a new length."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n0 = len(a)
    if n0 == n_new:
        return a / a.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

        parent = None
    
    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_work = 8192
        parent = (np.arange(n_work, dtype=np.float64) + 1) * (n_work - np.arange(n_work, dtype=np.float64))
        parent = parent / parent.sum()
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, triangular init n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    print(f"Working with n={n_work}")
    
    # === Phase 0: Diverse initialization (30s) ===
    phase0_end = min(deadline, t_start + 30.0)
    profiles = _make_profiles(n_work)
    
    best_prof = None
    best_prof_sc = best_score
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        if sc < best_prof_sc:
            best_prof_sc = sc
            best_prof = p.copy()
        if sc < best_score:
            update_best(p, sc, f"prof{idx}")
    
    if best_prof is not None and best_prof_sc < _score_norm_fn(parent):
        start_a = best_prof
    else:
        start_a = parent.copy()
    
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # === Phase 1: Main gradient optimization (700s) ===
    phase1_end = min(deadline, t_start + 730.0)
    print(f"Phase 1: Main gradient opt, start score={_score_norm_fn(start_a):.10f}")
    a_best, sc_best = _gradient_optimize(start_a, phase1_end, rng,
                                          max_lr=1.5, K_start=30, K_end=2,
                                          momentum=0.92, phase_name="main")
    update_best(a_best, sc_best, "grad_main")
    
    # === Phase 2: Fine-grained gradient (300s) ===
    phase2_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"Phase 2: Fine gradient, start score={_score_norm_fn(best_seq):.10f}")
        a_best2, sc_best2 = _gradient_optimize(best_seq, phase2_end, rng,
                                                max_lr=0.5, K_start=5, K_end=1,
                                                momentum=0.95, phase_name="fine")
        update_best(a_best2, sc_best2, "grad_fine")
    
    # === Phase 3: Peak shaving + local search (300s) ===
    phase3_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Peak shaving")
        dl_shave = min(phase3_end - 100, time.monotonic() + 120.0)
        a_best3, sc_best3 = _peak_shave(best_seq, dl_shave, rng, n_trials=10000)
        update_best(a_best3, sc_best3, "peak_shave")
        
        # Gradient polish after shaving
        if time.monotonic() < phase3_end - 50:
            dl_grad3 = min(phase3_end - 30, time.monotonic() + 80.0)
            a_best3b, sc_best3b = _gradient_optimize(best_seq, dl_grad3, rng,
                                                      max_lr=0.3, K_start=3, K_end=1,
                                                      momentum=0.9, phase_name="shave_grad")
            update_best(a_best3b, sc_best3b, "shave_grad")
        
        if time.monotonic() < phase3_end:
            print("Phase 3b: Local search")
            dl_ls = phase3_end - 5.0
            a_best4, sc_best4 = _local_search(best_seq, dl_ls, rng, n_trials=50000)
            update_best(a_best4, sc_best4, "local_search")
    
    # === Phase 4: Multi-restart (200s) ===
    phase4_end = min(deadline, t_start + 1530.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Multi-restart")
        n_restarts = 5
        per_restart = max(15.0, (phase4_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break
            
            noise_scale = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise_scale * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase4_end - 5, time.monotonic() + per_restart)
            a_r, sc_r = _gradient_optimize(p_start, dl, rng,
                                            max_lr=1.0, K_start=10, K_end=1,
                                            momentum=0.9, phase_name=f"restart{i}")
            update_best(a_r, sc_r, f"restart{i}")
    
    # === Phase 5: Final ultra-fine polish ===
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = max(20.0, deadline - time.monotonic() - 5.0)
        print(f"Phase 5: Final polish, remaining={remaining:.0f}s")
        
        # Very fine gradient with K=1
        dl_final = time.monotonic() + remaining * 0.5
        a_fin, sc_fin = _gradient_optimize(best_seq, dl_final, rng,
                                            max_lr=0.2, K_start=1, K_end=1,
                                            momentum=0.98, phase_name="final",
                                            min_improve=1e-14)
        update_best(a_fin, sc_fin, "final_grad")
        
        # Final local search
        if time.monotonic() < deadline - 15:
            dl_ls2 = min(deadline - 10, time.monotonic() + remaining * 0.3)
            a_fin2, sc_fin2 = _local_search(best_seq, dl_ls2, rng, n_trials=30000)
            update_best(a_fin2, sc_fin2, "final_ls")
        
        # One more gradient pass
        if time.monotonic() < deadline - 10:
            dl_fin3 = min(deadline - 5, time.monotonic() + remaining * 0.15)
            a_fin3, sc_fin3 = _gradient_optimize(best_seq, dl_fin3, rng,
                                                  max_lr=0.1, K_start=1, K_end=1,
                                                  momentum=0.99, phase_name="ultra_final")
            update_best(a_fin3, sc_fin3, "ultra_final")
    
    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            n_fb = 8192
            final_seq = (np.arange(n_fb, dtype=np.float64) + 1) * (n_fb - np.arange(n_fb, dtype=np.float64))
            final_seq = final_seq / final_seq.sum()
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_norm_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        # Validate with evaluator if feasible
        if len(final_seq) <= 32768:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned: {sc_eval}")
            except Exception as ex:
                print(f"Evaluate failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    n_fb = 8192
    fallback = (np.arange(n_fb, dtype=np.float64) + 1) * (n_fb - np.arange(n_fb, dtype=np.float64))
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n={n_fb}, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
