def _fast_minimax_step(a, n_steps=50, deadline=None):
    """Fast approximate minimax using multiplicative updates."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = c.max()
        
        # Compute "effectiveness" of each coordinate
        # g_j = sum_k w_k * a[k-j] (soft cross-correlation)
        # Use uniform weights (simplest approximation)
        # g_j = (1/(2n-1)) * sum_k a[k-j] = (1/(2n-1)) * sum of valid a values
        # This is just a constant! So uniform weights don't help.
        
        # Better: use the actual convolution as weights
        w = c / c.sum()
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1 : 2*n-1]
        W = float(np.dot(w, c))
        
        # Multiplicative update: a_j <- a_j * exp(-eta * (h_j - W) / M)
        eta = 0.5
        update = np.exp(-eta * (h - W) / (M + 1e-15))
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def _peaking_reduction(a, deadline, max_rounds=100):
    """Targeted reduction of the peak convolution value."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        
        # Try to reduce c[k_star] by adjusting elements
        # c[k_star] = sum_m a[m] * a[k_star - m]
        # To reduce this, we can decrease a[m] for m where a[m]*a[k_star-m] is large
        # and increase a[m'] for m' not contributing to c[k_star]
        
        # Gradient of c[k_star] w.r.t. z_j:
        # dc[k_star]/dz_j = 2 * a[j] * (a[k_star - j] - c[k_star])
        # (from our earlier derivation, with w concentrated at k_star)
        
        a_kj = np.zeros(n)
        valid_j = np.arange(n)
        k_minus_j = k_star - valid_j
        valid_mask = (k_minus_j >= 0) & (k_minus_j < n)
        a_kj[valid_mask] = a[k_minus_j[valid_mask]]
        
        # Gradient direction to reduce c[k_star]
        grad_peak = 2.0 * a * (a_kj - c[k_star])
        
        # But we also need to not increase other c[k] values
        # Use a smoothed gradient
        w = np.zeros(2*n-1)
        w[k_star] = 1.0
        # Add some smoothing
        for dk in [-2, -1, 1, 2]:
            if 0 <= k_star + dk < 2*n-1:
                w[k_star + dk] = 0.2
        
        w = w / w.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1 : 2*n-1]
        
        # Take a step in the negative gradient direction
        lr = 0.1 / (1.0 + rnd * 0.1)
        z = z - lr * 4.0 * a * (h - W)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Uniform
    profiles.append(np.ones(n) / n)
    
    # Cosine-like
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1, 2, 3, 5, -1, -2, -3]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    
    # Quadratic
    for c in [0, 0.5, 1, 2]:
        a = 1.0 + c * (t - 0.5)**2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Sine-based
    for freq in [2, 4, 8]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    return profiles

"""
Multi-phase optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses L-BFGS with aggressive log-sum-exp beta continuation, peaking-directed
local search, diverse restarts, and multiple sequence length exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = c.max()
    vals = np.clip((c - cm) / beta, -50, 100)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = c.max()
    vals = np.clip((c - cm) / beta, -50, 100)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1: 2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-16,
                         'maxls': 50, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        if verbose:
                            print(f"        beta={beta:.2e}: score={s_new:.10f}")
                z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _peaking_reduction(a_init, deadline, max_rounds=200):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = c.max()

        # Build soft weights centered on the peak
        k_star = int(np.argmax(c))
        w = np.zeros(2 * n - 1)
        sigma = max(1, int(0.5 * n / max(1, np.sqrt(n))))
        for dk in range(-sigma * 3, sigma * 3 + 1):
            k = k_star + dk
            if 0 <= k < 2 * n - 1:
                w[k] = np.exp(-0.5 * (dk / sigma) ** 2)
        w = w / w.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1: 2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        # Adaptive learning rate
        lr = 0.15 / (1.0 + 0.05 * rnd)
        z = z - lr * grad / (np.linalg.norm(grad) + 1e-15) * 0.3
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _fast_minimax(a_init, deadline, n_steps=100):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        w = c / c.sum()
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1: 2 * n - 1]
        W = float(np.dot(w, c))
        M = float(c.max())

        eta = 0.3 / (1.0 + 0.01 * step)
        update = np.exp(-eta * (h - W) / (M + 1e-15))
        a_new = a * update
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    profiles.append(np.ones(n) / n)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    for rate in [1, 2, 3, 5, -1, -2, -3]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for c in [0.5, 1, 2, 4]:
        a = 1.0 + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for freq in [2, 3, 5, 8]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u ** 2)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _update_best(state, p, sc, label=""):
    if p is not None and np.isfinite(sc) and sc < state['best_score']:
        state['best_seq'] = np.asarray(p, dtype=np.float64).copy()
        state['best_score'] = float(sc)
        elapsed = time.monotonic() - state['t_start']
        print(f"  NEW BEST: {state['best_score']:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")


def _coord_descent_fine(a_init, deadline, max_rounds=50):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        improved = False
        order = np.random.default_rng(rnd + 42).permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.05, -0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break

    return best_a, best_score
