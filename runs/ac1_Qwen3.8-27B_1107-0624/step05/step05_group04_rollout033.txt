        x = np.linspace(-1, 1, n)
        w = np.maximum(0, 1 - np.abs(x)) ** p
        if w.sum() > 0:
            candidates.append(w / w.sum())

def _convolve_equalize(a_init, deadline, max_iters=50000):
    """FFT-based convolution equalization: minimize variance of conv(a,a)."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    N = _next_pow2(3 * n)  # Enough for cross-correlation without wraparound
    
    # Pre-allocate
    e_padded = np.zeros(N)
    a_padded = np.zeros(N)
    
    # Cosine learning rate schedule
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute convolution spectrum
        fa = np.fft.rfft(a_padded * 0.0 + np.concatenate([a, np.zeros(N - n)]))
        c_full = np.fft.irfft(fa * fa, N)[:2 * n - 1]
        
        # Target
        t_target = 1.0 / (2 * n - 1)
        
        # Error
        e = c_full - t_target
        
        # Compute gradient: grad[j] = 4 * sum_k e[k] * a[k-j]
        # Using FFT cross-correlation
        E = np.fft.rfft(e_padded * 0.0 + np.concatenate([e, np.zeros(N - (2 * n - 1))]))
        A = np.fft.rfft(np.concatenate([a, np.zeros(N - n)]))
        nc = np.fft.irfft(E * np.conj(A), N)
        grad = 4.0 * nc[:n]
        
        # Normalize gradient and take step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Cosine decay learning rate
        lr = 0.5 * (1.0 + np.cos(np.pi * it / max_iters)) / (gnorm / n + 1e-10)
        lr = min(lr, 2.0)
        
        a = a - lr * grad / n
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        # Check improvement every 100 iterations
        if it % 100 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Final check
    sc = _score_norm_fn(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))
