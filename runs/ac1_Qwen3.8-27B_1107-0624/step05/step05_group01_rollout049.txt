"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS continuation, projected subgradient, LP peak balancing,
and adaptive restarts to escape local minima.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    """Assumes sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * c.max()

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

        m0 = c0.max()
        beta_list = [m0 * f for f in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        def obj(z_in):
            a_in = _softmax(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = c_in.max()
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = shifted.max()
            lse = cm + beta * (max_s + np.log(np.sum(np.exp(shifted - max_s))))
            return 2.0 * n * lse
        
        def grad(z_in):
            a_in = _softmax(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = c_in.max()
            if cm < 1e-300:

            shifted = (c_in - cm) / beta
            max_s = shifted.max()
            e = np.exp(shifted - max_s)

            a_rev = a_in[::-1]

            h = w_conv[n-1:2*n-1]

def _lp_phase(a_init, deadline, K=50, max_rounds=20):

        M = c.max()

        active = sorted_idx[:K]

            i_hi = min(n - 1, k)
            if i_lo <= i_hi:

"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS continuation, projected subgradient with backtracking,
LP peak balancing, adaptive Adam, and random restarts.
Key improvements: direct simplex projection (allows zeros), smarter LP,
multi-scale peak targeting, and better time allocation.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    idx = np.nonzero(cond)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    rho = idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _lbfgs_phase(a_init, deadline, beta_list=None, maxiter=6000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        beta_list = [m0 * f for f in [0.05, 0.03, 0.015, 0.008, 0.004,
                                       0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                       0.00005, 0.00002]]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def obj_fn(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return 2.0 * n * lse

        def grad_fn(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())

def _proj_grad_phase(a_init, deadline, lr_init=0.005, max_iter=200000,

        peak_idx = np.where(c >= threshold)[0]

        I = np.zeros(2 * n - 1)
        I[peak_idx] = 1.0
        I_rev = I[::-1]
        grad_full = _full_convolve(I_rev, a)
        grad = grad_full[n - 1:2 * n - 1]

        # Normalize gradient to have unit norm for stable step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad = grad / gnorm
        lr = lr * gnorm  # scale back

        # Backtracking line search
        improved = False
        for _ls in range(max_ls):
            a_trial = _project_simplex(a - lr * grad)
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())

                M = M_trial

    s = best_a.sum()
    if s < 1e-15:
        return best_a, _score_norm(best_a)
    best_a = best_a / s
    best_score = _score_norm(best_a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]

        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

                          b_ub=np.full(len(active), M * slack),
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 15})
            if res.status != 0:
                break
            b_new = np.asarray(res.x, dtype=np.float64)

        except Exception:
            break

            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)

            if st < 1e-15:

            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())

                best_score = sc_trial
                M = M_trial

                K = max(10, K // 2)
            else:
                break

    return best_a, best_score


def _adam_phase(a_init, deadline, lr=0.008, max_iter=300000):
    """Adam with adaptive log-sum-exp and restarts."""
    n = len(a_init)
    z = np.log(np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15))
    z = z - z.mean()
    best_a = np.asarray(a_init, dtype=np.float64).copy()
    best_score = _score_norm(best_a)

    c0 = _full_convolve(best_a, best_a)
    beta0 = 0.01 * float(c0.max())
    beta = beta0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    stall = 0
    rng = np.random.default_rng(777)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-15 * cm, beta0 * (0.97 ** (t // 40)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 40 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                stall = 0
            else:
                stall += 1
            if stall > 60:
                z = z + 0.04 * rng.standard_normal(n)
                z = z - z.mean()
                stall = 0

    return best_a, best_score


def _active_peak_opt(a_init, deadline, K=5, max_rounds=50):
    """Directly optimize top-K peaks using L-BFGS on their sum."""
    n = len(a_init)
    z = np.log(np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15))
    z = z - z.mean()
    best_a = np.asarray(a_init, dtype=np.float64).copy()
    best_score = _score_norm(best_a)

    for outer in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_idx = np.argsort(c)[::-1][:K]

        def obj_apeak(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_idx]
            return float(np.sum(peaks ** 2))  # L2^2 of top peaks

        def grad_apeak(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_idx]
            # d/d a_i of sum_k peaks[k]^2 = 2 * sum_k peaks[k] * a_{k-i}
            I_w = np.zeros(2 * n - 1)
            for idx_k, k in enumerate(peak_idx):
                I_w[k] = 2.0 * peaks[idx_k]
            I_rev = I_w[::-1]
            grad_a_full = _full_convolve(I_rev, a_in)
            grad_a = grad_a_full[n - 1:2 * n - 1]
            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            res = minimize(obj_apeak, z, jac=grad_apeak, method='L-BFGS-B',
                           options={'maxiter': 2000, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': 10000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score - 1e-12:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = res.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: L-BFGS continuation, projected gradient,
    LP peak balancing, active peak optimization, and Adam polish.
    Uses direct simplex projection to allow boundary solutions.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = np.inf

    def update(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s < 0.01:
            return
        p = p / s
        sc = float(sc)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if parent_raw.sum() > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
    else:
        n = 8192
        parent = _greedy_init(n)

    print(f"Starting: n={n}")
    update(parent, _score_norm(parent), "parent")

    # === Phase 1: L-BFGS continuation (350s) ===
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 350.0)
    a1, s1 = _lbfgs_phase(best_seq, p1_end, maxiter=8000)
    update(a1, s1, "lbfgs_1")

    # === Phase 2: LP peak balancing (250s) ===
    print("Phase 2: LP peak balancing")
    p2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < p2_end:
        a2, s2 = _lp_phase(best_seq, p2_end, K=40, max_rounds=12)
        update(a2, s2, "lp_1")

    # === Phase 3: Projected gradient (300s) ===
    print("Phase 3: Projected gradient")
    p3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < p3_end:
        a3, s3 = _proj_grad_phase(best_seq, p3_end, lr_init=0.003,
                                   max_iter=200000, peak_frac=0.97,
                                   max_ls=12, restart_every=80)
        update(a3, s3, "projgrad")

    # === Phase 4: L-BFGS round 2 (200s) ===
    print("Phase 4: L-BFGS round 2")
    p4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < p4_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
        a4, s4 = _lbfgs_phase(best_seq, p4_end, beta_list=tiny_betas, maxiter=6000)
        update(a4, s4, "lbfgs_2")

    # === Phase 5: Active peak optimization (150s) ===
    print("Phase 5: Active peak optimization")
    p5_end = min(deadline, t_start + 1250.0)
    if time.monotonic() < p5_end:
        for K in [3, 5, 8]:
            if time.monotonic() > p5_end - 20:
                break
            a5, s5 = _active_peak_opt(best_seq, min(p5_end, time.monotonic() + 40), K=K, max_rounds=20)
            update(a5, s5, f"active_K{K}")

    # === Phase 6: LP round 2 (100s) ===
    print("Phase 6: LP round 2")
    p6_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < p6_end:
        a6, s6 = _lp_phase(best_seq, p6_end, K=60, max_rounds=8, slack=0.999)
        update(a6, s6, "lp_2")

    # === Phase 7: Projected gradient round 2 (150s) ===
    print("Phase 7: Projected gradient round 2")
    p7_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < p7_end:
        a7, s7 = _proj_grad_phase(best_seq, p7_end, lr_init=0.002,
                                   max_iter=150000, peak_frac=0.95,
                                   max_ls=10, restart_every=60)
        update(a7, s7, "projgrad_2")

    # === Phase 8: Adam polish (remaining ~270s) ===
    print("Phase 8: Adam polish")
    if time.monotonic() < deadline - 20:
        a8, s8 = _adam_phase(best_seq, deadline - 50, lr=0.006, max_iter=300000)
        update(a8, s8, "adam_1")

        if time.monotonic() < deadline - 30:
            a9, s9 = _adam_phase(best_seq, deadline - 10, lr=0.003, max_iter=200000)
            update(a9, s9, "adam_2")

    # === Phase 9: Try larger n if time allows ===
    if time.monotonic() < deadline - 120 and best_seq is not None:
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"Phase 9: Interpolating to n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            update(a_interp, sc_in, f"interp_n{n_new}")
            if time.monotonic() < deadline - 30:
                a_in_b, sc_in_b = _lbfgs_phase(a_interp, deadline - 10, maxiter=3000)
                update(a_in_b, sc_in_b, f"lbfgs_n{n_new}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 100000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(1024)
    print(f"FALLBACK: n=1024, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
