"""
Nesterov-accelerated gradient descent with adaptive soft-max temperature,
Fourier parametric exploration, and coordinate descent refinement for
minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines fast low-dimensional parametric search with full gradient fine-tuning.
"""
import numpy as np
import time
from scipy.optimize import minimize_scalar


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_and_loss(z, beta):
    """Compute gradient and loss for L(z) = 2n * beta * logsumexp(conv(softmax(z))/beta)."""
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s_e = float(e.sum())
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    # dL/dc = 2n * w
    # dL/da[i] = 4n * conv(w, flip(a))[i+n-1]
    a_flip = a[::-1]
    wa = _full_convolve(w, a_flip)
    dlda = 4.0 * n * wa[n - 1:2 * n - 1]
    # dL/dz = a * (dlda - a . dlda)
    a_dot_dlda = float(np.dot(a, dlda))
    grad = a * (dlda - a_dot_dlda)
    loss = 2.0 * n * beta * (np.log(s_e) + c_max / beta) if s_e > 0 else float(c_max)
    return grad, float(loss), a, c_max


def _nesterov_optimize(a_init, deadline, beta_init=None, lr_init=0.005,
                       momentum=0.9, beta_floor=1e-6, n_phases=8):
    """Nesterov-accelerated gradient descent with geometric beta cooling."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    best_a = a_init.copy()
    best_mc = max_c0

    if beta_init is None:
        beta_init = 0.05 * max_c0

    v = np.zeros(n)
    z_prev = z.copy()
    t = 0
    lr = lr_init

    # Geometric beta schedule
    beta_ratio = (beta_floor / beta_init) ** (1.0 / max(1, n_phases))

    for phase in range(n_phases):
        beta = max(beta_init * (beta_ratio ** phase), beta_floor)
        phase_start = time.monotonic()
        while time.monotonic() < deadline:
            t += 1
            # Nesterov lookahead
            z_look = z + momentum * (z - z_prev)
            z_prev = z.copy()

            grad, loss, a, c_max = _compute_grad_and_loss(z_look, beta)
            if not np.all(np.isfinite(grad)):
                break

            # Check actual score periodically
            if t % 100 == 0:
                sc = float(_full_convolve(a, a).max())
                if np.isfinite(sc) and sc < best_mc:
                    best_mc = sc
                    best_a = a.copy()

            # RMSProp-style adaptive learning rate
            v = 0.9 * v + 0.1 * grad
            v_hat = v / (1.0 - 0.9 ** t) if t > 1 else v
            step = lr * v_hat / (np.sqrt(v_hat ** 2).mean() + 1e-8)

            # Clip step size

            if step_norm > 1.0:
                step = step * (1.0 / step_norm)

            z = z - step
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            # Decay learning rate
            if t % 5000 == 0:
                lr = max(lr_init * 0.5, lr * 0.9)

        # End of phase
        a_end = _softmax_fn(z)
        sc_end = float(_full_convolve(a_end, a_end).max())
        if np.isfinite(sc_end) and sc_end < best_mc:
            best_mc = sc_end
            best_a = a_end.copy()

    # Final evaluation
    a_final = _softmax_fn(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_mc:
        best_mc = sc_final
        best_a = a_final.copy()

    return best_a, best_mc


def _fourier_search(a_init, deadline, n_modes=64):

    best_score = float(_full_convolve(a_norm, a_norm).max())

    def _reconstruct(coeffs):

            a += coeffs[k] * np.cos(2.0 * np.pi * (k + 1) * j / n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s

    def _obj(coeffs):
        a = _reconstruct(coeffs)
        c = _full_convolve(a, a)
        return float(c.max())

    # Coordinate descent on Fourier coefficients
    coords = np.arange(n_modes)
    rng = np.random.default_rng(12345)
    rng.shuffle(coords)

    iter_count = 0
    while time.monotonic() < deadline:

        for idx in coords:
            if time.monotonic() > deadline:
                break
            best_c = coeffs[idx]
            best_val = best_score
            # Try a few perturbations
            for delta in [0.5, -0.5, 0.2, -0.2, 0.1, -0.1]:
                coeffs[idx] = best_c + delta * (1.0 / (1.0 + idx * 0.1))
                a = _reconstruct(coeffs)
                val = float(_full_convolve(a, a).max())
                if val < best_val:
                    best_val = val
                    best_c = coeffs[idx]
                coeffs[idx] = best_c
            if np.isfinite(best_val) and best_val < best_score:
                best_score = best_val
                best_coeffs = coeffs.copy()
                improved = True
        if not improved:

                if val < best_score:

                    best_coeffs = coeffs.copy()
                else:
                    coeffs[idx] = old
        iter_count += 1
        if iter_count > 200:

def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_fn(z)


def _interp_profile(p, n_new):
    """Interpolate a normalized profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

            remaining = phase3_end - time.monotonic() - 40
            if remaining < 60:
                break

            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            if remaining > 100:
                a_r, _ = _adam_refine(
                    p_interp, time.monotonic() + min(remaining, phase3_end - time.monotonic() - 30),
                    lr=0.003
                )
                update_best(a_r)

            # Nesterov on the interpolated version
            if time.monotonic() < phase3_end - 60:
                remaining2 = phase3_end - time.monotonic() - 40
                if remaining2 > 80:
                    a_n, _ = _nesterov_optimize(
                        p_interp, time.monotonic() + min(remaining2, phase3_end - time.monotonic() - 30),
                        beta_init=0.02 * float(_full_convolve(p_interp, p_interp).max()),
                        lr_init=0.005,
                        momentum=0.9,
                        n_phases=5
                    )
                    update_best(a_n)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Diversification (1400s - 1700s) ===
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 4: Diversification ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic() - 20) / max(1, n_restarts)
        sigmas = [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            deadline_i = time.monotonic() + min(per_restart, phase4_end - time.monotonic() - 10)
            if deadline_i <= time.monotonic():
                break

            # Perturb
            p_start = _perturb_log(a_best_norm, sigmas[i], rng)

            # Also try structured perturbation (sine wave)
            if i % 2 == 1:
                freq = 2 + i
                p_struct = a_best_norm * (1.0 + 0.1 * sigmas[i] * np.sin(2 * np.pi * freq * np.arange(len(a_best_norm)) / len(a_best_norm)))
                p_struct = np.maximum(p_struct, 1e-15)
                p_struct = p_struct / p_struct.sum()
                p_start = p_struct

            a_r, sc_r = _adam_refine(p_start, deadline_i, lr=0.003)
            update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Final polish (1700s - deadline) ===
    print(f"\n=== Phase 5: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 8
        if remaining > 30:
            a_final, _ = _adam_refine(a_best_norm, time.monotonic() + remaining, lr=0.001)
            update_best(a_final)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # --- Final validation ---
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
