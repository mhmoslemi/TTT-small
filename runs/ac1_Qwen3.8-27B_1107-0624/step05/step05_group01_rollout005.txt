                return 0.0

            sh = (peaks - pm) / beta

            return float(2.0 * n * lse)

def _adam_backtrack(a_init, deadline, lr_init=0.05, beta_init_frac=0.01,
                    decay=0.98, max_iter=500000):
    """Adam with backtracking line search and adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = beta_init_frac * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-10
    no_improve = 0
    check_interval = 20

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-10 * cm, cm0 * beta_init_frac * (decay ** (t // check_interval)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        # Backtracking line search
        lr = lr_init / (1.0 + 0.1 * t / check_interval)
        step = lr * mh / (np.sqrt(vh) + eps)

        obj_current = _lse_obj(z, beta, n)
        alpha = 1.0
        improved = False
        for _bt in range(10):
            z_trial = z - alpha * step
            z_trial = np.clip(z_trial, -50, 50)
            z_trial = z_trial - z_trial.mean()
            obj_trial = _lse_obj(z_trial, beta, n)
            if obj_trial < obj_current:
                z = z_trial
                improved = True
                break
            alpha *= 0.5

        if not improved:
            # Try without the mean subtraction constraint
            no_improve += 1
            if no_improve > 20:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0
                m = np.zeros(n)
                v = np.zeros(n)
            continue
        else:
            no_improve = 0

        if t % check_interval == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()

    return best_a, best_score


def _greedy_init(n):
    """Binomial-based initialization."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype='exp'):
    t = np.linspace(0, 1, n)
    if etype == 'exp3':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'linear':
        a = 0.1 + 0.9 * t
    elif etype == 'exp7':
        a = np.exp(7.0 * (t - 0.5))
    elif etype == 'cubic':
        a = 0.05 + 0.5 * t + 0.45 * t**3
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()
