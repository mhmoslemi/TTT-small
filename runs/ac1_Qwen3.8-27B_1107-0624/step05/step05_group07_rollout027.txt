"""

Uses: (1) LP-based best-response iteration, (2) Nesterov-accelerated (FISTA)

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f, max_time=30):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_val = g_val / g_val.sum()
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters=50, deadline=None, step_size=0.3):

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _fista_logsumexp(a_init, deadline, beta_list=None):

    n = len(a_init)

    best_a = a_init.copy()

                 0.0004, 0.0003, 0.0002, 0.00015, 0.0001, 0.00008]

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:

        # FISTA parameters
        t_fista = 1.0
        z_prev = z.copy()
        z_curr = z.copy()

        # Estimate Lipschitz constant conservatively
        # For log-sum-exp with parameter beta, L ~ O(n/beta)
        L_est = 4.0 * n / beta
        step = 1.0 / L_est
        # Cap step to avoid instability
        step = min(step, 0.5)

        max_iters = int(time_per * 20)
        max_iters = max(100, min(max_iters, 20000))

        for it in range(max_iters):

                break
            # FISTA extrapolation
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
            momentum = (t_fista - 1.0) / t_new
            y = z_curr + momentum * (z_curr - z_prev)

            # Gradient step
            grad = _logsumexp_grad(y, beta, n)
            if not np.all(np.isfinite(grad)):

def _peak_targeting_search(a_init, deadline, rng, n_rounds=200):
    """
    Directly target the peak of the autocorrelation:
    - Find the lag k* with max autocorrelation
    - Identify which elements contribute most to c[k*]
    - Reduce those elements and redistribute mass to "safe" positions
    """
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a = a_init.copy()

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Compute "responsibility" of each element for the peak
        # c[k*] = sum_i a[i] * a[k* - i] (where valid)
        # Element j contributes a[j] * a[k* - j] to c[k*]
        valid_start = max(0, k_star - (n - 1))
        valid_end = min(n, k_star + 1)
        resp = np.zeros(n)
        for j in range(valid_start, valid_end):
            partner = k_star - j
            if 0 <= partner < n:
                resp[j] = a[j] * a[partner]

        # Find elements with high responsibility
        # We want to reduce them
        top_k = max(1, n // 50)  # top 2% of elements by responsibility
        top_indices = np.argsort(resp)[-top_k:]

        # Find "safe" elements (low responsibility) to receive mass
        bottom_indices = np.argsort(resp)[:top_k]

        # Compute how much to transfer
        total_to_move = 0.01 * peak_val  # small transfer
        # Scale down high-resp elements
        scale_down = 0.02  # reduce by 2%
        for j in top_indices:
            if a[j] > 1e-10:
                a[j] *= (1 - scale_down)

        # Redistribute to safe elements
        mass_moved = sum(a_init[j] * scale_down for j in top_indices if a_init[j] > 1e-10)
        if mass_moved > 0:
            # Add to safe elements proportionally to their "capacity"
            safe_weights = np.array([1.0 / (resp[j] + 1e-10) for j in bottom_indices])
            safe_weights = safe_weights / safe_weights.sum()
            for idx, j in enumerate(bottom_indices):
                a[j] += mass_moved * safe_weights[idx]

        # Add small random perturbation for diversity
        noise = 0.005 * rng.normal(size=n)
        a = a + noise * a  # relative noise
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s

        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        elif sc > best_score * 1.01:
            # Revert if too bad
            a = best_a.copy()

    return best_a, best_score


def _subgradient_method(a_init, deadline, n_iters=5000):
    """
    Subgradient method with simplex projection.
    Directly uses the subgradient of max(conv(a,a)).
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    eta = 0.01
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        # Average over top few peaks for stability
        top_k = min(3, len(c))
        sorted_idx = np.argsort(c)[-top_k:]
        weights = c[sorted_idx]
        weights = weights / weights.sum()

        # Subgradient: convex combination of subgradients at top peaks
        grad = np.zeros(n)
        for k in sorted_idx:
            w_k = c[k] / weights.sum()
            # Subgradient at lag k: 2*a[k-j] for valid j
            for j in range(n):
                partner = k - j
                if 0 <= partner < n:
                    grad[j] += 2.0 * w_k * a[partner]

        # Normalized step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        step = eta / (it + 1) ** 0.5 / grad_norm

        a_new = a - step * grad
        # Project onto simplex
        a_new = _project_simplex(a_new)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    k = np.max(ind[cond])
    theta = cssv[k - 1] / k
    w = np.maximum(v - theta, 0)
    return w

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    n_restarts_lbfgs = 3
    per_restart = (phase3_end - time.monotonic()) / max(1, n_restarts_lbfgs)

    for r in range(n_restarts_lbfgs):
        if time.monotonic() > phase3_end - 20:
            break
        if r == 0:
            a_cur = best_seq.copy()
        else:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            noise_scale = 0.05 * (r + 1)
            z = z + noise_scale * rng.normal(size=len(z))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
        per_dl = min(phase3_end - 10, time.monotonic() + per_restart)
        p_best, sc_best = _run_lbfgs_continuation(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Peak-targeting search (250s)
    phase4_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 4: Peak targeting (n={len(best_seq)})")

    per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.8)
    p_best, sc_best = _peak_targeting_search(best_seq, per_dl, rng, n_rounds=300)
    update_best(p_best, sc_best, "peak_target")

    # Also try subgradient method
    if time.monotonic() < phase4_end - 30:
        per_dl2 = min(phase4_end - 10, time.monotonic() + 100.0)
        p_best2, sc_best2 = _subgradient_method(best_seq, per_dl2, n_iters=3000)
        update_best(p_best2, sc_best2, "subgrad")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Try different n values (200s)
    phase5_end = min(deadline, t_start + 1450.0)
    print(f"\nPhase 5: Different n exploration")

    for n_try in [3072, 5120, 6144]:
        if time.monotonic() > phase5_end - 30:
            break
        if n_try == len(best_seq):
            continue
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")

        # Quick L-BFGS on this n
        if time.monotonic() < phase5_end - 20:
            per_dl = min(phase5_end - 10, time.monotonic() + 60.0)
            p_best, sc_best = _run_lbfgs_continuation(p_interp, per_dl)
            update_best(p_best, sc_best, f"lbfgs_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Phase 6: Fine polishing (remaining time)
    phase6_end = deadline
    print(f"\nPhase 6: Final polishing (n={len(best_seq)})")

    if time.monotonic() < phase6_end and best_seq is not None:
        # Coordinate descent
        per_dl = min(phase6_end - 30, time.monotonic() + 120.0)
        p_best, sc_best = _coord_descent(best_seq, per_dl, max_rounds=80)
        update_best(p_best, sc_best, "coord_cd")

        # Tiny-beta L-BFGS
        if time.monotonic() < phase6_end - 30:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_tiny = 1e-5 * max_c_final
            per_dl2 = min(phase6_end - 10, time.monotonic() + 60.0)

            n = len(best_seq)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta_tiny, n),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-16,
                             'maxls': 20, 'maxfun': 10000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    sc_new = _score_norm_fn(a_new)
                    update_best(a_new, sc_new, "tiny_lbfgs")
            except Exception:
                pass

        # One more FISTA with very small beta
        if time.monotonic() < phase6_end - 20:
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_ultra = [2e-5 * max_c_final, 1e-5 * max_c_final, 5e-6 * max_c_final]
            per_dl3 = min(phase6_end - 5, time.monotonic() + 40.0)
            p_best, sc_best = _fista_logsumexp(best_seq, per_dl3, beta_list=beta_ultra)
            update_best(p_best, sc_best, "ultra_fista")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 8000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
