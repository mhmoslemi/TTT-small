"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses softmax parameterization, adaptive L-BFGS with beta-continuation,
subgradient polishing with Polyak step, and perturbation-based exploration.
Key improvements: adaptive beta schedule, Polyak step size, coarser perturbations for escape,
and a "peak-directed" refinement phase.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _active_gradient(a, c, n, tol_frac=0.005):
    """Exact subgradient using active set (lags within tol_frac of max)."""
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(len(a))
    threshold = c_max * (1.0 - tol_frac)
    active = c >= threshold
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W), c_max, active

def _lbfgs_phase(z_start, beta, n, max_iter, ftol=1e-13, gtol=1e-10):
    """Run one L-BFGS phase. Returns (z_out, success, niter, fun_out)."""
    def fun(z):
        return _smoobj(z, beta, n)
    def jac(z):
        return _smograd(z, beta, n)
    try:
        result = minimize(
            fun, z_start, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                     'maxcor': 30, 'maxls': 80, 'disp': False}
        )
        return result.x, result.success, int(result.nit), float(result.fun)
    except Exception:
        return z_start, False, 0, _smoobj(z_start, beta, n)

def _perturb_z(z, sigma, rng):
    z = z + sigma * rng.standard_normal(len(z))
    z = z - z.mean()
    return z

def _perturb_fourier_z(z, rng, n_modes=3, amp_scale=0.02):
    n = len(z)
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 4))
        amp = amp_scale * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return z

def _perturb_block_swap(z, rng, n_blocks=4):
    """Swap blocks in z-space to create structural changes."""
    n = len(z)
    z = z.copy()
    block_size = n // (2 * n_blocks)
    for _ in range(n_blocks):
        i = rng.integers(0, n - 2 * block_size)
        j = rng.integers(0, n - 2 * block_size)
        if abs(i - j) > block_size:
            z[i:i+block_size], z[j:j+block_size] = z[j:j+block_size].copy(), z[i:i+block_size].copy()
    z = z - z.mean()
    return z

        return [float(x) for x in final]

    # Fallback
    fallback = _make_initial(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
