import numpy as np

import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(4, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Score assuming sum(a) = 1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    """Build sparse matrix A where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    # A is (2n-1) x n, A[k,j] = f[k-j] if 0 <= k-j < n else 0
    row_idx = []
    col_idx = []
    data = []
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                row_idx.append(k)
                col_idx.append(j)
                data.append(f[idx])
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _build_conv_matrix_fast(f):
    """Build sparse convolution matrix more efficiently."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for j in range(n):

        # conv[k] += f[k-j] * g[j], so A[k,j] = f[k-j]
        for k in range(j, j + n):

            rows.append(k)
            cols.append(j)
            vals.append(f[idx])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)
    row_idx = np.tile(np.arange(n, dtype=np.int64), n) + np.repeat(np.arange(n, dtype=np.int64), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _build_conv_rows(f, lags):
    n = len(f)
    rows_list = []
    cols_list = []
    data_list = []
    for k in lags:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows_list.append(k)
            cols_list.append(j)
            data_list.append(f[k - j])
    if len(rows_list) == 0:
        return csr_matrix((len(lags), n))
    A = csr_matrix((data_list, (rows_list, cols_list)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, max_time=30.0, use_cutting_plane=True):
    n = len(f)
    t_start = time.monotonic()

    g = cp.Variable(n)
    t_var = cp.Variable()

    if use_cutting_plane:
        c = _full_convolve(f, f)
        top_k = min(100, 2 * n - 1)
        active_lags = np.argsort(c)[-top_k]
        constraints = [cp.sum(g) == 1, g >= 0]
        best_g_val = None
        best_t_val = None

        for cp_round in range(15):
            if time.monotonic() - t_start > max_time:
                break
            A_act = _build_conv_rows(f, active_lags)
            constraints = [A_act @ g <= t_var, cp.sum(g) == 1, g >= 0]
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            try:
                prob.solve(solver=cp.CLARABEL, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS, verbose=False)
                except Exception:
                    break
            if g.value is None:
                break
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s < 1e-15:
                break
            g_val = g_val / s
            t_val = float(t_var.value) if t_var.value is not None else float(_full_convolve(f, g_val).max())
            if best_g_val is None or t_val < best_t_val:
                best_g_val = g_val.copy()
                best_t_val = t_val

            f_conv_g = _full_convolve(f, g_val)
            violated = np.where(f_conv_g > t_val + 1e-10)[0]
            if len(violated) == 0:
                break
            active_lags = np.unique(np.concatenate([active_lags, violated]))
            if len(active_lags) > 300:
                top_idx = np.argsort(f_conv_g[active_lags])[-300:]
                active_lags = active_lags[top_idx]
        return best_g_val, best_t_val
    else:
        A = _build_conv_matrix(f)
        constraints = [A @ g <= t_var, cp.sum(g) == 1, g >= 0]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None, None
        if g.value is None:
            return None, None
        g_val = np.maximum(g.value, 0)
        s = g_val.sum()
        if s < 1e-15:
            return None, None
        g_val = g_val / s
        t_val = float(t_var.value) if t_var.value is not None else float(_full_convolve(f, g_val).max())
        return g_val, t_val


def _lp_iterate(f, n_iters=100, deadline=None, use_momentum=True):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    f_prev = f.copy()
    momentum = 0.5

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        elapsed_ratio = it / max(1, n_iters)
        step = max(0.05, 0.4 * (1 - 0.7 * elapsed_ratio))

        g, t_val = _lp_solve(f, max_time=20.0)
        if g is None:
            break

        if use_momentum:
            direction = step * (g - f + momentum * (f - f_prev))
            f_new = f + direction
        else:
            f_new = (1 - step) * f + step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()

        f_prev = f.copy()
        f = f_new

        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad


def _logsumexp_obj_active(z, beta, n, active_mask):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_active = c[active_mask]
    cm_active = float(c_active.max())
    vals = np.clip((c_active - cm_active) / beta, -50, 50)
    lse = cm_active + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _run_lbfgs_continuation(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.01, 0.007, 0.005, 0.003, 0.002,
                 0.001, 0.0007, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00007, 0.00005, 0.00003, 0.00002, 0.00001]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 90.0)
        time_per = max(time_per, 5.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 20))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=30, rng=None):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    c = _full_convolve(a_init, a_init)
    a = a_init.copy()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        improved = False
        order = np.arange(n)
        if rng is not None and rnd > 0:
            rng.shuffle(order)

        for j in order:
            if time.monotonic() > deadline:
                break
            s_j = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                val = 0.0
                ij = k - j
                if 0 <= ij < n:
                    val += a[ij]
                ij2 = j - k
                if 0 <= ij2 < n:
                    val += a[ij2]
                s_j[k] = val

            e_2j = np.zeros(2 * n - 1)
            if 2 * j < 2 * n - 1:
                e_2j[2 * j] = 1.0

            a_j = a[j]
            best_delta = 0.0
            best_local = float(c.max())

            candidates = []
            for frac in [-0.5, -0.2, -0.1, -0.05, -0.02, -0.01, 0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
                d = frac * max(a_j, 1e-6)
                if a_j + d >= 0:
                    candidates.append(d)
            # Also try absolute values
            for d in [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01, 0.05, 0.1]:
                if a_j + d >= 0:
                    candidates.append(d)

            for delta in candidates:
                new_max = float(np.max(c + delta * s_j + delta * delta * e_2j))
                if new_max < best_local:
                    best_local = new_max
                    best_delta = delta

            if best_delta != 0.0:
                a[j] += best_delta
                a[j] = max(a[j], 0.0)
                s_total = a.sum()
                if s_total > 1e-15:
                    a = a / s_total
                    c = _full_convolve(a, a)
                    sc = _score_norm_fn(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
        if not improved and rnd > 2:
            break

    return best_a, best_score
