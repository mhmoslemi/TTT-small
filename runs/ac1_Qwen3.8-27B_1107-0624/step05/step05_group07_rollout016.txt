def _logsumexp_obj(z, beta, n):

    c = _conv_self(a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)

def _logsumexp_grad(z, beta, n):

    c = _conv_self(a)
    cm = float(c.max())
    vals = (c - cm) / beta

    e = np.exp(vals)

    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:

        a = _softmax_fn(z)
        current_score = _score_norm_fn(a)
        improved = False
        order = np.random.default_rng(rnd * 12345).permutation(n)

        for j in order:

                break
            best_delta = 0.0
            best_local = current_score

                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)

                current_score = best_local

        if current_score < best_score:
            best_score = current_score
            best_a = _softmax_fn(z).copy()
        if not improved:

def _interp_profile(p0, n_new):

        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u ** 2)
    profiles.append(a / a.sum())
    # Asymmetric profiles
    a = np.exp(-2.0 * u)
    profiles.append(a / a.sum())
    a = np.exp(-5.0 * u)
    profiles.append(a / a.sum())
    a = 1.0 / (1.0 + np.exp(-10.0 * u))
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = _make_profiles(n_init)[5]  # exp profile
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    a_cur = parent.copy()

    # Phase 1: Exponentiated gradient (top-k deflation) - 650s
    phase1_end = min(deadline, t_start + 650.0)
    print(f"\nPhase 1: Exponentiated gradient (n={n_parent})")

    a_cur, sc1 = _exp_gradient_optimize(a_cur, phase1_end, eta_init=0.8, eta_decay=0.9995,
                                        top_k_cycle=[1, 2, 3, 5, 10, 20, 50])
    update_best(a_cur, sc1, "exp_grad")

    # Phase 1b: Second pass with different parameters
    if time.monotonic() < phase1_end - 50:
        phase1b_end = min(phase1_end - 10, time.monotonic() + 200.0)
        a_cur2, sc1b = _exp_gradient_optimize(a_cur, phase1b_end, eta_init=0.3, eta_decay=0.9998,
                                              top_k_cycle=[1, 3, 5, 10, 20])
        update_best(a_cur2, sc1b, "exp_grad2")
        if sc1b < sc1:
            a_cur = a_cur2

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS fine continuation - 500s
    phase2_end = min(deadline, t_start + 1150.0)
    print(f"\nPhase 2: L-BFGS fine optimization")

    # Main L-BFGS run
    if time.monotonic() < phase2_end - 60:
        per_dl = min(phase2_end - 60, time.monotonic() + 350.0)
        beta_list_main = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                          0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
        p_best, sc_best = _lbfgs_fine(best_seq, per_dl, beta_fracs=beta_list_main)
        update_best(p_best, sc_best, "lbfgs_main")

    # L-BFGS with perturbation
    if time.monotonic() < phase2_end - 30:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        noise = 0.02
        z = z + noise * rng.normal(size=len(z))
        z = z - z.mean()
        a_pert = _softmax_fn(z)
        per_dl2 = min(phase2_end - 10, time.monotonic() + 150.0)
        p_best2, sc_best2 = _lbfgs_fine(a_pert, per_dl2,
                                         beta_fracs=[0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
        update_best(p_best2, sc_best2, "lbfgs_pert")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Try different n values - 300s
    phase3_end = min(deadline, t_start + 1450.0)
    print(f"\nPhase 3: Different n exploration")

    for n_try in [2048, 3072, 6144]:
        if time.monotonic() > phase3_end - 60:
            break
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")

        if time.monotonic() < phase3_end - 30:
            per_dl = min(phase3_end - 20, time.monotonic() + 100.0)
            p_best, sc_best = _lbfgs_fine(p_interp, per_dl,
                                           beta_fracs=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
            update_best(p_best, sc_best, f"lbfgs_n{n_try}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Final polishing - remaining time
    phase4_end = deadline
    print(f"\nPhase 4: Final polishing")

    if time.monotonic() < phase4_end and best_seq is not None:
        # Coordinate descent
        per_dl = min(phase4_end - 60, time.monotonic() + 150.0)
        p_best, sc_best = _coord_descent_fine(best_seq, per_dl, max_rounds=15)
        update_best(p_best, sc_best, "coord_desc")

        # Final ultra-fine L-BFGS
        if time.monotonic() < phase4_end - 20:
            c_final = _conv_self(best_seq)
            max_c_final = float(c_final.max())
            per_dl2 = min(phase4_end - 10, time.monotonic() + 80.0)
            p_best2, sc_best2 = _lbfgs_fine(best_seq, per_dl2,
                                             beta_fracs=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002, 0.000001])
            update_best(p_best2, sc_best2, "ultra_lbfgs")

        # One more exponentiated gradient pass with very small eta
        if time.monotonic() < phase4_end - 15:
            per_dl3 = min(phase4_end - 5, time.monotonic() + 40.0)
            p_best3, sc_best3 = _exp_gradient_optimize(best_seq, per_dl3, eta_init=0.05,
                                                       eta_decay=0.9999, top_k_cycle=[1, 2])
            update_best(p_best3, sc_best3, "final_eg")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 8000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_profiles(n_fb)[5]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
