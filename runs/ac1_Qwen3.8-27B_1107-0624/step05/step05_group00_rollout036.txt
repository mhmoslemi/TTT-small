def _fista_backtrack(a_init, beta, deadline, max_iter):
    """FISTA with backtracking line search and periodic restarts."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    
    # Initial Lipschitz constant estimate
    L = 4.0 * n / beta
    
    # Compute initial state
    c_curr = _full_convolve(a_curr, a_curr)
    best_score = float(c_curr.max())
    best_a = a_curr.copy()
    
    # Smoothed objective value (for tracking)
    def smoothed_obj(x):
        c = _full_convolve(x, x)
        cm = float(c.max())
        w = np.exp((c - cm) / beta)
        ws = float(w.sum())
        if ws < 1e-300 or not np.isfinite(ws):
            return cm
        return cm + beta * np.log(ws)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = float(y.sum())
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        # Compute gradient at y
        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum
        
        grad = _compute_grad(y, w)
        
        # Backtracking line search
        step = 1.0 / L
        f_y = smoothed_obj(y)
        
        for bt in range(20):
            a_next = _proj_simplex(y - step * grad)
            f_next = smoothed_obj(a_next)
            if f_next <= f_y - 0.5 * step * np.dot(grad, a_next - y):
                break
            step *= 0.5
            L *= 2.0
        
        # Periodic restart
        if it % 50 == 0:
            f_curr = smoothed_obj(a_curr)
            if f_next > f_curr + 1e-8:
                a_prev = a_curr.copy()
                t_val = 1.0
            else:
                a_prev = a_curr.copy()
                a_curr = a_next
                t_val = t_next
        else:
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
        
        # Track best

            sc = float(a_curr.max() * 0)  # placeholder

            sc = float(c_chk.max())

    sc_final = float(c_final.max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_score
