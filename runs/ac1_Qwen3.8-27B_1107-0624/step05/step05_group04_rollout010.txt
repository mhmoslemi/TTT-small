"""
Improved multi-scale optimization with:
1. Diverse initializations (ramps, Gaussians, cosines, random smooth)
2. FISTA (accelerated gradient) for log-sum-exp smoothed objective
3. Adaptive peak-shaving with subgradient on true max
4. Longer sequences (up to 8192)
5. Block-coordinate refinement
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

def _true_max_grad(z, n):
    """Compute subgradient of max(conv(a,a)) w.r.t. z (via softmax parameterization)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    # Find all indices where c is within epsilon of max
    tol = 1e-10 * cm
    active = np.where(c >= cm - tol)[0]
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z, cm

def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(50, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.01 * cm:
        return max(1e-10, 0.01 * cm)
    elif gap > 0.3 * cm:
        return max(1e-10, 0.005 * cm)
    else:
        return max(1e-10, 0.1 * gap)

def _run_fista(z_init, deadline, rng, beta, max_iter=50000):
    """FISTA (accelerated gradient descent) for log-sum-exp objective."""
    n = len(z_init)
    z = z_init.copy()
    z_prev = z.copy()
    t_k = 1.0
    best_z = z.copy()
    best_val = _logsumexp_obj(z, beta, n)
    
    # Initial step size estimate
    a0 = _softmax_fn(z)
    c0 = _full_convolve(a0, a0)
    cm0 = float(c0.max())
    # Lipschitz constant estimate
    L = 4.0 * cm0 / beta  # rough estimate
    step = 1.0 / L
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        grad = _logsumexp_grad(z, beta, n)
        if not np.all(np.isfinite(grad)):
            break
        
        z_new = z - step * grad
        z_new = np.clip(z_new, -60, 60)
        z_new = z_new - z_new.mean()
        
        f_new = _logsumexp_obj(z_new, beta, n)
        
        # Backtracking if needed
        if f_new > _logsumexp_obj(z, beta, n) + 0.5 * L * step * step * np.dot(grad, grad):
            step *= 0.5
            z_new = z - step * grad
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            f_new = _logsumexp_obj(z_new, beta, n)
        
        t_next = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2
        z = z_new + ((t_k - 1) / t_next) * (z_new - z)
        z_prev = z
        t_k = t_next
        
        if f_new < best_val:
            best_val = f_new
            best_z = z.copy()
        
        if it % 10000 == 0 and it > 0:
            # Reset momentum if not improving
            z = best_z.copy()
            t_k = 1.0
    
    return best_z, best_val

def _run_lbfgs_continuation(a_init, deadline, rng, n_betas=15):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_start = 0.1 * max_c0
    beta_end = 1e-7 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 240.0)
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 1000 if beta > 0.05 * max_c0 else (3000 if beta > 0.005 * max_c0 else 8000)
        maxfun = maxiter * 5
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 100, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                z_new = np.clip(z_new, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _run_adam_gd(a_init, deadline, rng, lr_init=0.05, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        
        if it % 50 == 0:
            beta = _adaptive_beta(c)
        
        cm = float(c.max())
        if cm < 1e-30:
            break
        
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        
        a_rev = a[::-1]
        w_conv = _full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if it % 10 == 0:
            sc = _score_norm_fn(a)
            a_check = _softmax_fn(z)
            sc = _score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)
        
        if t % 20000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score

def _peak_shave(a, deadline, rng, max_rounds=500):
    """Aggressive peak shaving: identify top peaks and reduce contributing elements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        
        # Find top peaks
        top_k = min(10, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices[:5]:
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:10]
            
            for idx_pos in order[:5]:
                idx = valid_i[idx_pos]
                for delta in [0.002, 0.005, 0.01, 0.02, 0.03]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            
            # Also try increasing the other element of the pair
            for idx_pos in order[:3]:
                idx = valid_i[idx_pos]
                other = pk - idx
                if 0 <= other < n and other != idx:
                    for delta in [0.002, 0.005, 0.01]:

def _make_ramp(n, slope=1.5, curvature=0.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t + curvature * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_gaussian(n, sigma=0.3, center=0.5):
    t = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a / a.sum()

def _make_cosine(n, freq=1.0, phase=0.5):
    t = np.linspace(0, 1, n)
    a = phase + (1 - phase) * np.cos(freq * np.pi * (t - 0.5))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_profiles(n, rng):
    profiles = []
    # Ramps
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        profiles.append(_make_ramp(n, s))
    # Quadratic
    for s, c in [(1.5, 0.5), (2.0, -0.3), (1.0, 1.0), (3.0, 0.5)]:
        t = np.linspace(0, 1, n)
        a = 1.0 + s * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Gaussian
    for sig in [0.2, 0.3, 0.4, 0.5]:
        profiles.append(_make_gaussian(n, sig))
    # Cosine
    for freq in [1.0, 1.5, 2.0]:
        for phase in [0.3, 0.5, 0.7]:
            profiles.append(_make_cosine(n, freq, phase))
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0]:
        t = np.linspace(0, 1, n)
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Random smooth
    for _ in range(3):
        z = rng.normal(size=n) * 0.5
        # Smooth it
        from scipy.ndimage import gaussian_filter1d
        z_smooth = gaussian_filter1d(z, sigma=5)
        profiles.append(_softmax_fn(z_smooth))
    return profiles

"""
Multi-scale optimization with aggressive continuation, subgradient refinement,
and coordinate descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
- More aggressive beta continuation (down to 1e-9 * max_c)
- Top-K subgradient averaging for smoother peak reduction
- Coordinate descent polish
- Try n=8192 if time allows
- Multiple restart strategies
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    return a * (grad_a - offset)

def _topk_grad(z, n, K=20):
    """Subgradient using top-K peaks averaged."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n), cm
    K_actual = min(K, len(c))
    top_idx = np.argsort(c)[::-1][:K_actual]
    tol = cm * 0.1  # include peaks within 10% of max
    active = np.where(c >= cm - tol)[0]
    active = np.setdiff1d(active, np.arange(len(c)), assume_unique=False)  # no wait
    # Actually just use top K
    w = np.zeros(len(c))
    w[top_idx] = 1.0 / K_actual
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    gz = a * (grad_a - offset)
    return gz, cm

def _run_lbfgs_cont(a_init, deadline, rng, n_betas=20):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_start = 0.15 * max_c0
    beta_end = 1e-8 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 300.0)
        level_dl = time.monotonic() + time_budget
        
        if beta > 0.05 * max_c0:
            maxiter = 500
        elif beta > 0.005 * max_c0:
            maxiter = 2000
        else:
            maxiter = 10000
        maxfun = maxiter * 5
        
        try:
            res = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxls': 100, 'maxfun': maxfun}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                z_new = np.clip(res.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _run_subgrad_descent(a_init, deadline, rng, lr_init=0.01, K=20):
    """Subgradient descent on true max with top-K averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Running average of gradients (Nesterov's method)
    z_avg = z.copy()
    t = 0
    lr = lr_init
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        
        t += 1
        grad, cm = _topk_grad(z, n, K=K)
        if not np.all(np.isfinite(grad)):
            break
        
        # Normalized step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-20:
            break
        grad = grad / grad_norm
        
        z_avg = z_avg + (1.0 / (t + 1)) * (z - z_avg)
        z = z_avg - lr / np.sqrt(t) * grad
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if it % 50 == 0:
            a = _softmax_fn(z)
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Decay lr
        if t % 10000 == 0:
            lr = max(0.001, lr * 0.8)
    
    return best_a, best_score
