Let me also think about the "peak targeting" more carefully.

Suppose the max of conv(a,a) is at position k*. Then:
b_{k*} = sum_j a[j] a[k*-j]

To reduce b_{k*}, we want to reduce the sum. Since all terms are non-negative, we need to reduce some a[j] values. But we need to maintain sum(a) = 1, so we increase other a[j] values.

The key insight: if we reduce a[i] by δ and increase a[m] by δ, the change in b_{k*} is:
Δb_{k*} ≈ δ * (a[k*-m] - a[k*-i]) * 2 (using the gradient formula)

Wait, more carefully:
- Reducing a[i] by δ changes b_{k*} by -δ * (gradient of b_{k*} w.r.t. a[i]) = -δ * 2*a[k*-i] (if valid)
- Increasing a[m] by δ changes b_{k*} by +δ * 2*a[k*-m] (if valid)

Total change: 2δ * (a[k*-m] - a[k*-i])

To minimize the increase (or maximize the decrease), we want a[k*-m] < a[k*-i], i.e., we want to move mass from position i to position m where a[k*-m] is smaller than a[k*-i].

This means: move mass from positions i where a[k*-i] is large to positions m where a[k*-m] is small.

In other words: move mass from positions "reflected" around k*/2 to positions where the "reflected" value is small.

This is a nice geometric interpretation! The peak at k* is sensitive to pairs of positions (i, k*-i) that both have large values. To reduce the peak, we break up such pairs by moving one of them.

**Algorithm for peak targeting:**
1. Find k* = argmax(conv(a,a))
2. For each pair (i, k*-i) where both are in range and a[i]*a[k*-i] is large:
   - Move a small amount of mass from the larger of a[i], a[k*-i] to a position where the "reflected" value is small
3. Repeat

This is essentially a form of gradient descent on the max (using the subgradient at the peak).

def next_pow2(n):

    while k < n:

    return max(2, k)

def full_convolve(a, b):

    fb = np.fft.rfft(b, N)

    s = float(a.sum())

    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP minimax iteration, log-sum-exp L-BFGS, power-sum optimization,
peak targeting with active-set identification, and coordinate descent.
Explores multiple sequence lengths and uses dual variables from LP
to guide targeted optimization.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _exp_neg_clip(x):
    return np.exp(np.clip(x, -80, 80))


def _lp_minimax(f, solver=cp.CLARABEL):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    out_n = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for k in range(out_n):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            row_idx.append(k)
            col_idx.append(j)
            data.append(f[j])
    A = csr_matrix((data, (row_idx, col_idx)), shape=(out_n, n))
    constraints = [A @ g <= t, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
        if g.value is not None and np.all(np.isfinite(g.value)):
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 0:
                g_val = g_val / gs
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate_adaptive(f, deadline, max_rounds=50):

    best_score = _score_norm(f)
    a_cur = f.copy()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_cur, a_cur)
        M = float(c.max())
        g, t_val = _lp_minimax(a_cur)
        if g is None:
            break

        # Adaptive step size: try multiple step sizes, use best
        best_step = 0.3
        best_sc = np.inf
        for step in [0.8, 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1 - step) * a_cur + step * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_sc:

                best_step = step

        if rnd % 5 == 0:

    return best_a, best_score


def _lse_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    e = np.exp(vals)
    lse = cm + beta * float(np.log(np.sum(e)))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 4.0 * n * a * (h - W)


def _powsum_obj(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    return 2.0 * n * float(np.sum(c_pos ** p) ** (1.0 / p))


def _powsum_grad(z, p, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_pos = np.maximum(c, 1e-30)
    S = float(np.sum(c_pos ** p))
    norm = S ** (1.0 / p)
    w = (c_pos ** (p - 1)) / S
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 2.0 * n * (2.0 / norm) * a * (h - W)
    return grad

def _peak_target(a_init, deadline, n_iters=200):
    """Directly target the peak of the convolution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    a = a_init.copy()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        c_max = float(c[k_star])

        # Find pairs (i, k_star - i) with large product
        # Gradient of c[k_star] w.r.t. a[i] is 2*a[k_star - i]
        # We want to reduce a[i] where a[k_star-i] is large
        # and increase a[j] where a[k_star-j] is small

        # Compute sensitivity: for each position i, how much does a[i] contribute to peak
        # contribution[i] = a[i] * a[k_star - i] (if valid)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        # Find positions to reduce (high contribution to peak)
        # and positions to increase (low contribution to peak, meaning a[k_star - j] is small)
        # We want to move mass from high-contrib to low-contrib positions

        # Score for moving mass from i to j: change in peak ≈ 2*delta*(a[k_star-j] - a[k_star-i])
        # We want a[k_star-j] < a[k_star-i], i.e., the "reflected" value at j is smaller

        # Simple heuristic: reduce the top few contributors and increase the bottom few
        sorted_idx = np.argsort(contrib)[::-1]
        n_reduce = max(1, n // 50)
        reduce_set = sorted_idx[:n_reduce]

        # For increase positions, pick those where a[k_star - j] is smallest
        reflected = np.zeros(n)
        for j in range(n):
            i = k_star - j
            if 0 <= i < n:
                reflected[j] = a[i]
            else:
                reflected[j] = 0.0

        sorted_inc = np.argsort(reflected)
        n_incr = n_reduce
        incr_set = sorted_inc[:n_incr]

        # Exclude overlap
        mask = np.ones(n, dtype=bool)
        mask[reduce_set] = False
        incr_set = incr_set[mask[incr_set]] if len(incr_set) > 0 else sorted_inc[n_reduce:n_reduce + n_incr]
        if len(incr_set) == 0:
            incr_set = sorted_inc[n_reduce:n_reduce + n_incr]
        incr_set = incr_set[:n_incr]

        # Determine amount to move
        total_reduce = 0.0
        for i in reduce_set:
            total_reduce += a[i] * 0.05
        total_reduce = min(total_reduce, 0.01)
        if total_reduce < 1e-10:
            break

        # Move mass
        for i in reduce_set:
            a[i] -= total_reduce / len(reduce_set)
        for j in incr_set:
            if len(incr_set) > 0:
                a[j] += total_reduce / len(incr_set)

        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=30):

def _adaptive_gd(a_init, deadline, rng, lr_init=0.15, max_iter=50000):

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_init_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    # Flat
    profiles.append(np.ones(n) / n)
    # Linear tapers
    for slope in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Cosine-like
    for p in [1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * (t - 0.5)), 1e-12) ** p
        profiles.append(a / a.sum())
    # V-shaped
    a = np.abs(t - 0.5) + 0.1
    profiles.append(a / a.sum())
    return profiles

    n_lp = min(n_parent, 2048)
    if n_parent > n_lp:
        a_lp = _interp_profile(parent, n_lp)
    else:
        a_lp = parent.copy()
    sc_lp = _score_norm(a_lp)
    update_best(a_lp, sc_lp, f"lp_start_n{n_lp}")

    # Run LP with decreasing "aggressiveness"
    cur = a_lp.copy()
    for lp_phase in range(3):
        if time.monotonic() > phase1_end:
            break
        cur, sc_cur = _lp_iterate_adaptive(cur, phase1_end, max_rounds=25)
        update_best(cur, sc_cur, f"lp_p{lp_phase}_n{n_lp}")
        # Interpolate back to parent size
        if n_parent > n_lp:
            rec = _interp_profile(cur, n_parent)
            sc_rec = _score_norm(rec)
            update_best(rec, sc_rec, f"lp_rec_n{n_parent}")

    # Also try LP at full n if time permits
    if time.monotonic() < phase1_end - 30 and n_parent <= 4096:
        cur_full = _interp_profile(cur, n_parent) if n_parent > n_lp else cur.copy()
        cur_full, sc_full = _lp_iterate_adaptive(cur_full, phase1_end, max_rounds=10)
        update_best(cur_full, sc_full, f"lp_full_n{n_parent}")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: LSE Continuation (500s) ===
    phase2_end = min(deadline, t_start + 850.0)
    print(f"\n=== Phase 2: LSE Continuation ===")

    n_opt = len(best_seq)
    a_start = best_seq.copy()

    # Run LSE continuation from best
    per_dl = min(phase2_end - 30, time.monotonic() + 350.0)
    p2_best, p2_sc = _run_lse_continuation(a_start, per_dl)
    update_best(p2_best, p2_sc, "lse_main")

    # Try from a perturbed version
    if time.monotonic() < phase2_end - 60:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.03 * rng.normal(size=n_opt)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_pert = _softmax(z)
        per_dl2 = min(phase2_end - 20, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
        p2b_best, p2b_sc = _run_lse_continuation(a_pert, per_dl2)
        update_best(p2b_best, p2b_sc, "lse_perturbed")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Power-sum + Peak Targeting (300s) ===
    phase3_end = min(deadline, t_start + 1150.0)
    print(f"\n=== Phase 3: Power-sum + Peak Targeting ===")

    # Power-sum optimization
    if time.monotonic() < phase3_end - 100:
        per_dl = min(phase3_end - 100, time.monotonic() + 200.0)
        p3_best, p3_sc = _run_powsum_opt(best_seq, per_dl)
        update_best(p3_best, p3_sc, "powsum")

    # Peak targeting
    if time.monotonic() < phase3_end:
        per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.8)
        p3b_best, p3b_sc = _peak_target(best_seq, per_dl, n_iters=500)
        update_best(p3b_best, p3b_sc, "peak_target")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Adaptive GD + Fine LSE (200s) ===
    phase4_end = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 4: Fine Optimization ===")

    if time.monotonic() < phase4_end:
        per_dl = min(phase4_end - 80, time.monotonic() + 120.0)
        p4_best, p4_sc = _adaptive_gd(best_seq, per_dl, rng, lr_init=0.1)
        update_best(p4_best, p4_sc, "adaptive_gd")

    # Very fine LSE with tiny beta
    if time.monotonic() < phase4_end - 20:
        n_f = len(best_seq)
        c_f = _full_convolve(best_seq, best_seq)
        max_cf = float(c_f.max())
        beta_tiny = 5e-6 * max_cf
        z_f = np.log(np.maximum(best_seq, 1e-15))
        z_f = z_f - z_f.mean()
        per_dl = min(phase4_end - 10, time.monotonic() + 60.0)
        try:
            result = minimize(
                _lse_obj, z_f, args=(beta_tiny, n_f),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-16,
                         'maxls': 30, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_f = _softmax(result.x)
                sc_f = _score_norm(a_f)
                if np.isfinite(sc_f) and sc_f < best_score:
                    update_best(a_f, sc_f, "tiny_lse")
        except Exception:
            pass

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Try larger/smaller n (150s) ===
    phase5_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 5: Multi-n Exploration ===")

    current_n = len(best_seq)

    # Try n = 8192 if current is 4096
    if current_n == 4096 and time.monotonic() < phase5_end - 60:
        a_8192 = _interp_profile(best_seq, 8192)
        sc_8192 = _score_norm(a_8192)
        update_best(a_8192, sc_8192, "interp_8192")
        if time.monotonic() < phase5_end - 40:
            per_dl = min(phase5_end - 20, time.monotonic() + 80.0)
            p5_best, p5_sc = _run_lse_continuation(a_8192, per_dl,
                beta_schedule=[0.02*float(_full_convolve(a_8192,a_8192).max()),
                               0.01*float(_full_convolve(a_8192,a_8192).max()),
                               0.005*float(_full_convolve(a_8192,a_8192).max())])
            update_best(p5_best, p5_sc, "opt_8192")

    # Try n = 2048 if current is 4096 or 8192
    if current_n >= 4096 and time.monotonic() < phase5_end - 30:
        a_2048 = _interp_profile(best_seq, 2048)
        sc_2048 = _score_norm(a_2048)
        update_best(a_2048, sc_2048, "down_2048")
        if time.monotonic() < phase5_end - 20:
            per_dl = min(phase5_end - 10, time.monotonic() + 30.0)
            p5b_best, p5b_sc = _run_lse_continuation(a_2048, per_dl,
                beta_schedule=[0.01*float(_full_convolve(a_2048,a_2048).max()),
                               0.005*float(_full_convolve(a_2048,a_2048).max())])
            update_best(p5b_best, p5b_sc, "opt_2048")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Coordinate descent polish (remaining time) ===
    phase6_end = deadline
    print(f"\n=== Phase 6: Coordinate Descent Polish ===")

    if time.monotonic() < phase6_end and best_seq is not None:
        per_dl = min(phase6_end - 15, time.monotonic() + (phase6_end - time.monotonic()) * 0.7)
        p6_best, p6_sc = _coord_descent(best_seq, per_dl, max_rounds=40)
        update_best(p6_best, p6_sc, "coord_cd")

        # Final tiny-beta LSE
        if time.monotonic() < phase6_end - 10:
            n_f = len(best_seq)
            c_f = _full_convolve(best_seq, best_seq)
            max_cf = float(c_f.max())
            beta_ultra = 2e-6 * max_cf
            z_f = np.log(np.maximum(best_seq, 1e-15))
            z_f = z_f - z_f.mean()
            per_dl2 = min(phase6_end - 5, time.monotonic() + 25.0)
            try:
                result = minimize(
                    _lse_obj, z_f, args=(beta_ultra, n_f),
                    method='L-BFGS-B', jac=_lse_grad,
                    options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxls': 20, 'maxfun': 10000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_f = _softmax(result.x)
                    sc_f = _score_norm(a_f)
                    if np.isfinite(sc_f) and sc_f < best_score:
                        update_best(a_f, sc_f, "ultra_lse")
            except Exception:
                pass

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Final: One more round of peak targeting on the best ===
    if time.monotonic() < deadline - 5 and best_seq is not None:
        per_dl = min(deadline - 2, time.monotonic() + 10.0)
        p_final, sc_final = _peak_target(best_seq, per_dl, n_iters=100)
        update_best(p_final, sc_final, "final_peak")

    # Validate and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_init_profiles(n_fb)[0]
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n={n_fb}, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
