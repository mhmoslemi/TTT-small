"""

1. Adaptive top-K peak objective with L-BFGS continuation
2. Structured mass-transfer local search exploiting convolution structure

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N, s=N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - separation / 2.0
    c2 = 0.5 + separation / 2.0
    a = np.exp(-(t - c1)**2 / (2*width**2)) + np.exp(-(t - c2)**2 / (2*width**2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_u_profile(n, sharpness):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + sharpness * u**2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _topk_objective(z, n, K, beta):
    """Objective: beta * log(sum(exp(c_k/beta) for top-K) / K) + (1-beta_norm)*max_c"""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    if K >= len(c):
        return 2.0 * n * max_c
    # Smooth top-K: use log-sum-exp as smooth max over all, but weight top-K more
    # Actually just use log-sum-exp with aggressive beta for focusing
    c_shifted = (c - max_c) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * max_c
    log_sum = float(np.log(s))
    return 2.0 * n * (max_c + beta * log_sum)


def _topk_gradient(z, n, K, beta):
    """Gradient of the top-K smooth objective."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    c_shifted = (c - max_c) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g


def _run_lbfgs_topk(a_init, deadline, beta_starts, per_level_time=30.0):
    """L-BFGS with top-K smooth continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    for beta_frac in beta_starts:
        if time.monotonic() > deadline - 2.0:
            break
        beta = beta_frac * max_c0
        level_deadline = min(deadline, time.monotonic() + per_level_time)

        def obj(z_):
            return _topk_objective(z_, n, len(z_), beta)

        def grad(z_):
            return _topk_gradient(z_, n, len(z_), beta)

        try:
            result = minimize(
                obj, z.copy(), jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxls': 100, 'maxfun': 20000, 'maxiter': 3000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
                else:
                    # Check if z improved the smooth objective at least
                    a_check = _softmax_fn(z_new)
                    sc_check = _score_norm_fn(a_check)
                    if np.isfinite(sc_check) and sc_check < best_score * 1.001:
                        z = z_new.copy()
                        if sc_check < best_score:
                            best_score = sc_check
                            best_a = a_check.copy()
        except Exception:
            pass
    return best_a, best_score


def _multiplicative_weights(a_init, deadline, n_iters=50000, K=10, lr=0.01):
    """Mirror descent (multiplicative weights) on the simplex targeting top-K peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        max_c = float(c.max())

        # Get top K lags
        K_actual = min(K, len(c))
        top_idx = np.argsort(c)[-K_actual:]
        # Weight: emphasize peaks more
        top_vals = c[top_idx]
        w = np.zeros(len(c))
        w[top_idx] = (top_vals / max_c) ** 4  # Sharpen weights toward peaks
        w = w / w.sum()

        # Compute pressure: p_i = sum_k w_k * a_{k-i} (circular part, truncated)
        w_rev_trimmed = w[n-1:2*n-1]  # This is the "flat" part
        # Actually compute pressure properly
        # pressure[i] = sum_k w_k * a_{k-i} for valid k-i
        # Use convolution: pressure = convolve(w, a_reversed)[n-1:2n-1]
        a_rev = a[::-1]
        p_conv = _full_convolve(w, a_rev)
        pressure = p_conv[n-1:2*n-1]

        # Update: z_i -= lr * (pressure_i - mean(pressure))
        avg_p = pressure.mean()
        z = z - lr * (pressure - avg_p)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        # Anneal learning rate
        if it % 5000 == 0 and it > 0:
            lr *= 0.85

        if it % 10 == 0:
            a_curr = _softmax_fn(z)
            sc = _score_norm_fn(a_curr)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_curr.copy()

    return best_a, best_sc


def _structured_mass_transfer(a_init, deadline, n_trials=50000, K_peaks=8):

    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    rng = np.random.default_rng(12345)

    for it in range(n_trials):

            break

        c = _full_convolve(a, a)
        K = min(K_peaks, len(c))
        top_lags = np.argsort(c)[-K:]

        # Strategy 1: Move mass from high-product pairs at peak lags
        if it % 4 == 0:
            # Pick a random peak lag
            lag = int(top_lags[rng.integers(len(top_lags))])
            i_min = max(0, lag - (n - 1))
            i_max = min(n - 1, lag)
            if i_min > i_max:
                continue
            pairs = np.arange(i_min, i_max + 1)
            partners = lag - pairs
            products = a[pairs] * a[partners]
            # Pick the pair with largest product (or random among top few)
            top_pair_idx = np.argsort(products)[-min(3, len(products)):]
            idx = pairs[top_pair_idx[rng.integers(len(top_pair_idx))]]

            # Reduce this position
            frac = rng.uniform(0.05, 0.25)
            delta = a[idx] * frac

            # Find safe position: low contribution to all top lags
            contrib = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    partners_kl = kl - np.arange(lo, hi + 1)
                    contrib[lo:hi+1] += a[partners_kl]
            
            # Add small random component for diversity
            contrib += 0.001 * rng.random(n)
            safe_sorted = np.argsort(contrib)
            # Pick from bottom 20%
            n_safe = max(1, n // 5)
            add_idx = int(safe_sorted[rng.integers(n_safe)])

            a_trial = a.copy()
            a_trial[idx] -= delta
            a_trial[idx] = max(0.0, a_trial[idx])
            a_trial[add_idx] += delta
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

            a_trial = a.copy()
            mult = rng.uniform(0.75, 1.25, size=block_size)

            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:

                best_a = a_trial.copy()
                a = a_trial.copy()

        # Strategy 3: Pair swap with annihilation
        elif it % 4 == 2:
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt = min(a_trial[i1], a_trial[i2]) * frac

            a_trial[i2] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

def _spectral_smooth(a_init, deadline, n_iters=200, alpha=0.1):
    """Iterative spectral smoothing: adjust magnitude spectrum to flatten autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    N = _next_pow2(2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break

        A = rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # The autocorrelation is IFFT(|A|^2). We want it flat.
        # Power spectrum P = |A|^2
        P = mag ** 2

        # Compute current autocorrelation from spectrum
        acorr = irfft(P, N, s=N)[:2*n-1]
        max_ac = float(acorr.max())
        min_ac = float(acorr.min())

        if max_ac < 1e-15:
            break

        # Desired: flat autocorrelation => flat power spectrum (in some sense)
        # But we want to reduce peaks in acorr, which means reducing P at frequencies
        # that constructively interfere at the peak lags.
        # Simpler: apply a soft-clip to the power spectrum
        # Target: P_new = P * (1 - alpha * (acorr_peak_contribution - mean))
        # Approximate: reduce P where it's large (corresponds to low-freq energy)
        P_mean = P.mean()
        P_ratio = P / max(P_mean, 1e-30)
        # Soft compression
        P_compressed = P * (1.0 - alpha * np.maximum(0, P_ratio - 1.0) / (P_ratio.max() - 1.0 + 1e-10))
        P_compressed = np.maximum(P_compressed, 1e-30)

        mag_new = np.sqrt(P_compressed)

        # Reconstruct: a_new = IFFT(mag_new * exp(j*phase))
        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, N, s=N)[:n]

        # Project to non-negative simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()

        # Anneal alpha
        if it % 50 == 0 and it > 0:
            alpha *= 0.8

    return best_a, best_sc


def _adam_topk(a_init, deadline, K=5, lr=0.015, n_iters=100000):
    """Adam optimizer with top-K peak targeting."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1

        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        max_c = float(c.max())

        # Top-K gradient
        K_actual = min(K, len(c))
        top_idx = np.argsort(c)[-K_actual:]
        # Weight top K lags
        w = np.zeros(len(c))
        top_vals = c[top_idx]
        w[top_idx] = 1.0 / K_actual

        W = float(np.dot(w, c))
        a_rev = a_curr[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n-1:2*n-1]
        grad = 4.0 * n * a_curr * (h - W)

        # Smooth the gradient slightly
        if not np.all(np.isfinite(grad)):
            grad = np.zeros(n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)

        # Anneal learning rate
        lr_t = lr * (0.95 ** (t // 5000))
        z = z - lr_t * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

    return best_a, best_sc


def _multi_resolution_polish(a_init, deadline, rng):
    """Optimize at coarser resolution, then refine at full resolution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score_norm_fn(a_init)

    # Try several coarse resolutions
    for coarse_factor in [4, 8]:
        n_coarse = max(128, n // coarse_factor)
        if n_coarse >= n:
            continue

        a_coarse = _interp_profile(a_init, n_coarse)
        sc_coarse = _score_norm_fn(a_coarse)

        # Optimize at coarse level

        # Upsample and compare

        if sc_up < best_sc:
            best_sc = sc_up
            best_a = a_up.copy()

            a_best2, sc_best2 = _run_lbfgs_topk(parent, phase1_end - 10, betas2, per_level_time=25.0)
            update_best(a_best2, sc_best2, "lbfgs_parent")

    print(f"Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Multiplicative weights / Mirror descent (200s) ===
    phase2_end = min(deadline, t_start + 530.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"Phase 2: Multiplicative weights, start score={best_score:.10f}")
        dl = phase2_end - 5.0
        a_best, sc_best = _multiplicative_weights(best_seq, dl, n_iters=30000, K=8, lr=0.008)
        update_best(a_best, sc_best, "mirror_descent")

        # Second pass with different K
        if time.monotonic() < phase2_end - 30:
            a_best2, sc_best2 = _multiplicative_weights(best_seq, phase2_end - 10, n_iters=20000, K=16, lr=0.005)
            update_best(a_best2, sc_best2, "mirror_K16")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Adam top-K (200s) ===
    phase3_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"Phase 3: Adam top-K, start score={best_score:.10f}")
        dl = phase3_end - 5.0
        a_best, sc_best = _adam_topk(best_seq, dl, K=5, lr=0.012, n_iters=80000)
        update_best(a_best, sc_best, "adam_K5")

        if time.monotonic() < phase3_end - 40:
            a_best2, sc_best2 = _adam_topk(best_seq, phase3_end - 10, K=12, lr=0.008, n_iters=50000)
            update_best(a_best2, sc_best2, "adam_K12")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Structured local search (300s) ===
    phase4_end = min(deadline, t_start + 1030.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: Structured local search, start score={best_score:.10f}")
        dl = phase4_end - 5.0
        a_best, sc_best = _structured_mass_transfer(best_seq, dl, n_trials=40000, K_peaks=6)
        update_best(a_best, sc_best, "local_search_1")

        # Second round with more peaks
        if time.monotonic() < phase4_end - 30:
            a_best2, sc_best2 = _structured_mass_transfer(best_seq, phase4_end - 10, n_trials=25000, K_peaks=12)
            update_best(a_best2, sc_best2, "local_search_2")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Spectral smoothing (150s) ===
    phase5_end = min(deadline, t_start + 1180.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"Phase 5: Spectral smoothing, start score={best_score:.10f}")
        dl = phase5_end - 5.0
        a_best, sc_best = _spectral_smooth(best_seq, dl, n_iters=100, alpha=0.05)
        update_best(a_best, sc_best, "spectral_1")

        if time.monotonic() < phase5_end - 20:
            a_best2, sc_best2 = _spectral_smooth(best_seq, phase5_end - 10, n_iters=80, alpha=0.15)
            update_best(a_best2, sc_best2, "spectral_2")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Multi-resolution polish (150s) ===
    phase6_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print(f"Phase 6: Multi-resolution, start score={best_score:.10f}")
        dl = phase6_end - 5.0
        a_best, sc_best = _multi_resolution_polish(best_seq, dl, rng)
        update_best(a_best, sc_best, "multi_res")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Intensive fine-tuning (remaining time) ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Fine-tuning, remaining={remaining:.0f}s, score={best_score:.10f}")

        # 7a: Very fine L-BFGS
        if remaining > 80:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            betas_fine = [0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001]
            a_best, sc_best = _run_lbfgs_topk(best_seq, dl, betas_fine, per_level_time=12.0)
            update_best(a_best, sc_best, "final_lbfgs")

        # 7b: More local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.3)
            a_best, sc_best = _structured_mass_transfer(best_seq, dl, n_trials=20000, K_peaks=4)
            update_best(a_best, sc_best, "final_local")

        # 7c: Fine Adam
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.25)
            a_best, sc_best = _adam_topk(best_seq, dl, K=3, lr=0.005, n_iters=40000)
            update_best(a_best, sc_best, "final_adam")

        # 7d: Final mirror descent
        if time.monotonic() < deadline - 15:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.15)
            a_best, sc_best = _multiplicative_weights(best_seq, dl, n_iters=15000, K=4, lr=0.003)
            update_best(a_best, sc_best, "final_mirror")

    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular_weight(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
