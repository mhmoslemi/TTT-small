Power-law ramp optimizer.
Starts from power-law profiles a_i ∝ (i+1)^p, finds optimal p via sweep,
then refines with L-BFGS logsumexp continuation and peak-targeted local search.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _fft_convolve(a, b):
    """Fast full convolution using FFT."""
    n, m = len(a), len(b)
    out_len = n + m - 1
    N = 1
    while N < out_len:
        N <<= 1

def _make_power_ramp(n, p):
    """a_i ∝ (i+1)^p, normalized to sum 1."""
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** p
    return a / a.sum()

def _make_clipped_ramp(n, p, clip_frac):
    """a_i ∝ min((i+1)^p, (n*clip_frac)^p), normalized."""
    i = np.arange(1, n+1, dtype=np.float64)
    cap = (n * clip_frac) ** p
    a = np.minimum(i ** p, cap)
    return a / a.sum()

def _make_offset_ramp(n, p, offset):
    """a_i ∝ (i+1)^p + offset, normalized."""
    i = np.arange(1, n+1, dtype=np.float64)

    e = np.exp(shifted)
    s = float(np.sum(e))
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s))


def _make_power(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p
    return a / a.sum()


def _make_clipped(n, p, clip_frac):
    i = np.arange(1, n + 1, dtype=np.float64)
    cap = (n * clip_frac) ** p
    a = np.minimum(i ** p, cap)
    return a / a.sum()


def _make_offset(n, p, off):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** p + off
    return a / a.sum()


def _make_two_segment(n, p1, p2, split):
    i = np.arange(1, n + 1, dtype=np.float64)
    split_idx = int(n * split)
    a = np.empty(n)
    a[:split_idx] = (np.arange(1, split_idx + 1)) ** p1
    a[split_idx:] = (split_idx + 1) ** p1 * (np.arange(1, n - split_idx + 1)) ** p2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_refine(a_init, deadline, beta_scale=1.0, max_wall=100.0):
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    c0 = _fft_conv(a_init, a_init)
    max_c0 = float(np.max(c0))

    betas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * beta_scale for b in betas]

    per_level = max(2.0, max_wall / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_end = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax(z_)
            return _logsumexp_score(a_, beta)

        def grad(z_):
            a_ = _softmax(z_)
            c_ = _fft_conv(a_, a_)
            cm_ = float(np.max(c_))
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = float(np.sum(e_))
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            # h[i] = sum_k w[k] * a[k-i] for i=0..n-1
            # = IFFT(FFT(w) * conj(FFT(a_padded)))[:n]
            N2 = 1
            while N2 < 3 * n:
                N2 <<= 1
            Wa = np.fft.rfft(w, N2)
            Aa = np.fft.rfft(a_, N2)
            h = np.fft.irfft(Wa * np.conj(Aa), N2)[:n]
            g_a = 4.0 * n * beta * h
            grad_z = a_ * (g_a - float(np.sum(g_a)) * a_)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_)
            return grad_z

        try:
            res = minimize(
                obj, z0.copy(), jac=grad,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxfs': 50, 'maxfun': 10000}
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax(res.x)
                sc = _score(a_new)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_new.copy()
                    z0 = res.x.copy()
        except Exception:
            pass

    return best_a, best_sc


def _peak_local_search(a_init, deadline, rng, n_trials=50000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _fft_conv(a, a)
        max_idx = int(np.argmax(c))

        if it % 4 == 0:
            # Reduce peak contributor
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            if lo > hi:
                continue
            js = np.arange(lo, hi + 1)
            partners = max_idx - js
            prods = a[js] * a[partners]
            peak_j = int(js[np.argmax(prods)])
            peak_p = int(max_idx - peak_j)

            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.4)
            amt = a_trial[peak_j] * frac
            a_trial[peak_j] = max(0.0, a_trial[peak_j] - amt)

            # Compute safety score (contribution to top lags)
            top_k = min(5, len(c))
            top_lags = np.argsort(c)[-top_k:]
            danger = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                l = max(0, kl - (n - 1))
                h = min(n - 1, kl)
                for j in range(l, h + 1):
                    p = kl - j
                    if 0 <= p < n:
                        danger[j] += a[p]
            safe = np.argsort(danger)[:max(1, n // 4)]
            add_idx = int(safe[rng.integers(len(safe))])
            a_trial[add_idx] += amt
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        elif it % 4 == 1:
            # Block perturbation
            blen = int(rng.integers(3, min(32, n // 4)))
            start = int(rng.integers(0, n - blen))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=blen)
            a_trial[start:start + blen] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        elif it % 4 == 2:
            # Pair swap
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.05, 0.3)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2)
            a_trial[i1] += amt2
            a_trial[i2] += amt1
            a_trial /= a_trial.sum()
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        else:
            # Mirror descent step (power iteration)
            c = _fft_conv(a, a)
            top_k = min(3, len(c))
            top_lags = set(int(x) for x in np.argsort(c)[-top_k:])
            h = np.zeros(n)
            for kl in top_lags:
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                for j in range(lo, hi + 1):
                    p = kl - j
                    h[j] += a[p]
            h /= len(top_lags)
            eta = 0.3
            a_trial = a * np.exp(-eta * h)
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_sc


def _adam_refine(a_init, deadline, rng, lr=0.01, k_top=5, max_iters=30000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_sc = _score(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1
        a_curr = _softmax(z)
        c = _fft_conv(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        h = np.zeros(n)
        for kl in top_idx:
            kl = int(kl)
            lo = max(0, kl - (n - 1))
            hi = min(n - 1, kl)
            for j in range(lo, hi + 1):
                p = kl - j
                h[j] += a_curr[p]
        h /= k_actual
        h = h - h.mean()
        g = 4.0 * n * a_curr * h

        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 10 == 0:
            a_new = _softmax(z)
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()

        if t % 3000 == 0:
            lr = max(0.001, lr * 0.9)

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_work = 8192
    if parent is not None:
        update_best(parent, _score(parent), "parent")
        n_work = len(parent)
        print(f"Parent: n={n_work}, score={best_score:.10f}")
    else:
        n_work = 8192
        print(f"No parent, using n={n_work}")

    # === Phase 1: Profile sweep (30s) ===
    phase1_end = min(deadline, t_start + 30.0)
    print("Phase 1: Profile sweep")

    candidates = []
    for p in [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 1.75, 2.0, 2.5, 3.0, 4.0]:
        if time.monotonic() > phase1_end - 1:
            break
        a = _make_power(n_work, p)
        sc = _score(a)
        candidates.append((sc, a.copy(), f"pow_{p}"))
        update_best(a, sc, f"pow_{p}")

    for p in [1.0, 1.5, 2.0, 2.5]:
        for cf in [0.3, 0.5, 0.7]:
            if time.monotonic() > phase1_end - 1:
                break
            a = _make_clipped(n_work, p, cf)
            sc = _score(a)
            candidates.append((sc, a.copy(), f"clip_{p}_{cf}"))
            update_best(a, sc, f"clip_{p}_{cf}")

    for p in [1.0, 1.5, 2.0]:
        for off in [0.01, 0.05, 0.1, 0.2]:
            if time.monotonic() > phase1_end - 1:
                break
            a = _make_offset(n_work, p, off)
            sc = _score(a)
            candidates.append((sc, a.copy(), f"off_{p}_{off}"))
            update_best(a, sc, f"off_{p}_{off}")

    for p1 in [1.0, 1.5, 2.0]:
        for p2 in [0.5, 0.0]:
            for sp in [0.3, 0.5, 0.7]:
                if time.monotonic() > phase1_end - 1:
                    break
                a = _make_two_segment(n_work, p1, p2, sp)
                sc = _score(a)
                candidates.append((sc, a.copy(), f"2seg_{p1}_{p2}_{sp}"))
                update_best(a, sc, f"2seg_{p1}_{p2}_{sp}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f} [{candidates[0][2] if candidates else 'none'}]")

    # Use the best candidate for refinement
    start_a = candidates[0][1].copy() if candidates else _make_power(n_work, 2.0)
    if best_seq is not None:
        start_a = best_seq.copy()

    # === Phase 2: L-BFGS refinement (400s) ===
    phase2_end = min(deadline, t_start + 430.0)
    print(f"Phase 2: L-BFGS, start score={_score(start_a):.10f}")

    a_best, sc_best = _lbfgs_refine(start_a, phase2_end - 5, beta_scale=1.0,
                                     max_wall=200.0)
    update_best(a_best, sc_best, "lbfgs_1")

    # Second pass with smaller beta
    if time.monotonic() < phase2_end - 30:
        a_b2, sc_b2 = _lbfgs_refine(best_seq, phase2_end - 5, beta_scale=0.4,
                                     max_wall=120.0)
        update_best(a_b2, sc_b2, "lbfgs_2")

    print(f"Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak-targeted local search (300s) ===
    phase3_end = min(deadline, t_start + 730.0)
    if time.monotonic() < phase3_end:
        print("Phase 3: Peak local search")
        a_best, sc_best = _peak_local_search(best_seq, phase3_end - 5, rng, n_trials=80000)
        update_best(a_best, sc_best, "local_search")
    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Adam refinement (200s) ===
    phase4_end = min(deadline, t_start + 930.0)
    if time.monotonic() < phase4_end:
        print("Phase 4: Adam")
        a_best, sc_best = _adam_refine(best_seq, phase4_end - 5, rng, lr=0.008, k_top=3)
        update_best(a_best, sc_best, "adam_k3")
        if time.monotonic() < phase4_end - 20:
            a_b2, sc_b2 = _adam_refine(best_seq, phase4_end - 5, rng, lr=0.005, k_top=7)
            update_best(a_b2, sc_best, "adam_k7")
    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: L-BFGS on best from local search (250s) ===
    phase5_end = min(deadline, t_start + 1180.0)
    if time.monotonic() < phase5_end:
        print("Phase 5: L-BFGS polish")
        a_best, sc_best = _lbfgs_refine(best_seq, phase5_end - 5, beta_scale=0.3,
                                         max_wall=180.0)
        update_best(a_best, sc_best, "lbfgs_3")
        if time.monotonic() < phase5_end - 20:
            a_b2, sc_b2 = _lbfgs_refine(best_seq, phase5_end - 5, beta_scale=0.15,
                                         max_wall=60.0)
            update_best(a_b2, sc_b2, "lbfgs_4")
    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Multi-restart (200s) ===
    phase6_end = min(deadline, t_start + 1380.0)
    if time.monotonic() < phase6_end:
        print("Phase 6: Multi-restart")
        n_restarts = 4
        per_restart = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax(z)
            dl = min(phase6_end - 5, time.monotonic() + per_restart * 0.7)
            a_best, sc_best = _lbfgs_refine(p_start, dl, beta_scale=0.5,
                                             max_wall=per_restart * 0.7)
            update_best(a_best, sc_best, f"restart_{i}")
    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final local search + mirror descent (200s) ===
    phase7_end = min(deadline, t_start + 1580.0)
    if time.monotonic() < phase7_end:
        print("Phase 7: Final polish")
        a_best, sc_best = _peak_local_search(best_seq, phase7_end - 5, rng, n_trials=50000)
        update_best(a_best, sc_best, "final_local")
        if time.monotonic() < phase7_end - 10:
            a_b2, sc_b2 = _adam_refine(best_seq, phase7_end - 5, rng, lr=0.003, k_top=2)
            update_best(a_b2, sc_b2, "final_adam")
    print(f"Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Final intensive L-BFGS (remaining) ===
    if time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic() - 30)
        print(f"Phase 8: Final L-BFGS, remaining={remaining:.0f}s")
        a_best, sc_best = _lbfgs_refine(best_seq, deadline - 30, beta_scale=0.1,
                                         max_wall=remaining)
        update_best(a_best, sc_best, "final_lbfgs")
        if time.monotonic() < deadline - 15:
            a_b2, sc_b2 = _peak_local_search(best_seq, deadline - 15, rng, n_trials=20000)
            update_best(a_b2, sc_b2, "very_final_local")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_power(4096, 2.0)
            s = final_seq.sum()
        final_seq = final_seq / s

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={float(final_seq.sum()):.6f}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_power(8192, 2.0)
    sc_fb = _score(fallback)
    print(f"FALLBACK: n=8192, p=2.0, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
