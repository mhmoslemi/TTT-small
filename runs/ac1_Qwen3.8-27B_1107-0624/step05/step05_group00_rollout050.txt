def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    n_full = n_a + n_b - 1
    N = _next_pow2(n_full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_full]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):

    n = len(a)
    c = _full_convolve(a, a)

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_softmax_weights(c, beta):

    return w, cm

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        center = 0.5
        sigma = 0.25
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a = np.maximum(a, 1e-15)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _smooth_perturb(a, sigma, rng, kernel_size=5):
    """Perturb with smooth noise to avoid high-frequency artifacts."""
    z = np.log(np.maximum(a, 1e-15))
    noise = rng.standard_normal(len(a))
    # Smooth the noise

    noise = np.convolve(noise, kernel, mode='same')
    z = z + sigma * noise
    z = z - z.mean()
    return _softmax_compute(z)

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    n_full = n_a + n_b - 1
    N = _next_pow2(n_full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_full]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_weights(c, beta):
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w, cm


def _grad_simplex(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _grad_logspace(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

        # Subgradient: grad_j = 2 * a[k_star - j] (for valid j)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * a[idx]
            idx2 = j - k_star
            if 0 <= idx2 < n:
                pass  # already handled by symmetry since conv is symmetric
        # Actually for self-convolution: d/d a_j (a*a)(k*) = a_{k*-j} + a_{j-k*} but since (a*a)(k) = sum_i a_i a_{k-i},

                v += a[j - k_star]
            grad[j] = 2.0 * v

        print(f"No parent, tapered n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"max_conv={max_c0:.10f}")

    # === Phase 1: FISTA coarse-to-fine ===
    print("Phase 1: FISTA adaptive-beta")
    phase1_end = min(deadline, t_start + 250.0)
    a_curr = a0.copy()

    betas_1 = []
    beta = 0.025 * max_c0
    while beta > 0.00005 * max_c0:
        betas_1.append(beta)
        beta *= 0.55

    a_curr, _ = _fista_adaptive(a_curr, phase1_end, beta_schedule=betas_1, max_iter=150000)
    a_curr = a_curr / a_curr.sum()
    sc = _score(a_curr)
    if sc < best_score:
        best_score = sc
        best_seq = a_curr.copy()
        print(f"  FISTA: score={sc:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam log-space fine ===
    print("Phase 2: Adam log-space")
    phase2_end = min(deadline, t_start + 650.0)
    c_curr = _full_convolve(a_curr, a_curr)
    max_cc = float(c_curr.max())

    betas_2 = []
    beta = 0.0015 * max_cc
    while beta > 0.000003 * max_c0:
        betas_2.append(beta)
        beta *= 0.5
    if len(betas_2) > 10:
        betas_2 = betas_2[:10]

    per_time = phase2_end - time.monotonic() - 5
    if per_time > 20:
        a_opt, _ = _adam_logspace(
            a_curr, time.monotonic() + per_time,
            betas_2, lr_init=0.012, max_iter_total=350000
        )
        sc_opt = _score(a_opt)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"  Adam: score={sc_opt:.10f}, t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy()

    # === Phase 3: Multi-restart ===
    print("Phase 3: Multi-restart")
    phase3_end = min(deadline, t_start + 1050.0)

    if time.monotonic() < phase3_end - 50:
        n_restarts = 8
        per_restart = (phase3_end - time.monotonic() - 30) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 15:
                break

            if i % 4 == 0:
                p_start = _perturb_random(a_work, 0.015 + 0.004 * i, rng)
            elif i % 4 == 1:
                p_start = _perturb_smooth(a_work, 0.03, rng, ksize=9)
            elif i % 4 == 2:
                p_start = _perturb_freq(a_work, 0.02, rng)
            else:
                shift = rng.integers(1, max(2, n_main // 512))
                shifted = np.roll(a_work, shift)
                p_start = 0.6 * a_work + 0.4 * shifted
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0008 * max_cs, 0.0004 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs]

            p_best, _ = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                bs, lr_init=0.008, max_iter_total=70000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                a_work = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Exact subgradient polish ===
    print("Phase 4: Exact subgradient polish")
    phase4_end = min(deadline, t_start + 1350.0)

    if time.monotonic() < phase4_end - 10 and best_seq is not None:
        a_sg = best_seq.copy()
        a_sg = a_sg / a_sg.sum()
        for attempt in range(3):
            if time.monotonic() > phase4_end - 5:
                break
            a_sg, _ = _exact_subgrad_step(a_sg, lr=0.002, n_steps=200)
            a_sg = a_sg / a_sg.sum()
            sc_sg = _score(a_sg)
            if sc_sg < best_score:
                best_score = sc_sg
                best_seq = a_sg.copy()
                print(f"  Subgrad {attempt}: score={sc_sg:.10f}")

    # === Phase 5: Try n=8192 ===
    print("Phase 5: n=8192")
    phase5_end = min(deadline, t_start + 1650.0)

    if time.monotonic() < phase5_end - 50 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)
        sc_big = _score(a_big)
        print(f"  Interp n=8192: score={sc_big:.10f}")

        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < phase5_end - 30:
            c_big = _full_convolve(a_big, a_big)
            max_cb = float(c_big.max())
            betas_big_f = [0.004 * max_cb, 0.002 * max_cb, 0.001 * max_cb]
            a_big_opt, _ = _fista_adaptive(
                a_big, time.monotonic() + 30.0,
                beta_schedule=betas_big_f, max_iter=40000
            )
            if time.monotonic() < phase5_end - 15:
                c_tmp = _full_convolve(a_big_opt, a_big_opt)
                max_ct = float(c_tmp.max())
                betas_big_a = [0.0008 * max_ct, 0.0004 * max_ct, 0.0002 * max_ct, 0.0001 * max_ct]
                per_t = phase5_end - time.monotonic() - 5
                if per_t > 15:
                    a_big_opt, _ = _adam_logspace(
                        a_big_opt, time.monotonic() + per_t,
                        betas_big_a, lr_init=0.006, max_iter_total=80000
                    )
            sc_big_full = _score(a_big_opt)
            if sc_big_full < best_score:
                best_score = sc_big_full
                best_seq = a_big_opt.copy()
                print(f"  Optimized n=8192: score={sc_big_full:.10f}")

    # === Phase 6: Final ultra-fine polish ===
    print("Phase 6: Final polish")

    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00001 * max_cp, 0.000005 * max_cp, 0.000002 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, _ = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.002, max_iter_total=100000
            )
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Final: score={sc_pol:.10f}")

    # === Final validation ===
    print(f"Done: best_score={best_score:.10f}")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")

        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
