```python
def fista(a_init, deadline, mu_decay=0.999, mu_min=1e-6, max_iter=1000000):
    n = len(a_init)
    a = a_init.copy()
    a_prev = a_init.copy()
    t = 1.0
    mu = mu_init  # start with some initial smoothing
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        # Compute smoothed objective and its gradient
        c = _full_convolve(a, a)
        c_max = c.max()
        c_shifted = (c - c_max) / mu
        e = np.exp(c_shifted)
        w = e / e.sum()
        
        # Gradient of smoothed objective
        grad = np.zeros(n)
        # Weighted sum of gradient_k
        for k in range(2*n-1):
            if w[k] > 1e-10:
                j_start = max(0, k-(n-1))
                j_end = min(k, n-1)
                m = k - np.arange(j_start, j_end+1)
                grad[j_start:j_end+1] += 2.0 * w[k] * a[m]
        
        # Lipschitz constant of the smoothed objective
        # grad f_μ is Lipschitz with constant ~ 2/max(a) + O(1/mu)
        # Use a conservative bound
        L = 4.0 / mu  # conservative
        
        # FISTA update
        y = a_prev + ((t-1)/(t+2)) * (a_prev - a)
        a_new = y - (1.0/L) * grad
        a_prev = a.copy()
        a = _project_simplex(a_new)
        t = 1.0 + np.sqrt(1.0 + 4.0*t*t)/2.0
        
        # Decrease smoothing
        mu = max(mu_min, mu * mu_decay)
        
        # Track best
        if it % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score
```

```python
Convex optimization approach to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses FISTA (accelerated proximal gradient) with log-sum-exp smoothing,
active set refinement, and multiple initialization strategies.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    # Algorithm from Duchi et al. (2012)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _smoothed_grad(a, w, N):
    """
    Compute gradient of smoothed objective.
    a: sequence (length n)
    w: softmax weights over lags (length 2n-1)
    Returns gradient (length n)
    """
    n = len(a)
    N = _next_pow2(3 * n)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(W * np.conj(A), N)
    return 2.0 * corr[:n]

def _fista_smooth(a_init, deadline, mu_init, mu_final, mu_decay, max_iter=5000000):
    """
    FISTA with log-sum-exp smoothing.
    """
    n = len(a_init)
    a = a_init.copy()
    a_prev = a_init.copy()
    t_k = 1.0
    mu = mu_init
    best_a = a_init.copy()
    best_score = _score(a_init)
    N = _next_pow2(3 * n)
    
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        # Compute convolution
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Softmax weights
        c_shifted = (c - c_max) / mu
        e = np.exp(c_shifted)
        w = e / e.sum()
        
        # Gradient
        grad = _smoothed_grad(a, w, N)
        
        # Step size: 1/L where L ~ 2/mu (conservatively)
        L = 4.0 / mu
        step = 1.0 / L
        
        # FISTA momentum
        y = a_prev + ((t_k - 1.0) / (t_k + 2.0)) * (a_prev - a)
        a_new = y - step * grad
        a_prev = a.copy()
        a = _project_simplex(a_new)
        t_k = 1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k) / 2.0
        
        # Decrease mu
        mu = max(mu_final, mu * mu_decay)
        
        # Check score periodically
        if it % 500 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score
```
