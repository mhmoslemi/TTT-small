"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses: (1) FISTA with softmax-smoothed max, (2) Top-K subgradient with Nesterov acceleration,
(3) Adam in log-space for fine-tuning, (4) multi-restart diversification.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1, so score = 2*n*max(conv)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max of convolution."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

def _topk_nesterov(a_init, deadline, K=10, resample_interval=500):

def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    """Adam optimizer in log-space with softmax-smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.001, current_lr * 0.97)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _lbfgs_smooth(a_init, beta, deadline, max_iter=5000):
    """Use scipy L-BFGS-B on the smooth approximation with log parameters."""
    from scipy.optimize import minimize

    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def objective(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            return 2.0 * n * cm
        w = e / s
        return 2.0 * n * float(np.dot(w, c))

    def gradient(z):
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        c_shifted = (c - cm) / beta
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300:
            return np.zeros(n)
        w = e / s
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)

    result = minimize(
        objective, z0, method='L-BFGS-B',
        jac=gradient,
        options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8,
                 'maxfun': max_iter * 2}
    )

    if result.success or result.fun < objective(z0):
        a_opt = _softmax_compute(result.x)
        return a_opt, _score_norm(a_opt)
    else:
        return a_init, _score_norm(a_init)

        a_opt2 = np.maximum(a_opt2, 0)
        s = a_opt2.sum()
        if s > 1e-15:
            a_opt2 = a_opt2 / s
        sc2_full = _score(a_opt2)
        if sc2_full < best_score:
            best_score = sc2_full
            best_seq = a_opt2.copy()
            print(f"  TopK Nesterov: score={sc2_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam log-space (400s) ===
    print("Phase 3: Adam log-space fine")
    phase3_end = min(deadline, t_start + 900.0)
    a_work = best_seq.copy()
    a_work = np.maximum(a_work, 0)
    s = a_work.sum()
    if s > 1e-15:
        a_work = a_work / s

    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())
    betas_fine = []
    beta = 0.002 * max_cc
    while beta > 0.0000005 * max_c0:
        betas_fine.append(beta)
        beta *= 0.5
    if len(betas_fine) > 12:
        betas_fine = betas_fine[:12]

    per_time = phase3_end - time.monotonic() - 10
    if per_time > 20:
        a_opt3, sc3 = _adam_logspace(
            a_work, time.monotonic() + per_time,
            betas_fine, lr_init=0.015, max_iter_total=600000
        )
        sc3_full = _score(a_opt3)
        if sc3_full < best_score:
            best_score = sc3_full
            best_seq = a_opt3.copy()
            print(f"  Adam log-space: score={sc3_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: L-BFGS attempts (200s) ===
    print("Phase 4: L-BFGS")
    phase4_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase4_end - 10:
        a_l = best_seq.copy()
        a_l = np.maximum(a_l, 0)
        s = a_l.sum()
        if s > 1e-15:
            a_l = a_l / s
        c_l = _full_convolve(a_l, a_l)
        max_cl = float(c_l.max())
        beta_lb = 0.0003 * max_cl
        per_t = phase4_end - time.monotonic() - 5
        if per_t > 30:
            a_opt4, sc4 = _lbfgs_smooth(a_l, beta_lb, time.monotonic() + per_t, max_iter=3000)
            sc4_full = _score(a_opt4)
            if sc4_full < best_score:
                best_score = sc4_full
                best_seq = a_opt4.copy()
                print(f"  L-BFGS: score={sc4_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Multi-restart (300s) ===
    print("Phase 5: Multi-restart")
    phase5_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase5_end - 30:
        n_restarts = 5
        per_restart = (phase5_end - time.monotonic() - 20) / n_restarts
        a_work = best_seq.copy()
        a_work = np.maximum(a_work, 0)
        s = a_work.sum()
        if s > 1e-15:
            a_work = a_work / s
        n_main = len(a_work)

        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            if i % 5 == 0:
                p_start = _perturb(a_work, 0.025 + 0.01 * i, rng)
            elif i % 5 == 1:
                sz = max(1, n_main // 1024)
                shift = int(rng.integers(1, sz + 1))
                p_start = np.roll(a_work, shift).copy()
                p_start = np.maximum(p_start, 0)
                ps = p_start.sum()
                if ps > 1e-15:
                    p_start = p_start / ps
            elif i % 5 == 2:
                p_start = _perturb(a_work, 0.05, rng)
            elif i % 5 == 3:
                # Mix with a different profile
                p_alt = _make_profile(n_main, "sinusoidal")
                mix = 0.7 * a_work + 0.3 * p_alt
                p_start = np.maximum(mix, 1e-10)
                p_start = p_start / p_start.sum()
            else:
                # Random restart
                p_start = _make_profile(n_main, "bimodal")
                p_start = _perturb(p_start, 0.1, rng)

            sc_s = _score(p_start)
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.0015 * max_cs, 0.0008 * max_cs, 0.0004 * max_cs,
                  0.0002 * max_cs, 0.0001 * max_cs, 0.00005 * max_cs]
            bs = bs[:4]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                bs, lr_init=0.012, max_iter_total=80000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {i} (sc_start={sc_s:.6f}): score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i} (sc_start={sc_s:.6f}): no improvement (sc={sc_full:.6f}) t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Try different length (200s) ===
    print("Phase 6: Different length exploration")
    phase6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase6_end - 20 and best_seq is not None:
        # Try n = 8192 by interpolation
        n_new = len(best_seq) * 2
        if n_new <= 16384 and time.monotonic() < phase6_end - 30:
            a_new = _interp_profile(best_seq, n_new)
            sc_new = _score(a_new)
            print(f"  Interpolated n={n_new}: score={sc_new:.10f}")
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                # Optimize the new length
                if time.monotonic() < phase6_end - 20:
                    c_new = _full_convolve(a_new, a_new)
                    max_cn = float(c_new.max())
                    betas_new = [0.001 * max_cn, 0.0005 * max_cn, 0.0002 * max_cn, 0.0001 * max_cn]
                    per_t = phase6_end - time.monotonic() - 5
                    if per_t > 15:
                        a_new_opt, sc_new_opt = _adam_logspace(
                            a_new, time.monotonic() + per_t,
                            betas_new, lr_init=0.01, max_iter_total=100000
                        )
                        sc_new_full = _score(a_new_opt)
                        if sc_new_full < best_score:
                            best_score = sc_new_full
                            best_seq = a_new_opt.copy()
                            print(f"  Optimized n={n_new}: score={sc_new_full:.10f}")

        # Also try n = len(best_seq) // 2 (shorter)
        n_half = len(best_seq) // 2
        if n_half >= 256 and time.monotonic() < phase6_end - 30:
            a_half = _interp_profile(best_seq, n_half)
            sc_half = _score(a_half)
            print(f"  Interpolated n={n_half}: score={sc_half:.10f}")
            if sc_half < best_score:
                best_score = sc_half
                best_seq = a_half.copy()
                if time.monotonic() < phase6_end - 15:
                    c_half = _full_convolve(a_half, a_half)
                    max_ch = float(c_half.max())
                    betas_half = [0.001 * max_ch, 0.0005 * max_ch, 0.0002 * max_ch]
                    per_t = phase6_end - time.monotonic() - 5
                    if per_t > 10:
                        a_half_opt, sc_half_opt = _adam_logspace(
                            a_half, time.monotonic() + per_t,
                            betas_half, lr_init=0.01, max_iter_total=80000
                        )
                        sc_half_full = _score(a_half_opt)
                        if sc_half_full < best_score:
                            best_score = sc_half_full
                            best_seq = a_half_opt.copy()
                            print(f"  Optimized n={n_half}: score={sc_half_full:.10f}")

    # === Phase 7: Final ultra-fine polish (remaining time) ===
    print("Phase 7: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_pol = best_seq.copy()
        a_pol = np.maximum(a_pol, 0)
        s = a_pol.sum()
        if s > 1e-15:
            a_pol = a_pol / s

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00003 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 3
        if per_t > 10:
            a_pol_opt, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.004, max_iter_total=300000
            )
            sc_pol_full = _score(a_pol_opt)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol_opt.copy()
                print(f"  Final polish: score={sc_pol_full:.10f}")

    # === Final validation ===
    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}, time={time.monotonic()-t_start:.1f}s")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        result_list = [float(x) for x in final]
        sc_final = _score(result_list)
        print(f"Return score: {sc_final:.10f}")
        try:
            sc_eval = float(evaluate_sequence(result_list))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return result_list

    # Fallback
    fallback = _make_profile(256, "uniform")
    return [float(x) for x in fallback]
