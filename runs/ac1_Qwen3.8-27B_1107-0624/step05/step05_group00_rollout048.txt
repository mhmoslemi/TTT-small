"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FISTA for coarse optimization, Adam in log-space with adaptive beta for fine optimization,
top-k gradient at low beta for accuracy, and multi-restart with structured perturbations.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_a(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, ga, n):
    """Convert gradient w.r.t. a to gradient w.r.t. z (log-space)."""
    sz = np.dot(ga, a)
    return a * (ga - sz)


def _softmax_weights(c, beta):
    """Compute softmax weights for convolution values."""
    cm = float(c.max())
    z = (c - cm) / beta
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        mask = (c >= cm - 1e-12)
        w[mask] = 1.0 / mask.sum()
        return w
    return e / s


def _topk_weights(c, k):
    """Uniform weights on top-k peaks."""
    if k >= len(c):
        return np.ones_like(c) / len(c)
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros_like(c)
    w[idx] = 1.0 / k
    return w


def _fista_coarse(a_init, deadline, max_iter=50000):
    """FISTA on simplex with softmax-smoothed max. Coarse-to-fine beta."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c0 = float(c_init.max())
    best_score = _score_unnorm(a_curr)
    best_a = a_curr.copy()

    # Beta schedule: geometric decay
    betas = []
    beta = 0.04 * max_c0
    while beta > 0.0002 * max_c0:
        betas.append(beta)
        beta *= 0.55
    if len(betas) > 12:
        betas = betas[:12]

    for beta in betas:
        if time.monotonic() > deadline:
            break
        a_curr = a_curr / a_curr.sum()
        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())
        beta_eff = max(beta, 1e-10)

        L = 8.0 * n / beta_eff
        step = 1.0 / L
        t_val = 1.0

        for it in range(max_iter):
            if time.monotonic() > deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            w = _softmax_weights(c_y, beta_eff)
            grad = _compute_grad_a(y, w)

            a_next = _proj_simplex(y - step * grad)

            if it % 200 == 0:
                sc = _score_unnorm(a_next)
                if sc < best_score:
                    best_score = sc
                    best_a = a_next.copy()

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

    sc_final = _score_unnorm(a_curr)
    if sc_final < best_score:

        best_a = a_curr.copy()

def _make_cosine_profile(n, m=1, alpha=0.1):
    """Create a cosine-modulated profile."""
    t = np.linspace(0, 1, n, endpoint=False)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * m * t)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

                mod = 1.0 + 0.01 * np.cos(2.0 * np.pi * 3.0 * t_arr)
                p_start = a_work * mod
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()
            elif i == 2:
                # Shift perturbation
                shift = 1
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                # Slightly larger perturbation
                p_start = _perturb_logspace(a_work, 0.04, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            a_r, sc_r = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart),
                beta_start=0.0005 * max_cs,
                lr_init=0.006,
                max_iter=80000,
                beta_decay_period=1500,
                beta_decay_factor=0.6,
                lr_decay_period=5000,
                lr_decay_factor=0.95,
                topk_switch_ratio=0.0002,
                topk_k=2
            )
            sc_r_full = _score(a_r)
            print(f"  Restart {i}: score={sc_r_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = a_r.copy()
                a_work = a_r.copy()

    # === Phase 5: Ultra-fine final polish (remaining time) ===
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())

        per_t = deadline - time.monotonic() - 3
        if per_t > 10:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                beta_start=0.0001 * max_cp,
                lr_init=0.002,
                max_iter=200000,
                beta_decay_period=4000,
                beta_decay_factor=0.75,
                lr_decay_period=15000,
                lr_decay_factor=0.98,
                topk_switch_ratio=0.0001,
                topk_k=1
            )
            sc_pol_full = _score(a_pol)
            print(f"  Final polish: score={sc_pol_full:.10f}, t={time.monotonic()-t_start:.1f}s")
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()

    # === Final validation ===
    print(f"Best score: {best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
