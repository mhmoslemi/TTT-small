"""
Active-set equalization optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS warm-up followed by direct active-set linear equalization with

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _compute_active_gradients(p, active_indices, n):
    """
    Compute gradient matrix G where G[k_idx, i] = d(conv(p,p)_{active_indices[k_idx]})/dp[i]
    
    For the k-th convolution element:
    d(c_k)/dp[i] = p[k-i] (if 0<=k-i<n) + p[i-k] (if 0<=i-k<n)
    """
    L = len(active_indices)
    G = np.zeros((L, n))
    for idx, k in enumerate(active_indices):
        # First term: p[k-i] for i where 0 <= k-i < n, i.e., k-n+1 <= i <= k
        # Second term: p[i-k] for i where 0 <= i-k < n, i.e., k <= i <= k+n-1
        # Combined: for i in [0, n-1]:
        #   if i < k: G[idx, i] = p[k-i]
        #   if i == k: G[idx, i] = 2*p[0]  (but only if k < n)
        #   if i > k: G[idx, i] = p[i-k]
        #   (with bounds checking)
        
        # More carefully:
        # First term contributes to i in [max(0, k-n+1), min(n-1, k)]: G[idx, i] += p[k-i]
        # Second term contributes to i in [max(0, k), min(n-1, k+n-1)]: G[idx, i] += p[i-k]
        
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        if lo1 <= hi1:
            i_vals = np.arange(lo1, hi1 + 1)
            G[idx, i_vals] += p[k - i_vals]
        
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        if lo2 <= hi2:
            i_vals = np.arange(lo2, hi2 + 1)
            G[idx, i_vals] += p[i_vals - k]
    
    return G

def _lbfgs_optimize(a_init, beta, n, max_iter=3000):
    """Run L-BFGS-B for one beta phase."""
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={
                'maxiter': max_iter,
                'ftol': 1e-14,
                'gtol': 1e-10,
                'maxcor': 30,
                'maxls': 100,
                'disp': False,
            }
        )
        a_out = _softmax(result.x)
        return a_out
    except Exception:
        return a_init

def _try_lbfgs_schedule(a_start, deadline, min_time_remaining=50.0):

    betas = max_c * np.array([0.3, 0.1, 0.05, 0.02, 0.01, 0.005])
