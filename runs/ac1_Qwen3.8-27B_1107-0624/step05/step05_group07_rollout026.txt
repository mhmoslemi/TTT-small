"""
Hybrid optimization for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Combines LP best-response dynamics, adaptive L-BFGS with log-sum-exp smoothing,
multi-scale exploration, and coordinate descent polish.
Key improvements: diverse asymmetric initializations, adaptive LP step sizes,
finer beta continuation, and multi-n exploration.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    m = 2 * n - 1
    row_idx = []
    col_idx = []
    data = []
    for i in range(m):
        j_lo = max(0, i - n + 1)
        j_hi = min(i, n - 1)
        if j_lo > j_hi:
            continue
        cnt = j_hi - j_lo + 1
        row_idx.append(np.full(cnt, i, dtype=np.int32))
        col_idx.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
        data.append(f[i - j_hi:i - j_lo + 1])
    A = csr_matrix((np.concatenate(data), (np.concatenate(row_idx), np.concatenate(col_idx))), shape=(m, n))
    return A


def _lp_solve(f, M, max_time=None):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, sum(g)=1, g>=0. Returns g or None."""
    n = len(f)
    g_var = cp.Variable(n)
    A = _build_conv_matrix(f)
    constraints = [
        A @ g_var <= M,
        cp.sum(g_var) == 1,
        g_var >= 0
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        if max_time is not None:
            prob.solve(solver=cp.CLARABEL, max_time=max_time)
        else:
            prob.solve(solver=cp.CLARABEL)
        if g_var.value is not None and np.all(np.isfinite(g_var.value)):
            g_val = np.maximum(g_var.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                return g_val / s
    except Exception:
        pass
    return None


def _lp_minimax(f, max_time=30):
    """Find the minimum achievable max-convolution via LP."""
    n = len(f)
    g_var = cp.Variable(n)
    t_var = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g_var <= t_var,
        cp.sum(g_var) == 1,
        g_var >= 0
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, max_time=max_time)
        if g_var.value is not None and np.all(np.isfinite(g_var.value)):
            g_val = np.maximum(g_var.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
                t_val = float(t_var.value) if t_var.value is not None else None
                return g_val, t_val
    except Exception:
        pass
    return None, None


def _lp_iterate_adaptive(f, n_iters, deadline, initial_step=0.4):
    """LP best-response iterations with adaptive step size and line search."""
    n = len(f)
    a = f.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    step = initial_step

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())
        g, t_val = _lp_minimax(a, max_time=20)
        if g is None:
            break

        # Line search for step size
        best_t = step
        best_sc = _score_norm_fn(a)
        for trial_t in [step * 0.5, step * 0.7, step, step * 1.2, step * 1.5]:
            trial_t = np.clip(trial_t, 0.01, 0.95)
            a_trial = (1 - trial_t) * a + trial_t * g
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_t = trial_t

        a_new = (1 - best_t) * a + best_t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_norm_fn(a_new)

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            step = min(0.9, best_t * 1.1)
        else:
            step = max(0.02, best_t * 0.7)

        a = a_new
        if it % 10 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, step={step:.3f}, t*={t_val if t_val else 'N/A'}")

    return best_a, best_score


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, beta_list=None, n_restarts_noise=0.0):
    """L-BFGS with geometric beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005,
                 0.004, 0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0006,
                 0.0005, 0.0004, 0.0003, 0.0002, 0.00015, 0.0001,
                 0.00008, 0.00005, 0.00003, 0.00002, 0.00001]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    overall_rng = np.random.default_rng(12345)

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        time_per = max(time_per, 5.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 25))

        z_try = z.copy()
        if n_restarts_noise > 0 and idx > 0:
            z_try = z_try + n_restarts_noise * overall_rng.normal(size=n)
            z_try = z_try - z_try.mean()

        try:
            result = minimize(
                _lse_obj, z_try, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': maxiter * 4}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adaptive_gd(a_init, deadline, rng, lr_init=0.15, max_iter=50000):
    """Adaptive gradient descent with AMSGrad-style scheduling."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    v_hat_max = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-8, 0.02 * cm)
        vals = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        v_hat_max = np.maximum(v_hat_max, v)
        mh = m / (1 - b1 ** t)
        vh = v_hat_max / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)
        if t % 20000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=40):
    """Coordinate descent in log-space with adaptive steps."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    delta_scale = 0.05

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        order = np.random.default_rng(rnd + 1000).permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_local_score = _score_norm_fn(_softmax_fn(z))
            best_delta = 0.0
            for mult in [-2, -1, -0.5, 0.5, 1, 2]:
                delta = mult * delta_scale
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                sc = _score_norm_fn(_softmax_fn(z))
                if sc < best_score:
                    best_score = sc
                    best_a = _softmax_fn(z).copy()
                    improved = True
        if not improved:
            delta_scale *= 0.7
            if delta_scale < 0.001:
                break

    return best_a, best_score


def _make_diverse_profiles(n, rng):
    """Generate diverse initial profiles including asymmetric ones."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    # Uniform
    profiles.append(np.ones(n) / n)

    # U-shaped (various powers)
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = np.abs(u) ** p + 0.01
        profiles.append(a / a.sum())

    # Inverted U (peaked at center)
    for p in [1.0, 2.0, 4.0]:
        a = 1.0 - 0.8 * u ** 2
        a = np.maximum(a ** p, 1e-12)
        profiles.append(a / a.sum())

    # Exponential shapes (all four corners)
    for rate in [1, 2, 3, -1, -2, -3]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
        a = np.exp(rate * u)
        profiles.append(a / a.sum())

    # Bimodal
    for width in [0.1, 0.2, 0.3]:
        a = np.exp(-((t - 0.2) / width) ** 2) + np.exp(-((t - 0.8) / width) ** 2)
        profiles.append(a / a.sum())

    # Oscillatory (low frequency)
    for freq in [2, 3, 4, 5]:
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq * t)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Power law
    for alpha in [0.5, 1.0, 2.0]:
        a = (t + 0.01) ** alpha + ((1 - t) + 0.01) ** alpha
        profiles.append(a / a.sum())

    # Asymmetric linear
    for slope in [0.5, 1.0, 2.0, -0.5, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    # Product of shapes
    a = (np.abs(u) + 0.1) * (1.0 + 0.5 * u)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())

    # Random smooth profiles
    for _ in range(5):
        freqs = rng.integers(1, 8, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = np.ones(n)
        for f_, amp, ph in zip(freqs, amps, phases):
            a += amp * np.cos(2 * np.pi * f_ * t + ph)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _perturb_and_optimize(a_base, deadline, rng, n_perturb=5):
    """Perturb a base solution and re-optimize to escape local minima."""
    best_a = a_base.copy()
    best_score = _score_norm_fn(a_base)
    n = len(a_base)

    for i in range(n_perturb):
        if time.monotonic() > deadline:
            break
        noise_scale = [0.02, 0.05, 0.1, 0.15, 0.2][min(i, 4)]
        z = np.log(np.maximum(a_base, 1e-15))
        z = z - z.mean()
        z = z + noise_scale * rng.normal(size=n)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_try = _softmax_fn(z)

        # Quick L-BFGS with moderate beta
        c_try = _full_convolve(a_try, a_try)
        beta = 0.03 * float(c_try.max())
        time_per = min(60.0, (deadline - time.monotonic()) / max(1, n_perturb - i))
        if time_per < 10:
            break
        dl = time.monotonic() + time_per
        p_best, sc_best = _run_lbfgs_continuation(a_try, dl, beta_list=[beta, 0.5 * beta, 0.25 * beta, 0.1 * beta])
        if sc_best < best_score:
            best_score = sc_best
            best_a = p_best.copy()
            print(f"    Perturb {i}: improved to {sc_best:.10f} (noise={noise_scale})")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        profiles = _make_diverse_profiles(n_init, rng)
        best_p = profiles[0]
        best_sc = _score_norm_fn(profiles[0])
        for p in profiles:
            sc = _score_norm_fn(p)
            if sc < best_sc:
                best_sc = sc
                best_p = p
        update_best(best_p, best_sc, "profile_search")

    n_parent = len(best_seq)
    print(f"Starting with n={n_parent}, score={best_score:.10f}")

    # ===== Phase 1: LP iterations (500s) =====
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: LP iterations (n up to 4096) ===")

    # Try LP at the parent size (or 2048 if parent is larger)
    n_lp = min(n_parent, 2048)
    if n_parent > n_lp:
        a_lp = _interp_profile(best_seq, n_lp)
    else:
        a_lp = best_seq.copy()
    update_best(a_lp, _score_norm_fn(a_lp), f"lp_start_n{n_lp}")

    # Run LP with multiple rounds of decreasing step size
    a_cur = a_lp.copy()
    for lp_round in range(8):
        if time.monotonic() > phase1_end - 20:
            break
        step_size = 0.5 / (lp_round + 1)
        dl = min(phase1_end - 10, time.monotonic() + (phase1_end - time.monotonic()) / (8 - lp_round))
        a_cur, sc_cur = _lp_iterate_adaptive(a_cur, n_iters=50, deadline=dl, initial_step=step_size)
        update_best(a_cur, sc_cur, f"lp_r{lp_round}_n{n_lp}")

    # Interpolate back to parent size
    if n_parent > n_lp:
        a_rec = _interp_profile(a_cur, n_parent)
        update_best(a_rec, _score_norm_fn(a_rec), f"lp_interp_n{n_parent}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # ===== Phase 2: L-BFGS continuation (500s) =====
    phase2_end = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 2: L-BFGS continuation ===")

    n_curr = len(best_seq)
    n_candidates = sorted(set([n_curr, min(n_curr, 2048), 4096]))
    n_candidates = [x for x in n_candidates if x <= 4096]

    # Try L-BFGS at each candidate n
    for n_try in n_candidates:
        if time.monotonic() > phase2_end - 30:
            break
        if n_try > len(best_seq):
            a_start = _interp_profile(best_seq, n_try)
        elif n_try < len(best_seq):
            a_start = _interp_profile(best_seq, n_try)
        else:
            a_start = best_seq.copy()

        time_per = (phase2_end - time.monotonic()) / max(1, n_candidates.index(n_try) + 1)
        time_per = min(time_per, 150.0)
        time_per = max(time_per, 30.0)
        dl = time.monotonic() + time_per

        print(f"  L-BFGS at n={n_try} (time budget {time_per:.0f}s)")
        p_best, sc_best = _run_lbfgs_continuation(a_start, dl)
        update_best(p_best, sc_best, f"lbfgs_n{n_try}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # ===== Phase 3: Perturbation and re-optimization (300s) =====
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 3: Perturbation & re-optimization ===")

    if time.monotonic() < phase3_end and best_seq is not None:
        # Try perturbing the best solution
        dl = min(phase3_end - 20, time.monotonic() + 200.0)
        p_best, sc_best = _perturb_and_optimize(best_seq, dl, rng, n_perturb=6)
        update_best(p_best, sc_best, "perturbed")

        # Try a U-shaped initial at the best n
        n_best = len(best_seq)
        profiles = _make_diverse_profiles(n_best, rng)
        # Select U-shaped profiles and optimize the best one
        best_sub_sc = np.inf
        best_sub_p = None
        for p in profiles:
            sc = _score_norm_fn(p)
            if sc < best_sub_sc:
                best_sub_sc = sc
                best_sub_p = p
        if best_sub_p is not None and best_sub_sc < best_score:
            dl2 = min(phase3_end - 10, time.monotonic() + 60.0)
            p_opt, sc_opt = _run_lbfgs_continuation(best_sub_p, dl2)
            update_best(p_opt, sc_opt, "u_shaped_profile")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # ===== Phase 4: Multi-n exploration (200s) =====
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 4: Multi-n exploration ===")

    # Try larger n
    for n_large in [4096, 8192]:
        if time.monotonic() > phase4_end - 30:
            break
        if n_large <= len(best_seq):
            continue
        p_interp = _interp_profile(best_seq, n_large)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_large}")

        # Quick L-BFGS at larger n (fewer beta levels)
        if time.monotonic() < phase4_end - 30:
            c_tmp = _full_convolve(p_interp, p_interp)
            beta0 = 0.03 * float(c_tmp.max())
            beta_list = [b * beta0 for b in [1.0, 0.5, 0.3, 0.15, 0.08, 0.04]]
            time_per = min(80.0, phase4_end - time.monotonic() - 20)
            if time_per > 20:
                dl = time.monotonic() + time_per
                p_opt, sc_opt = _run_lbfgs_continuation(p_interp, dl, beta_list=beta_list)
                update_best(p_opt, sc_opt, f"lbfgs_quick_n{n_large}")

    # Try smaller n (might be better due to the 2n factor)
    for n_small in [1024, 2048]:
        if time.monotonic() > phase4_end - 15:
            break
        if n_small >= len(best_seq):
            continue
        p_ds = _interp_profile(best_seq, n_small)
        sc_ds = _score_norm_fn(p_ds)
        update_best(p_ds, sc_ds, f"downsample_n{n_small}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # ===== Phase 5: Fine polish (remaining) =====
    phase5_end = deadline
    print(f"\n=== Phase 5: Fine polish ===")

    if time.monotonic() < phase5_end and best_seq is not None:
        # Adaptive GD
        remaining = phase5_end - time.monotonic()
        if remaining > 30:
            gd_time = min(100.0, remaining * 0.4)
            dl_gd = time.monotonic() + gd_time
            p_best, sc_best = _adaptive_gd(best_seq, dl_gd, rng, lr_init=0.08)
            update_best(p_best, sc_best, "adaptive_gd")

        # Very fine L-BFGS
        if time.monotonic() < phase5_end - 20:
            c_final = _full_convolve(best_seq, best_seq)
            max_c = float(c_final.max())
            beta_tiny = 1e-5 * max_c
            beta_list_tiny = [b * max_c for b in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]]
            time_per = min(60.0, phase5_end - time.monotonic() - 15)
            if time_per > 15:
                dl = time.monotonic() + time_per
                p_best, sc_best = _run_lbfgs_continuation(best_seq, dl, beta_list=beta_list_tiny)
                update_best(p_best, sc_best, "ultra_fine_lbfgs")

        # Coordinate descent
        if time.monotonic() < phase5_end - 10:
            coord_dl = phase5_end - 10
            p_best, sc_best = _coord_descent(best_seq, coord_dl, max_rounds=30)
            update_best(p_best, sc_best, "coord_descent")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
