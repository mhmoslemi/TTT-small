def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _run_active_lbfgs(a_init, deadline, rng, max_top_k=200, n_continuation=25):

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    # Progressive top_k: start small, grow to max
    top_k_schedule = np.unique(np.concatenate([
        np.arange(3, 10, 2),       # 3, 5, 7, 9
        np.arange(10, 31, 5),      # 10, 15, 20, 25, 30
        np.arange(40, 101, 20),    # 40, 60, 80, 100
        np.arange(120, max_top_k + 1, 40)  # 120, 160, 200
    ]))
    top_k_schedule = top_k_schedule[top_k_schedule <= min(max_top_k, 2 * n - 1)]
    top_k_schedule = list(top_k_schedule)
    
    n_steps = len(top_k_schedule)
    
    for si, top_k in enumerate(top_k_schedule):
        if time.monotonic() > deadline:
            break
        
        remaining = n_steps - si
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget

        elif top_k <= 30:

        elif top_k <= 100:
            beta = 0.01 * max_c0
        else:
            beta = 0.003 * max_c0

            maxiter = 300
        elif top_k <= 30:

        else:
            maxiter = 2000
        maxfun = maxiter * 4

        except Exception as ex:

def _subgradient_minmax(a_init, deadline, rng, max_iter=50000):
    """Direct subgradient descent on max(conv(a,a))."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    lr = 0.01
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        
        # Subgradient of max: uniform weight on argmax (or top few)
        cm = c.max()
        # Use top-5 for smoother subgradient
        top_idx = np.argsort(c)[::-1][:5]
        weights = np.zeros(len(c))
        weights[top_idx] = 1.0 / len(top_idx)
        
        W = weights
        ar = a[::-1]
        Wa = _full_convolve(W, ar)
        h = 2.0 * Wa[n - 1:2 * n - 1]
        
        offset = float(np.dot(a, h))
        grad_z = a * (h - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        sc = _score_norm_fn(_softmax_fn(z))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _softmax_fn(z).copy()
        
        # Decay learning rate
        if t % 1000 == 0:
            lr = max(0.001, lr * 0.95)
        
        # Periodic restart
        if t % 5000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score


def _coordinate_descent(a_init, deadline, rng, max_rounds=2000):
    """Local search: try small perturbations to individual elements."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = c.max()
        
        # Find top peaks
        top_peaks = np.argsort(c)[::-1][:10]
        
        improved = False
        for pk in top_peaks[:5]:
            # Find top contributors to this peak
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:6]
            
            for pos in order[:3]:
                idx = valid_i[pos]
                # Try reducing
                for delta in [0.01, 0.03, 0.05, 0.1]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try increasing small values (to spread energy)
            small_idx = np.argsort(a_cur)[:5]
            for idx in small_idx:
                for delta in [0.01, 0.03, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and rnd > 100:
            # Try random perturbations
            for _ in range(10):
                idx = rng.integers(0, n)
                delta = rng.uniform(-0.05, 0.05)
                z_trial = z.copy()
                z_trial[idx] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if not improved and rnd > 300:
                break
    
    return best_a, best_score


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # S-curve
    a = 1.0 / (1.0 + np.exp(-5 * (t - 0.5)))
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        best_seq = parent.copy()
        best_score = sc_p
    else:
        # Generate from scratch
        print("No parent, generating from scratch...")
        a0 = _make_ramp(128, 1.5)
        best_seq = a0.copy()
        best_score = _score_norm_fn(a0)
    
    # --- Phase 1: Quick active-set L-BFGS at current scale (20% time) ---
    phase1_end = t_start + (budget_s - 30.0) * 0.20
    print(f"\nPhase 1: Active-set L-BFGS at n={len(best_seq)} (deadline +{phase1_end - t_start:.0f}s)")
    
    if time.monotonic() < phase1_end - 5:
        a_p1, sc_p1 = _run_active_lbfgs(best_seq, min(phase1_end - 2, time.monotonic() + (phase1_end - t_start)), rng, max_top_k=100, n_continuation=15)
        update_best(a_p1, sc_p1, "phase1_active")
    
    # --- Phase 2: Upsample to larger n and optimize (40% time) ---
    phase2_end = t_start + (budget_s - 30.0) * 0.60
    print(f"\nPhase 2: Multi-scale to n=8192 (deadline +{phase2_end - t_start:.0f}s)")
    
    n_current = len(best_seq)
    target_scales = [s for s in [2048, 4096, 8192, 16384] if s > n_current]
    if not target_scales:
        # Already at max scale, just do more optimization
        target_scales = [n_current]
    
    a_cur = best_seq.copy()
    
    for target_n in target_scales:
        if time.monotonic() > phase2_end - 30:
            break
        
        # Upsample
        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm_fn(a_target)
        print(f"  Upsampled to n={target_n}: score_norm={sc_target:.10f}")
        
        # Active-set L-BFGS
        alloc = min(300.0, (phase2_end - 10) - (time.monotonic() - t_start))
        opt_deadline = time.monotonic() + alloc
        a_opt, sc_opt = _run_active_lbfgs(a_target, opt_deadline, rng, max_top_k=150, n_continuation=20)
        print(f"  After n={target_n} active L-BFGS: score_norm={sc_opt:.10f}")
        
        if sc_opt < _score_norm_fn(a_cur):
            a_cur = a_opt
        update_best(a_cur, _score_norm_fn(a_cur), f"phase2_n{target_n}")
    
    # --- Phase 3: Subgradient min-max refinement (25% time) ---
    phase3_end = t_start + (budget_s - 30.0) * 0.85
    print(f"\nPhase 3: Subgradient min-max (deadline +{phase3_end - t_start:.0f}s)")
    
    if time.monotonic() < phase3_end - 5 and best_seq is not None:
        a_p3, sc_p3 = _subgradient_minmax(best_seq, phase3_end - 2, rng, max_iter=100000)
        update_best(a_p3, sc_p3, "phase3_subgrad")
    
    # --- Phase 4: Coordinate descent / local search (10% time) ---
    phase4_end = t_start + (budget_s - 30.0) * 0.95
    print(f"\nPhase 4: Coordinate descent (deadline +{phase4_end - t_start:.0f}s)")
    
    if time.monotonic() < phase4_end - 5 and best_seq is not None:
        a_p4, sc_p4 = _coordinate_descent(best_seq, phase4_end - 2, rng, max_rounds=3000)

    # --- Phase 5: Final tiny-beta L-BFGS polish ---

                         'maxls': 40, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Final L-BFGS: {best_score:.10f}")

    # --- Final validation ---
    if best_seq is not None:

        s = float(final_seq.sum())
        if s > 0.01:

        else:
            final_seq = _make_ramp(256, 1.5)
        
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence([float(x) for x in final_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception:
            pass

    # Fallback
    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
