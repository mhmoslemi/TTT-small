def _adam_run(a_init, deadline, lr=0.02, max_iter=300000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.02 * cm0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 100
    for _ in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.02 * (0.96 ** (t // 100)))
        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        if t % check_interval == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0
    return best_a, best_score


def _lp_phase(a_init, deadline, K=40, max_rounds=10):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.92 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = sorted_idx[:300]
        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        b_vec = np.full(len(active), M * 1.001)
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break
        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-13:
                best_a = a_trial.copy()
                best_score = _score_norm(a_trial)
                current_score = sc_trial
                improved = True
                break
        if not improved:
            break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: spectral init, Nesterov GD, L-BFGS continuation,
    active-set refinement, LP peak balancing, Adam polish, and multi-n exploration.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    np.random.seed(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_parent = len(parent) if parent is not None else 8192
    n = n_parent

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = np.ones(n) / n
        update_best(parent, _score_norm(parent), "uniform_default")

    print(f"Starting: n={n}, initial best={best_score:.10f}")

    # === Phase 1: Diversified initialization (180s) ===
    print("Phase 1: Diversified initialization")
    p1_end = min(deadline, t_start + 180.0)
    candidates = []

    # Spectral initializations
    spectral_params = [
        (2.0, 3.0, 0.5, 15),
        (2.0, 5.0, 0.4, 20),
        (3.0, 4.0, 0.6, 20),
        (1.5, 6.0, 0.3, 25),
        (2.5, 8.0, 0.5, 15),
    ]
    for alpha, bw, blend, n_it in spectral_params:
        if time.monotonic() > p1_end - 5:
            break
        a_spec = _spectral_init(n, n_it, alpha, bw, blend)
        sc = _score_norm(a_spec)
        candidates.append((a_spec, sc, f"spectral_a{alpha}_b{bw}"))
        update_best(a_spec, sc, f"spec_{alpha}_{bw}")

    # Parametric windows
    windows = _parametric_windows(n)
    for name, a_win in windows.items():
        if time.monotonic() > p1_end - 3:
            break
        sc = _score_norm(a_win)
        candidates.append((a_win, sc, name))
        update_best(a_win, sc, f"win_{name}")

    # Random restarts (a few)
    for _ in range(3):
        if time.monotonic() > p1_end - 5:
            break
        a_rand = np.random.dirichlet(np.ones(n) * 0.5)
        sc = _score_norm(a_rand)
        candidates.append((a_rand, sc, "random"))

    # Pick top 3 distinct candidates for further optimization
    candidates.sort(key=lambda x: x[1])
    top_cands = candidates[:5]
    print(f"  Phase 1 done: best init score={top_cands[0][1]:.10f} ({top_cands[0][2]})")

    print(f"  Current best: {best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 2: Nesterov gradient descent (500s) ===
    print("Phase 2: Nesterov gradient descent")
    p2_end = min(deadline, t_start + 680.0)

    # Run Nesterov from best candidate
    if best_seq is not None and time.monotonic() < p2_end - 60:
        nesterov_deadline = min(p2_end - 10, time.monotonic() + 300)
        a_nest, sc_nest = _nesterov_run(best_seq, nesterov_deadline)
        update_best(a_nest, sc_nest, "nesterov_main")

    # Run Nesterov from spectral init if it was good
    if len(top_cands) > 0 and top_cands[0][1] < best_score * 1.01 and time.monotonic() < p2_end - 60:
        a_nest2, sc_nest2 = _nesterov_run(top_cands[0][0], min(p2_end - 10, time.monotonic() + 200))
        update_best(a_nest2, sc_nest2, "nesterov_spec")

    # Run Nesterov from parent
    if parent is not None and time.monotonic() < p2_end - 30:
        a_nest3, sc_nest3 = _nesterov_run(parent, min(p2_end - 5, time.monotonic() + 150))
        update_best(a_nest3, sc_nest3, "nesterov_parent")

    print(f"  Phase 2 done: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 3: L-BFGS continuation (400s) ===
    print("Phase 3: L-BFGS continuation")
    p3_end = min(deadline, t_start + 1080.0)

    if best_seq is not None and time.monotonic() < p3_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        beta_list = [m_b * b for b in [0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                                        0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002,
                                        0.0001, 0.00005, 0.00003, 0.00001]]
        a_lb, sc_lb = _lbfgs_run(best_seq, min(p3_end - 30, time.monotonic() + 350),
                                  beta_list, maxiter_per_beta=4000)
        update_best(a_lb, sc_lb, "lbfgs_main")

    print(f"  Phase 3 done: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 4: LP peak balancing (200s) ===
    print("Phase 4: LP peak balancing")
    p4_end = min(deadline, t_start + 1280.0)

    if best_seq is not None and time.monotonic() < p4_end:
        a_lp, sc_lp = _lp_phase(best_seq, min(p4_end - 10, time.monotonic() + 180), K=50, max_rounds=8)
        update_best(a_lp, sc_lp, "lp_phase")

        # Second LP with different K
        if time.monotonic() < p4_end - 30:
            a_lp2, sc_lp2 = _lp_phase(best_seq, min(p4_end - 5, time.monotonic() + 80), K=20, max_rounds=5)
            update_best(a_lp2, sc_lp2, "lp_phase2")

    print(f"  Phase 4 done: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 5: Active-set refinement (150s) ===
    print("Phase 5: Active-set L-BFGS")
    p5_end = min(deadline, t_start + 1430.0)

    if best_seq is not None and time.monotonic() < p5_end:
        for K in [3, 5, 7]:
            if time.monotonic() > p5_end - 20:
                break
            a_act, sc_act = _active_set_lbfgs(best_seq, min(p5_end - 10, time.monotonic() + 60), K=K, rounds=15)
            update_best(a_act, sc_act, f"active_K{K}")

    print(f"  Phase 5 done: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 6: Final L-BFGS with tiny beta (150s) ===
    print("Phase 6: Final L-BFGS tiny beta")
    p6_end = min(deadline, t_start + 1580.0)

    if best_seq is not None and time.monotonic() < p6_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.0001, 0.00005, 0.00002, 0.00001, 0.000005]]
        a_f, sc_f = _lbfgs_run(best_seq, min(p6_end - 5, time.monotonic() + 140),
                                tiny_betas, maxiter_per_beta=6000)
        update_best(a_f, sc_f, "final_tiny")

    print(f"  Phase 6 done: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 7: Adam polish (100s) ===
    print("Phase 7: Adam polish")
    p7_end = min(deadline, t_start + 1680.0)

    if best_seq is not None and time.monotonic() < p7_end:
        a_ad, sc_ad = _adam_run(best_seq, min(p7_end - 10, time.monotonic() + 90), lr=0.015)
        update_best(a_ad, sc_ad, "adam_polish")

    print(f"  Phase 7 done: best={best_score:.10f} (t={time.monotonic()-t_start:.0f}s)")

    # === Phase 8: Try larger n if time allows ===
    if time.monotonic() < deadline - 120 and best_seq is not None and len(best_seq) <= 16384:
        n_new = min(2 * len(best_seq), 32768)
        if n_new > len(best_seq):
            print(f"Phase 8: Trying n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            print(f"  Interpolated score: {sc_in:.10f}")
            # Quick Nesterov on the interpolated
            if time.monotonic() < deadline - 30:
                a_in_b, sc_in_b = _nesterov_run(a_interp, min(deadline - 10, time.monotonic() + 100))
                update_best(a_in_b, sc_in_b, f"nesterov_n{n_new}")
            else:
                update_best(a_interp, sc_in, f"interp_n{n_new}")

    # === Final answer: pick the best ===
    print(f"\n=== FINAL: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.ones(256) / 256

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator if possible
        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = np.ones(512) / 512
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = np.ones(256) / 256
    print(f"FALLBACK: n=256, score={_score(fb):.10f}")
    return [float(x) for x in fb]
