"""
Optimized step function height sequence search using:
1. FFT-based mirror descent with Nesterov momentum and logsumexp continuation
2. Efficient O(n log n) gradient computation via cross-correlation
3. Peak-targeted local search with contribution-aware block moves
4. Parametric profile initialization and multi-restart strategy
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_smoothed_grad(a, beta):
    """Compute gradient of 2n*beta*log(sum(exp(c_k/beta))) w.r.t. a."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / beta
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    S = exp_vals.sum()
    if S < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = exp_vals / S
    # g_i = sum_{m=0}^{n-1} a_m * w_{i+m}
    # = conv(a_rev, w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    N = _next_pow2(conv_len)
    fa = np.fft.rfft(a_rev, N)
    fw = np.fft.rfft(w, N)
    conv_result = np.fft.irfft(fa * fw, N)[:conv_len]
    g = conv_result[n - 1:2 * n - 1]
    return 4.0 * n * g, c, c_max


def _mirror_descent_nesterov(a_init, deadline, beta_levels, lr_base=0.5,
                              momentum=0.9, max_iters_per_level=50000):
    """Mirror descent in log-space with Nesterov momentum and logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z_prev = z.copy()
    m = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for level, beta in enumerate(beta_levels):
        if time.monotonic() > deadline - 5.0:
            break
        level_start = time.monotonic()
        level_deadline = min(deadline - 3.0, level_start + 120.0)
        
        c_init = _full_convolve(_softmax_fn(z), _softmax_fn(z))
        c_max_init = float(c_init.max())
        lr = lr_base * c_max_init
        lr = min(lr, 0.1)
        
        iters_done = 0
        for it in range(max_iters_per_level):
            if time.monotonic() > level_deadline:
                break
            iters_done += 1
            
            a = _softmax_fn(z)
            grad_a, c, c_max = _compute_smoothed_grad(a, beta)
            
            # Gradient in z-space: a * (grad_a - dot(grad_a, a))
            grad_z = a * (grad_a - float(np.dot(grad_a, a)))
            
            # Nesterov momentum
            m = momentum * m + grad_z
            z_new = z - lr * m
            z_new = np.clip(z_new, -60, 60)
            z_new = z_new - z_new.mean()
            
            # Evaluate
            a_new = _softmax_fn(z_new)
            sc = _score_norm(a_new)
            
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            z_prev = z.copy()
            z = z_new
            lr = max(lr * 0.9999, 1e-6)
        
        print(f"    MD level {level}: beta={beta:.6f}, iters={iters_done}, best_score={best_score:.10f}")
    
    return best_a, best_score


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_logsumexp(a_init, deadline, beta_schedule, per_level_time=30.0):
    """L-BFGS with logsumexp continuation using FFT-based gradients."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    c_max0 = float(c0.max())
    betas = [b * c_max0 for b in beta_schedule]
    
    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_dl = min(deadline - 2.0, time.monotonic() + per_level_time)
        
        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            cs -= cs.max()
            e_ = np.exp(cs)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            grad_a, c_, c_max_ = _compute_smoothed_grad(a_, beta)
            grad_z = a_ * (grad_a - float(np.dot(grad_a, a_)))
            return grad_z
        
        try:
            result = minimize(obj, z.copy(), jac=grad_fn, method='L-BFGS-B',
                              options={'maxiter': 3000, 'ftol': 1e-18,
                                       'gtol': 1e-12, 'maxls': 50, 'maxfun': 5000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _peak_local_search(a_init, deadline, n_trials=50000, rng=None):
    """Local search targeting top convolution peaks with contribution-aware moves."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        K = min(8, len(c))
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])
        
        if it % 4 == 0:
            # Strategy 1: Reduce mass at positions contributing most to top peak
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                idx = pairs[int(np.argmax(products))]
                
                a_trial = a.copy()
                frac = rng.uniform(0.1, 0.4)
                delta = a_trial[idx] * frac
                a_trial[idx] = max(0.0, a_trial[idx] - delta)
                
                # Find safe position (low contribution to all top peaks)
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo = max(0, kl - (n - 1))
                    hi = min(n - 1, kl)
                    for j in range(lo, hi + 1):
                        partner = kl - j
                        if 0 <= partner < n:
                            contrib[j] += a[partner]
                
                safe = np.argsort(contrib)[:max(1, n // 10)]
                add_idx = int(safe[rng.integers(len(safe))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
        
        elif it % 4 == 1:
            # Strategy 2: Block multiplicative perturbation
            block_size = int(rng.integers(3, min(32, n // 4)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.6, 1.4, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        elif it % 4 == 2:
            # Strategy 3: Pair swap with normalization
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.05, 0.3)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2)
            a_trial[i1] += amt2
            a_trial[i2] += amt1
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        else:
            # Strategy 4: Smoothed block (blur a block and redistribute)
            block_size = int(rng.integers(5, min(64, n // 2)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            block = a_trial[start:start + block_size].copy()
            # Apply Gaussian smoothing to the block
            kernel_size = min(block_size, 5)
            if kernel_size >= 3:
                kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2)
                kernel /= kernel.sum()
                block = np.convolve(block, kernel, mode='same')
            block = np.maximum(block, 1e-15)
            block /= block.sum() * max(1e-15, a[start:start + block_size].sum())
            a_trial[start:start + block_size] = block
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_edge(n, alpha):
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = t ** alpha * (1.0 - t) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, sep, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2.0
    c2 = 0.5 + sep / 2.0
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining FFT-based mirror descent, L-BFGS continuation,
    and peak-targeted local search to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
            return True
        return False
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        n_work = n_parent
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_work = 8192
        parent = _make_triangular(n_work)
        sc_parent = _score_norm(parent)
        print(f"No parent, triangular n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    print(f"Working with n={n_work}")
    
    # --- Phase 0: Profile parameter sweep (60s) ---
    phase0_end = min(deadline, t_start + 60.0)
    print("Phase 0: Profile sweep")
    
    profiles = []
    # Various power-edge profiles
    for alpha in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]:
        profiles.append(_make_power_edge(n_work, alpha))
    # Bimodal
    for sep in [0.3, 0.4, 0.5, 0.6]:
        for width in [0.1, 0.15, 0.2]:
            profiles.append(_make_bimodal(n_work, sep, width))
    # W-profile
    for freq in [2, 3, 4]:
        for depth in [0.2, 0.4, 0.6]:
            profiles.append(_make_w_profile(n_work, freq, depth))
    # Quadratic
    for a1 in [0.3, 0.5, 0.8]:
        for a2 in [-0.2, 0.0, 0.2]:
            profiles.append(_make_quadratic(n_work, a1, a2))
    # Raised cosine
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        profiles.append(_make_raised_cosine(n_work, alpha))
    # Triangular
    profiles.append(_make_triangular(n_work))
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # Choose best starting point
    if candidates:
        best_start = candidates[0][1].copy()
        best_start_score = candidates[0][0]
    else:
        best_start = parent.copy()
        best_start_score = sc_parent
    
    if best_start_score > best_score:
        best_start = best_seq.copy()
        best_start_score = best_score
    
    # --- Phase 1: Mirror descent with Nesterov (500s) ---
    phase1_end = min(deadline, t_start + 560.0)
    print(f"Phase 1: Mirror descent, starting score={best_start_score:.10f}")
    
    c_init = _full_convolve(best_start, best_start)
    c_max_init = float(c_init.max())
    
    beta_schedule_md = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625, 0.0078]
    beta_levels_md = [b * c_max_init for b in beta_schedule_md]
    
    a_md, sc_md = _mirror_descent_nesterov(best_start, phase1_end, beta_levels_md,
                                            lr_base=0.3, momentum=0.85)
    update_best(a_md, sc_md, "mirror_descent")
    
    # --- Phase 2: L-BFGS continuation (400s) ---
    phase2_end = min(deadline, t_start + 960.0)
    if time.monotonic() < phase2_end - 30:
        print(f"Phase 2: L-BFGS continuation, current best={best_score:.10f}")
        start_a = best_seq.copy() if best_seq is not None else best_start
        beta_schedule_lbfgs = [0.5, 0.25, 0.125, 0.0625, 0.03125, 0.0156, 0.0078, 0.004]
        a_lbfgs, sc_lbfgs = _lbfgs_logsumexp(start_a, phase2_end, beta_schedule_lbfgs,
                                              per_level_time=40.0)
        update_best(a_lbfgs, sc_lbfgs, "lbfgs_main")
    
    # --- Phase 3: Peak-targeted local search (300s) ---
    phase3_end = min(deadline, t_start + 1260.0)
    if time.monotonic() < phase3_end - 10:
        print(f"Phase 3: Peak local search, current best={best_score:.10f}")
        a_ls, sc_ls = _peak_local_search(best_seq.copy(), phase3_end, n_trials=50000, rng=rng)
        update_best(a_ls, sc_ls, "local_search")
    
    # --- Phase 4: Second round of mirror descent from improved point (200s) ---
    phase4_end = min(deadline, t_start + 1460.0)
    if time.monotonic() < phase4_end - 20:
        print(f"Phase 4: Second MD round, current best={best_score:.10f}")
        c_now = _full_convolve(best_seq, best_seq)
        c_max_now = float(c_now.max())
        beta_levels_2 = [0.1, 0.05, 0.025, 0.0125, 0.00625, 0.003]
        beta_levels_2 = [b * c_max_now for b in beta_levels_2]
        a_md2, sc_md2 = _mirror_descent_nesterov(best_seq.copy(), phase4_end, beta_levels_2,
                                                  lr_base=0.2, momentum=0.9)
        update_best(a_md2, sc_md2, "md_round2")
    
    # --- Phase 5: More local search + L-BFGS fine-tuning (200s) ---
    phase5_end = min(deadline, t_start + 1660.0)
    if time.monotonic() < phase5_end - 10:
        print(f"Phase 5: Final refinement, current best={best_score:.10f}")
        
        # Fine L-BFGS
        if time.monotonic() < phase5_end - 60:
            beta_fine = [0.02, 0.01, 0.005, 0.002]
            a_fine, sc_fine = _lbfgs_logsumexp(best_seq.copy(), min(phase5_end - 60, time.monotonic() + 80),
                                                beta_fine, per_level_time=20.0)
            update_best(a_fine, sc_fine, "lbfgs_fine")
        
        # More local search
        if time.monotonic() < phase5_end - 10:
            a_ls2, sc_ls2 = _peak_local_search(best_seq.copy(), phase5_end, n_trials=30000, rng=rng)
            update_best(a_ls2, sc_ls2, "local_search2")
    
    # --- Phase 6: Multi-restart (remaining time) ---
    remaining = deadline - time.monotonic() - 30
    if remaining > 30:
        print(f"Phase 6: Multi-restart, remaining={remaining:.0f}s, best={best_score:.10f}")
        n_restarts = max(2, int(remaining / 60))
        per_restart = remaining / n_restarts
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 40:
                break
            noise = 0.01 + 0.005 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(deadline - 35, time.monotonic() + per_restart - 5)
            c_r = _full_convolve(p_start, p_start)
            c_max_r = float(c_r.max())
            beta_r = [0.2, 0.1, 0.05, 0.025]
            beta_r = [b * c_max_r for b in beta_r]
            a_r, sc_r = _mirror_descent_nesterov(p_start, dl, beta_r, lr_base=0.2, momentum=0.88)
            update_best(a_r, sc_r, f"restart{i}")
            
            # Quick local search after restart
            if time.monotonic() < deadline - 35:
                a_rl, sc_rl = _peak_local_search(best_seq.copy(), min(deadline - 35, time.monotonic() + 15),
                                                  n_trials=5000, rng=rng)
                update_best(a_rl, sc_rl, f"restart_ls{i}")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    fallback = _make_triangular(8192)
    print(f"FALLBACK: n=8192, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
