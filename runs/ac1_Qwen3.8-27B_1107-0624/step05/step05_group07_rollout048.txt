def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
            return g_val
    except Exception:
        pass
    return None

    return 2.0 * n * lse

                             'maxls': 100, 'maxfun': 30000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_seq = a_new.copy()
                        z = result.x.copy()
                        print(f"    Tiny beta {beta:.2e}: score={sc_new:.10f}")
            except Exception:
                pass

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Coordinate descent polish (remaining time) ===
    phase5_end = deadline
    print(f"\nPhase 5: Coordinate descent polish")

    if time.monotonic() < phase5_end and best_seq is not None:
        # Only do CD on smaller sequences for speed
        if len(best_seq) <= 4096:
            per_dl = min(phase5_end - 20, time.monotonic() + (phase5_end - time.monotonic()) * 0.6)
            p_best, sc_best = _coordinate_descent(best_seq, per_dl, max_rounds=15)
            _update_best(p_best, sc_best, "coord_descent")
        else:
            # Do subsampled CD
            n_sub = 2048
            if len(best_seq) > n_sub:
                p_sub = _interp_profile(best_seq, n_sub)
                per_dl = min(phase5_end - 20, time.monotonic() + (phase5_end - time.monotonic()) * 0.5)
                p_best_sub, sc_best_sub = _coordinate_descent(p_sub, per_dl, max_rounds=15)
                if sc_best_sub < _score_norm_fn(p_sub):
                    # Interpolate back
                    p_back = _interp_profile(p_best_sub, len(best_seq))
                    sc_back = _score_norm_fn(p_back)
                    _update_best(p_back, sc_back, "cd_interp_back")
                    _update_best(p_best_sub, sc_best_sub, "cd_sub")

        # Final tiny-beta L-BFGS
        if time.monotonic() < phase5_end - 15:
            n_final = len(best_seq)
            c_final = _full_convolve(best_seq, best_seq)
            max_c_final = float(c_final.max())
            beta_ultra = 3e-7 * max_c_final
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            per_dl2 = min(phase5_end - 5, time.monotonic() + 20.0)
            try:
                result = minimize(
                    _lse_obj, z, args=(beta_ultra, n_final),
                    method='L-BFGS-B', jac=_lse_grad,
                    options={'maxiter': 2000, 'ftol': 1e-25, 'gtol': 1e-16}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    sc_new = _score_norm_fn(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_seq = a_new.copy()
                        print(f"    Ultra final: score={sc_new:.10f}")
            except Exception:
                pass

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    profiles = _make_profiles(n_fb)
    fallback = profiles[8]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
