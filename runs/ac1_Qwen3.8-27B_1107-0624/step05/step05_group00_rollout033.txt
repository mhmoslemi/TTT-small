import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max() / (s * s)

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * c.max()

def _compute_grad_from_w(w, a):
    """
    Compute gradient of sum_k w[k] * c[k] where c = conv(a,a)
    grad[j] = 2 * sum_m w[j+m] * a[m] for m = 0..n-1
    This equals 2 * (w_rev * a)[2n-2-j] where w_rev is reversal of w
    """
    n = len(a)
    # w has length 2n-1
    w_rev = w[::-1]
    # Full convolution of w_rev and a has length (2n-1) + n - 1 = 3n-2
    wc = _fft_convolve(w_rev, a)
    # grad[j] = 2 * wc[2n-2-j] for j = 0..n-1
    # So grad = 2 * reverse(wc[n-1:2n-1])
    grad = 2.0 * wc[n-1:2*n-1][::-1]
    return grad

                            momentum=0.9, max_iter=200000):
    """
    Adam + momentum optimizer in log-space.
    a_init: normalized sequence (sum to 1)
    beta_schedule: list of beta values to use (annealing)
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z_centered = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init
    
    vel = np.zeros(n)  # momentum velocity
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        beta_iters = max(100, max_iter // len(beta_schedule))
        
        for it in range(beta_iters):
            if time.monotonic() > deadline:
                break
            
            t += 1
            a = _softmax(z_centered)
            
            c = _fft_convolve(a, a)
            cm = c.max()
            
            # Compute w
            if beta < 1e-10:
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
            else:
                c_shifted = (c - cm) / beta
                e = np.exp(c_shifted)
                s = e.sum()
                if s < 1e-300 or not np.isfinite(s):
                    w = np.zeros_like(c)
                    w[np.argmax(c)] = 1.0
                else:
                    w = e / s
            
            grad = _compute_grad_from_w(w, a)
            
            if not np.all(np.isfinite(grad)):
                break
            
            # Center the gradient (for scale invariance)
            grad_centered = grad - grad.mean()
            
            # Adam update
            m_buf = b1 * m_buf + (1 - b1) * grad_centered
            v_buf = b2 * v_buf + (1 - b2) * (grad_centered ** 2)
            mh = m_buf / (1 - b1**t)
            vh = v_buf / (1 - b2**t)
            
            update = current_lr * mh / (np.sqrt(vh) + eps)
            
            # Clip update for stability
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)
            
            # Update with momentum
            vel = momentum * vel - update
            z_centered = z_centered + vel
            z_centered = np.clip(z_centered, -30, 30)
            z_centered = z_centered - z_centered.mean()
            
            # Check progress periodically
            if t % 20 == 0:
                a_chk = _softmax(z_centered)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()
            
            # Decay learning rate
            if it % 2000 == 0 and it > 0:
                current_lr *= 0.95
    
    # Final check
    a_final = _softmax(z_centered)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

"""
Hybrid subgradient + Adam optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based exact subgradient computation in log-space with heavy ball momentum,
adaptive beta annealing, multi-phase structure, and local coordinate refinement.
Key innovations over previous approach:
- Direct subgradient (not FISTA) with momentum for sharper convergence
- Fourier-domain structured perturbations for escape from local minima
- Coordinate-wise refinement exploiting sparsity of active subgradient
- Adaptive two-timescale primal-dual updates
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_logits(a):
    """Gradient of smoothed max in log-space (same formula as previous code)."""
    n = len(a)
    c = _fft_convolve(a, a)
    a_rev = a[::-1]
    wc = _fft_convolve(w for w in [None])  # placeholder, won't be used
    return c


def _grad_logits(a, c, w, n):
    """
    Compute gradient in log-space of 2n * soft_max_beta(c).
    Returns grad_z[j] = 4n * a[j] * (h[j] - W)
    where h[j] = sum_m w[j+m]*a[m], W = w.c
    """
    a_rev = a[::-1]
    wc = _fft_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _compute_w(c, beta):
    """Compute softmax weights over convolution coefficients."""
    cm = float(c.max())
    if beta < 1e-12:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, cm
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w, cm
    return e / s, cm


def _momentum_logspace(a_init, deadline, beta_schedule, lr_init=0.008,
                       momentum=0.85, max_iter=300000, verbose=True):
    """
    Heavy-ball momentum + Adam hybrid in log-space.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    vel = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() > deadline:
            break
        iters_in_phase = max(200, max_iter // len(beta_schedule))

        for it in range(iters_in_phase):
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax(z)
            c = _fft_convolve(a, a)
            w, cm = _compute_w(c, beta)

            grad = _grad_logits(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            # Adam
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t)
            vh = v_buf / (1 - b2 ** t)
            adam_update = current_lr * mh / (np.sqrt(vh) + eps)

            # Clip
            un = np.linalg.norm(adam_update)
            if un > 0.8:
                adam_update = adam_update * (0.8 / un)

            # Momentum
            vel = momentum * vel - adam_update
            z = z + vel
            z = np.clip(z, -35.0, 35.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it % 5000 == 0 and it > 0:
                current_lr = max(0.001, current_lr * 0.97)

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _fista_simplex(a_init, beta, deadline, max_iter):
    """FISTA on simplex with softmax-smoothed max (from previous code, improved)."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-30:
            y = y / sy
        else:
            y = np.ones(n) / n

        c_y = _fft_convolve(y, y)
        c_max = float(c_y.max())
        w, _ = _compute_w(c_y, beta)

        y_rev = y[::-1]
        wc = _fft_convolve(w, y_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c_y))
        grad = 2.0 * (h - W)

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _fourier_perturb(a, rng, num_freqs=5):
    """Add low-frequency Fourier perturbations."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(num_freqs):
        freq = rng.integers(1, max(2, n // 100))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        t = np.arange(n)
        z = z + amp * np.cos(2 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _spike_perturb(a, rng, num_spikes=20):
    """Add small Gaussian spikes at random positions."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    for _ in range(num_spikes):
        pos = rng.integers(0, n)
        width = rng.integers(2, max(3, n // 200))
        amp = rng.uniform(-0.05, 0.05)
        t = np.arange(n)
        # Circular distance
        d = np.minimum(np.abs(t - pos), n - np.abs(t - pos))
        z = z + amp * np.exp(-d ** 2 / (2 * width ** 2))
    z = z - z.mean()
    return _softmax(z)


def _coordinate_descent(a_init, deadline, max_iter=50000):
    """
    Local coordinate descent: update small groups of coordinates.
    Exploits the fact that at near-optimum, only a few coordinates matter.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    # Group coordinates into blocks
    block_size = max(4, n // 512)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        # Pick random start
        start = rng.integers(0, n - block_size)
        # Random block of coordinates to update
        idx = np.arange(start, min(start + block_size, n))

        # Try small perturbations in this block
        for trial in range(3):
            delta = rng.standard_normal(len(idx)) * 0.02
            a_trial = a.copy()
            z_block = np.log(np.maximum(a_trial[idx], 1e-15)) + delta
            a_trial[idx] = np.exp(z_block)
            # Renormalize
            s = a_trial.sum()
            if s < 1e-15:
                a_trial = a.copy()
                break
            a_trial = a_trial / s
            sc = _score_norm(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_score


def _subgradient_descent(a_init, deadline, beta_init=0.001, beta_final=1e-6,
                          lr_init=0.005, momentum=0.9, max_iter=200000):
    """
    Pure subgradient descent with momentum in log-space.
    Uses exact or near-exact subgradient (very small beta).
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    vel = np.zeros(n)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    lr = lr_init

    total_iters = max_iter
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break

        # Anneal beta
        progress = it / total_iters
        beta = beta_init * (beta_final / beta_init) ** progress

        a = _softmax(z)
        c = _fft_convolve(a, a)
        w, cm = _compute_w(c, beta)

        grad = _grad_logits(a, c, w, n)
        if not np.all(np.isfinite(grad)):
            break

        # Step size decay
        lr_t = lr_init / (1.0 + 0.01 * it)

        vel = momentum * vel - lr_t * grad
        z = z + vel
        z = np.clip(z, -35.0, 35.0)
        z = z - z.mean()

        if it % 100 == 0:
            a_chk = _softmax(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

            lr_init=0.002, momentum=0.93, max_iter=100000
        )
        sc_pol_full = _score(a_pol2)
        if sc_pol_full < best_score:
            best_score = sc_pol_full
            best_seq = a_pol2.copy()
            print(f"  Final polish: score={sc_pol_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # === Validation and return ===
    print(f"Best score found: {best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
