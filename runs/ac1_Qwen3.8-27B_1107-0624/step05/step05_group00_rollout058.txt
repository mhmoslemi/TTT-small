def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    """Assume sum(a)=1, return 2n*max(a*a)"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)

    p_new = np.maximum(p_new, 1e-12)

        w_top = w_top / w_top.sum()

    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Compute weights
        w = np.exp((c - cm) / beta)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws < 1e-300:

            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws

**What if I use a completely different approach: simulated annealing / stochastic gradient MCMC?**

The idea: treat the problem as finding the minimum of an energy function E(a) = 2n * max(a*a) (with sum(a)=1). Use a Markov chain that:
1. Proposes a random change to a (e.g., move mass from one position to another)

This is different from gradient-based methods because it can escape local minima.

**Algorithm:**
1. Start from parent
2. Repeat:
   a. Choose two random positions i, j
   b. Propose: a_i' = a_i + δ, a_j' = a_j - δ (for small random δ)
   c. Compute change in max(a*a) (approximately)
   d. Accept with probability exp(-ΔE/T)
3. Slowly decrease T

The challenge is computing the energy change efficiently. If we change a_i by δ, the convolution changes by:
Δc_k = δ * a_{k-i} + a_i * δ' * ... (where δ' is the change at j)

For a single position change (a_i += δ, others unchanged, then renormalize):
Δc_k = δ * a_{k-i} (approximately, before renormalization)

The new max is max(old_max, max_k (c_k + δ*a_{k-i}))... this is complex.

A simpler variant: use the "swap" move. Pick i, j, and swap a small amount of mass between them.

"""
Adam log-space optimizer with adaptive softmax smoothing, symmetrization,
and multi-length exploration for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Improves upon previous FISTA+Adam approach with better hyperparameter
schedules and n=8192 exploration.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    out_n = n_a + n_b - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

def _adam_phase(a_init, deadline, n_steps, beta_init, beta_final, lr_init, lr_final, symmetrize=True):
    """
    Adam log-space optimization with linearly interpolated beta schedule.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    if symmetrize:
        z = 0.5 * (z + z[::-1])
        z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        t += 1
        frac = step / max(1, n_steps - 1)
        beta = beta_init * (beta_final / beta_init) ** frac
        lr = lr_init + (lr_final - lr_init) * frac
        
        a = _softmax_compute(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        if cm < 1e-30:
            break
        
        # Softmax weights
        w = np.exp((c - cm) / beta)
        ws = float(w.sum())
        if not np.isfinite(ws) or ws < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / ws
        
        # Gradient in z-space: 2n * a * (h - W)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
