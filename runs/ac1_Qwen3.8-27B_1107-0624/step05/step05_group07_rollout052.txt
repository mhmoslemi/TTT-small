```python
2*n*max(conv(a,a))/sum(a)^2. Uses Fichtenholz-Hedge accelerated steps,
exact subgradient method, and structured initialization.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    """Returns 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    """Build matrix A such that (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    # (f*g)[k] = sum_i f[k-i] * g[i] for valid i
    # = sum_i A[k,i] * g[i]
    # A[k,i] = f[k-i] if 0 <= k-i < n, else 0
    # So A[k,i] = f[k-i] where 0 <= k-i <= n-1
    # i.e., k-(n-1) <= i <= k, and 0 <= i <= n-1
    row_idx = []
    col_idx = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            row_idx.append(k)
            col_idx.append(i)
            data.append(f[k-i])
    return csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))

def _lp_solve(f, deadline=None):
    """Solve: min t s.t. A@f@g <= t, sum(g)=1, g>=0. Returns g, t."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [A @ g <= t, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, max_time=30)
        if g.value is not None and t.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 0:
                g_val = g_val / s
            return g_val, float(t.value)
    except Exception as e:
        print(f"  LP failed: {e}")
    return None, None

def _lp_minimax_accelerated(f_init, deadline, n_iters=500):
    """
    Accelerated LP minimax using Hedge-style averaging.
    """
    n = len(f_init)
    f = f_init.copy()
    f_avg = f_init.copy()  # Exponentially weighted average
    best_a = f_init.copy()
    best_score = _score_norm_fn(f_init)
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Find best response to f_avg
        g, t_val = _lp_solve(f_avg)
        if g is None:
            break
        
        # Hedge update with step size
        t_step = 2.0 / (it + 2)  # O(1/k) step for accelerated
        f_new = (1 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        f_new = f_new / s
        
        # Update average (Nesterov-style)
        f_avg = (it / (it + 1.0)) * f_avg + (1.0 / (it + 1.0)) * f_new
        
        # Check score
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        
        # Also check the average
        sc_avg = _score_norm_fn(f_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = f_avg.copy()
        
        if it % 50 == 0:
            print(f"    LP-acc iter {it}: score={sc:.10f}, avg_score={sc_avg:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    
    return best_a, best_score

        if time.monotonic() > deadline:
            break
        t += 1

        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        
        c = _full_convolve(a, a)
        cm = c.max()
        
        # Find subgradient: positions where c is near max
        threshold = cm - 1e-10
        peak_mask = (c >= threshold)

        top_indices = np.argsort(c)[-top_k:]

        weights = weights / weights.sum()

        # d/d(a[i]) (a*a)[k] = a[k-i] (for valid k-i) + a[i+k-(n-1)] ...

        # Let me be more careful:
        # (a*a)[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j] * a[k-j]
        # d/d(a[i]) = sum over terms where i appears:
        #   - a[j]*a[k-j] where j=i: term is a[i]*a[k-i] (if 0<=k-i<n), derivative w.r.t. a[i] is a[k-i]
        #   - a[j]*a[k-j] where k-j=i, i.e., j=k-i: term is a[k-i]*a[i] (if 0<=k-i<n), derivative w.r.t. a[i] is a[k-i]

        # More precisely:
        # d/d(a[i]) (a*a)[k] = a[k-i] (from the j=i term, if valid) + a[k-i] (from the k-j=i term, if valid and k-j != i)
        # = a[k-i] if k-i is valid and k-i == i (i.e., k=2i)

        # But this is expensive. Let's use FFT:
        # The "un-weighted" gradient of (a*a)[k] w.r.t. a[i] is related to the cross-correlation
        # of a with the "peak indicator"
        
        # Simpler approach: for each peak position k, the gradient contribution to a[i] is:
        #   grad_k[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if k-i != i and 0<=k-i<n)
        #             = 2*a[k-i] (if 0<=k-i<n and k != 2*i)
        #             = a[k-i] (if k == 2*i)
        
        # Using FFT: if we define w[k] for the peaks, then
        # sum_k w[k] * a[k-i] (for valid k-i) = (w_reversed * a)[i] (approximately)
        # where w_reversed is w placed in a (2n-1)-long array
        
        # Let's just compute it directly for the top peaks
        grad = np.zeros(n)
        for k_idx in top_indices:
            k = int(k_idx)
            w = weights[top_indices == k_idx][0] if (top_indices == k_idx).any() else 0
            # Actually, let me just use the weights array properly
        # Let me redo this
        grad = np.zeros(n)
        for wi, ki in enumerate(top_indices):
            k = int(ki)
            w = weights[wi]
            # k ranges from 0 to 2n-2
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                # a[i] appears as a[j] with j=i: contributes a[k-i] if 0<=k-i<n
                # a[i] appears as a[k-j] with k-j=i, j=k-i: contributes a[k-i] if 0<=k-i<n and k-i != i
                ki_val = k - i
                if 0 <= ki_val < n:
                    if k == 2 * i:
                        grad[i] += w * a[ki_val]
                    else:
                        grad[i] += w * 2 * a[ki_val]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s
        
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Decay learning rate
        if t % 500 == 0:
            lr = max(0.001, lr * 0.95)
        
        if it % 2000 == 0:
            print(f"    SubGD iter {it}: score={sc:.10f}, best={best_score:.10f}, lr={lr:.6f}")
    
    return best_a, best_score
```

Actually, I just realized something important. The previous code's LP approach fixes f and solves for g, then does f <- (1-t)*f + t*g. But this is NOT the same as the "Hedge" algorithm. In the Hedge algorithm, you maintain a probability distribution over actions, and the "average" is what's used for the guarantee.

The key difference: in the LP minimax, the "best response" g is a probability distribution (sum=1, g>=0), and the update f <- (1-t)*f + t*g is a "fictitious play" update. The convergence of fictitious play is known to be O(1/sqrt(T)) for the "average" payoff, but the "current" payoff might oscillate.

To get the O(1/T) rate, we need to use the "Hedge" (multiplicative weights) update:
f_new[i] = f[i] * exp(-t * grad_i) / Z

where grad_i is the "loss" of action i.

In our context, the "loss" of a[i] is the contribution of a[i] to the peak. Specifically, if the peak is at position k*, then the loss of a[i] is proportional to a[k*-i] (the "partner" of a[i] in the peak convolution).

So the multiplicative weights update would be:
f_new[i] = f[i] * exp(-t * a[k*-i]) / Z (if we use a single peak)
or
f_new[i] = f[i] * exp(-t * sum_k w[k] * a[k-i]) / Z (if we use multiple peaks)

```python
def _project_simplex(x):
    """Project x onto {y: sum(y)=1, y>=0}."""
    n = len(x)
    # Bisect on lambda
    lo = x.min() - 1
    hi = x.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.maximum(x - mid, 0)
        s = y.sum()
        if s > 1:
            lo = mid
        else:
            hi = mid
    y = np.maximum(x - hi, 0)
    return y / y.sum()
```

```python
"""
Accelerated LP-minimax (Nesterov averaging) + subgradient + L-BFGS continuation
+ coordinate descent for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Exploits multiple step-size schedules, dual QP approach, and multi-n exploration.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        i_lo = max(0, k - (n - 1))
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[k - i])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, time_limit=20.0):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [A @ g <= t, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, max_time=time_limit)
        if g.value is not None and t.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-12:
                g_val = g_val / s
            else:
                g_val = np.ones(n) / n
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None

    f = f_init.copy()
    f_avg = f_init.copy()

    best_score = _score_norm_fn(f_init)
    step_mode = 0  # 0: 2/(k+2), 1: 0.3 constant, 2: 1/sqrt(k)
    step_schedules = [
        lambda k: 2.0 / (k + 2),
        lambda k: 0.3,
        lambda k: 1.0 / np.sqrt(k + 1),
    ]

    for it in range(5000):
        if time.monotonic() > deadline:
            break
        g, t_val = _lp_solve(f_avg, time_limit=time_per_lp)
        if g is None:
            break

        t_step = step_schedules[step_mode](it)
        t_step = min(t_step, 0.8)
        f_new = (1 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        f_avg = (it / float(it + 1)) * f_avg + (1.0 / float(it + 1)) * f_new

        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        sc_avg = _score_norm_fn(f_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = f_avg.copy()

        if it % 30 == 0:
            print(f"    LP-Nesterov[{step_mode}] iter {it}: sc={sc:.10f} avg={sc_avg:.10f} best={best_score:.10f} t*={t_val:.8f}")

    return best_a, best_score


def _subgradient_adam(a_init, deadline, top_k=30, lr_init=0.05):
    n = len(a_init)
    a = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t_adam += 1
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        c = _full_convolve(a, a)
        cm = c.max()

        top_idx = np.argsort(c)[-top_k:]
        vals = c[top_idx]
        vmin = vals.min()
        weights = np.maximum(vals - vmin, 1e-12)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for wi in range(len(top_idx)):
            k = int(top_idx[wi])
            w_val = weights[wi]
            i_lo = max(0, k - (n - 1))
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                ki = k - i
                if k == 2 * i:
                    grad[i] += w_val * a[ki]
                else:
                    grad[i] += w_val * 2.0 * a[ki]

        gn = np.linalg.norm(grad)
        if gn > 1e-15:
            grad = grad / gn

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t_adam)
        vh = v / (1 - b2 ** t_adam)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s

        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t_adam % 1000 == 0:
            lr = max(0.002, lr * 0.97)
        if t_adam % 5000 == 0:
            print(f"    SubGD iter {it}: sc={sc:.10f} best={best_score:.10f} lr={lr:.6f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_fine(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005,
                 0.004, 0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0006,
                 0.0005, 0.0004, 0.0003, 0.0002, 0.0001, 0.00005,
                 0.00002, 0.00001]
        beta_list = [b * max_c0 for b in fracs]

    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = len(beta_list) - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        maxiter = max(200, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 30, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=40):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    deltas = [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            base_score = _score_norm_fn(a)
            best_delta = 0.0
            best_local = base_score
            for d in deltas:
                z[j] += d
                a_trial = _softmax_fn(z)
                sc = _score_norm_fn(a_trial)
                z[j] -= d
                if sc < best_local:
                    best_local = sc
                    best_delta = d
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -60, 60)
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 4.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # U-shape profiles
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = np.abs(u) ** p + 0.3
        profiles.append(a / a.sum())
    for p in [0.3, 0.5, 1.0, 2.0]:
        a = (1 - u * u) ** p + 0.1
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _peak_shave(a, deadline, n_iters=2000):
    """Directly shave peaks by reducing a at positions that contribute to peaks."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a = a.copy()

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = c.max()
        peak_idx = int(np.argmax(c))

        # Find which a[i] contribute most to c[peak_idx]
        i_lo = max(0, peak_idx - (n - 1))
        i_hi = min(n - 1, peak_idx)
        contributions = a[i_lo:i_hi + 1] * a[peak_idx - i_lo:peak_idx - i_hi][::-1]

        # Reduce the largest contributors
        top_contrib = np.argsort(contributions)[-3:]
        reduction = 0.001 * (it % 50 + 1) / 50.0  # gradually increase
        for tc in top_contrib:
            a[i_lo + tc] *= (1.0 - reduction)

        a = np.maximum(a, 0)
        s = a.sum()
        a = a / s
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            a = best_a.copy()
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        profiles = _make_profiles(n_init)
        parent = profiles[0]
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, best_score={best_score:.10f}")

    # === Phase 1: LP minimax with Nesterov (multiple step schedules) ===
    print("\n=== Phase 1: LP Minimax ===")
    n_lp = min(n_parent, 4096)
    if n_parent > n_lp:
        a_lp = _interp_profile(parent, n_lp)
    else:
        a_lp = parent.copy()

    # Try step schedule 0: 2/(k+2)
    phase1a_end = min(deadline, t_start + 250.0)
    print(f"  Schedule 0 (2/(k+2)):")
    a_best, sc_best = _lp_minimax_nesterov(a_lp, phase1a_end, time_per_lp=10.0)
    update_best(a_best, sc_best, "lp_sched0")

    # Try step schedule 1: constant 0.3 (restart from best)
    if time.monotonic() < deadline - 300:
        phase1b_end = min(deadline, t_start + 450.0)
        print(f"  Schedule 1 (const 0.3):")
        # Modify the function by calling with different internal state
        f_init = best_seq.copy()
        if n_parent > n_lp:
            f_init = _interp_profile(f_init, n_lp)
        a_best2, sc_best2 = _lp_minimax_const(f_init, phase1b_end, time_per_lp=10.0)
        update_best(a_best2, sc_best2, "lp_sched1")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Subgradient with Adam ===
    print("\n=== Phase 2: Subgradient ===")
    phase2_end = min(deadline, t_start + 700.0)
    if time.monotonic() < phase2_end:
        a_sg = best_seq.copy()
        if n_parent > n_lp:
            a_sg = _interp_profile(a_sg, n_lp)
        p_best, sc_best = _subgradient_adam(a_sg, phase2_end, top_k=20, lr_init=0.03)
        update_best(p_best, sc_best, "subgrad")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: L-BFGS fine continuation ===
    print("\n=== Phase 3: L-BFGS ===")
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end:
        for restart in range(3):
            if time.monotonic() > phase3_end - 30:
                break
            if restart == 0:
                a_cur = best_seq.copy()
            else:
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                noise = 0.02 * (restart + 1)
                z = z + noise * rng.normal(size=len(z))
                z = np.clip(z, -50, 50)
                z = z - z.mean()
                a_cur = _softmax_fn(z)
            per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) / max(1, 3 - restart))
            p_best, sc_best = _lbfgs_fine(a_cur, per_dl)
            update_best(p_best, sc_best, f"lbfgs_r{restart}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Try larger n ===
    print("\n=== Phase 4: Larger n ===")
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        n_large = 8192
        p_interp = _interp_profile(best_seq, n_large)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_large}")

        if time.monotonic() < phase4_end - 30:
            # Quick LP at larger n
            phase4b_end = min(phase4_end, t_start + 1500.0)
            a_large, sc_large = _lp_minimax_nesterov(p_interp, phase4b_end, time_per_lp=20.0)
            update_best(a_large, sc_large, f"lp_n{n_large}")

        # Also try n=2048 (downsample)
        if time.monotonic() < phase4_end - 20:
            p_2048 = _interp_profile(best_seq, 2048)
            sc_2048 = _score_norm_fn(p_2048)
            update_best(p_2048, sc_2048, "downsample_n2048")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Peak shaving + coordinate descent ===
    print("\n=== Phase 5: Polish ===")
    phase5_end = deadline
    if time.monotonic() < phase5_end and best_seq is not None:
        # Peak shaving
        per_dl = min(phase5_end - 60, time.monotonic() + 120.0)
        p_ps, sc_ps = _peak_shave(best_seq, per_dl)
        update_best(p_ps, sc_ps, "peak_shave")

        # Coordinate descent
        if time.monotonic() < phase5_end - 30:
            per_dl2 = min(phase5_end - 15, time.monotonic() + (phase5_end - time.monotonic()) * 0.6)
            p_cd, sc_cd = _coord_descent(best_seq, per_dl2, max_rounds=20)
            update_best(p_cd, sc_cd, "coord_descent")

        # Final tiny L-BFGS
        if time.monotonic() < phase5_end - 10:
            n_final = len(best_seq)
            c_final = _full_convolve(best_seq, best_seq)
            max_c = float(c_final.max())
            beta_tiny = 1e-5 * max_c
            per_dl3 = min(phase5_end - 5, time.monotonic() + 30.0)
            p_tiny, sc_tiny = _lbfgs_fine(best_seq, per_dl3, beta_list=[beta_tiny, 5e-6 * max_c])
            update_best(p_tiny, sc_tiny, "tiny_lbfgs")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    return [float(x) for x in fallback]


def _lp_minimax_const(f_init, deadline, time_per_lp=10.0):
    """LP minimax with constant step size 0.3."""
    n = len(f_init)
    f = f_init.copy()
    f_avg = f_init.copy()
    best_a = f_init.copy()
    best_score = _score_norm_fn(f_init)

    for it in range(5000):
        if time.monotonic() > deadline:
            break
        g, t_val = _lp_solve(f_avg, time_limit=time_per_lp)
        if g is None:
            break

        t_step = 0.3
        f_new = (1 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        f_avg = (it / float(it + 1)) * f_avg + (1.0 / float(it + 1)) * f_new

        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        sc_avg = _score_norm_fn(f_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = f_avg.copy()

        if it % 30 == 0:
            print(f"    LP-const iter {it}: sc={sc:.10f} avg={sc_avg:.10f} best={best_score:.10f}")

    return best_a, best_score
```
