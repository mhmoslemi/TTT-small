            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}, ratio={max_c0 / (a0.sum()**2):.10f}")

    # === Phase 1: FISTA with top-K softmax ===
    print("Phase 1: FISTA top-K on n={}...".format(n_main))
    phase1_end = t_start + 200.0
    a_work = a0.copy()

    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    # Beta schedule: geometric from coarse to fine
    beta = 0.02 * max_cc
    K = 50
    while beta > 0.000005 * max_c0 and time.monotonic() < phase1_end - 5:
        phase_timeout = min(time.monotonic() + 40.0, phase1_end - 3)
        max_it = 50000
        a_work, sc_norm = _fista_topk(a_work, beta, K, phase_timeout, max_it)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={beta:.8f} K={K}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Gradually increase K and decrease beta
        beta *= 0.5
        if beta < 0.001 * max_c0:
            K = min(100, K * 2)

    print(f"End Phase 1: best_score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Adam log-space (long phase) ===
    print("Phase 2: Adam log-space...")
    phase2_end = t_start + 700.0
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    # Fine beta schedule
    betas_fine = []
    bf = 0.002 * max_cw
    while bf > 0.000002 * max_c0:
        betas_fine.append(bf)
        bf *= 0.6  # Slower decay for more time at each level
    if len(betas_fine) > 15:
        betas_fine = betas_fine[:15]

    per_time = phase2_end - time.monotonic() - 10
    a_opt, sc_opt = _adam_logspace_v2(
        a_work, time.monotonic() + max(10.0, per_time),
        betas_fine, lr_init=0.015, max_iter_total=800000, lr_schedule="cosine"
    )
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam v2: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = best_seq.copy() / best_seq.sum()
    print(f"End Phase 2: best_score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: n=8192 exploration ===
    print("Phase 3: n=8192 exploration...")
    phase3_end = t_start + 1100.0
    n_8192 = 8192

    if time.monotonic() < phase3_end - 30 and best_seq is not None:
        a_8192 = _interp_profile(best_seq, n_8192)
        sc_8192 = _score(a_8192)
        print(f"  Interpolated n=8192: score={sc_8192:.10f}")

        if sc_8192 < best_score:
            best_score = sc_8192
            best_seq = a_8192.copy()

        # Optimize the 8192 version with FISTA
        if time.monotonic() < phase3_end - 40:
            c_8192 = _full_convolve(a_8192, a_8192)
            mc_8192 = float(c_8192.max())
            beta_8192 = 0.01 * mc_8192
            t_left = phase3_end - 60 - time.monotonic()

            # Quick FISTA pass
            a_8192_f, sc_8192_f = _fista_topk(
                a_8192, beta_8192, 50,
                time.monotonic() + max(5.0, t_left * 0.4), 30000
            )
            a_8192_f = a_8192_f / a_8192_f.sum()
            sc_8192_f_full = _score(a_8192_f)
            if sc_8192_f_full < best_score:
                best_score = sc_8192_f_full
                best_seq = a_8192_f.copy()
                print(f"  FISTA n=8192: score={sc_8192_f_full:.10f}")

            # Adam on 8192
            if time.monotonic() < phase3_end - 20:
                c_8192f = _full_convolve(a_8192_f, a_8192_f)
                mc_8192f = float(c_8192f.max())
                betas_8192 = [0.001 * mc_8192f, 0.0005 * mc_8192f,
                              0.0002 * mc_8192f, 0.0001 * mc_8192f]
                t_left2 = phase3_end - 20 - time.monotonic()
                if t_left2 > 20:
                    a_8192_a, sc_8192_a = _adam_logspace_v2(
                        a_8192_f, time.monotonic() + t_left2,
                        betas_8192, lr_init=0.008, max_iter_total=200000,
                        lr_schedule="cosine"
                    )
                    sc_8192_a_full = _score(a_8192_a)
                    if sc_8192_a_full < best_score:
                        best_score = sc_8192_a_full
                        best_seq = a_8192_a.copy()
                        print(f"  Adam n=8192: score={sc_8192_a_full:.10f}")

    print(f"End Phase 3: best_score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse initializations ===
    print("Phase 4: Multi-restart...")
    phase4_end = t_start + 1550.0
    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        n_best = len(best_seq)
        n_restarts = 8
        per_restart = (phase4_end - time.monotonic() - 20) / n_restarts

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 10:
                break

            # Different restart strategies
            if i % 4 == 0:
                # Small perturbation
                p_start = _perturb(best_seq, 0.02 + 0.01 * i, rng)
            elif i % 4 == 1:
                # Frequency-domain perturbation
                p_start = best_seq.copy()
                F = np.fft.rfft(p_start)
                F = F * (1.0 + 0.05 * rng.standard_normal(len(F)))
                p_start = np.fft.irfft(F, len(p_start))
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()
            elif i % 4 == 2:
                # Blend with a different profile
                alt_profile = _make_profile(n_best, "raised_cosine")
                p_start = 0.7 * best_seq + 0.3 * alt_profile
                p_start = p_start / p_start.sum()
            else:
                # Larger perturbation
                p_start = _perturb(best_seq, 0.05 + 0.02 * i, rng)

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.001 * max_cs, 0.0005 * max_cs,
                       0.0002 * max_cs, 0.0001 * max_cs]

            a_r, sc_r = _adam_logspace_v2(
                p_start, time.monotonic() + max(5.0, per_restart),
                betas_r, lr_init=0.01, max_iter_total=150000,
                lr_schedule="cosine"
            )
            sc_r_full = _score(a_r)
            if sc_r_full < best_score:
                best_score = sc_r_full
                best_seq = a_r.copy()
                print(f"  Restart {i}: score={sc_r_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 4: best_score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Peak shaving + coordinate descent polish ===
    print("Phase 5: Peak shaving polish...")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        # Peak shaving
        t_left = deadline - 20 - time.monotonic()
        if t_left > 10:
            a_shaved, sc_shaved = _peak_shaving(a_pol, n_trials=200, deadline=time.monotonic() + t_left * 0.4)
            sc_shaved_full = _score(a_shaved)
            if sc_shaved_full < best_score:
                best_score = sc_shaved_full
                best_seq = a_shaved.copy()
                a_pol = a_shaved.copy()
                print(f"  Peak shaving: score={sc_shaved_full:.10f}")

        # Coordinate descent
        if time.monotonic() < deadline - 15:
            a_cd, sc_cd = _coordinate_descent_polish(a_pol, deadline - 15)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  Coord descent: score={sc_cd_full:.10f}")

        # One more final Adam polish with very small betas
        if time.monotonic() < deadline - 10:
            a_final = best_seq.copy() / best_seq.sum()
            c_fin = _full_convolve(a_final, a_final)
            mcf = float(c_fin.max())
            betas_ult = [0.00001 * mcf, 0.000005 * mcf, 0.000002 * mcf]
            t_left2 = deadline - 10 - time.monotonic()
            if t_left2 > 5:
                a_ult, sc_ult = _adam_logspace_v2(
                    a_final, time.monotonic() + t_left2,
                    betas_ult, lr_init=0.003, max_iter_total=100000,
                    lr_schedule="cosine"
                )
                sc_ult_full = _score(a_ult)
                if sc_ult_full < best_score:
                    best_score = sc_ult_full
                    best_seq = a_ult.copy()
                    print(f"  Final ultra polish: score={sc_ult_full:.10f}")

    # === Final validation ===
    print(f"\nBest score: {best_score:.10f}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s_final = float(final.sum())
        if s_final < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
