"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 targeting < 1.5027.
Strategies:
1. Multiplicative majorization (Wiberg-style) - fast O(n log n) iterations

import numpy as np
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):

    z = z - np.max(z)

def _simplex_proj(v, target_sum=1.0):
    """Project v onto {x: x>=0, sum(x)=target_sum}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho) == 0:
        return np.maximum(v, 0.0) / max(1e-30, np.sum(np.maximum(v, 0.0))) * target_sum
    cap = rho[-1]
    theta = (cssv[cap] - target_sum) / (cap + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-15:
        return np.full(n, target_sum / n)
    return w * (target_sum / ws)

        per_level = max(2.0, max_wall / len(betas))
    else:
        per_level = min(30.0, max(5.0, (deadline - time.monotonic()) / max(1, len(betas))))
    
    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline - 2.0, time.monotonic() + per_level)
        
        def obj_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = float(e_.sum())
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = float(e_.sum())
            if s_ < 1e-300:
                w_p = np.zeros_like(c_)
                w_p[int(np.argmax(c_))] = 1.0
            else:
                w_p = e_ / s_
            a_rev_ = a_[::-1]
            cross_ = _full_convolve(w_p, a_rev_)
            h = cross_[n - 1:2 * n - 1]
            grad_a = 4.0 * n * h  # gradient w.r.t. a
            # Convert to gradient w.r.t. z via softmax Jacobian
            # da/dz = a*(I - a^T 1) so grad_z = a * (grad_a - a . grad_a)
            W = float(np.dot(a_, grad_a))
            grad_z = a_ * (grad_a - W)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_)
            return grad_z
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                    print(f"    LSE beta={beta:.2e}: score={best_score:.10f}")
        except Exception:
            pass
    
    return best_a, best_score


def _projected_subgradient(a_init, deadline, n_iters=100000):
    """Projected subgradient descent on the exact max."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        
        # Gradient of (a*a)[k_star]: 2 * a[k_star - m]
        a_rev = a[::-1]
        cross = _full_convolve(np.zeros(2 * n - 1), a_rev)
        # Actually, gradient is simply: grad[m] = 2 * a[k_star - m] if 0 <= k_star - m < n else 0
        grad = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for m in range(lo, hi + 1):
            grad[m] = 2.0 * a[k_star - m]
        
        # Backtracking line search
        base_step = 0.1 / (it + 1) * (1.0 / max(c.max(), 1e-10))
        found = False
        for trial in range(10):
            eta = base_step * (0.5 ** trial)
            a_trial = a - eta * grad
            a_trial = _simplex_proj(a_trial, 1.0)
            sc_trial = _score_norm_fn(a_trial)
            if sc_trial < best_sc or sc_trial < _score_norm_fn(a):
                a = a_trial
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a_trial.copy()
                found = True
                break
        
        if not found:
            # Try a smaller base step
            base_step *= 0.5
            if base_step < 1e-15:
                break
    
    return best_a, best_sc


def _peak_local_search(a_init, deadline, rng, n_trials=30000):
    """Local search targeting the highest convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    a /= s
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        K = min(5, len(c))
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])
        
        if it % 4 == 0:
            # Strategy 1: Move mass away from peak-contributing positions
            lo = max(0, max_lag - (n - 1))
            hi = min(n - 1, max_lag)
            if lo <= hi:
                pairs = np.arange(lo, hi + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                src_idx = int(pairs[np.argmax(products)])
                
                # Find safe destination (low contribution to top lags)
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo_k = max(0, kl - (n - 1))
                    hi_k = min(n - 1, kl)
                    for j in range(lo_k, hi_k + 1):
                        partner = kl - j
                        if 0 <= partner < n:
                            contrib[j] += a[partner]
                
                safe_idx = np.argsort(contrib)[:max(1, n // 4)]
                dst_idx = int(safe_idx[rng.integers(len(safe_idx))])
                
                if src_idx == dst_idx:
                    continue
                a_trial = a.copy()
                frac = rng.uniform(0.05, 0.3)
                delta = a_trial[src_idx] * frac
                a_trial[src_idx] -= delta
                a_trial[src_idx] = max(0.0, a_trial[src_idx])
                a_trial[dst_idx] += delta
                tt = a_trial.sum()
                if tt > 1e-10:
                    a_trial /= tt
                    sc = _score_norm_fn(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
        
        elif it % 4 == 1:
            # Strategy 2: Block perturbation
            bs = int(rng.integers(3, min(32, n // 4)))
            start = int(rng.integers(0, max(1, n - bs)))
            a_trial = a.copy()
            mult = rng.uniform(0.8, 1.2, size=bs)
            a_trial[start:start + bs] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        elif it % 4 == 2:
            # Strategy 3: Pair swap
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.4)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1 + amt2)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2 + amt1)
            tt = a_trial.sum()
            if tt > 1e-10:
                a_trial /= tt
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        else:
            # Strategy 4: Gaussian smoothing at peak positions
            sigma = rng.uniform(1.0, 8.0)
            a_trial = a.copy()
            # Subtract a small Gaussian from peak-contributing positions
            lo = max(0, max_lag - (n - 1))
            hi = min(n - 1, max_lag)
            mid = (lo + hi) / 2.0
            positions = np.arange(n, dtype=np.float64)
            gauss = np.exp(-(positions - mid) ** 2 / (2 * sigma ** 2))
            frac = rng.uniform(0.01, 0.05)
            a_trial = a_trial - frac * gauss * a_trial.max()
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc


def _adam_topk(a_init, deadline, rng, lr=0.01, k_top=5, max_iters=50000):
    """Adam optimizer with top-K smoothed gradient on log-sum-exp."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        
        # Top-K weighted gradient
        K = min(k_top, len(c))
        top_idx = np.argsort(c)[-K:]
        weights = np.exp((c[top_idx] - c[top_idx].max()) / (c[top_idx].max() * 0.05 + 1e-10))
        weights /= weights.sum()
        
        p = np.zeros(2 * n - 1)
        p[top_idx] = weights
        a_rev = a_curr[::-1]
        cross = _full_convolve(p, a_rev)
        h = cross[n - 1:2 * n - 1]
        grad_a = 4.0 * n * h
        W = float(np.dot(a_curr, grad_a))
        grad_z = a_curr * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros(n)
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        lr_eff = lr * (1.0 - 0.0001 * it)
        z = z - max(lr_eff, 0.001) * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabolic(n, width=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-((t - 0.5) / width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal(n, sep=0.3, w=0.12):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep / 2.0
    c2 = 0.5 + sep / 2.0
    a = np.exp(-((t - c1) ** 2) / (2 * w ** 2)) + np.exp(-((t - c2) ** 2) / (2 * w ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_shape(n, freq=2, depth=0.3):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining multiplicative majorization,
    L-BFGS continuation, projected subgradient, and local search.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent sequence ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_triangular_weight(n_parent)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, triangular n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    n_work = n_parent
    print(f"Working with n={n_work}")
    
    # --- Phase 0: Diverse initializations (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    profiles = [
        _make_triangular_weight(n_work),
        _make_parabolic(n_work, 0.8),
        _make_parabolic(n_work, 1.0),
        _make_parabolic(n_work, 1.2),
        _make_bimodal(n_work, 0.3, 0.12),
        _make_bimodal(n_work, 0.4, 0.15),
        _make_w_shape(n_work, 2, 0.2),
        _make_w_shape(n_work, 3, 0.15),
    ]
    
    best_init = None
    best_init_sc = np.inf
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        if sc < best_init_sc:
            best_init_sc = sc
            best_init = p.copy()
        update_best(p, sc, f"prof{idx}")
    
    start_a = best_init if best_init is not None else parent.copy()
    print(f"Phase 0 done: best init score={best_init_sc:.10f}")
    
    # --- Phase 1: Multiplicative majorization (300s) ---
    phase1_end = min(deadline, t_start + 330.0)
    print(f"Phase 1: Multiplicative majorization (start score={_score_norm_fn(start_a):.10f})")
    a_mm, sc_mm = _multiplicative_majorization(start_a, phase1_end, rng, alpha=0.8)
    update_best(a_mm, sc_mm, "majorization")
    
    # Also try from parent
    if parent is not None and time.monotonic() < phase1_end - 30:
        a_mm2, sc_mm2 = _multiplicative_majorization(parent, phase1_end, rng, alpha=1.2)
        update_best(a_mm2, sc_mm2, "majorization_parent")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: L-BFGS continuation (400s) ---
    phase2_end = min(deadline, t_start + 730.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"Phase 2: L-BFGS continuation (start score={_score_norm_fn(best_seq):.10f})")
        dl = phase2_end - 10.0
        a_l, sc_l = _lbfgs_lsegap(best_seq, dl, rng, beta_scale=1.0, max_wall=350.0)
        update_best(a_l, sc_l, "lbfgs_main")
        
        # Second pass with different scale
        if time.monotonic() < phase2_end - 40:
            dl2 = phase2_end - 10.0
            a_l2, sc_l2 = _lbfgs_lsegap(best_seq, dl2, rng, beta_scale=0.5, max_wall=100.0)
            update_best(a_l2, sc_l2, "lbfgs_fine")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Adam top-K (200s) ---
    phase3_end = min(deadline, t_start + 930.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"Phase 3: Adam top-K")
        dl = phase3_end - 5.0
        a_ad, sc_ad = _adam_topk(best_seq, dl, rng, lr=0.008, k_top=3)
        update_best(a_ad, sc_ad, "adam_k3")
        
        if time.monotonic() < phase3_end - 30:
            dl2 = phase3_end - 5.0
            a_ad2, sc_ad2 = _adam_topk(best_seq, dl2, rng, lr=0.005, k_top=7)
            update_best(a_ad2, sc_ad2, "adam_k7")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Projected subgradient (150s) ---
    phase4_end = min(deadline, t_start + 1080.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: Projected subgradient")
        dl = phase4_end - 5.0
        a_sg, sc_sg = _projected_subgradient(best_seq, dl, n_iters=50000)
        update_best(a_sg, sc_sg, "subgradient")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Peak-targeted local search (250s) ---
    phase5_end = min(deadline, t_start + 1330.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"Phase 5: Peak-targeted local search")
        dl = phase5_end - 5.0
        a_ls, sc_ls = _peak_local_search(best_seq, dl, rng, n_trials=40000)
        update_best(a_ls, sc_ls, "local_search")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Multi-restart (150s) ---
    phase6_end = min(deadline, t_start + 1480.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print(f"Phase 6: Multi-restart L-BFGS")
        n_restarts = 3
        per_restart = max(30.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 15:
                break
            noise = 0.01 + 0.008 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase6_end - 5, time.monotonic() + per_restart)
            a_r, sc_r = _lbfgs_lsegap(p_start, dl, rng, beta_scale=0.4, max_wall=per_restart)
            update_best(a_r, sc_r, f"restart{i}")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Try different n (100s) ---
    phase7_end = min(deadline, t_start + 1580.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        print(f"Phase 7: Multi-resolution (current n={n_curr})")
        
        # Try 2x
        if n_curr * 2 <= 16384 and time.monotonic() < phase7_end - 30:
            p2 = _interp_profile(best_seq, n_curr * 2)
            sc2 = _score_norm_fn(p2)
            print(f"  n={n_curr*2}: score={sc2:.10f}")
            if sc2 < best_score:
                update_best(p2, sc2, f"interp_n{n_curr*2}")
                if time.monotonic() < phase7_end - 20:
                    dl = min(phase7_end - 10, time.monotonic() + 60.0)
                    a2, sc2b = _lbfgs_lsegap(p2, dl, rng, beta_scale=0.5, max_wall=50.0)
                    update_best(a2, sc2b, f"lbfgs_n{n_curr*2}")
        
        # Try n//2
        if n_curr // 2 >= 1024 and time.monotonic() < phase7_end - 30:
            pH = _interp_profile(best_seq, n_curr // 2)
            scH = _score_norm_fn(pH)
            print(f"  n={n_curr//2}: score={scH:.10f}")
            if scH < best_score:
                update_best(pH, scH, f"interp_n{n_curr//2}")
    
    print(f"Phase 7 done: best={best_score:.10f}")
    
    # --- Phase 8: Final intensive polishing (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(10.0, deadline - time.monotonic())
        print(f"Phase 8: Final polishing, remaining={remaining:.0f}s")
        
        # 8a: Very fine L-BFGS
        if remaining > 50:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.4)
            a_f, sc_f = _lbfgs_lsegap(best_seq, dl, rng, beta_scale=0.2, max_wall=remaining * 0.35)
            update_best(a_f, sc_f, "final_lbfgs")
        
        # 8b: Local search
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.3)
            a_fs, sc_fs = _peak_local_search(best_seq, dl, rng, n_trials=25000)
            update_best(a_fs, sc_fs, "final_local")
        
        # 8c: One more Adam pass with tiny lr
        if time.monotonic() < deadline - 15:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.2)
            a_fa, sc_fa = _adam_topk(best_seq, dl, rng, lr=0.002, k_top=2)
            update_best(a_fa, sc_fa, "final_adam")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_full(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={float(final_seq.sum()):.6f}")
        
        # Validate with evaluator
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval != sc_final:
                        print(f"  WARNING: evaluator={sc_eval:.10f} vs computed={sc_final:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular_weight(4096)
    sc_fb = _score_fn if hasattr(_score_full, '__call__') else None
    print(f"FALLBACK: n=4096")
    return [float(x) for x in fallback]
