2. The gradient of `c[k]` w.r.t. `p[i]` is `2*p[k-i]` (for valid indices) - a clean formula
3. At the optimum, the "pressure" (subgradient) should be constant on the support of p
4. Anti-symmetric perturbation of a symmetric profile reduces the center peak at second order

My approach combines:
- **Mirror descent** with top-K subgradient (efficient, uses the clean gradient structure)
- **Equalization steps** that directly balance the top convolution values
- **Controlled asymmetry injection** via anti-symmetric perturbations
- **Multi-scale optimization** and targeted local refinement

"""
Mirror descent + equalization optimizer for minimizing 2n*max(conv(p,p))/sum(p)^2.
Key insights: symmetric p has score >= 2, so asymmetry is essential.
Uses clean gradient formula dc[k]/dp[i] = 2*p[k-i], top-K subgradient,
anti-symmetric perturbation for exploration, and direct equalization steps.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_gradient_topk(p, topK_idx, K):
    """Compute subgradient: (1/K) * sum_{k in topK} dc[k]/dp = (2/K) * sum_{k in topK} p[k-i]."""
    n = len(p)
    g = np.zeros(n)
    for k in topK_idx:
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # g[i] += 2*p[k-i] for i in [i_lo, i_hi]
            # p[k-i] for i in [i_lo, i_hi] is p[k-i_hi : k-i_lo+1] reversed
            g[i_lo:i_hi + 1] += 2.0 * p[k - i_hi:k - i_lo + 1][::-1]
    g /= K
    return g


def _make_symmetric_profile(n, shape='triangle', tilt=0.0):
    """Generate a profile with optional asymmetry tilt."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    if shape == 'triangle':
        base = (t + 1.0 / n) * (1.0 - t + 1.0 / n)
    elif shape == 'raised_cos':
        base = 0.5 * (1.0 + np.cos(np.pi * u))
    elif shape == 'quartic':
        base = (1.0 - u ** 2) ** 2
    elif shape == 'bspline4':
        v = np.clip(t, 0.02, 0.98)
        base = v ** 3 * (1.0 - v) ** 3
    elif shape == 'bspline6':
        v = np.clip(t, 0.02, 0.98)
        base = v ** 5 * (1.0 - v) ** 5
    elif shape == 'uniform':
        base = np.ones(n)
    else:
        base = (t + 1.0 / n) * (1.0 - t + 1.0 / n)
    
    # Add anti-symmetric tilt
    if tilt != 0.0:
        # Anti-symmetric component: s[i] = u[i] (odd around center)
        s = u.copy()
        base = base * (1.0 + tilt * s)
    
    base = np.maximum(base, 1e-15)
    return base / base.sum()


def _make_asymmetric_profile(n, bias=0.3, width=0.4):
    """Create a profile with controlled asymmetry by shifting a bump."""
    t = np.linspace(0.0, 1.0, n)
    center = 0.5 + bias * 0.3
    a = np.exp(-(t - center) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _mirror_descent(p_init, deadline, K=8, lr_init=0.5, eval_interval=20, max_iters=200000):
    """
    Mirror descent with top-K subgradient for minimizing max(conv(p,p)).
    Uses the clean gradient formula: dc[k]/dp[i] = 2*p[k-i].
    """
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-20))
    z = z - z.mean()
    
    best_p = p_init.copy()
    best_sc = _score_norm_fn(p_init)
    lr = lr_init
    
    p = p_init.copy()
    c = _full_convolve(p, p)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Find top K
        topK_idx = np.argsort(c)[-K:]
        
        # Compute subgradient
        g = _compute_gradient_topk(p, topK_idx, K)
        
        # Center the gradient (doesn't change the simplex update)
        pg = float(np.dot(p, g))
        g_c = g - pg
        
        # Mirror descent update in z-space
        z = z - lr * g_c
        z = z - z.mean()
        
        # Periodically update p and evaluate
        if it % eval_interval == 0:
            p = _softmax_fn(z)
            c = _full_convolve(p, p)
            sc = float(2.0 * n * c.max())
            if sc < best_sc:
                best_sc = sc
                best_p = p.copy()
        
        # Adaptive learning rate
        if it % 500 == 0:
            lr *= 0.97
        
        # Occasionally recompute c more frequently for better topK tracking
        if it % (eval_interval * 5) == 0 and it > 0:
            p = _softmax_fn(z)
            c = _full_convolve(p, p)
    
    # Final evaluation
    p = _softmax_fn(z)
    sc = _score_norm_fn(p)
    if sc < best_sc:
        best_sc = sc
        best_p = p.copy()
    
    return best_p, best_sc


def _equalization_step(p, K=5, n_inner=50, lr=0.1):
    """
    Direct equalization: iteratively adjust p to make top K convolution values equal.
    At optimum, the subgradient should be constant on support.
    """
    n = len(p)
    p = p.copy()
    best_p = p.copy()
    best_sc = _score_norm_fn(p)
    
    for outer in range(n_inner):
        c = _full_convolve(p, p)
        topK_idx = np.argsort(c)[-K:]
        
        # Compute subgradient
        g = _compute_gradient_topk(p, topK_idx, K)
        
        # We want g to be constant on support. 
        # Move in direction that equalizes g.
        # Use: delta_p proportional to -(g - mean_g_on_support)
        support = p > 1e-10
        if support.sum() < 2:
            break
        g_mean = g[support].mean()
        delta = -(g - g_mean)
        
        # Scale to maintain sum = 1
        s = p[support].sum()
        delta[support] -= delta[support].mean() * p[support].sum() / max(s, 1e-15)
        
        # Update
        p_new = p + lr * delta
        p_new = np.maximum(p_new, 0.0)
        s_new = p_new.sum()
        if s_new < 1e-10:
            break
        p_new = p_new / s_new
        
        sc = _score_norm_fn(p_new)
        if sc < best_sc:
            best_sc = sc
            best_p = p_new.copy()
        
        p = p_new
        lr *= 0.95
    
    return best_p, best_sc


def _peak_directed_refinement(p_init, deadline, n_trials=50000):
    """
    Targeted local search: identify specific (i,j) pairs responsible for peaks
    and make surgical adjustments.
    """
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_sc = _score_norm_fn(p)
    rng_local = np.random.default_rng(12345)
    
    for it in range(n_trials):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(p, p)
        M = c.max()
        max_k = int(np.argmax(c))
        
        # Find the dominant pair at the peak
        i_min = max(0, max_k - (n - 1))
        i_max = min(n - 1, max_k)
        if i_min <= i_max:
            pairs = np.arange(i_min, i_max + 1)
            partners = max_k - pairs
            products = p[pairs] * p[partners]
            idx = pairs[int(np.argmax(products))]
            partner = max_k - idx
            
            # Strategy 1: Reduce the larger of the pair
            if it % 4 == 0:
                p_trial = p.copy()
                if p[idx] >= p[partner]:
                    reduce_i = idx
                    keep_i = partner
                else:
                    reduce_i = partner
                    keep_i = idx
                
                # Move mass to a "safe" position
                # Safe = position that minimizes contribution to top lags
                top_lags = np.argsort(c)[-5:]
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo2 = max(0, kl - (n - 1))
                    hi2 = min(n - 1, kl)
                    contrib[lo2:hi2 + 1] += p[kl - hi2:kl - lo2 + 1][::-1]
                
                safe_positions = np.argsort(contrib)[:max(3, n // 10)]
                target = int(safe_positions[rng_local.integers(len(safe_positions))])
                
                frac = rng_local.uniform(0.05, 0.3)
                delta = p_trial[reduce_i] * frac
                p_trial[reduce_i] -= delta
                p_trial[reduce_i] = max(0.0, p_trial[reduce_i])
                p_trial[target] += delta
                p_trial = p_trial / p_trial.sum()
                
                sc = _score_norm_fn(p_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p_trial.copy()
                    p = p_trial.copy()
            
            # Strategy 2: Split the peak pair's contribution
            elif it % 4 == 1:
                p_trial = p.copy()
                # Reduce both members of the pair slightly
                frac = rng_local.uniform(0.03, 0.15)
                d1 = p_trial[idx] * frac
                d2 = p_trial[partner] * frac
                p_trial[idx] = max(0.0, p_trial[idx] - d1)
                p_trial[partner] = max(0.0, p_trial[partner] - d2)
                # Distribute to neighbors of each
                for target_off in [-2, -1, 1, 2]:
                    t1 = idx + target_off
                    t2 = partner + target_off
                    if 0 <= t1 < n:
                        p_trial[t1] += d1 * 0.25
                    if 0 <= t2 < n:
                        p_trial[t2] += d2 * 0.25
                p_trial = np.maximum(p_trial, 0.0)
                p_trial = p_trial / p_trial.sum()
                
                sc = _score_norm_fn(p_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p_trial.copy()
                    p = p_trial.copy()
            
            # Strategy 3: Block perturbation on the "responsible" region
            elif it % 4 == 2:
                # Identify the region around the peak-contributing indices
                center = (idx + partner) // 2
                radius = max(5, n // 50)
                lo = max(0, center - radius)
                hi = min(n, center + radius)
                
                p_trial = p.copy()
                n_block = hi - lo
                mult = rng_local.uniform(0.85, 1.15, size=n_block)
                p_trial[lo:hi] *= mult
                p_trial = np.maximum(p_trial, 1e-15)
                p_trial = p_trial / p_trial.sum()
                
                sc = _score_norm_fn(p_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p_trial.copy()
                    p = p_trial.copy()
            
            # Strategy 4: Global tilt adjustment
            else:
                p_trial = p.copy()
                t = np.linspace(0, 1, n)
                tilt_strength = rng_local.uniform(0.01, 0.08)
                direction = rng_local.choice([-1, 1])
                multiplier = 1.0 + direction * tilt_strength * (2.0 * t - 1.0)
                p_trial *= multiplier
                p_trial = np.maximum(p_trial, 1e-15)
                p_trial = p_trial / p_trial.sum()
                
                sc = _score_norm_fn(p_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p_trial.copy()
                    p = p_trial.copy()
    
    return best_p, best_sc


def _adaptive_mirror_descent(p_init, deadline, K_start=3, K_end=15, max_iters=300000):
    """
    Mirror descent with adaptive K (start with few peaks, increase as we get closer to optimum).
    Also uses a more aggressive initial step size.
    """
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-20))
    z = z - z.mean()
    
    best_p = p_init.copy()
    best_sc = _score_norm_fn(p_init)
    
    p = p_init.copy()
    c = _full_convolve(p, p)
    
    lr = 1.0
    total_iters = max_iters
    eval_interval = 10
    
    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive K: start small, grow
        progress = it / total_iters
        K = K_start + int((K_end - K_start) * progress)
        K = max(1, min(K, len(c)))
        
        topK_idx = np.argsort(c)[-K:]
        
        g = _compute_gradient_topk(p, topK_idx, K)
        pg = float(np.dot(p, g))
        g_c = g - pg
        
        z = z - lr * g_c
        z = z - z.mean()
        z = np.clip(z, -50, 50)
        
        if it % eval_interval == 0:
            p = _softmax_fn(z)
            c = _full_convolve(p, p)
            sc = float(2.0 * n * c.max())
            if sc < best_sc:
                best_sc = sc
                best_p = p.copy()
        
        # Learning rate schedule: aggressive early, gentle late
        if it < total_iters // 4:
            lr = 1.0 * (1.0 - 0.5 * progress)
        elif it < 3 * total_iters // 4:
            lr = 0.5 * (1.0 - 0.5 * (progress - 0.25) / 0.75)
        else:
            lr = max(0.05, 0.25 * (1.0 - (progress - 0.75) / 0.25))
        
        # Update c every few iterations for tracking
        if it % (eval_interval * 3) == 0:
            p = _softmax_fn(z)
            c = _full_convolve(p, p)
    
    p = _softmax_fn(z)
    sc = _score_norm_fn(p)
    if sc < best_sc:
        best_sc = sc
        best_p = p.copy()
    
    return best_p, best_sc


def _asymmetry_search(p_sym, deadline, n_trials=200):
    """
    Search over anti-symmetric perturbations of a symmetric profile.
    p[i] = f[i] * (1 + eps * s[i]) where s is anti-symmetric.
    """
    n = len(p_sym)
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0  # Anti-symmetric
    
    best_p = p_sym.copy()
    best_sc = _score_norm_fn(p_sym)
    rng_local = np.random.default_rng(777)
    
    # Try different anti-symmetric patterns
    patterns = [
        u,                              # Linear tilt
        np.sin(np.pi * u),              # Sine tilt
        np.sin(2 * np.pi * u),          # Higher freq
        u ** 3,                         # Cubic
        np.sign(u) * (1 - np.abs(u)),   # Triangular
        np.sin(3 * np.pi * u),
    ]
    
    for pat in patterns:
        for eps in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            for sign in [1, -1]:
                if time.monotonic() > deadline:
                    return best_p, best_sc
                
                p_trial = p_sym * (1.0 + sign * eps * pat)
                p_trial = np.maximum(p_trial, 1e-15)
                p_trial = p_trial / p_trial.sum()
                
                sc = _score_norm_fn(p_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_p = p_trial.copy()
    
    return best_p, best_sc


def _local_refine_gradient(p_init, deadline, lr=0.01, max_iters=50000):
    """
    Fine-tuning with very small steps using exact top-1 gradient.
    """
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-20))
    z = z - z.mean()
    
    best_p = p_init.copy()
    best_sc = _score_norm_fn(p_init)
    p = p_init.copy()
    c = _full_convolve(p, p)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        max_k = int(np.argmax(c))
        
        # Gradient of c[max_k]
        i_lo = max(0, max_k - n + 1)
        i_hi = min(n - 1, max_k)
        g = np.zeros(n)
        g[i_lo:i_hi + 1] = 2.0 * p[max_k - i_hi:max_k - i_lo + 1][::-1]
        
        # Center
        pg = float(np.dot(p, g))
        g_c = g - pg
        
        z = z - lr * g_c
        z = z - z.mean()
        
        if it % 30 == 0:
            p = _softmax_fn(z)
            c = _full_convolve(p, p)
            sc = float(2.0 * n * c.max())
            if sc < best_sc:
                best_sc = sc
                best_p = p.copy()
        
        # Very gentle decay
        if it % 2000 == 0:
            lr *= 0.95
    
    return best_p, best_sc

            p_md3, sc_md3 = _adaptive_mirror_descent(best_seq, dl_md, K_start=5, K_end=20, max_iters=200000)
            update_best(p_md3, sc_md3, "md_post_eq")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Peak-directed local search (300s) ---
    phase4_end = min(deadline, t_start + 1220.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        dl_ps = phase4_end - 10
        p_ps, sc_ps = _peak_directed_refinement(best_seq, dl_ps, n_trials=80000)
        update_best(p_ps, sc_ps, "peak_directed")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Multi-restart with different asymmetries (250s) ---
    phase5_end = min(deadline, t_start + 1470.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        n_restarts = 5
        per_restart = max(20.0, (phase5_end - time.monotonic()) / n_restarts)
        rng = np.random.default_rng(int(seed) % (2**63 - 1))
        
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 10:
                break
            
            # Create perturbed start
            noise_level = 0.02 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            
            # Add structured perturbation (not just noise)
            if i % 2 == 0:
                # Add tilt
                t = np.linspace(0, 1, len(best_seq))
                z += noise_level * (2 * t - 1) * rng.normal()
            else:
                # Add local perturbation
                k = int(rng.integers(0, len(z) - 100))
                z[k:k+100] += noise_level * rng.normal(size=100)
            
            z = np.clip(z, -40, 40)
            z = z - z.mean()
            p_start = _softmax_fn(z)
            
            dl = min(phase5_end - 10, time.monotonic() + per_restart)
            p_r, sc_r = _adaptive_mirror_descent(p_start, dl, K_start=3, K_end=12, max_iters=100000)
            update_best(p_r, sc_r, f"restart_{i}")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Fine-tuning (200s) ---
    phase6_end = min(deadline, t_start + 1670.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        # Very fine gradient steps
        dl_fine = min(phase6_end - 30, time.monotonic() + 80.0)
        p_fine, sc_fine = _local_refine_gradient(best_seq, dl_fine, lr=0.005, max_iters=100000)
        update_best(p_fine, sc_fine, "fine_tune")
        
        # Another round of peak-directed
        if time.monotonic() < phase6_end - 30:
            dl_ps2 = phase6_end - 20
            p_ps2, sc_ps2 = _peak_directed_refinement(best_seq, dl_ps2, n_trials=30000)
            update_best(p_ps2, sc_ps2, "peak_fine")
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Try different n (final 100s) ---
    if best_seq is not None and time.monotonic() < deadline - 40:
        n_curr = len(best_seq)
        
        # Try 2x resolution
        for target_n in [n_curr * 2]:
            if target_n > 32768 or target_n <= n_curr:
                continue
            if time.monotonic() > deadline - 60:
                break
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp to n={target_n}: score={sc_interp:.10f}")
            if sc_interp < best_score:
                update_best(p_interp, sc_interp, f"interp_n{target_n}")
            
            if time.monotonic() < deadline - 40:
                dl_int = min(deadline - 35, time.monotonic() + 40.0)
                p_int, sc_int = _adaptive_mirror_descent(p_interp, dl_int, K_start=5, K_end=15, max_iters=50000)
                update_best(p_int, sc_int, f"md_n{target_n}")
    
    # --- Final: One more polishing pass ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        dl_final = deadline - 15
        p_final, sc_final = _adaptive_mirror_descent(best_seq, dl_final, K_start=5, K_end=20, max_iters=100000)
        update_best(p_final, sc_final, "final_polish")
    
    # --- Validate and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_symmetric_profile(8192, 'triangle', 0.3)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        if len(final_seq) <= 32768:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_symmetric_profile(8192, 'triangle', 0.3)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
