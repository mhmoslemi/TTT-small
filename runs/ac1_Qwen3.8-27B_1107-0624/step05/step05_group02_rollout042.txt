    n = len(a)

    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]

def _conv_self(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _conv(a, b):
    n, m = len(a), len(b)
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -100, 100))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_lse(a, beta):
    n = len(a)
    c = _conv_self(a)
    c_max = c.max()
    sh = np.clip((c - c_max) * beta, -50.0, 50.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        val = c_max
    else:
        w = e / s
        val = c_max + np.log(s) / beta
    af = a[::-1]
    cw = _conv(w, af)
    grad = 2.0 * n * cw[n - 1:2 * n - 1]
    return grad, val


def _adam_run(a0, dl, lr, b0, b1, nph, iph):

    z = np.log(np.maximum(a0, 1e-15))
    z -= z.mean()
    ba = a0.copy()
    bs = _score(a0)
    mb = np.zeros(n)
    vb = np.zeros(n)
    ts = 0
    B1, B2, ep = 0.9, 0.999, 1e-8
    lr_c = lr
    for ph in range(nph):
        if time.monotonic() > dl:
            break
        bt = b0 * (b1 / b0) ** (ph / max(1, nph - 1))
        it = 0
        pat = 0
        while it < iph:
            if time.monotonic() > dl:
                break
            it += 1
            ts += 1
            a = _softmax(z)
            ga, _ = _grad_lse(a, bt)
            mg = float(np.dot(a, ga))
            gz = a * (ga - mg)
            if not np.all(np.isfinite(gz)):
                break
            mb = B1 * mb + (1 - B1) * gz
            vb = B2 * vb + (1 - B2) * gz * gz
            mh = mb / (1 - B1 ** ts)
            vh = vb / (1 - B2 ** ts)
            up = lr_c * mh / (np.sqrt(vh) + ep)
            un = np.linalg.norm(up)
            if un > 2.0:
                up *= 2.0 / un
            z -= up
            z = np.clip(z, -50, 50)
            z -= z.mean()
            if it % 100 == 0:
                sc = _score(a)
                if sc < bs:
                    bs = sc
                    ba = a.copy()
                    pat = 0
                else:
                    pat += 1
                if pat > 80:

                lr_c *= 0.93
        af = _softmax(z)
        sf = _score(af)
        if sf < bs:

            ba = af.copy()
    return ba, bs

def _upscale(a, nn):
    n = len(a)
    if n == nn:
        return a.copy()
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, nn)
    an = np.interp(x1, x0, a)
    an = np.maximum(an, 1e-12)
    return an / an.sum()

    z = np.log(np.maximum(a, 1e-15))
    z += sg * rg.standard_normal(len(a))

Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS and Adam on softmax parameterization with adaptive log-sum-exp
smoothing, diverse perturbations (log-space, frequency-domain, kernel
convolution), multi-restart, and multi-n exploration.

import numpy as np
import time
from scipy.optimize import minimize


def _np2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _cself(a):
    n = len(a)
    N = _np2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _cconv(a, b):
    n, m = len(a), len(b)
    N = _np2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]


def _sc(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _cself(a)
    return float(2.0 * n * c.max() / (s * s))


def _smx(z):
    z = z - z.max()
    e = np.exp(np.clip(z, -100.0, 100.0))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _zlog(a):
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    return z


def _grl(a, bt):
    """Gradient of 2n*LSE_beta(conv(a,a)) w.r.t. a, and the LSE value."""
    n = len(a)
    c = _cself(a)
    cm = float(c.max())
    sh = np.clip((c - cm) * bt, -50.0, 50.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        v = cm
    else:
        w = e / s
        v = cm + np.log(s) / bt
    af = a[::-1]
    cw = _cconv(w, af)
    g = 4.0 * n * cw[n - 1: 2 * n - 1]
    return g, v


def _lbf(z, n, bt):
    """Objective for L-BFGS: 2n * LSE_beta(conv(softmax(z), softmax(z)))."""
    a = _smx(z)
    c = _cself(a)
    cm = float(c.max())
    sh = np.clip((c - cm) * bt, -50.0, 50.0)
    e = np.exp(sh)
    s = float(e.sum())
    if s < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + np.log(s) / bt)

    """Gradient of L-BFGS objective w.r.t. z (softmax parameter)."""
    a = _smx(z)
    ga, _ = _grl(a, bt)
    mg = float(np.dot(a, ga))
    return a * (ga - mg)
