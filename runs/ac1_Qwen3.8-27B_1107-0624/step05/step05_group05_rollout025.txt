Multi-phase gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient computation with log-sum-exp smoothing,
Adam optimizer with adaptive learning rate, and random restarts.

import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    out_n = len(a) + len(b) - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unnorm(a):
    """Score assuming sum(a) = 1, i.e., 2*n*max(c)."""
    n = len(a)
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _grad_from_w(a, w):
    """Compute grad_a[j] = 2 * sum_m a[m]*w[m+j] for j=0..n-1.
    a has length n, w has length 2n-1."""
    n = len(a)
    N = _next_pow2(3 * n)
    fw = np.fft.rfft(w, N)
    fa = np.fft.rfft(a, N)
    corr = np.fft.irfft(fw * np.conj(fa), N)
    return 2.0 * corr[:n]

def _adam_loop(a_init, deadline, beta_start, beta_end, max_iter, lr_start, lr_end):
    """Adam optimizer with exponential beta and lr decay."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_unnorm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    prev_best = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax(z)
        c = _fft_convolve(a, a)
        c_max = float(c.max())
        
        beta = beta_start * (max(beta_end, beta_start * 1e-6) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-12)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = e / s
        
        grad_a = _grad_from_w(a, w)
        
        # Check for numerical issues
        if not np.all(np.isfinite(grad_a)):
            break

        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)

        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)

        vh = v_buf / (1.0 - b2 ** t)

        un = np.linalg.norm(update)

    print(f"  Phase 1: score_unnorm={sc1:.10f}, full_score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a1.copy()

    # Phase 2: Extended Adam with slower beta decay (200-600s)
    print("Phase 2: Extended Adam (slow beta decay)")
    p2_end = min(deadline, t_start + 600.0)
    a_best = best_seq.copy()
    c_best = _fft_convolve(a_best, a_best)
    max_cb = float(c_best.max())
    a2, sc2 = _adam_loop_adaptive(a_best, p2_end, beta_start=0.005 * max_cb,
                                  beta_end=0.00005 * max_cb, max_iter=10000000, lr=0.01)
    full_sc = _score(a2)
    print(f"  Phase 2: score_unnorm={sc2:.10f}, full_score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a2.copy()

    # Phase 3: Try different sequence lengths (600-900s)
    print("Phase 3: Multi-length exploration")
    p3_end = min(deadline, t_start + 900.0)
    for n_try in [1024, 4096]:
        if time.monotonic() > p3_end - 60:
            break
        a_try = _resample_profile(best_seq, n_try)
        c_try = _fft_convolve(a_try, a_try)
        max_ct = float(c_try.max())
        deadline_try = min(p3_end - 10, time.monotonic() + 120)
        a_try, sc_try = _adam_loop(a_try, deadline_try, beta_start=0.01 * max_ct,
                                   beta_end=0.0005 * max_ct, max_iter=5000000,
                                   lr_start=0.015, lr_end=0.003, eval_every=100)
        full_sc_try = _score(a_try)
        print(f"  n={n_try}: score={full_sc_try:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_try) and full_sc_try < best_score:
            best_score = full_sc_try
            best_seq = a_try.copy()
            print(f"  New best from n={n_try}!")

    # Phase 4: Fine Adam on best (900-1300s)
    print("Phase 4: Fine Adam refinement")
    p4_end = min(deadline, t_start + 1300.0)
    a_best = best_seq.copy()
    c_best = _fft_convolve(a_best, a_best)
    max_cb = float(c_best.max())
    a4, sc4 = _adam_loop_adaptive(a_best, p4_end, beta_start=0.002 * max_cb,
                                  beta_end=0.00001 * max_cb, max_iter=10000000, lr=0.005)
    full_sc = _score(a4)
    print(f"  Phase 4: score_unnorm={sc4:.10f}, full_score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a4.copy()

    # Phase 5: Random restarts with perturbation (1300-1650s)
    print("Phase 5: Random restarts")
    p5_end = min(deadline, t_start + 1650.0)
    n_restarts = 8
    restart_time = max(30.0, (p5_end - time.monotonic()) / n_restarts)
    for r in range(n_restarts):
        if time.monotonic() > p5_end - 10:
            break
        sigma = 0.02 + 0.015 * r
        a_r = _perturb(best_seq, sigma, rng)
        c_r = _fft_convolve(a_r, a_r)
        max_cr = float(c_r.max())
        deadline_r = min(p5_end - 10, time.monotonic() + restart_time)
        a_r, sc_r = _adam_loop(a_r, deadline_r, beta_start=0.005 * max_cr,
                               beta_end=0.0001 * max_cr, max_iter=2000000,
                               lr_start=0.01, lr_end=0.002, eval_every=100)
        full_sc_r = _score(a_r)
        print(f"  Restart[{r}] sigma={sigma:.3f}: score={full_sc_r:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc_r) and full_sc_r < best_score:
            best_score = full_sc_r
            best_seq = a_r.copy()
            print(f"  New best from restart {r}!")

    # Phase 6: Ultra-fine final Adam (1650 to deadline)
    print("Phase 6: Ultra-fine final Adam")
    if time.monotonic() < deadline - 20:
        a_best = best_seq.copy()
        c_best = _fft_convolve(a_best, a_best)
        max_cb = float(c_best.max())
        a6, sc6 = _adam_loop_adaptive(a_best, deadline - 5, beta_start=0.0005 * max_cb,
                                      beta_end=0.000001 * max_cb, max_iter=10000000, lr=0.003)
        full_sc = _score(a6)
        print(f"  Phase 6: score_unnorm={sc6:.10f}, full_score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a6.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
