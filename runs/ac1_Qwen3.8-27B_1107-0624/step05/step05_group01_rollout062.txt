"""
Optimizes non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines L-BFGS log-sum-exp continuation, Nesterov-accelerated gradient,
active-set peak shaving, and adaptive beta scheduling.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)
    return r[:out_n]


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(z, beta, n):
    """Compute gradient of 2n * LSE(c/beta) w.r.t. z, where a=softmax(z)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z), 0.0, np.zeros(len(c))
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return grad, 2.0 * n * (cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))), w


def _lsei_obj(z, beta, n):
    grad, obj, _ = _compute_gradient(z, beta, n)
    return obj


def _lbfgs_phase(a_init, deadline, beta_list=None, maxiter=5000):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        beta_list = [m0 * f for f in [0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                                       0.003, 0.002, 0.001, 0.0005, 0.0003,
                                       0.0001, 0.00005, 0.00002, 0.00001]]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(_lsei_obj, z, args=(beta, n),
                           method='L-BFGS-B', jac=lambda zz, bb, nn: _compute_gradient(zz, bb, nn)[0],
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxls': 80, 'maxfun': 30000})
            if res.x is not None and np.all(np.isfinite(res.x)):
                a_new = _softmax_fn(res.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = res.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _nesterov_phase(a_init, deadline, lr_init=0.04, beta_init_frac=0.01,
                    momentum=0.92, max_iter=500000):
    """Nesterov-accelerated gradient on log-sum-exp surrogate with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = beta_init_frac * cm0
    beta_floor = 1e-8 * cm0

    v = np.zeros(n)
    y = z.copy()
    t = 0
    lr = lr_init
    no_improve = 0
    check_interval = 30
    patience = 50

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a_y = _softmax_fn(y)
        grad_y, _, _ = _compute_gradient(y, beta, n)

        v = momentum * v - lr * grad_y
        z_new = z + v
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()

        a_z = _softmax_fn(z_new)
        sc_check = _score_norm(a_z)

        if np.isfinite(sc_check) and sc_check < best_score - 1e-14:
            best_score = sc_check
            best_a = a_z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            lr *= 0.7
            v *= 0.5
            beta = max(beta * 0.8, beta_floor)
            no_improve = 0

        if t % (check_interval * 5) == 0:
            beta = max(beta * 0.85, beta_floor)
            lr = max(lr * 1.05, lr_init * 0.3)

        z = z_new
        y = z + momentum * v

    return best_a, best_score


def _peak_shaving(a_init, deadline, n_peaks=8, max_rounds=200):
    """Direct peak shaving: iteratively reduce the top peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    S = float(a.sum())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        current_score = 2.0 * n * cm / (S * S)

        # Get top peaks
        sorted_idx = np.argsort(c)[::-1][:n_peaks]
        peak_vals = c[sorted_idx]
        # Weight by how close to max
        weights = np.maximum(peak_vals - 0.95 * cm, 0.0)
        if weights.sum() < 1e-30:
            weights = np.ones(n_peaks)
        weights = weights / weights.sum()

        # Compute combined gradient direction to reduce peaks
        # dc_k/da_j = 2 * a[k-j] (out of range = 0)
        grad = np.zeros(n)
        for idx, pi in enumerate(sorted_idx):
            wi = weights[idx]
            # grad[j] += wi * 2 * a[pi - j] for valid indices
            j_lo = max(0, pi - (n - 1))
            j_hi = min(n - 1, pi)
            if j_lo <= j_hi:
                a_slice = a[pi - j_hi:pi - j_lo + 1][::-1]
                grad[j_lo:j_hi + 1] += wi * 2.0 * a_slice

        # Project onto sum=0 subspace
        grad = grad - np.mean(grad)

        # Step size: small enough to not overshoot
        eps = 0.01 * cm / (np.linalg.norm(grad) + 1e-30)
        eps = min(eps, 0.05)

        a_new = a - eps * grad
        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new * S

        sc_new = _score_norm(a_new)
        if np.isfinite(sc_new) and sc_new < best_score - 1e-15:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_try = a - 0.3 * eps * grad
            a_try = np.maximum(a_try, 0.0)
            s_try = float(a_try.sum())
            if s_try > 1e-15:
                a_try = a_try / s_try * S
                sc_try = _score_norm(a_try)
                if np.isfinite(sc_try) and sc_try < best_score - 1e-15:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                else:
                    break
            else:
                break

    return best_a, best_score


def _adam_phase(a_init, deadline, lr=0.015, beta_init_frac=0.005):
    """Adam optimizer on log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = beta_init_frac * cm0
    beta_floor = 1e-9 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps_adam = 0.9, 0.9999, 1e-8
    no_improve = 0
    check_interval = 30
    patience = 80

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1

        grad, _, _ = _compute_gradient(z, beta, n)

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps_adam)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1

            beta = max(beta * 0.97, beta_floor)

            if no_improve > patience:
                lr *= 0.75
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, p):
    """Various parametric initializations."""
    t = np.linspace(0, 1, n)
    if p == 'power2':
        a = (0.1 + 0.9 * t) ** 2
    elif p == 'power3':
        a = (0.05 + 0.95 * t) ** 3
    elif p == 'exp3':
        a = np.exp(3.0 * (t - 0.5))
    elif p == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif p == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif p == 'linear':
        a = 0.1 + 0.9 * t
    elif p == 'cubic':
        a = 0.05 + 0.3 * t + 0.65 * t**3
    elif p == 'quartic':
        a = 0.02 + 0.1 * t + 0.88 * t**4
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0.copy()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: L-BFGS continuation, Nesterov gradient,
    peak shaving, Adam polish, and alternative initializations.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        print(f"No parent. Using greedy init n={n}")
        update_best(parent, _score_norm(parent), "greedy_default")

    print(f"Starting: n={n}, best={best_score:.10f}")

    # === Phase 1: L-BFGS continuation (450s) ===
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 450.0)
    a_s = best_seq.copy()
    a_b1, sc1 = _lbfgs_phase(a_s, p1_end, maxiter=8000)
    update_best(a_b1, sc1, "lbfgs_main")

    # Try greedy init through L-BFGS
    if time.monotonic() < p1_end - 80:
        a_g = _greedy_init(n)
        sc_g = _score_norm(a_g)
        if sc_g < best_score * 1.05:
            a_gb, sc_gb = _lbfgs_phase(a_g, min(p1_end - 20, time.monotonic() + 120))
            update_best(a_gb, sc_gb, "lbfgs_greedy")

    # Try a parametric init
    if time.monotonic() < p1_end - 60:
        a_p = _parametric_init(n, 'power3')
        sc_p = _score_norm(a_p)
        if sc_p < best_score * 1.05:
            a_pb, sc_pb = _lbfgs_phase(a_p, min(p1_end - 10, time.monotonic() + 80))
            update_best(a_pb, sc_pb, "lbfgs_param")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov accelerated gradient (400s) ===
    print("Phase 2: Nesterov gradient")
    p2_end = min(deadline, t_start + 850.0)
    if time.monotonic() < p2_end:
        a_nv, sc_nv = _nesterov_phase(best_seq, p2_end, lr_init=0.03,
                                      beta_init_frac=0.005, momentum=0.90)
        update_best(a_nv, sc_nv, "nesterov")

        # Second Nesterov round with different params
        if time.monotonic() < p2_end - 60:
            a_nv2, sc_nv2 = _nesterov_phase(best_seq, p2_end, lr_init=0.015,
                                            beta_init_frac=0.002, momentum=0.93)
            update_best(a_nv2, sc_nv2, "nesterov_slow")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Peak shaving (200s) ===
    print("Phase 3: Peak shaving")
    p3_end = min(deadline, t_start + 1050.0)
    if time.monotonic() < p3_end:
        for k_peaks in [5, 10, 15]:
            if time.monotonic() > p3_end - 30:
                break
            a_ps, sc_ps = _peak_shaving(best_seq, p3_end, n_peaks=k_peaks, max_rounds=100)
            update_best(a_ps, sc_ps, f"peak_shave_K{k_peaks}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Adam polish (250s) ===
    print("Phase 4: Adam polish")
    p4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < p4_end:
        a_ad, sc_ad = _adam_phase(best_seq, p4_end, lr=0.012, beta_init_frac=0.003)
        update_best(a_ad, sc_ad, "adam_012")

        if time.monotonic() < p4_end - 60:
            a_ad2, sc_ad2 = _adam_phase(best_seq, p4_end, lr=0.005, beta_init_frac=0.001)
            update_best(a_ad2, sc_ad2, "adam_slow")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Final L-BFGS with tiny beta (250s) ===
    print("Phase 5: Final L-BFGS tiny beta")
    p5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p5_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.0003, 0.0001, 0.00005, 0.00002, 0.00001]]
        a_f, sc_f = _lbfgs_phase(best_seq, p5_end, beta_list=tiny_betas, maxiter=10000)
        update_best(a_f, sc_f, "final_tiny")

        # Another peak shave after final L-BFGS
        if time.monotonic() < p5_end - 30:
            a_ps2, sc_ps2 = _peak_shaving(best_seq, p5_end, n_peaks=5, max_rounds=50)
            update_best(a_ps2, sc_ps2, "final_peak_shave")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try different n or re-optimization (remaining) ===
    if time.monotonic() < deadline - 50:
        print("Phase 6: Additional attempts")

        # Try re-optimizing from best with small perturbation
        a_pert = best_seq + 0.001 * rng.standard_normal(len(best_seq))
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        sc_pert = _score_norm(a_pert)
        if sc_pert < best_score * 1.01:
            a_pe, sc_pe = _nesterov_phase(a_pert, min(deadline - 20, time.monotonic() + 100),
                                          lr_init=0.02, beta_init_frac=0.001)
            update_best(a_pe, sc_pe, "perturbed_nesterov")

        # Try interpolation to larger n
        if time.monotonic() < deadline - 60:
            n_new = min(2 * n, 16384)
            if n_new > n:
                a_interp = _interp_profile(best_seq, n_new)
                sc_in = _score_fn(a_interp)
                print(f"  Interp to n={n_new}: score={sc_in:.10f}")
                if sc_in < best_score:
                    update_best(a_interp, _score_norm(a_interp), f"interp_n{n_new}")
                    a_ib, sc_ib = _lbfgs_phase(a_interp, min(deadline - 20, time.monotonic() + 120))
                    update_best(a_ib, sc_ib, f"lbfgs_n{n_new}")

    # === Final validation ===
    print(f"\nFinal best_score (normalized) = {best_score:.10f}")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator if possible
        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
