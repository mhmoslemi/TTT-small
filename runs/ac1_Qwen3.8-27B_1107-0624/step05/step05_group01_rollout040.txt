"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategies: geometric/binomial init, L-BFGS log-sum-exp continuation,
coordinate descent polish with exact 1D minimization, and fine-tuning.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Normalizing score: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _greedy_init(n):
    """Binomial-like initialization (reversed to make increasing)."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _geometric_init(n, c=2.0):
    """Geometric initialization: a[i] = exp(c*i/(n-1)) for c > 0."""
    t = np.arange(n) / (n - 1)
    a = np.exp(c * t)
    return a / a.sum()


def _power_init(n, p=1.0):
    """Power law: a[i] = (i+1)^p."""
    t = np.arange(1, n + 1)
    a = t ** p
    return a / a.sum()


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lbfgs_phase(a_init, deadline, beta_fracs=None, maxiter=5000):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    if beta_fracs is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        beta_fracs = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    # Actually, let me use absolute beta values based on m0
    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    betas = [m0 * f for f in beta_fracs]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def obj(z_in, beta=beta, n=n):
            a_in = _softmax(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)
        
        def grad(z_in, beta=beta, n=n):
            a_in = _softmax(z_in)
            c_in = _full_convolve(a_in, a_in)

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses geometric/power-law initialization, L-BFGS log-sum-exp continuation,
coordinate descent with exact 1D ternary-search minimization, and fine-tuning.
Key insight: asymmetric (increasing) sequences can achieve score < 2 by
reducing the central convolution peak below sum(a_i^2).

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _geometric_init(n, c=2.0):
    t = np.arange(n) / max(1, n - 1)
    a = np.exp(c * t)
    return a / a.sum()


def _power_init(n, p=2.0):
    t = np.arange(1, n + 1, dtype=np.float64)
    a = t ** p
    return a / a.sum()


def _binom_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _lp_init(n, c_target=2.0):
    """Semi-analytic: geometric shape optimized for given c."""
    return _geometric_init(n, c_target)


def _obj_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _grad_lse(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g

            n_di = hi_i - lo_i + 1
            n_di1 = hi_i1 - lo_i1 + 1
            dir_i[lo_i:hi_i + 1] = 2.0 * a[0:n_di]
            dir_i1[lo_i1:hi_i1 + 1] = 2.0 * a[0:n_di1]

            for xi in candidates_i:
                di = xi - ai
                for xi1 in candidates_i1:
                    di1 = xi1 - ai1
                    c_trial_max = float(np.max(c + di * dir_i + di1 * dir_i1))
                    if c_trial_max < best_f:
                        best_f = c_trial_max
                        best_pair = (xi, xi1)

            if best_pair is not None:
                xi, xi1 = best_pair
                di = xi - ai
                di1 = xi1 - ai1
                c += di * dir_i + di1 * dir_i1
                # Quadratic corrections
                if 2 * i < total_len:
                    c[2 * i] += di * di
                if 2 * (i + 1) < total_len:
                    c[2 * (i + 1)] += di1 * di1
                a[i] = xi
                a[i + 1] = xi1
                any_improve = True

        if sweep % 20 == 0:
            cur = _score_norm(a)
            if cur < best_score:
                best_score = cur
        if not any_improve and sweep > 3:
            break

    final_score = _score_norm(a)
    if final_score < best_score:
        best_score = final_score
    return a, best_score


def _adam_refine(a_init, deadline, lr=0.01, n_iters=None):
    """Adam-based refinement with adaptive log-sum-exp."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.05 * cm0

    if n_iters is None:
        n_iters = 200000

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_every = 20

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-14 * cm, cm0 * 0.05 * (0.98 ** (t // 20)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_every == 0:
            a_cur = _softmax(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _interp_to_n(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Multiple initializations (geometric, power, binomial)
    2. L-BFGS log-sum-exp continuation
    3. Coordinate descent polish (single + paired)
    4. Adam refinement
    5. Final L-BFGS with tiny beta
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_p = _score_norm(parent)
        print(f"Parent: n={n}, norm_score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n = 8192
        parent = _geometric_init(n, 2.0)
        update_best(parent, _score_norm(parent), "geo_default")

    print(f"Starting n={n}, best={best_score:.10f}")

    # === Phase 1: Multiple initializations + quick L-BFGS ===
    print("Phase 1: Initialization + quick optimization")
    p1_deadline = min(deadline, t_start + 300.0)

    inits = {}
    inits['parent'] = parent.copy()
    inits['geo_2'] = _geometric_init(n, 2.0)
    inits['geo_3'] = _geometric_init(n, 3.0)
    inits['geo_4'] = _geometric_init(n, 4.0)
    inits['power_1'] = _power_init(n, 1.0)
    inits['power_2'] = _power_init(n, 2.0)
    inits['power_3'] = _power_init(n, 3.0)
    inits['binom'] = _binom_init(n)

    # Quick score all inits
    init_scores = {}
    for name, a in inits.items():
        sc = _score_norm(a)
        init_scores[name] = sc
        update_best(a, sc, f"init_{name}")

    # Sort by score and optimize top ones
    sorted_inits = sorted(init_scores.items(), key=lambda x: x[1])
    print(f"  Init scores: {[(k, f'{v:.6f}') for k, v in sorted_inits[:4]]}")

    for name, _ in sorted_inits[:5]:
        if time.monotonic() > p1_deadline - 30:
            break
        a_cur = inits[name]
        a_opt, sc_opt = _lbfgs_run(a_cur, min(time.monotonic() + 50, p1_deadline - 5),
                                   maxiter=3000)
        update_best(a_opt, sc_opt, f"lbfgs_{name}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Aggressive L-BFGS on best ===
    print("Phase 2: Aggressive L-BFGS")
    p2_deadline = min(deadline, t_start + 700.0)

    if best_seq is not None:
        a_lb, sc_lb = _lbfgs_run(best_seq, p2_deadline,
                                  beta_list=None, maxiter=10000)
        update_best(a_lb, sc_lb, "lbfgs_aggressive")

        # Try from a couple other inits
        for name in ['geo_3', 'power_2']:
            if time.monotonic() > p2_deadline - 60:
                break
            a_tmp, sc_tmp = _lbfgs_run(inits[name],
                                       min(time.monotonic() + 80, p2_deadline - 10),
                                       maxiter=5000)
            update_best(a_tmp, sc_tmp, f"lbfgs2_{name}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Coordinate descent ===
    print("Phase 3: Coordinate descent")
    p3_deadline = min(deadline, t_start + 1200.0)

    if best_seq is not None:
        a_cd, sc_cd = _coord_descent(best_seq, min(p3_deadline - 60, time.monotonic() + 400),
                                     max_sweeps=300)
        update_best(a_cd, sc_cd, "coord_descent")

        if time.monotonic() < p3_deadline - 30 and best_seq is not None:
            a_cp, sc_cp = _coord_descent_pairs(best_seq,
                                               min(p3_deadline - 30, time.monotonic() + 200),
                                               max_sweeps=200)
            update_best(a_cp, sc_cp, "coord_pairs")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Adam refinement ===
    print("Phase 4: Adam refinement")
    p4_deadline = min(deadline, t_start + 1500.0)

    if best_seq is not None:
        a_ad, sc_ad = _adam_refine(best_seq, p4_deadline, lr=0.01)
        update_best(a_ad, sc_ad, "adam_l001")

        if time.monotonic() < p4_deadline - 30 and best_seq is not None:
            a_ad2, sc_ad2 = _adam_refine(best_seq, p4_deadline, lr=0.003)
            update_best(a_ad2, sc_ad2, "adam_l0003")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Final L-BFGS with tiny beta ===
    print("Phase 5: Final fine-tuning")
    p5_deadline = min(deadline, t_start + 1650.0)

    if best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in
                      [0.001, 0.0005, 0.0002, 0.0001, 0.00005,
                       0.00002, 0.00001, 0.000005]]
        a_f, sc_f = _lbfgs_run(best_seq, p5_deadline,
                                beta_list=tiny_betas, maxiter=15000)
        update_best(a_f, sc_f, "final_lbfgs")

        # Coordinate descent again on the L-BFGS result
        if time.monotonic() < p5_deadline - 10:
            a_cd2, sc_cd2 = _coord_descent(best_seq, p5_deadline, max_sweeps=200)
            update_best(a_cd2, sc_cd2, "final_coord")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Try larger n if time ===
    if time.monotonic() < deadline - 100 and best_seq is not None:
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"Phase 6: Trying n={n_new}")
            p6_deadline = deadline - 10
            a_interp = _interp_to_n(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")
            if time.monotonic() < p6_deadline - 30:
                a_in_opt, sc_in_opt = _lbfgs_run(a_interp, min(time.monotonic() + 120, p6_deadline - 10),
                                                  maxiter=5000)
                update_best(a_in_opt, sc_in_opt, f"lbfgs_n{n_new}")
                if time.monotonic() < p6_deadline - 10:
                    a_in_cd, sc_in_cd = _coord_descent(a_in_opt, p6_deadline, max_sweeps=100)
                    update_best(a_in_cd, sc_in_cd, f"coord_n{n_new}")

    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _geometric_init(512, 2.0)

        sc_final = _score(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _geometric_init(512, 2.0)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _geometric_init(1024, 2.0)
    print(f"FALLBACK: n=1024, score={_score(fb):.10f}")
    return [float(x) for x in fb]
