= a[k-i] + a[k-i] (when 0 ≤ k-i < n)

But wait, when k = 2i, the term j = i appears in BOTH sums:
- First sum: j=i gives a[k-i] = a[i]
- Second sum: k-j=i gives j=k-i=i, gives a[j] = a[i]
So total = 2*a[i]. That's correct - the term a[i]^2 has derivative 2*a[i].

When k ≠ 2i (but k-i is valid):
- First sum: j=i gives a[k-i]
- Second sum: j=k-i gives a[k-i]
These are different terms (a[i]*a[k-i] and a[k-i]*a[i]), so total = 2*a[k-i]. Correct.

So: ∂conv(k)/∂a[i] = 2*a[k-i] (if 0 ≤ k-i < n), else 0.

Therefore, the subgradient of max(conv) at a is:
For K* = argmax(conv):
sg[i] = max_{k in K*} 2*a[k-i] (if we take the "max" subgradient)
Or more properly, any convex combination: sg[i] = sum_{k in K*} λ_k * 2*a[k-i] for λ_k ≥ 0, sum λ_k = 1.

The "mean" subgradient (uniform over K*) or the "max" subgradient (using only the k that gives the max at coordinate i) are both valid.

a[i] ← a[i] - η * sg[i]

"""
Multi-method optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines log-sum-exp L-BFGS continuation, subgradient peak reduction,
and pair-exchange refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _conv(a):
    return _full_convolve(a, a)

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    n = len(a)
    c = _conv(a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm_fn(a / s)

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _conv(a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _conv(a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # h[i] = sum_k w[k] * a[k-i] for valid k-i
    # This is cross-correlation of w and a
    N = _next_pow2(2 * n - 1)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    h_full = np.fft.irfft(W * np.conj(A), N)
    h = h_full[:n]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

def _run_lbfgs(a_init, deadline, rng, n_betas=15, verbose=False):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _conv(a_init)
    max_c0 = float(c0.max())
    
    beta_start = 0.5 * max_c0
    beta_end = max(1e-8 * max_c0, 1e-12)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 3000)
        
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxls': 40, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z = np.clip(result.x, -70, 70)
                z = z - z.mean()
                a_new = _softmax_fn(z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _subgradient_step(a, c, lr, top_k=5):
    """
    Perform a subgradient step to reduce the peak of c = conv(a,a).
    Returns updated a.
    """
    n = len(a)
    # Find top_k peak positions
    peak_indices = np.argsort(c)[::-1][:top_k]
    peak_vals = c[peak_indices]
    
    # Weights proportional to peak values
    w = peak_vals / peak_vals.sum()
    
    # Compute weighted subgradient
    # sg[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    sg = np.zeros(n)
    for idx, pk in enumerate(peak_indices):
        # a[k-i] for i = 0..n-1: valid when 0 <= pk-i < n, i.e., pk-n+1 <= i <= pk
        lo = max(0, pk - n + 1)
        hi = min(n - 1, pk)
        if lo <= hi:
            sg[lo:hi+1] += 2.0 * w[idx] * a[pk - np.arange(lo, hi+1)]
    
    # Center the subgradient (to preserve sum)
    sg_centered = sg - sg.mean()
    
    # Update
    a_new = a - lr * sg_centered
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new

def _subgradient_descent(a_init, deadline, rng, initial_lr=0.01, decay=0.995, top_k=5):
    """Iterative subgradient descent with adaptive step size."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = initial_lr
    
    c = _conv(a)
    
    for it in range(100000):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient step
        a_new = _subgradient_step(a, c, lr, top_k)
        sc = _score_norm_fn(a_new)
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            c = _conv(a)
        else:
            # Step was bad, reduce learning rate
            lr *= 0.5
        
        # Periodically restore best and try different lr
        if it % 500 == 0:
            a = best_a.copy()
            c = _conv(a)
            lr = initial_lr * (decay ** (it // 500))
        
        # Check for stagnation
        if it % 1000 == 0 and it > 0:
            # Add small perturbation
            z = np.log(np.maximum(a, 1e-15))
            z += 0.01 * rng.normal(size=n)
            z = z - z.mean()
            a = _softmax_fn(z)
            c = _conv(a)
    
    return best_a, best_score

def _pair_exchange_refine(a_init, deadline, rng, max_rounds=5000):
    """
    Pair-exchange refinement: transfer small amounts of mass between pairs
    of indices to reduce the convolution peak.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    c = _conv(a)
    cm = float(c.max())
    
    # Precompute: for each index i, which peak positions does it contribute to?
    # Peak positions
    peak_thresh = cm * 0.95
    peak_mask = c >= peak_thresh
    peak_indices = np.where(peak_mask)[0]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Find the current peak
        pk = int(np.argmax(c))
        c_pk = float(c[pk])
        
        # Find donor: index that contributes most to the peak
        # Contribution of a[i] to c[pk] is a[i] * a[pk-i] (times 2 if i ≠ pk-i)
        valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
        if len(valid_i) == 0:
            break
        contributions = a[valid_i] * a[pk - valid_i]
        # Weight by whether it's a "double" term
        double_mask = (valid_i == pk - valid_i)
        effective_contrib = contributions.copy()
        effective_contrib[~double_mask] *= 2.0
        
        donor_local = int(np.argmax(effective_contrib))
        donor = int(valid_i[donor_local])
        
        # Find recipient: index that contributes least to peaks
        # Score each index by its total contribution to top peaks
        if len(peak_indices) > 0:
            peak_list = peak_indices[:10]  # top 10 peaks
            contrib_to_peaks = np.zeros(n)
            for p in peak_list:
                vi = np.arange(max(0, p - n + 1), min(n, p + 1))
                if len(vi) > 0:
                    contrib_to_peaks[vi] += a[vi] * a[p - vi]
            
            # Recipient should have low contribution to peaks and be far from peak positions
            # Try indices with low contribution
            candidate_indices = np.argsort(contrib_to_peaks)[:20]
        else:
            candidate_indices = np.argsort(a)[:20]
        
        # Try exchanging: move small delta from donor to recipient
        improved = False
        for recipient in candidate_indices:
            if recipient == donor:
                continue
            
            for delta in [0.001, 0.005, 0.01, 0.02]:
                if a
