def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    out_n = n + m - 1

def _build_lp_matrix(n):
    """Build sparse matrix for LP constraints."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _self_convolve(a):
    n = len(a)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _score_norm(a):
    n = len(a)
    c = _self_convolve(a)
    return float(2.0 * n * c.max())


def _score(a):

    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _self_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _build_lp_matrix(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=45):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _golden_line_search(f, g, max_evals=15):
    """Golden section search for optimal blend t in (0, 1)."""
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - (b - a) / gr
    d = a + (b - a) / gr
    fc = _score_norm((1.0 - c) * f + c * g)
    fd = _score_norm((1.0 - d) * f + d * g)
    evals = 2
    for _ in range(max_evals - 2):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - (b - a) / gr
            fc = _score_norm((1.0 - c) * f + c * g)
        else:
            a = c
            c = d
            fc = fd
            d = a + (b - a) / gr
            fd = _score_norm((1.0 - d) * f + d * g)
        evals += 1
        if b - a < 0.01:
            break
    t = (a + b) / 2.0
    return t


def _lp_step_adaptive(f, rows, cols, time_limit=45):
    """LP step with adaptive M reduction and golden-section line search."""
    n = len(f)
    c = _self_convolve(f)
    M = float(c.max())
    
    # Try to reduce M to get a tighter LP
    M_used = M
    g_0 = None
    for factor in [0.995, 0.99, 0.98, 0.96]:
        M_try = M * factor
        g_0 = _solve_lp(f, M_try, rows, cols, time_limit)
        if g_0 is not None and g_0.sum() > 1e-15:
            M_used = M_try
            break
    
    if g_0 is None or g_0.sum() < 1e-15:
        g_0 = _solve_lp(f, M, rows, cols, time_limit)
        if g_0 is None or g_0.sum() < 1e-15:
            return f, M
    
    g = g_0 / g_0.sum()
    
    # Golden section line search
    best_sc = _score_norm(f)
    t_opt = _golden_line_search(f, g, max_evals=12)
    
    f_new = (1.0 - t_opt) * f + t_opt * g
    sc_new = _score_norm(f_new)
    
    if sc_new < best_sc:
        return f_new, M
    else:
        # Try smaller steps
        for t in [0.05, 0.1, 0.15, 0.2]:
            f_try = (1.0 - t) * f + t * g
            sc_try = _score_norm(f_try)
            if sc_try < best_sc:
                return f_try, M
        return f, M


def _subgrad_step(a, k_star, eta, momentum, velocity, t_iter):
    """One sub-gradient step with Nesterov momentum."""
    n = len(a)
    # Sub-gradient: d[i] = a[k_star - i] if 0 <= k_star - i < n, else 0
    d = np.zeros(n)
    valid = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
    if len(valid) > 0:
        d[valid] = a[k_star - valid]
    
    # Nesterov: look-ahead
    a_look = a + momentum * velocity
    # Recompute d with look-ahead
    d2 = np.zeros(n)
    valid2 = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
    if len(valid2) > 0:
        d2[valid2] = a_look[k_star - valid2]
    
    # Step
    a_new = a_look - eta * d2
    a_new = _project_simplex(a_new)
    
    # Update velocity
    velocity = momentum * velocity + (a - a_new)
    
    return a_new, velocity


def _subgrad_phase(a, deadline, max_iter=500000):
    """Sub-gradient optimization with adaptive learning rate."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    velocity = np.zeros(n)
    momentum = 0.9
    t_iter = 0
    prev_sc = best_sc
    stall = 0
    
    while t_iter < max_iter:
        if time.monotonic() > deadline:
            break
        t_iter += 1
        
        c = _self_convolve(a)
        c_max = float(c.max())
        
        # Learning rate schedule
        eta = 0.5 / (1.0 + 0.01 * t_iter)
        
        # Find all near-max positions for smoother gradient
        threshold = c_max * 0.999
        near_max = np.where(c >= threshold)[0]
        
        if len(near_max) == 1:
            a, velocity = _subgrad_step(a, near_max[0], eta, momentum, velocity, t_iter)
        else:
            # Average sub-gradient over near-max positions
            d = np.zeros(n)
            for k_star in near_max[:5]:  # Limit to avoid too much averaging
                valid = np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)
                if len(valid) > 0:
                    d[valid] += a[k_star - valid]
            d /= min(len(near_max), 5)
            a_new = a - eta * d
            a_new = _project_simplex(a_new)
            velocity = momentum * velocity + (a - a_new)
            a = a_new
        
        # Check improvement periodically
        if t_iter % 200 == 0:
            sc = _score_norm(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
            if stall > 20:
                momentum = max(0.5, momentum * 0.95)
                stall = 0
    
    # Final check
    sc = _score_norm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc, t_iter


def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=500000, lr=0.01):
    """Adam optimizer with log-sum-exp smoothing of the max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        beta = beta_start * (max(beta_end, 1e-15) / beta_start) ** (t / max_iter)
        
        c = _self_convolve(a)
        c_max = float(c.max())
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient computation
        a_rev = a[::-1]
        conv_aw_rev = _full_convolve(a_rev, w)
        grad = conv_aw_rev[n - 1: 2 * n - 1]
        
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * grad_z * grad_z
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t % 500 == 0:
            a_check = _softmax(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > 15:
                current_lr = max(0.001, current_lr * 0.9)
                patience = 0
            elif best_score < prev_best * 0.999:
                current_lr = min(0.05, current_lr * 1.05)
                prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _interp_profile(p, n_new):
    """Interpolate a normalized profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()
