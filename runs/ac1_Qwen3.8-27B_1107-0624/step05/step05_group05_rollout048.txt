"""
Hybrid LP + Adaptive Smooth Gradient optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:
1. Better beta annealing schedule for log-sum-exp smoothing
2. Nesterov acceleration in the smooth phase
3. Active-set focusing for near-convergent iterations
4. More aggressive multi-restart with diverse perturbations
5. Adaptive learning rate with backtracking
6. Symmetry exploitation option for reduced search space

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, best_sc


def _smooth_grad_logdomain(a, beta):
    """Compute gradient of log-sum-exp(beta * conv(a,a)) in log-domain (natural gradient)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / max(beta, 1e-15)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        s = 1.0
    else:
        w = e / s
    # grad_a_j = 2 * sum_m w_{j+m} * a_m / s_total
    # This is 2 * cross_correlation(w, a) / s
    # cross_correlation(w, a)_j = sum_m w_{j+m} * a_m
    # = convolution of reversed w with a, shifted
    w_rev = w[::-1]
    conv_wa = _full_convolve(w_rev, a)
    # (w_rev * a)_k = sum_i w_rev[i] * a[k-i] = sum_i w[2n-2-i] * a[k-i]
    # Let m = 2n-2-i, then i = 2n-2-m:
    # = sum_m w[m] * a[k-2n+2+m]
    # We want cross_corr_j = sum_m w[j+m] * a[m]
    # Set k - 2n + 2 = m => k = m + 2n - 2 - m = 2n-2... hmm
    # Actually: cross_corr_j = sum_{m=0}^{n-1} w_{j+m} * a_m
    # conv_wa_k = sum_i w_rev[i] * a[k-i] = sum_i w[2n-2-i] * a[k-i]
    # Let p = 2n-2-i: = sum_p w[p] * a[k-p-2n+2+p] = sum_p w[p] * a[k-2n+2]
    # That's not right. Let me just compute it directly using the relation:
    # cross_corr(w, a)_j = (w * a_rev)_j + (2n-2-n+1) ... 
    # Simpler: just use the fact that cross_corr(w,a) = conv(w_rev_padded, a) with appropriate indexing.

    # cross_corr_j = sum_{m=0}^{n-1} w_{j+m} * a_m where w has length 2n-1
    # This equals: the (j)-th element of the "anti-convolution"

    # cross_corr_j = sum_m w_{j+m} * a_m = sum_m w_{j+m} * a_m

    # (w_flipped * a)_k = sum_i w_flipped[i] * a[k-i] = sum_i w[2n-2-i] * a[k-i]

    N_w = len(c)  # 2n-1

    # Normalize

                 nesterov=True):

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    # Nesterov momentum
    a_prev = a_init.copy()
    z_prev = z.copy()
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Nesterov extrapolation
        if nesterov and t > 1:
            momentum = (t - 1) / (t + 2.0)
            z = z + momentum * (z - z_prev)
        
        z_prev = z.copy()
        a = _softmax(z)
        
        # Compute gradient
        beta_frac = t / max_iter
        beta = beta_start * (beta_end / beta_start) ** beta_frac if beta_start > 0 else beta_end
        beta = max(beta, 1e-10)
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Cross-correlation w with a
        N_fft = _next_pow2(len(c) + n - 1)
        W = np.fft.rfft(w, N_fft)
        A = np.fft.rfft(a, N_fft)
        cc = np.fft.irfft(np.conj(W) * A, N_fft)[:n]
        grad_a = 2.0 * cc / max(s, 1e-300)
        
        W_dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W_dot)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        max_step = 0.3
        if update_norm > max_step:
            update = update * (max_step / update_norm)
        
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if t % 1000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        # Adaptive learning rate
        if t % 5000 == 0:
            if best_score < prev_best * 0.9995:
                current_lr = min(0.05, current_lr * 1.08)
            elif best_score > prev_best:
                current_lr = max(0.0005, current_lr * 0.85)
            else:
                current_lr = max(0.0005, current_lr * 0.97)
            prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _active_set_smooth(a_init, deadline, top_k=10, lr=0.005, max_iter=2000000):
    """Optimize using only the top-k active convolution positions."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Find top-k active positions
        if top_k < len(c):
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            c_active = c[top_idx]
        else:
            c_active = c
            top_idx = np.arange(len(c))
        
        # Smooth max of active positions with very sharp beta
        beta = 50.0
        c_sh = (c_active - c_max) / beta
        e = np.exp(c_sh)
        s = e.sum()
        w_active = e / s
        
        # Build full w from active
        w = np.zeros(len(c))
        w[top_idx] = w_active
        
        # Cross-correlation
        N_fft = _next_pow2(len(c) + n - 1)
        W = np.fft.rfft(w, N_fft)
        A = np.fft.rfft(a, N_fft)
        cc = np.fft.irfft(np.conj(W) * A, N_fft)[:n]
        grad_a = 2.0 * cc / max(s, 1e-300)
        
        W_dot = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W_dot)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > 0.1:
            update = update * (0.1 / update_norm)
        
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if t % 2000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if t % 10000 == 0:
            if best_score < prev_best * 0.9999:
                current_lr = min(0.02, current_lr * 1.05)
            elif best_score > prev_best * 1.0001:
                current_lr = max(0.0001, current_lr * 0.7)
            prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _perturb_structured(a, rng):
    """Structured perturbation: add a smooth random function."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    # Add a few random sinusoids
    for _ in range(3):
        freq = rng.integers(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        x = np.arange(n)
        z += amp * np.sin(2 * np.pi * freq * x / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _quick_lp_refine(a, rows, cols, max_iters=15):
    """Quick LP refinement from current point."""
    for _ in range(max_iters):
        a, sc = _lp_step(a, rows, cols, time_limit=15)
        if time.monotonic() > 0:  # placeholder, actual time check in caller

    return a


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    # Build LP indices
    rows, cols = _build_lp_indices(n)
    print(f"LP indices built: {len(rows)} nonzeros, t={time.monotonic()-t_start:.1f}s")

    # === Phase 1: LP iterations (0 - 200s) ===
    phase1_end = min(deadline, t_start + 200.0)
    lp_count = 0
    prev_sc = _score_norm(a)
    stall = 0

    while time.monotonic() < phase1_end:
        t0 = time.monotonic()
        a, sc = _lp_step(a, rows, cols, time_limit=20)
        lp_count += 1

        if sc < prev_sc - 1e-12:
            stall = 0
        else:
            stall += 1

        if lp_count % 5 == 0 or stall > 10:
            full_sc = _score(a)
            if lp_count % 10 == 0 or stall > 10:
                print(f"  LP[{lp_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a.copy()

        if stall > 30:
            print(f"  LP stalled at iter {lp_count}")
            break
        prev_sc = sc

    full_sc = _score(a)
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 2: Smooth gradient with aggressive beta annealing (200 - 600s) ===
    phase2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < phase2_end:
        a_work = best_seq.copy()
        c_init = _full_convolve(a_work, a_work)
        c_max_init = float(c_init.max())
        
        # Multiple annealing stages
        stages = [
            (0.05 * c_max_init, 0.01 * c_max_init, 0.012),
            (0.01 * c_max_init, 0.002 * c_max_init, 0.008),
            (0.002 * c_max_init, 0.0005 * c_max_init, 0.005),
            (0.0005 * c_max_init, 0.0001 * c_max_init, 0.003),
            (0.0001 * c_max_init, 0.00002 * c_max_init, 0.002),
        ]
        
        for si, (bs, be, lr_s) in enumerate(stages):
            if time.monotonic() > phase2_end - 10:
                break
            stage_deadline = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) / (len(stages) - si))
            a_polished, sc_polished = _adam_smooth(
                a_work, stage_deadline,
                beta_start=bs, beta_end=be,
                max_iter=5000000, lr=lr_s,
                nesterov=True
            )
            full_sc = _score(a_polished)
            print(f"  Smooth stage {si}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a_polished.copy()
            a_work = a_polished

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: LP refinement from best (600 - 900s) ===
    phase3_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase3_end:
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp3_count = 0
        while time.monotonic() < phase3_end:
            a, sc = _lp_step(a, rows, cols, time_limit=20)
            lp3_count += 1
            if sc < prev_sc - 1e-12:
                stall = 0
            else:
                stall += 1
            if lp3_count % 5 == 0:
                full_sc = _score(a)
                print(f"  LP2[{lp3_count}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:

                    best_seq = a.copy()
            if stall > 25:
                break
            prev_sc = sc
        print(f"Phase 3 done: {lp3_count} LP iters, best={best_score:.10f}")

    # === Phase 4: Active-set focused optimization (900 - 1300s) ===
    phase4_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase4_end:
        a_work = best_seq.copy()

            a_work = a_polished

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 5: Multi-restart with diverse perturbations (1300 - 1650s) ===
    phase5_end = min(deadline, t_start + 1650.0)

            # Quick smooth (remaining time)

            p_polished, sc_pp = _adam_smooth(

    if time.monotonic() < deadline - 10:

    return [float(x) for x in fallback]
