def _next_pow2(n):

def _next_pow2(n):

def _full_convolve(a, b):

    p_schedule = [2, 4, 8, 16, 32, 64, 128, 256, 512]

        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        est = 0.004 * (n / 2048.0)
        max_iter = min(3000, max(100, int(remaining / (est * 5))))
        a_new = _lbtfgs_power(current, p, deadline, max_iter)
        sc_new = _score(a_new)
        sc_cur = _score(current)
        if sc_new < sc_cur + 1e-12:
            current = a_new
    return current

def _run_ls_schedule(a_start, deadline, min_remaining=60.0):
    """Continuation with log-sum-exp beta."""
    n = len(a_start)
    c = _full_convolve(a_start, a_start)
    mc = float(c.max())
    if mc < 1e-15:
        return a_start
    current = a_start.copy()
    betas = mc * np.array([0.2, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005])
    for beta in betas:
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break
        max_iter = min(3000, max(100, int(remaining / 0.03)))
        a_new = _lbtfgs_ls(current, beta, deadline, max_iter)
        sc_new = _score(a_new)
        sc_cur = _score(current)
        if sc_new < sc_cur + 1e-12:
            current = a_new
    return current

def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining Lp-continuation, log-sum-exp continuation,
    subgradient polishing, symmetry enforcement, and multi-restart diversification.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.1f}s")
    
    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw / parent_raw.sum()
    except Exception:
        pass
    
    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        print(f"Parent: n={n_main}, score={_score(parent):.10f}")
        update_best(parent, _score(parent), "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"Init uniform n={n_main}, score={_score(a0):.10f}")
        update_best(a0, _score(a0), "init")
    
    # Phase 1: Power-continuation optimization
    print(f"\nPhase 1: Power continuation (t={time.monotonic()-t_start:.1f}s)")
    a_pw = _run_power_schedule(a0, deadline, min_remaining=400.0)
    update_best(a_pw, _score(a_pw), "power_cont")
    
    # Phase 2: Log-sum-exp continuation from best so far
    print(f"\nPhase 2: LS continuation (t={time.monotonic()-t_start:.1f}s)")
    a_ls = _run_ls_schedule(best_seq, deadline, min_remaining=300.0)
    update_best(a_ls, _score(a_ls), "ls_cont")
    
    # Phase 3: Subgradient polishing
    print(f"\nPhase 3: Subgradient polish (t={time.monotonic()-t_start:.1f}s)")
    sg_dl = min(deadline - 200.0, time.monotonic() + 150.0)
    a_sg, sc_sg = _subgrad_polish(best_seq, sg_dl, max_steps=5000, lr_init=0.04)
    update_best(a_sg, sc_sg, "subgrad")
    
    # Phase 4: Symmetry enforcement + re-optimize
    print(f"\nPhase 4: Symmetry (t={time.monotonic()-t_start:.1f}s)")
    a_sym = _symmetrize(best_seq)
    sc_sym = _score(a_sym)
    if sc_sym < best_score + 0.001:
        # Re-optimize the symmetric version
        if time.monotonic() < deadline - 100:
            a_sym2 = _lbtfgs_power(a_sym, 256, deadline - 100, max_iter=2000)
            sc_sym2 = _score(a_sym2)
            if sc_sym2 < best_score:
                a_sym = a_sym2
            update_best(a_sym, _score(a_sym), "symmetric")
        # Subgradient on symmetric
        if time.monotonic() < deadline - 50:
            a_sym3, sc_sym3 = _subgrad_polish(a_sym, deadline - 50, max_steps=2000, lr_init=0.02)
            update_best(a_sym3, sc_sym3, "symmetric_sg")
    
    # Phase 5: Fine L-BFGS with alternating methods
    print(f"\nPhase 5: Fine optimization (t={time.monotonic()-t_start:.1f}s)")
    for method_i in range(3):
        if time.monotonic() > deadline - 30:
            break
        if method_i == 0:
            # Power with very high p
            a_fin = _lbtfgs_power(best_seq, 512, deadline - 30, max_iter=2000)
        elif method_i == 1:
            # LS with tiny beta
            c_b = _full_convolve(best_seq, best_seq)
            mb = float(c_b.max())
            a_fin = _lbtfgs_ls(best_seq, mb * 0.0002, deadline - 30, max_iter=2000)
        else:
            # Subgradient with small lr
            a_fin, _ = _subgrad_polish(best_seq, deadline - 30, max_steps=3000, lr_init=0.01)
        update_best(a_fin, _score(a_fin), f"fine_{method_i}")
    
    # Phase 6: Multi-restart with diverse perturbations
    print(f"\nPhase 6: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        tries = min(4, max(1, int((deadline - 40 - time.monotonic()) / 60)))
        for i in range(tries):
            if time.monotonic() > deadline - 35:
                break
            if i == 0:
                p_start = _perturb_structured(best_seq, rng, n_modes=2)
            elif i == 1:
                p_start = _perturb_structured(best_seq, rng, n_modes=5)
            elif i == 2:
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + 0.02 * rng.standard_normal(len(z))
                p_start = _softmax(z)
            else:
                # Try a different profile blended with best
                alt = _make_profile(len(best_seq), "raised_cosine")
                p_start = 0.8 * best_seq + 0.2 * alt
                p_start = p_start / p_start.sum()
            
            sc_start = _score(p_start)
            # Quick optimization
            a_r = _lbtfgs_power(p_start, 64, deadline - 35, max_iter=1000)
            sc_r = _score(a_r)
            if sc_r < best_score + 0.01:
                a_r2, _ = _subgrad_polish(a_r, deadline - 35, max_steps=2000, lr_init=0.02)
                update_best(a_r2, _score(a_r2), f"restart_{i}")
            else:
                update_best(a_r, sc_r, f"restart_{i}")
    
    # Phase 7: Final polish
    print(f"\nPhase 7: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _subgrad_polish(best_seq, deadline - 3, max_steps=3000, lr_init=0.005)
        update_best(a_fin, sc_fin, "final_polish")
    
    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
