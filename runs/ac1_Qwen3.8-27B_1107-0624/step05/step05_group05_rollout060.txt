"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2:
- Improved LP iterations with golden section line search
- Frequency-domain QP relaxation for diverse starting points
- Smoothed subgradient (mirror descent) with adaptive parameters
- Perturbation restarts to escape local minima

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

    import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit,
                               'primal_feasibility_tolerance': 1e-9,
                               'dual_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None
        return np.maximum(res.x, 0.0), res
    except Exception:
        return None, None


def _golden_line_search(f, g, best_sc_norm, max_iter=60):
    """Golden section search on t in [0, 1] for (1-t)*f + t*g."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _score_norm((1.0 - c) * f + c * g)
    fd = _score_norm((1.0 - d) * f + d * g)
    for _ in range(max_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = _score_norm((1.0 - c) * f + c * g)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = _score_norm((1.0 - d) * f + d * g)
        if b - a < 1e-6:
            break
    t_opt = (a + b) / 2.0
    sc_opt = _score_norm((1.0 - t_opt) * f + t_opt * g)
    if sc_opt >= best_sc_norm:
        return 0.0, best_sc_norm
    return t_opt, sc_opt


def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0, lp_res = _solve_lp(f, M, rows, cols, time_limit=time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M, None
    g = g_0 / g_0.sum()
    best_sc = _score_norm(f)
    t_opt, sc_after = _golden_line_search(f, g, best_sc)
    if t_opt > 0.0:
        f = (1.0 - t_opt) * f + t_opt * g
    return f, M, t_opt


def _mwu_step(a, beta, eta):
    n = len(a)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / max(beta, 1e-10)
    w = np.exp(c_shifted)
    w /= w.sum()
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    z = np.log(np.maximum(a, 1e-300)) - eta * grad
    z -= z.mean()
    return _softmax(z)


def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=2000000, lr=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        if t % 20000 == 0:
            if best_score < prev_best * 0.9995:
                current_lr = min(0.05, current_lr * 1.15)
            else:
                current_lr = max(0.001, current_lr * 0.92)
            prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _freq_domain_qp(a_init, deadline, n_freq=None):
    """Solve the frequency-domain convex relaxation:
    min t s.t. |DFT(a)_k|^2 <= t for sampled k, sum(a)=1, a>=0
    """
    if not HAS_CVXPY:
        return None
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    if n_freq is None:
        n_freq = min(N, max(512, 2 * n))
    # Sample n_freq frequencies evenly
    freq_idx = np.linspace(0, N - 1, n_freq, dtype=int)
    j = np.arange(n, dtype=np.float64)
    k = freq_idx.astype(np.float64)
    F_real = np.cos(2.0 * np.pi * np.outer(k, j) / N)
    F_imag = -np.sin(2.0 * np.pi * np.outer(k, j) / N)

    print(f"  QP: n={n}, N={N}, n_freq={n_freq}, matrix shape={F_real.shape}")
    t0 = time.monotonic()

    a_var = cp.Variable(n)
    t_var = cp.Variable()
    A_real = F_real @ a_var
    A_imag = F_imag @ a_var
    constraints = [
        cp.sum(a_var) == 1,
        a_var >= 0,
        t_var >= 0
    ]
    # Add quadratic constraints: A_real[k]^2 + A_imag[k]^2 <= t
    constraints.append(A_real**2 + A_imag**2 <= t_var)

    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)

    try:
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    break
            except Exception:
                continue
    except Exception as e:
        print(f"  QP failed: {e}")
        return None

    elapsed = time.monotonic() - t0
    print(f"  QP status={prob.status}, t={t_var.value:.8f}, time={elapsed:.1f}s")

    if prob.status in ['optimal', 'optimal_inaccurate'] and a_var.value is not None:
        a_sol = np.maximum(a_var.value, 0.0)
        a_sol = a_sol / a_sol.sum()
        sc = _score(a_sol)
        print(f"  QP solution score: {sc:.10f}")
        if time.monotonic() < deadline:
            return a_sol
    return None


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))

def _dual_informed_step(f, rows, cols, M, lp_res):
    """Use LP dual variables to construct a weighted subgradient step."""
    n = len(f)
    if lp_res is None or lp_res.ineqlin is None:
        return None
    duals = np.abs(lp_res.ineqlin)
    duals_norm = duals / (duals.sum() + 1e-30)
    # Build weighted gradient: sum_k duals[k] * (T_k f)
    c = _full_convolve(f, f)
    # Weighted "importance" of each lag
    w_lag = duals_norm
    w_rev = w_lag[::-1]
    conv_fw = _full_convolve(f, w_rev)
    grad = 2.0 * conv_fw[n - 1:2 * n - 1][::-1]
    # Project onto sum=0 subspace
    grad -= grad.mean()
    # Take small step in -grad direction (in log space)
    step_size = 0.02
    z = np.log(np.maximum(f, 1e-15))
    z -= step_size * grad
    z -= z.mean()

            eta = 0.05 if bi < 2 else 0.02
            for it in range(200):
                if time.monotonic() > phase2_end:
                    break
                a = _mwu_step(a, beta, eta)
            sc = _score(a)
            print(f"  MWU beta={beta:.6e}: score={sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a.copy()

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: More LP from best (550 - 950s)
    phase3_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase3_end:
        print("\n=== Phase 3: Continued LP ===")
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        lp3_count = 0
        while time.monotonic() < phase3_end:
            a, M, t_opt = _lp_step(a, rows, cols, time_limit=25)
            sc = _score_norm(a)
            lp3_count += 1
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if lp3_count % 5 == 0:
                full_sc = _score(a)
                print(f"  LP2[{lp3_count}] score={full_sc:.10f} stall={stall} "
                      f"t={time.monotonic()-t_start:.0f}s")
                if np.isfinite(full_sc) and full_sc < best_score:
                    best_score = full_sc
                    best_seq = a.copy()
            if stall > 35:
                # Try perturbation
                a_pert = _perturb(a, 0.015, rng)
                a_pert, _, _ = _lp_step(a_pert, rows, cols, time_limit=20)
                sc_pert = _score_norm(a_pert)
                if sc_pert < prev_sc:
                    a = a_pert
                    stall = 0
                else:
                    break
            prev_sc = sc
        print(f"Phase 3 done: {lp3_count} LP iters, best={best_score:.10f}")

    # Phase 4: Adam smoothing (950 - 1350s)
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end:
        print("\n=== Phase 4: Adam smoothing ===")
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        deadline4 = phase4_end - 10
        a_polished, sc_polished = _adam_smooth(
            best_seq, deadline4,
            beta_start=0.0005 * max_cb, beta_end=0.00001 * max_cb,
            max_iter=2000000, lr=0.005
        )
        full_sc = _score(a_polished)
        print(f"Phase 4 Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_polished.copy()
        else:
            print(f"  Adam didn't improve (best was {best_score:.10f})")

    # Phase 5: Second Adam pass with different params (1350 - 1550s)
    phase5_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < phase5_end:
        print("\n=== Phase 5: Second Adam pass ===")
        c_best2 = _full_convolve(best_seq, best_seq)
        max_cb2 = float(c_best2.max())
        deadline5 = phase5_end - 10
        a_p2, sc_p2 = _adam_smooth(
            best_seq, deadline5,
            beta_start=0.0002 * max_cb2, beta_end=0.000005 * max_cb2,
            max_iter=1500000, lr=0.003
        )
        full_sc2 = _score(a_p2)
        print(f"Phase 5 Adam2: score={full_sc2:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc2) and full_sc2 < best_score:
            best_score = full_sc2
            best_seq = a_p2.copy()

    # Phase 6: Perturbation + short LP (1550 - 1700s)
    phase6_end = min(deadline, t_start + 1700.0)
    if time.monotonic() < phase6_end:
        print("\n=== Phase 6: Perturbation restarts ===")
        for attempt in range(5):
            if time.monotonic() > phase6_end - 30:
                break
            sigma = 0.02 + 0.01 * attempt
            p_start = _perturb(best_seq, sigma, rng)
            prev_sc = _score_norm(p_start)
            stall = 0
            for it in range(15):
                if time.monotonic() > phase6_end - 10:
                    break
                p_start, _, t_opt = _lp_step(p_start, rows, cols, time_limit=20)
                sc = _score_norm(p_start)
                if sc < prev_sc - 1e-10:
                    stall = 0
                else:
                    stall += 1
                if stall > 8:
                    break
                prev_sc = sc
            full_sc = _score(p_start)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc:.10f} "
                  f"t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_start.copy()

    # Phase 7: Final LP polish (1700 - 1780s)
    phase7_end = min(deadline, t_start + 1780.0)
    if time.monotonic() < phase7_end:
        print("\n=== Phase 7: Final LP polish ===")
        a = best_seq.copy()
        prev_sc = _score_norm(a)
        stall = 0
        for it in range(20):
            if time.monotonic() > phase7_end - 5:
                break
            a, M, t_opt = _lp_step(a, rows, cols, time_limit=20)
            sc = _score_norm(a)
            if sc < prev_sc - 1e-10:
                stall = 0
            else:
                stall += 1
            if stall > 10:
                break
            prev_sc = sc
        full_sc = _score(a)
        print(f"Phase 7: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
