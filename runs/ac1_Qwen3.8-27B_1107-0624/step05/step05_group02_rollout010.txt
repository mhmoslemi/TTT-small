```python
"""
Projected gradient with Nesterov momentum + L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses hard/near-hard peak targeting with efficient FFT-based gradient computation.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(x):
    """Project x onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(x)
    if x.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al. 2008
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(x - theta, 0)
    s = out.sum()
    if s > 0:
        out = out / s  # Safety renormalization
    else:
        out = np.ones(n) / n
    return out


def _compute_grad(a, w):
    """
    Compute grad[i] = 2 * sum_j a[j] * w[i+j]
    using the formula: grad = 2 * convolve(reverse(a), w)[n-1 : 2n-1]
    """
    n = len(a)
    b = a[::-1]
    # convolve(b, w) has length n + len(w) - 1 = n + 2n - 1 - 1 = 3n - 2
    bw = _full_convolve(b, w)
    return 2.0 * bw[n - 1:2 * n - 1]

    subject to a >= 0, sum(a) = 1.
    """

    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)
    lr = initial_lr
    
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_score = float(c.max())
    
    beta_scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    
    for beta_scale in beta_scales:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * best_score
        iters = 0
        max_iters = 50000
        while iters < max_iters:
            if time.monotonic() > deadline:
                break
            iters += 1
            
            # Nesterov lookahead
            y = a + momentum * v * (-1)  # y = a - momentum * v (v is the "step")
            # Actually, standard Nesterov: y_{k+1} = x_k + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})
            # Simpler version (heavy-ball): y = a + momentum * (a - a_prev)
            # Let me use the simpler heavy-ball momentum:
            # v = momentum * v + grad
            # a_new = a - lr * v
            # This is more like "heavy ball" than true Nesterov, but works well in practice.
            
            # Evaluate at y
            c_y = _full_convolve(y, y)
            M_y = float(c_y.max())
            
            # Compute weights
            c_shifted = (c_y - M_y) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300 or not np.isfinite(s_e):
                w = np.zeros_like(c_y)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = e / s_e
            
            # Compute gradient at y
            grad = _compute_grad(y, w)
            
            if not np.all(np.isfinite(grad)):
                break
            
            # Momentum update
            v = momentum * v + grad
            v_norm = np.linalg.norm(v)
            if v_norm > 1.0:

            # Step
            a_new = a - lr * v
            a = _project_simplex(a_new)
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` over non-negative sequences. The previous approach achieved 1.5059 using LP + Adam. I'll take a different approach using:

1. **Heavy-ball momentum** with hard/near-hard peak targeting (simpler than Adam, potentially escapes different local minima)
2. **L-BFGS** with log-sum-exp smoothing for fine polishing (better line search than Adam)
3. **Multi-n exploration** including n=16384
4. **Sharp temperature schedules** to target peaks more aggressively

Key insight: The gradient of the smoothed max is `4*n*a*(h - W)` where `h[j] = sum_k w[k]*a[k-j]` and `W = dot(w, conv)`. I'll compute this efficiently via FFT and use both momentum-based and quasi-Newton methods.

```python
"""
Hybrid optimization: heavy-ball momentum + L-BFGS polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses FFT-based gradient computation, simplex
projection, sharp peak targeting, and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _project_simplex(x):
    """Project x onto {a >= 0, sum(a) = 1}."""
    n = len(x)
    if n == 0:
        return x
    if x.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    rho_mask = u * idx > (cssv - 1)
    if not np.any(rho_mask):
        return np.ones(n) / n
    rho = np.nonzero(rho_mask)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    out = np.maximum(x - theta, 0)
    s = out.sum()
    if s > 1e-15:
        out = out / s
    else:
        out = np.ones(n) / n
    return out


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z), z
    return e / s, z


def _compute_h_and_W(a, w):
    """
    Compute h[j] = sum_k w[k]*a[k-j] (valid k-j) and W = dot(w, conv(a,a)).
    h is the cross-correlation of w and a, shifted appropriately.
    """
    n = len(a)
    # h[j] = sum_{k: 0<=k-j<2n-1} w[k] * a[k-j]
    # Let m = k-j, so k = j+m: h[j] = sum_m w[j+m] * a[m]
    # This is the cross-correlation: h = cross_corr(w, a)
    # Using convolution: h[j] = (a_rev * w)[n-1+j] where a_rev = reverse(a)
    a_rev = a[::-1]
    aw = _full_convolve(a_rev, w)
    h = aw[n - 1:2 * n - 1]
    # W = sum_k w[k] * conv(a,a)[k]
    c = _full_convolve(a, a)
    W = float(np.dot(w, c))
    return h, W


def _grad_softmax(z, beta, n):
    """Gradient of beta*log(sum(exp(conv(softmax(z),softmax(z))/beta))) w.r.t. z."""
    a, _ = _softmax_fn(z)
    c = _full_convolve(a, a)
    M = float(c.max())
    c_shifted = (c - M) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    h, W = _compute_h_and_W(a, w)
    grad = 4.0 * n * a * (h - W)
    return grad, a


def _value_softmax(z, beta):
    """Value of the smoothed objective."""
    a, _ = _softmax_fn(z)
    c = _full_convolve(a, a)
    M = float(c.max())
    c_shifted = (c - M) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return M
    return M + beta * np.log(s)


def _heavy_ball(a_init, deadline, lr0=0.3, momentum=0.85, beta_scale0=0.01):
    """Heavy-ball momentum method on the simplex."""
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_cmax = float(c.max())
    lr = lr0
    beta = beta_scale0 * best_cmax

    iters = 0
    while time.monotonic() < deadline:
        iters += 1

        # Compute weights
        c_a = _full_convolve(a, a)
        M = float(c_a.max())
        c_shifted = (c_a - M) / beta
        e = np.exp(c_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(c_a)
            w[int(np.argmax(c_a))] = 1.0
        else:
            w = e / s_e

        # Compute gradient
        h, W = _compute_h_and_W(a, w)
        grad = 2.0 * h  # gradient of weighted sum, not the full softmax grad

        if not np.all(np.isfinite(grad)):
            break

        # Momentum
        v = momentum * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > 2.0:
            v = v * (2.0 / v_norm)

        # Step and project
        a_new = a - lr * v
        a = _project_simplex(a_new)

        # Decay
        lr = lr0 / (1.0 + 0.005 * iters)
        if iters % 2000 == 0:
            beta = beta_scale0 * best_cmax * (1.0 - 0.1 * min(iters / 20000, 1.0))

        # Check improvement
        if iters % 50 == 0:
            c_check = _full_convolve(a, a)
            sc = float(c_check.max())
            if sc < best_cmax:
                best_cmax = sc
                best_a = a.copy()
                if iters % 500 == 0:
                    print(f"    HB: iter={iters}, cmax={sc:.10f}, lr={lr:.6f}, beta={beta:.8f}")

    return best_a, best_cmax


def _lbfgs_optimize(a_init, deadline, beta_scale=0.005):
    """L-BFGS optimization using softmax parameterization."""
    n = len(a_init)
    a_norm = a_init / a_init.sum()
    z = np.log(np.maximum(a_norm, 1e-15))
    z = z - z.mean()

    c_init = _full_convolve(a_norm, a_norm)
    M_init = float(c_init.max())
    beta = beta_scale * M_init

    best_a = a_norm.copy()
    best_val = M_init

    state = {'count': 0, 'beta': beta}

    def fun_and_grad(z_vec):
        state['count'] += 1
        a, _ = _softmax_fn(z_vec)
        c = _full_convolve(a, a)
        M = float(c.max())
        c_shifted = (c - M) / state['beta']
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            val = M
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            val = M + state['beta'] * np.log(s)
            w = e / s
        h, W = _compute_h_and_W(a, w)
        grad = 4.0 * n * a * (h - W)
        # Check for improvement in actual objective
        if M < best_val:
            nonlocal best_a, best_val
            best_val = M
            best_a = a.copy()
        return val, grad

    # Fix best_val in closure - use a list instead
    best_val_list = [M_init]
    best_a_list = [a_norm.copy()]

    def fun_and_grad2(z_vec):
        state['count'] += 1
        a, _ = _softmax_fn(z_vec)
        c = _full_convolve(a, a)
        M = float(c.max())
        c_shifted = (c - M) / state['beta']
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            val = M
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            val = M + state['beta'] * np.log(s)
            w = e / s
        h, W = _compute_h_and_W(a, w)
        grad = 4.0 * n * a * (h - W)
        if M < best_val_list[0]:
            best_val_list[0] = M
            best_a_list[0] = a.copy()
        return val, grad

    try:
        res = minimize(fun_and_grad2, z, method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 5000, 'maxfun': 5000,
                                'ftol': 1e-15, 'gtol': 1e-12,
                                'maxls': 50, 'maxcor': 20})
    except Exception as ex:
        print(f"    L-BFGS error: {ex}")

    # Final check
    a_final = best_a_list[0]
    c_final = _full_convolve(a_final, a_final)
    M_final = float(c_final.max())
    if M_final < best_val_list[0]:
        best_val_list[0] = M_final
        best_a_list[0] = a_final

    print(f"    L-BFGS: {state['count']} evals, best cmax={best_val_list[0]:.10f}")
    return best_a_list[0], best_val_list[0]


def _lbfgs_multiscale(a_init, deadline, scale_list=None):
    """L-BFGS with decreasing beta (multi-scale / continuation method)."""
    n = len(a_init)
    if scale_list is None:
        scale_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    a = a_init / a_init.sum()
    best_a = a.copy()
    c = _full_convolve(a, a)
    best_cmax = float(c.max())
    M_ref = best_cmax

    for i, bs in enumerate(scale_list):
        if time.monotonic() > deadline:
            break
        beta = bs * M_ref
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Set up L-BFGS
        z = np.log(np.maximum(a, 1e-15))
        z = z - z.mean()
        state = {'beta': beta, 'best_cmax': best_cmax, 'best_a': best_a.copy(), 'n': n}

        def obj_grad(z_vec):
            a_s, _ = _softmax_fn(z_vec)
            c_s = _full_convolve(a_s, a_s)
            M_s = float(c_s.max())
            c_shifted = (c_s - M_s) / state['beta']
            e_s = np.exp(c_shifted)
            s_s = float(e_s.sum())
            if s_s < 1e-300 or not np.isfinite(s_s):
                val = M_s
                w_s = np.zeros_like(c_s)
                w_s[int(np.argmax(c_s))] = 1.0
            else:
                val = M_s + state['beta'] * np.log(s_s)
                w_s = e_s / s_s
            h_s, W_s = _compute_h_and_W(a_s, w_s)
            grad_s = 4.0 * state['n'] * a_s * (h_s - W_s)
            if M_s < state['best_cmax']:
                state['best_cmax'] = M_s
                state['best_a'] = a_s.copy()
            return val, grad_s

        max_iter = min(3000, max(100, int(remaining * 200)))
        try:
            minimize(obj_grad, z, method='L-BFGS-B', jac=True,
                     options={'maxiter': max_iter, 'maxfun': max_iter,
                              'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 40, 'maxcor': 25})
        except Exception:
            pass

        a = state['best_a']
        best_a = a.copy()
        c = _full_convolve(a, a)
        sc = float(c.max())
        if sc < best_cmax:
            best_cmax = sc
        M_ref = best_cmax
        print(f"    LBFGS-MS: scale={bs:.4f}, cmax={best_cmax:.10f}, t={time.monotonic()-t_ref:.1f}s")

    return best_a, best_cmax


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    """Perturb in log-space and renormalize."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_fn(z)[0]
```
