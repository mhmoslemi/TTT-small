def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):

    c = _full_convolve(a, a)

def _lbfgs_continuation(a_init, deadline, beta_scale=1.0, max_wall_time=None):
    """L-BFGS with logsumexp continuation using FFT-based gradients."""
    n = len(a_init)

    max_c0 = float(c0.max())

    betas = [b * max_c0 * beta_scale for b in beta_starts]
    
    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:

    for bi, beta in enumerate(betas):
        if time.monotonic() > deadline - 2.0:

            return o

            _, g = _logsumexp_obj_and_grad(z_, beta, n)
            return g
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',

def _peak_equalization(a_init, deadline, n_iter=5000):
    """Directly equalize top-K peaks using multiplicative updates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    K_top = 10  # Number of peaks to target
    
    for it in range(n_iter):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        # Find top K peaks
        top_idx = np.argsort(c)[-K_top:]
        
        # Compute gradient direction: for each position i, how much does it
        # contribute to the top K peaks?
        contrib = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            for j in range(lo, hi + 1):
                partner = k_val - j
                if 0 <= partner < n:
                    contrib[j] += a[partner]
        
        contrib /= K_top
        
        # Multiplicative update: reduce positions with high contribution
        eps = 1e-10
        eta = 0.05  # Step size
        a_trial = a * (1.0 - eta * (contrib - contrib.mean()))
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _score_norm_fn(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
        
        # Decay step size
        if it % 500 == 0:
            eta = max(0.001, eta * 0.9)
    
    return best_a, best_sc

        mean_g = float(np.dot(a_curr, grad_a))
        grad_z = a_curr * (grad_a - mean_g)
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        
        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        # Adaptive learning rate
        if t % 3000 == 0:
            lr = max(0.001, lr * 0.9)
    
    return best_a, best_sc


def _local_pair_search(a_init, deadline, n_trials=20000):
    """Local search: exchange mass between positions to reduce peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    rng = np.random.default_rng(12345)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        max_lag = int(np.argmax(c))
        
        # Find the pair with largest product at the max lag
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        if i_min <= i_max:
            pairs = np.arange(i_min, i_max + 1)
            partners = max_lag - pairs
            products = a[pairs] * a[partners]
            idx = pairs[int(np.argmax(products))]
            
            # Try moving mass from idx to a "safe" position
            # Safe = low contribution to top peaks
            top5 = np.argsort(c)[-5:]
            contrib = np.zeros(n)
            for kl in top5:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                for j in range(lo, hi + 1):
                    partner = kl - j
                    if 0 <= partner < n:
                        contrib[j] += a[partner]
            
            safe = np.argsort(contrib)[:max(1, n // 4)]
            add_idx = int(safe[rng.integers(len(safe))])
            
            a_trial = a.copy()
            frac = rng.uniform(0.15, 0.5)
            delta = a_trial[idx] * frac
            a_trial[idx] = max(0.0, a_trial[idx] - delta)
            a_trial[add_idx] += delta
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        # Also try random block perturbation
        elif it % 3 == 1:
            block_size = int(rng.integers(3, min(20, n // 8)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc


def _spectral_flattening(a_init, deadline, n_iter=200):
    """Try to flatten the power spectrum to reduce convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Compute the convolution (autocorrelation)
        c = _full_convolve(a, a)
        
        # We want to reduce the peaks. In frequency domain:
        # c has DFT |A|^2 (approximately, for circular conv)
        # We want |A|^2 to be less "peaky" in time domain
        # Strategy: reduce the variance of |A|^2
        
        A = np.fft.rfft(a, 2 * n)
        power = np.abs(A) ** 2
        power_mean = power.mean()
        
        # Attenuate the peaks in the power spectrum
        # This should make the autocorrelation more concentrated... 
        # Wait, that's the wrong direction.
        # Actually, we want the TIME domain (autocorrelation) to be flat.
        # The autocorrelation is ifft(|A|^2).
        # To make it flat, we want |A|^2 to be as "peaked" as possible (concentrated at DC).
        # But that would make a concentrate at DC (be constant), which gives triangular autocorr.
        # 
        # Alternatively: the problem is that for non-negative a, we can't easily
        # make the autocorrelation perfectly flat. Let's just try gentle corrections.
        
        # Compute the "excess" at each autocorrelation lag
        c_mean = c.mean()
        c_max = c.max()
        
        # If max is much larger than mean, we have a problem
        # Try: a[i] *= (1 - eta * (local_excess[i])) where local_excess measures
        # how much a[i] contributes to the excess peaks
        
        if c_max / c_mean < 1.01:
            break  # Already quite flat
        
        # Compute contribution of each a[i] to the excess
        excess = np.maximum(c - c_mean, 0)
        # For each i, contribution to excess = sum_k excess[k] * a[k-i] / sum(excess)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        
        # FFT-based: contrib[i] = sum_k excess[k] * a[k-i] (for valid k-i)
        # Similar to before: contrib[i] = sum_j a[j] * excess[i+j]
        ex_rev = excess[::-1].copy()
        N = _next_pow2(3 * n)
        fa = np.fft.rfft(a, N)
        fe = np.fft.rfft(ex_rev, N)
        conv = np.fft.irfft(fa * fe, N)[:2*n-1]
        contrib = conv[n-1:2*n-1][::1]
        # Wait, I need to be careful with the indexing
        # conv[m] = sum_j a[j] * ex_rev[m-j] = sum_j a[j] * excess[2n-2-(m-j)] = sum_j a[j] * excess[2n-2-m+j]
        # I want contrib[i] = sum_j a[j] * excess[i+j]
        # Set 2n-2-m+j = i+j => 2n-2-m = i => m = 2n-2-i
        # So contrib[i] = conv[2n-2-i]
        contrib = conv[2*n-2 : n-2 : -1]
        
        contrib /= (total_excess + 1e-15)
        
        eta = 0.02 * (1.0 - it / n_iter)
        a_trial = a * (1.0 - eta * (contrib - contrib.mean()))
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()
        
        sc = _score_norm_fn(a_trial)
        if sc < best_sc:
            best_sc = sc
            best_a = a_trial.copy()
            a = a_trial.copy()
    
    return best_a, best_sc


def _try_multiple_n(a_best, deadline, n_options):
    """Try interpolating to different n values and re-optimizing briefly."""
    best_seq = a_best.copy()
    best_sc = _score_norm_fn(a_best)
    
    for target_n in n_options:
        if time.monotonic() > deadline - 10:
            break
        if target_n == len(a_best) or target_n < 1024 or target_n > 16384:
            continue
        
        p_interp = _interp_profile(a_best, target_n)
        sc_interp = _score_norm_fn(p_interp)
        
        if sc_interp < best_sc:
            best_sc = sc_interp
            best_seq = p_interp.copy()
            print(f"  Interp n={target_n}: score={best_sc:.10f}")
        
        # Quick refinement
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 20, time.monotonic() + 30.0)
            a_r, sc_r = _lbfgs_continuation(p_interp, dl, beta_scale=0.3, max_wall_time=25.0)
            if sc_r < best_sc:
                best_sc = sc_r
                best_seq = a_r.copy()
                print(f"  Refined n={target_n}: score={best_sc:.10f}")
    
    return best_seq, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimizer combining:
    - FFT-accelerated L-BFGS with logsumexp continuation
    - Peak equalization via multiplicative updates
    - Spectral flattening
    - Adam with top-K subgradient
    - Local pair search
    - Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    n_parent = 8192 if parent is None else len(parent)
    
    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        parent = _make_triangular_weight(8192)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, triangular n=8192, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    # Determine working n (use parent's n or 8192)
    n_work = n_parent if 2048 <= n_parent <= 16384 else 8192
    
    # --- Phase 0: Diverse initialization (50s) ---
    phase0_end = min(deadline, t_start + 50.0)
    
    profiles = [
        _make_triangular_weight(n_work),
        _make_cosine_window(n_work),
        _make_raised_cosine_alpha(n_work, 0.7),
        _make_raised_cosine_alpha(n_work, 0.85),
        _make_raised_cosine_alpha(n_work, 1.0),
        _make_raised_cosine_alpha(n_work, 1.2),
        _make_raised_cosine_alpha(n_work, 1.5),
        _make_raised_cosine_alpha(n_work, 2.0),
        _make_kaiser_window(n_work, 4.0),
        _make_kaiser_window(n_work, 6.0),
        _make_kaiser_window(n_work, 8.0),
        _make_bimodal(n_work, 0.3, 0.15),
        _make_bimodal(n_work, 0.5, 0.2),
        _make_bimodal(n_work, 0.6, 0.25),
        _make_power_law(n_work, 2.0),
        _make_power_law(n_work, 3.0),
        _make_w_profile(n_work, 2, 0.3),
        _make_w_profile(n_work, 3, 0.2),
        _make_w_profile(n_work, 4, 0.15),
        _make_w_profile(n_work, 6, 0.1),
    ]
    
    # Also include parent if different n
    if parent is not None and n_parent != n_work:
        profiles.append(_interp_profile(parent, n_work))
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: {len(candidates)} profiles, best={best_score:.10f}")
    
    # Select top 3 for main optimization
    top_starts = [c[1].copy() for c in candidates[:3]]
    if parent is not None:
        p_norm = _interp_profile(parent, n_work)
        sc_p = _score_norm_fn(p_norm)
        if sc_p < _score_norm_fn(top_starts[-1]):
            top_starts[-1] = p_norm
    
    # --- Phase 1: L-BFGS continuation on best starts (700s) ---
    phase1_end = min(deadline, t_start + 750.0)
    print(f"Phase 1: L-BFGS continuation, deadline in {phase1_end - time.monotonic():.0f}s")
    
    per_start = max(30.0, (phase1_end - time.monotonic()) / len(top_starts))
    
    for si, start in enumerate(top_starts):
        if time.monotonic() > phase1_end - 10:
            break
        print(f"  Start {si+1}/{len(top_starts)}: initial score={_score_norm_fn(start):.10f}")
        dl = min(phase1_end - 10, time.monotonic() + per_start)
        a_best, sc_best = _lbfgs_continuation(start, dl, beta_scale=1.0, max_wall_time=max(10.0, per_start - 10))
        update_best(a_best, sc_best, f"lbfgs_s{si}")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Peak equalization (250s) ---
    phase2_end = min(deadline, t_start + 1000.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: Peak equalization")
        dl = phase2_end - 5.0
        a_best, sc_best = _peak_equalization(best_seq, dl, n_iter=3000)
        update_best(a_best, sc_best, "peak_eq")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Spectral flattening (200s) ---
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Spectral flattening")
        dl = phase3_end - 5.0
        a_best, sc_best = _spectral_flattening(best_seq, dl, n_iter=300)
        update_best(a_best, sc_best, "spectral")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Adam top-K (200s) ---
    phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Adam top-K")
        dl = phase4_end - 5.0
        a_best, sc_best = _adam_topk(best_seq, dl, lr=0.012, k_top=5)
        update_best(a_best, sc_best, "adam_k5")
        
        if time.monotonic() < phase4_end - 20:
            dl2 = phase4_end - 5.0
            a_best2, sc_best2 = _adam_topk(best_seq, dl2, lr=0.008, k_top=10)
            update_best(a_best2, sc_best2, "adam_k10")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Local pair search (150s) ---
    phase5_end = min(deadline, t_start + 1550.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Local pair search")
        dl = phase5_end - 5.0
        a_best, sc_best = _local_pair_search(best_seq, dl, n_trials=15000)
        update_best(a_best, sc_best, "local")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Multi-n exploration (150s) ---
    phase6_end = min(deadline, t_start + 1700.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Multi-n exploration")
        n_curr = len(best_seq)
        n_options = [n_curr * 2, n_curr // 2, 4096, 8192, 16384]
        n_options = [x for x in n_options if x != n_curr and 1024 <= x <= 16384]
        a_best, sc_best = _try_multiple_n(best_seq, phase6_end, n_options)
        update_best(a_best, sc_best, "multi_n")
        n_work = len(best_seq) if best_seq is not None else n_work
    
    print(f"Phase 6 done: best={best_score:.10f}")
    
    # --- Phase 7: Final intensive polish (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 7: Final polishing, remaining={remaining:.0f}s")
        
        # 7a: Very fine L-BFGS
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl, beta_scale=0.3,
                                                   max_wall_time=remaining * 0.35)
            update_best(a_best, sc_best, "final_lbfgs")
        
        # 7b: More local search
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.25)
            a_best, sc_best = _local_pair_search(best_seq, dl, n_trials=10000)
            update_best(a_best, sc_best, "final_local")
        
        # 7c: Final peak equalization
        if time.monotonic() < deadline - 25:
            dl = min(deadline - 20, time.monotonic() + remaining * 0.2)
            a_best, sc_best = _peak_equalization(best_seq, dl, n_iter=2000)
            update_best(a_best, sc_best, "final_peak_eq")
        
        # 7d: One more L-BFGS with very low beta
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.15)
            a_best, sc_best = _lbfgs_continuation(best_seq, dl, beta_scale=0.1,
                                                   max_wall_time=remaining * 0.15)
            update_best(a_best, sc_best, "final_fine_lbfgs")
    
    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")
        
        # Validate with evaluator if feasible
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular_weight(8192)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
