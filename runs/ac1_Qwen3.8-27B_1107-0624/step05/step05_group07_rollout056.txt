"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP-based minimax iterations, fine log-sum-exp L-BFGS,
peak-shaving heuristics, and multi-scale approaches to improve on
the previous state of the art of 1.505966, targeting 1.5027.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 1e-15:
                g_val = g_val / gs
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate_armijo(f, n_iters, deadline, step_size=0.3):
    """LP iterations with Armijo step-size backtracking."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        g, t_val = _lp_minimax(f)
        if g is None:
            break

        # Armijo backtracking
        best_step = 0.0
        current_score = _score_norm_fn(f)
        for alpha in [step_size, step_size * 0.5, step_size * 0.25, step_size * 0.1, 0.05, 0.02, 0.01]:
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-15:
                continue
            f_new = f_new / s
            sc = _score_norm_fn(f_new)
            if sc < current_score:
                best_step = alpha
                current_score = sc
                break

        if best_step > 0:
            f_new = (1 - best_step) * f + best_step * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            f_new = f_new / s
            sc = _score_norm_fn(f_new)
            if sc < best_score:
                best_score = sc
                best_a = f_new.copy()
            f = f_new
        else:
            # Try a very small step
            f_new = 0.99 * f + 0.01 * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            f_new = f_new / s
            f = f_new

        if it % 10 == 0:
            print(f"    LP iter {it}: score={_score_norm_fn(f):.10f}, best={best_score:.10f}")

    return best_a, best_score


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lse_lbfgs(a_init, deadline, beta_list=None):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003,
                 0.00002, 0.00001, 0.000005, 0.000002, 0.000001]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        maxiter = max(100, int(time_per * 40))

        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-16,
                         'maxls': 60, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                    if idx % 5 == 0:
                        print(f"    LSE beta={beta:.2e}: score={best_score:.10f}")
        except Exception:
            pass

    return best_a, best_score


def _peak_shave(a_init, deadline, max_iters=200):
    """Directly reduce the peak of the convolution via targeted gradient steps."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a = a_init.copy()
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Find all lags near the max
        threshold = cm * 0.999
        active = np.where(c >= threshold)[0]

        # Average gradient over active lags
        grad = np.zeros(n)
        for k in active:
            # grad[j] = a[k-j] * 1[0<=k-j<n] + a[j-k] * 1[0<=j-k<n]
            j_arr = np.arange(n)
            term1_idx = k - j_arr
            term1 = np.where((term1_idx >= 0) & (term1_idx < n), a[term1_idx], 0.0)
            term2_idx = j_arr - k
            term2 = np.where((term2_idx >= 0) & (term2_idx < n), a[term2_idx], 0.0)
            grad += term1 + term2

        # Take a step to reduce the peak
        # In log-space: z_new = z - lr * grad / a (to get a_step proportional to -grad)
        # Actually, in log-space, da_j = a_j * dz_j, so dz_j = -lr * grad_j / a_j
        lr = 1e-4
        dz = -lr * grad / np.maximum(a, 1e-15)
        z = z + dz
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            # Adaptive: if improving, keep going
        else:
            # Revert and try smaller step
            z = z - dz
            if lr < 1e-8:
                break
            lr *= 0.5
            if it % 50 == 0:
                print(f"    PeakShave iter {it}: score={best_score:.10f}, lr={lr:.2e}")

    return best_a, best_score


def _adaptive_gd(a_init, deadline, rng, lr_init=0.15, max_iter=50000):
    """Adaptive gradient descent with momentum on log-softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-9, 0.02 * cm)
        vals = np.clip((c - cm) / beta, -100, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 5000 == 0:
            lr = max(0.01, lr * 0.85)
            print(f"    ADG iter {t}: score={best_score:.10f}, lr={lr:.4f}")
        if t % 20000 == 0 and t > 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=30):
    """Coordinate descent in log-space with fine deltas."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    deltas = [-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        current_score = _score_norm_fn(a)
        improved = False
        # Random permutation for diversity
        order = np.arange(n)
        rng_local = np.random.default_rng(rnd)
        rng_local.shuffle(order)

        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = current_score
            for d in deltas:
                z[j] += d
                a_t = _softmax_fn(z)
                sc = _score_norm_fn(a_t)
                if sc < best_local:
                    best_local = sc
                    best_delta = d
                z[j] -= d

            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                current_score = best_local
                improved = True
                if current_score < best_score:
                    best_score = current_score
                    best_a = _softmax_fn(z).copy()

        if not improved:
            break
        if rnd % 5 == 0:
            print(f"    CD round {rnd}: score={best_score:.10f}")

    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Linear ramps
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0, 5.0, -1.0, -2.0, -3.0, -5.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Cosine
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    # Sine modulation
    a = 1.0 + 0.5 * np.sin(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = 1.0 + 0.8 * np.sin(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    # Block patterns
    half = n // 2
    for ratio in [0.5, 0.6, 0.7, 0.8, 1.5, 2.0, 3.0]:
        a = np.ones(n)
        a[:half] = 1.0
        a[half:] = ratio
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # U-shaped
    a = 1.0 / (0.1 + u ** 2)
    profiles.append(a / a.sum())
    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _multi_scale_optimize(a_init, deadline, n_scales=3, scale_ratios=[4, 2, 1]):
    """Optimize at multiple scales: coarse to fine."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    # Work from coarse to fine
    current = a_init.copy()
    for scale in scale_ratios:
        if time.monotonic() > deadline:
            break
        n_coarse = max(64, n // scale)
        # Downsample
        coarse = _interp_profile(current, n_coarse)
        # Optimize at this scale with L-BFGS
        time_slice = (deadline - time.monotonic()) / max(1, len(scale_ratios) - scale_ratios.index(scale) + 1)
        per_dl = time.monotonic() + min(time_slice, 120.0)

        c0 = _full_convolve(coarse, coarse)
        beta_start = 0.05 * float(c0.max())
        beta_list = [beta_start * (0.5 ** i) for i in range(10)]

        p_best, sc_best = _lse_lbfgs(coarse, per_dl, beta_list=beta_list)
        if sc_best < best_score:
            best_score = sc_best
            # Upsample back
            best_a = _interp_profile(p_best, n)
            current = best_a.copy()
            print(f"    MultiScale coarse_n={n_coarse}: score={best_score:.10f}")

    return best_a, best_score

            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            try:
                result = minimize(
                    _lse_obj, z, args=(beta_ultra, n_final),
                    method='L-BFGS-B', jac=_lse_grad,
                    options={'maxiter': 3000, 'ftol': 1e-25, 'gtol': 1e-16,
                             'maxls': 80, 'maxfun': 15000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_seq = a_new.copy()
                        print(f"  Final tiny-beta: {best_score:.10f}")
            except Exception:
                pass

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
