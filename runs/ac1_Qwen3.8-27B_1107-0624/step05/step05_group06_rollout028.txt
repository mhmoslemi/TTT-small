"""
SLP (Sequential Linear Programming) with active set refinement for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines L-BFGS smoothing, SLP with LP subproblems, and subgradient polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_exact(a, c, n):
    """Gradient using exact active set (coeffs within tolerance of max)."""
    c_max = c.max()
    tol = 0.001 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _lp_solve_slep(a_cur, c_cur, active_indices, n):
    """
    SLP: Solve LP for the linearized problem at a_cur.
    
    minimize t
    subject to: 2*(C_k a_cur)^T a - t <= c_cur[k]  for k in active_indices
                 sum(a) = 1
                 a >= 0, t >= 0
    """
    m = len(active_indices)
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequality constraints
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    
    for idx, k in enumerate(active_indices):
        # (C_k a_cur)[i] = a_cur[k-i] for max(0, k-n+1) <= i <= min(n-1, k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            # Coefficient of a[i] is 2 * a_cur[k-i]
            # For i in [i_min, i_max], k-i ranges from k-i_min to k-i_max
            coeffs = a_cur[k - i_max : k - i_min + 1]  # a_cur[k-i] for i from i_min to i_max
            A_ub[idx, i_min:i_max+1] = 2.0 * coeffs
        
        A_ub[idx, n] = -1.0
        b_ub[idx] = c_cur[k]

    bounds = [(0, None)] * (n + 1)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            return result.x[:n], result.x[n]
    except Exception:
        pass
    
    return None, None

def _slp_iteration(a_cur, deadline, max_iter=200, active_frac_start=0.2, active_frac_end=0.02):
    """
    SLP iterations with gradually refining active set.
    a_cur is normalized to sum=1.
    """
    n = len(a_cur)
    total_conv_len = 2 * n - 1
    
    a = a_cur.copy()
    best_a = a.copy()
    best_score = _score(a)
    
    n_stages = 5
    for stage in range(n_stages):
        frac = active_frac_start * (active_frac_end / active_frac_start) ** (stage / max(1, n_stages - 1))
        n_active = max(10, int(frac * total_conv_len))
        
        for it in range(max_iter // n_stages):
            if time.monotonic() > deadline:
                return best_a, best_score
            
            c = _full_convolve(a, a)
            c_sorted = np.sort(c)[::-1]
            threshold = c_sorted[min(n_active, len(c_sorted)) - 1]
            active_mask = c >= threshold
            active_indices = np.where(active_mask)[0]
            
            a_opt, t_opt = _lp_solve_slep(a, c, active_indices, n)
            
            if a_opt is None:
                break
            
            a_opt = np.maximum(a_opt, 0)
            s_opt = a_opt.sum()
            if s_opt < 1e-15:
                break
            a_opt = a_opt / s_opt
            
            # Backtracking line search
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                a_try = (1 - alpha) * a + alpha * a_opt
                a_try = a_try / a_try.sum()
                sc_try = _score(a_try)
                if sc_try < best_score * (1 + 1e-10):
                    a = a_try
                    best_a = a_try.copy()
                    best_score = sc_try
                    break
            else:
                # No improvement, break
                break
            
            if it > 0 and abs(best_score - _score(a)) < 1e-12:
                break
    
    return best_a, best_score

def _lbfgs_optimize(a_init, beta, n, deadline, max_iter=5000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return _smoobj(z, beta, n)
    
    def jac(z):
        return _smograd(z, beta, n)
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()

def _subgrad_polish(a_init, deadline, max_steps=3000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    prev_sc = best_sc
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_exact(a, c, n)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Polyak-style step: use gap to adjust
        step = lr / (1.0 + 0.5 * i * 0.01)
        z = z - step * grad
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                lr = min(lr * 1.05, 0.1)
            else:
                lr *= 0.9
                # Rollback
                z = z + (step / 2) * grad
                z = np.clip(z, -60.0, 60.0)
                z = z - z.mean()
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
SLP (Sequential Linear Programming) with active set refinement combined with
L-BFGS log-sum-exp smoothing and subgradient polishing for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovation: At each SLP iteration, the non-convex quadratic constraint
a^T C_k a <= t is linearized at the current point, yielding a linear program
in the original (non-log) space. The active set (near-maximal convolution
positions) is gradually refined across stages for convergence.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_exact(a, c, n):
    c_max = c.max()
    tol = 0.001 * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lp_solve_slep(a_cur, c_cur, active_indices, n):
    """
    Solve the SLP sub-problem (LP):
      minimize t
      subject to: 2*(C_k a_cur)^T a - t <= c_cur[k]  for k in active_indices
                   sum(a) = 1
                   a >= 0, t >= 0
    """
    m = len(active_indices)
    if m == 0:
        return None, None

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, k in enumerate(active_indices):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            coeffs = a_cur[k - i_max:k - i_min + 1][::-1]
            A_ub[idx, i_min:i_max + 1] = 2.0 * coeffs
        A_ub[idx, n] = -1.0
        b_ub[idx] = c_cur[k]

    bounds = [(0, None)] * (n + 1)

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            return result.x[:n], result.x[n]
    except Exception:
        pass

    return None, None


def _slp_refine(a_init, deadline, max_stages=5, max_iter_per_stage=60,
                active_frac_start=0.15, active_frac_end=0.01):
    """
    SLP with gradually refining active set.
    a_init should be normalized to sum=1.
    Returns (best_a, best_score).
    """
    n = len(a_init)
    total_conv_len = 2 * n - 1

    a = a_init.copy()
    best_a = a.copy()
    best_score = _score(a)

    n_stages = max_stages
    iters_per_stage = max_iter_per_stage

    for stage in range(n_stages):
        if time.monotonic() > deadline:
            break
        frac = active_frac_start * (active_frac_end / active_frac_start) ** (
            stage / max(1, n_stages - 1))
        n_active = max(10, int(frac * total_conv_len))
        n_active = min(n_active, total_conv_len)

        for it in range(iters_per_stage):
            if time.monotonic() > deadline:
                return best_a, best_score

            c = _full_convolve(a, a)
            c_sorted = np.sort(c)[::-1]
            threshold = c_sorted[min(n_active, len(c_sorted)) - 1]
            active_mask = c >= threshold
            active_indices = np.where(active_mask)[0]

            if len(active_indices) == 0:
                break

            a_opt, t_opt = _lp_solve_slep(a, c, active_indices, n)

            if a_opt is None:
                break

            a_opt = np.maximum(a_opt, 0.0)
            s_opt = a_opt.sum()
            if s_opt < 1e-15:
                break
            a_opt = a_opt / s_opt

            improved = False
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.25, 0.1]:
                a_try = (1.0 - alpha) * a + alpha * a_opt
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                sc_try = _score(a_try)
                if sc_try <= best_score * (1.0 + 1e-12):
                    a = a_try
                    if sc_try < best_score:
                        best_score = sc_try
                        best_a = a.try.copy() if hasattr(a_try, 'copy') else a_try.copy()
                        best_a = a.copy()
                        best_score = sc_try
                    improved = True
                    break

            if not improved:
                if it >= 3:
                    break

    return best_a, best_score


def _lbfgs_optimize(a_init, beta, n, deadline, max_iter=5000):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()

    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return a_init.copy()


def _subgrad_polish(a_init, deadline, max_steps=3000, lr_init=0.03):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    patience = 0
    max_patience = 500

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_exact(a, c, n)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        step = lr / (1.0 + 0.3 * i * 0.005)
        z_new = z - step * grad
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        if i % 50 == 0:
            a_check = _softmax(z_new)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
                z = z_new
                patience = 0
                lr = min(lr * 1.02, 0.08)
            else:
                patience += 50
                lr *= 0.92
                z = z_new
        else:
            z = z_new

        if patience > max_patience:
            break

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    SLP with active set refinement + L-BFGS + subgradient polishing.
    The SLP phase solves LP subproblems at each iteration (linearizing the
    quadratic autocorrelation constraints), which provides a novel direction
    not captured by pure gradient methods.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.2e} t={time.monotonic() - t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent, uniform n={n_main}")
        update_best(a0, _score(a0), "init")

    n = len(a0)

    # Phase 1: L-BFGS warm-up
    print(f"\nPhase 1: L-BFGS warm-up (t={time.monotonic() - t_start:.1f}s)")
    c_cur = _full_convolve(a0, a0)
    max_c = float(c_cur.max())
    betas_warm = max_c * np.array([0.08, 0.04, 0.02, 0.01])
    current_a = a0.copy()

    for i, beta in enumerate(betas_warm):
        if time.monotonic() > deadline - 600:
            break
        remaining = deadline - 600 - time.monotonic()
        if remaining < 10:
            break
        est_time = 0.015 * (n / 2048.0)
        max_iter = min(3000, max(100, int(remaining / (est_time * 6))))
        current_a = _lbfgs_optimize(current_a, beta, n, deadline - 600, max_iter)

    sc1 = _score(current_a)
    update_best(current_a, sc1, "lbfgs_ph1")

    # Phase 2: SLP with active set (main novel component)
    print(f"\nPhase 2: SLP active set (t={time.monotonic() - t_start:.1f}s)")
    a_slp, sc_slp = _slp_refine(current_a, deadline - 400,
                                max_stages=6, max_iter_per_stage=50,
                                active_frac_start=0.15, active_frac_end=0.008)
    update_best(a_slp, sc_slp, "slp_main")

    # Phase 3: L-BFGS polish on SLP result
    print(f"\nPhase 3: L-BFGS polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 250:
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        fine_betas = mb * np.array([0.005, 0.002, 0.001, 0.0004])
        current_a = best_seq.copy()
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 200:
                break
            remaining = deadline - 200 - time.monotonic()
            if remaining < 10:
                break
            est_time = 0.015 * (n / 2048.0)
            max_iter = min(2500, max(100, int(remaining / (est_time * 6))))
            a_new = _lbfgs_optimize(current_a, beta, n, deadline - 200, max_iter)
            sc = _score(a_new)
            if sc < _score(current_a) - 1e-15:
                current_a = a_new
            update_best(a_new, sc, f"polish_{i}")

    # Phase 4: Second SLP with tighter active set
    print(f"\nPhase 4: SLP tighten (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 150:
        a_slp2, sc_slp2 = _slp_refine(best_seq, deadline - 130,
                                      max_stages=5, max_iter_per_stage=40,
                                      active_frac_start=0.08, active_frac_end=0.003)
        update_best(a_slp2, sc_slp2, "slp_tight")

    # Phase 5: Subgradient polishing
    print(f"\nPhase 5: Subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        a_sg, sc_sg = _subgrad_polish(best_seq, deadline - 50,
                                      max_steps=2000, lr_init=0.015)
        update_best(a_sg, sc_sg, "subgrad")

    # Phase 6: Multi-restart
    print(f"\nPhase 6: Multi-restart (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        remaining = deadline - 80 - time.monotonic()
        n_restarts = min(3, max(1, int(remaining / 150)))
        for i in range(n_restarts):
            if time.monotonic() > deadline - 70:
                break

            if i == 0:
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z + 0.03 * rng.standard_normal(len(z))
                z = z - z.mean()
                a_pert = _softmax(z)
            elif i == 1:
                z = np.log(np.maximum(best_seq, 1e-15))
                t_arr = np.arange(n)
                freq = rng.integers(1, max(2, n // 4))
                amp = 0.015 * rng.uniform(0.5, 2.0)
                phase = rng.uniform(0, 2 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * freq * t_arr / n + phase)
                z = z - z.mean()
                a_pert = _softmax(z)
            else:
                a_pert = best_seq * (1.0 + 0.02 * rng.standard_normal(n))
                a_pert = np.maximum(a_pert, 1e-10)
                a_pert = a_pert / a_pert.sum()

            # Quick SLP from perturbed
            a_r, sc_r = _slp_refine(a_pert, deadline - 70,
                                    max_stages=3, max_iter_per_stage=30,
                                    active_frac_start=0.1, active_frac_end=0.01)
            update_best(a_r, sc_r, f"restart_{i}_slp")

            # Quick L-BFGS
            if time.monotonic() < deadline - 60:
                c_r = _full_convolve(a_r, a_r)
                mb_r = float(c_r.max())
                a_r2 = _lbfgs_optimize(a_r, mb_r * 0.003, n, deadline - 60, max_iter=1500)
                update_best(a_r2, _score(a_r2), f"restart_{i}_lbfgs")

    # Phase 7: Final SLP + L-BFGS cycle
    print(f"\nPhase 7: Final cycle (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a_fin, sc_fin = _slp_refine(best_seq, deadline - 4,
                                    max_stages=3, max_iter_per_stage=30,
                                    active_frac_start=0.05, active_frac_end=0.002)
        update_best(a_fin, sc_fin, "final_slp")

        if time.monotonic() < deadline - 3:
            c_fin = _full_convolve(best_seq, best_seq)
            mb = float(c_fin.max())
            a_fin2 = _lbfgs_optimize(best_seq, mb * 0.0003, n, deadline - 3,
                                     max_iter=800)
            update_best(a_fin2, _score(a_fin2), "final_lbfgs")

        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _subgrad_polish(best_seq, deadline - 2,
                                              max_steps=500, lr_init=0.008)
            update_best(a_fin3, sc_fin3, "final_sg")

    # Final validation
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
