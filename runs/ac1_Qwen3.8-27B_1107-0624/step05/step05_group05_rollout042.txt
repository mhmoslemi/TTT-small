                                'gtol': 1e-10, 'maxls': 50})
        a_new = _softmax(res.x)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"    L-BFGS improved: score={sc:.10f}")
    except Exception as e:
        print(f"    L-BFGS error: {e}")

    elapsed = time.monotonic() - t_start
    print(f"    L-BFGS done in {elapsed:.1f}s, beta={beta:.2e}, best={best_score:.10f}")
    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_passes=200):
    """Coordinate descent with efficient C-tracking."""
    n = len(a_init)
    a = a_init.copy()
    C = _full_convolve(a, a)
    best_score = float(C.max())
    best_a = a.copy()

    for pass_num in range(max_passes):
        if time.monotonic() > deadline:
            break

        any_improvement = False

        for j in range(n):
            if time.monotonic() > deadline:
                break

            old_val = a[j]
            if old_val < 1e-15:
                # Try a small positive value
                candidates = [1e-6, 1e-5, 1e-4]
            else:
                delta = old_val * 0.15
                candidates = [
                    0.0,
                    max(0.0, old_val - delta),
                    max(0.0, old_val - 0.5 * delta),
                    old_val,
                    old_val + 0.5 * delta,
                    old_val + delta,
                    old_val * 0.5,
                    old_val * 1.5,
                ]

            best_cand = old_val
            best_max_c = float(C.max())

            for val in candidates:
                if val == old_val:
                    continue
                # Update C efficiently
                # Change in a[j]: dv = val - old_val
                dv = val - old_val
                # Affected k values: k in [j, j+n-1] intersect [0, 2n-2]
                k_lo = max(0, j)
                k_hi = min(2 * n - 1, j + n)
                # For k in this range, the index into a is k-j (in [0, n-1])
                idx_in_a = np.arange(k_lo - j, k_hi - j)
                
                # New C[k] = C[k] + 2*dv*a[k-j] for k != 2j
                # New C[2j] = C[2j] + dv*(val + old_val) = C[2j] + dv*2*old_val + dv^2
                # Actually: C[k] has term 2*a[j]*a[k-j] (for k-j != j) or a[j]^2 (for k-j=j)
                # Change: 2*dv*a[k-j] (for k-j != j) or (val^2 - old_val^2) = dv*(val+old_val) (for k-j=j)
                
                new_C_affected = C[k_lo:k_hi] + 2.0 * dv * a[idx_in_a]
                # Fix the k=2j entry if it's in range
                if k_lo <= 2 * j < k_hi:
                    pos_in_range = 2 * j - k_lo
                    new_C_affected[pos_in_range] = C[2 * j] + (val * val - old_val * old_val)

                # Compute new max
                new_max = max(float(new_C_affected.max()), 
                              float(np.delete(C, np.arange(k_lo, k_hi)).max()) if k_hi > k_lo < 2*n-1 else 0.0)
                # Simpler: just take max of the affected part and the rest
                if k_lo == 0 and k_hi >= 2 * n - 1:
                    new_max = float(new_C_affected.max())
                else:
                    # The unaffected part's max is at most the current max
                    # (since we're increasing some entries)
                    # For safety, just use current max as upper bound for unaffected
                    unaffected_max = best_max_c  # conservative
                    new_max = max(float(new_C_affected.max()), unaffected_max)

                if new_max < best_max_c - 1e-12:
                    best_max_c = new_max
                    best_cand = val

            if best_cand != old_val:
                # Apply the best candidate
                dv = best_cand - old_val
                k_lo = max(0, j)
                k_hi = min(2 * n - 1, j + n)
                idx_in_a = np.arange(k_lo - j, k_hi - j)
                C[k_lo:k_hi] += 2.0 * dv * a[idx_in_a]
                if k_lo <= 2 * j < k_hi:
                    pos_in_range = 2 * j - k_lo
                    C[2 * j] = C[2 * j] + (best_cand * best_cand - old_val * old_val)  # already added 2*dv*a[j] above, fix
                    # Actually let's just recompute C[2j] properly
                    # C[2j] should have base + best_cand^2
                    # We added 2*dv*a[idx] which includes 2*dv*old_val at position 2j
                    # We need C[2j] += dv*(best_cand + old_val) instead of 2*dv*old_val
                    # Correction: subtract 2*dv*old_val, add dv*(best_cand+old_val)
                    C[2 * j] += dv * (best_cand + old_val) - 2.0 * dv * old_val
                a[j] = best_cand
                any_improvement = True

        if best_max_c < best_score - 1e-10:
            best_score = best_max_c
            best_a = a.copy()

        if not any_improvement:
            # No improvement in this pass
            # Try a random subset of coordinates with larger steps
            if pass_num % 5 == 0:
                rng_local = np.random.default_rng(pass_num * 12345)
                for _ in range(50):
                    j = int(rng_local.integers(0, n))
                    old_val = a[j]
                    new_val = old_val * (1.0 + 0.3 * rng_local.standard_normal())
                    new_val = max(0.0, min(new_val, 2.0))
                    if abs(new_val - old_val) < 1e-10:
                        continue
                    # Quick check
                    a[j] = new_val
                    c_max_new = float(_full_convolve(a, a).max())
                    if c_max_new > best_max_c * 1.001:
                        a[j] = old_val
                    elif c_max_new < best_max_c:
                        C = _full_convolve(a, a)
                        best_max_c = c_max_new
                # Refresh C
                C = _full_convolve(a, a)

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    """Multi-phase optimizer: Adam + L-BFGS + Coordinate Descent."""
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        a = parent.copy()
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"Parent: n={n}, score={init_sc:.10f}")
    else:
        n = 2048
        a = np.ones(n) / n
        init_sc = _score(a)
        best_seq = a.copy()
        best_score = init_sc
        print(f"No parent, uniform n={n}, score={init_sc:.10f}")

    print(f"Total budget: {budget_s}s, deadline at t={deadline - t_start:.1f}s")

    # === Phase 1: Adam with moderate-to-fine beta (0 to 40% of time) ===
    phase1_end = t_start + 0.40 * (deadline - t_start)
    print(f"Phase 1 (Adam): {t_start - t_start:.0f}s to {phase1_end - t_start:.0f}s")

    c_init = _full_convolve(a, a)
    c_max_init = float(c_init.max())

    # Beta schedule: (end_fraction_of_phase, beta_relative_to_c_max)
    # Start with moderate smoothing, decrease to fine
    betas_phase1 = [
        (0.0, 0.05),    # beta = 0.05 * c_max
        (0.2, 0.03),
        (0.4, 0.02),
        (0.6, 0.01),
        (0.8, 0.005),
        (1.0, 0.002),
    ]
    betas_abs = [(f, b * c_max_init) for f, b in betas_phase1]

    a, sc = _adam_phase(a, phase1_end, betas_abs, lr_start=0.005, lr_end=0.001)
    full_sc = _score(a)
    print(f"Phase 1 done: score={full_sc:.10f}")
    if np.isfinite(full_sc) and full_sc < best_score:
        best_score = full_sc
        best_seq = a.copy()

    # === Phase 2: L-BFGS with small beta (40% to 55%) ===
    phase2_end = t_start + 0.55 * (deadline - t_start)
    print(f"Phase 2 (L-BFGS): {phase1_end - t_start:.0f}s to {phase2_end - t_start:.0f}s")

    c_now = _full_convolve(best_seq, best_seq)
    c_max_now = float(c_now.max())
    beta_lbfgs = 0.003 * c_max_now

    a2, sc2 = _lbfgs_phase(best_seq, phase2_end, beta_lbfgs)
    full_sc2 = _score(a2)
    print(f"Phase 2 done: score={full_sc2:.10f}")
    if np.isfinite(full_sc2) and full_sc2 < best_score:
        best_score = full_sc2
        best_seq = a2.copy()

    # === Phase 3: Adam with very fine beta (55% to 75%) ===
    phase3_end = t_start + 0.75 * (deadline - t_start)
    print(f"Phase 3 (Adam fine): {phase2_end - t_start:.0f}s to {phase3_end - t_start:.0f}s")

    c_now3 = _full_convolve(best_seq, best_seq)
    c_max3 = float(c_now3.max())
    betas_phase3 = [
        (0.0, 0.002),
        (0.3, 0.001),
        (0.6, 0.0005),
        (0.8, 0.0002),
        (1.0, 0.0001),
    ]
    betas_abs3 = [(f, b * c_max3) for f, b in betas_phase3]

    a3, sc3 = _adam_phase(best_seq, phase3_end, betas_abs3, lr_start=0.003, lr_end=0.0002)
    full_sc3 = _score(a3)
    print(f"Phase 3 done: score={full_sc3:.10f}")
    if np.isfinite(full_sc3) and full_sc3 < best_score:
        best_score = full_sc3
        best_seq = a3.copy()

    # === Phase 4: Coordinate Descent (75% to 92%) ===
    phase4_end = t_start + 0.92 * (deadline - t_start)
    print(f"Phase 4 (Coord Descent): {phase3_end - t_start:.0f}s to {phase4_end - t_start:.0f}s")

    a4, sc4 = _coordinate_descent(best_seq, phase4_end, max_passes=500)
    full_sc4 = _score(a4)
    print(f"Phase 4 done: score={full_sc4:.10f}")
    if np.isfinite(full_sc4) and full_sc4 < best_score:
        best_score = full_sc4
        best_seq = a4.copy()

    # === Phase 5: Final L-BFGS + Adam polish (92% to deadline) ===
    phase5_end = deadline - 5
    print(f"Phase 5 (Final polish): {phase4_end - t_start:.0f}s to {phase5_end - t_start:.0f}s")

    c_now5 = _full_convolve(best_seq, best_seq)
    c_max5 = float(c_now5.max())

    # Quick L-BFGS
    a5, sc5 = _lbfgs_phase(best_seq, min(phase5_end, t_start + 0.97 * (deadline - t_start)), 0.001 * c_max5)
    full_sc5 = _score(a5)
    if np.isfinite(full_sc5) and full_sc5 < best_score:
        best_score = full_sc5
        best_seq = a5.copy()

    # Final Adam with ultra-fine beta
    if time.monotonic() < phase5_end - 10:
        c_now6 = _full_convolve(best_seq, best_seq)
        c_max6 = float(c_now6.max())
        betas_final = [
            (0.0, 0.0005),
            (0.5, 0.0002),
            (1.0, 0.0001),
        ]
        betas_abs6 = [(f, b * c_max6) for f, b in betas_final]
        a6, sc6 = _adam_phase(best_seq, phase5_end, betas_abs6, lr_start=0.001, lr_end=0.0001)
        full_sc6 = _score(a6)
        if np.isfinite(full_sc6) and full_sc6 < best_score:
            best_score = full_sc6
            best_seq = a6.copy()

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
