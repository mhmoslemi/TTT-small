"""
Gradient-based optimization with Nesterov momentum, adaptive soft-max smoothing,
and multi-restart strategy for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses correct gradient computation via FFT, simplex parameterization via softmax,
and diverse restart strategies to escape local minima.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_gradient(a, beta):
    """Compute gradient of soft-max smoothed max(conv(a,a)) w.r.t. z (softmax param)."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # d_k = (c_k / beta) * w_k
    d = (c / beta) * w
    # grad_a_j = sum_m d_{j+m} * a_m = conv(d, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    dwc = _full_convolve(d, a_rev)
    grad_a = dwc[n - 1:2 * n - 1]
    # Gradient w.r.t. z: a * (grad_a - mean_a(grad_a))
    mean_grad = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - mean_grad)
    return grad_z, cm, w


def _nesterov_optimize(a_init, deadline, lr_init=0.05, beta_init=0.01, beta_min=0.0005,
                       momentum=0.9, max_clip=1.0):
    """Nesterov accelerated gradient descent with adaptive beta soft-max smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z = np.clip(z, -30.0, 30.0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a_init.copy()
    best_score = _score(a_init)

    # Nesterov state
    z_prev = z.copy()
    mu = momentum

    # Beta schedule parameters
    total_time_est = 600.0
    start_time = time.monotonic()

    lr = lr_init
    no_improve_count = 0
    patience = 500

    while time.monotonic() < deadline:
        t_step += 1
        elapsed = time.monotonic() - start_time

        # Adaptive beta: exponential decay
        progress = min(1.0, elapsed / (total_time_est * 0.7))
        beta = beta_init * (beta_min / beta_init) ** progress
        beta = max(beta, beta_min)

        # Nesterov lookahead
        z_look = z + mu * (z - z_prev)
        a_look = _softmax(z_look)

        # Compute gradient at lookahead point
        grad_z, cm, w = _compute_gradient(a_look, beta)

        if not np.all(np.isfinite(grad_z)):
            break

        # Adam-style update
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        # Nesterov update
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > max_clip:
            update = update * (max_clip / update_norm)

        z_prev = z.copy()
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Evaluate every 10 steps
        if t_step % 10 == 0:
            a_curr = _softmax(z)
            sc = _score(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                no_improve_count = 0
            else:
                no_improve_count += 10

        # Learning rate decay
        if t_step % 2000 == 0:
            lr = max(0.001, lr * 0.9)

        # If stuck for too long, increase momentum slightly and reset some state
        if no_improve_count > patience and t_step % 500 == 0:
            mu = min(0.95, mu + 0.02)
            # Small random kick
            z = z + 0.01 * np.random.default_rng(t_step).standard_normal(n)
            z = z - z.mean()
            z = np.clip(z, -50.0, 50.0)
            no_improve_count = 0

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _softmax_optimize_v2(a_init, deadline, lr_init=0.03, beta_init=0.02, beta_min=0.001,
                         momentum=0.9, max_clip=0.5):
    """Alternative optimizer with different schedule and more aggressive beta decay."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    z = np.clip(z, -30.0, 30.0)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.85, 0.995, 1e-8

    best_a = a_init.copy()
    best_score = _score(a_init)
    start_time = time.monotonic()

    mu = momentum
    lr = lr_init

    while time.monotonic() < deadline:
        t_step += 1
        elapsed = time.monotonic() - start_time

        # More aggressive beta decay
        progress = min(1.0, elapsed / 300.0)
        beta = beta_init * (beta_min / beta_init) ** progress
        beta = max(beta, beta_min)

        a_curr = _softmax(z)
        grad_z, cm, w = _compute_gradient(a_curr, beta)

        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        # Use momentum on the velocity
        update = mu * (v_buf - v_buf * b2) + (1 - mu) * lr * mh / (np.sqrt(vh) + eps)
        # Simpler: just use Adam with step decay
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > max_clip:
            update = update * (max_clip / update_norm)

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if t_step % 20 == 0:
            a_eval = _softmax(z)
            sc = _score(a_eval)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_eval.copy()

        if t_step % 1000 == 0:
            lr = max(0.002, lr * 0.92)

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coordinate_descent(a, deadline, max_rounds=100):
    """Simple coordinate descent: for each element, find optimal value given others fixed."""
    n = len(a)
    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Random order for coordinate descent
        order = rng.permutation(n)
        improved = False
        for idx in order:
            if time.monotonic() > deadline:
                break
            # Try a few values for a[idx]
            a_orig = a[idx]
            best_local = a_orig
            best_local_score = _score(a)
            # Try perturbation
            for factor in [0.5, 0.7, 1.3, 1.5, 0.3, 2.0]:
                a_trial = a.copy()
                a_trial[idx] = a_orig * factor
                a_trial = np.maximum(a_trial, 0)
                if a_trial.sum() < 0.01:
                    continue
                sc = _score(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = a_orig * factor
            if best_local != a_orig:
                a[idx] = best_local
                improved = True
        if _score(a) < best_score:
            best_score = _score(a)
            best_a = a.copy()
        if not improved:
            break

    return best_a, best_score


def _perturb(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_seeded_patterns(n, rng):
    """Generate diverse initial patterns."""
    patterns = []
    # Uniform
    patterns.append(np.ones(n) / n)
    # Raised cosine
    t = np.linspace(0, 1, n)
    rc = 0.5 + 0.5 * np.cos(2 * np.pi * t)
    patterns.append(rc / rc.sum())
    # Inverse raised cosine (U-shape)
    irc = 0.5 - 0.5 * np.cos(2 * np.pi * t) + 0.1
    patterns.append(irc / irc.sum())
    # Gaussian
    center = n / 2
    g = np.exp(-((np.arange(n) - center) ** 2) / (2 * (n / 4) ** 2))
    patterns.append(g / g.sum())
    # Two peaks
    p2 = np.exp(-((np.arange(n) - n * 0.25) ** 2) / (2 * (n / 8) ** 2)) + \
         np.exp(-((np.arange(n) - n * 0.75) ** 2) / (2 * (n / 8) ** 2))
    patterns.append(p2 / p2.sum())
    # Random sparse-ish
    r = np.random.default_rng(42).random(n)
    r = r ** 3  # Make it sparser
    patterns.append(r / r.sum())
    return patterns


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase gradient optimization with Nesterov momentum, adaptive soft-max,
    diverse restarts, and coordinate descent polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(a0)
    else:
        n_main = 8192
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")
        update_best(a0)

    # === Phase 1: Main Nesterov optimization (0 - 500s) ===
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: Nesterov optimization (n={n_main}) ===")

    a_opt, sc_opt = _nesterov_optimize(
        a0, phase1_end, lr_init=0.05, beta_init=0.015, beta_min=0.0005,
        momentum=0.9, max_clip=0.8
    )
    update_best(a_opt)
    print(f"Phase 1 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: Second optimizer variant (500s - 900s) ===
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: Alternative optimizer ===")

    if best_seq is not None and time.monotonic() < phase2_end:
        a_opt2, sc_opt2 = _softmax_optimize_v2(
            best_seq / best_seq.sum(), phase2_end,
            lr_init=0.04, beta_init=0.01, beta_min=0.0008,
            momentum=0.9, max_clip=0.6
        )
        update_best(a_opt2)
        print(f"Phase 2 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Multi-restart with diverse initializations (900s - 1300s) ===
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 3: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase3_end:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 5
        per_restart = (phase3_end - time.monotonic() - 30) / max(1, n_restarts)

        # Diverse perturbations
        sigmas = [0.02, 0.05, 0.1, 0.2, 0.3]
        for i in range(n_restarts):
            if time.monotonic() > phase3_end - 20:
                break
            if per_restart < 30:
                break
            deadline_i = time.monotonic() + min(per_restart, phase3_end - time.monotonic() - 10)
            if i < 3:
                # Perturb best
                p_start = _perturb(a_best_norm, sigmas[i], rng)
            elif i == 3:
                # Try a seeded pattern
                p_start = _make_seeded_patterns(len(a_best_norm), rng)[1]  # raised cosine
            else:
                # Larger perturbation
                p_start = _perturb(a_best_norm, 0.5, rng)

            beta_i = 0.01 * (1 + i * 0.5)
            a_r, sc_r = _nesterov_optimize(
                p_start, deadline_i, lr_init=0.05, beta_init=min(beta_i, 0.05),
                beta_min=0.0003, momentum=0.9, max_clip=0.8
            )
            update_best(a_r)

        print(f"Phase 3 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Coordinate descent polish (1300s - 1450s) ===
    phase4_end = min(deadline, t_start + 1450.0)
    print(f"\n=== Phase 4: Coordinate descent polish ===")

    if best_seq is not None and time.monotonic() < phase4_end:
        a_cd, sc_cd = _coordinate_descent(best_seq / best_seq.sum(), phase4_end, max_rounds=50)
        update_best(a_cd)
        print(f"Phase 4 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try different n (1450s - 1650s) ===
    phase5_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 5: Different n exploration ===")

    if best_seq is not None and time.monotonic() < phase5_end:
        target_ns = []
        if len(best_seq) < 4096:
            target_ns = [4096, 8192]
        elif len(best_seq) < 16384:
            target_ns = [16384]
        else:
            target_ns = [4096]  # Try smaller in case it's better

        for target_n in target_ns:
            if time.monotonic() > phase5_end - 60:
                break
            p_interp = _interp_profile(best_seq, target_n)
            update_best(p_interp)

            remaining = phase5_end - time.monotonic() - 30
            if remaining > 80:
                a_r, sc_r = _nesterov_optimize(
                    p_interp, time.monotonic() + remaining,
                    lr_init=0.04, beta_init=0.01, beta_min=0.0003,
                    momentum=0.9, max_clip=0.7
                )
                update_best(a_r)

        print(f"Phase 5 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final intensive polish (1650s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 8
        if remaining > 30:
            # Final Nesterov with very small beta
            a_fin, sc_fin = _nesterov_optimize(
                a_best_norm, time.monotonic() + remaining,
                lr_init=0.02, beta_init=0.003, beta_min=0.0001,
                momentum=0.93, max_clip=0.3
            )
            update_best(a_fin)

    print(f"Phase 6 done: score={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
