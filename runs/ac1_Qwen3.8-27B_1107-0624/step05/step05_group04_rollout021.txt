    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]

    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return cm, grad_z


def _obj_grad_weighted(z, top_k=5):
    """Compute weighted peak objective focusing on top-k peaks."""
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0, np.zeros(n)

    sorted_idx = np.argsort(c)[::-1]
    top_indices = sorted_idx[:top_k]
    top_vals = c[top_indices]

    # Exponential weighting on top peaks
    weights = np.exp(10.0 * (top_vals - top_vals[0]) / max(cm, 1e-15))
    weights = weights / weights.sum()

    w = np.zeros_like(c)
    w[top_indices] = weights

    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]

    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return float(np.dot(weights, top_vals)), grad_z


def _adam_update(z, grad, m, v, m_hat, v_hat, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update step."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * (grad * grad)
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v, m_hat, v_hat


def _adaptive_beta(c, min_beta=1e-10):
    """Adaptively choose beta based on peak distribution."""
    cm = float(c.max())
    if cm < 1e-15:
        return min_beta
    sorted_c = np.sort(c)[::-1]
    k = min(10, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.02 * cm:
        return max(min_beta, 0.01 * cm)
    elif gap > 0.3 * cm:
        return max(min_beta, 0.05 * cm)
    else:
        return max(min_beta, 0.2 * gap)


def _make_profiles(n):
    """Generate diverse initialization profiles."""
    t = np.linspace(0, 1, n)
    profiles = []
    # Linear ramps with different rates
    for rate in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = 1.0 + rate * t
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Exponential profiles
    for rate in [1.0, 2.0, 3.0, 5.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    # Sigmoid profiles
    for k in [5, 10, 20]:
        a = 1.0 / (1.0 + np.exp(-k * (t - 0.3)))
        profiles.append(a / a.sum())
    # Quadratic
    for c in [0.5, 1.0, 2.0]:
        a = 1.0 + 1.5 * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-10)
        profiles.append(a / a.sum())
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = t ** p + 1e-10
        profiles.append(a / a.sum())
    return profiles


def _upsample(a, target_n):
    """Upsample a sequence to target length using interpolation."""
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _run_adam_optimize(a_init, deadline, rng, lr_init=0.05, verbose=True):
    """Main Adam optimization with adaptive beta and Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    beta = _adaptive_beta(c)
    min_beta = 1e-8 * float(c.max())

    m = np.zeros(n)
    v = np.zeros(n)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    lr = lr_init

    patience = 0
    patience_limit = 50000

    last_best_iter = 0

    while time.monotonic() < deadline:
        t += 1
        a = _softmax_fn(z)

        # Adaptive beta: slowly decrease
        if t % 500 == 0:
            c_check = _full_convolve(a, a)
            beta_new = _adaptive_beta(c_check, min_beta)
            beta = max(min_beta, 0.95 * beta_new)

        # Choose method based on beta
        if beta < 1e-6 * float(_full_convolve(a, a).max()):
            obj, grad = _obj_grad_hard(z)
            # Add small smoothing to avoid getting stuck
            obj_s, grad_s = _obj_grad_smooth(z, 1e-4 * float(_full_convolve(a, a).max()))
            grad = 0.7 * grad + 0.3 * grad_s
        elif beta < 0.01:
            obj, grad = _obj_grad_weighted(z, top_k=5)
            # Blend with smooth gradient
            obj_s, grad_s = _obj_grad_smooth(z, beta)
            grad = 0.5 * grad + 0.5 * grad_s
        else:
            obj, grad = _obj_grad_smooth(z, beta)

        if not np.all(np.isfinite(grad)):
            break

        # Nesterov: look ahead
        z_lookahead = z - lr * (m_hat / (np.sqrt(v_hat) + 1e-8)) if t > 0 else z
        if beta < 0.01:
            _, grad_nest = _obj_grad_smooth(z_lookahead, max(beta, 0.01))
            grad = grad_nest

        z, m, v, m_hat, v_hat = _adam_update(z, grad, m, v, m_hat, v_hat, t, lr)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        # Learning rate schedule
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.95)

        # Check score periodically
        if t % 1000 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1000

        # Restart if stuck
        if patience >= patience_limit:
            if verbose:
                print(f"    Restart at iter {t}, score={best_score:.10f}")
            patience = 0
            # Structured perturbation
            noise_scale = 0.05
            z = z + noise_scale * rng.normal(size=n)
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            # Also try a "shape" perturbation
            if rng.random() < 0.5:
                freq = rng.integers(2, 8)
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.02 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                z = z + pert
                z = z - z.mean()
            lr = lr_init * 0.5

        # Final convergence check
        if t > 50000 and t % 10000 == 0:
            lr = max(0.0005, lr * 0.8)

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coordinate_descent(a_init, deadline, rng, max_rounds=5000):
    """Coordinate descent: adjust individual elements to reduce peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    # Precompute which elements affect which peaks
    c = _full_convolve(best_a, best_a)
    cm = float(c.max())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        peak_idx = int(np.argmax(c))

        # Find top contributors to this peak
        valid_i = np.arange(max(0, peak_idx - n + 1), min(n, peak_idx + 1))
        if len(valid_i) == 0:
            continue
        prod = a_cur[valid_i] * a_cur[peak_idx - valid_i]
        top_contributors = valid_i[np.argsort(prod)[::-1][:10]]

        improved = False
        for idx in top_contributors:
            # Try reducing this element
            for delta in [0.001, 0.003, 0.01, 0.03]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing small elements (to spread out)
            a_cur = _softmax_fn(z)
            small_idx = np.argsort(a_cur)[:5]
            for idx in small_idx:
                for delta in [0.003, 0.01, 0.03]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rnd > 50:
            # Try random coordinate adjustments
            for _ in range(20):
                idx = rng.integers(0, n)
                direction = rng.choice([-1, 1])
                delta = rng.uniform(0.001, 0.02)
                z_trial = z.copy()
                z_trial[idx] += direction * delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if not improved and rnd > 200:
                break

    return best_a, best_score


def _fine_adam(a_init, deadline, lr=0.001, max_iter=100000):
    """Fine Adam optimization with very small learning rate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0

    c = _full_convolve(a_init, a_init)
    beta = max(1e-6, 0.001 * float(c.max()))

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax_fn(z)
        obj, grad = _obj_grad_smooth(z, beta)

        if not np.all(np.isfinite(grad)):
            break

        z, m, v, m_hat, v_hat = _adam_update(z, grad, m, v, m_hat, v_hat, t, lr)
        z = np.clip(z, -50, 50)
        z = z - z.mean()

        if it % 2000 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()

    a_final = _softmax_fn(z)
    sc_final = _score_norm_fn(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_profiles(4096)[4]  # Linear ramp
        update_best(a0, _score_norm_fn(a0), "default_ramp")
        best_seq = a0.copy()
        best_score = _score_norm_fn(a0)

    n_best = len(best_seq)

    # Phase 1: Multi-start at current size (120s)
    phase1_end = t_start + 120.0
    print(f"\nPhase 1: Multi-start optimization (deadline t={phase1_end - t_start:.0f}s)")

    profiles = _make_profiles(n_best)
    # Add perturbed parent
    for scale in [0.01, 0.05, 0.1]:
        pert = best_seq * (1.0 + scale * rng.normal(size=n_best))
        pert = np.maximum(pert, 1e-15)
        profiles.append(pert / pert.sum())

    best_p1 = best_seq.copy()
    best_sc_p1 = best_score

    for idx, prof in enumerate(profiles[:12]):
        if time.monotonic() > phase1_end - 10:
            break
        sc = _score_norm_fn(prof)
        if sc >= best_sc_p1 * 1.05:
            continue  # Skip obviously bad starts
        alloc_time = 8.0
        dl = min(phase1_end - 5, time.monotonic() + alloc_time)
        a_opt, sc_opt = _run_adam_optimize(prof, dl, rng, lr_init=0.03, verbose=False)
        if sc_opt < best_sc_p1:
            best_sc_p1 = sc_opt
            best_p1 = a_opt.copy()
            print(f"    Profile {idx}: score={sc_opt:.10f}")

    update_best(best_p1, best_sc_p1, "phase1_multistart")

    # Phase 2: Main Adam optimization (800s)
    phase2_end = t_start + 920.0
    print(f"\nPhase 2: Main Adam optimization (deadline t={phase2_end - t_start:.0f}s)")

    if time.monotonic() < phase2_end - 30:
        # Run main optimization from best
        dl_main = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.8)
        a_main, sc_main = _run_adam_optimize(best_seq, dl_main, rng, lr_init=0.05, verbose=True)
        update_best(a_main, sc_main, "phase2_main")

        # Try from perturbed best
        if time.monotonic() < phase2_end - 60:
            for attempt in range(3):
                if time.monotonic() > phase2_end - 30:
                    break
                z = np.log(np.maximum(best_seq, 1e-15))
                z = z - z.mean()
                # Structured perturbation
                freq = rng.integers(1, 6)
                phase = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.02, 0.08)
                pert = amp * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                z_pert = z + pert + 0.02 * rng.normal(size=n_best)
                z_pert = np.clip(z_pert, -50, 50)
                z_pert = z_pert - z_pert.mean()
                a_pert = _softmax_fn(z_pert)
                dl_sub = min(phase2_end - 10 - attempt * 10, time.monotonic() + 60.0)
                a_sub, sc_sub = _run_adam_optimize(a_pert, dl_sub, rng, lr_init=0.03, verbose=False)
                update_best(a_sub, sc_sub, f"phase2_perturb{attempt}")

    # Phase 3: Fine-tuning (400s)
    phase3_end = t_start + 1320.0
    print(f"\nPhase 3: Fine-tuning (deadline t={phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 20 and best_seq is not None:
        # Fine Adam
        dl_fine = min(phase3_end - 100, time.monotonic() + (phase3_end - time.monotonic()) * 0.4)
        a_fine, sc_fine = _fine_adam(best_seq, dl_fine, lr=0.002)
        update_best(a_fine, sc_fine, "phase3_fine_adam")

        # Coordinate descent
        if time.monotonic() < phase3_end - 50:
            dl_cd = min(phase3_end - 20, time.monotonic() + 80.0)
            a_cd, sc_cd = _coordinate_descent(best_seq, dl_cd, rng)
            update_best(a_cd, sc_cd, "phase3_coord_desc")

        # Very fine Adam
        if time.monotonic() < phase3_end - 10:
            dl_vf = min(phase3_end - 5, time.monotonic() + 40.0)
            a_vf, sc_vf = _fine_adam(best_seq, dl_vf, lr=0.0005)
            update_best(a_vf, sc_vf, "phase3_very_fine")

    # Phase 4: Size exploration (300s)
    phase4_end = deadline
    print(f"\nPhase 4: Size exploration (deadline t={phase4_end - t_start:.0f}s)")

    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        target_sizes = [5120, 6144, 8192]
        for target_n in target_sizes:
            if time.monotonic() > phase4_end - 40:
                break
            if target_n < n_best:
                continue
            a_up = _upsample(best_seq, target_n)
            sc_up = _score_norm_fn(a_up)
            print(f"    Upsampled to n={target_n}: score={sc_up:.10f}")

            # Quick optimization at this size
            dl_size = min(phase4_end - 20, time.monotonic() + 60.0)
            a_sz, sc_sz = _run_adam_optimize(a_up, dl_size, rng, lr_init=0.02, verbose=False)
            update_best(a_sz, sc_sz, f"size_{target_n}")

            # Fine-tune at this size
            if time.monotonic() < phase4_end - 20:
                dl_sz_fine = min(phase4_end - 10, time.monotonic() + 30.0)
                a_sz2, sc_sz2 = _fine_adam(best_seq if sc_sz2 < sc_sz else a_up, dl_sz_fine, lr=0.001)
                # Only update if we're at the right size
                if len(a_sz2) == target_n:
                    update_best(a_sz2, sc_sz2, f"size_{target_n}_fine")

    # Phase 5: Final polish (remaining time)
    print(f"\nPhase 5: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())

        # One more round of coordinate descent
        dl_final_cd = min(deadline - 3, time.monotonic() + 20.0)
        a_final_cd, sc_final_cd = _coordinate_descent(best_seq, dl_final_cd, rng, max_rounds=2000)
        update_best(a_final_cd, sc_final_cd, "final_cd")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_profiles(256)[3]

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        # Verify with evaluate_sequence if available
        try:
            eval_score = evaluate_sequence([float(x) for x in final_seq])
            print(f"Verified with evaluate_sequence: {eval_score:.10f}")
        except Exception:
            pass
        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_profiles(256)[3]
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
