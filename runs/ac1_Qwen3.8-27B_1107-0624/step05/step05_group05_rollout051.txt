"""
Soft-max Adam optimizer with multi-beta annealing, multiple sequence lengths,
and perturbation-based restarts for minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _adam_optimize(a_init, deadline, beta_start, beta_end, total_iters, lr_start, lr_end, wd=0.0):
    """
    Adam optimization in log-space with soft-max objective and cosine LR schedule.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while t < total_iters:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Cosine annealing on beta
        progress = t / total_iters
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Soft-max gradient
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient of soft-max w.r.t. a
        # d(softmax-beta)/da[m] = (1/beta) * sum_k w[k] * 2*a[k-m] (for valid k-m)
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n-1:2*n-1][::-1] / beta
        
        # Mirror descent gradient in z-space
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        
        # Cosine LR
        lr = lr_start + 0.5 * (lr_end - lr_start) * (1 + np.cos(np.pi * progress))
        
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Gradient clipping
        update_norm = np.linalg.norm(update)
        max_norm = 1.0
        if update_norm > max_norm:
            update = update * (max_norm / update_norm)
        
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if t % 500 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
    
    # Final check
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

"""
Multi-phase optimizer: soft-max Adam with adaptive beta annealing, gradient-at-max
regularization, perturbation restarts, and multi-length exploration for minimizing
2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad(a, c, beta, alpha_exact):
    """Compute gradient of soft-max objective in a-space.
    Mixes the smooth soft-max gradient with the exact subgradient at the max."""
    n = len(a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / max(beta, 1e-10)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # Soft-max gradient: (2/beta) * cross_correlation(w, a)
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad_smooth = 2.0 * conv_aw[n - 1:2 * n - 1][::-1] / beta

    # Exact subgradient at the max index
    k_star = int(np.argmax(c))
    grad_exact = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    for j in range(j_start, j_end + 1):
        grad_exact[j] += 2.0 * a[k_star - j]

    grad = (1.0 - alpha_exact) * grad_smooth + alpha_exact * grad_exact
    return grad, w


def _adam_optimize(a_init, deadline, beta_start, beta_end, total_iters,
                   lr_start, lr_end, alpha_exact=0.3, clip_norm=0.8):
    """Adam optimization in log-space with soft-max + exact gradient mixing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    while t < total_iters:
        if time.monotonic() > deadline:
            break
        t += 1

        a = _softmax(z)
        c = _full_convolve(a, a)

        # Adaptive beta: exponential decay
        progress = t / total_iters
        beta = beta_start * (beta_end / beta_start) ** progress

        # Also increase alpha_exact as beta decreases (transition from smooth to sharp)
        alpha = min(1.0, alpha_exact * (1.0 + 2.0 * progress))
        alpha = min(alpha, 0.8)  # cap at 0.8 to keep some smoothness

        grad, w = _compute_grad(a, c, beta, alpha)

        # Mirror descent: gradient in z-space
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)

        if not np.all(np.isfinite(grad_z)):
            break

        # Adam
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)

        # Cosine LR with warmup
        warmup_frac = 0.05
        if progress < warmup_frac:
            lr = lr_start * (progress / warmup_frac)
        else:
            adj = (progress - warmup_frac) / (1.0 - warmup_frac)
            lr = lr_start + (lr_end - lr_start) * 0.5 * (1.0 + np.cos(np.pi * adj))

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Periodic evaluation
        if t % 1000 == 0 or t == total_iters:
            a_check = _softmax(z)
            sc = _score(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _adam_optimize_multi(a_init, deadline, max_cb):
    """Run Adam at multiple beta levels, each starting from the best so far."""
    current = a_init.copy()
    best_a = a_init.copy()
    best_score = _score(a_init)

    # Beta levels (as fractions of max_cb)
    beta_levels = [
        (0.05, 0.01, 0.03, 0.005, 0.5),
        (0.01, 0.003, 0.015, 0.002, 0.4),
        (0.003, 0.0008, 0.01, 0.001, 0.35),
        (0.0008, 0.0002, 0.005, 0.0005, 0.3),
        (0.0002, 0.00005, 0.002, 0.0002, 0.25),
        (0.00005, 0.00001, 0.001, 0.0001, 0.2),
    ]

    for bi, (bs, be, lr_s, lr_e, alpha) in enumerate(beta_levels):
        if time.monotonic() > deadline:
            break
        remaining_frac = 1.0 - bi / len(beta_levels)
        est_iters = int(200000 * remaining_frac)

        a_out, sc_out = _adam_optimize(
            current, deadline,
            beta_start=bs * max_cb, beta_end=be * max_cb,
            total_iters=est_iters,
            lr_start=lr_s, lr_end=lr_e,
            alpha_exact=alpha
        )
        full_sc = _score(a_out)
        print(f"    Beta level {bi}: bs={bs}, be={be}, score={full_sc:.10f}, t={time.monotonic()-t_start_global:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_a = a_out.copy()

    return best_a, best_score


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _interpolate(a, n_new):
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, a)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()
