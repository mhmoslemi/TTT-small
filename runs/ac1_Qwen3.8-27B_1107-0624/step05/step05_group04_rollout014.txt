"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses: (1) multi-scale L-BFGS with log-sum-exp, (2) top-k gradient descent,
(3) direct peak equalization with line search, (4) final polish.

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _topk_obj(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if k >= len(c):
        return float(c.mean())
    idx = np.argpartition(c, -k)[-k:]
    return float(np.mean(c[idx]))


def _topk_grad(z, k, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    if k >= len(c):
        k = len(c)
    idx = np.argpartition(c, -k)[-k:]
    a_rev = a[::-1]
    v = np.zeros(2 * n - 1)
    v[idx] = 1.0
    w_conv = _full_convolve(a_rev, v)
    h = w_conv[n - 1:2 * n - 1] * (2.0 / k)
    offset = float(np.dot(a, h))
    grad_z = a * (h - offset)
    return grad_z


def _adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(20, len(sorted_c) - 1)

    elif gap > 0.5 * cm:
        return max(1e-10, 0.01 * cm)
    else:

def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _adam_topk(a_init, deadline, rng, lr_init=0.05, max_iter=50000):

        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a = _softmax(z)

        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        if t % 2000 == 0:
            lr = max(0.005, lr * 0.85)

        if t % 5000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except NameError:
        parent = None
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(256, 1.5)
        update_best(a0, _score_norm(a0), "default_ramp")

    # --- Phase 1: Multi-scale from scratch (25% of time) ---
    phase1_end = t_start + (budget_s - 30.0) * 0.25
    print(f"\nPhase 1: Multi-scale optimization")

    if parent is None or time.monotonic() < phase1_end:
        n_start = 128
        profiles = []
        for s in [1.0, 1.5, 2.0, 3.0]:
            profiles.append(_make_ramp(n_start, s))
        # Add exponential
        t_arr = np.linspace(0, 1, n_start)
        for rate in [2.0, 4.0]:
            a_exp = np.exp(rate * t_arr)
            profiles.append(a_exp / a_exp.sum())

        a_cur = profiles[0]
        sc_cur = _score_norm(a_cur)
        for prof in profiles:
            sc = _score_norm(prof)
            if sc < sc_cur:
                sc_cur = sc
                a_cur = prof.copy()

        quick_dl = min(phase1_end, t_start + 15.0)
        a_cur, sc_cur = _run_lbfgs_lse(a_cur, quick_dl, rng, n_betas=6)

        scales = [256, 512, 1024, 2048, 4096]
        for target_n in scales:
            if time.monotonic() > phase1_end - 5:
                break
            a_target = _upsample(a_cur, target_n)
            alloc = min(30.0, (phase1_end - time.monotonic()) * 0.7)
            if alloc < 5:
                break
            opt_dl = time.monotonic() + alloc
            a_opt, sc_opt = _run_lbfgs_lse(a_target, opt_dl, rng, n_betas=10)
            if sc_opt < sc_cur:
                a_cur = a_opt
                sc_cur = sc_opt
            print(f"  n={target_n}: score={sc_cur:.10f}")

        if sc_cur < best_score:
            update_best(a_cur, sc_cur, "multi_scale")

    print(f"\n  Phase 1 done: best_score={best_score:.10f}")

    # --- Phase 2: L-BFGS log-sum-exp on best (25% of time) ---
    phase2_end = t_start + (budget_s - 30.0) * 0.50
    print(f"\nPhase 2: L-BFGS LSE refinement")

    if best_seq is not None and time.monotonic() < phase2_end:
        dl2 = min(phase2_end - 2, time.monotonic() + (phase2_end - time.monotonic()))
        a_p2, sc_p2 = _run_lbfgs_lse(best_seq, dl2, rng, n_betas=20)
        update_best(a_p2, sc_p2, "phase2_lse")

    print(f"\n  Phase 2 done: best_score={best_score:.10f}")

    # --- Phase 3: Top-k L-BFGS (20% of time) ---
    phase3_end = t_start + (budget_s - 30.0) * 0.70
    print(f"\nPhase 3: Top-k L-BFGS")

    if best_seq is not None and time.monotonic() < phase3_end:
        dl3 = min(phase3_end - 2, time.monotonic() + (phase3_end - time.monotonic()))
        a_p3, sc_p3 = _run_topk_lbfgs(best_seq, dl3, rng)
        update_best(a_p3, sc_p3, "phase3_topk")

    print(f"\n  Phase 3 done: best_score={best_score:.10f}")

    # --- Phase 4: Peak equalization (20% of time) ---
    phase4_end = t_start + (budget_s - 30.0) * 0.90
    print(f"\nPhase 4: Peak equalization")

    if best_seq is not None and time.monotonic() < phase4_end:
        dl4 = min(phase4_end - 5, time.monotonic() + (phase4_end - time.monotonic()) * 0.6)
        a_p4, sc_p4 = _peak_equalize(best_seq, dl4, rng)
        update_best(a_p4, sc_p4, "phase4_peak_eq")

        if time.monotonic() < phase4_end - 5:
            dl4b = min(phase4_end - 2, time.monotonic() + 60.0)
            a_p4b, sc_p4b = _coordinate_descent_peaks(best_seq, dl4b, rng)
            update_best(a_p4b, sc_p4b, "phase4_coord")

    print(f"\n  Phase 4 done: best_score={best_score:.10f}")

    # --- Phase 5: Final polish (10% of time) ---
    print(f"\nPhase 5: Final polish")

    if best_seq is not None and time.monotonic() < deadline:
        # Tiny beta L-BFGS
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = max(1e-7, 1e-5 * max_c_final)

        dl5 = min(deadline - 10, time.monotonic() + 60.0)
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny, n_final),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(np.clip(result.x, -60, 60) - np.log(np.maximum(best_seq, 1e-15)).mean())
                a_new = _softmax(np.clip(result.x, -60, 60))
                sc_new = _score_norm(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    update_best(a_new, sc_new, "phase5_lse_tiny")
        except Exception as ex:
            print(f"  L-BFGS tiny failed: {ex}")

        # Final peak equalization
        if time.monotonic() < deadline - 10:
            dl5b = min(deadline - 3, time.monotonic() + 45.0)
            a_p5, sc_p5 = _peak_equalize(best_seq, dl5b, rng)
            update_best(a_p5, sc_p5, "phase5_peak")

        # Adam topk
        if time.monotonic() < deadline - 5:
            dl5c = min(deadline - 2, time.monotonic() + 30.0)
            a_p5b, sc_p5b = _adam_topk(best_seq, dl5c, rng, lr_init=0.01, max_iter=20000)
            update_best(a_p5b, sc_p5b, "phase5_adam")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Try evaluate_sequence for validation
        try:
            eval_score = evaluate_sequence([float(x) for x in final_seq])
            print(f"  evaluate_sequence confirms: {eval_score:.10f}")
        except Exception:
            pass

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
