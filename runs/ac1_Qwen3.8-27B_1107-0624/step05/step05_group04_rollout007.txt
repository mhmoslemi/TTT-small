def next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return e / s

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def score_norm_fn(a):
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())


def upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def logsumexp_obj(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def logsumexp_grad(z, beta, n):
    a = softmax_fn(z)
    c = full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def adaptive_beta(c):
    cm = float(c.max())
    if cm < 1e-15:
        return 1e-10
    sorted_c = np.sort(c)[::-1]
    k = min(30, len(sorted_c) - 1)
    gap = sorted_c[0] - sorted_c[k]
    if gap < 0.02 * cm:
        return max(1e-10, 0.02 * cm)
    elif gap > 0.3 * cm:
        return max(1e-10, 0.05 * cm)
    else:
        return max(1e-10, 0.2 * gap)


def run_lbfgs_continuation(a_init, deadline, rng, n_betas=12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)

    c0 = full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_start = 0.5 * max_c0
    beta_end = max(1e-7 * max_c0, 1e-10)
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget

        if beta > 0.1 * max_c0:
            maxiter = 300
        elif beta > 0.01 * max_c0:
            maxiter = 800
        else:
            maxiter = 2500
        maxfun = maxiter * 5

        try:
            result = minimize(
                logsumexp_obj, z.copy(), args=(beta, n),
                method='L-BFGS-B', jac=logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxls': 40, 'maxfun': maxfun}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = softmax_fn(z_new)
                s_new = score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def run_adam(a_init, deadline, rng, lr_init=0.08, max_iter=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t += 1
        a = softmax_fn(z)
        c = full_convolve(a, a)
        beta = adaptive_beta(c)

        cm = float(c.max())
        if cm < 1e-30:
            break

        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()

        a_rev = a[::-1]
        w_conv = full_convolve(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]

        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 50 == 0:
            a_check = softmax_fn(z)
            sc = score_norm_fn(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

        if t % 5000 == 0:
            lr = max(0.003, lr * 0.85)

        if t % 30000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def peak_peel(a_init, deadline, rng, max_rounds=500):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = softmax_fn(z)
        c = full_convolve(a_cur, a_cur)

        pk = int(np.argmax(c))
        valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
        if len(valid_i) == 0:
            break
        prod = a_cur[valid_i] * a_cur[pk - valid_i]
        top_idx = valid_i[np.argsort(prod)[::-1][:8]]

        improved = False
        for idx in top_idx:
            for delta in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = softmax_fn(z_trial)
                sc = score_norm_fn(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            small_idx = np.argsort(a_cur)[:6]
            for idx in small_idx:
                for delta in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = softmax_fn(z_trial)
                    sc = score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and rnd > 30:
            z = z + 0.04 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def multi_peak_peel(a_init, deadline, rng, max_rounds=300):
    """Target multiple peaks simultaneously for better balance."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = score_norm_fn(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        a_cur = softmax_fn(z)
        c = full_convolve(a_cur, a_cur)
        cm = float(c.max())

        # Find top 3 peaks
        top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for pk in top_peaks:
            valid_i = np.arange(max(0, int(pk) - n + 1), min(n, int(pk) + 1))
            if len(valid_i) == 0:
                continue
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            top_idx = valid_i[np.argsort(prod)[::-1][:4]]

            for idx in top_idx:
                for delta in [0.003, 0.008, 0.015, 0.03]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = softmax_fn(z_trial)
                    sc = score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            small_idx = np.argsort(a_cur)[:4]
            for idx in small_idx:
                for delta in [0.003, 0.008, 0.015, 0.03]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = softmax_fn(z_trial)
                    sc = score_norm_fn(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break

    return best_a, best_score


def make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        profiles.append(make_ramp(n, s))
    a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5)))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0]:
        a = np.exp(rate * t)
        profiles.append(a / a.sum())
    for p in [1.5, 2.0, 2.5, 3.0, 4.0]:
        a = t ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
        a = (1 - t) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Skewed shapes
    a = t + 0.5 * np.sin(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = 1.0 + np.sin(2 * np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    n_target = 4096

    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None and len(parent) == n_target:
        sc_p = score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score_norm={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    elif parent is not None:
        parent_resized = upsample(parent, n_target)
        sc_p = score_norm_fn(parent_resized)
        print(f"Parent resized to n={n_target}: score_norm={sc_p:.10f}")
        update_best(parent_resized, sc_p, "parent_resized")
    else:
        a0 = make_ramp(n_target, 1.5)
        update_best(a0, score_norm_fn(a0), "default")

    # Phase 1: Multi-scale from scratch (28% of time)
    phase1_end = t_start + (deadline - t_start) * 0.28
    print(f"\nPhase 1: Multi-scale (deadline t={phase1_end - t_start:.0f}s)")

    n_start = 128
    profiles = make_profiles(n_start)
    best_prof = None
    best_prof_sc = np.inf
    for prof in profiles:
        sc = score_norm_fn(prof)
        if sc < best_prof_sc:
            best_prof_sc = sc
            best_prof = prof.copy()
    print(f"  Best initial profile (n={n_start}): score_norm={best_prof_sc:.10f}")

    dl = min(phase1_end, t_start + 12.0)
    a_cur, sc_cur = run_lbfgs_continuation(best_prof, dl, rng, n_betas=8)
    print(f"  After n={n_start}: score_norm={sc_cur:.10f}")

    scales = [256, 512, 1024, 2048, 4096]
    for target_n in scales:
        if time.monotonic() > phase1_end - 5:
            break
        a_up = upsample(a_cur, target_n)
        sc_up = score_norm_fn(a_up)
        alloc = min(20 + 8 * (target_n // 256), 100)
        opt_dl = min(phase1_end - 3, time.monotonic() + alloc)
        a_opt, sc_opt = run_lbfgs_continuation(a_up, opt_dl, rng, n_betas=10)
        print(f"  n={target_n}: up={sc_up:.8f} opt={sc_opt:.8f}")
        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")

    # Phase 2: L-BFGS refinement from best (24% of time)
    phase2_end = t_start + (deadline - t_start) * 0.52
    print(f"\nPhase 2: L-BFGS refinement (deadline t={phase2_end - t_start:.0f}s)")

    if best_seq is not None and time.monotonic() < phase2_end - 10:
        n_best = len(best_seq)
        opt_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()))
        a_p2, sc_p2 = run_lbfgs_continuation(best_seq, opt_dl, rng, n_betas=15)
        update_best(a_p2, sc_p2, "phase2_lbfgs")

        if time.monotonic() < phase2_end - 20:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.04 * rng.normal(size=n_best)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_pert = softmax_fn(z)
            opt_dl2 = min(phase2_end - 3, time.monotonic() + 50.0)
            a_p2b, sc_p2b = run_lbfgs_continuation(a_pert, opt_dl2, rng, n_betas=8)
            update_best(a_p2b, sc_p2b, "phase2_restart")

    # Phase 3: Adam + Peak peeling (32% of time)
    phase3_end = t_start + (deadline - t_start) * 0.84
    print(f"\nPhase 3: Adam + Peak peeling (deadline t={phase3_end - t_start:.0f}s)")

    if best_seq is not None and time.monotonic() < phase3_end - 10:
        gd_dl = min(phase3_end - 80, time.monotonic() + (phase3_end - time.monotonic()) * 0.45)
        a_gd, sc_gd = run_adam(best_seq, gd_dl, rng, lr_init=0.08)
        update_best(a_gd, sc_gd, "adam_coarse")

        if time.monotonic() < phase3_end - 50:
            pp_dl = min(phase3_end - 30, time.monotonic() + 40.0)
            a_pp, sc_pp = peak_peel(best_seq, pp_dl, rng, max_rounds=300)
            update_best(a_pp, sc_pp, "peak_peel")

        if time.monotonic() < phase3_end - 30:
            mp_dl = min(phase3_end - 15, time.monotonic() + 30.0)
            a_mp, sc_mp = multi_peak_peel(best_seq, mp_dl, rng, max_rounds=200)
            update_best(a_mp, sc_mp, "multi_peak")

        if time.monotonic() < phase3_end - 10:
            gd_dl2 = min(phase3_end - 3, time.monotonic() + 30.0)
            a_gd2, sc_gd2 = run_adam(best_seq, gd_dl2, rng, lr_init=0.02)
            update_best(a_gd2, sc_gd2, "adam_fine")

    # Phase 4: Final tiny-beta L-BFGS + peak peeling (16% of time)
    print(f"\nPhase 4: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_final = len(best_seq)
        c_final = full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = max(1e-6 * max_c_final, 1e-10)

        try:
            result = minimize(
                logsumexp_obj, z, args=(beta_tiny, n_final),
                method='L-BFGS-B', jac=logsumexp_grad,
                options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': 25000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = softmax_fn(np.clip(result.x, -80, 80))
                sc_new = score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_seq = a_new.copy()
                    print(f"  Tiny-beta L-BFGS: {best_score:.10f}")
        except Exception as ex:
            print(f"  Tiny-beta failed: {ex}")

    # Final peak peeling with remaining time
    if best_seq is not None and time.monotonic() < deadline - 3:
        pp_dl = deadline - 2
        a_pp, sc_pp = peak_peel(best_seq, pp_dl, rng, max_rounds=200)
        update_best(a_pp, sc_pp, "final_peel")

    # Validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        result_list = [float(x) for x in final_seq]
        sc_final = evaluate_sequence(result_list)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return result_list

    fallback = make_ramp(256, 1.5)
    fl = [float(x) for x in fallback]
    print(f"FALLBACK: score={evaluate_sequence(fl):.10f}")
