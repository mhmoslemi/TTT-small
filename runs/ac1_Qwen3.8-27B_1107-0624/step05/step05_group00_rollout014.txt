        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        bc1 = 1.0 - b1 ** t_eff
        bc2 = 1.0 - b2 ** t_eff
        m_hat = m_buf / bc1
        v_hat = v_buf / bc2

        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Track best
        if t % 25 == 0:
            a_chk = _softmax_safe(z)
            sc = _score_norm(a_chk)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_chk.copy()

    a_final = _softmax_safe(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _fista_coarse(a_init, beta, deadline, max_iter=50000):
    """FISTA on simplex with softmax-smoothed max for coarse optimization."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    L = 8.0 * n  # Lipschitz constant estimate
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        # Gradient
        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _basin_hop(a_init, deadline, num_hops=5, hop_sigma=0.02, rng=None):
    """Basin hopping: optimize, perturb, optimize, ..."""
    if rng is None:
        rng = np.random.default_rng()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    n = len(a_init)

    c_init = _full_convolve(a_init, a_init)
    max_c = float(c_init.max())

    time_per_hop = (deadline - time.monotonic()) / num_hops

    for hop in range(num_hops):
        if time.monotonic() >= deadline:
            break

        # Perturb in log-space
        z = np.log(np.maximum(a_init, 1e-15))
        z = z + hop_sigma * rng.standard_normal(n)
        z = z - z.mean()
        a_perturbed = _softmax_safe(z)

        # Optimize from perturbed point
        hop_deadline = min(deadline, time.monotonic() + time_per_hop)
        beta_start = 0.005 * max_c
        beta_end = 0.0002 * max_c
        a_opt, sc_opt = _adam_logspace(
            a_perturbed, hop_deadline,
            beta_start, beta_end,
            lr_init=0.015, lr_final=0.001,
            target_ipr=12.0, clip_norm=0.5,
            reset_period=3000, rng=rng
        )
        sc_full = _score_norm(a_opt)
        if sc_full < best_score:
            best_score = sc_full
            best_a = a_opt.copy()
            a_init = a_opt.copy()
            print(f"    Basin hop {hop}: improved to {best_score:.10f}")
        else:
            # Keep original as starting point for next hop
            pass

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = np.ones(n_main) / n_main
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"max_conv={max_c0:.10f}")

    a_work = a0.copy()

    # === Phase 1: Coarse FISTA (20s) ===
    print("Phase 1: FISTA coarse optimization")
    p1_end = min(deadline, t_start + 20.0)
    if p1_end > t_start + 2:
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())
        a_work, _ = _fista_coarse(a_work, 0.02 * max_cc, p1_end)

        sc = _score(a_work)
        if sc < best_score:
            best_score = sc
            best_seq = a_work.copy()
            print(f"  After FISTA: score={sc:.10f}")

    # === Phase 2: Main Adam optimization (55% of time) ===
    print("Phase 2: Adam log-space main optimization")
    p2_end = min(deadline, t_start + 0.55 * (budget_s - 20.0))
    c_w = _full_convolve(a_work, a_work)
    max_cw = float(c_w.max())

    a_opt, sc_opt = _adam_logspace(
        a_work, p2_end,
        beta_init=0.008 * max_cw,
        beta_final=0.00005 * max_cw,
        lr_init=0.018, lr_final=0.0005,
        target_ipr=12.0,
        clip_norm=0.5,
        reset_period=5000,
        noise_scale=0.0,
        rng=rng
    )
    sc_full = _score(a_opt)
    print(f"  After main Adam: score={sc_full:.10f}")
    if sc_full < best_score:
        best_score = sc_full
        best_seq = a_opt.copy()
        a_work = a_opt.copy()

    # === Phase 3: Basin hopping (20% of time) ===
    print("Phase 3: Basin hopping")
    p3_end = min(deadline, t_start + 0.75 * (budget_s - 20.0))
    if time.monotonic() < p3_end - 10:
        c_b = _full_convolve(a_work, a_work)
        max_cb = float(c_b.max())
        a_hopped, sc_hopped = _basin_hop(
            a_work, p3_end,
            num_hops=4, hop_sigma=0.015, rng=rng
        )
        sc_full2 = _score(a_hopped)
        print(f"  After basin hopping: score={sc_full2:.10f}")
        if sc_full2 < best_score:
            best_score = sc_full2
            best_seq = a_hopped.copy()
            a_work = a_hopped.copy()

    # === Phase 4: Try n=8192 (10% of time) ===
    print("Phase 4: n=8192 exploration")
    p4_end = min(deadline, t_start + 0.88 * (budget_s - 20.0))
    if time.monotonic() < p4_end - 15 and best_seq is not None:
        n_big = 8192
        a_big = _interp_profile(best_seq, n_big)

        if sc_big < best_score:
            best_score = sc_big
            best_seq = a_big.copy()

        if time.monotonic() < p4_end - 10:
            c_big = _full_convolve(a_big, a_big)
            max_cbig = float(c_big.max())
            a_big_opt, _ = _adam_logspace(
                a_big, min(p4_end - 5, time.monotonic() + 0.10 * (budget_s - 20.0)),
                beta_init=0.006 * max_cbig,
                beta_final=0.00005 * max_cbig,
                lr_init=0.008, lr_final=0.0003,
                target_ipr=10.0,
                clip_norm=0.4,
                reset_period=3000,
                rng=rng
            )
            sc_big_full = _score(a_big_opt)
            print(f"  Optimized n=8192: score={sc_big_full:.10f}")
            if sc_big_full < best_score:
                best_score = sc_big_full
                best_seq = a_big_opt.copy()

    # === Phase 5: Final ultra-fine polish (7% of time) ===
    print("Phase 5: Final polish")
    if best_seq is not None and time.monotonic() < deadline - 5:

            a_pol, deadline,
            beta_init=0.0003 * max_cp,
            beta_final=0.00001 * max_cp,
            lr_init=0.004, lr_final=0.0001,
            target_ipr=5.0,
            clip_norm=0.2,
            reset_period=2000,
            noise_scale=0.0,
            rng=rng
        )
        sc_pol = _score(a_pol_opt)
        print(f"  After final polish: score={sc_pol:.10f}")
        if sc_pol < best_score:
            best_score = sc_pol
            best_seq = a_pol_opt.copy()

    # === Final validation ===
    print(f"\nBest score: {best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)

            final = np.ones(256) / 256.0
        sc_final = _score(final)

        try:

    # Fallback
    fallback = np.ones(256) / 256.0
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
