         sum(a) = 1, a >= 0

This is a QCQP with quadratic constraints. If the active set is small (say 5-20 constraints), we can solve it efficiently using sequential quadratic programming (SQP) or by reformulating.

Actually, for a small active set, we can use the following approach:
- The KKT conditions require: sum of gradients of active C[k] (weighted by Lagrange multipliers) + lambda * 1 + mu = 0 (where mu is for the non-negativity)
- This is a system that can be solved iteratively

But this is complex to implement correctly. Let me stick with the log-sum-exp approach but make it more aggressive.

**Final approach:**

1. Use log-sum-exp with a very aggressive continuation (start at beta = 0.1 * max_C, decrease by factor 2 each round, go down to beta = 1e-6 * max_C)
2. At each beta level, run L-BFGS to convergence (or near convergence)
3. Periodically (every few iterations), check if the solution has "stabilized" (the top few C[k] values are well-separated)
4. Once stabilized, switch to a "direct max" approach: just optimize the max of the top 3-5 C[k] values
5. Use random restarts to avoid local minima
6. Try multiple n values

For the "direct max" approach, I'll use a subgradient method:
- Compute C, find top k indices
- Average the gradients over the active set
- Take a step in the negative gradient direction
- Project onto the simplex (a >= 0, sum(a) = 1)

Projection onto the simplex is O(n log n) and well-known.

def _next_pow2(n):

    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _lbfgs_run(z, beta, n, deadline, maxiter):
    def obj(zv):
        return _lse_obj(zv, beta, n)

    def grad(zv):
        return _lse_grad(zv, beta, n)

    try:
        result = minimize(
            obj, z, jac=grad, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                     'maxls': 40, 'maxfun': maxiter * 5}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            return result.x
    except Exception:
        pass
    return z


def _continuation(a_init, deadline, max_beta_frac=0.1, min_beta_frac=1e-7):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init)
    max_c0 = float(c0.max())

    n_betas = 50
    log_betas = np.linspace(np.log(max_beta_frac * max_c0),
                            np.log(min_beta_frac * max_c0), n_betas)

    for idx, log_b in enumerate(log_betas):
        if time.monotonic() > deadline:
            break
        beta = float(np.exp(log_b))

        remaining = n_betas - idx
        time_left = deadline - time.monotonic()
        time_per = time_left / max(1, remaining)
        time_per = min(time_per, 30.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(100, int(time_per * 25))

        z = _lbfgs_run(z, beta, n, level_dl, maxiter)
        a_new = _softmax(z)
        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if idx % 10 == 0:
            elapsed = time.monotonic() - (deadline - time_left)
            print(f"    cont iter {idx}/{n_betas}: beta={beta:.2e}, score={sc:.10f}")

    return best_a, best_score


def _subgradient_refine(a_init, deadline, top_k=20, n_steps=20000):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = 0.5 / (n ** 0.5)
    momentum = 0.85
    vel = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a)
        top_indices = np.argsort(c)[-top_k:]
        grad = np.zeros(n)
        for k in top_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * a[idx]
        grad /= len(top_indices)

        vel = momentum * vel - lr * grad
        a_new = a + vel
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        if step % 500 == 0:
            lr *= 0.92
        if step % 2000 == 0:
            top_k = min(40, top_k + 5)

    return best_a, best_score


def _coord_descent(a_init, deadline, max_rounds=100, n_trials=12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        cur_score = _score_norm(a)
        improved = False
        step_sizes = [0.05, 0.02, 0.01, 0.005, -0.005, -0.01, -0.02, -0.05,
                      0.1, -0.1, 0.002, -0.002]

        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = cur_score
            for delta in step_sizes:
                z[j] += delta
                a_trial = _softmax(z)
                sc = _score_norm(a_trial)
                z[j] -= delta
                if sc < best_local:
                    best_local = sc
                    best_delta = delta

            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                a = _softmax(z)
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    cur_score = sc
                    improved = True

        if not improved:
            break

    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []

    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    for rate in [0.5, 1.0, 2.0, 3.0, 5.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())

    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = np.maximum(1.0 - t, 1e-12) ** p
        profiles.append(a / a.sum())

    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())

    a = np.ones(n)
    profiles.append(a / n)

    a = np.abs(u) + 0.1
    profiles.append(a / a.sum())

    a = 1.0 / (0.1 + u ** 2)
    profiles.append(a / a.sum())

    # Two-bump asymmetric
    for center in [0.3, 0.4, 0.6, 0.7]:
        a = np.exp(-50 * (t - center) ** 2) + 0.1 * np.exp(-50 * (t - (1 - center)) ** 2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _perturb(a, rng, sigma=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    z = z + sigma * rng.normal(size=n)
    z = np.clip(z, -60, 60)
    z = z - z.mean()
    return _softmax(z)


def _block_optimize(a_init, deadline, block_size=3, n_rounds=200):
    """Optimize groups of coordinates together."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        cur_score = _score_norm(a)

        # Pick random blocks
        n_blocks = min(20, n // block_size)
        for _ in range(n_blocks):
            start = rng.integers(0, max(1, n - block_size))
            end = min(start + block_size, n)
            size = end - start

            # Try small perturbations in this block
            improved_block = False
            for trial in range(6):
                if size == 1:
                    dz = np.zeros(n)
                    dz[start] = rng.normal(0, 0.03)
                else:
                    dz = np.zeros(n)
                    dz[start:end] = rng.normal(0, 0.03, size)

                z_trial = z + dz
                a_trial = _softmax(z_trial)
                sc = _score_norm(a_trial)
                if sc < cur_score:
                    z = z_trial
                    cur_score = sc
                    a = a_trial
                    improved_block = True
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()

        if not improved_block:
            # Try larger step
            z = z + 0.1 * rng.normal(size=n)
            z = np.clip(z, -80, 80)
            z = z - z.mean()
            a = _softmax(z)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = _make_profiles(n_init)[15]  # exponential decay
        update_best(parent, _score_norm(parent), "default")

    # === Phase 1: Multi-scale quick scan (180s) ===
    phase1_end = min(deadline, t_start + 180.0)
    print(f"\n=== Phase 1: Multi-scale scan ===")

    n_values = [2048, 4096, 8192]
    n_best = len(parent)

    for n_try in n_values:
        if time.monotonic() > phase1_end:
            break
        if n_try == len(parent):
            a_cur = parent.copy()
        else:
            a_cur = _interp_profile(parent, n_try)
        sc = _score_norm(a_cur)
        update_best(a_cur, sc, f"scan_n{n_try}")

        # Quick optimization: a few L-BFGS iterations with moderate beta
        c_cur = _full_convolve(a_cur)
        beta = 0.05 * float(c_cur.max())
        z_cur = np.log(np.maximum(a_cur, 1e-15))
        z_cur = z_cur - z_cur.mean()
        sub_dl = min(phase1_end - 5, time.monotonic() + 30.0)
        z_cur = _lbfgs_run(z_cur, beta, n_try, sub_dl, 500)
        a_new = _softmax(z_cur)
        sc_new = _score_norm(a_new)
        update_best(a_new, sc_new, f"scan_opt_n{n_try}")

    print(f"  Phase 1 done: best={best_score:.10f}, n={len(best_seq)}")

    # === Phase 2: Aggressive continuation at best n (500s) ===
    phase2_end = min(deadline, t_start + 680.0)
    print(f"\n=== Phase 2: Continuation (n={len(best_seq)}) ===")

    n_curr = len(best_seq)
    a_start = best_seq.copy()

    # Main continuation run
    p_best, sc_best = _continuation(a_start, phase2_end, max_beta_frac=0.08, min_beta_frac=5e-8)
    update_best(p_best, sc_best, "continuation_main")

    # Perturbed restart
    if time.monotonic() < phase2_end - 60:
        a_pert = _perturb(best_seq, rng, sigma=0.03)
        sub_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
        p_best2, sc_best2 = _continuation(a_pert, sub_dl, max_beta_frac=0.03, min_beta_frac=5e-8)
        update_best(p_best2, sc_best2, "continuation_perturb")

    # Another restart with different sigma
    if time.monotonic() < phase2_end - 30:
        a_pert2 = _perturb(best_seq, rng, sigma=0.08)
        sub_dl2 = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.5)
        p_best3, sc_best3 = _continuation(a_pert2, sub_dl2, max_beta_frac=0.02, min_beta_frac=1e-7)
        update_best(p_best3, sc_best3, "continuation_perturb2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Subgradient refinement (300s) ===
    phase3_end = min(deadline, t_start + 980.0)
    print(f"\n=== Phase 3: Subgradient refinement ===")

    if time.monotonic() < phase3_end and best_seq is not None:
        a_sg = best_seq.copy()
        sub_dl = min(phase3_end - 10, time.monotonic() + 250.0)
        p_best, sc_best = _subgradient_refine(a_sg, sub_dl, top_k=20, n_steps=30000)
        update_best(p_best, sc_best, "subgradient")

    # === Phase 4: Block optimization + more continuation (300s) ===
    phase4_end = min(deadline, t_start + 1280.0)
    print(f"\n=== Phase 4: Block optimization ===")

    if time.monotonic() < phase4_end and best_seq is not None:
        n_block = len(best_seq)
        sub_dl = min(phase4_end - 60, time.monotonic() + 200.0)
        p_best, sc_best = _block_optimize(best_seq, sub_dl, block_size=4, n_rounds=500)
        update_best(p_best, sc_best, "block_opt")

    # Another continuation from the current best
    if time.monotonic() < phase4_end - 30 and best_seq is not None:
        a_cur = best_seq.copy()
        c_cur = _full_convolve(a_cur)
        beta_small = 0.005 * float(c_cur.max())
        z_cur = np.log(np.maximum(a_cur, 1e-15))
        z_cur = z_cur - z_cur.mean()
        sub_dl2 = min(phase4_end - 10, time.monotonic() + 100.0)
        z_cur = _lbfgs_run(z_cur, beta_small, len(a_cur), sub_dl2, 8000)
        a_new = _softmax(z_cur)
        sc_new = _score_norm(a_new)
        update_best(a_new, sc_new, "late_continuation")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Try different n values for final push (200s) ===
    phase5_end = min(deadline, t_start + 1480.0)
    print(f"\n=== Phase 5: Multi-n final push ===")

    n_parent_best = len(best_seq)
    for n_try in [2048, 4096, 8192]:
        if time.monotonic() > phase5_end:
            break
        if n_try == n_parent_best:
            continue
        a_try = _interp_profile(best_seq, n_try)
        sc_try = _score_norm(a_try)
        update_best(a_try, sc_try, f"interp_n{n_try}")

        # Quick optimize at this n
        if time.monotonic() < phase5_end - 30:
            c_try = _full_convolve(a_try)
            beta_try = 0.01 * float(c_try.max())
            z_try = np.log(np.maximum(a_try, 1e-15))
            z_try = z_try - z_try.mean()
            sub_dl = min(phase5_end - 10, time.monotonic() + 40.0)
            z_try = _lbfgs_run(z_try, beta_try, n_try, sub_dl, 3000)
            a_new = _softmax(z_try)
            sc_new = _score_norm(a_new)
            update_best(a_new, sc_new, f"opt_n{n_try}")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Coordinate descent polish (remaining time) ===
    print(f"\n=== Phase 6: Polish ===")

    if time.monotonic() < deadline and best_seq is not None:
        sub_dl = min(deadline - 10, time.monotonic() + (deadline - time.monotonic()) * 0.6)
        p_best, sc_best = _coord_descent(best_seq, sub_dl, max_rounds=200)
        update_best(p_best, sc_best, "coord_descent")

    # Final tiny-beta L-BFGS
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq)
        max_c_final = float(c_final.max())
        beta_ultra = 1e-6 * max_c_final
        z_final = np.log(np.maximum(best_seq, 1e-15))
        z_final = z_final - z_final.mean()
        sub_dl2 = min(deadline - 5, time.monotonic() + 40.0)
        z_final = _lbfgs_run(z_final, beta_ultra, n_final, sub_dl2, 10000)
        a_final = _softmax(z_final)
        sc_final = _score_norm(a_final)
        update_best(a_final, sc_final, "ultra_tiny_beta")

    # One more subgradient pass
    if time.monotonic() < deadline - 5 and best_seq is not None:
        sub_dl3 = min(deadline - 2, time.monotonic() + 30.0)
        p_best, sc_best = _subgradient_refine(best_seq, sub_dl3, top_k=15, n_steps=10000)
        update_best(p_best, sc_best, "final_subgradient")

    # Final validation and return
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_full(final_seq)
        print(f"Final: score={sc_final:.10f}, n={len(final_seq)}")

        # Verify with evaluator if reasonable size
        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_profiles(n_fb)[15]
    print(f"FALLBACK: n={n_fb}, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
