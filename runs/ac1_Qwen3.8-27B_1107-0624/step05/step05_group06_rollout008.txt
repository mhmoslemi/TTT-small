"""
Self-consistent field optimization for minimizing 2*n*max(self-convolution)/sum^2.
Combines power iteration on the convolution response operator with L-BFGS polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize

# Helper functions (top-level, no closures)

def next_pow2(n):
    """Next power of 2 >= n."""
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def softmax(z):
    """Numerically stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def compute_response(a, active_weights):
    """Compute the convolution response r_j = sum_k w_k * a_{k-j}.
    
    active_weights: array of length 2n-1 with weights for each lag.
    Returns: array of length n.
    """
    n = len(a)
    # r_j = sum_k w_k * a_{k-j} for valid (k-j) in [0, n-1]
    # Using: r = convolve(w_flipped, a)[n-1:2n-1][::-1]
    w_flipped = active_weights[::-1]
    conv = full_convolve(w_flipped, a)
    r = conv[n-1:2*n-1][::-1]
    return r

def scf_iteration(a, gamma, active_tol):
    """One self-consistent field iteration.
    
    a: current sequence (sum=1)
    gamma: concentration parameter
    active_tol: tolerance for active set
    """
    n = len(a)
    c = full_convolve(a, a)
    c_max = c.max()
    
    # Active set: lags within active_tol of max
    active_mask = c >= (c_max - active_tol * c_max)
    if not active_mask.any():
        active_mask[np.argmax(c)] = True
    
    # Uniform weights on active set
    w = np.zeros(2*n-1)
    w[active_mask] = 1.0 / active_mask.sum()
    
    # Compute response
    r = compute_response(a, w)
    
    # Update: a_new ∝ exp(-gamma * r)
    z = -gamma * r
    a_new = softmax(z)
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    
    return a_new, c_max, active_mask.sum()

    """Self-consistent field optimization with annealing."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = score(a)
    
    gamma = gamma_init
    n_active_total = max_iters
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Anneal gamma
        frac = i / n_active_total
        gamma_cur = gamma_init * (gamma_end / gamma_init) ** frac
        
        # Active tolerance also anneals
        tol_cur = active_tol * (1 + frac * 5)  # widen tolerance over time
        
        a_new, c_max, n_active = scf_iteration(a, gamma_cur, tol_cur)
        
        # Check improvement
        if i % 100 == 0:
            sc = score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        a = a_new
    
    # Final check
    sc = score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc

Hmm wait, I realize the SCF iteration might not work well as a standalone method because:
1. The active set changes each iteration, making it non-smooth.
2. The "power iteration" form `a ∝ exp(-gamma * r)` might not converge to the right fixed point.

Let me think about this differently. The fixed point condition is: at the optimum, the gradient of `max_k conv_k` should be constant on the support of `a`. The gradient is proportional to the response `r_j`. So at the optimum, `r_j = constant` for all j with `a_j > 0`.

This means the optimal `a` is supported on the set where `r_j` is at its minimum. This is a "water-filling" type condition.

A better iteration:
1. Compute `r = T_S(a)` (response).
2. Find the minimum of `r` over the support: `r_min = min(r_j : a_j > 0)`.
3. Update: `a_j = max(0, 1 - r_j / r_min) / sum(...)` or `a_j = (r_max - r_j)_+ / sum(...)`.

Actually, the correct KKT condition for minimizing `f(a) = max_k conv_k(a)` subject to `a >= 0`, `sum a = 1` is:

At the optimum `a*`, there exists `lambda` such that:
- `∇f(a*)_j = lambda` for all j with `a*_j > 0`
- `∇f(a*)_j >= lambda` for all j with `a*_j = 0`

where `∇f(a*)_j = (2/|S|) * sum_{k in S} a*_{k-j} * [valid] = 2 * r_j`.

So the condition is `r_j = constant` on the support.

A natural iteration: **gradient projection**
1. Compute `r = T_S(a)`.
2. Compute gradient: `g = 2 * r`.
3. Project: `a_old = a`. Update `a = a - eta * (g - mean(g_on_support))`.
4. Project to simplex: `a = max(0, a)`, `a = a / a.sum()`.

This is essentially the subgradient method with a twist. The step size `eta` is crucial.

For the subgradient method, a good step size is `eta_t = c / sqrt(t)` (diminishing) or `eta_t = c * (f_t - f*) / ||g_t||^2` (Polyak step, requires knowing f*).

Since we don't know f*, we can use a heuristic: `eta_t = eta_0 / (1 + t/n0)` where n0 is a timescale.

Let me also think about the **Frank-Wolfe (conditional gradient)** method:

At each iteration:
1. Compute linearization: `g = ∇f(a)` (subgradient).
2. Find the vertex of the simplex that minimizes `<g, s>`: this is `s = e_{j*}` where `j* = argmin_j g_j`.
3. Update: `a = (1-t) * a + t * s` where `t = 2/(t_iter+2)` (standard FW step size).

This is simple and guarantees convergence (albeit slowly). It's natural for our problem since the feasible set is the simplex.

The advantage of FW: no need to project (the step is automatically in the feasible set). The disadvantage: slow convergence (O(1/t)).

A hybrid: use FW for the first phase (to get a good initial point), then switch to L-BFGS for polishing.

Actually, I think the most effective approach combining novelty and practicality is:

1. **Phase 1: SCF / Subgradient with adaptive step size** (fast, O(n log n) per iter)
2. **Phase 2: L-BFGS with log-sum-exp smoothing** (polishing)
3. **Phase 3: Fine subgradient with very small steps** (final squeeze)

And I'll try multiple n values (2048, 4096).

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _response(a, w):

    w has length 2n-1, a has length n, result has length n.
    """

    wf = w[::-1]
    conv = _full_conv(wf, a)
    return conv[n-1:2*n-1][::-1]

def _softmax_grad(z, beta, n):
    """Gradient of the smoothed objective w.r.t. z (log-a parameters)."""
    a = _softmax(z)
    s = float(a.sum())  # Should be 1 after softmax
    c = _full_conv(a, a)
    c_max = float(c.max())
    
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    w_sum = e.sum()
    if w_sum < 1e-300:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / w_sum
    
    r = _response(a, w)
    W = float(np.dot(w, c))
    
    # dM/da_i = 2*r_i, M = W (the expected value under w)
    # df/da_i = 2n/s^2 * (2*r_i - 2*W/s)
    # df/dz_i = a_i * (df/da_i - sum_j a_j * df/da_j)
    
    # sum_j a_j * df/da_j = (2n/s^2) * (2*sum_j a_j*r_j - 2*W/s * sum_j a_j)
    # = (2n/s^2) * (2*sum_j a_j*r_j - 2*W/s * s)
    # = (2n/s^2) * (2*sum_j a_j*r_j - 2*W)
    
    ar = float(np.dot(a, r))
    dropout = (2.0 * n / (s * s)) * (2.0 * ar - 2.0 * W)
    
    grad_a = (2.0 * n / (s * s)) * (2.0 * r - 2.0 * W / s)
    grad_z = a * (grad_a - dropout)
    
    # Objective value
    obj = 2.0 * n * (c_max + beta * np.log(max(w_sum, 1e-300))) / (s * s)
    
    return obj, grad_z

def _lbfgs_from_z(z0, beta, n, deadline, max_iter=5000):
    """Run L-BFGS-B from z0 for one beta phase."""
    def fun(z):
        obj, _ = _softmax_grad(z, beta, n)
        return obj
    
    def jac(z):
        _, g = _softmax_grad(z, beta, n)
        return g
    
    try:
        result = minimize(
            fun, z0, jac=jac, method='L-BFGS-B',

                     'maxcor': 30, 'maxls': 100, 'disp': False}
        )
        return _softmax(result.x)
    except Exception:
        return _softmax(z0)

def _subgrad_polish(a_init, deadline, max_steps=10000, lr_init=0.02):
    """Subgradient polishing in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax(z)
        c = _full_conv(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros(2*n-1)
        w[active] = 1.0 / active.sum()
        
        r = _response(a, w)
        W = float(np.dot(w, c))
        
        # Gradient in z-space
        grad_a = 4.0 * n * (r - W)
        grad_z = a * (grad_a - np.dot(a, grad_a))
        
        gn = np.linalg.norm(grad_z)
        if gn < 1e-15:
            break
        
        step = lr / (1.0 + i * 0.0005)
        z = z - step * grad_z / gn * lr  # Normalize gradient
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if i % 200 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            else:
                lr *= 0.95
    
    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc

def _fw_steps(a_init, deadline, max_steps=50000):
    """Frank-Wolfe (conditional gradient) method."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_conv(a, a)
        c_max = float(c.max())
        tol = 0.001 * c_max
        active = c >= (c_max - tol)
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros(2*n-1)
        w[active] = 1.0 / active.sum()
        
        r = _response(a, w)
        W = float(np.dot(w, c))
        grad = 4.0 * n * (r - W)
        
        # FW: move towards the vertex with minimum gradient
        j_star = int(np.argmin(grad))
        e_j = np.zeros(n)
        e_j[j_star] = 1.0
        
        # FW step size
        t = 2.0 / (i + 2)
        
        a_new = (1 - t) * a + t * e_j
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        if i % 500 == 0:
            sc = _score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_sc

def _scf_optimize(a_init, deadline, gamma_range=(1.0, 200.0), max_iters=200000):
    """Self-consistent field optimization."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        frac = i / max_iters
        gamma = gamma_range[0] * (gamma_range[1] / gamma_range[0]) ** frac
        tol = 0.01 + 0.04 * frac
        
        c = _full_conv(a, a)
        c_max = float(c.max())
        active = c >= (c_max - tol * c_max)
        if not active.any():
            active[int(np.argmax(c))] = True
        w = np.zeros(2*n-1)
        w[active] = 1.0 / active.sum()
        
        r = _response(a, w)
        
        # Update: a_new ∝ exp(-gamma * r)
        z = -gamma * r
        a_new = _softmax(z)
        
        # Blend with old (to maintain stability)
        alpha = 0.5 + 0.5 * frac  # Increase weight on new as we progress
        a = alpha * a_new + (1 - alpha) * a
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        if i % 500 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    sc_final = _score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    
    return best_a, best_sc
