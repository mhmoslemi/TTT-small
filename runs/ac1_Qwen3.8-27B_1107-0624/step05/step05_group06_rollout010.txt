"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using multi-phase optimization:
1) L-BFGS with power-sum smoothing and continuation
2) Adam-based projected gradient with adaptive active set
3) Multiplicative/coordinate descent refinement
4) Spectral-domain initialization and diversification restarts

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.full_like(v, target_sum / len(v))
    return v * (target_sum / s)


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _power_sum_obj(z, p, n):
    """Power-sum objective in log-softmax space."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)
    val = (np.sum(c_clipped ** p) / len(c)) ** (1.0 / p)
    return 2.0 * n * val

    """Gradient of power-sum objective."""
    a = _softmax(z)
    n_a = len(a)
    c = _full_convolve(a, a)

def _power_sum_grad_v2(z, p, n):
    """Corrected gradient of power-sum objective."""
    a = _softmax(z)
    n_a = len(a)
    c = _full_convolve(a, a)
    c_clipped = np.maximum(c, 1e-30)
    c_p = c_clipped ** p
    sum_c_p = np.sum(c_p) / len(c)
    if sum_c_p < 1e-300:
        return np.zeros_like(z)
    # factor = 2n * (sum c^p / L)^{1/p - 1} / L
    factor = 2.0 * n * sum_c_p ** (1.0 / p - 1.0) / len(c)
    c_pm1 = c_clipped ** (p - 1)
    
    # grad_c[j] = sum_k c_pm1[k] * (a[k-j] + a[j-k]) for valid indices
    # Part 1: sum_k c_pm1[k] * a[k-j]
    ca1 = _full_convolve(c_pm1, a[::-1])
    part1 = ca1[n - 1: n - 1 + n_a]
    
    # Part 2: sum_k c_pm1[k] * a[j-k] = (c_pm1 * a)[j]
    ca2 = _full_convolve(c_pm1, a)
    part2 = ca2[:n_a]
    
    grad_c = part1 + part2
    
    W = float(np.dot(grad_c, a))
    return factor * a * (grad_c - W)


def _log_sum_exp_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _log_sum_exp_grad(z, beta, n):
    a = _softmax(z)
    n_a = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc1 = _full_convolve(w, a_rev)
    part1 = wc1[n - 1: n - 1 + n_a]
    wc2 = _full_convolve(w, a)
    part2 = wc2[:n_a]
    grad_c = part1 + part2
    W = float(np.dot(w, c))
    return 4.0 * n * a * (grad_c - W)


def _adam_step(params, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update step."""
    t = state['t']
    state['t'] = t + 1
    m = state['m']
    v = state['v']
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** state['t'])
    v_hat = v / (1 - beta2 ** state['t'])
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)
    state['m'] = m
    state['v'] = v
    return params


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.4 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "triple":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * t) + 0.15 * np.cos(6.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _lbfgs_power_sum(a_init, p_schedule, deadline, min_time=10.0):
    """L-BFGS optimization with power-sum smoothing and continuation."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    current_z = z0.copy()
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, p in enumerate(p_schedule):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 5:
            break

        def fun(z):
            return _power_sum_obj(z, p, n)

        def jac(z):
            return _power_sum_grad_v2(z, p, n)

        try:
            result = minimize(
                fun, current_z, jac=jac, method='L-BFGS-B',
                options={
                    'maxiter': min(3000, max(100, int(remaining / 0.03))),
                    'ftol': 1e-12,
                    'gtol': 1e-9,
                    'maxcor': 40,
                    'maxls': 100,
                    'disp': False,
                }
            )
            current_z = result.x
            a_new = _softmax(current_z)
            sc = _score(a_new)
            if sc < current_sc:
                current_a = a_new
                current_sc = sc
                print(f"    p={p:.1f}: score={sc:.10f} (improved)")
        except Exception as e:
            print(f"    L-BFGS p={p} error: {e}")

    return current_a, current_sc

def _adaptive_subgrad(a_init, deadline, max_steps=5000, lr_init=0.02):
    """Adaptive subgradient with exact active set and momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    momentum = np.zeros(n)
    beta_m = 0.9
    step_size = np.zeros(n)  # RMSProp-style

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc1 = _full_convolve(w, a_rev)
        part1 = wc1[n - 1: n - 1 + n]
        wc2 = _full_convolve(w, a)
        part2 = wc2[:n]
        grad_c = part1 + part2
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (grad_c - W)

        # RMSProp + momentum
        step_size = 0.9 * step_size + 0.1 * grad ** 2
        momentum = beta_m * momentum - lr * grad / (np.sqrt(step_size) + 1e-8)
        z = z + momentum
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()
            else:
                lr *= 0.95

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _mult_descent(a_init, deadline, max_steps=5000):
    """Multiplicative descent for minimizing max autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = _score(a)
    lr = 0.1

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()
        # Gradient of max c_k w.r.t. a_j
        active = c >= c_max * 0.98
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        a_rev = a[::-1]
        wc1 = _full_convolve(w, a_rev)
        part1 = wc1[n - 1: n - 1 + n]
        wc2 = _full_convolve(w, a)
        part2 = wc2[:n]
        grad = part1 + part2

        # Multiplicative update: a_j *= (1 - lr * grad_j / c_max)
        # Keep non-negative
        update = 1.0 - lr * grad / max(c_max, 1e-10)
        update = np.maximum(update, 0.01)
        a = a * update
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

        if i % 50 == 0:
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            else:
                lr *= 0.9

    a_final = a / a.sum()
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n, dtype=np.float64)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def propagate_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: L-BFGS with power-sum and LSE smoothing,
    adaptive subgradient, multiplicative descent, and diverse restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"improve={prev - best_score:.4e} t={time.monotonic() - t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(parent)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(parent, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, using uniform n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "init_uniform")

    # --- Phase 1: L-BFGS with power-sum continuation ---
    print(f"\nPhase 1: L-BFGS power-sum (t={time.monotonic() - t_start:.1f}s)")
    c_cur = _full_convolve(a0, a0)
    max_c = float(c_cur.max())
    n = len(a0)
    # Scale p values relative to the magnitude of c
    p_values = [4, 6, 8, 12, 16, 24, 32, 48, 64, 96]
    a_psum, sc_psum = _lbfgs_power_sum(a0, p_values, deadline, min_time=300.0)
    update_best(a_psum, sc_psum, "power_sum")

    # --- Phase 2: L-BFGS with log-sum-exp continuation ---
    print(f"\nPhase 2: L-BFGS LSE (t={time.monotonic() - t_start:.1f}s)")
    a_cur = best_seq if best_seq is not None else a_psum
    c_cur = _full_convolve(a_cur, a_cur)
    max_cb = float(c_cur.max())
    betas = max_cb * np.array([0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
    a_lse, sc_lse = _lbfgs_lse(a_cur, betas, deadline, min_time=200.0)
    update_best(a_lse, sc_lse, "lse")

    # --- Phase 3: Adaptive subgradient polishing ---
    print(f"\nPhase 3: Adaptive subgradient (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None:
        sg_deadline = min(deadline - 100.0, time.monotonic() + 180.0)
        a_sg, sc_sg = _adaptive_subgrad(best_seq, sg_deadline, max_steps=8000, lr_init=0.015)
        update_best(a_sg, sc_sg, "adaptive_sg")

    # --- Phase 4: Multiplicative descent ---
    print(f"\nPhase 4: Multiplicative descent (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 80:
        md_deadline = min(deadline - 80.0, time.monotonic() + 120.0)
        a_md, sc_md = _mult_descent(best_seq, md_deadline, max_steps=5000)
        update_best(a_md, sc_md, "mult_descent")

    # --- Phase 5: Fine L-BFGS with very small beta ---
    print(f"\nPhase 5: Fine L-BFGS (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 60:
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        fine_betas = max_cb * np.array([0.0001, 0.00005, 0.00002, 0.00001])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 40:
                break
            remaining = deadline - 40 - time.monotonic()
            if remaining < 5:
                break
            a_new, sc_new = _lbfgs_lse(best_seq, [beta], deadline, min_time=20.0)
            update_best(a_new, sc_new, f"fine_{i}")

    # --- Phase 6: Diverse restarts ---
    print(f"\nPhase 6: Restarts (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining_total = deadline - 30 - time.monotonic()
        n_restarts = min(5, max(1, int(remaining_total / 120)))

        for i in range(n_restarts):
            if time.monotonic() > deadline - 25:
                break

            if i == 0:
                p_start = _perturb(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=3)
            elif i == 2:
                p_start = _perturb(best_seq, 0.05, rng)
            elif i == 3:
                p_start = _perturb_fourier(best_seq, rng, n_modes=5)
            else:
                # Try a different profile
                p_start = _make_profile(n_main, "triple")
                # Blend with best
                p_start = 0.5 * best_seq + 0.5 * p_start
                p_start = p_start / p_start.sum()

            sc_start = _score(p_start)
            print(f"  Restart {i}: start_score={sc_start:.10f}")

            # Run a quick LSE L-BFGS
            c_r = _full_convolve(p_start, p_start)
            max_cr = float(c_r.max())
            r_betas = max_cr * np.array([0.05, 0.02, 0.008, 0.003, 0.001])
            r_deadline = min(deadline - 25, time.monotonic() + 100)
            a_r, sc_r = _lbfgs_lse(p_start, r_betas, r_deadline, min_time=20.0)
            update_best(a_r, sc_r, f"restart_{i}")

            # Quick subgradient
            if sc_r < best_score + 0.01 and time.monotonic() < deadline - 20:
                a_r2, sc_r2 = _adaptive_subgrad(a_r, deadline - 20, max_steps=1000, lr_init=0.01)
                update_best(a_r2, sc_r2, f"restart_{i}_sg")

    # --- Phase 7: Final polishing cycle ---
    print(f"\nPhase 7: Final polish (t={time.monotonic() - t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        # One more fine LSE
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        if time.monotonic() < deadline - 3:
            a_fin, sc_fin = _lbfgs_lse(best_seq, [mb * 0.00001], deadline - 3, min_time=2.0)
            update_best(a_fin, sc_fin, "final_lse")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _adaptive_subgrad(best_seq, deadline - 2, max_steps=500, lr_init=0.005)
            update_best(a_fin2, sc_fin2, "final_sg")

        # Try multiplicative on final
        if time.monotonic() < deadline - 2:
            a_fin3, sc_fin3 = _mult_descent(best_seq, deadline - 2, max_steps=1000)
            update_best(a_fin3, sc_fin3, "final_mult")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        # Round to avoid floating point issues
        final = np.round(final, 15)
        final = np.maximum(final, 0.0)

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
