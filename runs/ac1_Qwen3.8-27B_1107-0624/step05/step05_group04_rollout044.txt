**Key mathematical insight:** For non-negative sequences with `sum(a) = S`, by Cauchy-Schwarz: `conv(a,a)[0] = sum(a_i^2) >= S^2/n`. Therefore `max(conv) >= S^2/n`, and `score >= 2n * S^2/n / S^2 = 2`. The uniform distribution achieves exactly 2.

"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key insight: By Cauchy-Schwarz, sum(a_i^2) >= (sum a_i)^2/n, so conv(a,a)[0] >= sum(a)^2/n,
meaning the score is always >= 2 for non-negative sequences. The uniform distribution
achieves exactly score = 2, which is the theoretical minimum.
Strategy: Return uniform distribution as primary candidate, evaluate parent, and run
lightweight optimization as safety net.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _adam_optimize(a_init, deadline, lr=0.1, max_iter=50000):
    """Adam gradient descent on log-space representation with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)

    c = _full_convolve(a_init, a_init)
    cm = float(c.max())
    beta = max(1e-10, 0.1 * cm)

    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Adapt beta: decrease as peaks concentrate
        if it % 500 == 0:
            beta = max(1e-10, 0.05 * cm)

        if cm < 1e-30:
            break

        grad_z = _logsumexp_grad(z, beta)
        if not np.all(np.isfinite(grad_z)):
            break

        m_v = b1 * m_v + (1 - b1) * grad_z
        v_v = b2 * v_v + (1 - b2) * (grad_z * grad_z)
        mh = m_v / (1 - b1 ** t)
        vh = v_v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)

        sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decay learning rate
        if t % 5000 == 0:
            lr = max(0.005, lr * 0.85)

    return best_a, best_score


def _lbfgs_refine(a_init, deadline, n_betas=10):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    beta_start = 0.2 * max_c0
    beta_end = 1e-6 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)

    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 60.0)
        level_deadline = time.monotonic() + time_budget

        maxiter = 1000 if beta > 0.05 * max_c0 else 3000
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': maxiter * 3}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -80, 80)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score
