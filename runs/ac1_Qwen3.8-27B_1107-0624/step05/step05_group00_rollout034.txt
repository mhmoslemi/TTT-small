```python
import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _hankel_product(w, a):
    """Compute H(w) @ a where H(w)_{j,m} = w_{j+m}"""
    n = len(a)
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, w)
    return conv[n-1:2*n-1]

def _minimax_qp_step(w, a_init, n_fista_iters=100):
    """Solve min a^T H(w) a s.t. a in simplex using FISTA"""
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 2.0  # Lipschitz constant of gradient 2*H(w)*a
    step = 1.0 / L
    
    for _ in range(n_fista_iters):
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(len(y)) / len(y)
        
        grad = 2.0 * _hankel_product(w, y)
        a_next = _proj_simplex(y - step * grad)
        
        a_prev = a_curr
        a_curr = a_next
        t_val = t_next
    
    return a_curr

def _minimax_alternate(a_init, deadline, max_rounds=200):
    """Alternating minimax: alternate between w update and a update"""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    for round in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Update w: softmax of conv(a, a)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Adaptive beta: start large, decrease
        beta = max(0.001 * c_max, 1e-10)
        w = np.exp((c - c_max) / beta)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum
        
        # Update a: FISTA on QP
        a = _minimax_qp_step(w, a, n_fista_iters=50)
        a = a / a.sum()
        
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if round % 20 == 0:
            print(f"  Minimax round {round}: score={sc:.10f}")
    
    return best_a, best_score
```

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses minimax alternating, FISTA with adaptive restarts, Adam in log-space,
and multi-length exploration. Key improvements: Hankel-matrix QP subproblems
solved via FISTA, adaptive beta schedules, and aggressive polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _hankel_product(w, a):
    n = len(a)
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, w)
    return conv[n - 1:2 * n - 1]


def _softmax_z(z):
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_maxsmooth(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()
    grad_norm_prev = 0.0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        a_rev = y[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        grad = 2.0 * h

        grad_norm = np.linalg.norm(grad)
        if grad_norm_prev > 0 and grad_norm > 1.5 * grad_norm_prev:
            t_val = 1.0

        a_next = _proj_simplex(y - step * grad)
        grad_norm_prev = grad_norm

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _minimax_qp(w, a_init, n_iters=200):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    step = 0.4

    for _ in range(n_iters):
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        grad = 2.0 * _hankel_product(w, y)
        a_next = _proj_simplex(y - step * grad)

        a_prev = a_curr
        a_curr = a_next
        t_val = t_next

    return a_curr


def _minimax_alternate(a_init, deadline, max_rounds=500):
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = max(0.005 * c_max / (1.0 + 0.02 * rnd), 1e-12)
        w = np.exp((c - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[int(np.argmax(c))] = 1.0
        else:
            w = w / w_sum

        a = _minimax_qp(w, a, n_iters=80)
        a = a / a.sum()

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if rnd % 50 == 0:
            print(f"    Minimax round {rnd}: score={sc:.10f}")

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total, lr_decay=0.995, decay_interval=1000):

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        a = _softmax_z(z)

        c_shifted = (c - cm) / beta

        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)

            c = _full_convolve(a, a)
            cm = float(c.max())

            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                break

            t += 1
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:

                current_lr = max(0.0005, current_lr * lr_decay)

    a_final = _softmax_z(z)
    sc_final = _score_norm(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_z(z)

    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bilateral":
        center = n / 2.0
        t = np.arange(n)
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / n)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
