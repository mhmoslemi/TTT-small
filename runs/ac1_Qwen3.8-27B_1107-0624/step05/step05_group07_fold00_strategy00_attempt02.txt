    except Exception as e:
        print(f"Critical error: {e}")
        # Fallback to basic candidate
        n = 2048
        basic_sequence = [{"profile": np.ones(n), "label": "basic"}]
        final_sequence, score = np.min(basic_sequence)[0], evaluate_sequence(final_sequence)
        print(score)
