"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses LP-based SQP direction finding, aggressive log-sum-exp continuation,
adaptive Adam polish with backtracking, and peak-specific refinement.
Key improvements over previous: LP direction finding for better geometry,
more aggressive continuation, adaptive step sizes, and longer runtime allocation.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _convolve_with_reversed(a, b):
    """Compute convolve(a, b[::-1]) efficiently, returning the part [n-1:2n-1]."""
    n = len(a)
    b_rev = b[::-1]
    c = _full_convolve(a, b_rev)
    return c[n-1:2*n-1]


def _lse_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    lse = cm + beta * (max_s + float(np.log(e.sum())))
    return float(2.0 * n * lse)


def _lse_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lp_direction(a, c, max_c, K=20):
    """
    Solve LP to find direction d that reduces the worst peak.
    minimize t
    subject to: sum_j a[k_i-j]*d[j] <= t - c[k_i] for i=1..K
                sum_j d[j] = 0
                d[j] >= -a[j]
    """
    n = len(a)
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:K]
    
    # Build constraint matrix
    # For peak at position k, constraint is: sum_j a[k-j] * d[j] <= t - c[k]
    # Rearranged: sum_j a[k-j] * d[j] - t <= -c[k]
    
    # Variables: [d[0], ..., d[n-1], t] -> n+1 variables
    # Objective: [0, 0, ..., 0, 1] (minimize t)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for i, k in enumerate(active):
        # sum_j a[k-j] * d[j] - t <= -c[k]
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_ub[i, i_lo:i_hi+1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        A_ub[i, -1] = -1.0  # coefficient of t
        b_ub[i] = -c[k]
    
    # Equality: sum d[j] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: d[j] >= -a[j], t unbounded
    bounds = [(-a[j], None) for j in range(n)]
    bounds.append((None, None))  # t unbounded
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status != 0:
            return None
        d = res.x[:n]
        t_val = res.x[n]
        return d, t_val
    except Exception:
        return None


def _sqp_phase(a_init, deadline, max_rounds=50, K=15):
    """SQP iterations with LP direction finding."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    print(f"  SQP start: score={best_score:.10f}")
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        a = best_a.copy()
        c = _full_convolve(a, a)
        max_c = float(c.max())
        
        # Get LP direction
        result = _lp_direction(a, c, max_c, K=K)
        if result is None:
            print(f"  SQP round {rnd}: LP failed")
            break
        
        d, t_val = result
        
        # Line search for step size
        best_alpha = 0.0
        best_trial_score = best_score
        
        for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = a + alpha * d
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            c_trial = _full_convolve(a_trial, a_trial)
            sc = float(2.0 * n * c_trial.max())
            if sc < best_trial_score:
                best_trial_score = sc
                best_alpha = alpha
        
        if best_alpha == 0.0:
            print(f"  SQP round {rnd}: no improvement (t={t_val:.6f})")
            # Try with fewer peaks
            K = max(5, K // 2)
            if K < 5:
                break
            continue
        
        best_a = (a + best_alpha * d)
        best_a = np.maximum(best_a, 0.0)
        best_a = best_a / best_a.sum()
        best_score = best_trial_score
        
        elapsed = time.monotonic() - t0
        print(f"  SQP round {rnd}: score={best_score:.10f}, alpha={best_alpha:.3f}, "
              f"t_val={t_val:.6f}, time={elapsed:.1f}s")
        
        if abs(best_alpha * t_val) < 1e-10:
            break
    
    print(f"  SQP end: score={best_score:.10f}")
    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, beta_schedule=None, maxiter_per_beta=10000):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    if beta_schedule is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001, 0.000005, 0.000003, 0.000001]
        beta_schedule = [b * m0 for b in fracs]
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        t0 = time.monotonic()
        try:
            result = minimize(
                _lse_objective, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_gradient,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-25,
                         'gtol': 1e-16, 'maxls': 50, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
        
        elapsed = time.monotonic() - t0
        print(f"  L-BFGS beta={beta:.2e}: score={best_score:.10f}, time={elapsed:.1f}s")
    
    return best_a, best_score
