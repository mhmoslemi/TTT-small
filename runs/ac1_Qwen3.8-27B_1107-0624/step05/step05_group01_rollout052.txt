"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses mirror descent (multiplicative update), L-BFGS with top-K peak smoothing,
simulated annealing, and interpolation to larger n.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _cross_corr(a, w):
    """Compute h[i] = sum_j w[j] * a[j - i] for i = 0..n-1.
    This is the gradient direction: grad[i] = 2 * h[i]."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    return wc[n - 1:2 * n - 1]


def _mirror_descent_step(a, c, beta, eta):
    """One step of mirror descent on the smoothed max objective.
    a_new[i] = a[i] * exp(-eta * grad[i]) / Z
    where grad[i] = 2 * sum_k w[k] * a[k-i]"""
    n = len(a)
    w = _softmax_fn(c / beta)
    h = _cross_corr(a, w)
    log_ratio = -eta * 2.0 * h
    log_ratio = log_ratio - log_ratio.max()
    ratio = np.exp(log_ratio)
    a_new = a * ratio
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def _mirror_descent_opt(a_init, deadline, beta_start=0.05, beta_end=0.001,
                        eta_start=0.1, max_iters=50000):
    """Mirror descent with adaptive beta and eta."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    t = 0
    no_improve = 0
    check_every = 10

    while t < max_iters and time.monotonic() < deadline:
        frac = min(1.0, t / (max_iters * 0.5))
        beta = beta_start * (beta_end / beta_start) ** frac
        eta = eta_start * (0.999 ** (t // 100))

        c = _full_convolve(a, a)
        a = _mirror_descent_step(a, c, beta, eta)
        t += 1

        if t % check_every == 0:
            sc = _score_norm(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                a = best_a.copy()
                eta = eta * 0.5
                beta = max(beta * 0.5, beta_end)
                no_improve = 0

        if time.monotonic() > deadline - 5:
            break

    return best_a, best_score


def _topk_obj(z, K, n):
    """Objective: 2n * smoothed_max of top K peaks of conv(softmax(z), softmax(z))."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    # Use top-K with softmax smoothing
    top_k = np.sort(c)[::-1][:K]
    beta = 1e-6 * max(top_k.max(), 1e-300)
    shifted = (top_k - top_k.max()) / beta
    lse = top_k.max() + beta * float(np.log(np.sum(np.exp(shifted - shifted.max()))))
    return float(2.0 * n * lse)


def _topk_grad(z, K, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    top_k_idx = np.argsort(c)[::-1][:K]
    top_k = c[top_k_idx]
    beta = 1e-6 * max(top_k.max(), 1e-300)
    shifted = (top_k - top_k.max()) / beta
    e = np.exp(shifted - shifted.max())
    w_top = e / e.sum()

    # Build full weight vector
    w = np.zeros(len(c))
    for i, idx in enumerate(top_k_idx):
        w[idx] = w_top[i]

    h = _cross_corr(a, w)
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_topk(a_init, deadline, K=5, maxiter=5000):
    """L-BFGS optimization with top-K peak surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    try:
        result = minimize(
            _topk_obj, z, args=(K, n),
            method='L-BFGS-B', jac=_topk_grad,
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                     'maxls': 80, 'maxfun': 10000}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score


def _lbfgs_lse(a_init, deadline, beta_list=None, maxiter_per_beta=3000):
    """L-BFGS with log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        beta_list = [b * m0 for b in fracs]

    def _lse_obj(z_in, beta, n_in):
        a_in = _softmax_fn(z_in)
        c_in = _full_convolve(a_in, a_in)
        cm = float(c_in.max())
        if cm < 1e-300:
            return 0.0
        shifted = (c_in - cm) / beta
        max_s = float(shifted.max())
        lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
        return float(2.0 * n_in * lse)

    def _lse_grad(z_in, beta, n_in):
        a_in = _softmax_fn(z_in)
        c_in = _full_convolve(a_in, a_in)
        cm = float(c_in.max())
        if cm < 1e-300:
            return np.zeros_like(z_in)
        shifted = (c_in - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c_in))
        h = _cross_corr(a_in, w)
        grad = 4.0 * n_in * a_in * (h - W)
        if not np.all(np.isfinite(grad)):
            return np.zeros_like(z_in)
        return grad

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _simulated_annealing(a_init, deadline, n_restarts=20, max_rounds=500):
    """Simulated annealing: random perturbations with acceptance criterion."""
    n = len(a_init)
    rng = np.random.default_rng(12345)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c = _full_convolve(best_a, best_a)
    cm = float(c.max())

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break
        a = best_a.copy()
        T = 0.01 * cm
        for it in range(max_rounds):
            if time.monotonic() > deadline:
                break
            # Perturb: pick a random contiguous block and scale it
            block_len = rng.integers(1, max(2, n // 100))
            start = rng.integers(0, n - block_len)
            perturbation = 1.0 + rng.random() * 0.3 * (2 * rng.random() - 1)
            a_trial = a.copy()
            a_trial[start:start + block_len] *= perturbation
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            c_trial = _full_convolve(a_trial, a_trial)
            sc_trial = float(c_trial.max())

            if sc_trial < cm or rng.random() < np.exp(-(sc_trial - cm) / T):
                a = a_trial
                cm = sc_trial
            T *= 0.999

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.02, max_iter=200000):
    """Adam with adaptive log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.005 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.005 * (0.98 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        h = _cross_corr(a, w)
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _subgradient_descent(a_init, deadline, max_iters=100000):
    """Subgradient descent with backtracking line search."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    eta = 0.01
    beta = 0.02
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta_cur = 0.01 * cm
        w = _softmax_fn(c / beta_cur)
        grad = 2.0 * _cross_corr(a, w)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        a_trial = None
        for trial in range(10):
            step = eta / (2.0 ** trial)
            a_cand = a - step * grad
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s < 1e-15:
                continue
            a_cand = a_cand / s
            sc = _score_norm(a_cand)
            if sc < best_score or sc < _score_norm(a) - 1e-15:
                a_trial = a_cand
                break
            a_cand = None

        if a_trial is not None:
            a = a_trial
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                eta = min(eta * 1.1, 0.1)
            else:
                no_improve += 1
        else:
            eta *= 0.5
            no_improve += 1

        if no_improve > 200:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization: mirror descent, subgradient descent,
    L-BFGS (top-K and LSE), simulated annealing, Adam polish, and
    interpolation to larger n. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _greedy_init(n)
        update_best(parent, _score_norm(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Mirror descent (multiplicative update) - 250s ===
    print("Phase 1: Mirror descent (multiplicative update)")
    p1_end = min(deadline, t_start + 250.0)

    a_md, sc_md = _mirror_descent_opt(
        best_seq, p1_end, beta_start=0.08, beta_end=0.002,
        eta_start=0.15, max_iters=100000
    )
    update_best(a_md, sc_md, "mirror_descent")

    # === Phase 2: L-BFGS with top-K (300s) ===
    print("Phase 2: L-BFGS top-K")
    p2_end = min(deadline, t_start + 550.0)

    if time.monotonic() < p2_end and best_seq is not None:
        for K in [3, 5, 8]:
            if time.monotonic() > p2_end - 30:
                break
            a_k, sc_k = _lbfgs_topk(best_seq, min(p2_end - 10, time.monotonic() + 80), K=K)
            update_best(a_k, sc_k, f"lbfgs_topK{K}")

    # === Phase 3: Subgradient descent (200s) ===
    print("Phase 3: Subgradient descent")
    p3_end = min(deadline, t_start + 750.0)

    if time.monotonic() < p3_end and best_seq is not None:
        a_sg, sc_sg = _subgradient_descent(best_seq, p3_end, max_iters=50000)
        update_best(a_sg, sc_sg, "subgrad")

    # === Phase 4: Simulated annealing (200s) ===
    print("Phase 4: Simulated annealing")
    p4_end = min(deadline, t_start + 950.0)

    if time.monotonic() < p4_end and best_seq is not None:
        a_sa, sc_sa = _simulated_annealing(best_seq, p4_end, n_restarts=15, max_rounds=400)
        update_best(a_sa, sc_sa, "sim_anneal")

    # === Phase 5: L-BFGS LSE continuation (250s) ===
    print("Phase 5: L-BFGS LSE continuation")
    p5_end = min(deadline, t_start + 1200.0)

    if time.monotonic() < p5_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        betas = [m_b * b for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]]
        a_lse, sc_lse = _lbfgs_lse(best_seq, p5_end, beta_list=betas, maxiter_per_beta=4000)
        update_best(a_lse, sc_lse, "lbfgs_lse")

    # === Phase 6: Adam polish (200s) ===
    print("Phase 6: Adam polish")
    p6_end = min(deadline, t_start + 1400.0)

    if time.monotonic() < p6_end and best_seq is not None:
        a_ad, sc_ad = _adam_polish(best_seq, p6_end, lr=0.015, max_iter=150000)
        update_best(a_ad, sc_ad, "adam_polish")

        if time.monotonic() < p6_end - 20:
            a_ad2, sc_ad2 = _adam_polish(best_seq, p6_end, lr=0.005, max_iter=80000)
            update_best(a_ad2, sc_ad2, "adam_slow")

    # === Phase 7: Try larger n (interpolation) (250s) ===
    print("Phase 7: Larger n")
    p7_end = min(deadline, t_start + 1650.0)

    if time.monotonic() < p7_end - 60 and best_seq is not None:
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"  Interpolating to n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")

            if time.monotonic() < p7_end - 30:
                a_md2, sc_md2 = _mirror_descent_opt(
                    a_interp, min(p7_end - 20, time.monotonic() + 60),
                    beta_start=0.05, beta_end=0.005, eta_start=0.1, max_iters=30000
                )
                update_best(a_md2, sc_md2, f"mirror_n{n_new}")

            if time.monotonic() < p7_end - 20:
                a_k2, sc_k2 = _lbfgs_topk(a_interp, min(p7_end - 5, time.monotonic() + 40), K=5)
                update_best(a_k2, sc_k2, f"topk_n{n_new}")

    # === Phase 8: Final fine polish (remaining time) ===
    print("Phase 8: Final polish")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_fin, sc_fin = _adam_polish(best_seq, deadline - 10, lr=0.003, max_iter=50000)
        update_best(a_fin, sc_fin, "final_polish")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(256)
    print(f"FALLBACK: n=256, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
