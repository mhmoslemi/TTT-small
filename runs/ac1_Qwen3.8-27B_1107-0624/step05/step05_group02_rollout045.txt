def _next_pow2(n):
    k = 1
    while k < n:

def _full_convolve(a, b):

    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

"""
Multi-phase gradient-based optimization with log-sum-exp smoothing,
adaptive temperature annealing, spectral equalization, and multi-restart
for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Key improvements: continuous temperature schedule, adaptive LR,

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _raised_cosine(n):
    x = np.linspace(0.0, np.pi, n)
    a = np.cos(x / 2.0) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _compute_smooth_grad(a, beta):
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad, c, w, cm

def _spectral_equalize(a, n_iters=50, alpha=0.5):
    """Spectral domain refinement: flatten the autocorrelation via PSD adjustment."""
    n = len(a)
    a_work = a.copy()
    N = _next_pow2(2 * n - 1)

    for _ in range(n_iters):
        fft_a = np.fft.rfft(a_work, N)
        psd = np.abs(fft_a) ** 2
        acorr = np.fft.irfft(fft_a * np.conj(fft_a), N)[:2 * n - 1]
        acorr = np.maximum(acorr, 1e-30)

        # Target: make acorr flatter
        target = np.full(len(acorr), 1.0 / len(acorr))
        ratio = target / acorr
        # Damp the adjustment
        adj = 1.0 + alpha * (ratio - 1.0)
        adj = np.maximum(adj, 0.1)
        adj = np.minimum(adj, 10.0)

        psd_new = psd * adj
        # Take square root (magnitude) and keep phase
        mag_new = np.sqrt(np.maximum(psd_new, 0))
        phase = np.angle(fft_a)
        fft_new = mag_new * np.exp(1j * phase)

        a_new = np.fft.irfft(fft_new, N)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        c_old = _full_convolve(a_work, a_work)
        c_new = _full_convolve(a_new, a_new)
        if c_new.max() < c_old.max():
            a_work = a_new
        else:
            alpha *= 0.8

    return a_work


def _interp_to_length(p, n_new):
    n0 = len(p)

    x0 = np.linspace(0, 1, n0)

    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)

            p_interp = _interp_to_length(best_seq, target_n)
            update_best(p_interp)

            remaining = phase5_end - time.monotonic() - 30
            if remaining > 60:
                a_r, _ = _optimize_gradient(p_interp, time.monotonic() + remaining,
                                            lr_init=0.02, beta_frac_init=0.01,
                                            beta_frac_min=0.0005, lr_min=0.001)
                update_best(a_r)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final polish (remaining time) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_final, _ = _optimize_gradient(a_best_norm, time.monotonic() + remaining,
                                            lr_init=0.005, beta_frac_init=0.002,
                                            beta_frac_min=0.0001, lr_min=0.0003)
            update_best(a_final)

            # One more spectral pass
            if time.monotonic() < deadline - 10 and best_seq is not None:
                a_b = best_seq / best_seq.sum()
                a_spec = _spectral_equalize(a_b, n_iters=100, alpha=0.1)
                update_best(a_spec)

    print(f"Phase 6 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _raised_cosine(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _raised_cosine(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
