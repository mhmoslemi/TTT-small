"""
Optimized step function sequence search using adaptive logsumexp continuation,
active-set subgradient refinement, and multi-scale diversification.
Key improvements: adaptive beta schedule, top-K active set tracking,
second-order L-BFGS with tighter tolerances, and n-doubling exploration.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - separation / 2.0
    c2 = 0.5 + separation / 2.0
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bspline_profile(n, order=4):
    t = np.linspace(0.0, 1.0, n)
    u = t * 0.9 + 0.05
    u = np.clip(u, 0.01, 0.99)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_cosine_basis(n, n_harmonics=20):
    t = np.linspace(0.0, 1.0, n)
    basis = np.ones((n_harmonics, n))
    for k in range(1, n_harmonics):
        basis[k] = np.cos(np.pi * k * t)
    return basis


def _project_onto_simplex(v, target_sum=1.0):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-15:
        return np.ones_like(v) / len(v)
    return v / s * target_sum


def _compute_gradient_weighted(a, weights):
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(weights, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_topk_grad(a, k_top):
    n = len(a)
    c = _full_convolve(a, a)
    k_actual = min(k_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    weights = np.zeros(len(c))
    weights[top_idx] = 1.0 / k_actual
    grad = _compute_gradient_weighted(a, weights)
    return grad * 4.0 * n


def _logsumexp_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * cm
    log_sum = float(np.log(s))
    return 2.0 * n * (cm + beta * log_sum)


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    W = float(np.dot(w, c))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    g = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(g)):
        return np.zeros_like(z)
    return g


def _run_continuation(a_init, deadline, max_levels=30, max_iter_per_level=3000,
                      beta_init=None, beta_decay=0.5, beta_floor=1e-6,
                      ftol=1e-18, gtol=1e-12):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if beta_init is None:
        beta_init = max_c0

    beta = beta_init
    levels_done = 0

    for level in range(max_levels):
        if time.monotonic() > deadline - 5.0:
            break
        if beta < beta_floor:
            break

        def obj(z_):
            return _logsumexp_objective(z_, beta, n)

        def grad_fn(z_):
            return _logsumexp_grad(z_, beta, n)

        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_level, 'ftol': ftol,
                         'gtol': gtol, 'maxls': 80, 'maxfun': max_iter_per_level * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                z = z_new.copy()
        except Exception:
            pass

        c_curr = _full_convolve(_softmax_fn(z), _softmax_fn(z))
        beta = max(beta_decay * beta, beta_floor)
        levels_done += 1

    # Final evaluation
    a_final = _softmax_fn(z)
    s_final = _score_norm_fn(a_final)
    if s_final < best_score:
        best_score = s_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_descent(a_init, deadline, max_iters=50000, lr_init=0.005,

def _active_set_refinement(a_init, deadline, k_values=None, lr=0.002, momentum=0.85):
    n = len(a_init)
    if k_values is None:
        k_values = [1, 2, 3, 5, 10, 20, 50, 100]

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    per_k_time = max(2.0, (deadline - time.monotonic()) / len(k_values))

    for kval in k_values:
        if time.monotonic() > deadline - 2.0:
            break
        k_deadline = min(deadline - 2.0, time.monotonic() + per_k_time)

        vel = np.zeros(n)
        cur_lr = lr
        cur_z = z.copy()

        for it in range(20000):
            if time.monotonic() > k_deadline:
                break
            a_curr = _softmax_fn(cur_z)
            c = _full_convolve(a_curr, a_curr)
            k_actual = min(kval, len(c))
            top_idx = np.argsort(c)[-k_actual:]
            weights = np.zeros(len(c))
            # Weight by how close each lag is to the max
            c_top = c[top_idx]
            c_max = c.max()
            c_min_top = c_top.min()
            if c_max - c_min_top > 1e-15:
                w_vals = (c_top - c_min_top) / (c_max - c_min_top)
                w_vals = w_vals ** 2  # Sharpen
            else:
                w_vals = np.ones(k_actual) / k_actual
            weights[top_idx] = w_vals / w_vals.sum()

            grad = _compute_gradient_weighted(a_curr, weights) * 4.0 * n
            grad = grad * a_curr  # Chain rule through softmax

            vel = momentum * vel + (1.0 - momentum) * grad
            cur_z = cur_z - cur_lr * vel
            cur_z = np.clip(cur_z, -80, 80)
            cur_z = cur_z - cur_z.mean()

            if it % 50 == 0:
                a_new = _softmax_fn(cur_z)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = cur_z.copy()

            cur_lr = max(1e-5, cur_lr * 0.999)

    return best_a, best_score


def _cosine_basis_optimization(a_init, deadline, n_harmonics=50, max_iter=5000):
    """Optimize in a cosine basis: a = base + sum(c_k * cos_k), then project to simplex."""
    n = len(a_init)
    base = a_init.copy()
    basis = _make_cosine_basis(n, n_harmonics)

    coeffs = np.zeros(n_harmonics)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    def obj_coeff(c_):
        a_ = base + basis.T @ c_
        a_ = np.maximum(a_, 0.0)
        s = a_.sum()
        if s < 1e-15:
            return 1e10
        a_ = a_ / s
        return _score_norm_fn(a_)

    def grad_coeff(c_):
        a_ = base + basis.T @ c_
        a_ = np.maximum(a_, 1e-15)
        s = a_.sum()
        a_norm = a_ / s
        c = _full_convolve(a_norm, a_norm)
        cm = c.max()
        # Subgradient at max
        idx = int(np.argmax(c))
        grad_a = np.zeros(n)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        for j in range(lo, hi + 1):
            grad_a[j] += 2.0 * a_norm[idx - j]
        grad_a = 2.0 * n * grad_a / s - 2.0 * n * a_norm * c.max() / s
        grad_c = basis @ grad_a
        return grad_c

    try:
        result = minimize(
            obj_coeff, coeffs.copy(), jac=grad_coeff,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 20000}
        )
        if result.x is not None:
            a_new = base + basis.T @ result.x
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-10:
                a_new = a_new / s
                sc = _score_norm_fn(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_score

        max_c_curr = float(c_curr.max())
        beta_start = max_c_curr * 0.3

        a_b, sc_b = _run_continuation(best_seq, phase3_end - 5.0,
                                       max_levels=30, max_iter_per_level=3000,
                                       beta_init=beta_start,
                                       beta_decay=0.45, beta_floor=1e-7,
                                       ftol=1e-19, gtol=1e-13)
        update_best(a_b, sc_b, "cont_fine")

        # Second fine pass with smaller beta start
        if time.monotonic() < phase3_end - 60:
            beta_start2 = max_c_curr * 0.1
            dl2 = phase3_end - 5.0
            a_b2, sc_b2 = _run_continuation(best_seq, dl2,
                                             max_levels=20, max_iter_per_level=2000,
                                             beta_init=beta_start2,
                                             beta_decay=0.4, beta_floor=1e-8,
                                             ftol=1e-19, gtol=1e-13)
            update_best(a_b2, sc_b2, "cont_fine2")

    print(f"Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Active-set / subgradient refinement (250s) ===
    phase4_end = min(deadline, t_start + 1170.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: active-set refinement, start={_score_norm_fn(best_seq):.10f}")
        dl = phase4_end - 5.0

        a_b, sc_b = _active_set_refinement(best_seq, dl,
                                            k_values=[1, 2, 3, 5, 10, 20, 50],
                                            lr=0.001, momentum=0.85)
        update_best(a_b, sc_b, "active_set")

        if time.monotonic() < phase4_end - 30:
            dl2 = phase4_end - 5.0
            a_b2, sc_b2 = _subgradient_descent(best_seq, dl2,
                                                max_iters=30000, lr_init=0.003,
                                                k_top=3, momentum=0.92, lr_decay=0.9997)
            update_best(a_b2, sc_b2, "subgrad")

    print(f"Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Stochastic Adam with adaptive K (200s) ===
    phase5_end = min(deadline, t_start + 1370.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"Phase 5: stochastic Adam, start={_score_norm_fn(best_seq):.10f}")
        dl = phase5_end - 5.0
        a_b, sc_b = _dual_processive_charge(best_seq, dl, rng, n_trials=40000)
        update_best(a_b, sc_b, "stochastic_adam")

    print(f"Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Cosine basis optimization (100s) ===
    phase6_end = min(deadline, t_start + 1470.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print(f"Phase 6: cosine basis, start={_score_norm_fn(best_seq):.10f}")
        dl = phase6_end - 5.0
        n_harm = min(40, len(best_seq) // 50)
        a_b, sc_b = _cosine_basis_optimization(best_seq, dl, n_harmonics=n_harm, max_iter=3000)
        update_best(a_b, sc_b, "cosine_basis")

    print(f"Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Diversification - noisy restarts and n-change (150s) ===
    phase7_end = min(deadline, t_start + 1620.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        print(f"Phase 7: diversification, best={best_score:.10f}")

        # Noisy restarts
        for i in range(3):
            if time.monotonic() > phase7_end - 20:
                break
            noise = 0.005 + 0.005 * i
            dl = min(phase7_end - 10, time.monotonic() + 30.0)
            a_b, sc_b = _noisy_restart_optimize(best_seq, dl, rng, noise_scale=noise)
            update_best(a_b, sc_b, f"restart_{i}")

        # Try n*2
        n_curr = len(best_seq)
        if n_curr <= 8192 and time.monotonic() < phase7_end - 40:
            n_new = n_curr * 2
            p_interp = _interp_profile(best_seq, n_new)
            sc_interp = _score_norm_fn(p_interp)
            print(f"  Interp to n={n_new}: score={sc_interp:.10f}")
            update_best(p_interp, sc_interp, f"interp_n{n_new}")

            if time.monotonic() < phase7_end - 20:
                dl = min(phase7_end - 15, time.monotonic() + 40.0)
                a_b, sc_b = _run_continuation(p_interp, dl, max_levels=10,
                                               max_iter_per_level=1000, beta_decay=0.5)
                update_best(a_b, sc_b, f"cont_n{n_new}")

    print(f"Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Final intensive polish (remaining time) ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic() - 30.0)
        print(f"Phase 8: final polish, remaining={remaining:.0f}s")

        # 8a: Very fine continuation
        if remaining > 40:
            c_curr = _full_convolve(best_seq, best_seq)
            max_c = float(c_curr.max())
            dl = min(deadline - 35, time.monotonic() + remaining * 0.4)
            a_b, sc_b = _run_continuation(best_seq, dl,
                                           max_levels=20, max_iter_per_level=2000,
                                           beta_init=max_c * 0.1,
                                           beta_decay=0.35, beta_floor=1e-9,
                                           ftol=1e-20, gtol=1e-14)
            update_best(a_b, sc_b, "final_cont")

        # 8b: Active set
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.25)
            a_b, sc_b = _active_set_refinement(best_seq, dl,
                                                k_values=[1, 2, 3, 5, 10, 30, 100],
                                                lr=0.0005, momentum=0.9)
            update_best(a_b, sc_b, "final_active")

        # 8c: Short stochastic
        if time.monotonic() < deadline - 25:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.15)
            a_b, sc_b = _dual_processive_charge(best_seq, dl, rng, n_trials=10000)
            update_best(a_b, sc_b, "final_stochastic")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = _score_norm_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 32768:
            try:
                sc_eval = evaluate_sequence([float(x) for x in final_seq])
                print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    fallback = _make_triangular_weight(8192)
    print(f"FALLBACK: n=8192, score={_score_norm_fn(fallback):.10f}")
    return [float(x) for x in fallback]
