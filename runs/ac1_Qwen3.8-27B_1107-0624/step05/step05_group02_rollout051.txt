def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def compute_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    z = z - np.max(z)

    if s < 1e-300 or not np.isfinite(s):

    return e / s

    bounds = [(0, None)] * n

        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g, M
    except Exception:
        pass
    
    return None, M

        vh = v_buf / (1.0 - b2 ** t_step)

        cur_lr = lr * (1.0 + 0.5 * np.cos(np.pi * progress))
        update = cur_lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > max_update:
            update = update * (max_update / un)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

    a_final = _safe_softmax(z)
    c_final = _full_convolve(a_final, a_final)
    cm_final = float(c_final.max())
    if cm_final < best_score:
        best_score = cm_final
        best_a = a_final.copy()

    return best_a, best_score


def _lp_descent(f, n, deadline, max_steps=12, lp_time=25):
    """LP-based descent with backtracking. Returns improved f."""
    f = f.astype(np.float64).copy()
    f = f / f.sum()
    current_t = 0.3
    A_cache = None
    f_for_matrix = None
    stuck = 0

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(f, f)
        M = float(c.max())

        if A_cache is None or np.max(np.abs(f - f_for_matrix)) > 1e-12:
            A_cache = _build_lp_matrix(f, n)
            f_for_matrix = f.copy()

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds_lp = [(0, None)] * n
        g = None
        try:
            res = linprog(c_obj, A_ub=A_cache, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'time_limit': lp_time, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
                if g.sum() < 1e-15:
                    g = None
        except Exception:
            g = None

        if g is None:
            stuck += 1
            if stuck > 3:
                break
            continue

        g = g / g.sum()
        improved = False
        t_try = current_t
        for _ in range(8):
            f_new = (1 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0)
            f_new_sum = f_new.sum()
            if f_new_sum < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / f_new_sum
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())
            if M_new < M * (1 - 1e-10):
                f = f_new
                current_t = min(1.0, t_try * 1.3)
                improved = True
                break
            else:
                t_try *= 0.5

        if not improved:
            stuck += 1
            current_t *= 0.5
            if stuck > 5:
                break

    return f


def _composite_objective_grad(a, c, top_k, beta):
    """Gradient using top-k weighted softmax for composite objective."""
    n = len(a)
    cm = c.max()
    c_norm = (c - cm) / beta
    idx = np.argsort(c)[::-1][:top_k]
    w = np.zeros_like(c)
    w[idx] = np.exp(c_norm[idx])
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / w_sum
    return _grad_from_weights(a, w, 2.0)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        print(f"Parent: n={n_main}, score={_score(a0):.10f}")
        update_best(a0)
    else:
        n_main = 8192
        a0 = _make_uniform(n_main)
        print(f"No parent, using uniform n={n_main}")
        update_best(a0)

    z0 = np.log(np.maximum(a0, 1e-15))
    z0 = z0 - z0.mean()

    # === Phase 1: Fast Adam exploration (0 - 400s) ===
    phase1_end = min(deadline, t_start + 400.0)
    print(f"\n=== Phase 1: Adam exploration (n={n_main}) ===")

    # Run 1: Broad beta (explore)
    a1, s1 = _adam_optimize(z0, phase1_end - 100, lr=0.05, beta_init=None,
                            beta_min=None, use_power_sum=False, max_update=1.5)
    update_best(a1)

    # Run 2: Sharper beta from result 1
    if time.monotonic() < phase1_end - 10:
        z1 = np.log(np.maximum(a1, 1e-15))
        z1 = z1 - z1.mean()
        a2, s2 = _adam_optimize(z1, phase1_end - 5, lr=0.02, beta_init=None,
                                beta_min=None, use_power_sum=False, max_update=0.8)
        update_best(a2)

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 2: LP-based descent (400s - 900s) ===
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\n=== Phase 2: LP descent ===")

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        f_lp = best_seq / best_seq.sum()
        f_lp = _lp_descent(f_lp, n_main, phase2_end - 10, max_steps=15, lp_time=25)
        update_best(f_lp)

        # Adam polish after LP
        if time.monotonic() < phase2_end - 20:
            z_lp = np.log(np.maximum(f_lp, 1e-15))
            z_lp = z_lp - z_lp.mean()
            a_lp, s_lp = _adam_optimize(z_lp, phase2_end - 5, lr=0.01,
                                        beta_init=None, beta_min=None,
                                        use_power_sum=False, max_update=0.5)
            update_best(a_lp)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Power-sum + composite objective (900s - 1300s) ===
    phase3_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 3: Power-sum & composite ===")

    if best_seq is not None and time.monotonic() < phase3_end - 30:
        z_best = np.log(np.maximum(best_seq / best_seq.sum(), 1e-15))
        z_best = z_best - z_best.mean()

        # Power-sum run
        a_ps, s_ps = _adam_optimize(z_best, min(phase3_end - 100, time.monotonic() + 200),
                                    lr=0.03, use_power_sum=True, p_start=4, p_end=32,
                                    max_update=1.0)
        update_best(a_ps)

        # Composite (top-k) approach
        if time.monotonic() < phase3_end - 20:
            a_cur = best_seq / best_seq.sum()
            n = len(a_cur)
            c_cur = _full_convolve(a_cur, a_cur)
            cm_cur = float(c_cur.max())

            # Manual loop with top-k
            z_cur = np.log(np.maximum(a_cur, 1e-15))
            z_cur = z_cur - z_cur.mean()
            m_buf = np.zeros(n)
            v_buf = np.zeros(n)
            t_step = 0
            b1, b2, eps = 0.9, 0.999, 1e-8
            top_k_list = [32, 64, 128, 256]
            lr_cur = 0.015

            for top_k in top_k_list:
                if time.monotonic() > phase3_end - 15:
                    break
                for it in range(3000):
                    if time.monotonic() > phase3_end - 15:
                        break
                    a = _safe_softmax(z_cur)
                    c = _full_convolve(a, a)
                    cm = float(c.max())
                    if cm < best_score:
                        best_score = cm
                        best_seq = a.copy()
                        print(f"  Composite top_k={top_k}: {best_score:.10f} t={time.monotonic()-t_start:.1f}s")

                    beta_k = cm * 0.05
                    grad_a = _composite_objective_grad(a, c, top_k, beta_k)
                    if not np.all(np.isfinite(grad_a)):
                        break
                    grad_z = a * (grad_a - float(np.dot(a, grad_a)))
                    t_step += 1
                    m_buf = b1 * m_buf + (1 - b1) * grad_z
                    v_buf = b2 * v_buf + (1 - b2) * (grad_z * grad_z)
                    mh = m_buf / (1 - b1 ** t_step)
                    vh = v_buf / (1 - b2 ** t_step)
                    upd = lr_cur * mh / (np.sqrt(vh) + eps)
                    un = np.linalg.norm(upd)
                    if un > 0.8:
                        upd = upd * (0.8 / un)
                    z_cur = z_cur - upd
                    z_cur = np.clip(z_cur, -50, 50)
                    z_cur = z_cur - z_cur.mean()
                lr_cur *= 0.7

            a_final3 = _safe_softmax(z_cur)
            update_best(a_final3)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart (1300s - 1650s) ===
    phase4_end = min(deadline, t_start + 1650.0)
    print(f"\n=== Phase 4: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        a_norm = best_seq / best_seq.sum()
        n_restarts = 6
        time_per_restart = (phase4_end - time.monotonic() - 20) / max(1, n_restarts)
        sigmas = [0.02, 0.04, 0.08, 0.12, 0.2, 0.35]
        use_ps = [False, False, True, False, True, False]

        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 20:
                break
            p_start_i = _perturb_log(a_norm, sigmas[i], rng)
            dl_i = min(time.monotonic() + time_per_restart, phase4_end - 5)
            a_r, s_r = _adam_optimize(np.log(np.maximum(p_start_i, 1e-15)),
                                      dl_i, lr=0.015,
                                      beta_init=None, beta_min=None,
                                      use_power_sum=use_ps[i],
                                      max_update=1.0)
            update_best(a_r)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Different-n exploration + final LP (1650s - 1770s) ===
    phase5_end = min(deadline, t_start + 1770.0)
    print(f"\n=== Phase 5: Different-n + final polish ===")

    if best_seq is not None and time.monotonic() < phase5_end - 20:
        # Try smaller n (might find better "shape" at lower resolution)
        if len(best_seq) > 4096:
            p_4096 = _interp_profile(best_seq, 4096)
            update_best(p_4096)
            if time.monotonic() < phase5_end - 30:
                z_4096 = np.log(np.maximum(p_4096, 1e-15))
                z_4096 = z_4096 - z_4096.mean()
                a_4096, _ = _adam_optimize(z_4096, min(time.monotonic() + 60, phase5_end - 10),
                                           lr=0.01, use_power_sum=False, max_update=0.5)
                update_best(a_4096)

        # Final LP on best
        if time.monotonic() < phase5_end - 30:
            f_final = best_seq / best_seq.sum()
            n_f = len(f_final)
            f_final = _lp_descent(f_final, n_f, phase5_end - 5, max_steps=5, lp_time=20)
            update_best(f_final)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final validation (1770s - deadline) ===
    print(f"\n=== Phase 6: Final ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
