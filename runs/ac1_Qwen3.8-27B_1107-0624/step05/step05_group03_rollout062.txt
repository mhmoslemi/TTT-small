"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy:
1. Initialize with binomial coefficient sequence (theoretically optimal for flat conv).
2. Optimize using Adam with logsumexp continuation (smooth -> exact max).
3. Peak-shaving local search for final refinement.
4. Multi-n exploration and multi-restart for diversity.
"""
import numpy as np
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    return irfft(rfft(a, N) * rfft(b, N), N)[:out_n]


def _conv_self(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:out_n]


def _grad_logsumexp_self(a, beta, n_scale=1.0):
    """Gradient of n_scale * beta * logsumexp(conv(a,a)/beta) w.r.t. a."""
    n = len(a)
    c = _conv_self(a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w = e / s
    # g[i] = 2 * n_scale * sum_j w[i+j] * a[j]
    # cross-correlation of w and a
    N = _next_pow2(len(w) + n - 1)
    W = rfft(w, N)
    A = rfft(a, N)
    g = irfft(W * A.conj(), N)[:n]
    g *= 2.0 * n_scale
    return g, c.max(), w


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _binom_seq(n):
    a = np.ones(n, dtype=np.float64)
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def _truncated_binom(n, taper_power):
    a = _binom_seq(n)
    t = np.linspace(0.0, 1.0, n)
    w = (1.0 - t) ** taper_power
    a = a * w
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return _binom_seq(n)
    return a / s


def _gaussian_tail(n, width_frac):
    a = _binom_seq(n)
    t = np.arange(n, dtype=np.float64)
    w = np.exp(-t / (width_frac * n))
    a = a * w
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s


def _adam_optimize(a_init, deadline, betas, lr_init=0.05, lr_decay=0.9995):
    """Adam optimizer with logsumexp continuation."""
    n = len(a_init)
    # Parameterize as log a (with exp to ensure positivity)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()  # Center for numerical stability
    
    best_a = a_init.copy()
    best_score = _score(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        iters_per_beta = 0
        while time.monotonic() < deadline - 5:
            t += 1
            a = np.exp(z - z.max())  # softmax-like for stability
            a = a / a.sum()
            
            grad, cmax, w = _grad_logsumexp_self(a, beta, n_scale=2.0 * n)
            
            # Ensure gradient is finite
            if not np.all(np.isfinite(grad)):
                break
            
            m = b1 * m + (1 - b1) * grad
            v = b2 * v + (1 - b2) * (grad * grad)
            mh = m / (1 - b1 ** t)
            vh = v / (1 - b2 ** t)
            z = z - lr * mh / (np.sqrt(vh) + eps)
            z = z - z.mean()
            z = np.clip(z, -50, 50)

                sc = _score(a_check)

                    best_a = a_check.copy()

        sc = _score(a_final)
        if np.isfinite(sc) and sc < best_score:

            best_a = a_final.copy()
    
    return best_a, best_score


def _peak_shave(a_init, deadline, n_trials=50000):
    """Local search that targets the highest convolution peaks."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _conv_self(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 2:
            break
        
        # Find the top peaks
        K = min(8, len(c))
        top_idx = np.argsort(c)[-K:]
        
        # For each top peak, find the contributing pair and "shave" it
        for k_val in top_idx:
            k_val = int(k_val)
            if time.monotonic() > deadline - 2:
                break
            
            # Find the pair (i, k-i) with the largest product
            i_min = max(0, k_val - (n - 1))
            i_max = min(n - 1, k_val)
            if i_min > i_max:
                continue
            
            pairs = np.arange(i_min, i_max + 1)
            partners = k_val - pairs
            products = a[pairs] * a[partners]
            best_pair = pairs[int(np.argmax(products))]
            best_partner = k_val - best_pair
            
            # Reduce the larger of the pair
            a_trial = a.copy()
            if a[best_pair] >= a[best_partner]:
                delta = a[best_pair] * 0.1
                a_trial[best_pair] -= delta
            else:
                delta = a[best_partner] * 0.1
                a_trial[best_partner] -= delta
            
            # Redistribute to a "safe" position (low contribution to peaks)
            # Find position with minimum total contribution to top peaks
            contrib = np.zeros(n)
            for kk in top_idx[:3]:
                kk = int(kk)
                lo = max(0, kk - (n - 1))
                hi = min(n - 1, kk)
                if lo <= hi:
                    contrib[lo:hi+1] += a[kk - np.arange(lo, hi+1)]  # a[partner]
            
            # Pick from the lowest contribution positions
            safe = np.argsort(contrib)[:max(5, n // 10)]
            add_idx = int(safe[np.random.randint(len(safe))])
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            
            s = a_trial.sum()
            if s > 1e-15:
                a_trial /= s
                sc = _score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    c = _conv_self(a)
                    break  # Restart the peak loop with new c
    
    return best_a, best_sc

"""
Optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Strategy:
1. Initialize with binomial coefficient sequence (theoretically gives pi/2 ≈ 1.5708).
2. Optimize using mirror descent (log-space Adam) with logsumexp continuation.
3. Peak-shaving local search targeting specific contributing pairs.
4. Multi-n exploration and multi-restart for diversity.
Key insight: gradient in log-space is a[k]*(g[k]-dot(a,g)) where g=2n*cross_corr(softmax(c/beta),a).
"""
import numpy as np
from scipy.fft import rfft, irfft
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _conv_self(a):
    n = len(a)
    out_n = 2 * n - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    return irfft(fa * fa, N)[:out_n]


def _cross_corr(w, a):
    """Compute cross-correlation: out[i] = sum_j w[i+j]*a[j] for i=0..len(a)-1."""
    n = len(a)
    N = _next_pow2(len(w) + n - 1)
    W = rfft(w, N)
    A = rfft(a, N)
    return irfft(W * A.conj(), N)[:n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _conv_self(a)
    return float(2.0 * n * c.max() / (s * s))


def _binom_seq(n):
    """a[k] = binom(2k,k)/4^k normalized to sum=1."""
    a = np.ones(n, dtype=np.float64)
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    s = a.sum()
    return a / s


def _trunc_binom(n, power):
    a = _binom_seq(n)
    t = np.linspace(0.0, 1.0, n)
    w = (1.0 - t) ** power
    a = a * w
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return _binom_seq(n)
    return a / s


def _power_seq(n, alpha):
    """a[k] = (1 + k)^(-alpha) normalized."""
    k = np.arange(n, dtype=np.float64)
    a = (1.0 + k) ** (-alpha)
    return a / a.sum()


def _exp_seq(n, rate):
    k = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * k / n)
    return a / a.sum()


def _mixed_init(n, frac_left):
    """Mix of left-concentrated and right-concentrated mass."""
    a_left = _binom_seq(n)
    a_right = a_left[::-1].copy()
    a = frac_left * a_left + (1.0 - frac_left) * a_right
    return a / a.sum()


def _mirror_adam(a_init, deadline, betas, lr_init=0.05, momentum=0.9, beta2=0.999):
    """Mirror descent (log-space) with Adam on logsumexp-conv objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.max()

    best_a = a_init.copy()
    best_score = _score(a_init)
    print(f"    Mirror-Adam start: score={best_score:.10f}")

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    eps = 1e-8

    for beta in betas:
        if time.monotonic() > deadline - 3:
            break

        # Adaptive max iterations per beta level
        max_iter = 50000
        stall = 0
        prev_sc = best_score

        for it in range(max_iter):
            if time.monotonic() > deadline - 3:
                break

            t += 1
            a = np.exp(z)
            a = a / a.sum()

            # Compute objective and gradient
            c = _conv_self(a)
            c_max = float(c.max())
            c_shifted = (c - c_max) / beta
            e = np.exp(c_shifted)
            s_e = e.sum()
            if s_e < 1e-300:
                w = np.zeros_like(c)
                w[int(c.argmax())] = 1.0
            else:
                w = e / s_e

            # g[i] = 2n * sum_j w[i+j] * a[j]
            cc = _cross_corr(w, a)
            g = 2.0 * n * cc

            # Gradient in z-space: a[k] * (g[k] - dot(a, g))
            dot_ag = float(np.dot(a, g))
            grad_z = a * (g - dot_ag)

            # Adam update
            m_buf = momentum * m_buf + (1.0 - momentum) * grad_z
            v_buf = beta2 * v_buf + (1.0 - beta2) * (grad_z * grad_z)
            m_hat = m_buf / (1.0 - momentum ** t)
            v_hat = v_buf / (1.0 - beta2 ** t)
            lr = lr_init / (1.0 + t * 0.001)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            z = z - z.max()
            z = np.clip(z, -60, 60)

            # Check improvement periodically
            if (it + 1) % 20 == 0:
                a_check = np.exp(z)
                a_check = a_check / a_check.sum()
                sc = _score(a_check)
                if sc < best_score:
                    if best_score - sc > 1e-6:
                        stall = 0
                    best_score = sc
                    best_a = a_check.copy()
                elif prev_sc - sc < 1e-8:
                    stall += 1
                else:
                    stall = 0
                prev_sc = sc

                # Staleness detection: restart with perturbation
                if stall > 30:
                    z = z + 0.1 * np.random.randn(n)
                    z = z - z.max()
                    stall = 0
                    m_buf *= 0.5
                    v_buf *= 0.5

        # End of beta level
        a_final = np.exp(z)
        a_final = a_final / a_final.sum()
        sc = _score(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
        print(f"    Beta={beta:.4f}: score={best_score:.10f}")

    return best_a, best_score


def _peak_shave(a_init, deadline, max_trials=50000, rng=None):
    """Local search: find top convolution peaks and reduce contributing pairs."""
    if rng is None:
        rng = np.random.default_rng(12345)

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    c = _conv_self(a)
    no_improve = 0

    for it in range(max_trials):
        if time.monotonic() > deadline - 2:
            break

        K = min(10, len(c))
        top_idx = np.argsort(c)[-K:]
        max_c = c[top_idx[-1]]

        # Randomly pick one of the top peaks to attack
        target_k = int(top_idx[rng.integers(K)])

        i_min = max(0, target_k - (n - 1))
        i_max = min(n - 1, target_k)
        if i_min > i_max:
            continue

        pairs = np.arange(i_min, i_max + 1)
        partners = target_k - pairs
        products = a[pairs] * a[partners]
        best_pi = pairs[int(np.argmax(products))]
        best_partner = target_k - best_pi

        # Compute "danger score" for each position (contribution to top peaks)
        danger = np.zeros(n)
        n_top = min(5, K)
        for kk in top_idx[:n_top]:
            kk = int(kk)
            lo = max(0, kk - (n - 1))
            hi = min(n - 1, kk)
            if lo <= hi:
                idxs = np.arange(lo, hi + 1)
                danger[idxs] += a[kk - idxs]

        # Strategy: reduce the largest contributor at the target peak
        a_trial = a.copy()
        if a[best_pi] >= a[best_partner]:
            frac = 0.05 + 0.1 * rng.random()
            delta = a[best_pi] * frac
            a_trial[best_pi] -= delta
        else:
            frac = 0.05 + 0.1 * rng.random()
            delta = a[best_partner] * frac
            a_trial[best_partner] -= delta
        a_trial = np.maximum(a_trial, 0.0)

        # Redistribute to a "safe" position (low danger)
        n_safe = max(10, n // 8)
        safe_idx = np.argsort(danger)[:n_safe]
        add_pos = int(safe_idx[rng.integers(len(safe_idx))])
        a_trial[add_pos] += delta

        s = a_trial.sum()
        if s > 1e-15:
            a_trial /= s
            sc = _score(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                c = _conv_self(a)
                no_improve = 0
            else:
                no_improve += 1

        # Periodically try random perturbations to escape
        if it % 500 == 499:
            a_trial = a * (1.0 + 0.05 * rng.standard_normal(n))
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial /= s
                sc = _score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    c = _conv_self(a)

    return best_a, best_sc


def _lbfgs_refine(a_init, deadline, beta_final=0.001):
    """Quick L-BFGS refinement using scipy."""
    from scipy.optimize import minimize

    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-30))
    z0 = z0 - z0.max()
    best_a = a_init.copy()
    best_sc = _score(a_init)

    def obj(z_):
        a_ = np.exp(z_ - z_.max())
        s = a_.sum()
        if s < 1e-300:
            return np.inf
        a_ = a_ / s
        c_ = _conv_self(a_)
        cm = c_.max()
        shifted = (c_ - cm) / beta_final
        e_ = np.exp(shifted)
        se = e_.sum()
        if se < 1e-300:
            return 2.0 * n * cm
        return 2.0 * n * (cm + beta_final * np.log(se))

    def grad(z_):
        a_ = np.exp(z_ - z_.max())
        s = a_.sum()
        if s < 1e-300:
            return np.zeros(n)
        a_ = a_ / s
        c_ = _conv_self(a_)
        cm = c_.max()
        shifted = (c_ - cm) / beta_final
        e_ = np.exp(shifted)
        se = e_.sum()
        if se < 1e-300:
            w = np.zeros_like(c_)
            w[int(c_.argmax())] = 1.0
        else:
            w = e_ / se
        cc = _cross_corr(w, a_)
        g = 2.0 * n * cc
        dot_ag = float(np.dot(a_, g))
        return a_ * (g - dot_ag)

    try:
        res = minimize(obj, z0.copy(), jac=grad, method='L-BFGS-B',
                       options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-10,
                                'maxfun': 5000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            a_new = np.exp(res.x - res.x.max())
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
    except Exception:
        pass

    return best_a, best_sc


def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a / a.sum()
    return a_new / s

    # Try smaller n
    for n_small in [2048, 4096]:
        if time.monotonic() > p0_end:
            break
        a_s = _binom_seq(n_small)
        sc_s = _score(a_s)
        init_list.append((sc_s, a_s, f"binom_n{n_small}"))
        if sc_s < best_score:
            best_score = sc_s
            best_seq = a_s.copy()

    init_list.sort(key=lambda x: x[0])
    print(f"  Init scores (top 5):")
    for i in range(min(5, len(init_list))):
        print(f"    {init_list[i][2]}: {init_list[i][0]:.10f}")
    print(f"  Best init: {best_score:.10f}, n={len(best_seq)}")

    # === Phase 1: Mirror-Adam optimization (700s) ===
    p1_end = min(deadline, t_start + 730.0)
    print("=== Phase 1: Mirror-Adam ===")

    # Use the best initialization
    a_start = best_seq.copy()
    c_start = _conv_self(a_start)
    max_c_start = float(c_start.max())

    # Beta schedule (relative to max_c)
    beta_fracs = [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    betas = [b * max_c_start for b in beta_fr

    a_opt, sc_opt = _mirror_adam(a_start, p1_end, betas, lr_init=0.04)
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
    print(f"  After Adam: {best_score:.10f}")

    # === Phase 2: L-BFGS polish (150s) ===
    p2_end = min(deadline, t_start + 880.0)
    print("=== Phase 2: L-BFGS polish ===")
    if time.monotonic() < p2_end - 10:
        a_lb, sc_lb = _lbfgs_refine(best_seq.copy(), p2_end, beta_final=0.001 * max_c_start)
        if sc_lb < best_score:
            best_score = sc_lb
            best_seq = a_lb.copy()
            print(f"  After L-BFGS: {best_score:.10f}")

    # === Phase 3: Peak shaving (350s) ===
    p3_end = min(deadline, t_start + 1230.0)
    print("=== Phase 3: Peak shaving ===")
    if time.monotonic() < p3_end - 10:
        a_sh, sc_sh = _peak_shave(best_seq.copy(), p3_end, max_trials=80000, rng=rng)
        if sc_sh < best_score:
            best_score = sc_sh
            best_seq = a_sh.copy()
            print(f"  After peak shave: {best_score:.10f}")

        # Another round with different seed
        if time.monotonic() < p3_end - 30:
            rng2 = np.random.default_rng(int(seed) + 777)
            a_sh2, sc_sh2 = _peak_shave(best_seq.copy(), min(p3_end, time.monotonic() + 120), max_trials=30000, rng=rng2)
            if sc_sh2 < best_score:
                best_score = sc_sh2
                best_seq = a_sh2.copy()
                print(f"  After peak shave 2: {best_score:.10f}")

    p4_end = min(deadline, t_start + 1430.0)
    print("=== Phase 4: Multi-n ===")
    if time.monotonic() < p4_end - 10:
        n_curr = len(best_seq)
        for target_n in [n_curr * 2, n_curr // 2, 16384, 4096]:
            if target_n < 1024 or target_n > 32768 or target_n == n_curr:
                continue
            if time.monotonic() > p4_end - 40:
                break

            a_interp = _interp(best_seq, target_n)
            sc_interp = _score(a_interp)
            print(f"  n={target_n} interp: {sc_interp:.10f}")

            if sc_interp < best_score:
                best_score = sc_interp
                best_seq = a_interp.copy()

            # Quick optimization at new n
            if time.monotonic() < p4_end - 30:
                dl_quick = min(p4_end - 20, time.monotonic() + 60.0)
                c_q = _conv_self(a_interp)
                mc_q = float(c_q.max())
                betas_q = [b * mc_q for b in [0.1, 0.05, 0.01, 0.005]]
                a_q, sc_q = _mirror_adam(a_interp, dl_quick, betas_q, lr_init=0.03)
                if sc_q < best_score:
                    best_score = sc_q
                    best_seq = a_q.copy()
                    print(f"  n={target_n} optimized: {best_score:.10f}")

    p5_end = min(deadline, t_start + 1580.0)
    print("=== Phase 5: Multi-restart ===")
    if time.monotonic() < p5_end - 10:
        n_restarts = 4
        per_restart = max(15.0, (p5_end - time.monotonic() - 10) / n_restarts)

        for i in range(n_restarts):

            z = np.log(np.maximum(best_seq, 1e-30))
            z = z - z.max()
            z = z + noise * rng.standard_normal(len(best_seq))
            z = np.clip(z, -50, 50)
            a_r = np.exp(z)
            a_r = a_r / a_r.sum()

            dl_r = min(p5_end - 5, time.monotonic() + per_restart)
            c_r = _conv_self(a_r)
            mc_r = float(c_r.max())
            betas_r = [b * mc_r for b in [0.1, 0.05, 0.01, 0.005, 0.002]]
            a_ro, sc_ro = _mirror_adam(a_r, dl_r, betas_r, lr_init=0.03)

    if time.monotonic() < deadline - 5:

            best_seq = a_fin.copy()

        sc_final = _score(final_seq)

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Evaluator error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _binom_seq(8192)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
