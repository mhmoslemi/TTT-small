```python
    Returns gradient of length n."""
    n = len(a)
    N = _next_pow2(2 * (2*n-1))  # Need N >= 2n-1 + n - 1 = 3n-2, but 2n-1 is enough for each conv
    
    # Actually for conv(a, w): length is n + (2n-1) - 1 = 3n-2
    # So N >= 3n-2
    N = _next_pow2(3 * n - 2)
    
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    fw_rev = np.fft.rfft(w[::-1], N)
    
    conv_aw = np.fft.irfft(fa * fw, N)  # length 3n-2, indices 0 to 3n-3
    conv_awr = np.fft.irfft(fa * fw_rev, N)  # length 3n-2
    
    # g2[m] = conv(a, w)[m] for m in [0, n-1]
    g2 = conv_aw[:n]
    
    # g1[m] = conv(a, w_rev)[2n-2-m] for m in [0, n-1]
    # 2n-2-m ranges from 2n-2 (m=0) to n-1 (m=n-1)
    g1 = conv_awr[n-1:2*n-1][::-1]
    
    return g1 + g2
```
