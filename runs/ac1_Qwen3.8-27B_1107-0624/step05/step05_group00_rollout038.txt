"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2:
Phase 1: FISTA with softmax-smoothed max (correct gradient)
Phase 2: Adam in log-space (correct gradient)
Phase 3: Nesterov-accelerated exact subgradient with Polyak step size
Phase 4: Final ultra-fine Adam polish
Uses n=8192 for maximum optimization surface.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _grad_w_dot_c(a, w):
    """Compute gradient of sum_m w[m]*c[m] where c=conv(a,a).
    Result: 2*xcorr(a,w) where xcorr[i]=sum_k a[k]*w[i+k]."""
    n = len(a)
    w_rev = w[::-1]
    N = _next_pow2(3 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w_rev, N)
    conv_full = np.fft.irfft(fa * fw, N)[:3 * n - 2]
    xcorr = conv_full[n - 1:2 * n - 1][::-1]
    return 2.0 * xcorr


def _grad_logspace(a, w, G):
    """Convert gradient w.r.t. a to gradient w.r.t. log-space z.
    dL/dz[i] = a[i]*(G[i] - dot(a,G))"""
    scalar = float(np.dot(a, G))
    return a * (G - scalar)


def _fista_phase(a_init, beta_list, deadline, max_iter_per_beta):
    """FISTA with softmax-smoothed max, cycling through beta values."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    best_a = a_curr.copy()
    best_sc = _score_norm(a_curr)
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        
        L = 8.0 * n / beta  # Lipschitz constant estimate
        step = 1.0 / L
        
        for it in range(max_iter_per_beta):
            if time.monotonic() > deadline:
                break
            
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            s_y = y.sum()
            if s_y > 1e-30:
                y = y / s_y
            else:
                y = np.ones(n) / n
            
            c_y = _full_convolve(y, y)
            c_max = float(c_y.max())
            
            # Compute softmax weights
            shifted = (c_y - c_max) / beta
            e = np.exp(shifted)
            w_sum = float(e.sum())
            if w_sum < 1e-300 or not np.isfinite(w_sum):
                w = np.zeros_like(c_y)
                w[int(np.argmax(c_y))] = 1.0
            else:
                w = e / w_sum
            
            G = _grad_w_dot_c(y, w)
            
            a_next = _proj_simplex(y - step * G)
            
            # Check score periodically
            if it % 200 == 0:
                sc = _score_norm(a_next)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_next.copy()
            
            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next
    
    sc = _score_norm(a_curr)
    if sc < best_sc:
        best_sc = sc
        best_a = a_curr.copy()
    
    return best_a, best_sc


def _adam_logspace_phase(a_init, beta_list, deadline, lr_init, max_iter):
