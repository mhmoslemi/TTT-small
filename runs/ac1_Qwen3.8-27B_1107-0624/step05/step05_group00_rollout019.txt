def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):

    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _soft_max_weights(c, beta):
    """Compute softmax weights for the autocorrelation."""
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s

def _grad_a(c, w, a):
    """Gradient of smoothed max w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    return 2.0 * h

def _grad_z(a, c, w, n):
    """Gradient of 2n * smoothed_max w.r.t. log-parameters z."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _fista_simplex(a_init, beta, deadline, max_iter=100000):
    """FISTA on the simplex with softmax-smoothed max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta  # Lipschitz constant (rough)
    step = 1.0 / L
    best_a = a_curr.copy()
    best_sc = 2.0 * n * float(_full_convolve(a_curr, a_curr).max())
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n
        
        c_y = _full_convolve(y, y)
        w = _soft_max_weights(c_y, beta)
        grad = _grad_a(c_y, w, y)
        
        a_next = _proj_simplex(y - step * grad)
        
        if it % 50 == 0:
            sc = _score_fast(a_next, _full_convolve(a_next, a_next))
            if sc < best_sc:
                best_sc = sc
                best_a = a_next.copy()
        
        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next
    
    sc_final = _score_fast(a_curr, _full_convolve(a_curr, a_curr))
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()
    
    return best_a, best_sc

def _adam_logspace(a_init, deadline, betas, lr_init=0.01, max_iter=200000):
    """Adam optimization in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score_fast(a_init, _full_convolve(a_init, a_init))
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        while t < max_iter and time.monotonic() < deadline:
            if time.monotonic() > deadline:
                break
            t += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            w = _soft_max_weights(c, beta)
            grad = _grad_z(a, c, w, n)
            
            if not np.all(np.isfinite(grad)):
                break
            
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * grad * grad
            mh = m_buf / (1 - b1**t)
            vh = v_buf / (1 - b2**t)
            
            update = lr * mh / (np.sqrt(vh) + eps)
            unorm = np.linalg.norm(update)
            if unorm > 0.3:
                update *= 0.3 / unorm
            
            z -= update
            z = np.clip(z, -50, 50)
            z -= z.mean()
            
            if t % 20 == 0:
                a_chk = _softmax(z)
                sc = _score_fast(a_chk, _full_convolve(a_chk, a_chk))
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a_chk.copy()
            
            if t % 10000 == 0:
                lr = max(0.0005, lr * 0.95)
    
    a_final = _softmax(z)
    sc_final = _score_fast(a_final, _full_convolve(a_final, a_final))
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()
    
    return best_a, best_sc
