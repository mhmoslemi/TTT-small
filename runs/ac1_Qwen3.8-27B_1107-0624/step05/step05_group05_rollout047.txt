    while k < n:
        k <<= 1
    return max(2, k)


def _convolve(a, a2):

    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(a2, N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    b = _convolve(a, a)
    return float(2.0 * n * b.max() / (s * s))

    """Score assuming sum(a)=1, so score = 2n * max(conv)."""
    n = len(a)
    b = _convolve(a, a)
    return float(2.0 * n * b.max())


def _grad_bk(a, k):
    """Compute gradient of conv(a,a)[k] w.r.t. a. Result = 2*a[k-i] for valid i."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        # grad[i] = 2*a[k-i] for i in [lo, hi]
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad

    n = len(a)

        return _grad_bk(a, k_max)

        hi = min(k, n - 1)

            grad[lo:hi + 1] += 2.0 * a[idx]

    return grad

        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        b_shifted = (b - b_max) / max(beta, 1e-10)
        e = np.exp(b_shifted)
        s_e = float(e.sum())
        if s_e < 1e-300 or not np.isfinite(s_e):
            w = np.zeros_like(b)
            w[int(np.argmax(b))] = 1.0
        else:
            w = e / s_e

        # Gradient: weighted average of gradients
        w_rev = w[::-1]
        conv_aw = _convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]

        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        if t % 10000 == 0:
            if best_score < prev_best * 0.9999:
                current_lr = min(0.05, current_lr * 1.05)
            else:
                current_lr = max(0.001, current_lr * 0.97)
            prev_best = best_score

    a_final = _softmax_z(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _pairwise_cd(a, deadline, n_pairs=1000):
    """Coordinate descent: try swapping mass between pairs of indices."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    rng = np.random.default_rng(12345)
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving mass from i to j (or vice versa)
        # a_i' = a_i - d, a_j' = a_j + d (keep sum = 1)
        best_d = 0.0
        best_val = best_sc
        for d in [-0.01, -0.005, -0.001, 0.001, 0.005, 0.01]:
            new_ai = a[i] + d
            new_aj = a[j] - d
            if new_ai < 0 or new_aj < 0:
                continue
            a_try = a.copy()
            a_try[i] = new_ai
            a_try[j] = new_aj
            sc = _score_norm(a_try)
            if sc < best_val:
                best_val = sc
                best_d = d

        if best_d != 0.0:
            a[i] += best_d
            a[j] -= best_d
            if best_sc > best_val:
                best_sc = best_val
                best_a = a.copy()

        if it % 5000 == 0:
            print(f"    CD iter {it}: best_sc={best_sc:.10f}")

    return best_a, best_sc


def _try_different_n(n, a_source, deadline, sub_budget):
    """Try a different sequence length by interpolating."""
    from scipy.interpolate import interp1d
    n0 = len(a_source)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n)
    a_new = np.interp(x1, x0, a_source)
    a_new = np.maximum(a_new, 1e-10)
    a_new = a_new / a_new.sum()

    # Quick FW
    sub_deadline = time.monotonic() + sub_budget
    a_opt, sc = _fw_phase(a_new, sub_deadline, start_K=1, end_K=20, use_greedy=True)
    return a_opt, sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        a = parent.copy()
        n = len(a)
    else:
        n = 2048
        a = np.ones(n) / n

    a = a / a.sum()
    init_sc = _score(a)
    best_seq = a.copy()
    best_score = init_sc
    print(f"Initial: n={n}, score={init_sc:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 1: FW with small K (0 - 400s)
    phase1_end = min(deadline, t_start + 400.0)
    print(f"Phase 1: FW K=1-5, t={time.monotonic()-t_start:.0f}s")
    a_p1, sc_p1 = _fw_phase(a, phase1_end, start_K=1, end_K=5, use_greedy=True)
    full_sc = _score(a_p1)
    if full_sc < best_score:
        best_score = full_sc
        best_seq = a_p1.copy()
    print(f"Phase 1 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 2: Smooth optimization (400 - 900s)
    phase2_end = min(deadline, t_start + 900.0)
    if time.monotonic() < phase2_end:
        print(f"Phase 2: Smooth Adam, t={time.monotonic()-t_start:.0f}s")
        b = _convolve(best_seq, best_seq)
        max_b = float(b.max())
        a_p2, sc_p2 = _smooth_phase(best_seq, phase2_end,
                                     beta_start=0.01 * max_b,
                                     beta_end=0.0001 * max_b,
                                     max_iter=5000000, lr=0.015)
        full_sc = _score(a_p2)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_p2.copy()
        print(f"Phase 2 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 3: FW with larger K on improved point (900 - 1300s)
    phase3_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase3_end:
        print(f"Phase 3: FW K=5-30, t={time.monotonic()-t_start:.0f}s")
        a_p3, sc_p3 = _fw_phase(best_seq, phase3_end, start_K=5, end_K=30, use_greedy=True)
        full_sc = _score(a_p3)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_p3.copy()
        print(f"Phase 3 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 4: Fine smooth polishing (1300 - 1600s)
    phase4_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase4_end:
        print(f"Phase 4: Fine Adam, t={time.monotonic()-t_start:.0f}s")
        b = _convolve(best_seq, best_seq)
        max_b = float(b.max())
        a_p4, sc_p4 = _smooth_phase(best_seq, phase4_end - 5,
                                     beta_start=0.001 * max_b,
                                     beta_end=0.00001 * max_b,
                                     max_iter=3000000, lr=0.005)
        full_sc = _score(a_p4)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_p4.copy()
        print(f"Phase 4 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Phase 5: Try different n values (1600 - 1750s)
    phase5_end = min(deadline, t_start + 1750.0)
    if time.monotonic() < phase5_end - 30:
        print(f"Phase 5: Try different n, t={time.monotonic()-t_start:.0f}s")
        for n_try in [4096, 1024]:
            if time.monotonic() > phase5_end - 10:
                break
            sub_budget = min(60.0, phase5_end - time.monotonic() - 5)
            if sub_budget < 10:
                break
            a_try, sc_try = _try_different_n(n_try, best_seq, phase5_end, sub_budget)
            full_sc = _score(a_try)
            print(f"    n={n_try}: score={full_sc:.10f}")
            if full_sc < best_score:
                best_score = full_sc
                best_seq = a_try.copy()

    # Phase 6: Final FW burst (1750 - deadline)
    if time.monotonic() < deadline - 5:
        print(f"Phase 6: Final FW, t={time.monotonic()-t_start:.0f}s")
        a_p6, sc_p6 = _fw_phase(best_seq, deadline - 5, start_K=2, end_K=15, use_greedy=True)
        full_sc = _score(a_p6)
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_p6.copy()
        print(f"Phase 6 done: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")

    # Final validation
    final = np.clip(best_seq, 0.0, 1000.0)
    s = float(final.sum())
    if s < 0.01:
        final = np.ones(256)
    sc_final = _score(final)
    print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
    try:
        sc_eval = float(evaluate_sequence([float(x) for x in final]))
        print(f"Evaluator score: {sc_eval:.10f}")
    except Exception as ex:
        print(f"Eval error: {ex}")
    return [float(x) for x in final]
