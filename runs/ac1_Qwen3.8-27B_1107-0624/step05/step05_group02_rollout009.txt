"""
Nesterov-accelerated mirror descent with adaptive temperature for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses efficient FFT-based gradient computation, Nesterov momentum, adaptive
temperature scheduling, and multi-restart exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_maxc(a):
    """Return max of auto-convolution (assumes sum=1)."""
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_sm_gradient(a, beta):
    """Compute gradient of soft-max smoothed max(conv(a,a)) w.r.t. log(a).
    
    Returns (grad_z, w, c) where grad_z is the gradient w.r.t. z=log(a).
    """
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    # = 2 * (a * w_flipped)[2n-2-j]
    w_flipped = w[::-1]
    ac = _full_convolve(a, w_flipped)
    grad_a = 2.0 * ac[2*n-2: n-1: -1]

    # Convert to gradient w.r.t. z = log(a) (softmax parameterization)
    dot = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - dot)

    return grad_z, w, c


def _nesterov_optimize(a_init, deadline, lr=0.005, beta_scale_init=0.05,
                       beta_scale_min=0.0005, momentum_decay=0.9995,
                       lr_decay=0.9999, check_interval=100):
    """Nesterov-accelerated gradient descent on the simplex (log-parameterization).
    
    Minimizes the soft-max approximation of max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_maxc = float(c_init.max())
    max_c_ref = best_maxc

    # Nesterov state
    z_prev = z.copy()
    t = 1.0
    beta_scale = beta_scale_init
    beta = beta_scale * max_c_ref

    lr_current = lr
    step = 0
    no_improve = 0
    best_step = 0

    while time.monotonic() < deadline:
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = (t - 1.0) / t_new
        z_nest = z + mu * (z - z_prev)
        a_nest = _softmax(z_nest)

        # Compute gradient at Nesterov point
        grad_z, w, c = _compute_sm_gradient(a_nest, beta)

        if not np.all(np.isfinite(grad_z)):
            # Fallback: use current z
            z_nest = z.copy()
            a_nest = _softmax(z_nest)
            grad_z, w, c = _compute_sm_gradient(a_nest, beta)
            if not np.all(np.isfinite(grad_z)):
                break

        # Gradient step with clipping
        update = lr_current * grad_z
        update_norm = np.linalg.norm(update)
        max_update = 2.0
        if update_norm > max_update:
            update = update * (max_update / update_norm)

        z_new = z_nest - update
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        a_new = _softmax(z_new)
        c_new = _full_convolve(a_new, a_new)
        maxc_new = float(c_new.max())

        if np.isfinite(maxc_new) and maxc_new < best_maxc:
            best_maxc = maxc_new
            best_a = a_new.copy()
            best_step = step
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive temperature: decrease beta_scale over time
        beta_scale *= momentum_decay
        if beta_scale < beta_scale_min:
            beta_scale = beta_scale_min
        beta = beta_scale * max_c_ref

        # Adaptive learning rate
        lr_current *= lr_decay
        if lr_current < 0.0001:
            lr_current = 0.0001

        # If stuck for a while, increase temperature and lr to escape
        if no_improve > 800:
            beta_scale = min(beta_scale * 3.0, beta_scale_init * 0.5)
            beta = beta_scale * max_c_ref
            lr_current = max(lr_current * 2.0, 0.003)
            no_improve = 0

        z_prev = z
        z = z_new
        t = t_new
        step += 1

        if step % (check_interval * 10) == 0:
            sc = _score(best_a)
            print(f"    NV step {step}: maxc={best_maxc:.12f}, score={sc:.10f}, "
                  f"beta_scale={beta_scale:.6f}, lr={lr_current:.6f}, "
                  f"no_improve={no_improve}")

    # Final check
    a_final = _softmax(z)
    maxc_final = float(_score_maxc(a_final))
    if maxc_final < best_maxc:
        best_maxc = maxc_final
        best_a = a_final.copy()

    return best_a, best_maxc


def _adam_optimize(a_init, deadline, lr=0.008):
    """Adam optimizer with log-sum-exp smoothed max (improved version)."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_maxc = float(c_init.max())
    max_c_ref = best_maxc

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr_current = lr

    # Adaptive beta schedule
    beta_scales = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    beta = beta_scales[0] * max_c_ref

    for beta_scale in beta_scales:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * max_c_ref
        while time.monotonic() < deadline:
            t_step += 1

            a = _softmax(z)
            grad_z, w, c = _compute_sm_gradient(a, beta)

            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = lr_current * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            if t_step % 200 == 0:
                a_check = _softmax(z)
                maxc_check = float(_score_maxc(a_check))
                if maxc_check < best_maxc:
                    best_maxc = maxc_check
                    best_a = a_check.copy()

            if t_step % 10000 == 0:
                lr_current = max(0.0005, lr_current * 0.92)

    a_final = _softmax(z)
    maxc_final = float(_score_maxc(a_final))
    if maxc_final < best_maxc:
        best_maxc = maxc_final
        best_a = a_final.copy()

    return best_a, best_maxc


def _make_uniform(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    if n0 == 1:
        return np.full(n_new, 1.0 / n_new)
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-30))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _build_lp_matrix(f, n):
    k = np.arange(2 * n - 1)
    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f, time_limit=20):
    """One LP-based step: find g maximizing sum(g) s.t. conv(f,g) <= max(conv(f,f))."""
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    A = _build_lp_matrix(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds_lp = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_lp,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

def _pow2_ge(n):
    k = 1
    while k < n:
        k <<= 1
    return k
