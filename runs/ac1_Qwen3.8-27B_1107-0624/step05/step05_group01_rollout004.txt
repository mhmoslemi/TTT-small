"""
Block LP optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses first-order linearization of the autocorrelation, solving LPs over
blocks of the sequence, combined with L-BFGS log-sum-exp polishing.
The LP directly minimizes the peak of the convolution while preserving
the sum, giving a principled descent direction for the non-smooth objective.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _lbfgs_optimize(a_init, deadline, maxiter=5000):
    """L-BFGS with log-sum-exp continuation for polishing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())
    fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
             0.00005, 0.00002]
    beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 80, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _block_lp_optimize(a, deadline, block_size=64):
    """
    Block coordinate descent via LP. For each block, solve:
      minimize t
      s.t. 2*sum_m delta_m * a[i-start-m] - t <= -conv[i]  (for relevant i)
           delta_m >= -a[start+m]
           sum(delta) = 0
    Then accept if actual score improves (with line search).
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    conv = _full_convolve(best_a, best_a)

    no_improve_rounds = 0
    max_no_improve = 5

    while time.monotonic() < deadline:
        if no_improve_rounds >= max_no_improve:
            break

        round_improved = False
        conv = _full_convolve(best_a, best_a)
        num_blocks = (n + block_size - 1) // block_size

        for start in range(0, n - 1, block_size):
            if time.monotonic() > deadline:
                return best_a, best_score

            end = min(start + block_size, n)
            B = end - start
            if B < 2:
                break

            # Build LP
            # Variables: x = [delta_0, ..., delta_{B-1}, t]  (B+1 vars)
            # Relevant i range: [start, start+B+n-2] intersect [0, 2n-2]
            i_min = max(0, start)
            i_max = min(2 * n - 2, start + B + n - 2)
            num_i = i_max - i_min + 1

            if num_i <= 0:
                continue

            # A_ub shape: (num_i + B, B + 1), b_ub shape: (num_i + B,)
            A_ub = np.zeros((num_i + B, B + 1))
            b_ub_vec = np.zeros(num_i + B)

            # Convolution constraints: 2*sum_m delta_m * a[i-start-m] - t <= -conv[i]
            for idx in range(num_i):
                i = i_min + idx
                # Coefficient of delta_m is 2*a[i-start-m] if 0 <= i-start-m < n
                # Vectorize: for m in 0..B-1, j = i-start-m
                js = i - start - np.arange(B)
                valid = (js >= 0) & (js < n)
                if np.any(valid):
                    A_ub[idx, valid] = 2.0 * best_a[js[valid]]
                A_ub[idx, B] = -1.0
                b_ub_vec[idx] = -conv[i]

            # Non-negativity: -delta_m <= a[start+m]
            A_ub[num_i:num_i + B, :B] = -np.eye(B)
            b_ub_vec[num_i:num_i + B] = best_a[start:end]

            # Objective: minimize t
            c_obj = np.zeros(B + 1)
            c_obj[B] = 1.0

            # Equality: sum(delta) = 0
            A_eq = np.zeros((1, B + 1))
            A_eq[0, :B] = 1.0
            b_eq = [0.0]

            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub_vec,
                              A_eq=A_eq, b_eq=b_eq,
                              bounds=[(None, None)] * B + [(None, None)],
                              method='highs', options={'time_limit': 5})
                if res.status != 0 or res.x is None:
                    continue

                delta = res.x[:B]
                # Check if delta is meaningful
                max_delta = float(np.max(np.abs(delta)))
                if max_delta < 1e-12:
                    continue

                # Line search: try full step, then fractions
                current_score = 2.0 * n * float(conv.max())  # since sum=1
                found = False
                for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                    a_trial = best_a.copy()
                    a_trial[start:end] = best_a[start:end] + alpha * delta
                    # Enforce non-negativity
                    a_trial = np.maximum(a_trial, 0.0)
                    s = float(a_trial.sum())
                    if s < 0.01:
                        continue
                    a_trial = a_trial / s
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc_trial = 2.0 * n * float(c_trial.max())
                    if sc_trial < current_score - 1e-10:
                        best_a = a_trial.copy()
                        best_score = sc_trial
                        conv = c_trial
                        current_score = sc_trial
                        round_improved = True
                        found = True
                        break

                if found:
                    break  # Restart block cycling with updated conv
                    # Actually, let's continue cycling but update conv
                    # For efficiency, update conv and continue
                    conv = _full_convolve(best_a, best_a)

            except Exception:
                continue

        if round_improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1

    return best_a, best_score

def _coord_descent(a, deadline, steps=50000):
    """Simple coordinate descent: adjust one element at a time to reduce max peak."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(best_a)
    conv = _full_convolve(best_a, best_a)

    for it in range(steps):
        if time.monotonic() > deadline:
            break
        # Find the peak
        peak_idx = int(np.argmax(conv))
        # The peak involves a[j]*a[peak_idx-j] for valid j
        # Try adjusting elements that contribute to the peak
        j_lo = max(0, peak_idx - n + 1)
        j_hi = min(n - 1, peak_idx)

        # Pick a random index in the contributing range
        j = int(j_lo + np.random.random() * (j_hi - j_lo + 1))
        # Try decreasing a[j] slightly
        old_val = best_a[j]
        if old_val < 1e-10:
            continue
        for frac in [0.9, 0.8, 0.95]:
            best_a[j] = old_val * frac
            # Update conv: conv changes only at positions involving j
            # conv[i] for i where a[j] appears: i = j + k for k in [0, n-1]
            # and i = k + j (same thing since conv is symmetric in the two args)
            # Actually conv[i] = sum_l a[l]*a[i-l], so a[j] appears when l=j or i-l=j
            # New conv[i] = old_conv[i] - a[j]*a[i-j] - a[j]*a[i-j] + new_a[j]*a[i-j] + new_a[j]*a[i-j]
            # = old_conv[i] - 2*a[j]*a[i-j] + 2*new_a[j]*a[i-j] (if i-j valid and i-j != j)
            # = old_conv[i] - 2*a[j]*a[i-j] + a[j]^2 (if i-j == j, i.e., i=2j)
            # This is getting complicated; just recompute for small n
            # For efficiency, only update affected entries
            new_val = best_a[j]
            delta_val = new_val - old_val
            for i in range(max(0, j), min(2 * n - 1, j + n)):
                other = i - j
                if 0 <= other < n:
                    if other == j:
                        conv[i] += 2 * delta_val * new_val  # (new^2 - old^2) = (new-old)(new+old)
                    else:
                        conv[i] += 2 * delta_val * best_a[other]
            # Recompute max
            max_c = float(np.max(conv))
            sc = 2.0 * n * max_c
            if sc < best_score - 1e-12:
                best_score = sc
                break
            else:
                best_a[j] = old_val  # revert
                # Recompute conv after revert
                for i in range(max(0, j), min(2 * n - 1, j + n)):
                    other = i - j
                    if 0 <= other < n:
                        if other == j:
                            conv[i] += 2 * (old_val - new_val) * old_val
                        else:
                            conv[i] += 2 * (old_val - new_val) * best_a[other]

    return best_a, best_score

        a1, sc1 = _lbfgs_optimize(best_seq, p1_end, maxiter=3000)
        update_best(a1, sc1, "lbfgs_warmup")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Block LP with small blocks (600s) ===
    p2_end = min(deadline, t_start + 850.0)
    print("Phase 2: Block LP (B=32)")
    if time.monotonic() < p2_end:
        a2, sc2 = _block_lp_optimize(best_seq, p2_end, block_size=32)
        update_best(a2, sc2, "block_lp_B32")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Block LP with larger blocks (400s) ===
    p3_end = min(deadline, t_start + 1250.0)
    print("Phase 3: Block LP (B=128)")
    if time.monotonic() < p3_end:
        a3, sc3 = _block_lp_optimize(best_seq, p3_end, block_size=128)
        update_best(a3, sc3, "block_lp_B128")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Peak-focused LP (300s) ===
    p4_end = min(deadline, t_start + 1550.0)
    print("Phase 4: Peak-focused LP")
    if time.monotonic() < p4_end:
        a4, sc4 = _peak_focused_lp(best_seq, p4_end, block_size=32, peak_count=30)
        update_best(a4, sc4, "peak_lp")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Final L-BFGS polish (150s) ===
    p5_end = min(deadline, t_start + 1700.0)
    print("Phase 5: Final L-BFGS polish")
    if time.monotonic() < p5_end:
        a5, sc5 = _lbfgs_optimize(best_seq, p5_end, maxiter=5000)
        update_best(a5, sc5, "final_lbfgs")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: One more block LP round if time allows ===
    if time.monotonic() < deadline - 20:
        print("Phase 6: Final block LP")
        a6, sc6 = _block_lp_optimize(best_seq, deadline - 10, block_size=64)
        update_best(a6, sc6, "final_block_lp")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Verify with evaluator if possible
        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, trying fallback")
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
