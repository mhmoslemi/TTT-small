def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Score with sum(a) = 1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0, np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    esum = e.sum()
    w = e / esum
    val = cm + beta * np.log(esum)

    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]

    w_2i = w[2 * np.arange(n)]
    grad_a = 2.0 * h - a * w_2i
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return float(val), grad_z


def _peak_target_val_grad(z, beta, top_k=5):
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0, np.zeros(n)
    # Focus on top_k peaks
    top_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[top_idx]
    shifted = (c_top - cm) / beta
    e = np.exp(shifted)
    esum = e.sum()
    w_top = e / esum

    # Expand w to full length (zero except at top peaks)
    w = np.zeros(2 * n - 1)
    w[top_idx] = w_top

    val = cm + beta * np.log(esum)

    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]

    w_2i = w[2 * np.arange(n)]
    grad_a = 2.0 * h - a * w_2i
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return float(val), grad_z


def _adam_run(z_init, deadline, beta_start, beta_end, lr_init=0.1,
              top_k=5, kick_interval=5000, kick_mag=0.05):
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    z = z - z.mean()
    best_z = z.copy()
    best_score = _score_norm(_softmax_fn(z))

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init

    n_total = max(1, int((deadline - time.monotonic()) / 0.005))
    for it in range(n_total):
        if time.monotonic() > deadline - 0.5:
            break
        t += 1

        # Adaptive beta: linear interpolation from beta_start to beta_end
        frac = min(1.0, t / max(1, n_total))
        beta = beta_start * (beta_end / beta_start) ** frac

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break

        if top_k <= 5:
            val, grad_z = _peak_target_val_grad(z, beta, top_k)
        else:
            val, grad_z = _logsumexp_val_grad(z, beta)

        if not np.all(np.isfinite(grad_z)):
            break

        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)

        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        a = _softmax_fn(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_z = z.copy()

        # Learning rate decay
        if t % 5000 == 0:
            lr = max(0.01, lr * 0.85)

        # Random kick to escape local minima
        if t % kick_interval == 0:
            z = z + kick_mag * np.random.standard_normal(n)
            z = z - z.mean()

    return best_z, best_score


def _coordinate_descent(a_init, deadline, n_trials=5000):
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    c = _full_convolve(a, a)
    cm = float(c.max())

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        # Find top peaks
        top_peaks = np.argsort(c)[::-1][:3]

        improved = False
        for pk in top_peaks:
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a[valid_i] * a[pk - valid_i]
            # Try adjusting top contributors
            order = np.argsort(prod)[::-1][:6]
            for idx_pos in order:
                idx = valid_i[idx_pos]
                # Try reducing
                for delta in [0.002, 0.005, 0.01, 0.02]:
                    a_trial = a.copy()
                    a_trial[idx] = max(1e-15, a_trial[idx] - delta * a.sum())
                    a_trial /= a_trial.sum()
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc = float(c_trial.max()) * 2 * n
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        c = c_trial
                        cm = float(c.max())
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random perturbation
            idx = rng.integers(0, n)
            for delta in [0.001, 0.005, -0.001, -0.005]:
                a_trial = a.copy()
                a_trial[idx] = max(1e-15, a_trial[idx] + delta)
                a_trial /= a_trial.sum()
                c_trial = _full_convolve(a_trial, a_trial)
                sc = float(c_trial.max()) * 2 * n
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    c = c_trial
                    cm = float(c.max())
                    improved = True
                    break
            if not improved and it > 500:
                # Reset to best
                a = best_a.copy()
                c = _full_convolve(a, a)
                cm = float(c.max())

    return best_a, best_score


def _make_ramp(n, slope=1.5, curvature=0.0):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t + curvature * (t - 0.5) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exp(n, rate=2.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()


def _make_freq_init(n):
    """Initialize using frequency-domain intuition: spread energy across frequencies."""
    # Create a sequence whose DFT has roughly flat magnitude (like a chirp)
    # This minimizes peak autocorrelation
    k = np.arange(n)
    # Linear chirp in time domain -> spread out autocorrelation
    a = np.cos(np.pi * k * (k + 1) / n)
    a = np.abs(a) + 0.1
    a = a / a.sum()
    return a


def _make_profile_set(n):
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        profiles.append(_make_ramp(n, s))
    for s, c in [(1.5, 0.5), (2.0, -0.3), (1.0, 1.0), (2.5, 0.3), (3.0, -0.5)]:
        t = np.linspace(0, 1, n)
        a = 1.0 + s * t + c * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 4.0, 6.0]:
        profiles.append(_make_exp(n, rate))
    profiles.append(_make_freq_init(n))
    # Symmetric bell
    t = np.linspace(0, 1, n)
    for w in [0.2, 0.3, 0.4, 0.5]:
        a = np.exp(-((t - 0.5) / w) ** 2)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0 = _make_ramp(256, 2.0)
        update_best(a0, _score_norm(a0), "default")

    # ============ Phase 1: Multi-scale from scratch (40% of time) ============
    phase1_end = t_start + (budget_s - 30.0) * 0.40
    print(f"\n=== Phase 1: Multi-scale (ends t={phase1_end - t_start:.0f}s) ===")

    # Try profiles at n=128
    n0 = 128
    profiles = _make_profile_set(n0)
    best_prof = None
    best_prof_sc = np.inf
    for prof in profiles:
        sc = _score_norm(prof)
        if sc < best_prof_sc:
            best_prof_sc = sc
            best_prof = prof.copy()

    print(f"  Best initial profile at n={n0}: score={best_prof_sc:.10f}")

    # Quick optimize at n=128
    z_cur = np.log(np.maximum(best_prof, 1e-15))
    z_cur = z_cur - z_cur.mean()
    quick_dl = min(phase1_end, t_start + 15.0)
    z_cur, sc_cur = _adam_run(z_cur, quick_dl, beta_start=0.3, beta_end=0.01, lr_init=0.5)
    a_cur = _softmax_fn(z_cur)
    print(f"  After n={n0} Adam: score={sc_cur:.10f}")

    # Multi-scale upsample
    scales = [256, 512, 1024, 2048, 4096]
    if time.monotonic() < phase1_end - 60:
        scales.append(8192)

    time_per_scale = (phase1_end - t_start - 15.0) / len(scales)

    for target_n in scales:
        if time.monotonic() > phase1_end - 10:
            break
        if target_n > 8192:
            continue

        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm(a_target)
        print(f"  Upsampled to n={target_n}: score={sc_target:.10f}")

        z_target = np.log(np.maximum(a_target, 1e-15))
        z_target = z_target - z_target.mean()

        alloc = min(time_per_scale, phase1_end - time.monotonic() - 5)
        if alloc < 5:
            break
        opt_dl = time.monotonic() + alloc

        # Beta schedule: start moderate, end small
        c_t = _full_convolve(a_target, a_target)
        max_c_t = float(c_t.max())
        beta_s = 0.2 * max_c_t
        beta_e = 0.001 * max_c_t

        # Learning rate depends on n
        lr = 0.3 if target_n <= 512 else (0.15 if target_n <= 2048 else 0.08)

        z_opt, sc_opt = _adam_run(z_target, opt_dl, beta_start=beta_s,
                                   beta_end=beta_e, lr_init=lr, top_k=5)
        a_opt = _softmax_fn(z_opt)
        print(f"  After n={target_n} Adam: score={sc_opt:.10f}")

        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt
            z_cur = z_opt

    if sc_cur < best_score:
        update_best(a_cur, sc_cur, "multi_scale")
    else:
        a_cur = best_seq.copy()
        sc_cur = best_score
        z_cur = np.log(np.maximum(a_cur, 1e-15))
        z_cur = z_cur - z_cur.mean()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # ============ Phase 2: Continue optimization at best scale (35% of time) ============
    phase2_end = t_start + (budget_s - 30.0) * 0.75

    if time.monotonic() < phase2_end - 20 and best_seq is not None:

        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        z_work = np.log(np.maximum(a_work, 1e-15))
        z_work = z_work - z_work.mean()

        c_work = _full_convolve(a_work, a_work)
        max_c_work = float(c_work.max())

        # Run multiple Adam passes with decreasing beta
        pass_configs = [
            (0.05 * max_c_work, 0.005 * max_c_work, 0.1, 5),
            (0.02 * max_c_work, 0.002 * max_c_work, 0.05, 3),
            (0.01 * max_c_work, 0.001 * max_c_work, 0.03, 3),
            (0.005 * max_c_work, 0.0005 * max_c_work, 0.02, 5),
        ]

        time_per_pass = (phase2_end - time.monotonic()) / len(pass_configs)

        for pidx, (bs, be, lr, tk) in enumerate(pass_configs):

                                       kick_interval=10000, kick_mag=0.03)
            a_new = _softmax_fn(z_new)
            update_best(a_new, sc_new, f"phase2_pass{pidx+1}")

            if sc_new < _score_norm(a_work):
                a_work = a_new.copy()
                z_work = z_new.copy()
            else:
                # Reset to best
                a_work = best_seq.copy() / best_seq.sum()
                z_work = np.log(np.maximum(a_work, 1e-15))
                z_work = z_work - z_work.mean()

    print(f"  Phase 2 done: best={best_score:.10f}")

    # ============ Phase 3: Coordinate descent + peak targeting (20% of time) ============
    phase3_end = t_start + (budget_s - 30.0) * 0.95

            print(f"  Coordinate descent (alloc={cd_alloc:.0f}s)")
            a_cd_out, sc_cd = _coordinate_descent(a_cd, dl_cd, n_trials=10000)
            update_best(a_cd_out, sc_cd, "coord_descent")

        # Final Adam with very small beta on the polished sequence
        if time.monotonic() < phase3_end - 10 and best_seq is not None:
            a_fin = best_seq.copy() / best_seq.sum()
            z_fin = np.log(np.maximum(a_fin, 1e-15))
            z_fin = z_fin - z_fin.mean()
            c_fin = _full_convolve(a_fin, a_fin)
            max_c_fin = float(c_fin.max())

            alloc_fin = min(60.0, phase3_end - time.monotonic() - 5)
            if alloc_fin > 5:
                dl_fin = time.monotonic() + alloc_fin
                print(f"  Final fine Adam (alloc={alloc_fin:.0f}s)")
                z_fin_out, sc_fin = _adam_run(z_fin, dl_fin,
                                               beta_start=0.001 * max_c_fin,
                                               beta_end=0.0001 * max_c_fin,
                                               lr_init=0.01, top_k=3,
                                               kick_interval=20000, kick_mag=0.01)
                a_fin_out = _softmax_fn(z_fin_out)
                update_best(a_fin_out, sc_fin, "final_adam")

    # ============ Phase 4: Try alternative n if time remains ============
    if time.monotonic() < deadline - 30 and best_seq is not None:
        n_cur = len(best_seq)
        # Try a different n that we haven't tried
        alt_n = 2 * n_cur if n_cur <= 4096 else (n_cur // 2 if n_cur > 8192 else 16384)

    print(f"FALLBACK: n=256, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
