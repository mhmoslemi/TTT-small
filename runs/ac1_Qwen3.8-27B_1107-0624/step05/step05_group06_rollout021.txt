Optimization of non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses power-smoothed L-BFGS with adaptive continuation, LP-based direction finding,
and exact active-set subgradient polishing.
"""
import numpy as np

import scipy.optimize

def _next_pow2_m(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fc(a, b):
    """Fast FFT convolution."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2_m(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fc(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _sm(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, N):
    """Log-sum-exp smoothed objective."""
    a = _sm(z)
    c = _fc(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-15:
        return 2.0 * N * cm
    return 2.0 * N * (cm + beta * np.log(s))

def _lse_grad(z, beta, N):
    """Gradient of LSE smoothed objective."""
    a = _sm(z)
    c = _fc(a, a)
    cm = c.max()
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-15:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_flip = a[::-1]
    wc = _fc(w, a_flip)
    h = wc[N-1:2*N-1]
    W = float(np.dot(w, c))
    return 4.0 * N * a * (h - W)

def _exact_subgrad(a, c, N, tol_frac=0.001):
    """Exact subgradient using active set."""
    cm = c.max()
    active = c >= (cm - tol_frac * cm)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_flip = a[::-1]
    wc = _fc(w, a_flip)
    h = wc[N-1:2*N-1]
    W = float(np.dot(w, c))
    return 4.0 * N * a * (h - W)

def _lp_direction(f, M):
    """Solve LP to find direction that maximizes sum(b) subject to f*b <= M, b >= 0."""
    n = len(f)
    # Constraint: (f * b)[x] <= M for x = 0, ..., 2n-2
    # (f*b)[x] = sum_j f[j] * b[x-j] for valid indices
    # A[x, j] = f[j] if 0 <= x-j < n, else 0
    
    # Build A matrix (sparsely)
    # For efficiency, use the structure: A[x, j] = f[j] if j <= x <= j+n-1
    # This is a Toeplitz-like matrix
    
    # For n=2048, A is 4095 x 2048. Let's build it.
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_vec = np.full(2*n-1, M)

    except Exception:
