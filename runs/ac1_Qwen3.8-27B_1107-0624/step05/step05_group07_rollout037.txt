"""
Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: (1) LP iterated minimax with adaptive line search,
(2) multi-scale L-BFGS with adaptive beta, (3) structured perturbation for local
minima escape, (4) n-sweep to find optimal dimension, (5) final coordinate descent polish.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, max_time=10):
    """Solve LP: min t s.t. conv(f,g)_k <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
            return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters=200, deadline=None, use_line_search=True):
    """LP iterated minimax with optional line search for step size."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    f = f.copy()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g, t_val = _lp_solve(f)
        if g is None:
            break

        if use_line_search:
            # Try multiple step sizes and pick the best
            best_tau = 0.3
            best_sc = _score_norm_fn(f)
            for tau in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                f_try = (1 - tau) * f + tau * g
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-15:
                    continue
                f_try = f_try / s
                sc = _score_norm_fn(f_try)
                if sc < best_sc:
                    best_sc = sc
                    best_tau = tau

            f_new = (1 - best_tau) * f + best_tau * g
            f_new = np.maximum(f_new, 0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
        else:
            f_new = (1 - 0.3) * f + 0.3 * g
            f_new = np.maximum(f_new, 0)

            best_a = f_new.copy()
        f = f_new

        if it % 50 == 0:
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_fine(a_init, deadline, max_betas=40):
    """L-BFGS with adaptive fine beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Adaptive beta schedule: geometric from 5% to 1e-6 of max_c0
    beta_start = 0.05 * max_c0
    beta_end = 1e-6 * max_c0
    ratios = np.geomspace(beta_start / max_c0, beta_end / max_c0, max_betas)
    beta_list = ratios * max_c0

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = max(1.0, (deadline - time.monotonic()) / remaining)
        time_per = min(time_per, 30.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(500, int(time_per * 50))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-15,
                         'maxls': 100, 'maxfun': maxiter * 10}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                if np.all(a_new >= 0) and np.all(np.isfinite(a_new)):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score

    z = z - z.mean()
    a = _softmax_fn(z)
    return _run_lbfgs_fine(a, deadline, max_betas=n_betas)


def _coord_descent_polish(a_init, deadline, max_rounds=50):
    """Coordinate descent in log-space to polish the solution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = _score_norm_fn(a)
            # Try decreasing steps
            for delta in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                z[j] += delta
                a_trial = _softmax_fn(z)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
                else:
                    z[j] -= delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
            # Restore z[j]
            z[j] -= best_delta
        if not improved:
            # Try smaller steps
            for j in range(n):
                if time.monotonic() > deadline:
                    break
                z[j] += 0.002
                a_trial = _softmax_fn(z)
                sc = _score_norm_fn(a_trial)
                z[j] -= 0.002
                if sc < best_score:
                    z[j] += 0.002
                    z = z - z.mean()
                    a = _softmax_fn(z)
                    sc = _score_norm_fn(a)
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        improved = True
        if not improved:
            break
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for slope in [0.3, 0.5, 1.0, 1.5, 2.0, -0.5, -1.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for rate in [0.5, 1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    a = np.abs(u) + 0.05
    profiles.append(a / a.sum())
    a = 1.0 / (0.05 + u**2)
    profiles.append(a / a.sum())
    # S-shaped
    a = 1.0 / (1.0 + np.exp(-10 * (t - 0.5)))
    profiles.append(a / a.sum())
    # Double bump
    a = np.exp(-20*(t-0.25)**2) + np.exp(-20*(t-0.75)**2)
    profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = _make_profiles(n_init)[20]  # use a cosine profile
        update_best(parent, _score_norm_fn(parent), "default_profile")

    n_parent = len(parent)
    print(f"Starting n={n_parent}")

    # Phase 1: LP iterated minimax with line search (500s)
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\nPhase 1: LP iterated minimax with line search")

    # Try at parent n and also at 2048 for speed
    n_lp = n_parent
    a_lp = parent.copy()
    update_best(a_lp, _score_norm_fn(a_lp), f"lp_start_n{n_lp}")

    # Run LP with line search
    a_cur, sc_cur = _lp_iterate(a_lp, n_iters=500, deadline=phase1_end, use_line_search=True)
    update_best(a_cur, sc_cur, f"lp_line_search_n{n_lp}")

    # Interpolate back if needed
    if n_parent != n_lp:
        a_rec = _interp_profile(a_cur, n_parent)
        update_best(a_rec, _score_norm_fn(a_rec), f"lp_interp_n{n_parent}")
    else:
        a_rec = a_cur.copy()

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: Fine L-BFGS continuation (400s)
    phase2_end = min(deadline, t_start + 900.0)
    print(f"\nPhase 2: Fine L-BFGS continuation (n={len(best_seq)})")

    a_start = best_seq.copy()
    p_best, sc_best = _run_lbfgs_fine(a_start, phase2_end, max_betas=50)
    update_best(p_best, sc_best, "lbfgs_fine")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Perturbation + re-optimization (350s)
    phase3_end = min(deadline, t_start + 1250.0)
    print(f"\nPhase 3: Perturbation + re-optimization")

    n_perturbations = 15
    per_perturb_time = (phase3_end - time.monotonic()) / n_perturbations

    for i in range(n_perturbations):
        if time.monotonic() > phase3_end - 10:
            break
        noise_scale = 0.01 * (1 + i * 0.3)
        per_dl = min(phase3_end - 5, time.monotonic() + max(per_perturb_time, 5.0))
        p_try, sc_try = _perturb_and_optimize(best_seq, rng, per_dl,
                                              noise_scale=noise_scale, n_betas=10)
        update_best(p_try, sc_try, f"perturb_{i}_ns{noise_scale:.3f}")

    # Also try structured perturbations (block perturbations)
    for i in range(5):
        if time.monotonic() > phase3_end - 10:
            break
        n = len(best_seq)
        block_size = n // 8
        start = int(rng.integers(0, n - block_size))
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z[start:start+block_size] += 0.05 * rng.normal(size=block_size)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        per_dl = min(phase3_end - 5, time.monotonic() + 15.0)
        p_try, sc_try = _run_lbfgs_fine(a, per_dl, max_betas=8)
        update_best(p_try, sc_try, f"block_perturb_{i}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: n-sweep (250s)
    phase4_end = min(deadline, t_start + 1500.0)
    print(f"\nPhase 4: n-sweep")

    n_candidates = [2048, 3072, 4096, 5120, 6144, 8192]
    n_candidates = [x for x in n_candidates if x != len(best_seq)]
    # Sort by distance from current best n (try closest first)
    n_candidates.sort(key=lambda x: abs(x - len(best_seq)))

    per_n_time = (phase4_end - time.monotonic()) / max(1, len(n_candidates))

    for n_try in n_candidates[:5]:
        if time.monotonic() > phase4_end - 10:
            break
        per_dl = min(phase4_end - 5, time.monotonic() + max(per_n_time, 20.0))
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm_fn(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")

        # Short L-BFGS at this n
        p_opt, sc_opt = _run_lbfgs_fine(p_interp, per_dl, max_betas=15)
        update_best(p_opt, sc_opt, f"lbfgs_n{n_try}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Final polish (remaining time)
    phase5_end = deadline
    print(f"\nPhase 5: Final polish (n={len(best_seq)})")

    # LP polish
    if time.monotonic() < phase5_end - 60:
        lp_dl = min(phase5_end - 60, time.monotonic() + 120.0)
        p_lp, sc_lp = _lp_iterate(best_seq, n_iters=100, deadline=lp_dl, use_line_search=True)
        update_best(p_lp, sc_lp, "lp_polish")

    # Very fine L-BFGS
    if time.monotonic() < phase5_end - 30:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_tiny = 1e-7 * max_c_final
        per_dl = min(phase5_end - 20, time.monotonic() + 60.0)
        p_try, sc_try = _single_beta_lbfgs(best_seq, beta_tiny, per_dl, maxiter=20000)
        update_best(p_try, sc_try, "ultra_fine_lbfgs")

    # Coordinate descent polish
    if time.monotonic() < phase5_end - 15:
        per_dl = min(phase5_end - 10, time.monotonic() + (phase5_end - time.monotonic()) * 0.6)
        p_cd, sc_cd = _coord_descent_polish(best_seq, per_dl, max_rounds=30)
        update_best(p_cd, sc_cd, "coord_descent")

    # One more tiny-beta L-BFGS after coordinate descent
    if time.monotonic() < phase5_end - 10:
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_ultra = 5e-8 * max_c_final
        per_dl = min(phase5_end - 5, time.monotonic() + 30.0)
        p_try, sc_try = _single_beta_lbfgs(best_seq, beta_ultra, per_dl, maxiter=10000)
        update_best(p_try, sc_try, "final_ultra_lbfgs")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final: one more LP round if time permits
    if time.monotonic() < phase5_end - 20:
        lp_dl = phase5_end - 10
        p_lp, sc_lp = _lp_iterate(best_seq, n_iters=50, deadline=lp_dl, use_line_search=True)
        update_best(p_lp, sc_lp, "final_lp")
        # Quick L-BFGS after LP
        if time.monotonic() < phase5_end - 5:
            c_f = _full_convolve(best_seq, best_seq)
            beta_f = 1e-7 * float(c_f.max())
            p_f, sc_f = _single_beta_lbfgs(best_seq, beta_f, phase5_end - 3, maxiter=5000)
            update_best(p_f, sc_f, "final_final")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = final_seq / max(s, 0.01)
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = _make_profiles(n_fb)[20]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
