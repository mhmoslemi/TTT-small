"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _convolution(a):
    return _full_convolve(a, a)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _symmetrize(a):
    return 0.5 * (a + a[::-1])


def _half_to_full(z_half):
    m = len(z_half)
    n = 2 * m
    a_half = _softmax_fn(z_half)
    a_full = np.zeros(n)
    a_full[:m] = a_half * 0.5
    a_full[m:] = a_half[::-1] * 0.5
    return a_full


def _full_to_half(a):
    n = len(a)
    m = n // 2
    a_sym = _symmetrize(a)
    a_half = a_sym[:m] * 2.0
    s = a_half.sum()
    if s < 1e-30:
        return np.zeros(m)
    return a_half / s


def _grad_wrt_a(a, w):
    """Compute grad_a where grad_a[j] = 4*n*sum_k w[k]*a[k-j].
    w has length 2n-1 (the logsumexp weights or delta weights)."""
    n = len(a)
    v = _full_convolve(w[::-1], a)
    # v has length 3n-2
    # grad_a[j] = 4n * v[2n-2-j] for j=0..n-1
    grad_a = 4.0 * n * v[n - 1:2 * n - 1][::-1]
    return grad_a


def _grad_wrt_z(a, grad_a):
    """Convert gradient w.r.t. a to gradient w.r.t. z (softmax param).
    grad_z[i] = a[i] * (grad_a[i] - sum_j a[j]*grad_a[j])"""
    a = _symmetrize(a)
    grad_a = _symmetrize(grad_a)
    shift = np.dot(a, grad_a)
    grad_z = a * (grad_a - shift)
    return _symmetrize(grad_z)


def _obj_and_grad_z(z_half, beta, n):
    """Objective and gradient for L-BFGS on half-sequence parameterization."""
    a = _half_to_full(z_half)
    c = _convolution(a)
    max_c = float(c.max())
    c_shifted = (c - max_c) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        obj = 2.0 * n * max_c
    else:
        w = e / s
        obj = 2.0 * n * (max_c + beta * float(np.log(s)))
    
    grad_a = _grad_wrt_a(a, w)
    # Map to half-sequence gradient
    # a_full[:m] = a_half * 0.5, a_full[m:] = a_half[::-1] * 0.5
    # da_full[j]/dz_half[i] for j < m: 0.5 * a_half[j]*(delta[j,i] - a_half[i])
    # da_full[j]/dz_half[i] for j >= m: 0.5 * a_half[n-1-j]*(delta[n-1-j,i] - a_half[i])
    m = len(z_half)
    a_half = _softmax_fn(z_half)
    
    # grad_z_half[i] = sum_j grad_a[j] * da_full[j]/dz_half[i]
    # = 0.5 * sum_{j<m} grad_a[j] * a_half[j]*(delta[j,i] - a_half[i])
    #   + 0.5 * sum_{j>=m} grad_a[j] * a_half[n-1-j]*(delta[n-1-j,i] - a_half[i])
    
    # First sum: 0.5 * (grad_a[:m] * a_half)[i] - 0.5 * a_half[i] * sum_{j<m} grad_a[j]*a_half[j]
    # Second sum: 0.5 * (grad_a[m:][::-1] * a_half)[i] - 0.5 * a_half[i] * sum_{j>=m} grad_a[j]*a_half[n-1-j]
    
    term1 = 0.5 * (grad_a[:m] * a_half)
    term2 = 0.5 * (grad_a[m:][::-1] * a_half)
    sum1 = 0.5 * np.dot(grad_a[:m], a_half)
    sum2 = 0.5 * np.dot(grad_a[m:], a_half[::-1])
    
    grad_z = term1 + term2 - a_half * (sum1 + sum2)
    
    if not np.all(np.isfinite(grad_z)):
        grad_z = np.zeros(m)
    
    return obj, grad_z

def _subgradient_step(a, step_size, n_active=3):
    """One subgradient step targeting the top n_active peaks."""
    n = len(a)
    c = _convolution(a)
    max_c = c.max()
    
    # Find active lags (within 0.5% of max)
    threshold = max_c * 0.995
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    active = active[:n_active]
    
    # Compute subgradient: mean of gradients at active lags
    # grad_a[j] = 2 * sum_{k in active} a[k-j] / |active|
    w = np.zeros(len(c))
    for k in active:
        w[k] = 1.0 / len(active)
    
    grad_a = _grad_wrt_a(a, w)
    # Project to simplex tangent
    grad_a = grad_a - np.dot(a, grad_a)
    # Enforce symmetry
    grad_a = _symmetrize(grad_a)
    
    # Step
    a_new = a - step_size * grad_a
    a_new = _symmetrize(a_new)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_c
    a_new = a_new / s
    return a_new, max_c


def _run_subgradient(a_init, deadline, n_active=3, initial_step=0.01):
    """Active-set subgradient method with decreasing step size."""
    n = len(a_init)
    a = _symmetrize(a_init).copy() / _symmetrize(a_init).sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    step = initial_step
    no_improve = 0
    
    while time.monotonic() < deadline - 1.0:
        a_new, max_c = _subgradient_step(a, step, n_active)
        sc = _score_norm(a_new)
        
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            a = a_new.copy()  # still move, but don't update best
            no_improve += 1
        
        # Decrease step size
        step *= 0.995
        if no_improve > 100:
            step *= 0.5
            no_improve = 0
        
        if step < 1e-10:
            break
    
    return best_a, best_sc


def _coord_descent(a_init, deadline, n_sweeps=None, top_k=200):
    """Coordinate descent with ternary search on each coordinate.
    Optimizes a[i] and a[n-1-i] jointly (enforcing symmetry)."""
    n = len(a_init)
    m = n // 2
    a = _symmetrize(a_init).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    if n_sweeps is None:
        n_sweeps = 1000000  # time-limited
    
    # Priority: optimize coordinates with largest values first
    # (they contribute most to the peaks)
    order = np.argsort(a[:m])[::-1]
    
    sweep = 0
    while time.monotonic() < deadline - 1.0 and sweep < n_sweeps:
        improved = False
        for idx in order:
            if time.monotonic() > deadline - 1.0:
                break
            i = int(idx)
            current_val = a[i]
            if current_val < 1e-15:
                continue
            
            # Ternary search for best value of a[i] (= a[n-1-i])
            lo = 0.0
            hi = current_val * 3.0 + 1e-4
            best_local_val = current_val
            best_local_sc = best_sc
            
            for _ in range(35):
                mid1 = lo + (hi - lo) / 3.0
                mid2 = hi - (hi - lo) / 3.0
                
                a_trial1 = a.copy()
                a_trial1[i] = mid1
                a_trial1[n - 1 - i] = mid1
                a_trial1 = np.maximum(a_trial1, 0.0)
                s1 = a_trial1.sum()
                if s1 > 1e-15:
                    a_trial1 /= s1
                    sc1 = _score_norm(a_trial1)
                else:
                    sc1 = np.inf
                
                a_trial2 = a.copy()
                a_trial2[i] = mid2
                a_trial2[n - 1 - i] = mid2
                a_trial2 = np.maximum(a_trial2, 0.0)
                s2 = a_trial2.sum()
                if s2 > 1e-15:
                    a_trial2 /= s2
                    sc2 = _score_norm(a_trial2)
                else:
                    sc2 = np.inf
                
                if sc1 < sc2:
                    hi = mid2
                else:
                    lo = mid1
                
                # Track best
                if sc1 < best_local_sc:
                    best_local_sc = sc1
                    best_local_val = mid1
                if sc2 < best_local_sc:
                    best_local_sc = sc2
                    best_local_val = mid2
            
            if best_local_sc < best_sc * (1 - 1e-12):
                a[i] = best_local_val
                a[n - 1 - i] = best_local_val
                s = a.sum()
                if s > 1e-15:
                    a /= s
                best_sc = _score_norm(a)
                best_a = a.copy()
                improved = True
        
        sweep += 1
        if not improved and sweep > 5:
            # Reorder by current values
            order = np.argsort(a[:m])[::-1]
    
    return best_a, best_sc


def _peak_balance(a_init, deadline, n_iter=5000):
    """Iteratively balance the top peaks by redistributing mass."""
    n = len(a_init)
    a = _symmetrize(a_init).copy()
    a /= a.sum()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_iter):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _convolution(a)
        max_c = c.max()
        
        # Find top peaks
        top = np.argsort(c)[-5:]
        max_lag = int(top[-1])
        
        # Find which pairs contribute most to the max lag
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        if i_min > i_max:
            continue
        
        pairs = np.arange(i_min, i_max + 1)
        partners = max_lag - pairs
        products = a[pairs] * a[partners]
        
        # Find the largest product (most responsible pair)
        top_pair_idx = int(np.argmax(products))
        i1 = int(pairs[top_pair_idx])
        i2 = int(partners[top_pair_idx])
        
        if i1 == i2:
            # Single element: reduce it slightly
            delta = a[i1] * 0.05
            a_trial = a.copy()
            a_trial[i1] -= delta
            a_trial[n - 1 - i1] -= delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        else:
            # Two different elements: reduce the larger, increase the smaller at a safe position
            if a[i1] >= a[i2]:
                reduce_idx = i1
                increase_source = i2
            else:
                reduce_idx = i2
                increase_source = i1
            
            delta = a[reduce_idx] * 0.03
            a_trial = a.copy()
            a_trial[reduce_idx] -= delta
            a_trial[n - 1 - reduce_idx] -= delta
            
            # Add to a position that doesn't contribute to top peaks
            # Find safe position
            contrib = np.zeros(n)
            for kl in top:
                kl = int(kl)
                lo_k = max(0, kl - (n - 1))
                hi_k = min(n - 1, kl)
                for j in range(lo_k, hi_k + 1):
                    partner = kl - j
                    if 0 <= partner < n:
                        contrib[j] += a[partner]
            
            # Pick from safe positions (lowest contribution), excluding reduce_idx
            contrib[reduce_idx] = np.inf
            contrib[n - 1 - reduce_idx] = np.inf
            safe_idx = int(np.argmin(contrib))
            
            a_trial[safe_idx] += delta
            a_trial[n - 1 - safe_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-15:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    return best_a, best_sc


def _make_symmetric_profile(m, profile_type='triangular'):
    """Create a symmetric profile with m elements in the first half."""
    if profile_type == 'triangular':
        t = np.linspace(0, 1, m)
        a = (1 - np.abs(2 * t - 1))
    elif profile_type == 'raised_cosine':
        t = np.linspace(0, 1, m)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif profile_type == 'gaussian':
        t = np.linspace(0, 1, m)
        a = np.exp(-8 * (t - 0.5) ** 2)
    elif profile_type == 'quartic':
        t = np.linspace(0, 1, m)
        u = np.abs(2 * t - 1)
        a = (1 - u) ** 4
    elif profile_type == 'sech':
        t = np.linspace(0, 1, m)
        a = 1.0 / np.cosh(3 * (t - 0.5))
    elif profile_type == 'power':
        t = np.linspace(0, 1, m)
        u = np.abs(2 * t - 1)
        a = (1 - u) ** 2
    else:
        a = np.ones(m)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_full_symmetric(a_half):
    """Create full symmetric sequence from half."""
    m = len(a_half)
    n = 2 * m
    a_full = np.zeros(n)
    a_full[:m] = a_half
    a_full[m:] = a_half[::-1]
    return a_full / a_full.sum()

            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    n_work = 8192
    m_work = n_work // 2
    
    if parent is not None:
        if len(parent) != n_work:
            # Interpolate to target size
            x0 = np.linspace(0, 1, len(parent))
            x1 = np.linspace(0, 1, n_work)
            parent = np.interp(x1, x0, parent)
            parent = np.maximum(parent, 0.0)
            parent /= parent.sum()
        parent = _symmetrize(parent)
        parent /= parent.sum()
        sc_parent = _score_norm(parent)
        print(f"Parent (symmetrized): n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent_sym")
    else:
        parent = _make_full_symmetric(_make_symmetric_profile(m_work, 'triangular'))
        sc_parent = _score_norm(parent)
        print(f"No parent, triangular: n={n_work}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "triangular")
    
    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)
    print(f"Phase 0: Diverse init, deadline={phase0_end - t_start:.0f}s")
    
    profile_types = ['triangular', 'raised_cosine', 'gaussian', 'quartic', 'sech', 'power']
    candidates = []
    for pt in profile_types:
        if time.monotonic() > phase0_end:
            break
        a_half = _make_symmetric_profile(m_work, pt)
        a_full = _make_full_symmetric(a_half)
        sc = _score_norm(a_full)
        candidates.append((sc, a_full.copy(), f"prof_{pt}"))
        update_best(a_full, sc, f"prof_{pt}")
    
    # Blend parent with best profile
    if candidates:
        candidates.sort(key=lambda x: x[0])
        blend = 0.5 * parent + 0.5 * candidates[0][1]
        blend = _symmetrize(blend)
        blend /= blend.sum()
        sc_blend = _score_norm(blend)
        update_best(blend, sc_blend, "blend")
    
    # Start from best
    start_a = best_seq.copy()
    start_sc = best_score
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # --- Phase 1: L-BFGS continuation (500s) ---
    phase1_end = min(deadline, t_start + 530.0)
    print(f"Phase 1: L-BFGS, start_score={start_sc:.10f}")
    
    a_l, sc_l = _run_lbfgs_symmetric(start_a, phase1_end, max_wall_time=500.0, beta_scale=1.0)
    update_best(a_l, sc_l, "lbfgs_1")
    
    # Second pass with different beta scale
    if time.monotonic() < phase1_end - 30:
        a_l2, sc_l2 = _run_lbfgs_symmetric(best_seq.copy(), phase1_end - 5, max_wall_time=120.0, beta_scale=0.3)
        update_best(a_l2, sc_l2, "lbfgs_2")
    
    print(f"Phase 1 done: best={best_score:.10f}")
    
    # --- Phase 2: Subgradient refinement (200s) ---
    phase2_end = min(deadline, t_start + 730.0)
    print(f"Phase 2: Subgradient")
    
    a_sg, sc_sg = _run_subgradient(best_seq, phase2_end, n_active=3, initial_step=0.005)
    update_best(a_sg, sc_sg, "subgrad_3")
    
    if time.monotonic() < phase2_end - 30:
        a_sg2, sc_sg2 = _run_subgradient(best_seq, phase2_end - 5, n_active=5, initial_step=0.003)
        update_best(a_sg2, sc_sg2, "subgrad_5")
    
    print(f"Phase 2 done: best={best_score:.10f}")
    
    # --- Phase 3: Coordinate descent (300s) ---
    phase3_end = min(deadline, t_start + 1030.0)
    print(f"Phase 3: Coordinate descent")
    
    a_cd, sc_cd = _coord_descent(best_seq, phase3_end, n_sweeps=500)
    update_best(a_cd, sc_cd, "coord_descent")
    
    print(f"Phase 3 done: best={best_score:.10f}")
    
    # --- Phase 4: Peak balancing + coord descent (200s) ---
    phase4_end = min(deadline, t_start + 1230.0)
    print(f"Phase 4: Peak balance")
    
    a_pb, sc_pb = _peak_balance(best_seq, min(phase4_end - 60, time.monotonic() + 120))
    update_best(a_pb, sc_pb, "peak_balance")
    
    if time.monotonic() < phase4_end - 30:
        a_cd2, sc_cd2 = _coord_descent(best_seq, phase4_end - 5, n_sweeps=200)
        update_best(a_cd2, sc_cd2, "coord_descent_2")
    
    print(f"Phase 4 done: best={best_score:.10f}")
    
    # --- Phase 5: Multi-restart (200s) ---
    phase5_end = min(deadline, t_start + 1430.0)
    print(f"Phase 5: Multi-restart")
    
    n_restarts = 3
    per_restart = max(30.0, (phase5_end - time.monotonic()) / max(1, n_restarts))
    
    for i in range(n_restarts):
        if time.monotonic() > phase5_end - 15:
            break
        noise_level = 0.005 + 0.003 * i
        z_half = np.log(np.maximum(_full_to_half(best_seq), 1e-15))
        z_half = z_half + noise_level * rng.normal(size=len(z_half))
        z_half = np.clip(z_half, -50, 50)
        p_start = _half_to_full(z_half)
        
        dl = min(phase5_end - 10, time.monotonic() + per_restart)
        a_r, sc_r = _run_lbfgs_symmetric(p_start, dl, max_wall_time=per_restart, beta_scale=0.5)
        update_best(a_r, sc_r, f"restart_{i}")
        
        # Quick coord descent polish
        if time.monotonic() < phase5_end - 20:
            a_rc, sc_rc = _coord_descent(a_r, min(phase5_end - 15, time.monotonic() + 30), n_sweeps=50)
            update_best(a_rc, sc_rc, f"restart_{i}_cd")
    
    print(f"Phase 5 done: best={best_score:.10f}")
    
    # --- Phase 6: Final intensive polish (remaining time) ---
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = max(10.0, deadline - time.monotonic())
        print(f"Phase 6: Final polish, remaining={remaining:.0f}s")
        
        # 6a: Very fine L-BFGS
        if remaining > 60:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.35)
            a_f, sc_f = _run_lbfgs_symmetric(best_seq, dl, max_wall_time=remaining * 0.35, beta_scale=0.2)
            update_best(a_f, sc_f, "final_lbfgs")
        
        # 6b: More coordinate descent
        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.35)
            a_cd_f, sc_cd_f = _coord_descent(best_seq, dl, n_sweeps=200)
            update_best(a_cd_f, sc_cd_f, "final_cd")
        
        # 6c: Tiny subgradient
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.25)
            a_sg_f, sc_sg_f = _run_subgradient(best_seq, dl, n_active=2, initial_step=0.001)
            update_best(a_sg_f, sc_sg_f, "final_subgrad")
    
    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_full_symmetric(_make_symmetric_profile(4096, 'triangular'))
            s = float(final_seq.sum())
        final_seq = final_seq / s
        
        sc_final = _score_norm(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        # Validate with evaluator
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
            else:
                print(f"Evaluator returned non-finite: {sc_eval}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_full_symmetric(_make_symmetric_profile(4096, 'triangular'))
    sc_fb = _score_norm(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
