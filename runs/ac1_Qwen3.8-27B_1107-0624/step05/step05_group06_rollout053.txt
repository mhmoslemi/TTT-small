def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):

    out_n = n + m - 1

def smoothed_objective(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def smoothed_gradient(z, beta, n):
    a = softmax(z)
    c = full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def active_set_gradient(a, tol_frac=0.001):
    n = len(a)
    c = full_convolve(a, a)
    c_max = c.max()
    if c_max < 1e-15:
        return np.zeros(n), c, 0
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = full_convolve(w, a_rev)
    h = wc[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad, c, int(active.sum())

def lbfgs_run(a_init, beta, n, deadline, max_iter=3000, ftol=1e-14):
    z0 = np.log(np.maximum(a_init, 1e-15))
    z0 = z0 - z0.mean()
    
    def fun(z):
        return smoothed_objective(z, beta, n)
    
    def jac(z):
        return smoothed_gradient(z, beta, n)
    
    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                         options={'maxiter': max_iter, 'ftol': ftol,
                                 'gtol': 1e-10, 'maxcor': 50, 'maxls': 200,
                                 'disp': False})
        a_out = softmax(result.x)
        return a_out, result.success
    except Exception:
        return a_init, False

def subgrad_polish_momentum(a_init, deadline, max_steps=10000, lr_init=0.02, momentum_init=0.9):
    """Subgradient polishing with Nesterov momentum."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = score(a_init)
    v = np.zeros(n)  # momentum
    lr = lr_init
    momentum = momentum_init
    stale = 0
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        a = softmax(z)
        grad, c, n_active = active_set_gradient(a)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm < 1e-14:
            break
        
        # Nesterov momentum
        v = momentum * v + grad
        step = lr / (1.0 + 0.01 * i)
        z_new = z - step * v
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        
        a_new = softmax(z_new)
        sc_new = score(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            z = z_new
            stale = 0
            # Slowly decrease momentum
            momentum = max(0.5, momentum * 0.999)
        else:
            stale += 1
            if stale > 100:
                lr *= 0.5
                momentum *= 0.9
                stale = 0
                if lr < 1e-6:
                    break
    
    return best_a, best_sc

def perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return softmax(z)

def make_seed(n, kind, rng):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2*np.pi*t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(-1, 1, n)
        a = np.exp(-t**2 * 3)
        return a / a.sum()
    elif kind == "square_root":
        t = np.linspace(0, 1, n)
        a = np.sqrt(t * (1-t)) + 0.1
        return a / a.sum()
    elif kind == "random_smooth":
        # Generate random smooth sequence via low-pass filtered noise
        freqs = np.fft.rfftfreq(n)
        spectrum = rng.standard_normal(len(freqs)) * np.exp(-freqs * 3)
        a = np.fft.irfft(spectrum, n)
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses L-BFGS with log-sum-exp smoothing, Nesterov momentum subgradient polishing,
adaptive active-set handling, and diverse restarts.

from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    r = irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

            stale += 1
            if stale == 50:
                lr = max(1e-7, lr * 0.7)
                momentum = max(0.5, momentum * 0.95)
                stale = 0
            if stale >= 500 and lr < 1e-7:
                break

    return best_a, best_sc


def _subgrad_adaptive(a_init, deadline, max_steps=20000, lr_init=0.02):
    """Subgradient with adaptive step size based on improvement ratio."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    lr = lr_init
    tol_frac = 0.001

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        grad, c, n_active = _active_grad(a, tol_frac)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break

        step = lr / grad_norm
        z_new = z - step * grad
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()

        a_new = _softmax(z_new)
        sc_new = _score(a_new)

        if sc_new < best_sc - 1e-15:
            best_sc = sc_new
            best_a = a_new.copy()
            z = z_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break

    return best_a, best_sc


def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _perturb_fourier(a, rng, n_modes=3):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        amp = 0.005 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _make_seed(n, kind, rng=None):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_envelope":
        t = np.linspace(-1, 1, n)
        a = np.sinc(2.0 * t) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "random_smooth":
        if rng is None:
            rng = np.random.default_rng(0)
        freqs = np.fft.rfftfreq(n)
        spectrum = rng.standard_normal(len(freqs)) * np.exp(-freqs * 2.0)
        a = np.fft.irfft(spectrum, n)
        a = np.abs(a) + 0.01
        return a / a.sum()
    else:
        return np.ones(n) / n


def _optimize_schedule(a_start, deadline, min_remaining=60.0):
    n = len(a_start)
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_start

    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                              0.0005, 0.0002, 0.0001, 0.00005])
    current_a = a_start.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 10:
            break

        est_time_per_iter = 0.004 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))

        a_new, success = _lbfgs_run(current_a, beta, n, deadline, max_iter=max_iter)
        sc = _score(a_new)

        if sc < current_sc - 1e-12:
            current_a = a_new
            current_sc = sc
        else:
            if i >= 3:
                break

    return current_a


def _polish_cycle(a_init, deadline, n_phases=3):
    """Multiple subgradient phases with decreasing learning rates."""
    current = a_init.copy()
    current_sc = _score(current)

    for phase in range(n_phases):
        if time.monotonic() > deadline:
            break
        lr = 0.01 * (0.5 ** phase)
        tol = 0.001 * (1.0 + 0.5 * phase)
        a_new, sc_new = _subgrad_momentum(current, deadline,
                                           max_steps=10000, lr_init=lr,
                                           momentum_init=0.90,
                                           tol_frac=tol)
        if sc_new < current_sc:
            current = a_new
            current_sc = sc_new

        # Also try adaptive step size
        if time.monotonic() < deadline - 5:
            a_alt, sc_alt = _subgrad_adaptive(current, deadline,
                                              max_steps=5000, lr_init=lr * 0.5)
            if sc_alt < current_sc:
                current = a_alt
                current_sc = sc_alt

    return current, current_sc


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 2048
        a0 = _make_seed(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    # Phase 1: Main L-BFGS optimization
    print(f"\nPhase 1: Main L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    a_main = _optimize_schedule(best_seq, deadline, min_remaining=400.0)
    sc_main = _score(a_main)
    if sc_main < best_score:
        best_seq = a_main
        best_score = sc_main
    print(f"  After L-BFGS: {best_score:.10f}")

    # Phase 2: Momentum subgradient polishing
    print(f"\nPhase 2: Momentum subgradient (t={time.monotonic()-t_start:.1f}s)")
    sg_deadline = min(deadline - 200.0, time.monotonic() + 200.0)
    a_sg, sc_sg = _subgrad_momentum(best_seq, sg_deadline,
                                     max_steps=30000, lr_init=0.008,
                                     momentum_init=0.92, tol_frac=0.0005)
    if sc_sg < best_score:
        best_seq = a_sg
        best_score = sc_sg
    print(f"  After subgrad: {best_score:.10f}")

    # Phase 3: Fine L-BFGS with very small beta
    print(f"\nPhase 3: Fine L-BFGS (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 100:
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        n = len(best_seq)
        fine_betas = mb * np.array([0.00005, 0.00002, 0.00001, 0.000005])
        for i, beta in enumerate(fine_betas):
            if time.monotonic() > deadline - 50:
                break
            a_new, _ = _lbfgs_run(best_seq, beta, n, deadline, max_iter=3000)
            sc = _score(a_new)
            if sc < best_score:
                best_seq = a_new
                best_score = sc
            print(f"  fine_{i} (beta={beta:.2e}): {sc:.10f}")

    # Phase 4: Multi-restart with diverse strategies
    print(f"\nPhase 4: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 40:
        remaining = deadline - 40 - time.monotonic()
        n_restarts = min(4, max(1, int(remaining / 180)))

        restart_strategies = [
            ("perturb_002", 0.02),
            ("perturb_fourier", None),
            ("perturb_001", 0.01),
            ("blend_cosine", None),
        ]

        for i in range(n_restarts):
            if time.monotonic() > deadline - 30:
                break

            label, param = restart_strategies[i % len(restart_strategies)]

            if label == "perturb_002":
                p_start = _perturb(best_seq, 0.02, rng)
            elif label == "perturb_fourier":
                p_start = _perturb_fourier(best_seq, rng, n_modes=3)
            elif label == "perturb_001":
                p_start = _perturb(best_seq, 0.01, rng)
            else:
                seed_shape = _make_seed(len(best_seq), "raised_cosine", rng)
                p_start = 0.7 * best_seq + 0.3 * seed_shape
                p_start = np.maximum(p_start, 0)
                p_start = p_start / p_start.sum()

            sc_start = _score(p_start)
            print(f"  Restart {i} ({label}): start={sc_start:.10f}")

            a_r = _optimize_schedule(p_start, deadline, min_remaining=40.0)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_seq = a_r.copy()
                best_score = sc_r
            print(f"    After L-BFGS: {sc_r:.10f}")

            if time.monotonic() < deadline - 25:
                a_r2, sc_r2 = _subgrad_momentum(a_r, deadline - 25,
                                                 max_steps=8000, lr_init=0.005,
                                                 momentum_init=0.88, tol_frac=0.0005)
                if sc_r2 < best_score:
                    best_seq = a_r2.copy()
                    best_score = sc_r2
                print(f"    After subgrad: {sc_r2:.10f}")

    # Phase 5: Final intensive polishing cycle
    print(f"\nPhase 5: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 3:
        a_fin, sc_fin = _polish_cycle(best_seq, deadline - 2, n_phases=4)
        if sc_fin < best_score:
            best_seq = a_fin
            best_score = sc_fin
        print(f"  Final: {best_score:.10f}")

    # Phase 6: One last attempt with adaptive method
    if best_seq is not None and time.monotonic() < deadline - 2:
        a_last, sc_last = _subgrad_adaptive(best_seq, deadline - 1,
                                            max_steps=5000, lr_init=0.003)
        if sc_last < best_score:
            best_seq = a_last
            best_score = sc_last
        print(f"  After adaptive: {best_score:.10f}")

    # Final validation
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_seed(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_seed(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
