"""
Optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Strategy: top-k LSE continuation, spectral perturbation, LP peak balancing,
Adam with adaptive beta, and L-BFGS final polish. Builds on parent (n=8192).
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _topk_lse_obj(z, beta, k, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if c.max() < 1e-300:
        return 0.0
    # Top-k log-sum-exp
    k_actual = min(k, len(c))
    # Get top-k indices
    if k_actual == len(c):
        topk = c
    else:
        part = np.partition(c, -k_actual)
        threshold = part[-k_actual]
        topk_indices = np.where(c >= threshold)[0]
        topk = c[topk_indices]
    # LSE of topk
    topk_shifted = topk / beta
    topk_shifted = topk_shifted - topk_shifted.max()
    lse = float(topk.max() + beta * np.log(np.sum(np.exp(topk_shifted))))
    return lse


def _topk_lse_grad(z, beta, k, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if c.max() < 1e-300:
        return np.zeros_like(z)
    k_actual = min(k, len(c))
    if k_actual == len(c):
        topk_mask = np.ones(len(c), dtype=np.float64)
        topk_vals = c
    else:
        part = np.partition(c, -k_actual)
        threshold = part[-k_actual]
        topk_mask = (c >= threshold).astype(np.float64)
        topk_vals = c * topk_mask
    # Weights on topk
    shifted = topk_vals / beta
    shifted = shifted - shifted.max()
    e = np.exp(shifted) * topk_mask
    total = e.sum()
    if total < 1e-300:
        return np.zeros_like(z)
    w = e / total
    # Cross-correlation of w and a
    N2 = _next_pow2((2 * n - 1) + n)
    Wf = np.fft.rfft(w, N2)
    Af_full = np.fft.fft(a, N2)
    corr = np.fft.ifft(Wf * np.conj(Af_full), N2).real
    grad_a = corr[:n]
    Wa = float(np.dot(grad_a, a))
    grad_z = (1.0 / beta) * a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    N2 = _next_pow2((2 * n - 1) + n)
    Wf = np.fft.rfft(w, N2)
    Af_full = np.fft.fft(a, N2)
    corr = np.fft.ifft(Wf * np.conj(Af_full), N2).real
    grad_a = corr[:n]
    Wa = float(np.dot(grad_a, a))
    grad_z = (1.0 / beta) * a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _spectral_perturb(a, n_freqs=50, max_trials=200):
    """Try adding small sinusoidal perturbations and keep improvements."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    t = np.arange(n, dtype=np.float64)

    for trial in range(max_trials):
        # Random frequency and type
        m = np.random.randint(1, max(2, n // 4))
        phase = np.random.uniform(0.0, 2.0 * np.pi)
        # Generate perturbation
        if np.random.rand() < 0.5:
            pert = np.cos(2.0 * np.pi * m * t / n + phase)
        else:
            pert = np.sin(2.0 * np.pi * m * t / n + phase)

        # Try different amplitudes
        improved = False
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            for sign in [1.0, -1.0]:
                a_trial = best_a + sign * eps * pert
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s * a.sum()
                sc = _score_norm(a_trial)
                if sc < best_score - 1e-14:
                    best_a = a_trial.copy()
                    best_score = sc
                    improved = True
                    break
            if improved:
                break

    return best_a, best_score


def _lp_peak_balance(a, deadline, K=50, max_rounds=12):
    """LP-based peak balancing with bisection on target peak."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm(a)
    S = float(best_a.sum())

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())

        # Get active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:min(K, len(c))]
        threshold = 0.9 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 150:
            active = active[:150]

        # Build constraint matrix: (a*b)[k] <= M for k in active
        # (a*b)[k] = sum_i a[k-i]*b[i] (with bounds)
        Kc = len(active)
        A_mat = np.zeros((Kc, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(Kc, M)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 30, 'ipm_gap_rel': 1e-8})
            if res.status != 0:
                continue
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                continue
            b_scaled = b_new * (S / sum_b)
        except Exception:
            continue

        # Try blending
        current_score = _score_norm(best_a)
        improved = False
        for t_blend in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t_blend) * best_a + t_blend * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            sc_trial = _score_norm(a_trial)
            if sc_trial < current_score - 1e-13:
                best_a = a_trial.copy()
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.015, max_iter=500000):
    """Adam optimizer with adaptive beta LSE surrogate and cosine annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta_start = 0.02 * cm0
    beta_end = 0.0001 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30

    total_iters = 500000

    for it in range(total_iters):
        if time.monotonic() > deadline:
            break
        t += 1

        # Cosine annealing for beta
        frac = min(1.0, t / total_iters)
        beta = beta_start + (beta_end - beta_start) * 0.5 * (1.0 + np.cos(np.pi * frac))

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        if cm < 1e-300:
            break

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))

        N2 = _next_pow2((2 * n - 1) + n)
        Wf = np.fft.rfft(w, N2)
        Af_full = np.fft.fft(a, N2)
        corr = np.fft.ifft(Wf * np.conj(Af_full), N2).real
        grad_a = corr[:n]

        # Gradient of LSE w.r.t. z (with 1/beta factor)
        grad = (4.0 * n / beta) * a * (grad_a - W * a)
        # Actually the gradient of 2n*LSE w.r.t. a is 2n/beta * (w correlation)
        # and w.r.t. z is 2n/beta * a * (corr_a - dot(corr_a, a))
        # Let me just use: grad_z = 2n/beta * a * (grad_a - Wa) where Wa = dot(grad_a, a)
        Wa2 = float(np.dot(grad_a, a))
        grad = (2.0 * n / beta) * a * (grad_a - Wa2)

        if not np.all(np.isfinite(grad)):
            z = z + 0.05 * np.random.randn(n)
            z = z - z.mean()
            continue

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                # Perturb and reset
                z = np.log(np.maximum(best_a, 1e-15))
                z = z - z.mean()
                z = z + 0.1 * np.random.randn(n)
                z = z - z.mean()
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

    return best_a, best_score


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _parametric_init(n, etype):
    t = np.linspace(0, 1, n)
    if etype == 'binom':
        return _greedy_init(n)
    elif etype == 'exp3':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'cubic':
        a = 0.05 + 0.5 * t + 0.4 * t ** 3
    elif etype == 'raised_cos':
        a = 0.5 + 0.5 * np.cos(np.pi * (t - 0.5)) ** 2
    elif etype == 'inv_sqrt':
        a = 1.0 / np.sqrt(0.1 + t)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _run_lbfgs_topk(a_init, deadline, n):
    """L-BFGS with top-k LSE continuation."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())

    # Schedule: (beta_frac, k_frac) - k is fraction of 2n-1
    schedule = [
        (0.3, 1.0), (0.2, 0.8), (0.15, 0.5), (0.1, 0.3), (0.08, 0.2),
        (0.05, 0.1), (0.04, 0.05), (0.03, 0.03), (0.02, 0.02), (0.015, 0.01),
        (0.01, 0.005), (0.008, 0.003), (0.005, 0.002), (0.003, 0.001),
        (0.002, 0.0005),
    ]

    total_len = 2 * n - 1
    for beta_frac, k_frac in schedule:
        if time.monotonic() > deadline:
            break
        beta = beta_frac * m0
        k = max(4, int(k_frac * total_len))
        try:
            result = minimize(
                _topk_lse_obj, z, args=(beta, k, n),
                method='L-BFGS-B', jac=_topk_lse_grad,
                options={'maxiter': 5000, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxls': 100, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _run_lbfgs_full_lse(a_init, deadline, n, beta_list=None):
    """L-BFGS with full LSE (all peaks)."""
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        beta_list = [m0 * b for b in
                     [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': 5000, 'ftol': 1e-22,
                         'gtol': 1e-15, 'maxls': 100, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: top-k LSE, spectral perturbation, LP balancing,
    Adam polish, and final L-BFGS. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    np.random.seed(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Top-k LSE continuation (500s) ===
    print("Phase 1: Top-k LSE continuation")
    p1_end = min(deadline, t_start + 500.0)

    a1, sc1 = _run_lbfgs_topk(best_seq, p1_end, n)
    update_best(a1, sc1, "topk_lse")

    # Also try from greedy
    if time.monotonic() < p1_end - 60:
        a_g = _greedy_init(n)
        a_g2, sc_g2 = _run_lbfgs_topk(a_g, min(p1_end - 10, time.monotonic() + 120), n)
        update_best(a_g2, sc_g2, "topk_greedy")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Spectral perturbation (200s) ===
    print("Phase 2: Spectral perturbation")
    p2_end = min(deadline, t_start + 700.0)

    if time.monotonic() < p2_end:
        a_sp, sc_sp = _spectral_perturb(best_seq, n_freqs=50, max_trials=300)
        update_best(a_sp, sc_sp, "spectral1")

        # Re-optimize after perturbation
        if time.monotonic() < p2_end - 30:
            a_sp2, sc_sp2 = _run_lbfgs_topk(a_sp, p2_end, n)
            update_best(a_sp2, sc_sp2, "topk_after_spectral")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: LP peak balancing (250s) ===
    print("Phase 3: LP peak balancing")
    p3_end = min(deadline, t_start + 950.0)

    if time.monotonic() < p3_end:
        a_lp, sc_lp = _lp_peak_balance(best_seq, p3_end, K=50, max_rounds=12)
        update_best(a_lp, sc_lp, "lp_bal")

        # Another round with different K
        if time.monotonic() < p3_end - 30:
            a_lp2, sc_lp2 = _lp_peak_balance(best_seq, p3_end, K=100, max_rounds=8)
            update_best(a_lp2, sc_lp2, "lp_bal2")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Adam polish (400s) ===
    print("Phase 4: Adam polish")
    p4_end = min(deadline, t_start + 1350.0)

    if time.monotonic() < p4_end:
        a_ad, sc_ad = _adam_polish(best_seq, p4_end, lr=0.012)
        update_best(a_ad, sc_ad, "adam1")

        if time.monotonic() < p4_end - 60:
            a_ad2, sc_ad2 = _adam_polish(best_seq, p4_end, lr=0.005)
            update_best(a_ad2, sc_ad2, "adam2")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Full LSE with tiny beta (200s) ===
    print("Phase 5: Full LSE tiny beta")
    p5_end = min(deadline, t_start + 1550.0)

    if time.monotonic() < p5_end:
        a_f, sc_f = _run_lbfgs_full_lse(best_seq, p5_end, n)
        update_best(a_f, sc_f, "full_lse_tiny")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: More spectral + final (remaining) ===
    if time.monotonic() < deadline - 30:
        print("Phase 6: Final spectral + polish")
        a_sp3, sc_sp3 = _spectral_perturb(best_seq, n_freqs=30, max_trials=100)
        update_best(a_sp3, sc_sp3, "spectral_final")

        if time.monotonic() < deadline - 15:
            a_fin, sc_fin = _adam_polish(best_seq, deadline - 10, lr=0.003)
            update_best(a_fin, sc_fin, "final_adam")

    # === Try different length if time allows ===
    if time.monotonic() < deadline - 120 and best_seq is not None:
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"Phase 7: Try n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")
            if time.monotonic() < deadline - 20:
                a_in_b, sc_in_b = _run_lbfgs_topk(a_interp, deadline - 10, n_new)
                update_best(a_in_b, sc_in_b, f"opt_n{n_new}")

    # === Final validation ===
    print(f"\n{'='*60}")
    print(f"Final best_score (norm) = {best_score:.10f}")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL actual score = {sc_final:.10f}, n={len(final_seq)}")

        # Verify with evaluator if possible
        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
