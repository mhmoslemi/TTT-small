    c = _full_convolve(a_curr, a_curr)
    max_c = c.max()

        # Build gradient matrix: each row is gradient of c[k] w.r.t. a

        prob.solve(solver=cp.ECOS)

"""
QP-based peak flattening optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. QP/SOCP formulation using exact linearization of convolution for direct peak reduction
2. Asymmetric profile initialization (exploiting the fact that symmetric sequences have score >= 2)
3. Adaptive multi-scale QP: coarse peaks first, then fine polishing
4. Improved L-BFGS logsumexp continuation with aggressive cooling
5. Direct subgradient descent with projected steps
6. Multi-n exploration (8192, 16384)
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time
import cvxpy as cp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_skew_profile(n, alpha):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 + alpha) ** (t * alpha)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_exponential_profile(n, rate):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-rate * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_asym(n, p_left, p_right):
    t = np.linspace(0.0, 1.0, n)
    mid = np.argmax(t >= 0.5)
    a = np.ones(n)
    a[:mid] = (1.0 - t[:mid] / 0.5) ** p_left
    a[mid:] = (t[mid:] / 0.5 - 0.5) ** p_right
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_asym(n, sep_left, sep_right, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - sep_left
    c2 = 0.5 + sep_right
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + 0.8 * np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, slope):

    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_quadratic_profile(n, a1, a2):
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + a1 * u + a2 * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth, phase=0.0):

    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t + phase)

def _make_bspline_profile(n, order=4):

    u = t * 0.9 + 0.05
    u = np.clip(u, 0.01, 0.99)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _build_gradient_matrix(a, top_lags):
    n = len(a)
    K = len(top_lags)
    G = np.zeros((K, n))
    for i, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_start <= j_end:
            vals = 2.0 * a[k - j_end:k - j_start + 1][::-1]
            G[i, j_start:j_end + 1] = vals
    return G


def _qp_peak_reduction(a, deadline, K=20, rho_start=100.0, rho_end=1000.0, max_iters=100, eta_frac=0.001):
    n = len(a)
    a_curr = a.copy().astype(np.float64)
    s = a_curr.sum()
    a_curr = a_curr / s

    c = _full_convolve(a_curr, a_curr)
    max_c = float(c.max())
    best_a = a_curr.copy()
    best_max_c = max_c

    it = 0
    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        # Adaptive K: more peaks when close to optimal
        K_actual = min(K, int(len(c) * 0.01) + 5)
        top_idx = np.argsort(c)[-K_actual:]
        top_vals = c[top_idx]
        max_val = float(top_vals[-1])

        # Adaptive eta: fraction of max to reduce
        eta = eta_frac * max_val
        # As we get closer to optimal, reduce eta
        if max_c > 0:
            progress = max(0.01, 1.0 - (max_c - max_c * 0.5) / max_c)
            eta *= progress

        # Adaptive rho
        frac = it / max(1, max_iters)
        rho = rho_start + (rho_end - rho_start) * frac * frac

        # Build gradient matrix
        G = _build_gradient_matrix(a_curr, top_idx)

        # QP: min t + (rho/2)*||da||^2
        #     s.t. G @ da <= t - c[top_idx]  (i.e., c[k] + G[k]@da <= t)
        #          sum(da) = 0
        #          da >= -a_curr
        da = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            G @ da - t_var <= -top_vals,  # c[k_i] + G[i]@da <= t
            cp.sum(da) == 0,
            da >= -a_curr + 1e-15,
        ]
        objective = cp.Minimize(t_var + 0.5 * rho * cp.sum_squares(da))
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, abstol=1e-10, reltol=1e-10)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=10000)
            except Exception:
                try:
                    prob.solve(solver=cp.CVXOPT)
                except Exception:
                    break

        if da.value is None:
            break

        da_val = np.asarray(da.value, dtype=np.float64).ravel()
        if not np.all(np.isfinite(da_val)):
            break

        # Safety: ensure non-negativity and zero sum
        a_trial = a_curr + da_val
        a_trial = np.maximum(a_trial, 0.0)
        s_trial = a_trial.sum()
        if s_trial < 1e-10:
            break
        a_trial = a_trial / s_trial

        c_trial = _full_convolve(a_trial, a_trial)
        max_trial = float(c_trial.max())

        if max_trial < best_max_c * (1.0 - 1e-10):
            best_max_c = max_trial
            best_a = a_trial.copy()
            a_curr = a_trial.copy()
            c = c_trial
            max_c = max_trial
            if it % 10 == 0:
                print(f"    QP iter {it}: max_c={max_c:.12f}, rho={rho:.1f}, K={K_actual}")
        else:
            # Try smaller step
            if it > 5:
                da_val *= 0.5
                a_trial = a_curr + da_val
                a_trial = np.maximum(a_trial, 0.0)
                s_trial = a_trial.sum()
                if s_trial > 1e-10:
                    a_trial /= s_trial
                    c_trial = _full_convolve(a_trial, a_trial)
                    max_trial = float(c_trial.max())
                    if max_trial < best_max_c * (1.0 - 1e-10):
                        best_max_c = max_trial
                        best_a = a_trial.copy()
                        a_curr = a_trial.copy()
                        c = c_trial
                        max_c = max_trial
                    else:
                        break
            else:
                break

    return best_a, best_max_c


def _subgradient_descent(a, deadline, max_iters=20000, lr_start=0.01, lr_end=0.001, K=5):
    n = len(a)
    a_curr = a.copy().astype(np.float64)
    a_curr /= a_curr.sum()

    c = _full_convolve(a_curr, a_curr)
    max_c = float(c.max())
    best_a = a_curr.copy()
    best_max_c = max_c

    for it in range(max_iters):
        if time.monotonic() > deadline - 3.0:
            break

        # Adaptive learning rate
        frac = it / max(1, max_iters)
        lr = lr_start + (lr_end - lr_start) * frac

        # Find top K peaks
        K_actual = min(K, len(c))
        top_idx = np.argsort(c)[-K_actual:]
        max_val = float(c[top_idx[-1]])

        # Average subgradient
        G = _build_gradient_matrix(a_curr, top_idx)
        # Weight by proximity to max
        weights = np.exp(-(c[top_idx] - max_val) / (max_val * 0.01))
        weights /= weights.sum()
        grad = weights @ G  # shape (n,)

        # Project onto sum=0
        grad -= grad.mean()

        # Update
        a_trial = a_curr - lr * grad
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial /= a_trial.sum()

        c_trial = _full_convolve(a_trial, a_trial)
        max_trial = float(c_trial.max())

        if max_trial < best_max_c * (1.0 - 1e-10):
            best_max_c = max_trial
            best_a = a_trial.copy()
            a_curr = a_trial.copy()
            c = c_trial

        if it % 5000 == 0:
            print(f"    Subgrad iter {it}: max_c={max_c:.12f}, lr={lr:.5f}")

    return best_a, best_max_c


def _lbfgs_continuation(a_init, deadline, beta_scale=1.0, min_beta_frac=0.0001, max_wall_time=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, min_beta_frac]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            log_sum = float(np.log(s_))
            return 2.0 * n * (cm_ + beta * log_sum)

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _coordinate_descent(a, deadline, max_sweeps=200):
    n = len(a)
    a_curr = a.copy().astype(np.float64)
    a_curr /= a_curr.sum()
    c = _full_convolve(a_curr, a_curr)
    max_c = float(c.max())
    best_a = a_curr.copy()
    best_max_c = max_c

    for sweep in range(max_sweps):
        if time.monotonic() > deadline - 5.0:
            break

        c = _full_convolve(a_curr, a_curr)
        if c.max() < best_max_c * (1.0 - 1e-12):
            best_max_c = c.max()
            best_a = a_curr.copy()

        # For each coordinate, try optimal 1D move
        # c[k] depends on a[j] through: c[k] contains a[j]*a[k-j] for valid k-j
        # So changing a[j] affects c[k] for all k where k-j is in [0, n-1], i.e., k in [j, j+n-1]
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 5.0:
                break

            # Current contribution: a[j] appears in c[k] for k = j to j+n-1
            # c[k] = sum_i a[i]*a[k-i], the term with a[j] is a[j]*a[k-j] (when i=j)
            #        plus a[k-j]*a[j] (when k-i=j, i.e., i=k-j) if k-j != j
            # So c[k] has contribution: a[j]*a[k-j] + (a[k-j]*a[j] if k-j != j else 0)
            # = 2*a[j]*a[k-j] if k != 2j, or a[j]^2 if k = 2j
            # For a small change da_j, delta_c[k] = 2*a[k-j]*da_j (from both terms) or a[k-j]*da_j if k=2j

            # To find optimal a[j], we need to minimize max over affected k of:
            # c[k] + (2*a[k-j] - 2*a[j]) * a[j]  ... this gets complex
            # Simpler: try a few trial values

            if a_curr[j] < 1e-15:
                continue

            # Try reducing a[j] by various fractions and redistributing
            # The most impactful reduction is where a[j] contributes most to the max
            # Find which k is most affected
            k_start = j
            k_end = min(2 * n - 2, j + n - 1)
            if k_start > k_end:
                continue

            # Get current c values for affected k
            affected_c = c[k_start:k_end + 1].copy()

            # Try halving a[j]
            a_trial = a_curr.copy()
            saved = a_trial[j]
            a_trial[j] *= 0.5
            # Redistribute to the coordinate with lowest "danger"
            # Danger of coordinate m: sum of a[k-m] over the top peaks
            top_K = min(10, len(c))
            top_peaks = np.argsort(c)[-top_K:]
            danger = np.zeros(n)
            for kp in top_peaks:
                kp = int(kp)
                ms = max(0, kp - (n - 1))
                me = min(n - 1, kp)
                danger[ms:me + 1] += a_curr[kp - me:kp - ms + 1][::-1]

            # Add to safest coordinate (not j)
            danger[j] = np.inf
            safe_idx = int(np.argmin(danger))
            a_trial[safe_idx] += saved * 0.5
            a_trial /= a_trial.sum()

            c_trial = _full_convolve(a_trial, a_trial)
            if c_trial.max() < max_c * (1.0 - 1e-10):
                a_curr = a_trial
                c = c_trial
                max_c = float(c.max())
                improved = True
            # Also try increasing
            elif a_curr[j] < 0.5 / n:
                a_trial2 = a_curr.copy()
                a_trial2[j] *= 1.5
                a_trial2[safe_idx] -= a_curr[j] * 0.5
                if a_trial2[safe_idx] >= 0:
                    a_trial2 /= a_trial2.sum()
                    c_trial2 = _full_convolve(a_trial2, a_trial2)
                    if c_trial2.max() < max_c * (1.0 - 1e-10):
                        a_curr = a_trial2
                        c = c_trial2
                        max_c = float(c_trial2.max())
                        improved = True

        if not improved and sweep > 10:
            break

    return best_a, best_max_c

                best_max_c = max_c
                best_a = a_trial.copy()
                a_curr = a_trial.copy()

        if trial % 10000 == 0:
            print(f"    Local search trial {trial}: max_c={max_c:.12f}")

    return best_a, best_max_c


def _multi_scale_refine(a, deadline):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    # Try different resolutions
    for scale in [2, 4]:
        n_coarse = max(256, n // scale)
        a_coarse = _interp_profile(a, n_coarse)
        sc_coarse = _score_norm_fn(a_coarse)
        print(f"    Multi-scale: coarse n={n_coarse}, score={sc_coarse:.10f}")

        # Optimize at coarse scale
        dl_coarse = min(deadline - 60, time.monotonic() + 60.0)
        a_c_best, sc_c = _lbfgs_continuation(a_coarse, dl_coarse, beta_scale=1.0, max_wall_time=55.0)

        # Upsample
        a_up = _interp_profile(a_c_best, n)
        sc_up = _score_norm_fn(a_up)
        if sc_up < best_sc:
            best_sc = sc_up
            best_a = a_up.copy()
            print(f"    Multi-scale: refined n={n}, score={sc_up:.10f}")

        if time.monotonic() > deadline - 30:
            break

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=1800):
    """
    QP-based peak flattening optimizer combining:
    - Asymmetric profile initialization
    - QP/SOCP peak reduction using exact convolution linearization
    - L-BFGS logsumexp continuation
    - Direct subgradient descent
    - Coordinate descent and local search
    - Multi-scale refinement
    - Multi-n exploration
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    # --- Phase 1: Diverse initialization (40s) ---
    phase1_end = min(deadline, t_start + 40.0)
    n_work = 8192

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
        if n_parent == n_work:
            parent_work = parent
        else:
            parent_work = _interp_profile(parent, n_work)
            sc_pw = _score_norm_fn(parent_work)
            update_best(parent_work, sc_pw, "parent_interp")
    else:
        parent_work = _make_triangular_weight(n_work)
        sc_pw = _score_norm_fn(parent_work)
        print(f"No parent, triangular n={n_work}, score={sc_pw:.10f}")
        update_best(parent_work, sc_pw, "init")

    # Generate diverse profiles
    profiles = [
        _make_skew_profile(n_work, 1.0),
        _make_skew_profile(n_work, 2.0),
        _make_skew_profile(n_work, 3.0),
        _make_skew_profile(n_work, 1.5),
        _make_exponential_profile(n_work, 2.0),
        _make_exponential_profile(n_work, 3.0),
        _make_exponential_profile(n_work, 5.0),
        _make_exponential_profile(n_work, 1.0),
        _make_power_asym(n_work, 1.0, 2.0),
        _make_power_asym(n_work, 2.0, 1.0),
        _make_power_asym(n_work, 1.5, 2.5),
        _make_bimodal_asym(n_work, 0.2, 0.3, 0.12),
        _make_bimodal_asym(n_work, 0.15, 0.25, 0.1),
        _make_bimodal_asym(n_work, 0.3, 0.2, 0.15),
        _make_linear_profile(n_work, 1.0),
        _make_linear_profile(n_work, 2.0),
        _make_linear_profile(n_work, 3.0),
        _make_quadratic_profile(n_work, 0.5, 0.3),
        _make_quadratic_profile(n_work, 1.0, -0.3),
        _make_quadratic_profile(n_work, 0.8, 0.5),
        _make_w_profile(n_work, 2, 0.3, 0.5),
        _make_w_profile(n_work, 3, 0.2, 1.0),
        _make_w_profile(n_work, 4, 0.2, 0.0),
        _make_triangular_weight(n_work),
        _make_bspline_profile(n_work, 3),
        _make_bspline_profile(n_work, 4),
    ]

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase1_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Include parent
    candidates.append((_score_norm_fn(parent_work), parent_work.copy(), "parent"))
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done: best={best_score:.10f}, top={candidates[0][2]}")

    # Select top 3 starting points
    top_starts = [c[1] for c in candidates[:3]]

    # --- Phase 2: QP peak reduction (450s) ---
    phase2_end = min(deadline, t_start + 490.0)
    print("Phase 2: QP peak reduction")

    for si, start_a in enumerate(top_starts[:2]):
        if time.monotonic() > phase2_end - 60:
            break
        print(f"  QP on start {si}: initial score={_score_norm_fn(start_a):.10f}")
        dl = min(phase2_end - 30, time.monotonic() + 200.0)
        a_q, mc_q = _qp_peak_reduction(start_a, dl, K=20, rho_start=50.0, rho_end=500.0, max_iters=80, eta_frac=0.002)
        sc_q = 2.0 * len(a_q) * mc_q
        update_best(a_q, sc_q, f"qp_start{si}")
        if a_q is not None and mc_q < best_score / (2.0 * len(a_q)):
            break  # Good enough, don't try next start

    # --- Phase 3: L-BFGS continuation (300s) ---
    phase3_end = min(deadline, t_start + 790.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: L-BFGS continuation")
        dl = min(phase3_end - 10, time.monotonic() + 250.0)
        a_best, sc_best = _lbfgs_continuation(best_seq, dl, beta_scale=1.0, max_wall_time=240.0)
        update_best(a_best, sc_best, "lbfgs_main")

        # Second pass with smaller beta
        if time.monotonic() < phase3_end - 40:
            dl2 = min(phase3_end - 10, time.monotonic() + 80.0)
            a_best2, sc_best2 = _lbfgs_continuation(best_seq, dl2, beta_scale=0.3, min_beta_frac=0.00005, max_wall_time=70.0)
            update_best(a_best2, sc_best2, "lbfgs_fine")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Subgradient descent (200s) ---
    phase4_end = min(deadline, t_start + 990.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Subgradient descent")
        dl = min(phase4_end - 10, time.monotonic() + 180.0)
        a_sg, mc_sg = _subgradient_descent(best_seq, dl, max_iters=30000, lr_start=0.005, lr_end=0.0005, K=5)
        sc_sg = 2.0 * len(a_sg) * mc_sg
        update_best(a_sg, sc_sg, "subgrad")

    # --- Phase 5: More QP polishing (250s) ---
    phase5_end = min(deadline, t_start + 1240.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: QP polishing")
        dl = min(phase5_end - 10, time.monotonic() + 230.0)
        a_q2, mc_q2 = _qp_peak_reduction(best_seq, dl, K=30, rho_start=200.0, rho_end=2000.0, max_iters=60, eta_frac=0.0005)
        sc_q2 = 2.0 * len(a_q2) * mc_q2
        update_best(a_q2, sc_q2, "qp_polish")

    # --- Phase 6: Coordinate descent (150s) ---
    phase6_end = min(deadline, t_start + 1390.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Coordinate descent")
        dl = min(phase6_end - 10, time.monotonic() + 130.0)
        a_cd, mc_cd = _coordinate_descent(best_seq, dl, max_sweeps=50)
        sc_cd = 2.0 * len(a_cd) * mc_cd
        update_best(a_cd, sc_cd, "coord_descent")

    # --- Phase 7: Local perturbation search (150s) ---
    phase7_end = min(deadline, t_start + 1540.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        print("Phase 7: Local perturbation search")
        dl = min(phase7_end - 10, time.monotonic() + 130.0)
        a_ls, mc_ls = _local_perturbation_search(best_seq, dl, n_trials=30000)
        sc_ls = 2.0 * len(a_ls) * mc_ls
        update_best(a_ls, sc_ls, "local_search")

    # --- Phase 8: Try n=16384 (120s) ---
    phase8_end = min(deadline, t_start + 1660.0)
    if best_seq is not None and time.monotonic() < phase8_end:
        n_curr = len(best_seq)
        if n_curr != 16384:
            print("Phase 8: Try n=16384")
            a_16k = _interp_profile(best_seq, 16384)
            sc_16k = _score_norm_fn(a_16k)
            print(f"  Interp to 16384: score={sc_16k:.10f}")
            update_best(a_16k, sc_16k, "interp_16k")

            if time.monotonic() < phase8_end - 60:
                dl = min(phase8_end - 20, time.monotonic() + 80.0)
                a_16k_best, sc_16k_best = _lbfgs_continuation(a_16k, dl, beta_scale=0.5, max_wall_time=70.0)
                update_best(a_16k_best, sc_16k_best, "lbfgs_16k")

    # --- Phase 9: Final intensive polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 9: Final polishing, remaining={remaining:.0f}s")

        # 9a: Fine QP
        if remaining > 60:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.35)
            a_fq, mc_fq = _qp_peak_reduction(best_seq, dl, K=40, rho_start=500.0, rho_end=5000.0, max_iters=40, eta_frac=0.0002)
            sc_fq = 2.0 * len(a_fq) * mc_fq
            update_best(a_fq, sc_fq, "final_qp")

        # 9b: Fine subgradient
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.25)
            a_fs, mc_fs = _subgradient_descent(best_seq, dl, max_iters=20000, lr_start=0.002, lr_end=0.0002, K=3)
            sc_fs = 2.0 * len(a_fs) * mc_fs
            update_best(a_fs, sc_fq if sc_fq < sc_fs else sc_fs, "final_subgrad")
            # Fix: use correct score
            if np.isfinite(sc_fs) and sc_fs < best_score:
                update_best(a_fs, sc_fs, "final_subgrad")

        # 9c: Final L-BFGS with very small beta
        if time.monotonic() < deadline - 25:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.2)
            a_fl, sc_fl = _lbfgs_continuation(best_seq, dl, beta_scale=0.1, min_beta_frac=0.00001, max_wall_time=remaining * 0.2)
            update_best(a_fl, sc_fl, "final_lbfgs")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        # Validate with evaluator
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                    if sc_eval < best_score:
                        best_score = sc_eval
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular_weight(8192)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
