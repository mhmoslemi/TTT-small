```python
# This is equivalent to: for each k, add v[k] * a shifted right by k
# a shifted right by k means: (shifted_a)[j] = a[k-j] if 0<=k-j<n else 0
# = a[k-j] for j in [max(0,k-n+1), min(k, n-1)]

# So: result = zeros(n)
# for k in range(2*n-1):
#     j_start = max(0, k-n+1)
#     j_end = min(k, n-1) + 1
#     # a[k-j] for j in [j_start, j_end) is a[k-j_start ... k-j_end+1] reversed
#     # = a[k-j_end+1 : k-j_start+1][::-1]
#     idx = k - np.arange(j_start, j_end)  # = k-j for j in range
#     # idx goes from k-j_start down to k-j_end+1
#     result[j_start:j_end] += v[k] * a[idx]
```

This is O(n^2) but with a Python loop over k. For n=2048, that's 4095 iterations. In Python, this might be slow (4095 * ~1μs = ~4ms per gradient evaluation). That's acceptable for a few thousand iterations.

Alternatively, I can use FFT:
`result[j] = sum_k v[k] * a[k-j]` = cross-correlation of v and a.
Using FFT: `result = ifft(fft(v, N) * conj(fft(a, N)))` appropriately cropped.

Actually, the cross-correlation `R(j) = sum_k v[k] * a[k-j]` can be computed as:
`R = ifft(fft(v, N) * conj(fft(a, N)))` where N >= 3n.

Let me verify: `ifft(fft(v, N) * conj(fft(a, N)))[j] = sum_k v[k] * a[(k-j) mod N]`.

For j in [0, n-1] and k in [0, 2n-2], we need `(k-j) mod N = k-j` when `0 <= k-j < n` and 0 otherwise (if we pad appropriately).

If N >= 3n-2, then for j in [0, n-1] and k in [0, 2n-2]:
- `k-j` ranges from `-(n-1)` to `2n-2`.
- `(k-j) mod N = k-j` if `k >= j`, and `= k-j+N` if `k < j`.
- For `k >= j`: `k-j` is in [0, 2n-2], and `a[k-j]` is valid if `k-j < n`.
- For `k < j`: `k-j+N` is in [N-(n-1), N-1], and since `a` is padded to length N with zeros beyond index n-1, `a[k-j+N] = 0`.

So `R[j] = sum_{k=j}^{min(2n-2, j+n-1)} v[k] * a[k-j]` which is exactly what we want!

Great, so I can use FFT for the gradient computation.

```python
"""
Hybrid optimizer: LP + Smooth DCA (log-domain) + Adam for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses FFT-based convolutions, geometric programming insights via DCA,
and adaptive learning rates.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _build_lp_sparse(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status not in (0, 3):  # 0=optimal, 3=time limit (use incumbent)
            return None
        if res.x is None:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=30):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.02, 0.05, 0.08]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M


def _dca_gradient(x, beta):
    """
    Compute gradient of f_beta(x) = beta*log(sum_k exp(g_k(x)/beta)) - 2*log(sum_j exp(x[j]))
    where g_k(x) = log(sum_j exp(x[j]+x[k-j])) = log((a*a)[k]), a = exp(x).
    
    Returns gradient vector of length n.
    """
    n = len(x)
    a = np.exp(x - x.max())  # stable exp, scaled
    a_log_scale = x.max()  # a_true = a * exp(a_log_scale)
    
    c = _full_convolve(a, a)  # c[k] = sum_j a[j]*a[k-j] (scaled)
    c = np.maximum(c, 1e-300)
    
    # g_k = log(c[k]) + 2*a_log_scale (but the 2*a_log_scale cancels in softmax)
    # w_k = exp(g_k/beta) / sum_l exp(g_l/beta) = c[k]^{1/beta} / sum_l c[l]^{1/beta}
    # (the additive constants cancel)
    log_c = np.log(c)
    w = np.exp(log_c / beta - np.max(log_c) / beta)
    w /= w.sum()
    
    # Gradient of g_smooth: sum_k w_k * a[j]*a[k-j] / c[k]
    # = a[j] * sum_k (w_k/c[k]) * a[k-j]
    v = w / c  # v[k] = w[k] / c[k], length 2n-1
    
    # Compute result[j] = sum_k v[k] * a[k-j] using FFT
    # This is cross-correlation of v and a
    N = _next_pow2(3 * n)
    V = np.fft.rfft(v, N)
    A = np.fft.rfft(a, N)
    # cross-correlation: ifft(fft(v) * conj(fft(a)))
    corr = np.fft.irfft(V * np.conj(A), N)[:n]
    # corr[j] = sum_k v[k] * a[(k-j) mod N]
    # For 0<=j<n and 0<=k<2n-1<N, this gives sum_{k: 0<=k-j<n} v[k]*a[k-j] (with a=0 outside [0,n))
    
    grad_g = a * corr  # a[j] * sum_k (w_k/c[k]) * a[k-j]
    
    # Gradient of h(x) = 2*log(sum exp(x[j])):
    p = a / a.sum()
    grad_h = 2.0 * p
    
    return grad_g - grad_h


def _dca_phase(a0, deadline, beta_start, beta_end, max_iter=500000, lr_init=0.01):
    """
    Smooth DCA phase in log-domain.
    """
    n = len(a0)
    x = np.log(np.maximum(a0, 1e-300))
    x = x - x.max()  # center for numerical stability
    
    best_a = a0.copy()
    best_score = _score_norm(a0)
    
    m = np.zeros(n)  # Adam momentum
    v = np.zeros(n)  # Adam velocity
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    prev_best = best_score
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Anneal beta
        frac = t / max_iter
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** frac
        
        # Compute gradient
        grad = _dca_gradient(x, beta)
        
        if not np.all(np.isfinite(grad)):
            break
        
        # Adam update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clip update
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        x = x - update
        x = np.clip(x, -100.0, 100.0)
        
        # Evaluate periodically
        if t % 100 == 0:
            a_cur = np.exp(x - x.max())
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > 20:
                # Reduce learning rate
                lr = max(0.0001, lr * 0.8)
                patience = 0
    
    a_final = np.exp(x - x.max())
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _adam_fine(a_init, deadline, max_iter=2000000, lr=0.005):
    """
    Fine Adam polishing in log-domain with log-sum-exp smoothing.
    """
    n = len(a_init)
    x = np.log(np.maximum(a_init, 1e-300))
    x = x - x.max()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = np.exp(x - x.max())
        c = _full_convolve(a, a)
        c = np.maximum(c, 1e-300)
        c_max = float(c.max())
        
        # Use very sharp softmax (small beta) for near-exact gradient
        beta = 0.001 * c_max
        log_c = np.log(c)
        w = np.exp((log_c - log_c.max()) / beta)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        
        # Gradient
        v = w / c
        N = _next_pow2(3 * n)
        V = np.fft.rfft(v, N)
        A = np.fft.rfft(a, N)
        corr = np.fft.irfft(V * np.conj(A), N)[:n]
        p = a / a.sum()
        grad = a * corr - 2.0 * p
        
        if not np.all(np.isfinite(grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        x = x - update
        x = np.clip(x, -100.0, 100.0)
        
        if t % 200 == 0:
            a_cur = np.exp(x - x.max())
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_cur.copy()
                patience = 0
            else:
                patience += 1
            if patience > 30:
                current_lr = max(0.0001, current_lr * 0.9)
                patience = 0
    
    a_final = np.exp(x - x.max())
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _perturb_log(a, sigma, rng):
    x = np.log(np.maximum(a, 1e-300))
    x = x + sigma * rng.standard_normal(len(a))
    x = x - x.max()
    return np.exp(x - x.max())
```
