def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(n, f):
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, M):
    n = len(f)
    try:
        A = _build_conv_matrix(n, f)
        g = cp.Variable(n, non_neg=True)
        constraints = [A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        for solver in [cp.GLOP, cp.PDLP, cp.ECOS, cp.CBC]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate'] and g.value is not None:
                    return np.maximum(g.value, 0.0)
            except Exception:
                continue
    except Exception:
        pass
    return None

        if s_g < 1e-15:

        # Self-concordant step size
        t = 2.0 / (it + 2.0)
        t = min(t, 0.5)

        f_new = np.maximum(f_new, 0.0)

def _subgradient_method(f_init, deadline, rng, max_iters=50000):

        gamma = gap / (2.0 * n * max_c + 1e-15)
        gamma = min(gamma, 0.5)

        # C_k = sum_i a_i * a_{k-i}, dC_k/da_j = 2*a_{k-j} (if j != k-j) or 2*a_j (if j==k-j)
