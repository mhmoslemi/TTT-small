"""
L-BFGS-B with log-sum-exp smoothing and beta annealing for minimizing
2*n*max(convolve(a,a))/sum(a)^2. Uses softmax parameterization for
unconstrained optimization, LP warmup for initial progress, coordinate
descent for final polishing, and multi-start diversification.
"""
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2n*max(c)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoothed_obj_grad_z(z, beta):
    """Compute smoothed objective and gradient w.r.t. z (softmax parameter)."""
    n = len(z)
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    shifted = (c - c_max) / max(beta, 1e-12)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse = c_max
    else:
        w = e / s
        lse = c_max + beta * np.log(s)
    
    # Gradient of lse w.r.t. a: g_p = sum_k w_k * 2*a[k-p] (valid indices)
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    g = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    
    # Gradient w.r.t. z: grad_z_p = 2n * a_p * (g_p - dot(g,a))
    W = float(np.dot(g, a))
    grad_z = 2.0 * n * a * (g - W)
    
    obj_val = 2.0 * n * lse
    return obj_val, grad_z


def _lbfgs_on_z(z_init, beta, maxiter=500, gtol=1e-8):
    """Run L-BFGS-B on the smoothed objective in z-space."""
    n = len(z_init)
    
    def obj_and_grad(z):
        z = np.clip(z, -60.0, 60.0)
        val, g = _smoothed_obj_grad_z(z, beta)
        if not np.isfinite(val):
            return 1e10, np.zeros_like(z)
        if not np.all(np.isfinite(g)):
            g = np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)
        return val, g
    
    try:
        res = scipy_minimize(obj_and_grad, z_init, method='L-BFGS-B',
                             options={'maxiter': maxiter, 'gtol': gtol,
                                      'maxfun': 50000, 'ftol': 1e-14})
        return np.clip(res.x, -60.0, 60.0), float(res.fun)
    except Exception:
        return z_init, np.inf


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp_for_direction(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(a, rows, cols, time_limit=30):
    c = _full_convolve(a, a)
    M = float(c.max())
    g_0 = _solve_lp_for_direction(a, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return a, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(a)
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * a + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * a + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        a = (1.0 - best_t) * a + best_t * g
    return a, M


def _coord_descent(a, max_passes=20, deadline=None):
    """Coordinate descent: for each coordinate, try several multipliers."""
    n = len(a)
    multipliers = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 0.0]
    best_sc = _score_norm(a)
    
    for p in range(max_passes):
        if deadline is not None and time.monotonic() > deadline:
            break
        any_improve = False
        for j in range(n):
            if deadline is not None and time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            best_cand = old_val
            best_cand_sc = best_sc
            for m in multipliers:
                a[j] = old_val * m
                sc = _score_norm(a)
                if sc < best_cand_sc - 1e-15:
                    best_cand_sc = sc
                    best_cand = old_val * m
            if best_cand != old_val:
                a[j] = best_cand
                best_sc = best_cand_sc
                any_improve = True
            else:
                a[j] = old_val
        if not any_improve:
            break
        a = a / a.sum()
    
    return a

    return z + sigma * rng.standard_normal(len(z))
