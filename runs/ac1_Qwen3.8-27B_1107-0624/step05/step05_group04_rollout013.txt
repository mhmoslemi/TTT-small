"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent, coordinate descent, Adam, and multi-scale approach
with different hyperparameters than previous attempts.
"""
import numpy as np
from scipy.optimize import minimize
import time

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def score_fn(a):
    """Compute the actual evaluation score."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def score_norm_fn(a):
    """Score assuming sum(a) = 1 (scale-invariant part)."""
    n = len(a)
    c = full_convolve(a, a)
    return float(2.0 * n * c.max())

def upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

def make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def make_exponential(n, rate=2.0):
    t = np.linspace(0, 1, n)
    a = np.exp(rate * t)
    return a / a.sum()

def subgradient_step(a, lr, n_top=1):
    """One subgradient step: reduce the top peak."""
    c = full_convolve(a, a)
    # Get top n_top peaks
    sorted_idx = np.argsort(c)[::-1]
    top_idx = sorted_idx[:n_top]
    
    # Weighted subgradient
    g = np.zeros_like(a)
    for k in top_idx:
        # grad of c[k] w.r.t. a[i] is 2*a[k-i] for valid i
        valid_i = np.arange(max(0, k - len(a) + 1), min(len(a), k + 1))
        g[valid_i] += 2.0 * a[k - valid_i]
    
    # Update with multiplicative weights (mirror descent)
    a_new = a * np.exp(-lr * g)
    s = a_new.sum()
    if s > 1e-30:
        a_new /= s
    return a_new

    cm = float(c.max())

    N = _next_pow2(out_n)

        if not improved and rnd % 200 == 199:
            # Add noise
            z = np.log(np.maximum(best_a, 1e-15))
            z += 0.02 * rng.normal(size=n)
            z -= z.mean()
            best_a = _softmax_fn(z)
    
    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, rng, n_betas=10):
    """L-BFGS with logsumexp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z -= z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    beta_start = 0.2 * max_c0
    beta_end = 1e-5 * max_c0
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        remaining = n_betas - idx
        time_budget = min(120.0, (deadline - time.monotonic()) / max(1, remaining))
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 1000 if beta > 0.05 * max_c0 else 3000
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11,
                         'maxls': 30, 'maxfun': maxiter * 4}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -60, 60)
                z_new -= z_new.mean()
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _peak_surgery(a_init, deadline, rng, max_rounds=500):
    """Surgically reduce peaks by adjusting specific elements."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z -= z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        cm = float(c.max())
        
        # Find top 3 peaks
        top_idx = np.argsort(c)[::-1][:3]
        
        improved = False
        for pk in top_idx:
            valid_start = max(0, pk - n + 1)
            valid_end = min(n, pk + 1)
            valid_i = np.arange(valid_start, valid_end)
            prod = a_cur[valid_i] * a_cur[pk - valid_i]
            order = np.argsort(prod)[::-1][:10]
            
            for pos in order[:5]:
                idx = valid_i[pos]
                # Try reducing this element
                for delta in [0.003, 0.007, 0.015, 0.03]:
                    z_trial = z.copy()
                    z_trial[idx] -= delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try adding to small elements (spreading)
            small_idx = np.argsort(a_cur)[:10]
            for idx in small_idx[:5]:
                for delta in [0.003, 0.007, 0.015]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and rnd > 50:
            # Larger perturbation
            z = z + 0.03 * rng.normal(size=n)
            z -= z.mean()
    
    return best_a, best_score


def _pair_swap_descent(a_init, deadline, rng, max_rounds=10000):
    """Pair swap descent: swap small amounts between pairs of elements."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try several swap amounts
        for frac in [0.001, 0.005, 0.01, 0.02]:
            delta = min(best_a[i], best_a[j]) * frac
            if delta < 1e-15:
                continue
            
            # Try i -> j
            a_trial = best_a.copy()
            a_trial[i] -= delta
            a_trial[j] += delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_trial.copy()
                break
            
            # Try j -> i
            a_trial = best_a.copy()
            a_trial[j] -= delta
            a_trial[i] += delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial /= s
            sc = _score_norm_fn(a_trial)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_trial.copy()
                break
    
    return best_a, best_score
