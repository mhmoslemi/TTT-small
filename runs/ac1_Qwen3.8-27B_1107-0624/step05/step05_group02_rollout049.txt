"""
Gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient with simplex projection, RMSProp-like adaptation,
and multi-scale exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _convolve(a, b):
    """Full convolution using FFT."""
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _gradient_subgradient(a, top_k=5):
    """
    Compute subgradient of max(conv(a,a)) with respect to a.
    Uses top-k averaging for stability.
    Returns gradient vector of same length as a.
    """
    n = len(a)
    c = _full_convolve(a, a)
    
    # Find top-k positions
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        # Get top k indices
        idx = np.argsort(c)[::-1][:top_k]
    
    # Weights proportional to exp((c[i] - c_max) / beta) for stability
    c_max = c[idx[0]]
    beta = 0.1 * (c_max + 1e-15)
    w = np.exp((c[idx] - c_max) / beta)
    w = w / w.sum()
    
    # Compute gradient: for each a_k, grad_k = sum_j w_j * a_{j-k} (if 0 <= j-k < n)
    # This is equivalent to convolving w (placed at positions idx) with reversed a
    grad = np.zeros(n)
    for i, j in enumerate(idx):
        # c_j = sum_k a_k * a_{j-k} (with 0 <= j-k < n)
        # dc_j / da_k = a_{j-k} (if 0 <= j-k < n)
        # So grad_k += w_i * a_{j-k}
        # The valid range of k for a given j: max(0, j-n+1) <= k <= min(n-1, j)
        k_lo = max(0, j - n + 1)
        k_hi = min(n - 1, j)
        if k_lo <= k_hi:
            # a_{j-k} for k in [k_lo, k_hi]
            # j-k ranges from j-k_hi to j-k_lo
            indices = j - np.arange(k_lo, k_hi + 1)
            grad[k_lo:k_hi+1] += w[i] * a[indices]
    
    return grad, c


def _project_simplex(v):
    """Project v onto the probability simplex (non-negative, sum=1)."""
    n = len(v)
    # Algorithm from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _soft_gradient(a, beta):
    """Compute soft-argmax gradient of max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    w = e / s
    
    # Gradient: grad_k = sum_j w_j * a_{j-k} (valid range)
    # This is a cross-correlation: conv(w, a_reversed)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    # The relevant part is wc[n-1 : 2n-1]
    grad = wc[n-1:2*n-1]
    
    return grad, c


def _optimize_subgradient(a_init, deadline, top_k=3, lr_init=0.001):
    """
    Optimize using exact subgradient with simplex projection and RMSProp.
    """
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    
    c = _full_convolve(a, a)
    best_score = float(c.max())
    best_a = a.copy()
    
    sq_avg = np.zeros(n)
    sq_avg_beta = 0.9
    lr = lr_init
    stuck = 0
    step_count = 0
    t_start = time.monotonic()
    
    while time.monotonic() < deadline:
        # Compute subgradient
        grad, c = _gradient_subgradient(a, top_k=top_k)
        
        # Center the gradient (tangent to simplex)
        grad = grad - grad.mean()
        
        # Add barrier for small values
        barrier = np.where(a < 0.01, -10.0 / (a + 1e-10), 0.0)
        grad = grad + barrier
        
        # RMSProp update
        sq_avg = sq_avg_beta * sq_avg + (1 - sq_avg_beta) * (grad * grad)
        rms = np.sqrt(sq_avg) + 1e-8
        step = lr * grad / rms
        
        # Limit step size
        step_norm = np.linalg.norm(step)
        max_step = 0.1
        if step_norm > max_step:
            step *= max_step / step_norm
        
        # Update
        a_new = a - step
        a_new = _project_simplex(a_new)
        
        # Evaluate
        c_new = _full_convolve(a_new, a_new)
        score_new = float(c_new.max())
        
        step_count += 1
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            stuck = 0
        else:
            a = a_new  # Still accept (to escape local structure)
            stuck += 1
        
        if stuck > 500:
            # Reduce learning rate
            lr *= 0.5
            stuck = 0
            if lr < 1e-12:
                break
        
        if step_count % 1000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  SubGrad step {step_count}: score={best_score:.10f}, lr={lr:.2e}, t={elapsed:.1f}s")
            lr *= 0.995  # Slow decay
    
    return best_a, best_score


def _optimize_adam_soft(a_init, deadline, lr=0.01, beta_start=0.05):
    """
    Adam optimization with soft-argmax gradient.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    # Adaptive beta
    beta = beta_start * best_score
    min_beta = 1e-6 * best_score
    
    while time.monotonic() < deadline:
        t_step += 1
        a = _softmax(z)
        
        grad, c = _soft_gradient(a, beta)
        cm = float(c.max())
        
        if cm < best_score:
            best_score = cm
            best_a = a.copy()
            
            # Decrease beta as we improve
            beta = max(min_beta, beta * 0.999)
        
        grad = grad - grad.mean()
        
        if not np.all(np.isfinite(grad)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)
        
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update *= 0.5 / update_norm
        
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        
        if t_step % 2000 == 0:
            elapsed = time.monotonic() - deadline + deadline  # just for timing
            sc = float(_full_convolve(_softmax(z), _softmax(z)).max())
            print(f"  Adam step {t_step}: best={best_score:.10f}, beta={beta:.2e}, t={time.monotonic()-deadline+deadline:.1f}s")
            lr = max(0.001, lr * 0.95)
    
    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _interp_profile(p, n_new):
    """Interpolate a profile to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    """Perturb a sequence in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _fourier_perturb(a, rng, n_modes=10):
    """Perturb using Fourier modes for structured exploration."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    
    # Random Fourier perturbation
    freqs = rng.integers(1, n // 2, size=n_modes)
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    amps = rng.uniform(0, 0.5, size=n_modes)
    
    x = np.arange(n)
    pert = np.zeros(n)
    for f, p, a_mag in zip(freqs, phases, amps):
        pert += a_mag * np.cos(2 * np.pi * f * x / n + p)
    
    z = z + pert
    z = z - z.mean()
    return _softmax(z)


def _lp_descent(f, M, n, time_limit=20):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    try:
        # Build constraint matrix
        k = np.arange(2 * n - 1)
        j = np.arange(n)
        diff = k[:, np.newaxis] - j[np.newaxis, :]
        mask = (diff >= 0) & (diff < n)
        rows, cols = np.where(mask)
        vals = f[diff[rows, cols]]
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None

"""
Multi-strategy gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines: exact subgradient with simplex projection, Adam with soft-argmax,
LP-based descent, multi-scale interpolation, and Fourier-mode exploration.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _topk_subgradient(a, top_k=5):
    """Compute weighted subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    c = _full_convolve(a, a)
    if top_k >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argsort(c)[::-1][:top_k]
    c_max = c[idx[0]]
    spread = c[idx[0]] - c[idx[-1]] + 1e-18
    beta = 0.3 * spread
    w = np.exp((c[idx] - c_max) / beta)
    w = w / w.sum()
    grad = np.zeros(n)
    for i in range(len(idx)):
        j = int(idx[i])
        wi = w[i]
        k_lo = max(0, j - n + 1)
        k_hi = min(n - 1, j)
        if k_lo <= k_hi:
            indices = j - np.arange(k_lo, k_hi + 1)
            grad[k_lo:k_hi + 1] += wi * a[indices]
    return grad, c


def _soft_gradient(a, beta):
    """Soft-argmax gradient using FFT-based cross-correlation."""
    n = len(a)
    c = _full_convolve(a, a)
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    grad = wc[n - 1:2 * n - 1]
    return grad, c


def _optimize_subgrad(a_init, deadline, top_k=5, lr_init=0.002, momentum=0.9):
    """Subgradient descent with RMSProp and simplex projection."""
    n = len(a_init)
    a = a_init.copy()
    s = float(a.sum())
    if s < 1e-15:
        return a, np.inf
    a = a / s
    c = _full_convolve(a, a)
    best_score = float(c.max())
    best_a = a.copy()
    sq_avg = np.zeros(n)
    sq_beta = 0.9
    lr = lr_init
    mom_buf = np.zeros(n)
    stuck = 0
    step_count = 0
    t_start = time.monotonic()

    while time.monotonic() < deadline:
        grad, c = _topk_subgradient(a, top_k=top_k)
        current_max = float(c.max())
        if current_max < best_score:
            best_score = current_max
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 2000:
            lr = max(1e-12, lr * 0.5)
            stuck = 0
            if lr < 1e-12:
                break

        grad = grad - grad.mean()
        barrier = np.where(a < 0.005, -5.0 / (a + 1e-10), 0.0)
        grad = grad + barrier

        mom_buf = momentum * mom_buf + (1.0 - momentum) * grad
        sq_avg = sq_beta * sq_avg + (1 - sq_beta) * (mom_buf * mom_buf)
        rms = np.sqrt(sq_avg) + 1e-8
        step = lr * mom_buf / rms

        step_norm = np.linalg.norm(step)
        max_step = 0.05
        if step_norm > max_step:
            step *= max_step / step_norm

        a_new = a - step
        a_new = _project_simplex(a_new)
        a = a_new
        step_count += 1

        if step_count % 5000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    SubGrad [{n}]: step={step_count}, best={best_score:.10f}, lr={lr:.2e}, t={elapsed:.1f}s")
            lr = max(1e-12, lr * 0.99)

    return best_a, best_score


def _optimize_adam(a_init, deadline, lr=0.01, beta_start=0.05):
    """Adam with soft-argmax gradient in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    best_a = a_init.copy()
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    beta = beta_start * best_score
    min_beta = 1e-7 * best_score

    while time.monotonic() < deadline:
        t_step += 1
        a = _softmax_fn(z)
        grad, c = _soft_gradient(a, beta)
        cm = float(c.max())
        if cm < best_score:
            best_score = cm
            best_a = a.copy()
            beta = max(min_beta, beta * 0.998)

        grad = grad - grad.mean()
        if not np.all(np.isfinite(grad)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update *= 0.3 / update_norm
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        if t_step % 3000 == 0:
            lr = max(0.001, lr * 0.97)
            a_check = _softmax_fn(z)
            sc = float(_full_convolve(a_check, a_check).max())
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

    a_final = _softmax_fn(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()


def _fourier_perturb(a, rng, n_modes=10, amp_scale=0.4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    freqs = rng.integers(1, max(2, n // 4), size=min(n_modes, n // 4))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    amps = rng.uniform(0, amp_scale, size=len(freqs))
    x = np.arange(n)
    pert = np.zeros(n)
    for i in range(len(freqs)):
        pert += amps[i] * np.cos(2 * np.pi * freqs[i] * x / n + phases[i])
    z = z + pert
    z = z - z.mean()
    return _softmax_fn(z)


def _lp_step(f, M, n, time_limit=15):
    try:
        k = np.arange(2 * n - 1)
        j = np.arange(n)
        diff = k[:, np.newaxis] - j[np.newaxis, :]
        mask = (diff >= 0) & (diff < n)
        rows, cols = np.where(mask)
        vals = f[diff[rows, cols]]
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _lp_descent_loop(f, deadline, max_steps=50):
    """LP-based descent loop."""
    n = len(f)
    f = f / f.sum()
    current_t = 0.3
    steps = 0
    stuck = 0
    while time.monotonic() < deadline - 20 and steps < max_steps:
        c = _full_convolve(f, f)
        M = float(c.max())
        g = _lp_step(f, M, n, time_limit=12)
        if g is None:
            stuck += 1
            if stuck > 6:
                break
            g = f.copy()
        else:
            stuck = 0
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_norm = g / g_sum
        t_try = current_t
        improved = False
        for _ in range(5):
            f_new = (1 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            fs = float(f_new.sum())
            if fs < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / fs
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())
            if M_new < M * (1 - 1e-10):
                f = f_new
                current_t = min(1.0, t_try * 1.3)
                improved = True
                steps += 1
                if steps % 3 == 0:
                    print(f"    LP: step={steps}, M={M:.10f}->{M_new:.10f}, t={time.monotonic()-deadline+deadline:.1f}s")
                break
            else:
                t_try *= 0.5
        if not improved:
            current_t *= 0.5
            stuck += 1
    return f


def _try_different_ns(a_best, deadline, rng):
    """Try interpolating to different lengths and optimizing."""
    t_start = time.monotonic()
    best_local = (a_best.copy(), _score(a_best))

    target_ns = []
    n_cur = len(a_best)
    for candidate in [4096, 8192, 16384, 32768]:
        if candidate != n_cur:
            target_ns.append(candidate)

    target_ns.sort(key=lambda x: abs(x - n_cur))
    max_targets = 2

    for target_n in target_ns[:max_targets]:
        if time.monotonic() > deadline - 60:
            break
        print(f"    Trying n={target_n}")
        p_interp = _interp_profile(a_best, target_n)
        sc_interp = _score(p_interp)
        if sc_interp < best_local[1]:
            best_local = (p_interp.copy(), sc_interp)
            print(f"      Interp improved: {sc_interp:.10f}")

        remaining = deadline - time.monotonic() - 30
        if remaining > 80:
            a_opt, _ = _optimize_subgrad(p_interp, time.monotonic() + remaining * 0.4, top_k=5, lr_init=0.001)
            sc_opt = _score(a_opt)
            if sc_opt < best_local[1]:
                best_local = (a_opt.copy(), sc_opt)
                print(f"      SubGrad n={target_n}: {sc_opt:.10f}")

            if time.monotonic() < deadline - 40:
                a_opt2, _ = _optimize_adam(a_opt, deadline, lr=0.006, beta_start=0.03)
                sc_opt2 = _score(a_opt2)
                if sc_opt2 < best_local[1]:
                    best_local = (a_opt2.copy(), sc_opt2)
                    print(f"      Adam n={target_n}: {sc_opt2:.10f}")

    return best_local


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-strategy optimization combining subgradient, Adam, LP,
    multi-scale interpolation, and Fourier-mode restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) t={time.monotonic()-t_start:.1f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        update_best(a0)
    else:
        n_main = 8192
        a0 = np.ones(n_main) / n_main
        print(f"No parent, using uniform n={n_main}")
        update_best(a0)

    # Phase 1: Subgradient optimization (0 - 550s)
    phase1_end = min(deadline, t_start + 550.0)
    print(f"\n=== Phase 1: Subgradient (n={n_main}) ===")
    if time.monotonic() < phase1_end - 30:
        a_opt, _ = _optimize_subgrad(a0, phase1_end, top_k=5, lr_init=0.003, momentum=0.9)
        update_best(a_opt)
        a0 = a_opt.copy()

    print(f"Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 2: Adam soft-argmax (550s - 850s)
    phase2_end = min(deadline, t_start + 850.0)
    print(f"\n=== Phase 2: Adam soft-argmax (n={n_main}) ===")
    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_best = best_seq / best_seq.sum()
        a_opt, _ = _optimize_adam(a_best, time.monotonic() + 150, lr=0.01, beta_start=0.05)
        update_best(a_opt)
        if time.monotonic() < phase2_end - 30:
            a_opt2, _ = _optimize_adam(a0, phase2_end, lr=0.008, beta_start=0.03)
            update_best(a_opt2)

    print(f"Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 3: LP descent (850s - 1050s)
    phase3_end = min(deadline, t_start + 1050.0)
    print(f"\n=== Phase 3: LP descent (n={min(n_main, 4096)}) ===")
    if best_seq is not None and time.monotonic() < phase3_end - 40:
        f_init = best_seq.copy()
        if len(f_init) > 4096:
            f_init = _interp_profile(f_init, 4096)
        f_opt = _lp_descent_loop(f_init / f_init.sum(), phase3_end)
        # Interpolate back
        if len(best_seq) > len(f_opt):
            f_opt_full = _interp_profile(f_opt, len(best_seq))
        else:
            f_opt_full = f_opt
        update_best(f_opt_full)

    print(f"Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 4: Multi-scale exploration (1050s - 1350s)
    phase4_end = min(deadline, t_start + 1350.0)
    print(f"\n=== Phase 4: Multi-scale exploration ===")
    if best_seq is not None and time.monotonic() < phase4_end - 60:
        a_best_norm = best_seq / best_seq.sum()
        best_after_ns, sc_after_ns = _try_different_ns(a_best_norm, phase4_end, rng)
        if sc_after_ns < best_score:
            update_best(best_after_ns)

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 5: Fourier restarts + subgrad (1350s - 1600s)
    phase5_end = min(deadline, t_start + 1600.0)
    print(f"\n=== Phase 5: Fourier restarts ===")
    if best_seq is not None and time.monotonic() < phase5_end - 60:
        a_best_norm = best_seq / best_seq.sum()
        n_restarts = 3
        per_restart = (phase5_end - time.monotonic() - 40) / max(1, n_restarts)
        for i in range(n_restarts):
            if time.monotonic() > phase5_end - 40:
                break
            p_start = _fourier_perturb(a_best_norm, rng, n_modes=5 + i * 8, amp_scale=0.3 + 0.1 * i)
            dl = time.monotonic() + min(per_restart, phase5_end - time.monotonic() - 20)
            a_r, _ = _optimize_subgrad(p_start, dl, top_k=5, lr_init=0.0015, momentum=0.9)
            update_best(a_r)

    print(f"Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # Phase 6: Final polish (1600s - deadline)
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_best_norm = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 15
        if remaining > 40:
            a_final, _ = _optimize_adam(a_best_norm, time.monotonic() + remaining, lr=0.003, beta_start=0.01)
            update_best(a_final)
        elif remaining > 15:
            a_final, _ = _optimize_subgrad(a_best_norm, deadline, top_k=3, lr_init=0.0005, momentum=0.95)
            update_best(a_final)

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, t={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
