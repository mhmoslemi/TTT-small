def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _self_convolve(a):
    return _full_convolve(a, a)


def _score_norm(a):
    """Score = 2*n*max(conv) assuming sum(a)=1"""
    n = len(a)
    c = _self_convolve(a)

    s = float(a.sum())

    c = _self_convolve(a)
    return float(2.0 * n * c.max() / (s * s))

            # grad_j = a_{j+k} + a_{j-k} with boundary handling
            g = np.zeros(n)
            # a_{j+k} for j=0..n-1-k
            g[:n-k] += a[k:]
            # a_{j-k} for j=k..n-1
            g[k:] += a[:n-k]
        grad += w * g
    return grad


def _subgradient_obj_and_grad(z, n, top_k=5, temperature=0.01):
    """Objective and gradient in z-space (softmax parameterization) for sum=1 normalization."""
    a = _softmax(z)
    c = _self_convolve(a)
    obj = float(2.0 * n * c.max())
    
    # Subgradient in a-space
    grad_a = _subgradient_at_peak(a, c, top_k, temperature)
    
    # Chain rule: d obj / d z_j = 2n * sum_i grad_a_i * a_i * (delta_ij - a_j)
    # = 2n * (grad_a_j * a_j - a_j * sum_i grad_a_i * a_i)
    dot = float(np.dot(grad_a, a))
    grad_z = 2.0 * n * a * (grad_a - dot)
    return obj, grad_z


def _subgradient_descent(a_init, deadline, lr_init=0.5, top_k=5, temp_init=0.05,
                         decay=0.999, n_max_iter=5000):
    """Pure subgradient descent in z-space with adaptive parameters."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    lr = lr_init
    temp = temp_init
    t = 0
    no_improve_count = 0
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _self_convolve(a)
        sc = float(2.0 * n * c.max())
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Adaptive temperature: decrease as we converge
        max_c = float(c.max())
        temp = max(0.001, temp * decay)
        
        # Compute gradient
        grad_a = _subgradient_at_peak(a, c, top_k, temp)
        dot = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - dot)
        
        # Adaptive learning rate
        if no_improve_count > 100:
            lr *= 0.9
            no_improve_count = 0
        
        # Gradient step
        z = z - lr * grad_z
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        
        # Periodic restart to escape local minima
        if it > 0 and it % 2000 == 0:
            z = z + 0.02 * np.random.randn(n)
            z = z - z.mean()
    
    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax(z)
    c = _self_convolve(a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, beta_list=None, verbose=False):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    c0 = _self_convolve(a_init)
    max_c0 = float(c0.max())
    
    if beta_list is None:
        fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005,
                 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005, 0.00003, 0.00002]
        beta_list = [b * max_c0 for b in fracs]
    
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = len(beta_list) - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(100, int(time_per * 25))
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': maxiter * 4}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                    if verbose:
                        print(f"    L-BFGS beta={beta:.6f}: score={s_new:.10f}")
        except Exception:
            pass
    
    return best_a, best_score


def _lp_minimax(f, max_time=30):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs > 0:
                g_val = g_val / gs
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate(f, n_iters=20, deadline=None, step_size=0.3):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm(f)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _self_convolve(f)
        M = float(c.max())
        g, t_val = _lp_minimax(f)
        if g is None:
            break
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        sc = _score_norm(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new
    
    return best_a, best_score


def _make_profiles(n, rng=None):
    """Generate diverse profile initializations."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Linear slopes
    for slope in [0.5, 1.0, 1.5, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    
    # Power of cos
    for p in [0.5, 1.0, 2.0, 3.0, 4.0, 6.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    
    # Uniform
    profiles.append(np.ones(n) / n)
    
    # Parabolic (U-shape)
    a = u**2 + 0.1
    profiles.append(a / a.sum())
    
    # Inverse parabolic (hat)
    a = 1.0 - u**2 + 0.1
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Lorentzian
    a = 1.0 / (0.1 + u**2)
    profiles.append(a / a.sum())
    
    # Two peaks
    a = np.exp(-20*(u-0.4)**2) + np.exp(-20*(u+0.4)**2)
    profiles.append(a / a.sum())
    
    # Three peaks
    a = np.exp(-30*(u-0.5)**2) + np.exp(-30*(u+0.5)**2) + np.exp(-30*u**2)
    profiles.append(a / a.sum())
    
    # Edge-concentrated
    a = np.exp(-5*t) + np.exp(-5*(1-t)) + 0.1
    profiles.append(a / a.sum())
    
    # Sinusoidal bump
    a = np.maximum(np.sin(np.pi * t), 1e-12) ** 2
    profiles.append(a / a.sum())
    
    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * t))
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Super-Gaussian
    for p in [4, 6, 8, 12]:
        a = np.exp(-np.abs(u)**p)
        profiles.append(a / a.sum())
    
    # Random smooth
    if rng is not None:
        z = rng.normal(0, 0.5, n)
        # Smooth with moving average
        k = max(1, n // 100)
        kernel = np.ones(k) / k
        z_smooth = np.convolve(z, kernel, mode='same')
        a = np.exp(z_smooth)
        profiles.append(a / a.sum())
    
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _coord_descent(a_init, deadline, max_rounds=20):
    """Coordinate descent in z-space with small perturbations."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    deltas = [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax(z)
        improved = False
        # Random subset of coordinates for speed
        num_coords = min(n, max(100, n // 4))
        coords = np.random.choice(n, num_coords, replace=False)
        
        for j in coords:
            if time.monotonic() > deadline:
                break
            current_sc = best_score
            for delta in deltas:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax(z_trial)
                sc = _score_norm(a_trial)
                if sc < current_sc:
                    current_sc = sc
                    z = z_trial.copy()
                    z = np.clip(z, -50, 50)
                    z = z - z.mean()
                    improved = True
        if not improved:
            # Try larger steps
            big_deltas = [-0.2, -0.1, 0.1, 0.2]
            for j in coords[:50]:
                if time.monotonic() > deadline:
                    break
                for delta in big_deltas:
                    z_trial = z.copy()
                    z_trial[j] += delta
                    a_trial = _softmax(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        z = np.clip(z, -50, 50)
                        z = z - z.mean()
                        improved = True
        if not improved:
            break
    
    return best_a, best_score


def _paired_swap_optimization(a_init, deadline, max_rounds=10):
    """Try swapping pairs of elements to reduce peak autocorrelation."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = best_a.copy()
        improved = False
        # Try random pairs
        n_trials = min(200, n // 10)
        for _ in range(n_trials):
            if time.monotonic() > deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            # Try swapping
            a_swap = a.copy()
            a_swap[i], a_swap[j] = a_swap[j], a_swap[i]
            sc_swap = _score_norm(a_swap)
            if sc_swap < best_score:
                best_score = sc_swap
                best_a = a_swap.copy()
                improved = True
                a = best_a.copy()
            # Try transferring mass
            for frac in [0.1, 0.2, 0.3]:
                for direction in [1, -1]:
                    a_trial = a.copy()
                    amt = a_trial[i] * frac * direction
                    if a_trial[i] - amt < 0 or a_trial[j] + amt < 0:
                        continue
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    s = a_trial.sum()
                    a_trial = a_trial / s
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        improved = True
                        a = best_a.copy()
        if not improved:
            break
    
    return best_a, best_score

        if time.monotonic() < phase7_end - 10:
            per_dl2 = min(phase7_end - 5, time.monotonic() + (phase7_end - time.monotonic()) * 0.5)
            p_best2, sc_best2 = _paired_swap_optimization(best_seq, per_dl2, max_rounds=5)
            update_best(p_best2, "paired_swap")
    
    print(f"  Phase 7 done: best={best_norm_score:.10f}")
    
    # === Phase 8: Final fine-tuning (80s) ===
    phase8_end = deadline
    print(f"\nPhase 8: Final fine-tuning")
    
    if best_seq is not None and time.monotonic() < phase8_end:
        # One more ultra-fine L-BFGS
        c_f = _self_convolve(best_seq)
        mc_f = float(c_f.max())
        beta_final = [b * mc_f for b in [0.0001, 0.00005, 0.00003, 0.00002, 0.00001]]
        per_dl = min(phase8_end - 5, time.monotonic() + (phase8_end - time.monotonic()) * 0.7)
        p_best, sc_best = _run_lbfgs_continuation(best_seq, per_dl, beta_list=beta_final)
        update_best(p_best, "final_lbfgs")
        
        # Very short subgradient
        if time.monotonic() < phase8_end - 10:
            per_dl2 = min(phase8_end - 3, time.monotonic() + 30.0)
            p_best2, sc_best2 = _subgradient_descent(best_seq, per_dl2, lr_init=0.05, top_k=1, temp_init=0.005)
            update_best(p_best2, "final_sgd")
    
    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
