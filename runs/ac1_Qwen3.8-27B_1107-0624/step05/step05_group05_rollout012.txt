Active-constraint LP + smooth-max gradient + multi-start optimizer
for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_full(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_convolve_rows_cols(n, k_indices):
    """Build sparse matrix indices for convolution constraints at given k positions."""
    rows_list = []
    cols_list = []
    for idx, k in enumerate(k_indices):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        # For constraint k, variable i has coefficient f[k-i]
        # i ranges from j_start to j_end
        i_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(i_range), idx, dtype=np.int32))
        cols_list.append(i_range.astype(np.int32))
    if not rows_list:
        return np.array([], dtype=np.int32), np.array([], dtype=np.int32)
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _active_lp_step(f, K=500, time_limit=15):
    """
    One LP step using only K most active constraints.
    f should be normalized to sum=1.
    Returns the LP solution g (normalized to sum=1) or None.
    """
    n = len(f)
    c = _full_convolve(f, f)
    M = float(c.max())
    
    # Select K most active constraint positions
    if K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argsort(c)[::-1][:K]
    
    # Build sparse constraint matrix
    rows, cols = _build_convolve_rows_cols(n, top_idx)
    # The values are f[k - i] for each (k, i) pair
    # For constraint index idx (corresponding to k = top_idx[idx]), variable i:
    # value = f[k - i] = f[top_idx[idx] - i]
    vals = f[top_idx[rows] - cols]
    
    A = sp.coo_matrix((vals, (rows, cols)), shape=(len(top_idx), n)).tocsr()
    c_obj = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
            return None
    except Exception:
        pass
    return None

    One full LP step: solve LP, find best interpolation t, return updated f.

    n = len(f)
    g = _active_lp_step(f, K, time_limit)
    if g is None:
        return f, 0.0
    
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:

def _next_pow2(n):

def _full_convolve(a, b):

    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

    """Build sparse matrix for convolution constraints at positions k_indices.
    Returns (rows, cols) for COO format. The values will be f[k - col] for each entry."""
    rows_list = []
    cols_list = []
    for idx, k in enumerate(k_indices):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start > i_end:

        i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
        rows_list.append(np.full(len(i_range), idx, dtype=np.int32))
        cols_list.append(i_range)

                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return None


def _line_search(f, g, n_points=15):
    """Find best t in [0,1] for interpolation f -> (1-t)f + t*g."""
    best_t = 0.0
    best_sc = _score_norm(f)
    ts = np.linspace(0.05, 1.0, n_points)
    for t in ts:
        f_c = (1.0 - t) * f + t * g
        sc = _score_norm(f_c)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    return best_t, best_sc


def _lp_iteration(f, K=500, time_limit=10, n_ls=15):
    """One LP iteration: solve LP, line search, return updated f."""
    g = _lp_solve(f, K, time_limit)
    if g is None:
        return f, 0.0
    best_t, best_sc = _line_search(f, g, n_ls)
    if best_t > 0:
        f = (1.0 - best_t) * f + best_t * g
    return f, best_t


def _compute_gradient(a, w):
    """Compute gradient of <w, a*a> with respect to a. Assumes w has length 2n-1."""
    n = len(a)
    # grad_i = sum_j a_j w_{i+j} + sum_j a_j w_{i-j} (with valid indices)
    # First term: conv(a[::-1], w)[n-1 : 2n-1]
    # Second term: conv(a, w)[n-1 : 2n-1]
    # By symmetry of the problem (autocorrelation), both terms are equal.
    # So grad = 2 * conv(a[::-1], w)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv = _full_convolve(a_rev, w)
    h = conv[n - 1: 2 * n - 1]
    return 2.0 * h


def _adam_smooth_max(a_init, deadline, beta_start, beta_end, lr=0.01, max_iter=500000):
    """Adam optimization with smooth max objective in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        
        # Smooth max weights
        beta = beta_start * (max(beta_end, 1e-14) / beta_start) ** (t / max_iter)
        c_shift = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shift)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        
        # Gradient
        grad = _compute_gradient(a, w)
        
        # Project to simplex tangent
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        un = np.linalg.norm(update)
        if un > 1.0:
            update *= 1.0 / un
        z -= update
        z = np.clip(z, -100.0, 100.0)
        z -= z.mean()
        
        # Track best
        if t % 1000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            if t % 10000 == 0:
                sc_now = _score_norm(a)
                if best_score < prev_best * 0.998:
                    current_lr = min(0.05, current_lr * 1.1)
                else:
                    current_lr = max(0.002, current_lr * 0.92)
                prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score


def _perturb_log(a, sigma, rng):
    """Perturb in log-space."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
