def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)

    r = np.fft.irfft(fa * fb, N)[:out_n]

    if s < 1e-15:

import numpy as np
from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    P_len = len(P_indices)
    c_P = c[P_indices]
    S_P = float(c_P.sum())

    # g_i = sum_{k in P} c_k * a_{k-i} = conv(c_masked, a_rev)[n-1:2n-1]
    c_masked = np.zeros(2 * n - 1)
    c_masked[P_indices] = c_P
    a_rev = a[::-1]
    g = _full_convolve(c_masked, a_rev)[n - 1:2 * n - 1]

    # h_i = sum_{k in P} a_{k-i} = conv(ind_P, a_rev)[n-1:2n-1]
    ind_P = np.zeros(2 * n - 1)
    ind_P[P_indices] = 1.0
    h = _full_convolve(ind_P, a_rev)[n - 1:2 * n - 1]

    grad = (4.0 / P_len) * g - (4.0 * S_P / (P_len * P_len)) * h
    return grad, c


def _max_subgradient(a, P_indices):
    """Subgradient of max(c_k for k in P) w.r.t. a.
    grad_i = (2/|P|) * sum_{k in P} a_{k-i}
    """
    n = len(a)
    P_len = len(P_indices)
    ind_P = np.zeros(2 * n - 1)
    ind_P[P_indices] = 1.0
    a_rev = a[::-1]
    h = _full_convolve(ind_P, a_rev)[n - 1:2 * n - 1]
    return (2.0 / P_len) * h


def _mirror_descent(a_init, deadline, lr_init=0.005, decay=0.9995, min_lr=1e-6,
                    peak_frac=0.98, min_peaks=10, max_peaks=200):
    """Mirror descent (multiplicative updates) minimizing max of top peaks."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    lr = lr_init
    no_improve = 0
    check_every = 20

    for it in range(1000000):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        threshold = peak_frac * cm
        P = np.where(c >= threshold)[0]
        if len(P) < min_peaks:
            P = np.argsort(c)[::-1][:min_peaks]
        if len(P) > max_peaks:
            P = np.argsort(c)[::-1][:max_peaks]

        grad = _max_subgradient(a, P)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-14:
            break

        # Adaptive step: scale by 1/gnorm for stability
        step = lr / (gnorm + 1e-12)

        # Multiplicative update
        a_new = a * np.exp(-step * grad)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        # Check score periodically
        if it % check_every == 0:
            sc = _score_norm_fn(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
                a = a_new.copy()
            else:
                a = a_new.copy()
                no_improve += 1
                if no_improve > 50:
                    lr *= 0.7
                    if lr < min_lr:
                        break
                    no_improve = 0

        else:
            a = a_new.copy()

        lr *= decay
        if lr < min_lr:
            break

    return best_a, best_score

def _spectral_shaping(a_init, deadline, n_iter=500, relax=1.5):
    """Iterative projection: flatten |A|^2 in frequency, project to non-negative in time.
    This pushes toward flat autocorrelation."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    N = _next_pow2(2 * n - 1)

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        # Forward DFT
        A = np.fft.fft(a, N)
        mag2 = np.abs(A) ** 2

        # Flatten: push energy from sidelobes toward DC
        # Target: reduce max sidelobe while preserving DC
        dc = mag2[0]
        sidelobes = mag2[1:]
        max_sl = float(np.max(sidelobes)) if len(sidelobes) > 0 else 0

        if max_sl < 1e-15:
            break

        # Soft flattening: reduce large sidelobes
        alpha = 0.3
        new_mag2 = mag2.copy()
        for j in range(1, N // 2 + 1):
            if j < N - j:  # avoid double-counting conjugate
                target = dc / N * 0.5  # target sidelobe level
                if mag2[j] > target:
                    new_mag2[j] = target + alpha * (mag2[j] - target)
                    new_mag2[N - j] = new_mag2[j]

        # Reconstruct with original phases
        new_A = np.sqrt(np.maximum(new_mag2, 0)) * (A / (np.abs(A) + 1e-30))

        # Inverse DFT and project
        a_new = np.fft.ifft(new_A).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _lp_refine(a_init, deadline, K=50, max_rounds=20):
    """LP-based peak equalization: maximize sum(b) s.t. (a*b)[k] <= M for active peaks."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Active set: top K and everything within 97% of max
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.97 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = sorted_idx[:300]

        # Build A_matrix: A[k, j] = a[k-j] if valid
        K_act = len(active)
        A_mat = np.zeros((K_act, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = a_init[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(K_act, M * 1.001)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = _score_norm_fn(best_a)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            sc_trial = _score_norm_fn(a_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = sc_trial
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _kkt_solve(a_init, deadline, max_rounds=50):
    """Solve KKT conditions: find a, mu such that sum_{k in P} mu_k a_{k-i} = C (const).
    Uses iterative linear system solving."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())
        # Active set: peaks within 96% of max
        threshold = 0.96 * cm
        P = np.where(c >= threshold)[0]
        if len(P) < 3:
            P = np.argsort(c)[::-1][:5]
        if len(P) > 500:
            P = np.argsort(c)[::-1][:500]

        P_len = len(P)

        # Build Hankel-like matrix B where B[i,j] = 1 if (i+j) in P else 0
        # Then B @ a should be constant at optimum
        # Solve: min ||B@a - C*1||^2 s.t. a >= 0, sum a = 1
        # Use iterative method: a <- project(a - lr * B^T (B@a - C*1))

        # Compute B@a efficiently: (B@a)_i = sum_j [i+j in P] a_j = sum_{k in P} a_{k-i}
        # This is the same as h from _max_subgradient computation
        ind_P = np.zeros(2 * n - 1)
        ind_P[P] = 1.0
        a_rev = a[::-1]
        Ba = _full_convolve(ind_P, a_rev)[n - 1:2 * n - 1]

        # Target: make Ba constant. Gradient of ||Ba - mean(Ba)||^2 w.r.t. a:
        # = 2 * B^T (Ba - mean(Ba) * 1)
        # B^T x = sum_i x_i [i+j in P] = for each j: sum_i x_i [i in P-j]
        # (B^T x)_j = sum_i x_i * [i+j in P]
        # = conv(x, ind_P_rev)[j + n - 1] ... let's compute directly

        # (B^T x)_j = sum_i x_i * indicator(i+j in P)
        # Let's compute: for each j, sum over i where i+j is in P of x_i
        # This is: sum_{k in P} x_{k-j} (with x_m = 0 if m outside [0,n-1])
        # = conv(x_rev, ind_P)[n-1-j] ... or use the same trick
        # (B^T x)_j = sum_i x_i * ind_P[i+j]
        # = [cross-correlation of x and ind_P] at lag j
        # = np.convolve(x, ind_P[::-1], 'full')[n-1+j] ... let me verify
        # np.convolve(x, ind_P[::-1], 'full')[m] = sum_k x_k * ind_P[::-1][m-k] = sum_k x_k * ind_P[len(ind_P)-1-m+k]
        # ind_P has length 2n-1, so: = sum_k x_k * ind_P[2n-2-m+k]
        # For m = n-1+j: = sum_k x_k * ind_P[2n-2-(n-1-j)+k] = sum_k x_k * ind_P[n-1+j+k]
        # Hmm, that's not quite right. Let me just use the direct formula.

        # Simpler: B^T x at index j = sum_i x_i * [i+j in P]
        # = sum_{k in P} x_{k-j} where x_m = 0 if m not in [0, n-1]
        # This is the same structure as before: conv(x, ind_P_rev_shifted)

        # Actually, let's just compute the gradient numerically via the structure:
        # grad_j = 2 * (Ba_j - mean_Ba) * (sum of something) ... 
        # Actually the gradient of (1/2)||Ba - C1||^2 w.r.t. a is B^T(Ba - C1)
        # and B^T(Ba - C1) at index j = sum_i (Ba_i - C) * [i+j in P]

        # Let's compute this as: for the vector d = Ba - mean(Ba)*1,
        # (B^T d)_j = sum_i d_i * [i+j in P]
        d = Ba - Ba.mean()
        # (B^T d)_j = sum_i d_i * ind_P[i+j]
        # = sum_{k in P} d_{k-j} (with d_m = 0 if m outside [0,n-1])
        d_ext = np.zeros(2 * n - 1)
        d_ext[:n] = d
        BTd = _full_convolve(d_ext[::-1], ind_P)[n - 1:2 * n - 1]

        # Wait, let me verify: _full_convolve(d_ext[::-1], ind_P)[m]
        # = sum_k d_ext[::-1][k] * ind_P[m-k] = sum_k d_ext[2n-2-k] * ind_P[m-k]
        # For m = n-1+j: = sum_k d_ext[2n-2-k] * ind_P[n-1+j-k]
        # = sum_k d_{2n-2-k} * [n-1+j-k in P]  (d_ext is zero outside [0,n-1])
        # Let k' = 2n-2-k, so k = 2n-2-k': = sum_{k'=n-1}^{2n-2} d_{k'} * [n-1+j-(2n-2-k') in P]
        # = sum_{k'=n-1}^{2n-2} d_{k'} * [k'-n+1+j in P]
        # Hmm this is getting confusing. Let me just use a different approach.

        # Direct computation: BTd[j] = sum_i d[i] * (1 if (i+j) in P else 0)
        # Let's compute using FFT: create a matrix-free product
        # Actually for n=8192 and |P| up to 500, direct computation is O(n*|P|) = 4M, which is fine.

        # Let's just do it directly for correctness:
        BTd_direct = np.zeros(n)
        for k in P:
            # BTd_direct[j] += d[k-j] for valid j, i.e., 0 <= k-j < n, so k-n+1 <= j <= k
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            if j_lo <= j_hi:
                BTd_direct[j_lo:j_hi + 1] += d[k - j_lo:k - j_lo + (j_hi - j_lo + 1)]

        # Gradient ascent on making Ba flat: a <- a - lr * BTd (then project)
        lr_kkt = 0.1 / (float(np.max(np.abs(BTd_direct))) + 1e-12)
        a_new = a - lr_kkt * BTd_direct
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = _score_norm_fn(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _lbfgs_continuation(a_init, deadline, beta_list=None, maxiter=5000):
    """L-BFGS with log-sum-exp smoothing of max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001, 0.00005, 0.00002]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def obj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)

        def grad(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return np.zeros_like(z_in)
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            e = np.exp(shifted - max_s)
            w = e / e.sum()
            W = float(np.dot(w, c_in))
            a_rev = a_in[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_in * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_in)
            return g

        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad,
                              options={'maxiter': maxiter, 'ftol': 1e-22,
                                       'gtol': 1e-15, 'maxls': 80, 'maxfun': 20000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_polish(a_init, deadline, lr=0.01, max_iter=200000):
    """Adam optimizer with log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta_base = 0.005 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-8 * cm, beta_base * (0.98 ** (t // 100)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % 50 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _random_init(n, rng, etype='gibbs'):
    """Generate diverse random initializations."""
    if etype == 'gibbs':
        a = np.exp(0.5 * np.sin(2 * np.pi * rng.uniform(1, 8) * np.linspace(0, 1, n))
                  + 0.3 * rng.normal(0, 1, n))
    elif etype == 'bimodal':
        centers = [rng.uniform(0, 0.3), rng.uniform(0.7, 1.0)]
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-0.5 * ((np.linspace(0, 1, n) - c) / rng.uniform(0.05, 0.2)) ** 2)
        a += 0.01 * np.random.uniform(0, 1, n)
    elif etype == 'oscillatory':
        f = rng.uniform(2, 20)
        a = 1.0 + 0.5 * np.cos(2 * np.pi * f * np.linspace(0, 1, n) + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n) * (0.5 + 0.5 * rng.random(n))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800):
    """

    1. Mirror descent (multiplicative) with adaptive peaks
    2. Variance-based mirror descent for flatter peaks

        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:

            s = parent_raw.sum()
            if s > 0.01:

    except Exception:

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 4096
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting: n={n}, initial best={best_score:.10f}")

    # === Phase 1: Mirror descent - multiplicative updates (350s) ===
    print("Phase 1: Mirror descent (multiplicative)")
    p1_end = min(deadline, t_start + 350.0)
    if time.monotonic() < p1_end and best_seq is not None:
        a_md, sc_md = _mirror_descent(best_seq, p1_end, lr_init=0.003,
                                      decay=0.9997, peak_frac=0.99,
                                      min_peaks=20, max_peaks=300)
        update_best(a_md, sc_md, "mirror_max")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Variance mirror descent (250s) ===
    print("Phase 2: Variance mirror descent")
    p2_end = min(deadline, t_start + 600.0)
    if time.monotonic() < p2_end and best_seq is not None:
        a_vm, sc_vm = _variance_mirror_descent(best_seq, p2_end, lr_init=0.005,
                                               decay=0.9995, peak_frac=0.95,
                                               min_peaks=10, max_peaks=150)
        update_best(a_vm, sc_vm, "variance_mirror")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Spectral shaping (200s) ===
    print("Phase 3: Spectral shaping")
    p3_end = min(deadline, t_start + 800.0)
    if time.monotonic() < p3_end and best_seq is not None:
        a_spec, sc_spec = _spectral_shaping(best_seq, p3_end, n_iter=200)
        update_best(a_spec, sc_spec, "spectral")

        # Also try from a different init
        if time.monotonic() < p3_end - 50:
            a_g = _greedy_init(n)
            a_spec2, sc_spec2 = _spectral_shaping(a_g, p3_end, n_iter=100)
            update_best(a_spec2, sc_spec2, "spectral_greedy")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: KKT-based solving (200s) ===
    print("Phase 4: KKT solving")
    p4_end = min(deadline, t_start + 1000.0)
    if time.monotonic() < p4_end and best_seq is not None:
        a_kkt, sc_kkt = _kkt_solve(best_seq, p4_end, max_rounds=100)
        update_best(a_kkt, sc_kkt, "kkt")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: LP refinement (200s) ===
    print("Phase 5: LP refinement")
    p5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p5_end and best_seq is not None:
        a_lp, sc_lp = _lp_refine(best_seq, p5_end, K=50, max_rounds=12)
        update_best(a_lp, sc_lp, "lp_main")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: L-BFGS polish (200s) ===
    print("Phase 6: L-BFGS polish")
    p6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < p6_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]
        a_lb, sc_lb = _lbfgs_continuation(best_seq, p6_end, beta_list=tiny_betas, maxiter=5000)
        update_best(a_lb, sc_lb, "lbfgs_polish")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Adam final polish (150s) ===
    print("Phase 7: Adam polish")
    p7_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p7_end and best_seq is not None:
        a_ad, sc_ad = _adam_polish(best_seq, p7_end, lr=0.005, max_iter=100000)
        update_best(a_ad, sc_ad, "adam_final")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Random restarts + local opt (180s) ===
    print("Phase 8: Random restarts")
    p8_end = min(deadline, t_start + 1730.0)
    if time.monotonic() < p8_end and best_seq is not None:
        for trial in range(5):
            if time.monotonic() > p8_end:
                break
            etype = ['gibbs', 'oscillatory', 'bimodal', 'gibbs', 'oscillatory'][trial]
            a_r = _random_init(n, rng, etype)
            # Blend with best
            a_blend = 0.5 * best_seq + 0.5 * a_r
            a_blend = a_blend / a_blend.sum()
            # Quick mirror descent
            a_rm, sc_rm = _mirror_descent(a_blend, min(p8_end - 10, time.monotonic() + 20),
                                          lr_init=0.002, peak_frac=0.98, min_peaks=10, max_peaks=100)
            update_best(a_rm, sc_rm, f"restart_{etype}_{trial}")

    # === Phase 9: Try different n values (170s) ===
    print("Phase 9: Multi-n exploration")
    p9_end = deadline - 10
    if time.monotonic() < p9_end and best_seq is not None:
        for n_try in [2 * n, n // 2]:
            if n_try < 64 or n_try > 16384:
                continue
            if time.monotonic() > p9_end - 30:
                break
            a_int = _interp_profile(best_seq, n_try)
            sc_int = _score_norm_fn(a_int)
            update_best(a_int, sc_int, f"interp_n{n_try}")
            if time.monotonic() < p9_end - 30:
                a_md_n, sc_md_n = _mirror_descent(a_int, min(p9_end - 10, time.monotonic() + 30),
                                                  lr_init=0.002, peak_frac=0.98,
                                                  min_peaks=10, max_peaks=100)
                update_best(a_md_n, sc_md_n, f"mirror_n{n_try}")

    print(f"  Phase 9 done: best={best_score:.10f}")

    # === Final: One more L-BFGS + Adam cycle on the global best ===
    if time.monotonic() < deadline - 15 and best_seq is not None:
        c_f = _full_convolve(best_seq, best_seq)
        m_f = float(c_f.max())
        final_betas = [m_f * b for b in [0.0001, 0.00005, 0.00002, 0.00001]]
        a_fin, sc_fin = _lbfgs_continuation(best_seq, deadline - 5, beta_list=final_betas, maxiter=3000)
        update_best(a_fin, sc_fin, "final_lbfgs")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    fb = _greedy_init(256)
    print(f"FALLBACK: n=256, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
