def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return obj, np.zeros_like(z)
    return obj, grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                lambda z_in: _lse_obj_and_grad(z_in, beta, n)[0],
                z, method='L-BFGS-B',
                jac=lambda z_in: _lse_obj_and_grad(z_in, beta, n)[1],
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _lp_peak_balance(a, deadline, K=50, max_rounds=20):
    """LP-based peak balancing with iterative blending."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())
        if S < 1e-15:
            break

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.90 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = sorted_idx[:300]

        # Build constraint matrix: A[k, j] = a[k-j] if 0<=k-j<n
        Kc = len(active)
        A_mat = np.zeros((Kc, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(Kc, M * 1.0005)
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 20})
            if res.status != 0:
                break
            b_new = np.maximum(res.x, 0)
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break
        if not improved:
            break

    return best_a, best_score


def _adam_optimize(a_init, deadline, lr=0.01, max_iter=500000, beta_decay=0.97):
    """Adam optimizer with adaptive beta and periodic L-BFGS restarts."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.02 * cm0

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 20
    lbnbfgs_interval = 5000
    lbnbfgs_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.02 * (beta_decay ** (t // 100)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1 - b1) * grad
        v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                z = z + 0.03 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0

        # Periodic L-BFGS restart
        if t % lbnbfgs_interval == 0:
            lbnbfgs_count += 1
            if lbnbfgs_count < 10 and time.monotonic() < deadline - 60:
                a_cur = _softmax_fn(z)
                c_b = _full_convolve(a_cur, a_cur)
                m_b = float(c_b.max())
                tiny_betas = [m_b * 0.001, m_b * 0.0005, m_b * 0.0002]
                a_r, sc_r = _run_lbfgs(a_cur, deadline - 30, beta_list=tiny_betas,
                                       maxiter_per_beta=2000)
                if sc_r < best_score:
                    best_score = sc_r
                    best_a = a_r.copy()
                    z = np.log(np.maximum(a_r, 1e-15))
                    z = z - z.mean()
                    m_buf = np.zeros(n)
                    v_buf = np.zeros(n)
                    t = 0
                    no_improve = 0

    return best_a, best_score


def _block_descent(a_init, deadline, block_sizes=(16, 32, 64, 128)):
    """Block coordinate descent with L-BFGS on each block."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for B in block_sizes:
        if time.monotonic() > deadline:
            break
        n_blocks = n // B
        for sweep in range(5):
            if time.monotonic() > deadline:
                break
            for blk in range(n_blocks):
                if time.monotonic() > deadline - 10:
                    break
                p = blk * B
                block = best_a[p:p+B].copy()
                rest_sum = float(best_a.sum()) - float(block.sum())

                def block_obj(x, p=p, B=B, rest_sum=rest_sum):
                    x = np.maximum(x, 0)
                    a_trial = best_a.copy()
                    a_trial[p:p+B] = x
                    s = rest_sum + x.sum()
                    if s < 1e-15:
                        return 1e10
                    c = _full_convolve(a_trial, a_trial)
                    return float(2.0 * n * c.max() / (s * s))

                try:
                    res = minimize(block_obj, block, method='L-BFGS-B',
                                   bounds=[(0, None)] * B,
                                   options={'maxiter': 50, 'ftol': 1e-15,
                                            'gtol': 1e-10, 'maxfun': 200})
                    new_block = np.maximum(res.x, 0)
                except Exception:
                    continue

                a_trial = best_a.copy()
                a_trial[p:p+B] = new_block
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                sc = float(2.0 * n * _full_convolve(a_trial, a_trial).max() / (s * s))
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_trial.copy()

    return best_a, best_score


def _peak_directed_optimize(a_init, deadline, K=8):
    """Focus optimization on top K peaks using active-set L-BFGS."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for outer in range(50):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]

        def pobj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            beta = 1e-6 * pm
            sh = (peaks - pm) / beta
            lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)

        def pgrad(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            beta = 1e-6 * pm
            sh = (peaks - pm) / beta
            e = np.exp(sh - sh.max())
            w = e / e.sum()
            grad_a = np.zeros(n)
            for idx_k in range(len(peak_indices)):
                k = peak_indices[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            result = minimize(pobj, z, method='L-BFGS-B', jac=pgrad,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 50, 'maxfun': 15000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):

        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()

            if s > 0.01:

    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
