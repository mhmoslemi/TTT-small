import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp

import sys

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    """Returns 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    """Returns the actual score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lp_minimax_active(f, active_indices, max_time=30):
    """Solve LP: min t s.t. (f*g)[k] <= t for k in active_indices, sum(g)=1, g>=0."""
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    
    # Build constraint matrix for active indices only
    # (f*g)[k] = sum_j f[max(0, k-n+1):min(n, k+1)] paired with g
    # For a given k, (f*g)[k] = sum_j f[j]*g[k-j] where 0<=j<n and 0<=k-j<n
    # = sum_j f[j]*g[k-j] where j ranges over max(0,k-n+1) to min(n-1,k)
    # Let me build the matrix rows for each active index k
    
    rows = []
    for k in active_indices:
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
        # j must satisfy: 0 <= j <= n-1 and 0 <= k-j <= n-1
        # So: max(0, k-(n-1)) <= j <= min(n-1, k)
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            row = np.zeros(n)
            row[js] = f[js]
            rows.append(row)
    
    if not rows:
        return None, None
    
    A = np.array(rows)  # shape (m, n)
    m = len(active_indices)
    
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None

def _lp_iterate_active(f, deadline, n_iters=1000, active_k=20, step_size_init=1.0, step_decay=0.95):
    """LP-based iterated minimax with active constraints."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    step_size = step_size_init
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(f, f)
        # Get active indices (top k)
        if active_k >= len(c):
            active_idx = np.arange(len(c))
        else:
            active_idx = np.argsort(c)[-active_k:]
        
        g, t_val = _lp_minimax_active(f, active_idx)
        if g is None:
            break
        
        f_new = (1 - step_size) * f + step_size * g
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s
        
        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
            # Reset step size when we make progress
            step_size = min(1.0, step_size * 1.2)
        else:
            # Decay step size when we don't improve
            step_size *= step_decay
        
        f = f_new
        
        if it % 50 == 0:
            print(f"    LP act {it}: score={sc:.10f}, best={best_score:.10f}, step={step_size:.3f}")
    
    return best_a, best_score

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _run_lbfgs(a_init, deadline, beta_start=None, beta_end=None, n_betas=15):
    """L-BFGS with geometric beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if beta_start is None:
        beta_start = 0.1 * max_c0
    if beta_end is None:
        beta_end = 1e-5 * max_c0
    
    # Geometric sequence of betas
    ratio = (beta_end / beta_start) ** (1.0 / (n_betas - 1))
    
    for idx in range(n_betas):
        if time.monotonic() > deadline:
            break
        beta = beta_start * (ratio ** idx)
        remaining_betas = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining_betas)
        time_per = min(time_per, 60.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(200, int(time_per * 30))
        
        try:
            result = minimize(
                _logsumexp_obj, z.copy(), args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _coord_descent(a_init, deadline, max_rounds=30):
    """Coordinate descent on log-scale parameters."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        # Random order for diversity
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            current_score = _score_norm_fn(_softmax_fn(z))
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < current_score:
                    current_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                improved = True
        if improved:
            a = _softmax_fn(z)
            sc = _score_norm_fn(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        else:
            break
    
    return best_a, best_score

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _make_initial_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Uniform
    profiles.append(np.ones(n) / n)
    
    # Various shapes
    for p in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = np.maximum(1.0 - np.abs(u)**p, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential shapes
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0, -3.0]:
        a = np.exp(rate * u)
        profiles.append(a / a.sum())
    
    # Semicircle-like
    a = np.sqrt(np.maximum(1 - u**2, 1e-12))
    profiles.append(a / a.sum())
    
    # Bimodal
    a = np.exp(-10*u**2) + 0.5 * np.exp(-2*(u-0.8)**2)
    profiles.append(a / a.sum())
    
    # Triangular
    a = np.maximum(1 - np.abs(u), 1e-12)
    profiles.append(a / a.sum())
    
    return profiles

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_active_matrix(f, active_indices):
    n = len(f)
    rows = []
    for k in active_indices:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            row = np.zeros(n)
            row[j_min:j_max + 1] = f[j_min:j_max + 1]
            rows.append(row)
    if not rows:
        return None
    return np.array(rows)


def _lp_solve_active(f, active_indices):
    n = len(f)
    A = _build_active_matrix(f, active_indices)
    if A is None:
        return None, None
    m = A.shape[0]
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False, tol_gap_abs=1e-12, tol_gap_rel=1e-12, tol_feas=1e-12)
        if g.value is not None and np.all(np.isfinite(g.value)):
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-15:
                g_val = g_val / s
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_iterate_growing(f, deadline, max_iters=5000, start_k=10, max_k=100, avg_window=5):
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    active_set = set()
    avg_history = []
    step = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(f, f)
        current_max = float(c.max())

        # Grow active set
        k_this = min(start_k + it // 10, max_k, len(c))
        top_idx = np.argsort(c)[-k_this:]
        active_set = active_set | set(top_idx.tolist())

        # Solve LP with current active set
        g, t_val = _lp_solve_active(f, sorted(active_set))
        if g is None:
            break

        # Update with Nesterov-style averaging
        avg_history.append(g)
        if len(avg_history) > avg_window:
            avg_history.pop(0)
        f_new = np.mean(avg_history, axis=0)
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s

        sc = _score_norm_fn(f_new)
        if sc < best_score:
            best_score = sc
            best_a = f_new.copy()
        f = f_new

        # Convergence check
        if it > 50 and abs(current_max - t_val) < 1e-8:
            pass
        if it % 100 == 0:
            print(f"    LPgrow {it}: score={sc:.10f}, best={best_score:.10f}, |K|={len(active_set)}, t*={t_val:.8f}")

    return best_a, best_score

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _run_lbfgs_continuation(a_init, deadline, n_betas=20, beta_frac_start=0.08, beta_frac_end=1e-5):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = max(float(c0.max()), 1e-10)
    beta_start = beta_frac_start * max_c0
    beta_end = beta_frac_end * max_c0

    ratio = (beta_end / beta_start) ** (1.0 / max(1, n_betas - 1))

    for idx in range(n_betas):
        if time.monotonic() > deadline:
            break
        beta = beta_start * (ratio ** idx)
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 45.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(100, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z.copy(), args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 40, 'maxfun': maxiter * 5}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score

def _coord_descent(a_init, deadline, max_rounds=40):

    best_a = a_init.copy()

    z = np.log(np.maximum(a_init, 1e-15))

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        current_score = _score_norm_fn(a)
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local = current_score
            for delta in [-0.08, -0.04, -0.02, -0.01, 0.01, 0.02, 0.04, 0.08]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
            if best_delta != 0.0:

                z = np.clip(z, -80, 80)
                z = z - z.mean()
                improved = True

            sc = _score_norm_fn(a)
            if sc < best_score:

                best_a = a.copy()
        else:

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else np.ones(n_new) / n_new
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        cm = float(c.max())
        beta = max(1e-8, 0.02 * cm)
        vals = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        if not np.all(np.isfinite(grad)):
            break
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        a = _softmax_fn(z)
        sc = _score_norm_fn(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 3000 == 0:
            lr = max(0.01, lr * 0.85)
        if t % 10000 == 0:
            z = z + 0.02 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Strategy: Growing active-constraint LP with Nesterov averaging, multi-n exploration,
    L-BFGS with geometric beta continuation, adaptive GD, and coordinate descent polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_init = 2048
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = _score_norm_fn(parent)
        print(f"Default init: n={n_init}, score={best_score:.10f}")

    n_parent = len(best_seq)

    # Helper to update best
    def try_update(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # ==========================================
    # Phase 1: Multi-n growing LP exploration (45%)
    # ==========================================
    phase1_end = t_start + 0.45 * (deadline - t_start)
    print(f"\nPhase 1: Growing LP (until {phase1_end - t_start:.0f}s)")

    n_values = sorted(set([2048, 3072, 4096]))
    if n_parent not in n_values:
        n_values.append(n_parent)
        n_values.sort()

    n_per = (phase1_end - time.monotonic()) / max(1, len(n_values))

    for n_val in n_values:
        if time.monotonic() > phase1_end - 20:
            break
        print(f"\n  n={n_val}:")
        a_init = _interp_profile(best_seq, n_val)
        sc = _score_norm_fn(a_init)
        try_update(a_init, sc, f"init_n{n_val}")

        per_dl = min(phase1_end - 10, time.monotonic() + n_per * 0.7)
        a_out, sc_out = _lp_iterate_growing(a_init, per_dl, max_iters=5000, start_k=10, max_k=80, avg_window=5)
        try_update(a_out, sc_out, f"grow_lp_n{n_val}")

        # Second pass with fixed active set
        if time.monotonic() < phase1_end - 30:
            per_dl2 = min(phase1_end - 10, time.monotonic() + n_per * 0.3)
            a_out2, sc_out2 = _lp_iterate_fixed(a_out, per_dl2, max_iters=2000, active_k=40)
            try_update(a_out2, sc_out2, f"fix_lp_n{n_val}")

    print(f"\n  Phase 1 done: best={best_score:.10f} (n={len(best_seq)})")

    # ==========================================
    # Phase 2: L-BFGS continuation (25%)
    # ==========================================
    phase2_end = t_start + 0.70 * (deadline - t_start)
    print(f"\nPhase 2: L-BFGS (until {phase2_end - t_start:.0f}s)")

    # L-BFGS at best n
    if time.monotonic() < phase2_end - 20:
        per_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.55)
        n_curr = len(best_seq)
        print(f"  L-BFGS at n={n_curr}:")
        a_out, sc_out = _run_lbfgs_continuation(best_seq, per_dl, n_betas=18, beta_frac_start=0.06, beta_frac_end=5e-6)
        try_update(a_out, sc_out, f"lbfgs_n{n_curr}")

    # Try n=4096 if not already
    if len(best_seq) != 4096 and time.monotonic() < phase2_end - 20:
        a_4096 = _interp_profile(best_seq, 4096)
        sc_4096 = _score_norm_fn(a_4096)
        try_update(a_4096, sc_4096, "interp_n4096")
        per_dl = min(phase2_end - 10, time.monotonic() + (phase2_end - time.monotonic()) * 0.4)
        print(f"  L-BFGS at n=4096:")
        a_out, sc_out = _run_lbfgs_continuation(a_4096, per_dl, n_betas=14, beta_frac_start=0.05, beta_frac_end=5e-6)
        try_update(a_out, sc_out, "lbfgs_n4096")

    # Try n=2048 if best is larger
    if len(best_seq) > 2048 and time.monotonic() < phase2_end - 15:
        a_2048 = _interp_profile(best_seq, 2048)
        sc_2048 = _score_norm_fn(a_2048)
        try_update(a_2048, sc_2048, "interp_n2048")
        if time.monotonic() < phase2_end - 10:
            per_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.3)
            a_out, sc_out = _run_lbfgs_continuation(a_2048, per_dl, n_betas=10, beta_frac_start=0.05, beta_frac_end=5e-6)
            try_update(a_out, sc_out, "lbfgs_n2048")

    print(f"\n  Phase 2 done: best={best_score:.10f} (n={len(best_seq)})")

    # ==========================================
    # Phase 3: Adaptive GD + LP restart (15%)
    # ==========================================
    phase3_end = t_start + 0.85 * (deadline - t_start)
    print(f"\nPhase 3: GD + restarts (until {phase3_end - t_start:.0f}s)")

    if time.monotonic() < phase3_end - 20:
        # Adaptive GD
        per_dl = min(phase3_end - 30, time.monotonic() + (phase3_end - time.monotonic()) * 0.4)
        print(f"  Adaptive GD:")
        a_out, sc_out = _adaptive_gd(best_seq, per_dl, rng, lr_init=0.1)
        try_update(a_out, sc_out, "adaptive_gd")

        # LP from perturbed state
        if time.monotonic() < phase3_end - 20:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z_pert = z + 0.03 * rng.normal(size=len(z))
            z_pert = np.clip(z_pert, -60, 60)
            z_pert = z_pert - z_pert.mean()
            a_pert = _softmax_fn(z_pert)
            per_dl2 = min(phase3_end - 10, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
            print(f"  LP from perturbation:")
            a_out2, sc_out2 = _lp_iterate_growing(a_pert, per_dl2, max_iters=3000, start_k=15, max_k=60, avg_window=4)
            try_update(a_out2, sc_out2, "perturb_lp")

    print(f"\n  Phase 3 done: best={best_score:.10f} (n={len(best_seq)})")

    # ==========================================
    # Phase 4: Final polish (10%)
    # ==========================================
    phase4_end = deadline
    print(f"\nPhase 4: Final polish")

    if time.monotonic() < phase4_end - 15 and best_seq is not None:
        # Tiny beta L-BFGS
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = max(float(c_final.max()), 1e-10)
        per_dl = min(phase4_end - 30, time.monotonic() + (phase4_end - time.monotonic()) * 0.4)
        print(f"  Tiny beta L-BFGS:")
        a_out, sc_out = _single_beta_lbfgs(best_seq, 5e-6 * max_cf, per_dl, maxiter=6000)
        try_update(a_out, sc_out, "tiny_beta")

        # Coordinate descent
        if time.monotonic() < phase4_end - 15:
            per_dl2 = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.5)
            print(f"  Coordinate descent:")
            a_out2, sc_out2 = _coord_descent(best_seq, per_dl2, max_rounds=25)
            try_update(a_out2, sc_out2, "coord_descent")

        # Ultra tiny beta
        if time.monotonic() < phase4_end - 10:
            per_dl3 = min(phase4_end - 5, time.monotonic() + (phase4_end - time.monotonic()) * 0.3)
            print(f"  Ultra tiny beta L-BFGS:")
            a_out3, sc_out3 = _single_beta_lbfgs(best_seq, 2e-6 * max_cf, per_dl3, maxiter=4000)
            try_update(a_out3, sc_out3, "ultra_tiny")

    print(f"\n  Phase 4 done: best={best_score:.10f} (n={len(best_seq)})")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
