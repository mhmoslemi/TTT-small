"""
Optimization of 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Combines L-BFGS continuation, aggressive LP peak balancing with line search,
Nesterov-accelerated gradient with BB step, and coordinate descent polish.
Focuses on exploiting the near-flatness of the arcsine convolution.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _greedy_init_tilted(n, tilt=1.0):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    t = np.linspace(0, 1, n)
    a = a * (1.0 + tilt * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = _greedy_init(n)
    else:
        a = a / s
    return a


def _greedy_init_boundary(n, boost=1.0, width_frac=0.1):
    a = _greedy_init(n)
    w = max(2, int(n * width_frac))
    a[:w] *= (1.0 + boost)
    a[-w:] *= (1.0 + boost)
    a = a / a.sum()
    return a


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)


def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 20000}

            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _build_lp_matrix_for_peaks(a, peak_indices, n):
    K = len(peak_indices)
    A_mat = np.zeros((K, n))
    for row, k in enumerate(peak_indices):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    return A_mat


def _lp_phase_advanced(a_init, deadline, K=50, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.98 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 300:
            active = active[:300]

        A_mat = _build_lp_matrix_for_peaks(best_a, active, n)
        b_vec = np.full(len(active), M * 1.0005)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 60})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        improved = False
        current_score = 2.0 * n * M / (S * S)
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = 2.0 * n * M_trial / (s_trial * s_trial)
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_score


def _nesterov_gradient_polish(a_init, deadline, lr=0.05, momentum=0.9):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.02 * cm0

    v = np.zeros(n)
    no_improve = 0
    check_interval = 30

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-10 * cm, cm0 * 0.02 * (0.96 ** (it // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        v = momentum * v + grad
        z = z - lr * v
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if it % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                lr *= 0.5
                if lr < 1e-6:
                    z = z + 0.05 * np.random.randn(n)
                    z = z - z.mean()
                    lr = 0.05
                    no_improve = 0

    return best_a, best_score


def _bb_gradient_polish(a_init, deadline):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    z_prev = z.copy()
    g_prev = None
    f_prev = None
    a_prev = None

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-10 * cm, cm0 * 0.01 * (0.95 ** (it // 20)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        f_cur = _softmin_obj(z, beta, n)

        if g_prev is not None:
            y = grad - g_prev
            s = z - z_prev
            sy = float(np.dot(s, y))
            yy = float(np.dot(y, y))
            if yy > 1e-30 and abs(sy) > 1e-30:
                lr = sy / yy
                lr = np.clip(lr, 1e-6, 1.0)
            else:
                lr = 0.01
        else:
            ng = float(np.dot(grad, grad))
            lr = 0.01 if ng < 1e-30 else 0.01

        z = z - lr * grad
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        z_prev = z.copy()
        g_prev = grad.copy()
        f_prev = f_cur

        if it % 20 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a_cur.copy()

    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_sweeps=50):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        cm = float(c.max())
        S = float(best_a.sum())
        current_score = 2.0 * n * cm / (S * S)
        improved = False

        step = 2 if sweep < 5 else 1
        for j in range(0, n, step):
            if time.monotonic() > deadline:
                break

            a_j = best_a[j]
            if a_j < 1e-15:
                continue

            lo = 0.0
            hi = 5.0 * a_j
            for _ in range(20):
                mid = 0.5 * (lo + hi)
                best_a[j] = mid
                c_new = _full_convolve(best_a, best_a)
                cm_new = float(c_new.max())
                S_new = float(best_a.sum())
                if S_new < 1e-15:
                    best_a[j] = a_j
                    break
                sc_new = 2.0 * n * cm_new / (S_new * S_new)
                if sc_new < current_score - 1e-14:
                    lo = mid
                    current_score = sc_new
                    improved = True
                else:
                    hi = mid
            best_a[j] = lo if improved and lo > 0 else a_j
            if improved and lo > 0:
                best_a[j] = lo

        if not improved:
            break

        c = _full_convolve(best_a, best_a)
        S = float(best_a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = best_a.copy()

    return best_a, best_score


def _pair_transfer_polish(a_init, deadline, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        S = float(best_a.sum())
        current_score = 2.0 * n * c.max() / (S * S)
        improved = False

        n_pairs = min(5000, n)
        for _ in range(n_pairs):
            if time.monotonic() > deadline:
                break
            i = int(rng.integers(0, n // 2))
            j = int(rng.integers(n // 2, n))
            if best_a[i] < 1e-10 or best_a[j] < 1e-10:
                continue

            for frac in [0.5, 0.3, 0.1, 0.05]:
                delta = frac * best_a[i]
                a_trial = best_a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                c_trial = _full_convolve(a_trial, a_trial)
                S_trial = float(a_trial.sum())
                if S_trial < 1e-15:
                    break
                sc_trial = 2.0 * n * c_trial.max() / (S_trial * S_trial)
                if sc_trial < current_score - 1e-14:
                    best_a = a_trial
                    current_score = sc_trial
                    improved = True
                    break

        if not improved:
            break
        c = _full_convolve(best_a, best_a)
        S = float(best_a.sum())
        sc = 2.0 * n * c.max() / (S * S)
        if sc < best_score - 1e-14:
            best_score = sc

    return best_a, best_score


def _active_set_phase(a_init, deadline, K=4):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for outer in range(100):
        if time.monotonic() > deadline:
            break

        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]

        def ao(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            beta = 1e-7 * pm
            sh = (peaks - pm) / beta
            lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)

        def ag(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[peak_indices]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            beta = 1e-7 * pm
            sh = (peaks - pm) / beta
            e = np.exp(sh - sh.max())
            w = e / e.sum()

            grad_a = np.zeros(n)
            for idx_k in range(len(peak_indices)):
                k = peak_indices[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z

        try:
            result = minimize(ao, z, method='L-BFGS-B', jac=ag,
                              options={'maxiter': 3000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 50, 'maxfun': 10000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization targeting score < 1.5027.
    Phases: L-BFGS continuation, aggressive LP, Nesterov gradient,
    BB-step gradient, coordinate descent, and pair transfer polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}, actual={_score_fn(parent):.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: L-BFGS continuation (500s) ===
    print("Phase 1: L-BFGS continuation")
    p1_end = min(deadline, t_start + 500.0)

    a_s = best_seq.copy()
    a_b1, sc1 = _run_lbfgs(a_s, p1_end)
    update_best(a_b1, sc1, "lbfgs_main")

    # Try tilted and boundary-boosted inits
    if time.monotonic() < p1_end - 60:
        for init_fn, label in [
            (lambda: _greedy_init_tilted(n, tilt=0.3), "tilted"),
            (lambda: _greedy_init_boundary(n, boost=0.5, width_frac=0.05), "boundary"),
        ]:
            if time.monotonic() > p1_end - 30:
                break
            a_alt = init_fn()
            sc_alt = _score_norm_fn(a_alt)
            if sc_alt < best_score * 1.02:
                a_alt_b, sc_alt_b = _run_lbfgs(a_alt, min(p1_end - 10, time.monotonic() + 80))
                update_best(a_alt_b, sc_alt_b, f"lbfgs_{label}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Aggressive LP (400s) ===
    print("Phase 2: LP peak balancing (advanced)")
    p2_end = min(deadline, t_start + 900.0)

    if time.monotonic() < p2_end and best_seq is not None:
        a_lp, sc_lp = _lp_phase_advanced(best_seq, p2_end, K=50, max_rounds=20)
        update_best(a_lp, sc_lp, "lp_adv1")

        # Second round with different K
        if time.monotonic() < p2_end - 60:
            a_lp2, sc_lp2 = _lp_phase_advanced(best_seq, p2_end, K=100, max_rounds=10)
            update_best(a_lp2, sc_lp2, "lp_adv2")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Active set refinement (200s) ===
    print("Phase 3: Active set refinement")
    p3_end = min(deadline, t_start + 1100.0)

    if time.monotonic() < p3_end and best_seq is not None:
        for K in [3, 5, 8]:
            if time.monotonic() > p3_end - 20:
                break
            a_act, sc_act = _active_set_phase(best_seq, p3_end, K=K)
            update_best(a_act, sc_act, f"active_K{K}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Nesterov gradient polish (250s) ===
    print("Phase 4: Nesterov gradient polish")
    p4_end = min(deadline, t_start + 1350.0)

    if time.monotonic() < p4_end and best_seq is not None:
        a_nest, sc_nest = _nesterov_gradient_polish(best_seq, p4_end, lr=0.04, momentum=0.9)
        update_best(a_nest, sc_nest, "nesterov_004")

        if time.monotonic() < p4_end - 30:
            a_nest2, sc_nest2 = _nesterov_gradient_polish(best_seq, p4_end, lr=0.01, momentum=0.95)
            update_best(a_nest2, sc_nest2, "nesterov_001")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: BB-step gradient (150s) ===
    print("Phase 5: BB-step gradient")
    p5_end = min(deadline, t_start + 1500.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_bb, sc_bb = _bb_gradient_polish(best_seq, p5_end)
        update_best(a_bb, sc_bb, "bb_step")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Coordinate descent (150s) ===
    print("Phase 6: Coordinate descent")
    p6_end = min(deadline, t_start + 1650.0)

    if time.monotonic() < p6_end and best_seq is not None:
        a_cd, sc_cd = _coordinate_descent(best_seq, p6_end, max_sweeps=10)
        update_best(a_cd, sc_cd, "coord_descent")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Pair transfer (100s) ===
    print("Phase 7: Pair transfer polish")
    p7_end = min(deadline, t_start + 1750.0)

    if time.monotonic() < p7_end and best_seq is not None:
        a_pt, sc_pt = _pair_transfer_polish(best_seq, p7_end, max_rounds=15)
        update_best(a_pt, sc_pt, "pair_transfer")

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Final: tiny L-BFGS polish ===
    if time.monotonic() < deadline - 10 and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.0001, 0.00005, 0.00002, 0.00001]]
        a_f, sc_f = _run_lbfgs(best_seq, deadline - 5, beta_list=tiny_betas, maxiter_per_beta=3000)
        update_best(a_f, sc_f, "final_tiny")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
