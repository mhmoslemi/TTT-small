```python
"""
LP-based Frank-Wolfe + FISTA + Adam multi-phase optimizer.
Key innovation: Uses a convex LP to find descent directions for the
non-convex auto-correlation peak minimization problem, combined with
gradient-based polishing. Explores multiple sequence lengths.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad(a, w):
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lp_direction(f, M, n, top_k=300):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M for top-k peaks of f*f, g >= 0.
    Returns normalized direction g or None on failure.
    """
    conv_ff = _full_convolve(f, f)
    # Select top-k indices by convolution value
    k_idx = np.argsort(conv_ff)[::-1][:top_k]

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(k_idx):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(idx)
            cols.append(j)
            vals.append(f[k - j])

    A = csr_matrix((vals, (rows, cols)), shape=(len(k_idx), n))
    b = np.full(len(k_idx), M)
    c = -np.ones(n)
    bounds = [(0.0, 1.0)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            g = res.x
            s = g.sum()
            if s > 1e-15:
                g = g / s
            return g
    except Exception:
        pass
    return None


def _line_search_mix(f, g, n, n_trials=21):
    """Find best mixing parameter t in [0,1] to minimize peak of mixed sequence."""
    weights = np.linspace(0.05, 0.95, n_trials)
    best_t = 0.1
    best_peak = np.inf
    ff = _full_convolve(f, f)
    fg = _full_convolve(f, g)
    gg = _full_convolve(g, g)
    for t in weights:
        conv_mix = (1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg
        peak = conv_mix.max()
        if peak < best_peak:
            best_peak = peak
            best_t = t
    # Refine around best_t
    lo = max(0.01, best_t - 0.1)
    hi = min(0.99, best_t + 0.1)
    for t in np.linspace(lo, hi, 15):
        conv_mix = (1 - t) ** 2 * ff + 2 * t * (1 - t) * fg + t ** 2 * gg
        peak = conv_mix.max()
        if peak < best_peak:
            best_peak = peak
            best_t = t
    return best_t


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad(y, w)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        while time.monotonic() < deadline:
            t += 1
            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.5:
                update = update * (0.5 / update_norm)
            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if t % 2000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "barker_like":
        # Alternating pattern with smooth envelope
        t = np.linspace(0, 1, n)
        envelope = np.sin(np.pi * t)
        pattern = 1.0 + 0.5 * np.cos(4.0 * np.pi * t)
        a = envelope * pattern
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0
    else:
        n_main = 4096
        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()
        best_score = sc0

    c0 = _full_convolve(a0, a0)
    max_c0 = float(c0.max())
    print(f"Initial max_conv={max_c0:.10f}")
    a_work = a0.copy() / a0.sum()

    # === Phase 1: LP-based Frank-Wolfe ===
    print("\n=== Phase 1: LP-based Frank-Wolfe ===")
    phase1_end = min(deadline, t_start + 180.0)
    a_fw = a_work.copy()
    n_fw = len(a_fw)

    for fw_iter in range(8):
        if time.monotonic() > phase1_end - 30:
            break
        conv_fw = _full_convolve(a_fw, a_fw)
        M_fw = float(conv_fw.max())
        print(f"  FW iter {fw_iter}: M={M_fw:.10f}")

        g_dir = _lp_direction(a_fw, M_fw, n_fw, top_k=400)
        if g_dir is None:
            print("  LP failed, trying with more constraints")
            g_dir = _lp_direction(a_fw, M_fw, n_fw, top_k=800)
        if g_dir is None:
            print("  LP failed, skipping to Phase 2")
            break

        # Line search for mixing parameter
        t_best = _line_search_mix(a_fw, g_dir, n_fw)
        a_new = (1 - t_best) * a_fw + t_best * g_dir
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc_new = _score(a_new)
        print(f"  FW iter {fw_iter}: t={t_best:.4f}, score={sc_new:.10f}")

        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new.copy()

        # Always accept if it improves over current
        sc_curr = _score(a_fw)
        if sc_new < sc_curr:
            a_fw = a_new.copy()
        else:
            # Try smaller step
            t_small = t_best * 0.5
            a_sm = (1 - t_small) * a_fw + t_small * g_dir
            a_sm = np.maximum(a_sm, 0.0) / a_sm.sum()
            sc_sm = _score(a_sm)
            if sc_sm < sc_curr:
                a_fw = a_sm.copy()
                sc_new2 = _score(a_fw)
                if sc_new2 < best_score:
                    best_score = sc_new2
                    best_seq = a_fw.copy()
            else:
                break

    a_work = a_fw.copy() / a_fw.sum()
    print(f"End Phase 1: best_score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 2: FISTA with coarse-to-fine beta ===
    print("\n=== Phase 2: FISTA smooth optimization ===")
    phase2_end = min(deadline, t_start + 500.0)
    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())

    betas_list = []
    beta = 0.03 * max_cc
    while beta > 0.00002 * max_c0:
        betas_list.append(beta)
        beta *= 0.45
    # Limit to reasonable number of phases
    if len(betas_list) > 12:
        betas_list = betas_list[:12]

    for i, beta in enumerate(betas_list):
        if time.monotonic() > phase2_end - 5:
            break
        per_time = phase2_end - time.monotonic() - 3
        phase_timeout = time.monotonic() + min(45.0, per_time)
        max_it = max(3000, int((phase_timeout - time.monotonic()) * 3000))
        a_work, sc = _fista_smoothed(a_work, beta, phase_timeout, max_it)
        a_work = a_work / a_work.sum()
        sc_full = _score(a_work)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_work.copy()
            print(f"  FISTA beta={beta:.6f}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
        c_curr = _full_convolve(a_work, a_work)
        max_cc = float(c_curr.max())

    print(f"End Phase 2: score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 3: Adam log-space fine optimization ===
    print("\n=== Phase 3: Adam log-space ===")
    phase3_end = min(deadline, t_start + 900.0)
    c_curr = _full_convolve(a_work, a_work)
    max_cc = float(c_curr.max())
    betas_fine = []
    beta = 0.002 * max_cc
    while beta > 0.000002 * max_c0:
        betas_fine.append(beta)
        beta *= 0.4
    if len(betas_fine) > 12:
        betas_fine = betas_fine[:12]

    per_time = phase3_end - time.monotonic() - 10
    a_opt, sc_opt = _adam_logspace(
        a_work, time.monotonic() + per_time,
        betas_fine, lr_init=0.015, max_iter_total=500000
    )
    sc_opt_full = _score(a_opt)
    if sc_opt_full < best_score:
        best_score = sc_opt_full
        best_seq = a_opt.copy()
        print(f"  Adam: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

    a_work = a_opt.copy() / a_opt.sum()
    print(f"End Phase 3: score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s")

    # === Phase 4: Multi-restart with diverse strategies ===
    print("\n=== Phase 4: Multi-restart ===")
    phase4_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < phase4_end - 60:
        n_restarts = 5
        per_restart = (phase4_end - time.monotonic() - 30) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > phase4_end - 30:
                break
            # Different perturbation strategies
            if i % 4 == 0:
                p_start = _perturb(a_work, 0.02 + 0.01 * i, rng)
            elif i % 4 == 1:
                # Cyclic shift
                sz = len(a_work)
                shift = rng.integers(1, max(2, sz // 1024))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            elif i % 4 == 2:
                # Frequency-domain perturbation
                F = np.fft.rfft(a_work)
                phase_pert = 0.05 * rng.standard_normal(len(F))
                F = F * np.exp(1j * phase_pert)
                p_start = np.fft.irfft(F, len(a_work))
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()
            else:
                # Scale perturbation (multiply by random smooth function)
                t_grid = np.linspace(0, 1, len(a_work))
                smooth_pert = 1.0 + 0.1 * np.sin(2 * np.pi * rng.uniform(1, 5) * t_grid + rng.uniform(0, 2 * np.pi))
                p_start = a_work * smooth_pert
                p_start = np.maximum(p_start, 1e-15)
                p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs, 0.0001 * max_cs]

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + max(5.0, per_restart * 0.85),
                bs, lr_init=0.012, max_iter_total=150000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                a_work = p_best.copy() / p_best.sum()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: score={sc_full:.10f} (not better) t={time.monotonic()-t_start:.1f}s")

    # === Phase 5: Try larger n ===
    print("\n=== Phase 5: Larger n exploration ===")
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end - 60 and best_seq is not None:
        for n_try in [len(best_seq) * 2]:
            if time.monotonic() > phase5_end - 60:
                break
            if n_try > 16384:
                continue
            a_try = _interp_profile(best_seq, n_try)
            sc_try = _score(a_try)
            print(f"  Interpolated n={n_try}: score={sc_try:.10f}")
            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()

            if time.monotonic() < phase5_end - 30:
                c_try = _full_convolve(a_try, a_try)
                max_ct = float(c_try.max())

                # Quick FISTA
                a_try, sc_t = _fista_smoothed(
                    a_try, 0.01 * max_ct,
                    time.monotonic() + min(30.0, phase5_end - time.monotonic() - 20),
                    20000
                )
                a_try = a_try / a_try.sum()

                # Adam polish
                if time.monotonic() < phase5_end - 15:
                    betas_t = [0.001 * max_ct, 0.0005 * max_ct, 0.0002 * max_ct]
                    a_try, sc_t2 = _adam_logspace(
                        a_try, time.monotonic() + max(5.0, phase5_end - time.monotonic() - 10),
                        betas_t, lr_init=0.01, max_iter_total=100000
                    )
                    a_try = a_try / a_try.sum()

                sc_try_opt = _score(a_try)
                print(f"  Optimized n={n_try}: score={sc_try_opt:.10f}")
                if sc_try_opt < best_score:
                    best_score = sc_try_opt
                    best_seq = a_try.copy()

    # === Phase 6: Final ultra-fine polish ===
    print("\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()
        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00003 * max_cp, 0.00001 * max_cp, 0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 15:
            a_pol, sc_pol = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.003, max_iter_total=200000
            )
            a_pol = np.clip(a_pol, 0, 1000)
            a_pol = a_pol / a_pol.sum()
            sc_pol_full = _score(a_pol)
            if sc_pol_full < best_score:
                best_score = sc_pol_full
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol_full:.10f}")

    # === Final validation ===
    print(f"\n=== Final: score={best_score:.10f}, time={time.monotonic()-t_start:.1f}s ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
