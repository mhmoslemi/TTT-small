def _next_pow2(n):

    n, m = len(a), len(b)
    out_len = n + m - 1
    N = _next_pow2(out_len)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_gradient(a, beta, n):
    """
    Compute gradient of soft-max objective with respect to a.
    beta is the temperature (smaller = sharper).
    Returns grad of length n.
    """
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0, 50.0)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    N = _next_pow2(2 * n + 2 * n - 1 + 2)  # enough for full conv of n and 2n-1

    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w[::-1], N)

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > max_step:
            update = update * (max_step / update_norm)

        z = z - update
        z = z - z.mean()
        z = np.clip(z, -60.0, 60.0)

        # Schedule
        if t_step % 1000 == 0:
            alpha = max(alpha_min, alpha * alpha_decay)
            current_lr = max(lr_floor, current_lr * lr_decay)

    # Final
    a_final = np.exp(z)
    a_final = a_final / a_final.sum()
    sc_final = float(_full_convolve(a_final, a_final).max())
    if sc_final < best_cm:
        best_cm = sc_final
        best_a = a_final.copy()

    return best_a, best_cm


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    a_new = np.exp(z)
    return a_new / a_new.sum()


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()

            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        a0 = parent / parent.sum()
        sc0 = _score(a0)
        print(f"Parent: n={len(a0)}, score={sc0:.10f}")

        best_seq = a0.copy()
        best_score = sc0

    print(f"\n=== Phase 1: Adam main (n={n}, alpha=0.1->0.01) ===")

    if time.monotonic() < phase1_end - 30:

    # === Phase 2: Sharper optimization (700s - 1200s) ===

    if best_seq is not None and time.monotonic() < phase2_end - 30:
        a_start = best_seq / best_seq.sum()
        # Run 1: moderate sharp
        run_dl = min(time.monotonic() + 200, phase2_end - 60)
