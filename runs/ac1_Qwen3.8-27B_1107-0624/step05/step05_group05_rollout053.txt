"""
Hybrid LP + smoothed-gradient optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses gradient descent with log-sum-exp smoothing (adaptive temperature), 
LP steps for large improvements, and careful hyperparameter scheduling.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_weights(z, beta):
    """Compute softmax(z/beta) with numerical stability."""
    z_shifted = (z - z.max()) / beta
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(z)
        w[np.argmax(z)] = 1.0
        return w
    return e / s

def _smoothed_gradient(a, c, beta):
    """
    Compute gradient of (1/beta)*log(sum exp(beta*c_k)) w.r.t. a.
   Returns grad[m] = 2 * sum_l a[l] * w[l+m] where w = softmax(beta*c).
    """
    w = _softmax_weights(c, beta)
    # cross_corr[m] = sum_{l=0}^{n-1} a[l] * w[l+m]
    # This is a cross-correlation of a with w
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # cross-correlation: ifft(fa * conj(fw)) gives sum_l a[l] * w[l+m] (circular)
    # We need the linear cross-correlation
    cc = np.fft.irfft(fa * np.conj(fw), N)[:2*n-1]
    # The linear cross-correlation: cc[m] = sum_l a[l] * w[l+m] for m=0..n-1
    # Actually, ifft(fft(a)*conj(fft(w))) gives the circular cross-correlation
    # For linear: we need to be careful about the indexing
    # Let me use a different approach: convolve a with reversed w
    # cross_corr[m] = sum_l a[l] * w[l+m]
    # = (a * w_rev)[m + len(w) - 1] where w_rev is reversed w
    # Actually, let me just compute it as:
    # pad a to length 2n-1, convolve with w
    # No, the simplest is:
    # cross_corr = np.correlate(w, a, mode='valid')  # gives sum_l a[l]*w[l+m], m=0..n-1
    # But for large n this might be slow. Let me use FFT properly.
    
    # The cross-correlation (a ⊛ w)[m] = sum_l a[l] * w[l+m]
    # = ifft(conj(fft(a)) * fft(w))[m] (circular, with appropriate zero-padding)
    # With zero-padding to length >= n + (2n-1) - 1 = 3n-2, we get linear result
    
    # Actually, the standard formula: 
    # ifft(fft(a, N) * conj(fft(w, N))) gives (a ⊛ w) circularly
    # For linear cross-correlation, we need N >= n + len(w) - 1 = n + (2n-1) - 1 = 3n-2
    
    # Let me just use a simpler approach:
    # cross_corr[m] = sum_{l=0}^{n-1} a[l] * w[l+m]
    # This is the m-th element of the "full" cross-correlation of a with w
    # We can compute it as: 
    #   Take the first n elements of ifft(fft(a, 2N2) * conj(fft(w, 2N2)))
    #   where N2 >= 2n-1 (to avoid circular aliasing)
    
    # Actually the cleanest way:
    # cross_corr[m] = <a, w[m:]> (inner product of a with w starting at position m)
    # = conv(a, w_rev)[m + len(w) - len(a)] ... this is getting confused.
    
    # Let me just do it with a direct FFT approach:
    # Define W = fft(w, N) and A = fft(a, N) with N >= 3n-2 for linearity
    # Then cross_corr = ifft(A * conj(W)) gives the circular cross-correlation
    # With sufficient zero-padding, the circular = linear for the first n values
    
    N2 = _next_pow2(3 * n)
    Fa = np.fft.rfft(a, N2)
    Fw = np.fft.rfft(w, N2)
    cc_full = np.fft.irfft(Fa * np.conj(Fw), N2)
    # cc_full[m] = sum_l a[l] * w[(l+m) mod N2] for circular
    # With N2 >= n + (2n-1) - 1 = 3n-2, for m = 0, ..., n-1:
    #   cc_full[m] = sum_{l=0}^{n-1} a[l] * w[l+m] (since l+m < n + n-1 = 2n-1 < N2)
    grad = 2.0 * cc_full[:n].copy()
    return grad

def _softmax_param(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)

def _solve_lp(f, M, rows, cols, time_limit=120):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None

def _lp_step(f, rows, cols, time_limit=120):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    # Line search
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M

def _phase_adam(a, deadline, beta_start, beta_end, lr_start, lr_end, max_iter=500000):
    """
    Adam optimizer with adaptive log-sum-exp smoothing.
    Optimizes in the z-space where a = softmax(z).
    """
    n = len(a)
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    prev_best = best_score
    patience = 0
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Adaptive beta
        progress = min(1.0, t / max_iter)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        # Current a
        a = _softmax_param(z)
        
        # Compute gradient
        c = _full_convolve(a, a)
        grad = _smoothed_gradient(a, c, beta)
        
        # Project gradient to be orthogonal to all-ones (maintain sum=1 in z-space)
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        lr = lr_start * (lr_end / lr_start) ** progress
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        # Clamp update
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        # Periodic evaluation
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
            
            if patience % 20 == 0 and patience > 0:
                print(f"    Adam t={t}, beta={beta:.6e}, score={sc:.10f}, lr={lr:.6e}")
                
            if patience > 200:
                # Reduce LR
                lr *= 0.5
                patience = 0
    
    # Final
    a_final = _softmax_param(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

def _phase_adam_active(a, deadline, top_k=50):
    """
    Adam optimizer that focuses on the top-k lags in the softmax.
    This gives a sharper approximation to the true max.
    """
    n = len(a)
    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.005
    
    while t < 300000:
        if time.monotonic() > deadline:
            break
        t += 1
        
        a = _softmax_param(z)
        c = _full_convolve(a, a)
        
        # Active set: only top-k lags
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(top_k, len(c)-1)]
        mask = c >= threshold
        c_active = c.copy()
        c_active[~mask] = -np.inf
        
        beta = 0.0001 * c.max()  # Very sharp
        w = _softmax_weights(c_active, beta)
        
        # Gradient
        N2 = _next_pow2(3 * n)
        Fa = np.fft.rfft(a, N2)
        Fw = np.fft.rfft(w, N2)
        cc_full = np.fft.irfft(Fa * np.conj(Fw), N2)
        grad = 2.0 * cc_full[:n].copy()
        
        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)
        
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        
        z = z - update
        z = np.clip(z, -100.0, 100.0)
        z = z - z.mean()
        
        if t % 1000 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"    Active-Adam t={t}, score={sc:.10f}, lr={lr:.6e}")
    
    a_final = _softmax_param(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    
    return best_a, best_score

"""
Hybrid LP + smoothed-gradient optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses FFT-based gradient computation with adaptive log-sum-exp smoothing,
interleaved with LP steps for large improvements, and active-set focusing
for final refinement.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_weights(z, beta):
    z_shifted = (z - z.max()) / beta
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(z)
        w[np.argmax(z)] = 1.0
        return w
    return e / s


def _softmax_param(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _gradient_convol(a, w):
    """Compute grad[m] = 2 * sum_l a[l] * w[l+m] using FFT cross-correlation."""
    n = len(a)
    N2 = _next_pow2(3 * n - 2)
    Fa = np.fft.rfft(a, N2)
    Fw = np.fft.rfft(w, N2)
    cc = np.fft.irfft(Fa * np.conj(Fw), N2)
    return 2.0 * cc[:n].copy()


def _smoothed_grad(a, c, beta):
    w = _softmax_weights(c, beta)
    return _gradient_convol(a, w)


def _active_grad(a, c, top_k):
    n = len(a)
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(top_k, len(c) - 1)]
    mask = c >= threshold
    c_mod = c.copy()
    c_mod[~mask] = -np.inf
    c_max = float(c.max())
    beta = max(1e-6, 0.001 * c_max)
    w = _softmax_weights(c_mod, beta)
    return _gradient_convol(a, w)


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=90):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols, time_limit=90):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols, time_limit)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M


def _adam_run(a_init, deadline, beta_start, beta_end, lr_start, lr_end,
              max_iter=500000, top_k=None):
    """Adam in z-space with adaptive log-sum-exp or active-set gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_start
    patience = 0
    prev_best = best_score

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        progress = min(1.0, t / max_iter)
        beta = beta_start * (beta_end / beta_start) ** progress

        a = _softmax_param(z)
        c = _full_convolve(a, a)

        if top_k is not None:
            grad = _active_grad(a, c, top_k)
        else:
            grad = _smoothed_grad(a, c, beta)

        W = float(np.dot(a, grad))
        grad_z = a * (grad - W)

        if not np.all(np.isfinite(grad_z)):
            break

        lr = current_lr
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = lr * mh / (np.sqrt(vh) + eps)

        update_norm = np.linalg.norm(update)
        if update_norm > 0.8:
            update = update * (0.8 / update_norm)

        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                patience = 0
            else:
                patience += 1

                current_lr *= 0.7
                patience = 0

    a_final = _softmax_param(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

            if time.monotonic() > sub_end:
                break
            a4, sc4 = _adam_run(
                best_seq, sub_end,
                beta_start=0.001, beta_end=0.0001,
                lr_start=0.006, lr_end=0.001,
                max_iter=400000, top_k=tk
            )
            full_sc = _score(a4)
            print(f"  Active-Adam (k={tk}): score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a4.copy()

    print(f"Phase 4: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 5: Final LP + Adam polish (1620s - deadline) ===
    if time.monotonic() < deadline - 15:
        print("=== Phase 5: Final polish ===")
        a = best_seq.copy()
        a, _ = _lp_step(a, rows, cols, time_limit=40)
        full_sc = _score(a)
        print(f"  Final LP: score={full_sc:.10f}")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()

        if time.monotonic() < deadline - 10:
            c_fin = _full_convolve(best_seq, best_seq)
            max_cf = float(c_fin.max())
            a_f, sc_f = _adam_run(
                best_seq, deadline - 5,
                beta_start=0.0005 * max_cf, beta_end=0.00001 * max_cf,
                lr_start=0.004, lr_end=0.0005,
                max_iter=300000, top_k=20
            )
            full_sc = _score(a_f)
            print(f"  Final Adam: score={full_sc:.10f}")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = a_f.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
