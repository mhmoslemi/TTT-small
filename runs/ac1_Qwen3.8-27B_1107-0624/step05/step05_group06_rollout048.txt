"""
Combined LP-based direction finding and L-BFGS optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses the LP from the SAA paper for coarse optimization, then L-BFGS with
log-sum-exp smoothing for fine-tuning, plus subgradient polishing.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.fft import fft, ifft


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = fft(a, N)
    fb = fft(b, N)
    r = ifft(fa * fb, N)[:out_n]
    return np.real(r)


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _normalize_sum(a, target=None):
    s = float(a.sum())
    if s < 1e-15:
        return a.copy()
    if target is None:
        target = np.sqrt(2.0 * len(a))
    return a * (target / s)


def _build_conv_matrix(n, a):
    """Build the (2n-1) x n convolution matrix A where c = A @ b gives conv(a, b)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp_direction(a, M, time_limit=10.0):
    """Solve LP: max sum(b) s.t. b >= 0, conv(a,b)(k) <= M for all k."""
    n = len(a)
    A = _build_conv_matrix(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_iteration(a, time_limit=10.0):
    """One LP-based iteration: find direction, line search, update."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = _normalize_sum(a, target_sum)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-15:
        return a

    b = _solve_lp_direction(a, M, time_limit)
    if b is None:
        return a

    sum_b = float(b.sum())
    if sum_b < 1e-15 or sum_b <= float(a.sum()) * (1 + 1e-10):
        return a

    g = b * (target_sum / sum_b)

    # Line search
    best_a = a.copy()
    best_obj = M
    for t in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
        trial = (1.0 - t) * a + t * g
        trial = np.maximum(trial, 0.0)
        trial = _normalize_sum(trial, target_sum)
        trial_obj = float(_full_convolve(trial, trial).max())
        if trial_obj < best_obj:
            best_obj = trial_obj
            best_a = trial.copy()
    return best_a


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _gradient_active(a, c, n, rel_tol=0.005):
    """Gradient using exact active set."""
    c_max = c.max()
    tol = rel_tol * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_optimize_z(z0, beta, n, max_iter=3000):
    """Run L-BFGS-B on the smoothed objective in z-space."""
    def fun(z):
        return _smoobj(z, beta, n)

    def jac(z):
        return _smograd(z, beta, n)

    try:
        result = minimize(fun, z0, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-15,
                                   'gtol': 1e-12, 'maxcor': 40,
                                   'maxls': 200, 'disp': False})
        return _softmax(result.x), result.success, int(result.nit)
    except Exception:
        return _softmax(z0), False, 0


def _lbfgs_schedule(a_init, deadline, min_time_remaining=60.0):
    """Run L-BFGS beta-continuation schedule."""
    n = len(a_init)
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    betas = max_c * np.array([0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    current_a = a_init.copy()

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time_remaining > deadline:

        remaining = deadline - min_time_remaining - time.monotonic()
        if remaining < 3:

        est_per_iter = 0.004 * (n / 2048.0)

        if i % 3 == 0:

def _peak_shave(a, n_shaves=20, deadline=None):
    """Peak shaving heuristic: iteratively reduce mass at positions contributing to peaks."""
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    a = _normalize_sum(a, target_sum).copy()

    for it in range(n_shaves):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = c.max()
        peak_idx = int(np.argmax(c))

        # Find positions contributing most to the peak
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[peak_idx - j]

        # Find the top contributors
        top_k = max(1, n // 50)
        top_indices = np.argsort(contrib)[-top_k:]

        # Find "safe" positions (low contribution to peak)
        safe_mask = np.ones(n, dtype=bool)
        safe_mask[top_indices] = False
        safe_positions = np.where(safe_mask)[0]
        if len(safe_positions) == 0:
            break

        # Move a small amount of mass from top contributors to safe positions
        total_move = 0.0
        move_per = M * 0.001 / len(top_indices)
        for j in top_indices:
            move = min(move_per, a[j] * 0.1)
            if move > 1e-15:
                a[j] -= move
                total_move += move

        if total_move < 1e-15:
            break

        # Distribute to safe positions proportional to current a
        safe_vals = a[safe_positions]
        safe_sum = safe_vals.sum()
        if safe_sum < 1e-15:
            a[safe_positions] += total_move / len(safe_positions)
        else:
            a[safe_positions] += total_move * safe_vals / safe_sum

        a = np.maximum(a, 0.0)
        a = _normalize_sum(a, target_sum)

    return a

            stall_count = 0
            if it % 10 == 0:
                print(f"  LP iter {it}: score={sc_lp:.10f} (dt={t_iter:.1f}s)")
        else:
            stall_count += 1
            if it % 10 == 0:
                print(f"  LP iter {it}: no improvement (stall={stall_count})")
            if stall_count > 15:
                print(f"  LP stalled after {it} iterations, stopping.")
                break

        # Reduce time limit as we converge
        if it > 50:
            lp_time_limit = max(3.0, lp_time_limit * 0.95)

    _update_best(a_lp, sc_lp, "lp_optimized")
    print(f"  LP final: score={_score(a_lp):.10f}")

    # --- Phase 2: Peak shaving ---
    print(f"\n{'='*60}")
    print(f"Phase 2: Peak shaving (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 500:
        a_shave = _peak_shave(best_seq, n_shaves=50, deadline=deadline - 500)
        _update_best(a_shave, None, "peak_shave")

    # --- Phase 3: L-BFGS fine-tuning ---
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS fine-tuning (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 200:
        a_lbg = _lbfgs_schedule(best_seq, deadline, min_time_remaining=200.0)
        _update_best(a_lbg, None, "lbfgs_polish")

    # --- Phase 4: Subgradient polishing ---
    print(f"\n{'='*60}")
    print(f"Phase 4: Subgradient polishing (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 100:
        sg_deadline = min(deadline - 50.0, time.monotonic() + 120.0)
        a_sg, sc_sg = _subgrad_polish(best_seq, sg_deadline, max_steps=5000, lr_init=0.015)
        _update_best(a_sg, sc_sg, "subgrad")

    # --- Phase 5: Multi-restart diversification ---
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-restart (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 120:
        remaining = deadline - 120 - time.monotonic()
        n_restarts = min(5, max(1, int(remaining / 120)))
        print(f"  Attempting {n_restarts} restarts ({remaining:.0f}s remaining)")

        for i in range(n_restarts):
            if time.monotonic() > deadline - 60:
                break

            if i == 0:
                p_start = _perturb_seq(best_seq, 0.02, rng)
            elif i == 1:
                p_start = _perturb_fourier(best_seq, rng, n_modes=2)
            elif i == 2:
                p_start = _perturb_seq(best_seq, 0.05, rng)
            elif i == 3:
                p_start = _perturb_fourier(best_seq, rng, n_modes=4)
            else:
                p_start = _perturb_seq(best_seq, 0.01, rng)

            sc_start = _score(p_start)
            print(f"  Restart {i}: start_score={sc_start:.10f}")

            # Quick L-BFGS on perturbed point
            if time.monotonic() < deadline - 30:
                a_r = _lbfgs_schedule(p_start, deadline, min_time_remaining=30.0)
                _update_best(a_r, None, f"restart_{i}")

        # Also try LP from a perturbed point
        if best_seq is not None and time.monotonic() < deadline - 30:
            a_pert = _perturb_seq(best_seq, 0.03, rng)
            sc_pert = _score(a_pert)
            if sc_pert < best_score + 0.01:
                print(f"  LP from perturbed: start_score={sc_pert:.10f}")
                a_lp2 = _normalize_sum(a_pert, target_sum).copy()
                for it in range(30):
                    if time.monotonic() > deadline - 25:
                        break
                    a_new = _lp_iteration(a_lp2, time_limit=5.0)
                    sc_new = _score(a_new)
                    if sc_new < _score(a_lp2) - 1e-12:
                        a_lp2 = a_new
                    else:
                        break
                _update_best(a_lp2, None, "lp_perturbed")

    # --- Phase 6: Final combined polish ---
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 5:
        # L-BFGS with very small beta
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())
        beta_final = mb * 0.0001
        if time.monotonic() < deadline - 3:
            z0 = np.log(np.maximum(best_seq, 1e-15))
            z0 = z0 - z0.mean()
            a_fin, _, _ = _lbfgs_optimize_z(z0, beta_final, n, max_iter=2000)
            _update_best(a_fin, None, "final_lbfgs")

        # Final subgradient
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _subgrad_polish(best_seq, deadline - 2, max_steps=1000, lr_init=0.008)
            _update_best(a_fin2, sc_fin2, "final_sg")

    # --- Phase 7: Try different n if time permits ---
    print(f"\n{'='*60}")
    print(f"Phase 7: Alternative n (t={time.monotonic() - t_start:.1f}s)")
    print(f"{'='*60}")

    if time.monotonic() < deadline - 60:
        # Try n = 4096 with a simple profile + quick LBFGS
        for n_try in [4096, 1024]:
            if time.monotonic() > deadline - 30:
                break
            a_try = _make_profile(n_try, "tapered")
            sc_try = _score(a_try)
            print(f"  Trying n={n_try}: initial score={sc_try:.10f}")
            _update_best(a_try, sc_try, f"init_n{n_try}")
            if sc_try < best_score + 0.1 and time.monotonic() < deadline - 20:
                a_try2 = _lbfgs_schedule(a_try, deadline, min_time_remaining=20.0)
                _update_best(a_try2, None, f"lbfgs_n{n_try}")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic() - t_start:.1f}s")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")

        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
