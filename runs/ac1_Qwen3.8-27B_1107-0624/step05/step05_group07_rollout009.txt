# Helper functions (all top-level)

def next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def score_norm_fn(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
