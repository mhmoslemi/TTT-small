def _make_structured_init(n, rng):
    """Generate various structured initializations and return the best."""
    t = np.linspace(0, 1, n)
    candidates = []
    
    # N-shape: low, rise, dip, rise to peak
    for p1, p2, p3 in [(0.2, 0.5, 0.3), (0.3, 0.6, 0.1), (0.1, 0.4, 0.5),
                        (0.15, 0.7, 0.15), (0.25, 0.4, 0.25)]:
        a = p1 + p2 * np.sin(np.pi * t) + p3 * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Bimodal
    for mu1, mu2, sig in [(0.25, 0.75, 0.15), (0.2, 0.8, 0.1), (0.3, 0.7, 0.2),
                           (0.15, 0.85, 0.12), (0.35, 0.65, 0.15)]:
        a = np.exp(-((t - mu1)**2) / (2*sig**2)) + np.exp(-((t - mu2)**2) / (2*sig**2))
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Sinusoidal variants
    for freq, phase in [(1, 0), (2, 0), (1, 0.5), (3, 0), (1, 1.0)]:
        a = 1.0 + 0.5 * np.sin(2*np.pi*freq*t + phase)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # U-shape
    for width in [0.3, 0.5, 0.7]:
        a = 1.0 / (width + (1-width) * np.sin(np.pi * t)**2)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Ramp with modulation
    for slope, mod in [(2.0, 0.3), (3.0, 0.2), (1.5, 0.5), (2.5, 0.4)]:
        a = 1.0 + slope * t + mod * np.sin(2 * np.pi * t)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Exponential decay from both ends
    for rate in [1.0, 2.0, 3.0]:
        a = np.exp(-rate * (t - 0.5)**2 / 0.25)
        a = np.maximum(a, 1e-10)
        candidates.append(a / a.sum())
    
    # Pick best
    best_a = None
    best_sc = np.inf
    for a in candidates:
        sc = _score_norm(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc


def _adam_smoothed_max(a_init, deadline, rng, n_max_iter=500000, lr_init=0.1):
    """Adam optimization of smoothed max(conv(a,a)) in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_init
    
    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    
    # Adaptive beta schedule
    beta = 0.3 * max_c
    beta_min = 1e-7 * max_c
    
    for it in range(n_max_iter):
        if time.monotonic() > deadline:
            break
        
        t_count += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        if cm < 1e-30:
            break
        
        # Adapt beta: make it track the "effective width" of the distribution over c
        if t_count % 50 == 0:
            sorted_c = np.sort(c)[::-1]
            k = min(50, len(sorted_c) - 1)
            gap = sorted_c[0] - sorted_c[k]
            if gap < 1e-10:
                beta = max(beta * 0.5, beta_min)
            else:
                beta = max(0.3 * gap, beta_min)
        
        # Compute gradient
        _, grad_a = _conv_and_grad(a, beta)
        
        # Convert to z-gradient: grad_z = a * (grad_a - dot(a, grad_a))
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        # Adam update
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t_count)
        vh = v / (1 - b2 ** t_count)
        
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        # Score check (every 10 iters for speed)
        if t_count % 10 == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        
        # Learning rate decay
        if t_count % 5000 == 0:
            lr = max(0.005, lr * 0.85)
        
        # Perturbation to escape local minima
        if t_count % 20000 == 0 and t_count > 0:
            noise_scale = 0.05 * np.exp(-t_count / 50000.0)
            z = z + noise_scale * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score


def _peak_equalize_refine(a_init, deadline, rng, max_rounds=500):
    """Targeted refinement: reduce the top peaks of the convolution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Find top peaks
        n_peaks = min(5, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        improved = False
        
        # For the top peak, find the dominant contributing index and reduce it
        for pk in peak_idx[:3]:
            # c_pk = sum_i a[i]*a[pk-i], valid i in [max(0, pk-n+1), min(n-1, pk)]
            i_lo = max(0, pk - n + 1)
            i_hi = min(n - 1, pk)
            if i_lo > i_hi:
                continue
            valid_i = np.arange(i_lo, i_hi + 1)
            # Avoid double counting when i == pk-i
            contrib = a[valid_i] * a[pk - valid_i]
            
            # Find the index with max contribution
            max_c_idx = np.argmax(contrib)
            idx = valid_i[max_c_idx]
            
            # Also consider the "partner" index pk - idx if different
            partner = pk - idx
            if 0 <= partner < n and partner != idx:
                partner_c = a[idx] * a[partner]
                if partner_c > contrib[max_c_idx]:
                    idx = partner  # Target the larger of the pair
            
            # Try reducing this index
            for delta in [0.02, 0.05, 0.1, 0.15, 0.2]:
                z_trial = z.copy()
                z_trial[idx] -= delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score * (1 - 1e-8):
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
                    break
            
            if improved:
                break
        
        # Also try increasing small values (redistributing mass)
        if not improved:
            small_idx = np.argsort(a)[:10]
            for idx in small_idx[:5]:
                for delta in [0.02, 0.05, 0.1]:
                    z_trial = z.copy()
                    z_trial[idx] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm(a_trial)
                    if sc < best_score * (1 - 1e-8):
                        best_score = sc
                        best_a = a_trial.copy()
                        z = z_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and rnd > 50:
            # Try a random perturbation
            if rnd % 10 == 0:
                n_pert = rng.integers(1, max(2, n // 100))
                idx_pert = rng.integers(0, n, size=n_pert)
                deltas = rng.uniform(-0.1, 0.1, size=n_pert)
                z_trial = z.copy()
                z_trial[idx_pert] += deltas
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score * (1 - 1e-8):
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
            
            if not improved and rnd > 100:
                break
    
    return best_a, best_score


def _coordinate_descent(a_init, deadline, rng, max_sweeps=200):
    """Cycle through coordinates and try moving each one."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    z = np.log(np.maximum(best_a, 1e-15))
    z = z - z.mean()
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        
        # Process indices in order of their current contribution to peaks
        # (those that contribute most to the max are optimized first)
        order = np.argsort(-a)  # process large values first
        
        improved_this_sweep = False
        for idx in order:
            if time.monotonic() > deadline:
                break
            
            # Try small moves
            for delta in [0.01, 0.05, -0.01, -0.05, 0.1, -0.1]:
                z_trial = z.copy()
                z_trial[idx] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm(a_trial)
                if sc < best_score * (1 - 1e-10):
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved_this_sweep = True
                    break
        
        if not improved_this_sweep and sweep > 20:
            break
    
    return best_a, best_score


def _lp_direction(a_fixed, M, n):
    """Solve LP: max sum(g) s.t. conv(a_fixed, g)_k <= M, g >= 0.
    Returns the optimal g (unnormalized) or None if infeasible/too slow."""
    # conv(a, g)_k = sum_i a[i] * g[k-i] <= M
    # This is a linear constraint: for each k, sum_{i: valid} a[i] * g[k-i] <= M
    # Let's build the constraint matrix A_eq (actually A_ub)
    # A_ub[g]_k = sum_{i: max(0,k-n+1) <= i <= min(n-1,k)} a[i] * g[k-i]
    # Note: g[k-i] = g[j] where j = k-i
    # So A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # This is a Toeplitz-like matrix. For n up to ~1024, it's feasible.
    if n > 2048:
        return None
    
    # Build A_ub (2n-1 x n matrix)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a_fixed[idx]
    
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 30})
        if res.status == 0:
            g = res.x
            if g is not None and np.all(np.isfinite(g)) and g.sum() > 0:
                return g
    except Exception:
        pass
    return None


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")
    
    # --- Get initial sequences ---
    print("Phase 0: Initialization")
    
    # Try parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        sc_p = _score_norm(parent)
        print(f"  Parent: n={len(parent)}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        a0, sc0 = _make_structured_init(512, rng)
        update_best(a0, sc0, "structured_init")
    
    # Also generate a structured init and compare
    a_struct, sc_struct = _make_structured_init(512, rng)
    if sc_struct < best_score:
        update_best(a_struct, sc_struct, "structured_512")
    
    # Quick optimization of the best 512-length candidate
    a_work = (best_seq / best_seq.sum()).copy()
    n_work = len(a_work)
    
    quick_deadline = t_start + 30.0
    a_work, sc_work = _adam_smoothed_max(a_work, quick_deadline, rng,
                                          n_max_iter=50000, lr_init=0.1)
    update_best(a_work, sc_work, "quick_adam_512")
    
    # --- Multi-scale continuation ---
    print("\nPhase 1: Multi-scale optimization")
    scales = [1024, 2048, 4096]
    time_alloc = {1024: 60, 2048: 120, 4096: 400}
    
    a_cur = a_work.copy()
    sc_cur = sc_work
    
    for target_n in scales:
        if time.monotonic() > deadline - 200:
            break
        
        a_target = _upsample(a_cur, target_n)
        sc_target = _score_norm(a_target)
        print(f"  Upsampled to n={target_n}: score={sc_target:.10f}")
        
        if sc_target < sc_cur:
            sc_cur = sc_target
            a_cur = a_target
        
        alloc = time_alloc.get(target_n, 100)
        opt_deadline = min(deadline - 200, time.monotonic() + alloc)
        
        # Adam optimization
        a_opt, sc_opt = _adam_smoothed_max(a_target, opt_deadline, rng,
                                            n_max_iter=200000, lr_init=0.05)
        print(f"  After Adam n={target_n}: score={sc_opt:.10f}")
        
        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt
            update_best(a_cur, sc_cur, f"adam_{target_n}")
    
    # --- Phase 2: Peak equalization + coordinate descent ---
    print(f"\nPhase 2: Refinement (current best: {best_score:.10f})")
    
    if time.monotonic() < deadline - 60 and best_seq is not None:
        # Peak equalization
        pe_deadline = min(deadline - 30, time.monotonic() + 120)
        a_pe, sc_pe = _peak_equalize_refine(best_seq / best_seq.sum(), pe_deadline, rng,
                                             max_rounds=300)
        update_best(a_pe, sc_pe, "peak_equalize")
        
        # Coordinate descent
        if time.monotonic() < deadline - 30:
            cd_deadline = min(deadline - 20, time.monotonic() + 90)
            a_cd, sc_cd = _coordinate_descent(best_seq / best_seq.sum(), cd_deadline, rng,
                                              max_sweeps=100)
            update_best(a_cd, sc_cd, "coord_descent")
    
    # --- Phase 3: Second round of Adam with smaller LR ---
    print(f"\nPhase 3: Fine Adam (current best: {best_score:.10f})")
    
    if time.monotonic() < deadline - 30 and best_seq is not None:
        a_fine = (best_seq / best_seq.sum()).copy()
        fine_deadline = min(deadline - 20, time.monotonic() + (deadline - time.monotonic()) * 0.4)
        a_f, sc_f = _adam_smoothed_max(a_fine, fine_deadline, rng,
                                        n_max_iter=300000, lr_init=0.02)
        update_best(a_f, sc_f, "fine_adam")
    
    # --- Phase 4: Peak equalization again ---
    if time.monotonic() < deadline - 20 and best_seq is not None:
        pe2_deadline = min(deadline - 10, time.monotonic() + 90)
        a_pe2, sc_pe2 = _peak_equalize_refine(best_seq / best_seq.sum(), pe2_deadline, rng,
                                               max_rounds=200)
        update_best(a_pe2, sc_pe2, "peak_equalize2")
    
    # --- Phase 5: Try different n values by resampling ---
    print(f"\nPhase 5: Resampling to different n (current best: {best_score:.10f})")
    
    if time.monotonic() < deadline - 10 and best_seq is not None:
        a_best = best_seq / best_seq.sum()
        # Try n values around the current one
        current_n = len(a_best)
        test_ns = set()
        for mult in [0.5, 0.75, 1.0, 1.25, 1.5]:
            test_n = int(round(current_n * mult))
            if 256 <= test_n <= 8192:
                test_ns.add(test_n)
        # Also try some specific values
        for tn in [1024, 1536, 2048, 3072, 4096, 5120, 6144]:
            if 256 <= tn <= 8192:
                test_ns.add(tn)
        
        for tn in sorted(test_ns):
            if time.monotonic() > deadline - 10:
                break
            a_res = _upsample(a_best, tn)
            sc_res = _score_norm(a_res)
            if sc_res < best_score:
                update_best(a_res, sc_res, f"resample_n{tn}")
            else:
                # Quick Adam on the resampled version
                res_deadline = min(deadline - 10, time.monotonic() + 30)
                a_r, sc_r = _adam_smoothed_max(a_res, res_deadline, rng,
                                                n_max_iter=30000, lr_init=0.03)
                update_best(a_r, sc_r, f"resample+adam_n{tn}")
    
    # --- Phase 6: Final peak equalization ---
    print(f"\nPhase 6: Final polish (current best: {best_score:.10f})")
    
    if time.monotonic() < deadline - 5 and best_seq is not None:
        fp_deadline = min(deadline - 3, time.monotonic() + 40)
        a_fp, sc_fp = _peak_equalize_refine(best_seq / best_seq.sum(), fp_deadline, rng,
                                             max_rounds=100)
        update_best(a_fp, sc_fp, "final_polish")
    
    # --- Return best ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_norm(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback, _ = _make_structured_init(512, rng)
    print(f"FALLBACK: n=512, score={_score_norm(fallback):.10f}")
    return [float(x) for x in fallback]
