"""
Multi-phase optimizer for minimizing 2*n*max(autocorr)/(sum)^2.
Uses LP minimax with line search, L-BFGS with log-sum-exp smoothing,
and targeted coordinate descent. Multi-scale n exploration with
diverse initializations and adaptive time allocation.
"""
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    return csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))


def _lp_solve(f):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, tol_gap_abs=1e-12, tol_gap_rel=1e-12, tol_feas=1e-12)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s < 1e-15:
                return None, None
            return g_val / s, float(t.value)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.PDP)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s < 1e-15:
                return None, None
            return g_val / s, float(t.value)
    except Exception:
        pass
    return None, None


def _lp_phase(a, deadline, max_iters=10000):

def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -80, 80)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -80, 80)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _interp_profile(p0, n_new):

    n0 = len(p0)
    if n0 == n_new:
        s = p0.sum()
        return p0 / s if s > 0 else p0
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-14)
    s = p.sum()
    return p / s if s > 0 else p


def _make_profiles(n, rng):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Flat
    profiles.append(np.ones(n) / n)
    # Linear ramps
    for slope in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0, -2.0, -3.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Cosine windows
    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        a = np.maximum(np.cos(np.pi * u / 2) ** 2, 1e-12) ** (p / 2)
        profiles.append(a / a.sum())
    # Gaussian
    for sigma in [0.1, 0.15, 0.2, 0.3]:

        profiles.append(a / a.sum())
    # Random
    for _ in range(5):
        a = np.abs(rng.standard_normal(n)) + 0.1
        profiles.append(a / a.sum())
    return profiles

        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        profiles = _make_profiles(n_init, rng)
        parent = profiles[0]
        update_best(parent, _score_norm(parent), "flat_init")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, score={best_score:.10f}")

    # === Phase 1: LP minimax at parent size ===
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: LP minimax (n={n_parent}, budget={phase1_end - t_start:.0f}s) ===")

    a_cur = parent.copy()
    a_lbl = "lp"
    a_cur, sc_cur = _lp_phase(a_cur, deadline=phase1_end, max_iters=5000)
    update_best(a_cur, sc_cur, f"lp_n{n_parent}")

    # === Phase 2: L-BFGS log-sum-exp refinement ===
    phase2_end = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 2: L-BFGS continuation (n={len(best_seq)}, budget={phase2_end - t_start:.0f}s) ===")

    a_start = best_seq.copy()
    p_best, sc_best = _lse_lbfgs(a_start, deadline=phase2_end)
    update_best(p_best, sc_best, f"lse_n{len(a_start)}")

    # Try a second L-BFGS run with perturbation
    if time.monotonic() < phase2_end - 60:
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + 0.03 * rng.normal(size=len(z))
        z = np.clip(z, -60, 60)
        z = z - z.mean()
        a_perturbed = _softmax_fn(z)
        per_dl = min(phase2_end - 10, time.monotonic() + 150.0)
        p2, sc2 = _lse_lbfgs(a_perturbed, deadline=per_dl)
        update_best(p2, sc2, "lse_perturbed")

    # === Phase 3: Try larger n ===
    phase3_end = min(deadline, t_start + 1400.0)
    print(f"\n=== Phase 3: Larger n exploration (budget={phase3_end - t_start:.0f}s) ===")

    for n_try in [8192, 16384]:
        if time.monotonic() > phase3_end:
            break
        if n_try <= len(best_seq):
            continue
        print(f"  Trying n={n_try}")
        p_interp = _interp_profile(best_seq, n_try)
        sc_interp = _score_norm(p_interp)
        update_best(p_interp, sc_interp, f"interp_n{n_try}")

        # Short LP at larger n
        lp_dl = min(phase3_end - 30, time.monotonic() + 120.0)
        p_lp, sc_lp = _lp_phase(p_interp, deadline=lp_dl, max_iters=2000)
        update_best(p_lp, sc_lp, f"lp_n{n_try}")

        # Short L-BFGS
        if time.monotonic() < phase3_end - 20:
            lse_dl = min(phase3_end - 10, time.monotonic() + 100.0)
            p_lse, sc_lse = _lse_lbfgs(p_lp, deadline=lse_dl)
            update_best(p_lse, sc_lse, f"lse_n{n_try}")

    # Also try smaller n (sometimes helps)
    if len(best_seq) > 2048 and time.monotonic() < phase3_end - 30:
        n_small = max(1024, len(best_seq) // 4)
        p_small = _interp_profile(best_seq, n_small)
        sc_small = _score_norm(p_small)
        update_best(p_small, sc_small, f"downsample_n{n_small}")
        lp_dl = min(phase3_end - 30, time.monotonic() + 60.0)
        p_small_lp, sc_small_lp = _lp_phase(p_small, deadline=lp_dl, max_iters=1000)
        update_best(p_small_lp, sc_small_lp, f"lp_n{n_small}")

    # === Phase 4: Final coordinate descent polish ===
    phase4_end = min(deadline, t_start + 1700.0)
    print(f"\n=== Phase 4: Coordinate descent polish (n={len(best_seq)}, budget={phase4_end - t_start:.0f}s) ===")

    if best_seq is not None:
        cd_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.8)
        p_cd, sc_cd = _coord_descent(best_seq, deadline=cd_dl, max_rounds=200)
        update_best(p_cd, sc_cd, "coord_descent")

        # One more LP pass on the polished result
        if time.monotonic() < phase4_end - 20:
            lp_dl2 = min(phase4_end - 10, time.monotonic() + 60.0)
            p_lp2, sc_lp2 = _lp_phase(best_seq, deadline=lp_dl2, max_iters=500)
            update_best(p_lp2, sc_lp2, "lp_final")

    # === Phase 5: Tiny-beta L-BFGS ===
    if time.monotonic() < deadline - 15 and best_seq is not None:
        print(f"\n=== Phase 5: Ultra-fine L-BFGS ===")
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c = float(c_final.max())
        beta_ultra = 5e-6 * max_c
        per_dl = min(deadline - 10, time.monotonic() + 40.0)

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        try:
            result = minimize(
                _lse_obj, z, args=(beta_ultra, n_final),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-16,
                         'maxls': 30, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_seq = a_new.copy()
                    print(f"  Ultra-beta improved to {best_score:.10f}")
        except Exception:
            pass

    # Final validation
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"  score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"  Evaluator confirms: {sc_eval:.10f}")
            except Exception as ex:
                print(f"  evaluate_sequence error: {ex}")

        elapsed = time.monotonic() - t_start
        print(f"  Total time: {elapsed:.1f}s")
        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
