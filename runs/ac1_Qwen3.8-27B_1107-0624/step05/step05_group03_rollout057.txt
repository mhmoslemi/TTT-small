```python
def _grad_conv_k(a, k):
    """Compute gradient of b[k] = sum_i a[i]*a[k-i] w.r.t. a."""
    n = len(a)
    # grad[j] = 2 * a[k-j] for valid j (0 <= k-j < n)

    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    g = np.zeros(n)
    if j_start <= j_end:
        indices = np.arange(j_start, j_end + 1)
        g[indices] = 2.0 * a[k - indices]
    return g
```

```python
"""
Optimized minimizer for 2*n*max(conv(a,a))/sum(a)^2 using:
1. Diverse U-shaped and bimodal profile initializations
2. L-BFGS with logsumexp continuation (coarse phase)
3. Nesterov-accelerated subgradient with exact O(n) gradient computation (fine phase)
4. Peak-targeted local search with mass redistribution
5. Adaptive top-K annealing for progressive sharpening of the max
Key innovation: exploits the O(n) gradient structure of conv_k to enable
millions of precise subgradient iterations in the fine optimization phase.

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Score assuming sum=1: 2n * max(convolve(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _grad_conv_k(a, k):
    """Compute gradient of b[k]=conv(a,a)[k] w.r.t. a. O(n)."""
    n = len(a)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    g = np.zeros(n)
    if j_start <= j_end:
        idx = np.arange(j_start, j_end + 1)
        g[idx] = 2.0 * a[k - idx]
    return g


def _topk_subgradient(a, K):
    """Compute top-K weighted subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    c = _full_convolve(a, a)
    K_actual = min(K, len(c))
    top_idx = np.argpartition(c, -K_actual)[-K_actual:]
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        g += _grad_conv_k(a, k)
    g /= float(K_actual)
    return g, c


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_u_profile(n, sharpness=2.0):
    """U-shaped profile: high at edges, low in center."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0
    a = 1.0 + sharpness * u ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):
    """Two Gaussian bumps."""
    t = np.linspace(0.0, 1.0, n)
    center1 = 0.5 - separation / 2.0
    center2 = 0.5 + separation / 2.0
    a = np.exp(-(t - center1) ** 2 / (2 * width ** 2)) + np.exp(-(t - center2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_u_profile(n, p):
    """U-shaped: (1-|2t-1|)^(-p) type."""
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - 0.8 * u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_triangular_weight(n):

    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _lbfgs_continuation(a_init, deadline, max_wall_time=300.0, beta_scale=1.0):
    """L-BFGS with logsumexp continuation for coarse optimization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    betas = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    betas = [b * max_c0 * beta_scale for b in betas]

    per_level = max(2.0, max_wall_time / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_dl = min(deadline - 1.0, time.monotonic() + per_level)

        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300 or not np.isfinite(s_):
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))

        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_normfn_safe(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _score_normfn_safe(a):
    try:
        return _score_norm_fn(a)
    except Exception:
        return np.inf


def _nesterov_subgradient(a_init, deadline, max_iters=2000000, K_start=50, K_end=1,
                           lr_start=0.05, lr_end=0.005, noise_level=0.0):
    """
    Nesterov-accelerated subgradient with multiplicative updates.
    Uses exact O(n) gradient computation for conv_k.
    Progressive K annealing from K_start to K_end.
    """
    n = len(a_init)
    a = a_init.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    t_nest = 1.0
    lr = lr_start
    K = K_start

    eval_interval = max(1, max_iters // 200)

    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break

        # Periodic score evaluation
        if it % eval_interval == 0:
            c = _full_convolve(a, a)
            sc = float(2.0 * n * c.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

            # Adaptive K annealing
            frac = float(it) / float(max_iters)
            K = max(K_end, int(K_start * (1.0 - frac) ** 1.5))

            # Adaptive learning rate
            lr = lr_start * (lr_end / lr_start) ** frac

        # Nesterov momentum
        momentum = (t_nest - 1.0) / (t_nest + 2.0)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 1e-15)
        s_y = y.sum()
        if s_y < 1e-10:
            y = a.copy()
            s_y = a.sum()
        y = y / s_y * a.sum()

        # Compute subgradient
        g, c = _topk_subgradient(y, K)

        # Multiplicative update
        # Scale: g has typical magnitude ~ 2*mean(a) ~ 2/n
        # We want relative change ~ lr per step
        g_scale = g * (n / 2.0)  # Normalize to O(1)
        g_scale = g_scale - np.mean(g_scale)  # Center to preserve sum
        g_scale = np.clip(g_scale, -50.0, 50.0)  # Safety clip

        a_candidate = y * np.exp(-lr * g_scale)
        a_candidate = np.maximum(a_candidate, 1e-15)
        s_c = a_candidate.sum()
        if s_c < 1e-10 or not np.isfinite(s_c):
            a_candidate = y.copy()
            s_c = y.sum()
        a_candidate = a_candidate / s_c * a.sum()

        # Optional: small noise to escape flat regions
        if noise_level > 0.0 and it % 100 == 0:
            noise = noise_level * np.random.randn(n)
            a_candidate = a_candidate + noise * a_candidate
            a_candidate = np.maximum(a_candidate, 1e-15)
            a_candidate /= a_candidate.sum() * a_candidate.sum() * n / a.sum()  # noqa
            a_candidate = np.maximum(a_candidate, 1e-15)
            a_candidate /= a_candidate.sum()
            a_candidate *= a.sum() / a_candidate.sum()  # Keep same sum

        a_prev = a.copy()
        a = a_candidate
        t_nest += 1.0

    # Final evaluation
    c = _full_convolve(a, a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _peak_targeted_refine(a_init, deadline, max_trials=50000):
    """
    Targeted local search: identify the peak lag, find contributing pairs,
    and perform targeted mass transfers to reduce the peak.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    rng = np.random.default_rng(12345)

    for it in range(max_trials):
        if time.monotonic() > deadline - 1.0:
            break

        c = _full_convolve(a, a)
        max_lag = int(np.argmax(c))
        max_val = c[max_lag]

        # Find the top contributing pairs at the max lag
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        if i_min > i_max:
            continue

        pairs = np.arange(i_min, i_max + 1)
        partners = max_lag - pairs
        products = a[pairs] * a[partners]

        # Strategy 1: Reduce the largest product pair
        if it % 3 == 0:
            top_pair_idx = int(np.argmax(products))
            i = int(pairs[top_pair_idx])
            j = int(partners[top_pair_idx])

            # Move mass from the larger of the two to a "safe" location
            if i == j:
                continue
            src = i if a[i] >= a[j] else j
            dst = j if src == i else i

            # Find safe destination: position with minimal contribution to top lags
            top_lags = np.argsort(c)[-5:]
            contrib = np.zeros(n)
            for kl in top_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    contrib[lo:hi + 1] += a[kl - np.arange(lo, hi + 1)]

            safe_positions = np.argsort(contrib)[:max(1, n // 10)]
            add_idx = int(safe_positions[rng.integers(len(safe_positions))])

            frac = rng.uniform(0.05, 0.25)
            delta = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= delta
            a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                a_trial *= a.sum()
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

        # Strategy 2: Equalize the top contributing pairs
        elif it % 3 == 1:
            if len(pairs) < 2:
                continue
            # Try to equalize a[pairs] values
            mean_prod = np.mean(products)
            for pi in range(len(pairs)):
                i = int(pairs[pi])
                j = int(partners[pi])
                if i == j:
                    continue
                if products[pi] > mean_prod * 1.2:
                    a_trial = a.copy()
                    # Reduce the product by reducing the larger element
                    if a[i] > a[j]:
                        target = np.sqrt(products[pi] * a[j] / a[i] * 0.8)
                        if target > 0 and target < a[i]:
                            a_trial[i] = target
                    else:
                        target = np.sqrt(products[pi] * a[i] / a[j] * 0.8)
                        if target > 0 and target < a[j]:
                            a_trial[j] = target
                    a_trial = np.maximum(a_trial, 0.0)
                    s = a_trial.sum()
                    if s > 1e-10:
                        a_trial = a_trial / s * a.sum()
                        sc = _score_norm_fn(a_trial)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            break

        # Strategy 3: Random small perturbation at the peak-contributing positions
        else:
            n_perturb = min(5, len(pairs))
            chosen = rng.choice(len(pairs), size=n_perturb, replace=False)
            a_trial = a.copy()
            for pi in chosen:
                i = int(pairs[pi])
                mult = rng.uniform(0.85, 1.15)
                a_trial[i] *= mult
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial = a_trial / s * a.sum()
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()

    return best_a, best_sc


def _multiplicative_descent(a_init, deadline, max_iters=500000, lr_start=0.1, lr_end=0.01):
    """
    Pure multiplicative descent with exact subgradient (no Nesterov).
    More stable than Nesterov for non-convex problems.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    eval_interval = max(1, max_iters // 500)

    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break

        if it % eval_interval == 0:
            c = _full_convolve(a, a)
            sc = float(2.0 * n * c.max())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        frac = float(it) / float(max_iters)
        lr = lr_start * (lr_end / lr_start) ** frac
        K = max(1, int(20 * (1.0 - frac)))

        # Subgradient
        g, c = _topk_subgradient(a, K)

        # Multiplicative update
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            break
        g_centered = g - np.mean(g)
        g_centered = np.clip(g_centered, -10.0 * g_norm, 10.0 * g_norm)

        a_new = a * np.exp(-lr * g_centered / g_norm)
        a_new = np.maximum(a_new, 1e-15)
        s = a_new.sum()
        if s < 1e-10 or not np.isfinite(s):
            break
        a_new = a_new / s * a.sum()
        a = a_new

    c = _full_convolve(a, a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _two_phase_refine(a_init, deadline):
    """Two-phase: Nesterov then multiplicative descent."""
    mid = time.monotonic() + (deadline - time.monotonic()) * 0.5

    # Phase 1: Nesterov
    a1, s1 = _nesterov_subgradient(a_init, mid, max_iters=500000,
                                    K_start=30, K_end=3, lr_start=0.08, lr_end=0.01)
    print(f"  Two-phase P1 (Nesterov): score={s1:.10f}")

    # Phase 2: Multiplicative
    a2, s2 = _multiplicative_descent(a1, deadline, max_iters=400000,
                                      lr_start=0.05, lr_end=0.005)
    print(f"  Two-phase P2 (Mult): score={s2:.10f}")

    return a2, s2


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer:
    1. Diverse U-shaped initializations
    2. L-BFGS logsumexp continuation (coarse)
    3. Nesterov subgradient with O(n) gradient (fine)
    4. Peak-targeted local search
    5. Final polishing
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_u_profile(n_parent, 1.5)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using U-profile n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    n_work = n_parent
    print(f"Working with n={n_work}")

    # --- Phase 0: Diverse initialization (30s) ---
    phase0_end = min(deadline, t_start + 30.0)

    profiles = [
        _make_u_profile(n_work, 1.0),
        _make_u_profile(n_work, 1.5),
        _make_u_profile(n_work, 2.0),
        _make_u_profile(n_work, 3.0),
        _make_u_profile(n_work, 5.0),
        _make_bimodal_profile(n_work, 0.3, 0.1),
        _make_bimodal_profile(n_work, 0.4, 0.12),
        _make_bimodal_profile(n_work, 0.5, 0.15),
        _make_bimodal_profile(n_work, 0.6, 0.2),
        _make_bimodal_profile(n_work, 0.7, 0.25),
        _make_power_u_profile(n_work, 2.0),
        _make_power_u_profile(n_work, 3.0),
        _make_power_u_profile(n_work, 5.0),
        _make_power_u_profile(n_work, 8.0),
        _make_edge_concentrated(n_work, 1.0),
        _make_edge_concentrated(n_work, 2.0),
        _make_edge_concentrated(n_work, 3.0),
        _make_quadratic_profile(n_work, 0.0, 1.0),
        _make_quadratic_profile(n_work, 0.0, 2.0),
        _make_quadratic_profile(n_work, 0.0, 3.0),
        _make_quadratic_profile(n_work, 1.0, 0.5),
        _make_quadratic_profile(n_work, -1.0, 1.5),
        _strcmp_profile(n_work, 1, 0.5),
        _strcmp_profile(n_work, 1, 0.8),
        _strcmp_profile(n_work, 2, 0.4),
        _make_triangular_weight(n_work),
        _interp_profile(parent, n_work) if parent is not None else _make_u_profile(n_work, 2.0),
    ]

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}, top={candidates[0][2] if candidates else 'none'}")

    # Pick best starting point for further optimization
    if candidates:
        start_a = candidates[0][1].copy()
    else:
        start_a = parent.copy() if parent is not None else _make_u_profile(n_work, 2.0)

    # Also compare with parent
    if parent is not None:
        sc_p = _score_norm_fn(parent)
        if sc_p < _score_normfn_safe(start_a):
            start_a = parent.copy()

    # --- Phase 1: L-BFGS continuation (300s) ---
    phase1_end = min(deadline, t_start + 330.0)
    if time.monotonic() < phase1_end:
        print(f"Phase 1: L-BFGS, start score={_score_normfn_safe(start_a):.10f}")
        a1, s1 = _lbfgs_continuation(start_a, phase1_end, max_wall_time=300.0, beta_scale=1.0)
        update_best(a1, s1, "lbfgs_1")

        # Second L-BFGS pass from a different starting point
        if len(candidates) > 1 and time.monotonic() < phase1_end - 30:
            start2 = candidates[1][1].copy()
            a1b, s1b = _lbfgs_continuation(start2, phase1_end - 10, max_wall_time=60.0, beta_scale=0.5)
            update_best(a1b, s1b, "lbfgs_1b")

    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: Nesterov subgradient (500s) ---
    phase2_end = min(deadline, t_start + 830.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print(f"Phase 2: Nesterov subgradient, start score={_score_normfn_safe(best_seq):.10f}")
        a2, s2 = _nesterov_subgradient(best_seq, phase2_end, max_iters=1500000,
                                        K_start=50, K_end=2, lr_start=0.1, lr_end=0.01,
                                        noise_level=0.0)
        update_best(a2, s2, "nesterov")

    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: Peak-targeted local search (300s) ---
    phase3_end = min(deadline, t_start + 1130.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print(f"Phase 3: Peak-targeted search, start score={_score_normfn_safe(best_seq):.10f}")
        a3, s3 = _peak_targeted_refine(best_seq, phase3_end, max_trials=50000)
        update_best(a3, s3, "peak_refine")

    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Fine Nesterov (300s) ---
    phase4_end = min(deadline, t_start + 1430.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print(f"Phase 4: Fine Nesterov, start score={_score_normfn_safe(best_seq):.10f}")
        a4, s4 = _nesterov_subgradient(best_seq, phase4_end, max_iters=1000000,
                                        K_start=10, K_end=1, lr_start=0.03, lr_end=0.003)
        update_best(a4, s4, "nesterov_fine")

    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Multiplicative descent (200s) ---
    phase5_end = min(deadline, t_start + 1630.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print(f"Phase 5: Mult descent, start score={_score_normfn_safe(best_seq):.10f}")
        a5, s5 = _multiplicative_descent(best_seq, phase5_end, max_iters=500000,
                                          lr_start=0.05, lr_end=0.005)
        update_best(a5, s5, "mult_descent")

    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Final peak-targeted polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"Phase 6: Final polishing, start score={_score_normfn_safe(best_seq):.10f}")
        a6, s6 = _peak_targeted_refine(best_seq, deadline - 10, max_trials=30000)
        update_best(a6, s6, "final_polish")

        # One more Nesterov pass with very small steps
        if time.monotonic() < deadline - 15 and best_score < 1.51:
            a7, s7 = _nesterov_subgradient(best_seq, deadline - 5, max_iters=300000,
                                            K_start=5, K_end=1, lr_start=0.01, lr_end=0.002)
            update_best(a7, s7, "final_nesterov")

    # --- Final validation ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_u_profile(4096, 2.0)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        # Compute actual score
        n_final = len(final_seq)
        c_final = _full_convolve(final_seq, final_seq)
        sc_final = float(2.0 * n_final * c_final.max() / (s * s))
        print(f"FINAL: score={sc_final:.10f}, n={n_final}, sum={s:.6f}")

        # Try to validate with evaluator if sequence is small enough
        if n_final <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_u_profile(4096, 2.0)
    sc_fb = _score_norm_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
```
