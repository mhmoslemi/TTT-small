def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_minimax(f, max_time=60):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gv = g_val.sum()
            if gv > 0:
                g_val = g_val / gv
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None

                if np.all(a_new >= 0):
                    s_new = _score_norm_fn(a_new)
                    if np.isfinite(s_new) and s_new < best_score:
                        best_score = s_new
                        best_a = a_new.copy()
                        z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _single_beta_lbfgs(a_init, beta, deadline, maxiter=8000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    try:
        result = minimize(
            _logsumexp_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_logsumexp_grad,
            options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                     'maxls': 40, 'maxfun': maxiter * 6}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            s_new = _score_norm_fn(a_new)
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_a = a_new.copy()
    except Exception:
        pass
    return best_a, best_score


def _coord_descent_fine(a_init, deadline, max_rounds=20):
    """Fine coordinate descent with more delta candidates."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    deltas = [-0.05, -0.03, -0.02, -0.01, -0.005, -0.002,
              0.002, 0.005, 0.01, 0.02, 0.03, 0.05]

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            best_delta = 0.0
            best_local_score = _score_norm_fn(a)
            for delta in deltas:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_local_score:
                    best_local_score = sc
                    best_delta = delta
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)
                z = z - z.mean()
                a = _softmax_fn(z)
                sc = _score_norm_fn(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        a = np.abs(u) ** p + 0.01
        profiles.append(a / a.sum())
    for slope in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + slope * np.abs(u)
        profiles.append(a / a.sum())
    for rate in [1.0, 2.0, 3.0, 4.0]:
        a = np.exp(rate * np.abs(u))
        profiles.append(a / a.sum())
    a = 1.0 / (0.1 + u ** 2)
    profiles.append(a / a.sum())
    a = np.ones(n)
    profiles.append(a / n)
    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        parent = _make_profiles(n_init)[3]
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    print(f"Starting: n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Extended Frank-Wolfe with LP (600s) ===
    phase1_end = min(deadline, t_start + 600.0)
    print(f"\nPhase 1: Frank-Wolfe with adaptive steps (target n={min(n_parent, 4096)})")

    n_fw = min(n_parent, 4096)
    if n_parent > n_fw:
        a_fw = _interp_profile(parent, n_fw)
    else:
        a_fw = parent.copy()
    sc_fw_init = _score_norm_fn(a_fw)
    update_best(a_fw, sc_fw_init, f"fw_init_n{n_fw}")

    a_cur, sc_cur = _fw_adaptive(a_fw, deadline=phase1_end, n_max_iters=3000)
    update_best(a_cur, sc_cur, f"fw_n{n_fw}")

    if time.monotonic() < phase1_end - 10:
        a_cur2, sc_cur2 = _fw_adaptive(a_cur, deadline=phase1_end, n_max_iters=2000)
        update_best(a_cur2, sc_cur2, f"fw2_n{n_fw}")

    if n_parent > n_fw:
        a_rec = _interp_profile(a_cur2, n_parent)
        sc_rec = _score_norm_fn(a_rec)
        update_best(a_rec, sc_rec, f"fw_interp_n{n_parent}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov-accelerated mirror descent (400s) ===
    phase2_end = min(deadline, t_start + 1000.0)
    print(f"\nPhase 2: Nesterov mirror descent (n={len(best_seq)})")

    n_curr = len(best_seq)
    a_start = best_seq.copy()
    c_start = _full_convolve(a_start, a_start)
    beta_init = 0.03 * float(c_start.max())

    per_dl = min(phase2_end - 10, time.monotonic() + 350.0)
    p_best, sc_best = _nesterov_mirror(a_start, per_dl, beta_init=beta_init)
    update_best(p_best, sc_best, f"nesterov_n{len(a_start)}")

    # Second Nesterov run with smaller initial beta
    if time.monotonic() < phase2_end - 20:
        beta_small = 0.01 * float(_full_convolve(best_seq, best_seq).max())
        per_dl2 = min(phase2_end - 5, time.monotonic() + 200.0)
        p_best2, sc_best2 = _nesterov_mirror(best_seq, per_dl2, beta_init=beta_small)
        update_best(p_best2, sc_best2, f"nesterov2_n{len(best_seq)}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: L-BFGS continuation (350s) ===
    phase3_end = min(deadline, t_start + 1350.0)
    print(f"\nPhase 3: L-BFGS fine continuation (n={len(best_seq)})")

    n_final = len(best_seq)
    a_for_lbfgs = best_seq.copy()

    # Multi-restart L-BFGS
    n_restarts = 4
    per_restart_time = (phase3_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase3_end - 20:
            break

        if r == 0:
            a_cur_r = a_for_lbfgs.copy()
        elif r == 1:
            z = np.log(np.maximum(a_for_lbfgs, 1e-15))
            z = z - z.mean()
            z = z + 0.02 * rng.normal(size=n_final)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur_r = _softmax_fn(z)
        elif r == 2:
            z = np.log(np.maximum(a_for_lbfgs, 1e-15))
            z = z - z.mean()
            z = z + 0.05 * rng.normal(size=n_final)
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur_r = _softmax_fn(z)
        else:
            # Try with different n
            n_alt = 2048 if n_final > 2048 else 3000
            a_alt = _interp_profile(a_for_lbfgs, n_alt)
            a_cur_r = a_alt

        per_dl = min(phase3_end - 10, time.monotonic() + per_restart_time)
        p_best_r, sc_best_r = _lbfgs_continuation(a_cur_r, per_dl)
        update_best(p_best_r, sc_best_r, f"lbfgs_r{r}_n{len(a_cur_r)}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Try alternative n values (200s) ===
    phase4_end = min(deadline, t_start + 1550.0)
    print(f"\nPhase 4: Alternative n exploration")

    n_alternatives = [n_final]
    if n_final != 2048:
        n_alternatives.append(2048)
    if n_final != 3072:
        n_alternatives.append(3072)
    if n_final != 5000:
        n_alternatives.append(5000)
    if n_final != 8192:
        n_alternatives.append(8192)

    for n_alt in n_alternatives:
        if time.monotonic() > phase4_end - 30:
            break
        if n_alt == n_final:
            continue
        p_alt = _interp_profile(best_seq, n_alt)
        sc_alt = _score_norm_fn(p_alt)
        update_best(p_alt, sc_alt, f"interp_n{n_alt}")

        if time.monotonic() < phase4_end - 20:
            c_alt = _full_convolve(p_alt, p_alt)
            beta_alt = 0.01 * float(c_alt.max())
            per_dl = min(phase4_end - 10, time.monotonic() + 60.0)
            p_pol, sc_pol = _single_beta_lbfgs(p_alt, beta_alt, per_dl, maxiter=3000)
            update_best(p_pol, sc_pol, f"lbfgs_n{n_alt}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Ultra-fine L-BFGS + coordinate descent (remaining) ===
    phase5_end = deadline
    print(f"\nPhase 5: Ultra-fine polishing")

    if time.monotonic() < phase5_end and best_seq is not None:
        # Ultra tiny beta L-BFGS
        n_fin = len(best_seq)
        c_fin = _full_convolve(best_seq, best_seq)
        max_c = float(c_fin.max())

        for beta_frac in [5e-5, 3e-5, 2e-5, 1e-5, 5e-6]:
            if time.monotonic() > phase5_end - 15:
                break
            beta_ultra = beta_frac * max_c
            per_dl = min(phase5_end - 10, time.monotonic() + 50.0)
            p_pol, sc_pol = _single_beta_lbfgs(best_seq, beta_ultra, per_dl, maxiter=5000)
            update_best(p_pol, sc_pol, f"ultra_beta_{beta_frac}")
            if p_pol is not None and np.isfinite(sc_pol) and sc_pol < best_score:
                best_seq = p_pol.copy()
                best_score = sc_pol
                c_fin = _full_convolve(best_seq, best_seq)
                max_c = float(c_fin.max())

        # Coordinate descent
        if time.monotonic() < phase5_end - 10:
            per_dl = min(phase5_end - 5, time.monotonic() + 80.0)
            p_cd, sc_cd = _coord_descent_fine(best_seq, per_dl, max_rounds=15)
            update_best(p_cd, sc_cd, "coord_descent_fine")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
