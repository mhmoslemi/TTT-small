"""
Hybrid optimizer: L-BFGS with log-sum-exp smoothing + active set refinement
+ multi-scale perturbation for minimizing 2n*max(conv(a,a))/(sum(a))^2.
"""
import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft

def _next_pow2(n):
    return 1 << (n - 1).bit_length()

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = rfft(a, N)
    fb = rfft(b, N)
    return irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1: 2n * max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _lse_objective(a, T):
    """Log-sum-exp smoothed objective (sum(a)=1 assumed)."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    # LSE approximation to max
    if T < 1e-10:
        return c_max
    shifted = (c - c_max) / T
    e = np.exp(shifted)
    lse = c_max + T * np.log(e.sum())
    return 2.0 * n * lse

def _lse_gradient(a, T):
    """Gradient of LSE smoothed objective."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / T
    e = np.exp(shifted)

def _adam_smoothed(a_init, deadline, T_start, T_end, lr=0.005, max_iter=500000):
    """Adam optimizer on the LSE-smoothed objective with annealing."""
    n = len(a_init)
    # Work in log-space for non-negativity
    z = np.log(np.maximum(a_init, 1e-300))
    z -= z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    while t < max_iter and time.monotonic() < deadline:
        t += 1
        a = np.exp(z)
        a = a / a.sum()  # normalize to sum=1
        
        T = T_start * (T_end / T_start) ** (t / max_iter)
        
        obj = _lse_objective(a, T)
        grad = _lse_gradient(a, T)
        
        # Project gradient to maintain sum=1: subtract mean weighted by a
        grad = grad - a * np.dot(a, grad)  # tangent space of simplex
        # Actually for the softmax parametrization, the gradient with respect to z is:
        # dz_j = a_j * (grad_j - sum_i a_i * grad_i)
        # Because a = softmax(z), so da_j/dz_k = a_j*(delta_jk - a_k)
        # d(obj)/dz_k = sum_j da_j/dz_k * d(obj)/da_j = a_k*(grad_k - sum_j a_j*grad_j)
        
        if not np.all(np.isfinite(grad)):
            break
        
        grad_z = a * (grad - np.dot(a, grad))
        
        m_buf = b1 * m_buf + (1 - b1) * grad_z
        v_buf = b2 * v_buf + (1 - b2) * grad_z ** 2
        mh = m_buf / (1 - b1 ** t)
        vh = v_buf / (1 - b2 ** t)
        
        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        
        z -= update
        z = np.clip(z, -30, 30)
        z -= z.mean()
        
        if t % 500 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    a_final = np.exp(z)
    a_final = a_final / a_final.sum()
    sc = _score_norm(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()
    
    return best_a, best_score

def _lbfgs_smoothed(a_init, deadline, T, max_iter=2000, tol=1e-10):
    """L-BFGS on the LSE-smoothed objective."""
    n = len(a_init)
    z0 = np.log(np.maximum(a_init, 1e-300))
    z0 -= z0.mean()
    
    def objective(z):
        a = np.exp(z - z.max())  # for numerical stability
        a = a / a.sum()
        return _lse_objective(a, T)
    
    def gradient(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        grad = _lse_gradient(a, T)
        grad_z = a * (grad - np.dot(a, grad))
        return grad_z
    
    start = time.monotonic()
    try:
        result = minimize(
            objective, z0, jac=gradient, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol,
                     'maxfun': 100000}
        )
    except Exception:
        result = None
    
    if result is not None and result.x is not None:
        a = np.exp(result.x - result.x.max())
        a = a / a.sum()
        return a
    return a_init
