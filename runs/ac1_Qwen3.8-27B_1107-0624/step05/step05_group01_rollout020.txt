"""
Optimized step function sequence minimization.
Uses iterative LP peak balancing, log-sum-exp L-BFGS continuation,
and Adam refinement to minimize 2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    """Score assuming sum(a) = 1 (or any fixed normalization)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j-1] * (2.0*j - 1.0) / (2.0*j)
    a = b[::-1].copy()
    return a / a.sum()

    """Intensive LP-based peak balancing."""
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    best_score = _score_norm(a)
    best_a = a.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        current_score = 2.0 * n * M
        
        # Find active peaks
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K_peaks]
        # Also include peaks within 90% of max
        threshold = 0.90 * M

        active = np.unique(np.concatenate([active, additional]))

def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)

def _softmin_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n-1:2*n-1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad

def _lbfgs_phase(a_init, deadline, beta_list=None, maxiter_per_beta=10000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001,
                 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        beta_list = [b * m0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),

            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)

                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:

    return best_a, best_score

"""
Step function sequence optimization minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses intensive LP peak balancing, gradient-based peak shaving,
L-BFGS log-sum-exp continuation, and Adam refinement.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _lp_intensive(a_init, deadline, K=200, max_rounds=1000):
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    best_a = a.copy()
    best_score = _score_norm(a)
    slack = 1.0005
    stall = 0
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        M = float(c.max())
        cur = 2.0 * n * M
        
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        thr = 0.90 * M
        extra = np.where(c >= thr)[0]
        active = np.unique(np.concatenate([active, extra]))
        if len(active) > 500:
            active = sorted_idx[:500]
        
        A_mat = np.zeros((len(active), n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
        
        b_vec = np.full(len(active), M * slack)
        
        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 15})
            if res.status != 0:
                break
            b_new = res.x
            sb = float(b_new.sum())
            if sb < 1e-15:
                break
            b_scaled = b_new * (S / sb)
        except Exception:
            break
        
        improved = False
        for t in [0.1, 0.05, 0.02, 0.01, 0.005]:
            a_t = (1.0 - t) * a + t * b_scaled
            st = float(a_t.sum())
            if st < 1e-15:
                continue
            ct = _full_convolve(a_t, a_t)
            Mt = float(ct.max())
            sc = 2.0 * n * Mt
            if sc < cur - 1e-14:

                cur = sc

def _grad_peak_shave(a_init, deadline, n_peaks=5):
    n = len(a_init)
    a = a_init.copy()
    S = float(a.sum())
    best_a = a.copy()
    best_score = _score_norm(a)
    lr = 0.05
    stall = 0
    
    while time.monotonic() < deadline and stall < 200:
        c = _full_convolve(a, a)
        M = float(c.max())
        cur = 2.0 * n * M
        
        peak_idx = np.argsort(c)[::-1][:n_peaks]
        
        grad = np.zeros(n)
        for k in peak_idx:
            w = c[k] / M
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                positions = np.arange(i_lo, i_hi + 1)
                grad[positions] += w * 2.0 * a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        if gn < 1e-15:
            break
        
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * scale * grad / gn
            a_t = a - step
            a_t = np.maximum(a_t, 0.0)
            st = float(a_t.sum())
            if st < 1e-15:

            ct = _full_convolve(a_t, a_t)

            if sc < cur - 1e-14:

                cur = sc

    return best_a, best_score


def _softmin_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)
