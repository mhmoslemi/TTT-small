```python
def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

"""
Optimized step function search using Nesterov-accelerated subgradient descent
with exact peak identification, log-sum-exp smoothing, and local peak surgery.
Key innovations over previous approach:
1. Exact subgradient (not just log-sum-exp) for final refinement

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * len(a) * beta * logsum


def _logsumexp_grad(z, beta):
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    # Gradient of sum_k w[k]*c[k] w.r.t. a[j]
    # = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # Compute h[j] = sum_k w[k] * a[k-j] for valid k-j in [0,n-1]
    # This is the convolution of w with a, evaluated at specific lags
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    # w_conv[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)]
    # We want h[j] = sum_k w[k] * a[k-j] for k-j in [0,n-1]
    # Let l = k-j, then k = j+l, so h[j] = sum_l w[j+l] * a[l] for valid j+l
    # = (w * a)[j] where * is cross-correlation
    # Actually: h[j] = sum_k w[k]*a[k-j] = (w cross-correlate a)[j]
    # In terms of convolution: cross-correlation can be done as convolution with reversed signal
    # h[j] = sum_k w[k]*a[k-j] = sum_l w[j+l]*a[l] (substituting l=k-j)
    # = (flip(w) * a)[n-1-j] ... this is getting complicated
    # Let me just do it directly using FFT
    # h[j] = sum_k w[k] * a[k-j] for 0<=k<n, 0<=k-j<n
    # = sum_{l=max(0,j-n+1)}^{min(j,n-1)} w[l+j] * a[l]  ... no
    # Let me just compute it as: for each j, h[j] = dot(w[valid_range], a[valid_range2])
    # Easier: use the fact that if we define B[k] = w[k] for k in [0, 2n-2] (zero-padded),
    # then h[j] = sum_k B[k]*a[k-j] = (B * a_flipped)[n-1-j] ... still confusing
    # Simplest: just use the gradient formula directly
    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for valid indices)
    # Let me compute this via FFT:
    # Define x = w (length 2n-1, zero-padded to N)
    # Define y = a (length n, zero-padded to N)
    # We want h[j] = sum_k x[k] * y[k-j] for valid k, k-j
    # This is the cross-correlation of x and y at lag j
    # cross_corr(j) = sum_k x[k] * y[k-j]
    # In FFT: cross_corr = IDFT(FFT(x) * conj(FFT(y)))
    N = _next_pow2(2 * n)
    X = np.fft.rfft(w, N)  # w has length 2n-1
    Y = np.fft.rfft(a, N)
    cc = np.fft.irfft(X * np.conj(Y), N)
    # cc[j] = sum_k x[k]*y[(k-j) mod N] for j=0,...,N-1
    # For j in [0, n-1]: cc[j] = sum_{k=j}^{j+2n-2} x[k]*y[k-j] (accounting for wrap-around)
    # Since x is zero outside [0, 2n-2] and y is zero outside [0, n-1]:
    # cc[j] = sum_{k=j}^{2n-2} x[k]*y[k-j] where k-j is in [0, n-1]
    # = sum_{k=j}^{j+n-1} w[k]*a[k-j]
    # This is what we want! (with w zero-padded to length 2n-1)
    # So h = cc[0:n]
    h = cc[:n]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _nesterov_gd(a_init, deadline, rng, lr_init=0.05, max_iter=500000):
    """Nesterov-accelerated gradient descent with exact subgradient of max(c)."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    # Nesterov state
    y = a.copy()
    t = 0
    lr = lr_init
    patience = 0
    best_patience_score = best_score
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        # Compute gradient at y
        c = _full_convolve(y, y)
        k_max = int(np.argmax(c))
        c_max = float(c[k_max])
        score = 2.0 * n * c_max
        
        # Gradient of c[k_max] w.r.t. a
        # grad[j] = 2*a[k_max-j] if 0 <= k_max-j < n, else 0
        valid_start = max(0, k_max - n + 1)
        valid_end = min(n, k_max + 1)
        grad = np.zeros(n)
        grad[valid_start:valid_end] = 2.0 * y[k_max - (np.arange(valid_start, valid_end))]
        
        # Nesterov step
        a_new = _simplex_project(a - lr * grad)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (beta - 1.0) / beta * (a_new - a)
        a = a_new
        
        # Track best
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate
        if patience > 5000:
            lr *= 0.95
            patience = 0
        if patience > 20000:
            # Perturb to escape
            z = np.log(np.maximum(a, 1e-15)) + 0.05 * rng.normal(size=n)
            a = _simplex_project(_softmax_fn(z))
            y = a.copy()
            patience = 0
            lr = min(0.1, lr * 1.2)
    
    return best_a, best_score


def _run_lbfgs(a_init, deadline, n_betas=10):

def _peak_surgery(a_init, deadline, rng, max_rounds=1000):
    """Local search: identify peaks and try to reduce them by adjusting contributors."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        score = 2.0 * n * c_max
        
        # Find top peaks
        top_k = min(5, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        
        improved = False
        for pk in peak_indices[:3]:
            # Find top contributors to c[pk]
            valid_i = np.arange(max(0, pk - n + 1), min(n, pk + 1))
            if len(valid_i) == 0:
                continue
            prod = a[valid_i] * a[pk - valid_i]
            order = np.argsort(prod)[::-1]
            
            for idx_pos in order[:6]:
                i = int(valid_i[idx_pos])
                j = pk - i
                if j < 0 or j >= n or j == i:
                    continue
                # Try reducing a[i] or a[j]
                for target in [i, j]:
                    for frac in [0.1, 0.05, 0.02, 0.01]:
                        a_trial = a.copy()
                        amount = a[target] * frac
                        a_trial[target] -= amount
                        # Redistribute to the smallest element
                        min_idx = int(np.argmin(a_trial))
                        a_trial[min_idx] += amount
                        a_trial = _simplex_project(a_trial)
                        sc = _score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            best_score = sc
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved and rnd > 50:
            break
    
    return best_a, best_score


def _upsample_smooth(a, target_n):
    """Upsample a to target_n using linear interpolation, keeping positivity."""
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _spectral_equalize(a_init, deadline, rng, iters=5000):
    """Spectral equalization: work in frequency domain to flatten the power spectrum."""
    n = len(a_init)
    N = _next_pow2(2 * n - 1)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    A = np.fft.rfft(a, N)
    lr = 0.1
    t = 0
    
    for it in range(iters):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Compute current autocorrelation
        c_full = _full_convolve(a, a)
        c_max = float(c_full.max())
        score = 2.0 * n * c_max
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Compute spectrum
        P = np.abs(A)**2
        P_mean = P.sum() / len(P)
        
        # Flatten: reduce peaks in P
        # Target: make P more uniform
        alpha = lr / (P_max_val + P_mean) if (P_max_val := P.max()) > 0 else 0.01
        # Reduce high components
        mask = P > 2.0 * P_mean
        A[mask] *= (1.0 - 0.01 * (P[mask] - P_mean) / P_mean)
        
        # Recover a
        a_new = np.fft.irfft(A, N)[:n]
        a = _simplex_project(a_new)
        A = np.fft.rfft(a, N)
        
        # Decay learning rate
        if t % 1000 == 0:
            lr *= 0.8
    
    return best_a, best_score

"""
Step function optimization using Nesterov-accelerated subgradient descent with
adaptive soft peaks, L-BFGS log-sum-exp continuation, and local peak surgery.
Key innovations:
1. Soft subgradient (weighted top-K peaks) for smoother Nesterov dynamics
2. Adaptive step-size with backtracking when argmax jumps
3. Multi-restart with diverse perturbation scales
4. Spectral-domain flattening as auxiliary refinement
5. Careful multi-phase time budget allocation
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n = len(a)
    m = len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _simplex_project(v):
    n = len(v)
    if n == 0:
        return np.array([])
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return 2.0 * n * beta * logsum


def _lse_grad(z, beta):
    n = len(z)
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    N = _next_pow2(2 * n)
    X = np.fft.rfft(w, N)
    Y = np.fft.rfft(a, N)
    cc = np.fft.irfft(X * np.conj(Y), N)
    h = cc[:n]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _run_lbfgs(a_init, deadline, n_betas=10, lr_scale=1.0):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_normalized(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    if max_c0 < 1e-15:
        return best_a, best_score
    beta_start = 0.15 * max_c0
    beta_end = max(1e-7 * max_c0, 1e-14)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    for idx, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = min((deadline - time.monotonic()) / max(1, remaining), 90.0)
        level_deadline = min(deadline, time.monotonic() + time_budget)
        if time.monotonic() >= level_deadline:
            continue
        maxiter = 5000
        try:
            result = minimize(
                _lse_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxls': 40, 'maxfun': maxiter * 8}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = np.clip(result.x, -60, 60)
                z_new = z_new - z_new.mean()
                a_new = _softmax_fn(z_new)
                sc_new = _score_normalized(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
    return best_a, best_score


def _nesterov_gd(a_init, deadline, rng, lr_init=0.02, top_k=3, max_iter=2000000):
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)
    y = a.copy()
    t = 0
    lr = lr_init
    stagnation = 0
    prev_kmax = -1
    jump_count = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        c = _full_convolve(y, y)
        sorted_idx = np.argsort(c)[::-1]
        k_peek = int(sorted_idx[0])
        c_max = float(c[k_peek])

        if k_peek != prev_kmax:
            jump_count += 1
        prev_kmax = k_peek

        if top_k == 1:
            grad = np.zeros(n)
            vs = max(0, k_peek - n + 1)
            ve = min(n, k_peek + 1)
            idx = np.arange(vs, ve)
            grad[idx] = 2.0 * y[k_peek - idx]
        else:
            k_top = sorted_idx[:top_k]
            c_top = c[k_top]
            w_top = c_top / c_top.sum()
            grad = np.zeros(n)
            for ki, wi in zip(k_top, w_top):
                vs = max(0, int(ki) - n + 1)
                ve = min(n, int(ki) + 1)
                idx = np.arange(vs, ve)
                grad[idx] += wi * 2.0 * y[int(ki) - idx]

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break

        a_new = _simplex_project(a - lr * grad)
        beta_mom = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (beta_mom - 1.0) / beta_mom * (a_new - a)
        a = a_new

        score = 2.0 * n * c_max
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 3000:
            lr *= 0.9
            stagnation = 0
        if stagnation > 15000:
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z + 0.03 * rng.normal(size=n)
            a = _simplex_project(_softmax_fn(z))
            y = a.copy()
            lr = min(0.05, lr * 1.5)
            stagnation = 0

        if jump_count > 50 and t > 1000:
            lr *= 0.7
            jump_count = 0

    return best_a, best_score


def _soft_nesterov(a_init, deadline, rng, lr_init=0.01, max_iter=1000000):
    """Nesterov with smooth approximation via top-K weighted gradient."""
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)
    y = a.copy()
    t = 0
    lr = lr_init
    stagnation = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        c = _full_convolve(y, y)
        cm = float(c.max())
        if cm < 1e-20:
            break
        shifted = (c - cm) / (0.1 * cm)
        e = np.exp(shifted)
        w = e / e.sum()
        N = _next_pow2(2 * n)
        X = np.fft.rfft(w, N)
        Y = np.fft.rfft(y, N)
        cc = np.fft.irfft(X * np.conj(Y), N)
        h = cc[:n]
        grad = 2.0 * h

        a_new = _simplex_project(a - lr * grad)
        beta_mom = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (beta_mom - 1.0) / beta_mom * (a_new - a)
        a = a_new

        score = 2.0 * n * cm
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 5000:
            lr *= 0.95
            stagnation = 0
        if stagnation > 30000:
            z = np.log(np.maximum(a, 1e-15))
            z = z - z.mean()
            z = z + 0.02 * rng.normal(size=n)
            a = _simplex_project(_softmax_fn(z))
            y = a.copy()
            lr = min(0.03, lr * 1.3)
            stagnation = 0

    return best_a, best_score


def _peak_surgery(a_init, deadline, rng, max_rounds=5000):
    n = len(a_init)
    a = _simplex_project(a_init.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        top_k = min(3, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]
        improved = False

        for pk in peak_indices:
            valid_i = np.arange(max(0, int(pk) - n + 1), min(n, int(pk) + 1))
            if len(valid_i) == 0:
                continue
            prod = a[valid_i] * a[int(pk) - valid_i]
            order = np.argsort(prod)[::-1]
            handled = set()
            for idx_pos in order[:8]:
                i = int(valid_i[idx_pos])
                j = int(pk) - i
                if i == j or i in handled or j in handled:
                    continue
                handled.add(i)
                handled.add(j)
                for frac in [0.02, 0.01, 0.005, 0.002]:
                    for target in [i, j]:
                        a_trial = a.copy()
                        amount = a_trial[target] * frac
                        a_trial[target] -= amount
                        min_idx = int(np.argmin(a_trial))
                        a_trial[min_idx] += amount
                        a_trial = _simplex_project(a_trial)
                        sc_trial = _score_normalized(a_trial)
                        if sc_trial < best_score - 1e-14:
                            best_score = sc_trial
                            best_a = a_trial.copy()
                            a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            continue
        if rnd > 100:
            break

    return best_a, best_score


def _spectral_flat(a_init, deadline, rng, max_iters=50000):

        a = _simplex_project(a_new)

def _upsample_smooth(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

    p3_end = t_start + (deadline - t_start) * 0.48
    print(f"\nPhase 3: Soft Nesterov (deadline +{p3_end-t_start:.0f}s)")
    a, sc = _soft_nesterov(best_seq, p3_end, rng, lr_init=0.008)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  Phase 3 result: {best_score:.10f}")
    else:
        print(f"  Phase 3 no improvement: {sc:.10f}")

    # --- Phase 4: L-BFGS fine-tuning (12% time) ---
    p4_end = t_start + (deadline - t_start) * 0.60
    print(f"\nPhase 4: L-BFGS fine (deadline +{p4_end-t_start:.0f}s)")
    a, sc = _run_lbfgs(best_seq, p4_end, n_betas=15)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  Phase 4 result: {best_score:.10f}")
    else:
        print(f"  Phase 4 no improvement: {sc:.10f}")

    # --- Phase 5: Peak surgery (10% time) ---
    p5_end = t_start + (deadline - t_start) * 0.70
    print(f"\nPhase 5: Peak surgery (deadline +{p5_end-t_start:.0f}s)")
    a, sc = _peak_surgery(best_seq, p5_end, rng, max_rounds=10000)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  Phase 5 result: {best_score:.10f}")
    else:
        print(f"  Phase 5 no improvement: {sc:.10f}")

    # --- Phase 6: Multi-restart (20% time) ---
    p6_end = t_start + (deadline - t_start) * 0.90
    print(f"\nPhase 6: Multi-restart (deadline +{p6_end-t_start:.0f}s)")
    n_best = len(best_seq)
    time_avail = p6_end - time.monotonic()
    n_restarts = 10
    time_per = max(5.0, time_avail / n_restarts)

    for ri in range(n_restarts):
        if time.monotonic() > p6_end - 10:
            break
        rd = min(time.monotonic() + time_per, p6_end - 5)
        noise_scale = 0.02 + 0.01 * ri
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        z = z + noise_scale * rng.normal(size=n_best)
        z = np.clip(z, -50, 50)
        z = z - z.mean()
        a_pert = _softmax_fn(z)
        a_r, sc_r = _nesterov_gd(a_pert, rd, rng, lr_init=0.008, top_k=2)
        if sc_r < best_score:
            best_score = sc_r
            best_seq = a_r.copy()
            print(f"  Restart {ri}: {best_score:.10f}")
        elif ri % 3 == 0:
            a_r2, sc_r2 = _run_lbfgs(a_r, min(rd + time_per * 0.5, p6_end - 5), n_betas=5)
            if sc_r2 < best_score:
                best_score = sc_r2
                best_seq = a_r2.copy()
                print(f"  Restart {ri}+lbfgs: {best_score:.10f}")

    # --- Phase 7: Final Nesterov + surgery polish (10% time) ---
    p7_end = deadline
    print(f"\nPhase 7: Final polish (deadline +{p7_end-t_start:.0f}s)")
    a, sc = _nesterov_gd(best_seq, min(deadline - 30, time.monotonic() + 200), rng, lr_init=0.003, top_k=2)
    if sc < best_score:
        best_score = sc
        best_seq = a.copy()
        print(f"  Final nesterov: {best_score:.10f}")

    if time.monotonic() < deadline - 15:
        a, sc = _peak_surgery(best_seq, deadline - 5, rng, max_rounds=5000)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  Final surgery: {best_score:.10f}")

    # --- Final validation ---
    print(f"\nTotal time: {time.monotonic() - t_start:.1f}s")
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        n = len(final_seq)
        c = _full_convolve(final_seq, final_seq)
        final_score = 2.0 * n * c.max() / (final_seq.sum() ** 2)
        print(f"FINAL: score={final_score:.10f}, n={n}")
        return [float(x) for x in final_seq]

    # Fallback
    n0 = 256
    t_arr = np.linspace(0, 1, n0)
    fb = 1.0 + 2.0 * t_arr
    fb = fb / fb.sum()
    print(f"FALLBACK: n={n0}")
    return [float(x) for x in fb]
