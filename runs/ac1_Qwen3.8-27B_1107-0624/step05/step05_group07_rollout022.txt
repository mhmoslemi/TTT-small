```python
import numpy as np
import time

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * c.max()

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return _score_normalized(a / s)

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x : sum(x) = target_sum, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    # Scale to make the problem standard (sum = 1)
    v_scaled = v / target_sum if target_sum > 0 else v
    # Standard projection onto sum=1, x>=0
    u = np.sort(v_scaled)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    result = np.maximum(v_scaled - theta, 0)
    return result * target_sum
```

```python
"""
Hybrid optimization for minimizing 2*n*max(self_convolution)/(sum)^2.
Uses: (1) L-BFGS with very fine log-sum-exp beta continuation,
(2) Nesterov-accelerated subgradient descent with exact max,
(3) Pair coordinate descent, (4) Multi-scale interpolation.
Key improvements over previous: tighter beta schedule, Nesterov momentum,
pair-wise updates, and adaptive step-size control.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score(a):
    """Full score for arbitrary a."""
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return _score_norm(a / s)


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_objective(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_gradient(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -100, 100)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # h[i] = sum_j a_j * w[i+j]  (cross-correlation)
    h = np.correlate(w, a, mode='valid')
    grad = (4.0 * n / beta) * a * (h - W)
    return grad


def _project_simplex(v, target=1.0):
    """Project v onto {x: sum(x)=target, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v / target)[::-1]
    cssv = np.cumsum(u)
    rho_mask = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(rho_mask):
        return np.full(n, target / n)
    rho = np.where(rho_mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v / target - theta, 0.0)
    return result * target


def _nesterov_subgrad(a_init, deadline, rng, max_iter=200000,
                      sharp_temp=0.005, init_step=None):
    """Nesterov-accelerated subgradient on the simplex with exact max."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_norm(a)

    if init_step is None:
        step = 2.0 / np.sqrt(n)

    y = a.copy()
    t_k = 1.0
    v = np.zeros(n)

    patience = 0
    max_patience = 5000

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Compute subgradient at y
        c = _full_convolve(y, y)
        cm = float(c.max())

        # Sharp softmax weights
        vals = (c - cm) / sharp_temp
        vals = np.clip(vals, -80, 80)
        w = np.exp(vals)
        w = w / w.sum()

        # Gradient in a-space: grad_i = 2 * sum_j a_j * w[i+j]
        h = np.correlate(w, y, mode='valid')
        W = float(np.dot(w, c))
        grad_a = 2.0 * y * (h - W)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new

        # Step
        step_k = step / t_k
        x_new = y - step_k * grad_a
        x_new = _project_simplex(x_new, 1.0)

        # Check improvement
        c_new = _full_convolve(x_new, x_new)
        score_new = 2.0 * n * float(c_new.max())

        if score_new < best_score:
            best_score = score_new
            best_a = x_new.copy()
            patience = 0
        else:
            patience += 1

        # Update Nesterov iterate
        y = x_new + alpha * (x_new - y) if it > 0 else x_new
        y = _project_simplex(y, 1.0)
        t_k = t_new

        # Adapt step size
        if patience > max_patience:
            step *= 0.8
            patience = 0
            # Reset momentum
            y = x_new.copy()
            t_k = 1.0

        if it % 5000 == 0:
            print(f"    Nesterov iter {it}: score={score_new:.10f}, best={best_score:.10f}, step={step_k:.6f}")

    return best_a, best_score

        cm = float(c.max())
        improved = False

        # Neighboring pairs
        for i in range(0, n - 1, 4):
            pairs.append((i, i + 1))
        # Random pairs
        n_rand = n // 8
        for _ in range(n_rand):
            i, j = rng.integers(0, n, size=2)
            if i != j:
                pairs.append((int(i), int(j)))

        rng.shuffle(pairs)

        for (i, j) in pairs:
            if time.monotonic() > deadline:
                break
            base_score = 2.0 * n * float(c.max())
            best_local = base_score
            best_deltas = (0.0, 0.0)

            # Try several delta combinations
            for di in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05]:
                for dj in [-0.05, -0.02, -0.01, 0.0, 0.01, 0.02, 0.05]:
                    a_trial = a.copy()
                    a_trial[i] = max(0.0, a_trial[i] + di * a.sum())
                    a_trial[j] = max(0.0, a_trial[j] + dj * a.sum())
                    s = a_trial.sum()
                    if s < 0.01:
                        continue
                    a_trial = a_trial / s
                    c_trial = _full_convolve(a_trial, a_trial)
                    sc = 2.0 * n * float(c_trial.max())
                    if sc < best_local:
                        best_local = sc
                        best_deltas = (di, dj)

            if best_deltas != (0.0, 0.0):
                a[i] = max(0.0, a[i] + best_deltas[0])
                a[j] = max(0.0, a[j] + best_deltas[1])
                s = a.sum()
                if s < 0.01:
                    break
                a = a / s
                improved = True
                c = _full_convolve(a, a)
                if 2.0 * n * float(c.max()) < best_score:
                    best_score = 2.0 * n * float(c.max())
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def _single_coord_descent(a_init, deadline, max_rounds=200):
    """Single coordinate descent."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        cm = float(c.max())
        improved = False

        for j in range(n):
            if time.monotonic() > deadline:
                break
            base_sc = 2.0 * n * cm
            best_local = base_sc
            best_delta = 0.0
            for delta in [-0.03, -0.01, -0.005, 0.005, 0.01, 0.03]:
                a[j] = max(0.0, a[j] + delta)
                s = a.sum()
                if s < 0.01:
                    a[j] -= delta
                    break
                a_norm = a / s
                c_trial = _full_convolve(a_norm, a_norm)
                sc = 2.0 * n * float(c_trial.max())
                if sc < best_local:
                    best_local = sc
                    best_delta = delta
                a[j] -= delta

            if best_delta != 0.0:
                a[j] = max(0.0, a[j] + best_delta)
                a = a / a.sum()
                c = _full_convolve(a, a)
                improved = True
                if 2.0 * n * c.max() < best_score:
                    best_score = 2.0 * n * float(c.max())
                    best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _smooth_and_refine(a_init, n_smooth, deadline):
    """Interpolate down to n_smooth, then back up, to smooth the sequence."""
    n = len(a_init)
    a_small = _interp_profile(a_init, n_smooth)
    a_back = _interp_profile(a_small, n)
    # Quick L-BFGS polish
    z = np.log(np.maximum(a_back, 1e-15))
    z = z - z.mean()
    c = _full_convolve(_softmax_fn(z), _softmax_fn(z))
    max_c = float(c.max())
    beta = 0.005 * max_c
    n_curr = n
    try:
        result = minimize(
            _lse_objective, z, args=(beta, n_curr),
            method='L-BFGS-B', jac=_lse_gradient,
            options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-15, 'maxls': 100}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            return a_new
    except Exception:
        pass
    return a_back


def _run_lbfgs_fine(a_init, deadline, n_restarts=5):
    """L-BFGS with very fine beta continuation and multiple restarts."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Very fine beta schedule
    beta_fracs = [0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006, 0.005,
                  0.004, 0.003, 0.002, 0.0015, 0.001, 0.0008, 0.0006,
                  0.0005, 0.0004, 0.0003, 0.0002, 0.00015, 0.0001,
                  0.00008, 0.00006, 0.00004, 0.00003, 0.00002,
                  0.00001, 0.000005]
    beta_list = [b * max_c0 for b in beta_fracs]

    for r in range(n_restarts):
        if time.monotonic() > deadline:
            break

        if r == 0:
            a_cur = best_a.copy()
        else:
            z_base = np.log(np.maximum(best_a, 1e-15))
            z_base = z_base - z_base.mean()
            noise_level = 0.02 * (r + 1)
            z_noisy = z_base + noise_level * np.random.default_rng(1234 + r * 7).standard_normal(n)
            z_noisy = np.clip(z_noisy, -50, 50)
            z_noisy = z_noisy - z_noisy.mean()
            a_cur = _softmax_fn(z_noisy)

        z = np.log(np.maximum(a_cur, 1e-15))
        z = z - z.mean()

        # Track best within this restart
        restart_best_a = a_cur.copy()
        restart_best_score = _score_norm(a_cur)

        n_betas = len(beta_list)
        for idx, beta in enumerate(beta_list):
            if time.monotonic() > deadline:
                break
            remaining = n_betas - idx
            time_per = (deadline - time.monotonic()) / max(1, remaining)
            time_per = min(time_per, 120.0)
            maxiter = max(200, int(time_per * 40))

            try:
                result = minimize(
                    _lse_objective, z, args=(beta, n),
                    method='L-BFGS-B', jac=_lse_gradient,
                    options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-16,
                             'maxcor': 50, 'maxls': 200, 'maxfun': maxiter * 8}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_new = _softmax_fn(result.x)
                    s_new = _score_norm(a_new)
                    if np.isfinite(s_new) and s_new < restart_best_score:
                        restart_best_score = s_new
                        restart_best_a = a_new.copy()
                        z = result.x.copy()
            except Exception:
                pass

        # Compare with global best
        if restart_best_score < best_score:
            best_score = restart_best_score
            best_a = restart_best_a.copy()
            print(f"    L-BFGS restart {r}: new best={best_score:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """Main search function for the best step function coefficients."""
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 2048
        t = np.linspace(0, 1, n_init)
        parent = np.ones(n_init) / n_init
        update_best(parent, _score_norm(parent), "uniform")

    n_parent = len(parent)
    print(f"Starting search with n={n_parent}, initial score={best_score:.10f}")

    # Check if parent is at a different n than expected
    n_work = n_parent
    a_work = parent.copy()

    # ============================================================
    # Phase 1: L-BFGS with fine beta continuation (50% of budget)
    # ============================================================
    phase1_end = t_start + 0.45 * (deadline - t_start)
    print(f"\n=== Phase 1: L-BFGS fine continuation (n={n_work}) ===")

    a_l, sc_l = _run_lbfgs_fine(a_work, phase1_end, n_restarts=4)
    update_best(a_l, sc_l, "lbfgs_phase1")

    # ============================================================
    # Phase 2: Nesterov subgradient (20% of budget)
    # ============================================================
    phase2_end = t_start + 0.65 * (deadline - t_start)
    print(f"\n=== Phase 2: Nesterov subgradient (n={len(best_seq)}) ===")

    a_n, sc_n = _nesterov_subgrad(best_seq, phase2_end, rng, max_iter=300000,
                                   sharp_temp=0.003, init_step=3.0 / np.sqrt(len(best_seq)))
    update_best(a_n, sc_n, "nesterov")

    # ============================================================
    # Phase 3: Pair coordinate descent (15% of budget)
    # ============================================================
    phase3_end = t_start + 0.80 * (deadline - t_start)
    print(f"\n=== Phase 3: Pair coordinate descent ===")

    a_p, sc_p = _coord_descent_pairs(best_seq, phase3_end, rng, max_rounds=50)
    update_best(a_p, sc_p, "pair_cd")

    # ============================================================
    # Phase 4: Smooth + refine + final L-BFGS (15% of budget)
    # ============================================================
    phase4_end = deadline
    print(f"\n=== Phase 4: Smooth + final refinement ===")

    # Try smoothing to smaller n and back
    n_smooth = max(1024, n_work // 4)
    if time.monotonic() < phase4_end - 60:
        a_sm = _smooth_and_refine(best_seq, n_smooth, phase4_end)
        sc_sm = _score_norm(a_sm)
        update_best(a_sm, sc_sm, "smooth_refine")

    # Try single coordinate descent
    if time.monotonic() < phase4_end - 30:
        sub_deadline = min(phase4_end - 10, time.monotonic() + 60.0)
        a_cd, sc_cd = _single_coord_descent(best_seq, sub_deadline, max_rounds=100)
        update_best(a_cd, sc_cd, "single_cd")

    # Final tiny-beta L-BFGS
    if time.monotonic() < phase4_end - 15 and best_seq is not None:
        n_f = len(best_seq)
        c_f = _full_convolve(best_seq, best_seq)
        max_cf = float(c_f.max())
        beta_ultra = 1e-6 * max_cf

        z_f = np.log(np.maximum(best_seq, 1e-15))
        z_f = z_f - z_f.mean()
        sub_dl = min(phase4_end - 5, time.monotonic() + 40.0)
        try:
            result = minimize(
                _lse_objective, z_f, args=(beta_ultra, n_f),
                method='L-BFGS-B', jac=_lse_gradient,
                options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-18,
                         'maxcor': 80, 'maxls': 200, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_final_try = _softmax_fn(result.x)
                sc_final_try = _score_norm(a_final_try)
                update_best(a_final_try, sc_final_try, "ultra_beta_lbfgs")
        except Exception:
            pass

    # ============================================================
    # Phase 5: Try different n values (if time allows)
    # ============================================================
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Phase 5: Multi-n exploration ===")
        for n_try in [2048, 4096, 8192]:
            if time.monotonic() > deadline - 15:
                break
            if n_try == len(best_seq):
                continue
            a_try = _interp_profile(best_seq, n_try)
            sc_try = _score_norm(a_try)
            print(f"    n={n_try}: score={sc_try:.10f}")
            if sc_try < best_score:
                update_best(a_try, sc_try, f"interp_n{n_try}")
                # Quick polish at this n
                if time.monotonic() < deadline - 20:
                    z_t = np.log(np.maximum(a_try, 1e-15))
                    z_t = z_t - z_t.mean()
                    c_t = _full_convolve(a_try, a_try)
                    b_t = 0.001 * float(c_t.max())
                    try:
                        res = minimize(
                            _lse_objective, z_t, args=(b_t, n_try),
                            method='L-BFGS-B', jac=_lse_gradient,
                            options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-15,
                                     'maxcor': 30, 'maxls': 100}
                        )
                        if res.x is not None and np.all(np.isfinite(res.x)):
                            a_pol = _softmax_fn(res.x)
                            sc_pol = _score_norm(a_pol)
                            update_best(a_pol, sc_pol, f"polish_n{n_try}")
                    except Exception:
                        pass

    # ============================================================
    # Final validation and return
    # ============================================================
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score(final_seq)
        print(f"\n{'='*60}")
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        print(f"{'='*60}")

        # Validate with evaluator if possible
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
