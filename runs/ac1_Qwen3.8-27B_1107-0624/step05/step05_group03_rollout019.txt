import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autoconvolve(a):
    """Compute a * a (autocorrelation-like, but linear convolution)."""
    return _full_convolve(a, a)


def _score_norm(a):
    """Score = 2n * max(conv) assuming sum(a)=1."""
    n = len(a)
    c = _autoconvolve(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Full score computation."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autoconvolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    t = np.linspace(0.0, 1.0, n)

    a = (1.0 - np.abs(2.0*t - 1.0)) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        beta_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    beta_list = [b * max_c0 for b in beta_list]
    
    if max_wall is not None:
        per_level = max(2.0, max_wall / len(beta_list))
    else:
        per_level = 30.0
    
    for beta in beta_list:
        if time.monotonic() > deadline - 2.0:
            break
        
        def obj_fn(z_):
            a_ = _softmax(z_)
            c_ = _autoconvolve(a_)
            cm_ = float(c_.max())
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s_))
        
        def grad_fn(z_):
            a_ = _softmax(z_)
            c_ = _autoconvolve(a_)
            cm_ = float(c_.max())
            shifted = (c_ - cm_) / beta
            e_ = np.exp(shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            # Gradient of sum_k w[k] * b[k] w.r.t. a[j]
            # = 2 * sum_k w[k] * a[k-j] * I(0<=k-j<n)
            # Build this efficiently
            grad_a = np.zeros(n)
            for k_idx in range(len(c_)):
                if w[k_idx] < 1e-15:
                    continue
                j_start = max(0, k_idx - (n - 1))
                j_end = min(n - 1, k_idx)
                # a[k-j] for j in [j_start, j_end]
                # k-j ranges from k-j_start to k-j_end
                partner_idx = k_idx - np.arange(j_start, j_end + 1)
                grad_a[j_start:j_end+1] += 2.0 * w[k_idx] * a_[partner_idx]
            
            # Chain rule: z -> a via softmax
            # da/dz_j = a_j * (delta(j,l) - a_l)
            # d_obj/dz_j = sum_l (d_obj/da_l) * (a_j*delta(j,l) - a_j*a_l)
            #            = a_j * (d_obj/da_j - sum_l a_l * d_obj/da_l)
            #            = a_j * (grad_a[j] - dot(a, grad_a))
            W = float(np.dot(a_, grad_a))
            grad_z = a_ * (grad_a - W)
            
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_)
            return grad_z
        
        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 60, 'maxfun': 50000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _subgradient_fw(a_init, deadline, max_iters=50000):
    """
    Frank-Wolfe method with away steps for min max(conv(a,a)) s.t. a>=0, sum(a)=1.
    Uses exact subgradient (uniform over active peaks).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        
        c = _autoconvolve(a)
        max_c = float(c.max())
        current_score = 2.0 * n * max_c
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Find active set: lags within 1% of max
        threshold = max_c * 0.999
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Compute subgradient: g[j] = 2/|S| * sum_{k in S} a[k-j] * I(0<=k-j<n)
        g = np.zeros(n)
        for k_idx in active:
            k_val = int(k_idx)
            j_start = max(0, k_val - (n - 1))
            j_end = min(n - 1, k_val)
            partner_idx = k_val - np.arange(j_start, j_end + 1)
            g[j_start:j_end+1] += 2.0 * a[partner_idx]
        g /= len(active)
        
        # Frank-Wolfe step: move towards argmin(g)
        # But use away-steps: also move away from argmax(g)
        j_min = int(np.argmin(g))
        j_max = int(np.argmax(g))
        
        if j_min == j_max:
            # All same, try random perturbation
            break
        
        # Step size: diminishing
        step = 2.0 / (float(it) + 2.0)
        step = min(step, 0.1)
        
        # Pairwise FW: move mass from j_max to j_min
        amt = step * a[j_max]
        if amt > 0:
            a[j_max] -= amt
            a[j_min] += amt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
    
    return best_a, best_score


def _peak_targeted_local_search(a_init, deadline, n_trials=100000, use_annealing=True):
    """
    Intensive local search that targets convolution peaks.
    Uses simulated annealing to escape local minima.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Temperature for simulated annealing
    T = 0.001
    T_min = 1e-8
    cooling = 0.9995
    
    c = _autoconvolve(a)
    max_c = float(c.max())
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        # Update temperature
        if use_annealing:
            T = max(T_min, T * cooling)
        
        # Strategy selection
        r = np.random.random()
        
        if r < 0.4:
            # Strategy A: Reduce a peak-contributing element
            # Find the max lag
            max_lag = int(np.argmax(c))
            
            # Find the pair with largest product at this lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                top_idx = pairs[int(np.argmax(products))]
                
                # Reduce this element
                frac = np.random.uniform(0.01, 0.1)
                delta = a[top_idx] * frac
                a_trial = a.copy()
                a_trial[top_idx] = max(0.0, a_trial[top_idx] - delta)
                
                # Add to a safe position (low gradient)
                # Compute "danger" score for each position
                danger = np.zeros(n)
                for k in [max_lag]:
                    js = max(0, k - (n-1))
                    je = min(n-1, k)
                    pidx = k - np.arange(js, je+1)
                    danger[js:je+1] += a[pidx]
                
                # Pick from lowest danger positions
                safe = np.argsort(danger)[:max(1, n // 10)]
                add_idx = int(safe[np.random.randint(len(safe))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm(a_trial)
                    
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        c = _autoconvolve(a)
                    elif use_annealing and np.random.random() < np.exp(-(sc - best_score) / max(T, 1e-15)):
                        a = a_trial.copy()
                        c = _autoconvolve(a)
        
        elif r < 0.7:
            # Strategy B: Pairwise mass transfer
            i1 = int(np.random.randint(n))
            i2 = int(np.random.randint(n))
            if i1 == i2:
                continue
            frac = np.random.uniform(0.01, 0.15)
            amt = min(a[i1], a[i2]) * frac
            a_trial = a.copy()
            if np.random.random() < 0.5:
                a_trial[i1] -= amt
                a_trial[i2] += amt
            else:
                a_trial[i1] += amt
                a_trial[i2] -= amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    c = _autoconvolve(a)
                elif use_annealing and np.random.random() < np.exp(-(sc - best_score) / max(T, 1e-15)):
                    a = a_trial.copy()
                    c = _autoconvolve(a)
        
        elif r < 0.85:
            # Strategy C: Block multiplicative perturbation
            block_size = int(np.random.randint(4, min(32, n // 16)))
            start = int(np.random.randint(0, max(1, n - block_size)))
            mult = np.random.uniform(0.85, 1.15, size=block_size)
            a_trial = a.copy()
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                c = _autoconvolve(a)
            
        else:
            # Strategy D: Top-K peak reduction (reduce multiple peaks simultaneously)
            K = int(np.random.randint(2, 6))
            top_lags = np.argsort(c)[-K:]
            a_trial = a.copy()
            total_delta = 0.0
            safe_indices = []
            
            for k_val in top_lags:
                k_val = int(k_val)
                i_min = max(0, k_val - (n - 1))
                i_max = min(n - 1, k_val)
                if i_min <= i_max:
                    pairs = np.arange(i_min, i_max + 1)
                    partners = k_val - pairs
                    products = a_trial[pairs] * a_trial[partners]
                    idx = pairs[int(np.argmax(products))]
                    frac = np.random.uniform(0.005, 0.05)
                    delta = a_trial[idx] * frac
                    a_trial[idx] = max(0.0, a_trial[idx] - delta)
                    total_delta += delta
                    safe_indices.append(idx)
            
            if total_delta > 0:
                # Distribute mass to safe positions
                # Find positions with low contribution to all top lags
                danger = np.zeros(n)
                for k_val in top_lags:
                    k_val = int(k_val)
                    js = max(0, k_val - (n-1))
                    je = min(n-1, k_val)
                    pidx = k_val - np.arange(js, je+1)
                    danger[js:je+1] += a_trial[pidx]
                
                # Pick safe positions
                safe = np.argsort(danger)[:max(1, n // 20)]
                # Distribute
                per_add = total_delta / len(safe[:min(len(safe), max(1, total_delta/a_trial[safe].max() if len(safe)>0 else 1))])
                for idx in safe[:min(len(safe), 10)]:
                    a_trial[idx] += per_add
                
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        c = _autoconvolve(a)
                    elif use_annealing and np.random.random() < np.exp(-(sc - best_score) / max(T, 1e-15)):
                        a = a_trial.copy()
                        c = _autoconvolve(a)
    
    return best_a, best_score


def _fine_adam(a_init, deadline, lr=0.005, k_top=3, max_iters=100000):
    """Adam optimizer with top-K peak awareness."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-10
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        
        t += 1
        a_curr = _softmax(z)
        c = _autoconvolve(a_curr)
        
        # Top-K gradient
        K = min(k_top, len(c))
        top_idx = np.argsort(c)[-K:]
        w = np.zeros(len(c))
        for idx in top_idx:
            w[idx] = 1.0 / K
        
        # Compute gradient w.r.t. a
        grad_a = np.zeros(n)
        for k_idx in top_idx:
            k_val = int(k_idx)
            js = max(0, k_val - (n-1))
            je = min(n-1, k_val)
            pidx = k_val - np.arange(js, je+1)
            grad_a[js:je+1] += 2.0 * w[k_idx] * a_curr[pidx]
        
        # Chain rule through softmax
        W = float(np.dot(a_curr, grad_a))
        grad_z = a_curr * (grad_a - W)
        
        if not np.all(np.isfinite(grad_z)):
            grad_z = np.zeros_like(z)
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        lr_t = lr * (1.0 - 0.1 * float(t) / max(1, max_iters))
        z = z - lr_t * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if t % 10 == 0:
            sc = _score_norm(a_curr)  # Actually should recompute
            a_check = _softmax(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
    
    return best_a, best_score

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
1. Start from parent construction
2. L-BFGS with logsumexp continuation for basin finding
3. Subgradient Frank-Wolfe with away-steps for direct max-conv minimization
4. Intensive peak-targeted local search with simulated annealing
5. Coordinate-descent refinement
6. Final Adam polishing with diminishing learning rate
Key improvements: better gradient computation, away-step FW, coordinate descent,
and more aggressive local search with annealing to escape local minima.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _autoconv(a):
    return _full_convolve(a, a)


def _score_norm(a):
    n = len(a)
    c = _autoconv(a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _autoconv(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cos(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(np.pi * alpha * (2.0 * t - 1.0)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


class _LbfgsProblem:
    """Stores parameters for L-BFGS logsumexp objective."""
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta

    def obj(self, z):
        a = _softmax_fn(z)
        c = _autoconv(a)
        cm = float(c.max())
        shifted = (c - cm) / self.beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            return 2.0 * self.n * cm
        return 2.0 * self.n * (cm + self.beta * np.log(s))

    def grad(self, z):
        a = _softmax_fn(z)
        c = _autoconv(a)
        cm = float(c.max())
        shifted = (c - cm) / self.beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        grad_a = np.zeros(self.n)
        for k_idx in range(len(c)):
            wk = w[k_idx]
            if wk < 1e-15:
                continue
            js = max(0, k_idx - (self.n - 1))
            je = min(self.n - 1, k_idx)
            pidx = k_idx - np.arange(js, je + 1)
            grad_a[js:je + 1] += 2.0 * wk * a[pidx]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            return np.zeros_like(z)
        return grad_z


def _lbfgs_logsumexp(a_init, deadline, beta_list, max_wall=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    c0 = _autoconv(a_init)
    max_c0 = float(c0.max())
    if max_wall is not None:
        per_level = max(2.0, max_wall / len(beta_list))
    else:
        per_level = 30.0
    for frac in beta_list:
        if time.monotonic() > deadline - 2.0:
            break
        beta = frac * max_c0
        prob = _LbfgsProblem(n, beta)
        try:
            result = minimize(
                prob.obj, z.copy(), jac=prob.grad,
                method='L-BFGS-B',
                options={'maxiter': 8000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 60, 'maxfun': 80000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
                    c0 = _autoconv(a_new)
                    max_c0 = float(c0.max())
        except Exception:
            pass
    return best_a, best_score


def _compute_subgrad(a, c, threshold_frac=0.999):
    """Compute subgradient of max(conv) w.r.t. a, uniform over active peaks."""
    n = len(a)
    max_c = float(c.max())
    threshold = max_c * threshold_frac
    active = np.where(c >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    g = np.zeros(n)
    for k_idx in active:
        k_val = int(k_idx)
        js = max(0, k_val - (n - 1))
        je = min(n - 1, k_val)
        pidx = k_val - np.arange(js, je + 1)
        g[js:je + 1] += 2.0 * a[pidx]
    g /= len(active)
    return g


def _subgrad_fw_away(a_init, deadline, max_iters=200000):
    """Frank-Wolfe with away-steps for min max(conv(a,a)) s.t. simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_norm(a)
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break
        c = _autoconv(a)
        sc = 2.0 * n * float(c.max())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        g = _compute_subgrad(a, c)
        j_min = int(np.argmin(g))
        j_max_away = int(np.argmax(g))
        if j_min == j_max_away:
            break
        step = 2.0 / (float(it) + 2.0)
        step = min(step, 0.05)
        amt = step * a[j_max_away]
        if amt > 1e-15:
            a[j_max_away] -= amt
            a[j_min] += amt
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_passes=200):
    """Coordinate descent: for each j, optimize a[j] given others fixed."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    for path in range(max_passes):
        if time.monotonic() > deadline - 2.0:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline - 3.0:
                break
            a_save = a[j]
            # Try several values of a[j]
            # The baseline (with a[j]=0) gives a certain max_conv
            # We want to find a[j] that minimizes max_conv
            # b[k] as function of a[j]: b[k] = b_base[k] + a[j]*(a[k-j] + a[j-k])
            # where b_base is conv with a[j]=0
            # We want max_k [b_base[k] + a[j] * delta_k] to be minimized
            # where delta_k = a[k-j] + a[j-k] (with appropriate zero-padding)
            # This is a 1D convex piecewise-linear minimization
            old_val = a[j]
            a[j] = 0.0
            c_base = _autoconv(a)
            # delta_k = a[k-j] + a[j-k] for valid indices
            # Actually when a[j]=0, the effect of restoring a[j]=v is:
            # b[k] += v * a[k-j] + v * a[j-k] (for valid k-j, j-k in [0,n))
            # But a[j]=0 now, so a[j-k]=0 when j-k=j, i.e., k=0... 
            # Let me just compute the effect differently.
            # With a[j] = v, b[k] = sum_i a[i]*a[k-i]
            # The terms involving a[j] are:
            #   i=j: a[j]*a[k-j] = v*a[k-j] (if 0<=k-j<n and k-j != j)
            #   k-i=j: a[k-j]*a[j] = a[k-j]*v (if 0<=k-j<n and k-j != j)
            #   i=j and k-i=j (k=2j): a[j]*a[j] = v^2
            # So b[k] = c_base[k] + v * (a[k-j] + a[j-k]) * I(valid) + v^2 * I(k==2j)
            # where a[j]=0 in c_base
            # Let d[k] = a[k-j]*I(0<=k-j<n, k-j!=j) + a[j-k]*I(0<=j-k<n, j-k!=j)
            # And e[k] = I(k==2j)
            # b[k] = c_base[k] + v*d[k] + v^2*e[k]
            # maximize over k: c_base[k] + v*d[k] + v^2*e[k]
            # For large k (bulk), e[k]=0, so it's linear in v: c_base[k] + v*d[k]
            # The max over k of (c_base[k] + v*d[k]) is convex in v
            # Minimum is where the "upper envelope" is lowest
            
            # Simpler approach: just try a few values
            candidates = [0.0, old_val * 0.5, old_val * 0.8, old_val, old_val * 1.2, old_val * 1.5]
            candidates.append(np.mean(a) * 2.0)
            
            best_v = old_val
            best_local_sc = np.inf
            for v in candidates:
                v = max(0.0, min(1.0, v))
                a[j] = v
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_val
                    break
                a_norm = a / s
                c_try = _autoconv(a_norm)
                sc_try = 2.0 * n * float(c_try.max())
                if sc_try < best_local_sc:
                    best_local_sc = sc_try
                    best_v = v
            a[j] = best_v
            if best_v != old_val:
                improved = True
                a = a / a.sum()
        
        sc_now = _score_norm(a)
        if sc_now < best_score:
            best_score = sc_now
            best_a = a.copy()
        
        if not improved:
            break
    return best_a, best_score


class _AdamProblem:
    def __init__(self, n, k_top):
        self.n = n
        self.k_top = k_top

    def grad(self, z):
        a = _softmax_fn(z)
        c = _autoconv(a)
        K = min(self.k_top, len(c))
        top_idx = np.argsort(c)[-K:]
        w = np.zeros(len(c))
        for idx in top_idx:
            w[idx] = 1.0 / K
        grad_a = np.zeros(self.n)
        for k_idx in top_idx:
            k_val = int(k_idx)
            js = max(0, k_val - (self.n - 1))
            je = min(self.n - 1, k_val)
            pidx = k_val - np.arange(js, je + 1)
            grad_a[js:je + 1] += 2.0 * w[k_idx] * a[pidx]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            return np.zeros_like(z)
        return grad_z


def _adam_refine(a_init, deadline, lr, k_top, max_iters=200000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-10
    prob = _AdamProblem(n, k_top)
    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        t += 1
        gz = prob.grad(z)
        m = b1 * m + (1 - b1) * gz
        v = b2 * v + (1 - b2) * (gz * gz)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        decay = max(0.1, 1.0 - 0.5 * float(t) / max(1, max_iters))
        lr_t = lr * decay
        z = z - lr_t * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        if t % 20 == 0:
            a_check = _softmax_fn(z)
            sc = _score_norm(a_check)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()
    return best_a, best_score


def _peak_local_search(a_init, deadline, n_trials, use_annealing, rng):

    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    c = _autoconv(a)
    T = 0.002
    T_min = 1e-10
    cooling = 0.9997
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        if use_annealing:
            T = max(T_min, T * cooling)
        r = float(rng.random())
        if r < 0.35:
            # Reduce top peak contributor
            c_cur = _autoconv(a)
            max_lag = int(np.argmax(c_cur))
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                top_idx = int(pairs[int(np.argmax(products))])
                frac = float(rng.uniform(0.005, 0.08))
                delta = a[top_idx] * frac
                a_trial = a.copy()
                a_trial[top_idx] = max(0.0, a_trial[top_idx] - delta)
                # Find safe position

                danger = np.zeros(n)
                for kv in top_K:
                    kv = int(kv)
                    js = max(0, kv - (n - 1))
                    je = min(n - 1, kv)
                    pidx = kv - np.arange(js, je + 1)
                    danger[js:je + 1] += a_trial[pidx]
                safe = np.argsort(danger)[:max(1, n // 10)]
                add_idx = int(safe[int(rng.integers(len(safe)))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        c = _autoconv(a)
                    elif use_annealing and float(rng.random()) < np.exp(-max(0, (sc - best_score)) / max(T, 1e-15)):
                        a = a_trial.copy()
                        c = _autoconv(a)
        elif r < 0.6:

            if float(rng.random()) < 0.5:

            else:
                a_trial[i2] = max(0.0, a_trial[i2] - amt)
                a_trial[i1] += amt
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    c = _autoconv(a)
                elif use_annealing and float(rng.random()) < np.exp(-max(0, (sc - best_score)) / max(T, 1e-15)):
                    a = a_trial.copy()
                    c = _autoconv(a)
        elif r < 0.8:

            start = int(rng.integers(0, max(1, n - bs)))

            a_trial = a.copy()
            a_trial[start:start + bs] *= mult
            a_trial = np.maximum(a_trial, 1e-15)

            sc = _score_norm(a_trial)

                best_score = sc

                a = a_trial.copy()
                c = _autoconv(a)
        else:

            a_trial = a.copy()
            total_delta = 0.0
            reduced_indices = []
            for kv in top_lags:

                i_min = max(0, kv - (n - 1))
                i_max = min(n - 1, kv)
                if i_min <= i_max:

                    partners = kv - pairs
                    products = a_trial[pairs] * a_trial[partners]
                    idx = int(pairs[int(np.argmax(products))])
                    frac = float(rng.uniform(0.003, 0.04))
                    delta = a_trial[idx] * frac
                    a_trial[idx] = max(0.0, a_trial[idx] - delta)
                    total_delta += delta
                    reduced_indices.append(idx)
            if total_delta > 1e-15:
                danger = np.zeros(n)
                for kv in top_lags:
                    kv = int(kv)
                    js = max(0, kv - (n - 1))
                    je = min(n - 1, kv)
                    pidx = kv - np.arange(js, je + 1)
                    danger[js:je + 1] += a_trial[pidx]
                safe = np.argsort(danger)[:max(1, n // 15)]
                n_add = min(5, len(safe))
                per_add = total_delta / n_add
                for idx in safe[:n_add]:
                    a_trial[idx] += per_add
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        c = _autoconv(a)
                    elif use_annealing and float(rng.random()) < np.exp(-max(0, (sc - best_score)) / max(T, 1e-15)):
                        a = a_trial.copy()
                        c = _autoconv(a)
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):

    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_work = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.10f}")
        if sc_parent < best_score:
            best_score = sc_parent
            best_seq = parent.copy()
            print(f"  SET BEST: {best_score:.10f} [parent]")
    else:
        n_work = 8192
        parent = _make_triangular(n_work)
        best_seq = parent.copy()
        best_score = _score_norm(parent)
        print(f"Triangular init: n={n_work}, score={best_score:.10f}")
    
    print(f"Starting optimization with n={n_work}, score={best_score:.10f}")
    
    def _try_update(p, sc, label):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # === Phase 1: L-BFGS warm-up (150s) ===
    phase1_end = min(deadline, t_start + 150.0)
    print(f"\n[Phase 1] L-BFGS warm-up ({phase1_end - t_start:.0f}s)")
    a1, sc1 = _lbfgs_logsumexp(best_seq, phase1_end,
                                beta_list=[0.1, 0.05, 0.02, 0.01, 0.005, 0.002],
                                max_wall=140.0)
    _try_update(a1, sc1, "lbfgs_warm")
    print(f"  Phase 1 done: {best_score:.10f}")
    
    # === Phase 2: Subgradient FW with away-steps (350s) ===
    phase2_end = min(deadline, t_start + 500.0)
    print(f"\n[Phase 2] Subgradient FW ({phase2_end - t_start:.0f}s)")
    a2, sc2 = _subgrad_fw_away(best_seq, phase2_end, max_iters=500000)
    _try_update(a2, sc2, "fw_away")
    print(f"  Phase 2 done: {best_score:.10f}")
    
    # === Phase 3: Fine L-BFGS (250s) ===
    phase3_end = min(deadline, t_start + 750.0)
    if time.monotonic() < phase3_end:
        print(f"\n[Phase 3] Fine L-BFGS ({phase3_end - t_start:.0f}s)")
        a3, sc3 = _lbfgs_logsumexp(best_seq, phase3_end,
                                    beta_list=[0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002],
                                    max_wall=240.0)
        _try_update(a3, sc3, "lbfgs_fine1")
        if time.monotonic() < phase3_end - 30:
            a3b, sc3b = _lbfgs_logsumexp(best_seq, phase3_end,
                                         beta_list=[0.0001, 0.00005, 0.00002, 0.00001, 0.000005],
                                         max_wall=50.0)
            _try_update(a3b, sc3b, "lbfgs_fine2")
    print(f"  Phase 3 done: {best_score:.10f}")
    
    # === Phase 4: Coordinate descent (200s) ===
    phase4_end = min(deadline, t_start + 950.0)
    if time.monotonic() < phase4_end:
        print(f"\n[Phase 4] Coordinate descent ({phase4_end - t_start:.0f}s)")
        a4, sc4 = _coordinate_descent(best_seq, phase4_end, max_passes=50)
        _try_update(a4, sc4, "coord_desc")
    print(f"  Phase 4 done: {best_score:.10f}")
    
    # === Phase 5: Intensive local search with annealing (350s) ===
    phase5_end = min(deadline, t_start + 1300.0)
    if time.monotonic() < phase5_end:
        print(f"\n[Phase 5] Local search + annealing ({phase5_end - t_start:.0f}s)")
        a5, sc5 = _peak_local_search(best_seq, phase5_end, n_trials=300000,
                                       use_annealing=True, rng=rng)
        _try_update(a5, sc5, "local_anneal")
    print(f"  Phase 5 done: {best_score:.10f}")
    
    # === Phase 6: Adam refinement (150s) ===
    phase6_end = min(deadline, t_start + 1450.0)
    if time.monotonic() < phase6_end:
        print(f"\n[Phase 6] Adam refinement ({phase6_end - t_start:.0f}s)")
        a6, sc6 = _adam_refine(best_seq, phase6_end, lr=0.003, k_top=3, max_iters=200000)
        _try_update(a6, sc6, "adam_k3")
        if time.monotonic() < phase6_end - 20:
            a6b, sc6b = _adam_refine(best_seq, phase6_end, lr=0.002, k_top=5, max_iters=50000)
            _try_update(a6b, sc6b, "adam_k5")
    print(f"  Phase 6 done: {best_score:.10f}")
    
    # === Phase 7: Final polishing (remaining ~350s) ===
    if time.monotonic() < deadline - 30:
        remaining = max(30.0, deadline - time.monotonic())
        print(f"\n[Phase 7] Final polishing ({remaining:.0f}s remaining)")
        
        # 7a: No-annealing local search (very precise)
        if remaining > 80:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.3)
            a7, sc7 = _peak_local_search(best_seq, dl, n_trials=200000,
                                          use_annealing=False, rng=rng)
            _try_update(a7, sc7, "final_local")
        
        # 7b: Very fine L-BFGS
        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.25)
            a7b, sc7b = _lbfgs_logsumexp(best_seq, dl,
                                         beta_list=[0.00005, 0.00002, 0.00001, 0.000005, 0.000002],
                                         max_wall=remaining * 0.25)
            _try_update(a7b, sc7b, "final_lbfgs")
        
        # 7c: Tiny Adam
        if time.monotonic() < deadline - 15:
            dl = min(deadline - 10, time.monotonic() + remaining * 0.2)
            a7c, sc7c = _adam_refine(best_seq, dl, lr=0.0008, k_top=2, max_iters=200000)
            _try_update(a7c, sc7c, "final_adam")
        
        # 7d: One more local search pass
        if time.monotonic() < deadline - 10:
            dl = min(deadline - 5, time.monotonic() + max(20.0, remaining * 0.15))
            a7d, sc7d = _peak_local_search(best_seq, dl, n_trials=100000,
                                            use_annealing=False, rng=rng)
            _try_update(a7d, sc7d, "final_local2")
    
    print(f"\nAll phases done. Best score: {best_score:.10f}")
    
    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        if len(final_seq) <= 16384:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirms: {sc_eval:.10f}")
                else:
                    print(f"WARNING: Evaluator returned {sc_eval}")
            except Exception as ex:
                print(f"Evaluator error: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    fallback = _make_triangular(8192)
    print(f"FALLBACK: n=8192, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
