Hybrid optimization combining LP-based direction finding, L-BFGS with 
log-sum-exp smoothing, and subgradient polishing to minimize
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences of length n.

from scipy.optimize import minimize, linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lbfgs_smo(z0, beta, n, max_iter=3000):
    """Run L-BFGS on the log-sum-exp smoothed objective."""
    def fun(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            return 2.0 * n * c_max
        return 2.0 * n * (c_max + beta * np.log(s))
    
    def jac(z):
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = c.max()
        shifted = (c - c_max) / beta
        e = np.exp(shifted)
        s = e.sum()
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:

        wc = _full_convolve(w, a_rev)
        h = wc[n-1:2*n-1]
        W = float(np.dot(w, c))
        return 4.0 * n * a * (h - W)
    
    result = minimize(fun, z0, jac=jac, method='L-BFGS-B',

def _subgrad_polish(a, deadline, max_steps=5000, lr_init=0.02):
    """Subgradient polishing in a-space with active set."""
    n = len(a)
    a = a / a.sum()
    M = float(_full_convolve(a, a).max())
    best_a = a.copy()
    best_M = M
    lr = lr_init
    
    for i in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = c.max()
        
        # Active set
        tol = 0.01 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Subgradient in a-space
        a_rev = a[::-1]
        wa = _full_convolve(w, a_rev)
        grad = 2.0 * wa[n-1:2*n-1]
        
        # Take step
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        c_new = _full_convolve(a_new, a_new)
        M_new = float(c_new.max())
        
        if M_new < M:
            a = a_new
            M = M_new
            if M < best_M:
                best_M = M
                best_a = a.copy()
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_a

"""
Hybrid LP + L-BFGS + subgradient optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
1. LP-based direction finding: solve max sum(b) s.t. (a*b)_k <= M, b>=0
   to generate descent directions that spread mass and reduce autocorrelation peaks
2. Palindromic averaging to exploit symmetry
3. More aggressive beta continuation with tighter convergence
4. Momentum-based subgradient for faster polishing
5. Multi-restart with diverse perturbations
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smooth_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_phase(z0, beta, n, max_iter=3000):
    """Run L-BFGS-B on the log-sum-exp smoothed objective."""
    result = minimize(
        _smooth_obj, z0, jac=_smooth_grad, method='L-BFGS-B',
        args=(beta, n),
        options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11,
                 'maxcor': 40, 'maxls': 200, 'disp': False}
    )
    return _softmax(result.x)


def _beta_cont(a, deadline, n, min_remaining=120):
    """Full beta-continuation schedule starting from sequence a."""
    c = _full_convolve(a, a)
    max_c = float(c.max())
    if max_c < 1e-15:
        return a

    betas = max_c * np.array([0.5, 0.25, 0.12, 0.06, 0.03, 0.015,
                              0.007, 0.0035, 0.0018, 0.0009, 0.0004,
                              0.0002, 0.0001])
    current_a = a.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_remaining > deadline:
            break
        remaining = deadline - min_remaining - time.monotonic()
        if remaining < 15:
            break
        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()
        est_time = 0.006 * (n / 2048.0)
        max_iter = min(8000, max(100, int(remaining / (est_time * 12))))
        a_new = _lbfgs_phase(z0, float(beta), n, max_iter)
        sc_new = _score(a_new)
        if sc_new < current_sc - 1e-13:
            current_a = a_new
            current_sc = sc_new

    return current_a


def _pal_avg(a):
    """Enforce approximate palindromic symmetry."""
    return 0.5 * (a + a[::-1])


def _lp_solve(a, n):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0. Returns (b_star, M)."""
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-15:
        return a.copy(), M

    # Build constraint matrix A: (2n-1) x n
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    k_idx = np.arange(2 * n - 1)[:, None]
    j_idx = np.arange(n)[None, :]
    diff = k_idx - j_idx
    mask = (diff >= 0) & (diff < n)
    A = a[None, :].copy() * mask
    del k_idx, j_idx, diff, mask

    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)

    # Add upper bounds for safety
    upper = min(M * n + 1.0, 1e6)
    bounds = [(0.0, upper)] * n

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
    except Exception:
        return a.copy(), M

    if result.status != 0 or result.x is None:
        return a.copy(), M

    return result.x, M

    n = len(a)
    a = a / a.sum()
    a = _pal_avg(a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    M = float(_full_convolve(a, a).max())
    best_a = a.copy()
    best_M = M
    stall_count = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if M < 1e-12:

        b_star, M_cur = _lp_solve(a, n)
        solve_time = time.monotonic() - t0

        sum_b = float(b_star.sum())
        if sum_b < 1e-15:
            break

        b_norm = b_star / sum_b
        b_norm = _pal_avg(b_norm)
        b_norm = np.maximum(b_norm, 0.0)
        b_norm = b_norm / b_norm.sum()

        # Multiple line search steps
        improved = False
        for t_candidate in [0.3, 0.5, 0.7, 0.9, 1.0, 0.1, 0.2]:
            a_test = (1.0 - t_candidate) * a + t_candidate * b_norm
            a_test = np.maximum(a_test, 0.0)
            s = a_test.sum()
            if s < 1e-15:
                continue
            a_test = a_test / s
            c_test = _full_convolve(a_test, a_test)
            m_test = float(c_test.max())
            if m_test < M - 1e-12:
                a = a_test
                M = m_test
                improved = True
                stall_count = 0
                break

        if improved:
            if M < best_M:
                best_M = M
                best_a = a.copy()
            if it % 100 == 0:
                print(f"    LP it={it}: M={M:.10f} score={2*n*M:.10f} "
                      f"t={time.monotonic()-time.monotonic()+solve_time:.3f}s "
                      f"elapsed={time.monotonic()-t0:.3f}s")
        else:
            stall_count += 1
            if stall_count > 30:
                break

    return best_a


def _subgrad_momentum(a, deadline, max_steps=20000, lr_init=0.05, momentum=0.5):
    """Subgradient method with momentum and adaptive step size."""
    n = len(a)
    a = a / a.sum()
    M = float(_full_convolve(a, a).max())
    best_a = a.copy()
    best_M = M
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        c_max = c.max()

        # Active set with adaptive tolerance
        tol = 0.02 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()

        # Subgradient
        a_rev = a[::-1]
        wa = _full_convolve(w, a_rev)
        grad = 2.0 * wa[n - 1:2 * n - 1]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad

        # Step
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a_new = _pal_avg(a_new)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        c_new = _full_convolve(a_new, a_new)
        M_new = float(c_new.max())

        if M_new < M - 1e-12:
            a = a_new
            M = M_new
            no_improve = 0
            if M < best_M:
                best_M = M
                best_a = a.copy()
            # Slowly decay momentum
            momentum = max(0.3, momentum * 0.999)
        else:
            lr *= 0.95
            no_improve += 1
            if no_improve > 200 or lr < 1e-12:
                break

        if i % 500 == 0 and i > 0:
            print(f"    SG it={i}: M={M:.10f} lr={lr:.6f} "
                  f"elapsed={time.monotonic()-t0_sg:.1f}s")

    return best_a


def _make_profile(n, kind="tapered"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(4.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "flat_taper":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(0.1 * n)
        for i in range(taper):
            f = 0.5 * (1 + np.cos(np.pi * i / taper))
            a[i] = f
            a[n - 1 - i] = f
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function combining LP direction finding, L-BFGS with
    log-sum-exp smoothing, and momentum subgradient polishing.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf
    n_main = 2048

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        a0 = parent.copy()
    else:
        a0 = _make_profile(n_main, "tapered")

    sc0 = _score(a0)
    best_seq = a0.copy()
    best_score = sc0
    print(f"Start: n={n_main}, score={sc0:.10f} t={time.monotonic()-t_start:.1f}s")

    # --- Phase 1: L-BFGS beta continuation (main optimization) ---
    print(f"\nPhase 1: L-BFGS beta continuation (t={time.monotonic()-t_start:.1f}s)")
    p1_deadline = t_start + 350
    a_p1 = _beta_cont(a0, min(p1_deadline, deadline), n_main, min_remaining=1300)
    sc_p1 = _score(a_p1)
    if sc_p1 < best_score:
        best_score = sc_p1
        best_seq = a_p1.copy()
        print(f"  Phase 1 result: {sc_p1:.10f}")
    else:
        print(f"  Phase 1: no improvement (score={sc_p1:.10f})")

    # --- Phase 2: LP-based refinement ---
    print(f"\nPhase 2: LP refinement (t={time.monotonic()-t_start:.1f}s)")
    p2_deadline = t_start + 850
    a_p2 = _lp_refine(best_seq, min(p2_deadline, deadline), max_iters=3000)
    sc_p2 = _score(a_p2)
    if sc_p2 < best_score:
        best_score = sc_p2
        best_seq = a_p2.copy()
        print(f"  Phase 2 result: {sc_p2:.10f}")
    else:
        print(f"  Phase 2: no improvement (score={sc_p2:.10f})")

    # --- Phase 3: L-BFGS polish after LP ---
    print(f"\nPhase 3: L-BFGS polish (t={time.monotonic()-t_start:.1f}s)")
    p3_deadline = t_start + 1150
    if time.monotonic() < p3_deadline:
        a_p3 = _beta_cont(best_seq, min(p3_deadline, deadline), n_main, min_remaining=500)
        sc_p3 = _score(a_p3)
        if sc_p3 < best_score:
            best_score = sc_p3
            best_seq = a_p3.copy()
            print(f"  Phase 3 result: {sc_p3:.10f}")
        else:
            print(f"  Phase 3: no improvement (score={sc_p3:.10f})")

    # --- Phase 4: Subgradient with momentum ---
    print(f"\nPhase 4: Subgradient momentum (t={time.monotonic()-t_start:.1f}s)")
    p4_deadline = t_start + 1450
    if time.monotonic() < p4_deadline:
        t0_sg = time.monotonic()
        a_p4 = _subgrad_momentum(best_seq, min(p4_deadline, deadline),
                                  max_steps=50000, lr_init=0.03, momentum=0.4)
        sc_p4 = _score(a_p4)
        if sc_p4 < best_score:
            best_score = sc_p4
            best_seq = a_p4.copy()
            print(f"  Phase 4 result: {sc_p4:.10f}")
        else:
            print(f"  Phase 4: no improvement (score={sc_p4:.10f})")

    # --- Phase 5: LP again (might find new directions) ---
    print(f"\nPhase 5: Second LP pass (t={time.monotonic()-t_start:.1f}s)")
    p5_deadline = t_start + 1600
    if time.monotonic() < p5_deadline:
        a_p5 = _lp_refine(best_seq, min(p5_deadline, deadline), max_iters=1000)
        sc_p5 = _score(a_p5)
        if sc_p5 < best_score:
            best_score = sc_p5
            best_seq = a_p5.copy()
            print(f"  Phase 5 result: {sc_p5:.10f}")

    # --- Phase 6: Multi-restart ---
    print(f"\nPhase 6: Multi-restart (t={time.monotonic()-t_start:.1f}s)")
    p6_deadline = deadline - 15
    if time.monotonic() < p6_deadline:
        remaining = p6_deadline - time.monotonic()
        n_restarts = min(3, max(1, int(remaining / 150)))

        for i in range(n_restarts):
            if time.monotonic() > p6_deadline - 10:
                break
            # Perturb
            t_pert = 0.015 * (i + 1)
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z + t_pert * rng.standard_normal(n_main)
            z = z - z.mean()
            p_start = _softmax(z)
            p_start = _pal_avg(p_start)
            p_start = np.maximum(p_start, 0.0)
            p_start = p_start / p_start.sum()

            # Quick optimization
            a_r = _beta_cont(p_start, p6_deadline, n_main, min_remaining=30)
            sc_r = _score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {i}: {sc_r:.10f}")

            # Subgradient polish on result
            if time.monotonic() < p6_deadline - 10:
                a_r2 = _subgrad_momentum(a_r, p6_deadline - 10,
                                          max_steps=5000, lr_init=0.02, momentum=0.3)
                sc_r2 = _score(a_r2)
                if sc_r2 < best_score:
                    best_score = sc_r2
                    best_seq = a_r2.copy()
                    print(f"  Restart {i} sg: {sc_r2:.10f}")

    # --- Phase 7: Final fine-tune ---
    print(f"\nPhase 7: Final tune (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 3 and best_seq is not None:
        # One more subgradient with very small lr
        a_fin = _subgrad_momentum(best_seq, deadline - 3,
                                   max_steps=3000, lr_init=0.005, momentum=0.3)
        sc_fin = _score(a_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_seq = a_fin.copy()
            print(f"  Final tune: {sc_fin:.10f}")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
