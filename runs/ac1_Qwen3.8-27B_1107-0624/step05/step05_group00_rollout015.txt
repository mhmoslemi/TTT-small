"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses KKT-guided updates, FISTA with continuation, Adam in log-space, peak-chipping heuristic,
and multi-length/multi-restart exploration. Key innovation: exploits the fact that for
non-palindromic sequences, the convolution peak need not be at the center, and uses
KKT optimality conditions to guide fine-tuning.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_smoothed(a, c, beta, n):
    """Compute gradient of smoothed max: sum w_k * (a*a)[k] where w = softmax(c/beta)."""
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return _compute_grad_weighted(a, w, n)


def _compute_grad_weighted(a, w, n):
    """Compute gradient of sum_k w[k] * (a*a)[k] w.r.t. a."""
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _compute_grad_z(a, c, beta, n):
    """Gradient in log-space (z = log(a), a = softmax(z))."""
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    grad_sm = _compute_grad_weighted(a, w, n)
    # Convert from gradient w.r.t. a to gradient w.r.t. z:
    # a_i = softmax(z)_i, da_i/dz_j = a_i*(delta_ij - a_j)
    # grad_z_j = sum_i grad_a_i * da_i/dz_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
    #          = a_j * grad_a_j - a_j * sum_i (grad_a_i * a_i)
    W = float(np.dot(grad_sm, a))
    return a * (grad_sm - W)


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 4.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = 2.0 * n * float(c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        grad = _compute_grad_smoothed(y, c_y, beta, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        remain_iter = max_iter_total - t
        if remain_iter <= 0:
            break
        it_in_phase = 0
        while it_in_phase < remain_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            grad = _compute_grad_z(a, c, beta, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 10000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _kkt_guided_update(a, deadline, max_iter, step_init):
    """KKT-guided update: drive the gradient of max toward being constant on active set."""
    n = len(a)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    step = step_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_curr, a_curr)
        c_max = float(c.max())
        # Find lags near the max (within 1%)
        threshold = c_max * 0.999
        active_lags = np.where(c >= threshold)[0]

        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(c))])

        # Compute KKT gradient: sum over active lags of (a[k-i] + a[i-k])
        g = np.zeros(n)
        for k in active_lags:
            # a[k-i] for valid i: need 0 <= k-i < n => k-n+1 <= i <= k
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += a_curr[k - np.arange(lo, hi + 1)]
            # a[i-k] for valid i: need 0 <= i-k < n => k <= i <= k+n-1
            lo2 = max(0, k)
            hi2 = min(n - 1, k + n - 1)
            if lo2 <= hi2:
                g[lo2:hi2+1] += a_curr[np.arange(lo2, hi2 + 1) - k]

        # Only update active (non-zero) elements
        active_mask = a_curr > 1e-12
        if not np.any(active_mask):
            break
        g_active = g[active_mask]
        g_mean = float(g_active.mean())
        g_std = float(g_active.std())
        if g_std < 1e-12:
            break

        # Mirror descent step in log-space
        z = np.log(np.maximum(a_curr, 1e-15))
        z_diff = g - g_mean
        z = z - step * z_diff
        z = z - z[active_mask].mean()  # renormalize on active set
        a_next = _softmax_compute(z)

        sc = _score_norm(a_next)
        if sc < best_score:
            best_score = sc
            best_a = a_next.copy()
            a_curr = a_next
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.9

        if it % 500 == 0:
            pass  # could print

    return best_a, best_score


def _peak_chipping(a, deadline, max_iter, step_init):
    """Iteratively reduce elements that contribute most to the peak lag."""
    n = len(a)
    a_curr = a.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)
    step = step_init

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a_curr, a_curr)
        k_star = int(np.argmax(c))

        # Sensitivity: how much a[i] contributes to c[k_star]
        sens = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            sens[lo:hi+1] = a_curr[k_star - np.arange(lo, hi + 1)]

        sens_max = float(sens.max())
        if sens_max < 1e-15:
            break

        # Reduce high-sensitivity elements, add mass to low-sensitivity ones
        reduction = step * sens / sens_max
        reduction = np.minimum(reduction, a_curr * 0.5)  # don't reduce by more than 50%
        total_reduced = float(reduction.sum())
        a_new = a_curr - reduction

        # Distribute mass to low-sensitivity elements
        low_sens = sens < 0.1 * sens_max
        if np.any(low_sens):
            n_low = int(low_sens.sum())
            a_new[low_sens] += total_reduced / n_low
        else:
            a_new += total_reduced / n

        a_new = np.maximum(a_new, 0.0)
        s_new = float(a_new.sum())
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = _score_norm(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_curr = a_new
            step = min(step * 1.02, 0.3)
        else:
            step *= 0.95

        if step < 1e-8:
            break

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "cosine_mod":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb_log(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _structured_perturb(a, rng):
    """Add a structured periodic perturbation."""
    n = len(a)
    period = rng.integers(n // 8, n // 2)
    phase = rng.random() * 2.0 * np.pi
    amplitude = rng.random() * 0.2 + 0.05
    t = np.arange(n)
    modulation = 1.0 + amplitude * np.cos(2.0 * np.pi * t / period + phase)
    a_new = a * modulation
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

        if len(betas_fine) > 12:
            betas_fine = betas_fine[:12]

        per_t = p4_end - time.monotonic() - 10
        a_opt, sc_opt = _adam_logspace(
            a_work, time.monotonic() + per_t,
            betas_fine, lr_init=0.008, max_iter_total=800000
        )
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")
        else:
            print(f"  Adam fine: no improvement (score={sc_opt_full:.10f})")
        a_work = best_seq.copy()

    # === Phase 5: Multi-restart with diverse perturbations ===
    print("\n=== Phase 5: Multi-restart ===")
    p5_end = min(deadline, t_start + 1350.0)
    if time.monotonic() < p5_end - 20:
        n_restarts = 8
        per_restart = (p5_end - time.monotonic() - 15) / n_restarts
        for i in range(n_restarts):
            if time.monotonic() > p5_end - 10:
                break
            # Different perturbation strategies
            if i % 4 == 0:
                p_start = _perturb_log(a_work, 0.02 + 0.01 * (i // 4), rng)
            elif i % 4 == 1:
                p_start = _structured_perturb(a_work, rng)
            elif i % 4 == 2:
                # Random shift
                sz = n_main // 256
                shift = int(rng.integers(1, max(2, sz)))
                p_start = np.roll(a_work, shift).copy()
                p_start = p_start / p_start.sum()
            else:
                # Blend with a structured profile
                profile = _make_profile(n_main, "cosine_mod")
                mix = 0.3 + 0.1 * (i // 4)
                p_start = (1 - mix) * a_work + mix * profile
                p_start = np.maximum(p_start, 1e-10)
                p_start = p_start / p_start.sum()

            # Quick Adam optimization for each restart
            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            bs = [0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs,
                  0.0001 * max_cs, 0.00005 * max_cs, 0.00002 * max_cs]

            per_t = min(per_restart, p5_end - time.monotonic() - 5)
            if per_t < 5:
                break

            p_best, sc_best = _adam_logspace(
                p_start, time.monotonic() + per_t,
                bs, lr_init=0.006, max_iter_total=100000
            )
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                a_work = p_best.copy()
                print(f"  Restart {i}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")
            else:
                print(f"  Restart {i}: no improvement (score={sc_full:.10f})")

    # === Phase 6: Try different lengths ===
    print("\n=== Phase 6: Multi-length exploration ===")
    p6_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < p6_end - 20 and best_seq is not None:
        for n_try in [2048, 8192]:
            if time.monotonic() > p6_end - 20:
                break
            a_try = _interp_profile(best_seq, n_try)
            sc_try = _score(a_try)
            print(f"  Interpolated n={n_try}: score={sc_try:.10f}")
            if sc_try < best_score:
                best_score = sc_try
                best_seq = a_try.copy()
                a_work = a_try.copy()
                print(f"  -> New best at n={n_try}!")

            # Optimize the new length
            if time.monotonic() < p6_end - 15:
                c_try = _full_convolve(a_work, a_work)
                max_ct = float(c_try.max())
                bt = [0.001 * max_ct, 0.0005 * max_ct, 0.0002 * max_ct,
                      0.0001 * max_ct, 0.00005 * max_ct]
                per_t = min(60.0, p6_end - time.monotonic() - 10)
                if per_t > 10:
                    a_try_opt, _ = _adam_logspace(
                        a_work, time.monotonic() + per_t,
                        bt, lr_init=0.005, max_iter_total=200000
                    )
                    sc_opt = _score(a_try_opt)
                    if sc_opt < best_score:
                        best_score = sc_opt
                        best_seq = a_try_opt.copy()
                        a_work = a_try_opt.copy()
                        print(f"  Optimized n={n_try}: score={sc_opt:.10f}")

    # === Phase 7: Final ultra-fine polish ===
    print("\n=== Phase 7: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        a_pol = best_seq.copy()
        a_pol = np.clip(a_pol, 0, 1000)
        a_pol = a_pol / a_pol.sum()

        c_pol = _full_convolve(a_pol, a_pol)
        max_cp = float(c_pol.max())
        betas_ult = [0.00005 * max_cp, 0.00002 * max_cp, 0.00001 * max_cp,
                     0.000005 * max_cp]
        per_t = deadline - time.monotonic() - 5
        if per_t > 10:
            a_pol, _ = _adam_logspace(
                a_pol, time.monotonic() + per_t,
                betas_ult, lr_init=0.002, max_iter_total=300000
            )
            sc_pol = _score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_seq = a_pol.copy()
                print(f"  Final polish: score={sc_pol:.10f}")

            # One more KKT pass
            if time.monotonic() < deadline - 5:
                a_kkt2, _ = _kkt_guided_update(a_pol, deadline - 2, 50000, 0.005)
                sc_k2 = _score(a_kkt2)
                if sc_k2 < best_score:
                    best_score = sc_k2
                    best_seq = a_kkt2.copy()
                    print(f"  Final KKT: score={sc_k2:.10f}")

    # === Final validation ===
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        c_final = _full_convolve(final, final)
        print(f"  max_conv={c_final.max():.10f}, sum={s:.6e}")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"  Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"  Eval error: {ex}")
        return [float(x) for x in final]

    # Fallback
    fallback = _make_profile(256, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
