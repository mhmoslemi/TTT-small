Wait, that's not right either. The LP maximizes sum(b) subject to the constraints. If b = a is feasible (which it is, since (a*a)_k <= M by definition of M), then the optimal sum(b) >= 1.

The optimal b* has sum(b*) >= 1 and (a * b*)_k <= M for all active k.

If I scale b' = b* / sum(b*), then (a * b')_k <= M / sum(b*) < M (if sum(b*) > 1).

So the peak of (a * b') is less than M! This means the "cross-correlation" peak is lower than the "autocorrelation" peak.

But I want to reduce the autocorrelation peak of b' itself, not just the cross-correlation with a. However, if I blend a and b':
a_new = (1-t)*a + t*b'

Then conv(a_new, a_new) = (1-t)^2 * conv(a,a) + 2t(1-t) * conv(a,b') + t^2 * conv(b',b')

The peak of this is at most:
(1-t)^2 * M + 2t(1-t) * M/sum(b*) + t^2 * max(conv(b',b'))

If max(conv(b',b')) is also small, then the blended sequence has a lower peak.

The LP guarantees that conv(a,b') is below M/sum(b*), but it doesn't guarantee that conv(b',b') is small. However, if b' is "similar" to a in shape, then conv(b',b') should also be small.

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses: diversified parametric initialization, Adam with adaptive log-sum-exp continuation,
LP-based peak balancing, combinatorial local search, and final L-BFGS polish.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _norm_score(a):
    """Score assuming sum(a)=1: returns 2n * max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def _parametric_init(n, etype, rng):
    t = np.linspace(0, 1, n)
    if etype == 'asym_exp':
        a = np.exp(3.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(0.01 + 0.99 * t)
    elif etype == 'exp5':
        a = np.exp(5.0 * (t - 0.5))
    elif etype == 'cubic':
        a = 0.05 + 0.5 * t + 0.4 * t**3
    elif etype == 'gaussian':
        mu = rng.uniform(0.3, 0.7)
        sig = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - mu) / sig) ** 2)
    elif etype == 'bimodal':
        mu1 = rng.uniform(0.15, 0.35)
        mu2 = rng.uniform(0.65, 0.85)
        sig = rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - mu1) / sig) ** 2) + np.exp(-0.5 * ((t - mu2) / sig) ** 2)
    elif etype == 'power_law':
        alpha = rng.uniform(1.5, 4.0)
        a = (0.01 + t) ** alpha
    elif etype == 'sinc':
        freq = rng.uniform(1.0, 4.0)
        a = np.abs(np.sinc(freq * (t - 0.5))) + 0.01
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _compute_grad_and_loss(z, beta, n):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0, np.zeros_like(z), a, cm, c

    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    W = float(np.dot(w, c))

    # h_i = sum_k w_k * a_{k-i}
    w_flip = w[::-1]
    h_full = _full_convolve(w_flip, a)
    h = h_full[2 * n - 2 : n - 2 : -1]

    loss = 2.0 * n * (cm + beta * (max_s + float(np.log(np.sum(e)))))
    grad = 2.0 * n * a * (h - W)

    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(z)
    return loss, grad, a, cm, c


def _adam_optimize(a_init, deadline, lr=0.05, max_iter=500000):
    """Adam optimizer with adaptive log-sum-exp continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_ns = _norm_score(a_init)

    c0 = _full_convolve(a_init, a_init)
    m0 = float(c0.max())

    beta = 0.15 * m0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 10

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1

        # Adaptive beta: geometric decay with min
        beta = max(1e-15, m0 * 0.15 * (0.997 ** (t // 5)))

        loss, grad, a, cm, c = _compute_grad_and_loss(z, beta, n)

        if not np.all(np.isfinite(grad)):
            break

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -60, 60)
        z_new = z_new - z_new.mean()

        # Check periodically
        if t % check_interval == 0:
            a_cur = _softmax_fn(z_new)
            ns = _norm_score(a_cur)
            if np.isfinite(ns) and ns < best_ns - 1e-12:
                best_ns = ns
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 40:
                # Reheat: add noise and increase beta
                z_new = z_new + 0.1 * np.random.randn(n)
                z_new = z_new - z_new.mean()
                beta = min(beta * 2, 0.3 * m0)
                no_improve = 0

        z = z_new

    return best_a, best_ns


def _lp_refine(a_init, deadline, K=40, max_rounds=20):
    """LP-based peak balancing to reduce the maximum convolution value."""
    n = len(a_init)
    best_a = a_init.copy()
    best_ns = _norm_score(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Active peaks: top K and those within 90% of max
        sorted_idx = np.argsort(c)[::-1]
        active = sorted_idx[:K]
        threshold = 0.90 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([active, additional]))
        if len(active) > 250:
            active = active[:250]

        # Build constraint matrix
        Ka = len(active)
        A_mat = np.zeros((Ka, n))
        for row, k in enumerate(active):
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                A_mat[row, i_lo:i_hi + 1] = best_a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]

        b_vec = np.full(Ka, M * 1.0005)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 15, 'dual_feasibility_tolerance': 1e-8})
            if res.status != 0 or res.x is None:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 0.5 * S:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        # Try blending
        improved = False
        current_ns = _norm_score(best_a)
        for t in [0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            a_trial = a_trial / s_trial
            ns_trial = _norm_score(a_trial)
            if ns_trial < current_ns - 1e-10:
                best_a = a_trial.copy()
                best_ns = ns_trial
                current_ns = ns_trial
                improved = True
                break

        if not improved:
            break

    return best_a, best_ns


def _local_search(a_init, deadline, n_trials=50000):
    """Combinatorial local search: try moving mass between pairs of indices."""
    n = len(a_init)
    best_a = a_init.copy()
    best_ns = _norm_score(a_init)
    rng = np.random.default_rng(12345)

    c = _full_convolve(best_a, best_a)
    cm = float(c.max())
    peak_idx = int(np.argmax(c))

    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        # Occasionally refresh peak
        if trial % 500 == 0:
            c = _full_convolve(best_a, best_a)
            cm = float(c.max())
            peak_idx = int(np.argmax(c))

        # Compute gradient direction for the peak
        # d c[peak]/d a[i] = a[peak - i] for valid i
        i_lo = max(0, peak_idx - n + 1)
        i_hi = min(n - 1, peak_idx)

        # Pick a random valid index to decrease and a random index to increase
        if i_hi - i_lo < 1:
            continue

        # Gradient says: decrease a[i] where a[peak-i] is large, increase where small
        contrib = np.zeros(n)
        contrib[i_lo:i_hi + 1] = best_a[peak_idx - i_lo:peak_idx - i_lo + (i_hi - i_lo + 1)]

        # Probability proportional to contribution (decrease high contributors)
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue

        # Increase from a random non-contributing position
        non_contrib_mask = (contrib < 0.1 * cm)
        non_contrib_indices = np.where(non_contrib_mask)[0]
        if len(non_contrib_indices) == 0:
            continue
        j = non_contrib_indices[rng.integers(len(non_contrib_indices))]

        # Decrease from a high-contributing position
        if rng.random() < 0.7:
            # Pick from top contributors
            top_k = min(50, i_hi - i_lo + 1)
            top_candidates = np.arange(i_lo, i_hi + 1)
            top_vals = contrib[top_candidates]
            top_candidates = top_candidates[np.argsort(top_vals)[::-1][:top_k]]
            i_dec = top_candidates[rng.integers(len(top_candidates))]
        else:
            i_dec = rng.integers(i_lo, i_hi + 1)

        if i_dec == j or best_a[i_dec] < 1e-10:
            continue

        # Try a small transfer
        eps = best_a[i_dec] * rng.uniform(0.01, 0.15)
        a_trial = best_a.copy()
        a_trial[i_dec] -= eps
        a_trial[j] += eps

        if a_trial[i_dec] < 0:
            continue

        ns_trial = _norm_score(a_trial)
        if ns_trial < best_ns - 1e-12:
            best_a = a_trial
            best_ns = ns_trial

    return best_a, best_ns


def _lbfgs_polish(a_init, deadline, beta_list=None, maxiter=5000):

    if beta_list is None:

        beta_list = [m0 * b for b in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]]

    best_a = a_init.copy()

            if res.x is not None and np.all(np.isfinite(res.x)):

def _spectral_init(n, rng):
    """Initialize using spectral flattening (Gerchberg-Saxton style)."""
    N = _next_pow2(2 * n - 1)
    a = np.zeros(n)
    a[:] = 1.0 / n

    for it in range(50):
        A = np.fft.rfft(a, N)
        # Target: |A|^2 should be flat, so |A| should be constant
        mag = np.abs(A)
        target_mag = mag.mean()
        # Update: keep phase, set magnitude to target
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # Ensure A_new is Hermitian (for real a)
        A_full = np.fft.fft(A_new, N)
        A_full[N // 2 + 1:] = np.conj(A_full[1:N // 2][::-1])
        if N % 2 == 0:
            A_full[N // 2] = np.abs(A_full[N // 2])
        a_new = np.fft.irfft(A_full, N)[:n]
        # Project to non-negative
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

    return a

    # === Final validation and return ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, total_time={time.monotonic()-t_start:.0f}s")

        if len(final_seq) <= 100000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                    return [float(x) for x in final_seq]
                else:
                    print("Evaluator returned inf, using fallback")
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    sc_fb = _score_fn(fb)
    print(f"FALLBACK: n=512, score={sc_fb:.10f}")
    return [float(x) for x in fb]
