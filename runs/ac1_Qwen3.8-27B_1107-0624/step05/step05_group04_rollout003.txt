        # = sum_k w[k] * a_rev[n-1-k+i]
        # = (w * a_rev)[i + n - 1]  ... let me verify
        # (w * a_rev)[m] = sum_k w[k] * a_rev[m-k]
        # We want sum_k w[k] * a_rev[(i+n-1)-k] = (w * a_rev)[i+n-1]
        # a_rev[n-1-k+i] = a_rev[i+n-1-k] ✓
        h = w_conv[n - 1: 2 * n - 1]
        
        scale = (float(np.sum(c_pos ** power)) ** (1.0/power - 1.0))
        grad_a = scale * 2.0 * h
    else:
        return obj, np.zeros(n)
    
    return obj, grad_a

def _topk_grad_z(a, z, power=10):
    """
    Compute gradient of top-k power objective w.r.t. z.
    Uses the chain rule: grad_z = a * (grad_a - sum_i a[i]*grad_a[i])
    """
    obj, grad_a = _topk_grad_and_obj(a, power=power)
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return obj, grad_z

def _run_adam(a_init, deadline, rng, lr_init=0.05, power=10, max_iter=500000):
    """Adam gradient descent in log-space with top-k power objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    
    no_improve_count = 0
    check_interval = 500
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        t += 1
        a = _softmax_fn(z)
        
        obj, grad_z = _topk_grad_z(a, z, power=power)
        
        if not np.all(np.isfinite(grad_z)):
            break
        
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        z_new = z - lr * mh / (np.sqrt(vh) + eps)
        z_new = np.clip(z_new, -50, 50)
        z_new = z_new - z_new.mean()
        
        a_new = _softmax_fn(z_new)
        sc = _score_norm_fn(a_new)
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z = z_new
            no_improve_count = 0
        else:
            no_improve_count += check_interval
            # Keep z as is (don't update if no improvement)
            # Actually for Adam, we should still update z to maintain momentum
            z = z_new
        
        # Decay learning rate periodically
        if t % 5000 == 0:
            lr = max(0.001, lr * 0.85)
        
        # Perturbation to escape local minima
        if t % 20000 == 0:
            z = z + 0.05 * rng.normal(size=n)
            z = z - z.mean()
    
    return best_a, best_score

def _coordinate_descent(a, deadline, rng, max_rounds=1000):

def _block_refine(a, deadline, rng, block_size=32, max_rounds=500):
    """
    Block-level refinement: adjust blocks of elements to reduce peaks.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    
    n_blocks = n // block_size
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        
        # Find worst peak
        peak_k = int(np.argmax(c))
        
        # Find which block contributes most to the peak
        valid_i = np.arange(max(0, peak_k - n + 1), min(n, peak_k + 1))
        if len(valid_i) == 0:
            continue
        prod = a_cur[valid_i] * a_cur[peak_k - valid_i]
        # Assign each valid_i to a block
        block_of = valid_i // block_size
        block_sums = np.zeros(n_blocks)
        np.add.at(block_sums, block_of, prod)
        
        # Target the block with highest contribution
        target_block = int(np.argmax(block_sums))
        block_start = target_block * block_size
        block_end = min(block_start + block_size, n)
        
        # Try adjusting the entire block
        improved = False
        for scale in [0.95, 0.9, 0.85, 1.05, 1.1]:
            z_trial = z.copy()
            z_trial[block_start:block_end] += np.log(scale)
            a_trial = _softmax_fn(z_trial)
            sc = _score_norm_fn(a_trial)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()
                z = z_trial.copy()
                improved = True
                break
        
        if not improved:
            # Try a gradient step on just this block
            obj, grad_z = _topk_grad_z(a_cur, z, power=10)
            block_grad = grad_z[block_start:block_end]
            if np.all(np.isfinite(block_grad)) and np.max(np.abs(block_grad)) > 1e-15:
                step = 0.01
                z_trial = z.copy()
                z_trial[block_start:block_end] -= step * block_grad / (np.max(np.abs(block_grad)) + 1e-15)
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    z = z_trial.copy()
                    improved = True
        
        if not improved and rnd > 30:
            # Do a global Adam step instead
            obj, grad_z = _topk_grad_z(a_cur, z, power=10)
            z -= 0.01 * grad_z / (np.max(np.abs(grad_z)) + 1e-15)
            z = z - z.mean()
    
    return best_a, best_score
