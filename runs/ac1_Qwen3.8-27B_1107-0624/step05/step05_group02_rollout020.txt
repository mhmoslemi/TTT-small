from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return float(_score_normalized(a_norm))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    j = np.arange(n)
    diff = k[:, np.newaxis] - j[np.newaxis, :]

    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

`a[i] * sum_j a[j] * (delta_{j, i+k} + delta_{j, i-k})` for each lag k

So `a[i]` appears in `conv(a,a)[k]` as `a[i] * a[k-i]` (if valid) + `a[i] * a[i-k]` (if valid).

The total "cost" involving `a[i]` is:
`a[i] * (sum over valid k of w_k * (a[k-i] + a[i-k]))`

where `w_k` are the weights from the log-sum-exp.

To minimize this with respect to `a[i]` (subject to `a[i] >= 0` and keeping other `a[j]` fixed), we'd want `a[i] = 0` if the coefficient is positive (which it always is since all `a[j] >= 0`). But we also need `sum(a) = 1`, so we can't set all to zero.

This suggests that coordinate descent isn't directly applicable in the simplex constraint. The softmax parameterization handles this automatically.

"""
Hybrid LP + enhanced Adam with mixed subgradient/smoothed gradient,
nudge steps, weight averaging, and adaptive temperature scheduling
for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
"""
import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    c = _full_convolve(a_norm, a_norm)
    return float(2.0 * n * c.max())


def _score_norm(a):
    """Assumes sum(a) ~ 1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _build_lp_matrix(f, n):
    k_arr = np.arange(2 * n - 1)
    j_arr = np.arange(n)
    diff = k_arr[:, np.newaxis] - j_arr[np.newaxis, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _subgrad_blame(a, c, n, top_k=3):
    """Compute blame vector for top-k lags of convolution."""
    blame = np.zeros(n)
    top_indices = np.argsort(c)[-top_k:]
    for k_star in top_indices:
        k_star = int(k_star)
        lo1 = max(0, k_star - n + 1)
        hi1 = min(n, k_star + 1)
        if lo1 < hi1:
            j_idx = np.arange(lo1, hi1)
            blame[j_idx] += a[k_star - j_idx]
        lo2 = max(0, k_star)
        hi2 = min(n, k_star + n)
        if lo2 < hi2:
            j_idx = np.arange(lo2, hi2)
            blame[j_idx] += a[j_idx - k_star]
    return blame


def _adam_enhanced(a_init, deadline, lr=0.01, mix_alpha=0.3,
                   nudge_interval=150, nudge_strength=0.08,
                   top_k_for_nudge=3, ema_decay=0.997):
    """
    Enhanced Adam with:
    - Mixed subgradient + log-sum-exp smoothed gradient
    - Periodic nudge steps targeting top convolution lags
    - Exponential moving average of iterates
    - Adaptive temperature cooling
    - Learning rate decay
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    beta_scales = [0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003,
                   0.002, 0.0015, 0.001, 0.0007, 0.0005, 0.0003, 0.0002,
                   0.0001, 0.00005, 0.00003]
    max_it_per_phase = 25000

    ema_a = a_init.copy()
    best_ema_score = best_score

    total_iters = 0
    for beta_scale in beta_scales:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * max_c_init
        iters = 0
        phase_start_lr = current_lr
        while iters < max_it_per_phase:
            if time.monotonic() > deadline:
                break
            iters += 1
            total_iters += 1
            t_step += 1

            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())

            # Smoothed gradient via log-sum-exp
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h_smooth = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            smooth_grad = 4.0 * n * a * (h_smooth - W)

            # Subgradient at top lags
            blame = _subgrad_blame(a, c, n, top_k=top_k_for_nudge)
            subgrad = 4.0 * n * a * blame

            # Mixed gradient
            grad = mix_alpha * subgrad + (1.0 - mix_alpha) * smooth_grad

            # Nudge step
            if iters % nudge_interval == 0:
                blame_nudge = _subgrad_blame(a, c, n, top_k=top_k_for_nudge)
                z -= nudge_strength * blame_nudge * a
                z = z - z.mean()

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            max_update = 0.3
            if update_norm > max_update:
                update = update * (max_update / update_norm)

            z = z - update
            z = np.clip(z, -60.0, 60.0)
            z = z - z.mean()

            # Update EMA
            a_curr = _softmax(z)
            ema_a = ema_decay * ema_a + (1.0 - ema_decay) * a_curr

            # Periodic evaluation
            if total_iters % 100 == 0:
                sc = float(_full_convolve(a_curr, a_curr).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_curr.copy()

                sc_ema = float(_full_convolve(ema_a, ema_a).max())
                if np.isfinite(sc_ema) and sc_ema < best_ema_score:
                    best_ema_score = sc_ema
                if np.isfinite(sc_ema) and sc_ema < best_score:
                    best_score = sc_ema
                    best_a = ema_a.copy()
                    if best_score < 0.0755:
                        print(f"    Adam iter {total_iters}: score={best_score:.10f} "
                              f"(beta_scale={beta_scale}, ema={'yes' if best_a is ema_a else 'no'})")

            # LR decay within phase
            if iters % 5000 == 0:
                current_lr = max(0.0005, phase_start_lr * (0.92 ** (iters // 5000)))

    # Final evaluation
    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    sc_ema_final = float(_full_convolve(ema_a, ema_a).max())
    if np.isfinite(sc_ema_final) and sc_ema_final < best_score:
        best_score = sc_ema_final
        best_a = ema_a.copy()

    return best_a, best_score

def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _make_uniform(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _lp_optimize(f, n, phase_end, max_lp_steps=100):
    """
    LP-based optimization: find descent directions by solving
    max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Returns optimized f and number of steps taken.
    """
    f = f / f.sum()
    current_M = float(_full_convolve(f, f).max())
    current_t = 0.3
    lp_steps = 0
    stuck = 0

    while time.monotonic() < phase_end and lp_steps < max_lp_steps:
        c = _full_convolve(f, f)
        M = float(c.max())

        A = _build_lp_matrix(f, n)
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        g = None
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': 20, 'presolve': True})
            if res.status == 0 and res.x is not None:
                g = np.maximum(np.asarray(res.x, dtype=np.float64), 0)
                if g.sum() < 1e-15:
                    g = None
        except Exception:
            g = None

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            current_t *= 0.5
            continue

        stuck = 0
        g_sum = float(g.sum())
        if g_sum < 1e-15:
            break
        g_norm = g * (f.sum() / g_sum)

        # Backtracking line search
        t_try = current_t
        improved = False
        for _ in range(5):
            f_new = (1.0 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            f_new_sum = float(f_new.sum())
            if f_new_sum < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / f_new_sum

            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())

            if M_new < M * (1.0 - 1e-10):
                f = f_new
                current_t = min(1.0, t_try * 1.3)
                improved = True
                lp_steps += 1
                if lp_steps % 5 == 0:
                    print(f"    LP step {lp_steps}: M={M:.10f} -> {M_new:.10f} "
                          f"(t={t_try:.4f}) t={time.monotonic():.0f}s")
                break
            else:
                t_try *= 0.5

        if not improved:
            current_t *= 0.5
            stuck += 1
            if stuck > 15:
                break

    return f, lp_steps

    update_best(f)
    print(f"Phase 1 done: {lp_steps} LP steps, best={best_score:.10f}, "
          f"time={time.monotonic() - t_start:.1f}s")

    # === Phase 2: Enhanced Adam (900s) ===
    phase2_end = min(deadline, t_start + 1100.0)
    print(f"\n=== Phase 2: Enhanced Adam ===")

    if best_seq is not None and time.monotonic() < phase2_end - 60:
        a_init = best_seq / best_seq.sum()
        remaining = phase2_end - time.monotonic() - 30
        a_best, sc_best = _adam_enhanced(
            a_init, time.monotonic() + remaining,
            lr=0.012, mix_alpha=0.3, nudge_interval=150,
            nudge_strength=0.08, top_k_for_nudge=3, ema_decay=0.997
        )
        update_best(a_best)

        # Second run with different parameters
        if time.monotonic() < phase2_end - 90:
            a_init2 = f / f.sum()
            remaining2 = phase2_end - time.monotonic() - 30
            a_best2, sc_best2 = _adam_enhanced(
                a_init2, time.monotonic() + remaining2,
                lr=0.008, mix_alpha=0.5, nudge_interval=100,
                nudge_strength=0.12, top_k_for_nudge=5, ema_decay=0.995
            )
            update_best(a_best2)

    print(f"Phase 2 done: best={best_score:.10f}, time={time.monotonic() - t_start:.1f}s")

    # === Phase 3: Multi-restart (400s) ===
    phase3_end = min(deadline, t_start + 1500.0)
    print(f"\n=== Phase 3: Multi-restart ===")

    if best_seq is not None and time.monotonic() < phase3_end - 60:
        a_best_norm = best_seq / best_seq.sum()
        configs = [
            (0.02, 0.010, 0.3),
            (0.04, 0.008, 0.4),
            (0.06, 0.007, 0.5),
            (0.10, 0.006, 0.6),
            (0.03, 0.012, 0.2),
            (0.08, 0.009, 0.4),
        ]
        n_restarts = len(configs)
        per_restart = (phase3_end - time.monotonic() - 40) / max(1, n_restarts)

        for i, (sigma, lr_r, alpha_r) in enumerate(configs):
            if time.monotonic() > phase3_end - 40:
                break
            p_start = _perturb(a_best_norm, sigma, rng)
            deadline_i = time.monotonic() + min(per_restart, phase3_end - time.monotonic() - 20)
            if deadline_i - time.monotonic() < 30:
                break
            print(f"  Restart {i + 1}/{n_restarts}: sigma={sigma}, lr={lr_r}, alpha={alpha_r}")
            a_r, sc_r = _adam_enhanced(
                p_start, deadline_i, lr=lr_r, mix_alpha=alpha_r,
                nudge_interval=120, nudge_strength=0.06,
                top_k_for_nudge=3, ema_decay=0.996
            )
            update_best(a_r)

    print(f"Phase 3 done: best={best_score:.10f}, time={time.monotonic() - t_start:.1f}s")

    # === Phase 4: Refined polishing (remaining time) ===
    phase4_end = deadline
    print(f"\n=== Phase 4: Refined polishing ===")

    if best_seq is not None and time.monotonic() < phase4_end - 30:
        a_best_norm = best_seq / best_seq.sum()
        remaining = phase4_end - time.monotonic() - 20
        if remaining > 60:
            a_final, sc_final = _adam_refined(
                a_best_norm, time.monotonic() + remaining, lr=0.002
            )
            update_best(a_final)

            # One more pass with slightly different LR
            if time.monotonic() < phase4_end - 40:
                remaining2 = phase4_end - time.monotonic() - 20
                if remaining2 > 40:
                    a_final2, sc_final2 = _adam_refined(
                        a_final / a_final.sum(),
                        time.monotonic() + remaining2, lr=0.001
                    )
                    update_best(a_final2)

    print(f"Phase 4 done: best={best_score:.10f}, time={time.monotonic() - t_start:.1f}s")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_uniform(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
              f"time={time.monotonic() - t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fallback = _make_uniform(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
