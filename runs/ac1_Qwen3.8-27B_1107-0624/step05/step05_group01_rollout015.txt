"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses peak-balancing gradient descent, LP refinement, spectral methods,
and diverse initialization to escape local minima.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _norm_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _init_binomial(n):

        b[j] = b[j-1] * (2.0*j - 1.0) / (2.0*j)
    a = b[::-1].copy()
    return a / a.sum()

def _lbfgs_lse(b_init, deadline, K=20):
    """L-BFGS on log-sum-exp of top K peaks."""
    n = len(b_init)
    z = np.log(np.maximum(b_init, 1e-15))
    z = z - z.mean()
    best_b = b_init.copy()
    best_score = _norm_score(best_b)
    
    # Continuation: start with large beta (smooth), end with small (sharp)
    c0 = _full_convolve(best_b, best_b)
    M0 = float(c0.max())
    betas = [0.1 * M0, 0.05 * M0, 0.02 * M0, 0.01 * M0, 0.005 * M0,
             0.002 * M0, 0.001 * M0, 0.0005 * M0]
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        def obj(z_in):
            b_in = _softmax(z_in)
            c_in = _full_convolve(b_in, b_in)
            # Log-sum-exp of all peaks (or top K)
            sorted_idx = np.argsort(c_in)[::-1][:K]
            peaks = c_in[sorted_idx]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            sh = (peaks - pm) / beta
            lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)
        
        def grad(z_in):
            b_in = _softmax(z_in)
            c_in = _full_convolve(b_in, b_in)
            sorted_idx = np.argsort(c_in)[::-1][:K]
            peaks = c_in[sorted_idx]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            sh = (peaks - pm) / beta
            e = np.exp(sh - sh.max())
            w = e / e.sum()
            
            # Weighted gradient
            w_full = np.zeros(2*n-1)
            w_full[sorted_idx] = w
            grad_b = _peak_grad(b_in, w_full)
            
            dot_b_grad = float(np.dot(b_in, grad_b))
            grad_z = b_in * (grad_b - dot_b_grad)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z
        
        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad,
                             options={'maxiter': 5000, 'ftol': 1e-20,
                                      'gtol': 1e-14, 'maxls': 50})
            if result.x is not None and np.all(np.isfinite(result.x)):
                b_new = _softmax(result.x)
                sc = _norm_score(b_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_b = b_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break
    
    return best_b, best_score

def _mirror_smooth(b):
    """Average b with its reverse for symmetry."""
    b_rev = b[::-1].copy()
    b_sym = 0.5 * (b + b_rev)
    b_sym = b_sym / b_sym.sum()
    return b_sym

def _interp_to(b, n_new):
    """Interpolate sequence to new length."""
    n = len(b)
    if n == n_new:
        return b.copy()
    x0 = np.linspace(0, 1, n)
    x1 = np.linspace(0, 1, n_new)
    b_new = np.interp(x1, x0, b)
    b_new = np.maximum(b_new, 1e-12)
    return b_new / b_new.sum()
