                        a_trial = a_trial / s
                    sc = _score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
        if not improved:
            # Try mirror transfer: reduce at i, increase at k0-i (if different)
            if i != j:
                for frac in [0.005, 0.01, 0.02]:
                    a_trial = a.copy()
                    red = a_trial[max(i, j)] * frac
                    a_trial[max(i, j)] = max(0, a_trial[max(i, j)] - red)
                    a_trial[min(i, j)] += red
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s > 1e-15:
                        a_trial = a_trial / s
                    sc = _score_norm(a_trial)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
        if not improved and rnd > 200:
            break
    return best_a, best_score


def _adam_gradient(a_init, deadline, lr_init=0.02, max_iter=50000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    t = 0
    c = _full_conv(_softmax(z), _softmax(z))
    beta = 0.05 * float(c.max())
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_conv(a, a)
        cm = float(c.max())
        if cm < 1e-30:
            break
        if t % 500 == 0:
            sorted_c = np.sort(c)[::-1]
            k = min(10, len(sorted_c) - 1)
            gap = sorted_c[0] - sorted_c[k]
            if gap < 0.02 * cm:
                beta = max(1e-10, 0.02 * cm)
            elif gap > 0.3 * cm:
                beta = max(1e-10, 0.005 * cm)
            else:
                beta = max(1e-10, 0.2 * gap)
        shifted = (c - cm) / beta
        e = np.exp(shifted)
        w = e / e.sum()
        a_rev = a[::-1]
        w_conv = _full_conv(a_rev, w)
        h = w_conv[n - 1:2 * n - 1]
        grad_a = 2.0 * h
        offset = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - offset)
        if not np.all(np.isfinite(grad_z)):
            break
        m = b1 * m + (1 - b1) * grad_z
        v = b2 * v + (1 - b2) * (grad_z * grad_z)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        a = _softmax(z)
        sc = _score_norm(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
        if t % 5000 == 0:
            lr = max(0.002, lr * 0.85)
    return best_a, best_score


def _make_profiles(n):
    t = np.linspace(0, 1, n)
    profiles = []
    for s in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        a = 1.0 + s * t
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    for s in [1.0, 2.0, 3.0]:
        a = np.exp(s * t)
        profiles.append(a / a.sum())
    a = np.sin(np.pi * t)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    a = np.cos(np.pi * t / 2)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    for q in [2, 3, 4]:
        a = 1.0 + q * (t - 0.5) ** 2
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    return profiles


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2 ** 63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_p = _score_full(parent)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n_init = 512
        profiles = _make_profiles(n_init)
        best_prof = None
        best_prof_sc = np.inf
        for p in profiles:
            sc = _score_norm(p)
            if sc < best_prof_sc:
                best_prof_sc = sc
                best_prof = p.copy()
        update_best(best_prof, _score_full(best_prof), "initial_profile")
        parent = best_prof

    n_best = len(parent)
    print(f"Starting optimization: n={n_best}, score={best_score:.10f}")

    # Phase 1: L-BFGS continuation on parent (40% of time)
    phase1_end = t_start + (budget_s - 30.0) * 0.40
    print(f"\nPhase 1: L-BFGS on parent (deadline t={phase1_end - t_start:.0f}s)")
    if time.monotonic() < phase1_end - 5:
        a_opt, sc_opt = _run_lbfgs(parent, phase1_end - 5, n_betas=20)
        update_best(a_opt, _score_full(a_opt), "phase1_lbfgs")

    # Phase 2: Adam gradient descent (20% of time)
    phase2_end = t_start + (budget_s - 30.0) * 0.60
    print(f"\nPhase 2: Adam GD (deadline t={phase2_end - t_start:.0f}s)")
    if time.monotonic() < phase2_end - 5 and best_seq is not None:
        a_adam, sc_adam = _adam_gradient(best_seq, phase2_end - 5, lr_init=0.01)
        update_best(a_adam, _score_full(a_adam), "phase2_adam")

    # Phase 3: Subgradient polishing (15% of time)
    phase3_end = t_start + (budget_s - 30.0) * 0.75
    print(f"\nPhase 3: Subgradient polish (deadline t={phase3_end - t_start:.0f}s)")
    if time.monotonic() < phase3_end - 5 and best_seq is not None:
        # Multiple subgradient runs with different lr
        for lr_try in [0.005, 0.001, 0.0005]:
            if time.monotonic() > phase3_end - 10:
                break
            sub_deadline = min(phase3_end - 5, time.monotonic() + 120.0)
            a_sg, sc_sg = _subgrad_polish(best_seq, sub_deadline, lr_init=lr_try)
            update_best(a_sg, _score_full(a_sg), f"subgrad_lr{lr_try}")

    # Phase 4: Peak coordinate descent (15% of time)
    phase4_end = t_start + (budget_s - 30.0) * 0.90
    print(f"\nPhase 4: Peak coord descent (deadline t={phase4_end - t_start:.0f}s)")
    if time.monotonic() < phase4_end - 5 and best_seq is not None:
        cd_deadline = min(phase4_end - 5, time.monotonic() + 200.0)
        a_cd, sc_cd = _peak_coord_descent(best_seq, cd_deadline)
        update_best(a_cd, _score_full(a_cd), "phase4_coord")

    # Phase 5: Final L-BFGS with tiny beta + subgradient (10% of time)
    print(f"\nPhase 5: Final polish")
    if time.monotonic() < deadline - 5 and best_seq is not None:
        # Tiny beta L-BFGS
        if time.monotonic() < deadline - 30:
            lbfgs_dl = min(deadline - 30, time.monotonic() + 120.0)
            a_final, sc_final = _run_lbfgs(best_seq, lbfgs_dl, n_betas=10)
            update_best(a_final, _score_full(a_final), "phase5_lbfgs")
        # Final subgradient
        if time.monotonic() < deadline - 5:
            sg_dl = min(deadline - 3, time.monotonic() + 60.0)
            a_sg2, sc_sg2 = _subgrad_polish(best_seq, sg_dl, lr_init=0.0001)
            update_best(a_sg2, _score_full(a_sg2), "phase5_subgrad")

    # Try upscaling if time remains and score could improve
    if time.monotonic() < deadline - 60 and best_seq is not None:
        n_cur = len(best_seq)
        if n_cur <= 2048:
            target_n = min(4096, n_cur * 2)
            print(f"\nTry upscaling to n={target_n}")
            a_up = _upsample(best_seq, target_n)
            sc_up = _score_full(a_up)
            print(f"  Upsampled score: {sc_up:.10f}")
            if sc_up < best_score * 1.01:  # Only if competitive
                up_deadline = min(deadline - 10, time.monotonic() + 120.0)
                a_up_opt, sc_up_opt = _run_lbfgs(a_up, up_deadline, n_betas=8)
                update_best(a_up_opt, _score_full(a_up_opt), "upscaled")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = np.linspace(0.1, 1.0, 256)
            final_seq = final_seq / final_seq.sum()
        sc_final = _score_full(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    # Fallback
    fallback = np.linspace(0.1, 1.0, 256)
    fallback = fallback / fallback.sum()
    print(f"FALLBACK: n=256, score={_score_full(fallback):.10f}")
    return [float(x) for x in fallback]
