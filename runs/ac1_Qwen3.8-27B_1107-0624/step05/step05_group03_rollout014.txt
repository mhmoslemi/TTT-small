        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))


def _fw_lp_solve(f, M, active_lags, solver_list):
    n = len(f)
    A = _build_active_matrix(n, f, active_lags)
    g = cp.Variable(n, non_neg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver, threads=1)
            if prob.status in ['optimal', 'optimal_inaccurate'] and g.value is not None:
                return np.maximum(g.value, 0.0)
        except Exception:
            continue
    return None


def _fw_method(f_init, deadline, rng, max_iters=50, active_frac=0.5):
    n = len(f_init)
    target_sum = np.sqrt(2.0 * n)
    f = f_init.copy()
    s = f.sum()
    f = f / s * target_sum
    
    c = _full_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    best_score = float(2.0 * n * M / target_sum**2)
    
    print(f"  FW start: M={M:.12f}, score={best_score:.10f}")
    
    solver_list = [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC, cp.GLPK]
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 15.0:
            break
        
        t0 = time.monotonic()
        
        # Compute active set
        c = _full_convolve(f, f)
        M = float(c.max())
        
        # Select active lags: keep top fraction by value
        n_active = max(20, int(active_frac * (2 * n - 1)))
        active = np.argsort(c)[-n_active:]
        
        # Solve LP
        g = _fw_lp_solve(f, M, active, solver_list)
        t_lp = time.monotonic() - t0
        
        if g is None or g.sum() < 1e-15:
            print(f"  FW iter {it}: LP failed ({t_lp:.1f}s)")
            # Try with larger active set
            n_active2 = max(50, int(0.8 * (2 * n - 1)))
            active2 = np.argsort(c)[-n_active2:]
            g = _fw_lp_solve(f, M, active2, solver_list)
            if g is None or g.sum() < 1e-15:
                print(f"  FW iter {it}: LP failed even with larger active set")
                break
        
        s_g = g.sum()
        g = g / s_g * target_sum
        
        # Step size
        t = 2.0 / (it + 2.0)
        t = min(t, 1.0)
        
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            break
        f_new = f_new / s_new * target_sum
        
        c_new = _full_convolve(f_new, f_new)
        M_new = float(c_new.max())
        
        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            best_score = float(2.0 * n * best_M / target_sum**2)
            print(f"  FW iter {it}: M={M_new:.12f}, score={best_score:.10f}, t={t:.3f}, lp_time={t_lp:.1f}s, active={len(active)}")
        else:
            print(f"  FW iter {it}: M={M_new:.12f} (no improvement), t={t:.3f}, lp_time={t_lp:.1f}s")
        
        f = f_new.copy()
    
    return best_f / best_f.sum(), best_score
