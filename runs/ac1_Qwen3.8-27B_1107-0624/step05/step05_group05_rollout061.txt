from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    # Find largest rho such that u[rho-1] > (cssv[rho-1] - 1) / rho

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]

$\nabla_p L = \frac{2}{\beta} \sum_k w_k p_{k-j} = \frac{2}{\beta} (w \star p)_j$.

To minimize, we do $p \leftarrow p - \eta \nabla_p L$ (projected onto simplex).

With the softmax parameterization $p_j = \exp(z_j) / \sum \exp(z_i)$:
$\frac{\partial p_j}{\partial z_l} = p_j (\delta_{jl} - p_l)$

$\frac{\partial L}{\partial z_l} = \sum_j \frac{\partial L}{\partial p_j} \frac{\partial p_j}{\partial z_l} = \sum_j \frac{\partial L}{\partial p_j} p_j (\delta_{jl} - p_l) = p_l \frac{\partial L}{\partial p_l} - p_l \sum_j p_j \frac{\partial L}{\partial p_j}$

$= p_l (g_l - \sum_j p_j g_j)$ where $g = \nabla_p L$.

So $\nabla_z L = p \odot (g - (p^T g) \mathbf{1})$.

To minimize: $z \leftarrow z - \eta \nabla_z L = z - \eta \cdot p \odot (g - (p^T g) \mathbf{1})$.

This makes sense: it's a multiplicative update on $p$.

1. Initialize $z$ (e.g., $z = \log(p_0)$ where $p_0$ is the initial probability).
2. Iterate:

   b. $c = p * p$ (full convolution)

   d. $g = (2/\beta) (w \star p)$
   e. $\nabla_z L = p \odot (g - (p^T g) \mathbf{1})$
   f. $z \leftarrow z - \eta \nabla_z L$
   g. $z \leftarrow z - \text{mean}(z)$ (centering)
3. Decrease $\beta$ over time (annealing).

This is clean and efficient. Each iteration is $O(n \log n)$ due to the FFT.

Key insight: the problem is to find a probability vector p minimizing max_k (p*p)_k,
where the gradient can be computed in O(n log n) via FFT-based convolution.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _convolve_weighted(w, p):
    """Compute (w *star* p)_j = sum_k w_k * p_{k-j} for j=0..n-1.
    w has length 2n-1, p has length n."""
    n = len(p)
    w_rev = w[::-1]
    conv = _full_convolve(w_rev, p)
    start = n - 1
    end = 2 * n - 1
    return conv[start:end][::-1]


def _smoothed_grad(p, beta):
    """Compute smoothed gradient of log-sum-exp((p*p)/beta) w.r.t. p."""
    n = len(p)
    c = _full_convolve(p, p)
    c_shifted = c / beta
    c_shifted -= c_shifted.max()
    w = np.exp(c_shifted)
    w /= w.sum()
    grad_p = (2.0 / beta) * _convolve_weighted(w, p)
    return grad_p, c, w


def _natural_grad_step(z, beta, lr):
    """One step of natural gradient descent on softmax parameterization."""
    p = _softmax(z)
    grad_p, c, w = _smoothed_grad(p, beta)
    W = float(np.dot(p, grad_p))
    grad_z = p * (grad_p - W)
    z = z - lr * grad_z
    z -= z.mean()
    return z, p, float(c.max())

def _lp_step(p, M, time_limit=30):
    """Solve LP: maximize sum(q) s.t. (p*q)_k <= M, q >= 0."""
    n = len(p)
    # Build constraint matrix A where A[k,j] = p[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(p[k - j])
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status == 0 and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def _lp_optimize(p, max_iters=100, stall_limit=30):

def _anneal_grad(p_init, budget_s, beta_init, beta_final, lr_init, lr_final,
                 momentum=0.9):
    """Smoothed natural gradient descent with annealing."""
    n = len(p_init)
    z = np.log(np.maximum(p_init, 1e-15)).copy()
    z -= z.mean()
    best_p = p_init.copy()
    best_sc = _score_norm(p_init)
    v = np.zeros(n)  # Momentum buffer
    
    t_start = time.monotonic()
    deadline = t_start + budget_s
    it = 0
    total_iters_estimate = max(1, int(budget_s / 0.5))  # ~2 iters per second
    prev_best = best_sc
    
    while time.monotonic() < deadline:
        it += 1
        # Anneal beta
        frac = min(1.0, it / total_iters_estimate)
        beta = beta_init * (beta_final / beta_init) ** frac
        lr = lr_init * (lr_final / lr_init) ** frac
        
        p = _softmax(z)
        grad_p, c, w = _smoothed_grad(p, beta)
        W = float(np.dot(p, grad_p))
        grad_z = p * (grad_p - W)
        
        # Momentum
        v = momentum * v + (1.0 - momentum) * grad_z
        update = v
        # Clip update
        upd_norm = np.linalg.norm(update)
        if upd_norm > 1.0:
            update = update * (1.0 / upd_norm)
        
        z = z - lr * update
        z -= z.mean()
        z = np.clip(z, -100.0, 100.0)
        
        # Periodic score check
        if it % 50 == 0:
            p_check = _softmax(z)
            sc = _score_norm(p_check)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_p = p_check.copy()
            # Adaptive LR
            if it % 500 == 0:
                if best_sc < prev_best * 0.999:
                    lr = min(0.5, lr * 1.05)
                elif best_sc > prev_best:
                    lr = max(0.001, lr * 0.9)
                prev_best = best_sc
    
    p_final = _softmax(z)
    sc_final = _score_norm(p_final)
    if np.isfinite(sc_final) and sc_final < best_sc:
        best_sc = sc_final
        best_p = p_final.copy()
    return best_p, best_sc


def _perturb_simplex(p, sigma, rng):

def _interp_to_n(p, n_new):
    """Interpolate a probability vector to a new length."""
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-15)
    return p_new / p_new.sum()

        print(f"\n=== Phase 2: Annealed gradient (deadline {phase2_end - t_start:.0f}s) ===")
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        budget2 = phase2_end - time.monotonic() - 5
        p_grad, sc_grad = _anneal_grad(
            p_work, budget2,
            beta_init=0.005 * max_cb, beta_final=0.00001 * max_cb,
            lr_init=0.1, lr_final=0.005, momentum=0.85
        )
        full_sc = _score(p_grad)
        print(f"  Gradient phase: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        _update_best(p_grad)
    
    # === Phase 3: More LP from improved point (800 - 1100s) ===
    phase3_end = min(deadline, t_start + 1100.0)
    if time.monotonic() < phase3_end:
        print(f"\n=== Phase 3: LP refinement (deadline {phase3_end - t_start:.0f}s) ===")
        p_work = best_seq.copy()
        stall = 0
        lp3_it = 0
        prev_sc = _score_norm(p_work)
        while time.monotonic() < phase3_end:
            c = _full_convolve(p_work, p_work)
            M = float(c.max())
            q = _lp_step(p_work, M, time_limit=20)
            if q is None or q.sum() < 1e-15:
                break
            q = q / q.sum()
            best_t = 0.0
            best_cand_sc = _score_norm(p_work)
            for t in [0.2, 0.4, 0.6, 1.0, 0.1]:
                cand = (1.0 - t) * p_work + t * q
                sc_cand = _score_norm(cand)
                if sc_cand < best_cand_sc:
                    best_cand_sc = sc_cand
                    best_t = t
            if best_t > 0:
                p_work = (1.0 - best_t) * p_work + best_t * q
                lp3_it += 1
                sc_now = _score_norm(p_work)
                if sc_now < prev_sc - 1e-12:
                    stall = 0
                else:
                    stall += 1
                if lp3_it % 10 == 0:
                    full_sc = _score(p_work)
                    print(f"  LP2[{lp3_it}] score={full_sc:.10f} stall={stall} t={time.monotonic()-t_start:.0f}s")
                    _update_best(p_work)
                if stall > 35:
                    break
                prev_sc = sc_now
            else:
                stall += 2
                if stall > 25:
                    break
        _update_best(p_work)
        print(f"Phase 3 done: {lp3_it} iters, best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")
    
    # === Phase 4: Fine gradient polishing (1100 - 1400s) ===
    phase4_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < phase4_end:
        print(f"\n=== Phase 4: Fine gradient (deadline {phase4_end - t_start:.0f}s) ===")
        c_best = _full_convolve(best_seq, best_seq)
        max_cb = float(c_best.max())
        budget4 = phase4_end - time.monotonic() - 5
        p_fine, sc_fine = _anneal_grad(
            best_seq, budget4,
            beta_init=0.001 * max_cb, beta_final=0.000005 * max_cb,
            lr_init=0.03, lr_final=0.001, momentum=0.9
        )
        full_sc = _score(p_fine)
        print(f"  Fine gradient: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        _update_best(p_fine)
    
    # === Phase 5: Try perturbation + short optimization (1400 - 1600s) ===
    phase5_end = min(deadline, t_start + 1600.0)
    if time.monotonic() < phase5_end:
        print(f"\n=== Phase 5: Perturbation restarts (deadline {phase5_end - t_start:.0f}s) ===")
        for attempt in range(4):
            if time.monotonic() > phase5_end - 40:
                break
            sigma = 0.02 + 0.01 * attempt
            p_pert = _perturb_simplex(best_seq, sigma, rng)
            # Quick LP
            stall = 0
            for it in range(15):
                if time.monotonic() > phase5_end - 10:
                    break
                c = _full_convolve(p_pert, p_pert)
                M = float(c.max())
                q = _lp_step(p_pert, M, time_limit=15)
                if q is None or q.sum() < 1e-15:
                    break
                q = q / q.sum()
                best_t = 0.0
                best_sc_cand = _score_norm(p_pert)
                for t in [0.3, 0.5, 0.7]:
                    cand = (1.0 - t) * p_pert + t * q
                    sc_c = _score_norm(cand)
                    if sc_c < best_sc_cand:
                        best_sc_cand = sc_c
                        best_t = t
                if best_t > 0:
                    p_pert = (1.0 - best_t) * p_pert + best_t * q
                    if _score_norm(p_pert) < _score_norm(p_pert):
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 2
                if stall > 8:
                    break
            full_sc = _score(p_pert)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            _update_best(p_pert)
    
    # === Phase 6: Try different n (1600 - deadline) ===
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 6: Different n exploration ===")
        for n_try in [1024, 4096]:
            if time.monotonic() > deadline - 60:
                break
            p_try = _interp_to_n(best_seq, n_try)
            sc_try = _score(p_try)
            print(f"  n={n_try}: initial score={sc_try:.10f}")
            # Quick anneal
            budget6 = min(120.0, deadline - time.monotonic() - 30)
            if budget6 > 30:
                c_t = _full_convolve(p_try, p_try)
                max_c_t = float(c_t.max())
                p_opt, sc_opt = _anneal_grad(
                    p_try, budget6,
                    beta_init=0.003 * max_c_t, beta_final=0.00001 * max_c_t,
                    lr_init=0.05, lr_final=0.003, momentum=0.85
                )
                full_sc = _score(p_opt)
                print(f"  n={n_try} optimized: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
                _update_best(p_opt)
    
    # === Final ultra-fine polish ===
    if time.monotonic() < deadline - 5:
        print(f"\n=== Final polish ===")
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        budget_f = deadline - time.monotonic() - 3
        if budget_f > 10:
            p_f, sc_f = _anneal_grad(
                best_seq, budget_f,
                beta_init=0.0003 * max_cf, beta_final=0.000003 * max_cf,
                lr_init=0.01, lr_final=0.0005, momentum=0.92
            )
            full_sc = _score(p_f)
            print(f"  Final: score={full_sc:.10f}")
            _update_best(p_f)
    
    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]
    
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
