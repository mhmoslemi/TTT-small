import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:

    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    c = _fft_convolve(a, a)

def _build_lp_sparse(f, n):
    """Build sparse matrix for LP: A[k,j] = f[|k-j|] if valid."""
    k_arr = np.arange(2 * n - 1)
    j_arr = np.arange(n)
    diff = k_arr[:, None] - j_arr[None, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))


def _lp_step(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    A = _build_lp_sparse(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(np.asarray(res.x), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _interp_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()

    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _max_conv(a):
    c = _fft_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z_shifted = z - np.max(z)
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_and_w(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*c_k)) w.r.t. a.
    grad_j = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    Returns (grad, w, c, cm).
    """
    n = len(a)
    c = _fft_convolve(a, a)
    cm = float(c.max())
    c_shifted = beta * (c - cm)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s

    # grad_j = 2 * sum_m a[m] * w[j+m] = 2 * conv(a_rev, w)[n-1+j]
    a_rev = a[::-1]
    N_conv = _next_pow2(3 * n - 1)
    fa = np.fft.rfft(a_rev, N_conv)
    fw = np.fft.rfft(w, N_conv)
    conv_aw = np.fft.irfft(fa * fw, N_conv)[:3 * n - 2]
    grad = 2.0 * conv_aw[n - 1:2 * n - 1]

    return grad, w, c, cm


def _adam_smoothed(a_init, t_end, beta_schedule, lr=0.02, name="",
                   check_interval=300, verbose_interval=10000,
                   clip_norm=0.3, b2=0.999):
    """
    Adam optimizer in log-space with log-sum-exp smoothed objective.
    beta_schedule: list of (beta_value, fraction_of_time) tuples.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_mc = _max_conv(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, eps = 0.9, 1e-8
    current_lr = lr

    t_start_local = time.monotonic()
    total_time = max(1.0, t_end - t_start_local)

    schedule = []
    t_cursor = t_start_local
    for beta_val, frac in beta_schedule:
        dur = frac * total_time
        schedule.append((beta_val, t_cursor, t_cursor + dur))
        t_cursor += dur

    sched_idx = 0

    while time.monotonic() < t_end:
        now = time.monotonic()
        while sched_idx < len(schedule) - 1 and now >= schedule[sched_idx][2]:
            sched_idx += 1
        beta = schedule[sched_idx][0]

        a = _softmax(z)
        grad, w, c, cm = _compute_grad_and_w(a, beta)

        # z-space gradient: a_j * (grad_j - <a, grad>)
        ag = float(np.dot(a, grad))
        grad_z = a * (grad - ag)

        if not np.all(np.isfinite(grad_z)):
            break

        t_step += 1
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > clip_norm:
            update = update * (clip_norm / update_norm)

        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        if t_step % check_interval == 0:
            a_check = _softmax(z)
            mc = _max_conv(a_check)
            if mc < best_mc:
                best_mc = mc
                best_a = a_check.copy()

            if t_step % verbose_interval == 0:
                elapsed = time.monotonic() - t_start_local
                print(f"  {name} step={t_step} beta={beta:.1f} max_c={mc:.10f} "

    mc_final = _max_conv(a_final)
    if mc_final < best_mc:
        best_mc = mc_final
        best_a = a_final.copy()

    return best_a, best_mc


def _build_lp_sparse(f, n):
    k_arr = np.arange(2 * n - 1)
    j_arr = np.arange(n)
    diff = k_arr[:, None] - j_arr[None, :]
    mask = (diff >= 0) & (diff < n)
    rows, cols = np.where(mask)
    vals = f[diff[rows, cols]]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f, M, time_limit=20):
    n = len(f)
    A = _build_lp_sparse(f, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit})
        if res.status == 0:
            g = np.maximum(np.asarray(res.x), 0)
            if g.sum() > 1e-15:
                return g
    except Exception:
        pass
    return None


def _interp_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n_old)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _make_gaussian(n, sigma_frac=0.25):
    x = np.linspace(-1, 1, n)
    sigma = sigma_frac
    a = np.exp(-x ** 2 / (2 * sigma ** 2))
    return a / a.sum()
