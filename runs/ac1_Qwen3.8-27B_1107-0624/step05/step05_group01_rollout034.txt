**The "water-filling" interpretation:**
At the optimum, the gradient of the objective (with respect to the constraint) should be "flat" in some sense. Specifically, if we write the problem as:
minimize max_k (a*a)[k] subject to sum a = 1, a >= 0

At the optimum, the active set of peaks (those achieving the max) should have "balanced" gradients. This is the KKT condition.

The KKT conditions say:
- For each active peak k: `d(a*a)[k]/da = lambda * 1 + mu` where `mu >= 0` is the dual variable for `a >= 0`.
- For each inactive peak: `d(a*a)[k]/da <= lambda * 1 + mu`

This means: at the optimum, the "influence" of each element `a[i]` on the active peaks should be roughly equal (up to the non-negativity constraint).

This gives us a practical algorithm:
1. Compute the active peaks.
2. Compute the gradient contribution from each active peak.
3. Find the element `a[i]` that has the highest "effective gradient" (contribution to active peaks) and reduce it.
4. Find the element with the lowest "effective gradient" and increase it.
5. Repeat.

This is essentially a form of coordinate descent / water-filling.

```python
"""
Multi-phase optimization of step function sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses Nesterov-accelerated gradient descent with log-sum-exp smoothing,
water-filling active-set refinement, and final coordinate polish.
"""
import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    zm = z - np.max(z)
    e = np.exp(zm)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j-1] * (2.0*j - 1.0) / (2.0*j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype):
    t = np.linspace(0, 1, n)

    elif etype == 'sqrt':

    elif etype == 'linear':

    elif etype == 'sin':

    else:

    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _compute_grad_h(a, w):
    """Compute h = (w * a_rev)[n-1:2n-1] which is the unnormalized gradient direction."""
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    return w_conv[n-1:2*n-1]

    """Nesterov-accelerated gradient descent on LSE surrogate."""
    n = len(a_init)

    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    
    # Nesterov momentum
    y = z.copy()  # lookahead point
    v = np.zeros(n)  # momentum
    
    # BB step size history
    prev_grad = None
    prev_y = None
    step = 0.1
    
    t_start = time.monotonic()
    it = 0
    no_improve = 0
    check_interval = 50

        if time.monotonic() > deadline:
            break

        frac = float(it) / total_iters_est
        beta = beta_init * (beta_final / beta_init) ** frac

        c = _full_convolve(a, a)
        cm = float(c.max())
```

The standard Nesterov accelerated gradient method is:
```
y_0 = x_0
for k = 0, 1, 2, ...:
```

```
x_0 = x_1
for k = 0, 1, 2, ...:
    x_{k+1} = x_k + momentum * (x_k - x_{k-1}) - alpha * grad(x_k)
```

Or equivalently with velocity:
```
v_0 = 0
for k = 0, 1, 2, ...:
    v_{k+1} = momentum * v_k - alpha * grad(x_k)
    x_{k+1} = x_k + v_{k+1}
```

I'll use the heavy-ball / velocity form, which is simpler and effective.

One issue: the step size `alpha` needs to be chosen carefully. If it's too large, the method diverges. If too small, it's slow.

The Barzilai-Borwein (BB) step size is a good choice:
- `alpha = ||s||^2 / |s^T y|` where `s = x_k - x_{k-1}` and `y = grad(x_k) - grad(x_{k-1})`.

But for our problem, the gradient changes nonlinearly, so BB might not be stable. A safer approach is to use a fixed step size with a "cooling" schedule, or to use backtracking line search.

Actually, for the log-sum-exp surrogate, the function is smooth with a known Lipschitz constant (related to `1/beta`), so we can use a fixed step size `alpha = 1/L` where `L` is the Lipschitz constant of the gradient.

```python
"""
Multi-phase optimization minimizing 2n*max(conv(a,a))/sum(a)^2 for non-negative a.
Combines Nesterov-accelerated gradient on LSE surrogate, water-filling active-set
refinement, spectral-domain exploration, and Adam final polish.
Key innovations over prior approach: momentum-based optimization (faster per second),

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    zm = z - np.max(z)
    e = np.exp(zm)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j-1] * (2.0*j - 1.0) / (2.0*j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype, rng):
    t = np.linspace(0, 1, n)
    if etype == 'exp':
        alpha = rng.uniform(2.0, 6.0)
        a = np.exp(alpha * (t - 0.5))

    elif etype == 'sin':

        a = 0.3 + 0.7*np.sin(np.pi*t + phase*0.1)
    elif etype == 'bimodal':
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        a = np.exp(-((t - center)**2) / (2*width**2)) + 0.1
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _compute_h(a, w):
    n = len(a)
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    return w_conv[n-1:2*n-1]


def _lse_grad_z(z, beta, n):
    """Compute gradient of 2n*LSE_beta(conv(softmax(z),softmax(z))) w.r.t. z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros(n), a, c, cm
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    h = _compute_h(a, w)
    Wa = float(np.dot(h, a))
    grad = 4.0 * n * a * (h - Wa)
    if not np.all(np.isfinite(grad)):
        return np.zeros(n), a, c, cm
    return grad, a, c, cm

def _smooth_perturbation(n, rng):
    """Generate a smooth random perturbation via low-frequency spectral synthesis."""
    N = _next_pow2(n)
    freq = np.fft.rfft(np.zeros(N))
    max_k = max(2, N // 64)
    for k in range(1, max_k):
        freq[k] = rng.randn() + 1j * rng.randn()
        freq[N-k] = np.conj(freq[k]) if N-k < N//2 else 0
    pert = np.fft.irfft(freq, N)[:n]
    pert = pert - pert.mean()
    pnorm = np.linalg.norm(pert)
    if pnorm > 1e-10:
        pert = pert / pnorm
    return pert

    def obj(z_in):
        a_in = _softmax_fn(z_in)
        c_in = _full_convolve(a_in, a_in)
        cm = float(c_in.max())
        if cm < 1e-300:
            return 0.0
        shifted = (c_in - cm) / beta
        max_s = float(shifted.max())
        lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
        return float(2.0 * n * lse)

    def grad_fn(z_in):
        g, _, _, _ = _lse_grad_z(z_in, beta, n)
        return g

    try:
        result = minimize(obj, z0, method='L-BFGS-B', jac=grad_fn,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxls': 50})
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc):
                return a_new, sc
    except Exception:
        pass

    return a_init, _score_norm_fn(a_init)


def _lbfgs_continuation(a_init, deadline, rng):
    """L-BFGS with geometric beta continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())

    fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    for frac in fracs:
        if time.monotonic() > deadline:
            break
        beta = frac * cm0

        def obj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)

        def grad_fn(z_in):
            g, _, _, _ = _lse_grad_z(z_in, beta, n)
            return g

        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad_fn,
                              options={'maxiter': 5000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 60, 'maxfun': 20000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: diverse init -> Nesterov -> water-filling -> L-BFGS continuation -> Adam polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**31 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Diverse initialization + quick optimization (250s) ===
    print("Phase 1: Diverse initialization")
    p1_end = min(deadline, t_start + 250.0)

    candidates = [best_seq.copy()]

    # Add parametric candidates
    inits = ['exp', 'pow', 'sqrt', 'bimodal', 'linear']
    for et in inits:
        a_p = _parametric_init(n, et, rng)
        candidates.append(a_p)

    # Greedy
    candidates.append(_greedy_init(n))

    # Perturb parent
    a_pert = best_seq * (1.0 + 0.1 * _smooth_perturbation(n, rng))
    a_pert = np.maximum(a_pert, 1e-12)
    a_pert = a_pert / a_pert.sum()
    candidates.append(a_pert)

    # Quick L-BFGS on each candidate
    per_cand_time = 30.0
    for idx, cand in enumerate(candidates):
        if time.monotonic() > p1_end - 10:
            break
        cand = np.maximum(cand, 1e-15)
        cand = cand / cand.sum()
        sc = _score_norm_fn(cand)
        update_best(cand, sc, f"init_{idx}")

        # Quick L-BFGS
        sub_deadline = min(p1_end - 5, time.monotonic() + per_cand_time)
        a_b, sc_b = _quick_lbfgs(cand, sub_deadline, maxiter=2000)
        update_best(a_b, sc_b, f"lbfgs_init_{idx}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Nesterov momentum (500s) ===
    print("Phase 2: Nesterov momentum optimization")
    p2_end = min(deadline, t_start + 750.0)

    if best_seq is not None and time.monotonic() < p2_end:
        c_b = _full_convolve(best_seq, best_seq)
        cm_b = float(c_b.max())
        beta_start = 0.05 * cm_b
        beta_end = 0.001 * cm_b

        a_nest, sc_nest = _nesterov_optimize(
            best_seq, p2_end, beta_start, beta_end, rng,
            momentum_init=0.8, momentum_end=0.95, lr_init=1.0)
        update_best(a_nest, sc_nest, "nesterov")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Water-filling (400s) ===
    print("Phase 3: Water-filling refinement")
    p3_end = min(deadline, t_start + 1150.0)

    if best_seq is not None and time.monotonic() < p3_end:
        # Multiple water-filling rounds with different K
        for K_val in [30, 50, 100]:
            if time.monotonic() > p3_end - 30:
                break
            sub_dl = min(p3_end - 10, time.monotonic() + (p3_end - time.monotonic()) / 3.0)
            a_wf, sc_wf = _waterfill_optimize(
                best_seq, sub_dl, K=K_val, lr_init=0.2, rng=rng)
            update_best(a_wf, sc_wf, f"waterfill_K{K_val}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: L-BFGS continuation (250s) ===
    print("Phase 4: L-BFGS continuation")
    p4_end = min(deadline, t_start + 1400.0)

    if best_seq is not None and time.monotonic() < p4_end:
        a_lbf, sc_lbf = _lbfgs_continuation(best_seq, p4_end, rng)
        update_best(a_lbf, sc_lbf, "lbfgs_cont")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: Adam final polish (300s) ===
    print("Phase 5: Adam final polish")
    p5_end = min(deadline, t_start + 1700.0)

    if best_seq is not None and time.monotonic() < p5_end:
        # Fine Adam
        sub_dl = min(p5_end - 30, time.monotonic() + (p5_end - time.monotonic()) * 0.6)
        a_ad, sc_ad = _adam_fine(best_seq, sub_dl, lr_init=0.003,
                                 beta_cooldown=1e-5, rng=rng)
        update_best(a_ad, sc_ad, "adam_fine")

        # Second Adam with different params
        if time.monotonic() < p5_end - 10:
            a_ad2, sc_ad2 = _adam_fine(best_seq, p5_end - 5, lr_init=0.001,
                                       beta_cooldown=1e-6, rng=rng)
            update_best(a_ad2, sc_ad2, "adam_very_fine")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Final Nesterov + waterfill combo on best (100s) ===
    print("Phase 6: Final combo")
    p6_end = deadline

    if best_seq is not None and time.monotonic() < p6_end - 20:
        c_b = _full_convolve(best_seq, best_seq)
        cm_b = float(c_b.max())

        sub_dl = min(p6_end - 10, time.monotonic() + 40.0)
        a_n2, sc_n2 = _nesterov_optimize(
            best_seq, sub_dl, 0.005*cm_b, 0.0005*cm_b, rng,
            momentum_init=0.93, momentum_end=0.97, lr_init=0.3)
        update_best(a_n2, sc_n2, "nesterov_final")

        if time.monotonic() < p6_end - 10:
            a_wf2, sc_wf2 = _waterfill_optimize(
                best_seq, p6_end - 5, K=80, lr_init=0.1, rng=rng)
            update_best(a_wf2, sc_wf2, "waterfill_final")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
```
