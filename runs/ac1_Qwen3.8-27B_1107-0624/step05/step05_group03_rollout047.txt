def _score_and_grad(a, beta):
    """Compute smoothed score and gradient w.r.t. a (not log a)."""
    n = len(a)
    c = _full_convolve(a, a)  # length 2n-1
    S = a.sum()
    
    # Softmax weights
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    lse = c_max + beta * np.log(e.sum())  # = logsumexp(c/beta) * ... 
    
    # Actually, LSE_beta(c) = c_max + beta * log(sum(exp((c - c_max)/beta)))
    # And the gradient of LSE w.r.t. c is w (the softmax)
    
    # Score: 2n * LSE_beta(c) / S^2
    score = 2.0 * n * lse / (S * S)
    
    # Gradient: d(score)/d(a_i) = (2n/S^2) * sum_k w_k * 2*a_{k-i} * 1[0<=k-i<n] - (4n*lse/S^3) * 1
    # The second term is constant across i, so for the log-param gradient we can ignore it
    # (it just shifts all gradient components by the same amount)
    
    # Compute g_i = sum_k w_k * 2 * a_{k-i} for valid k-i
    # Using the convolution trick: g_i/2 = (a_rev * w)[n-1+i]
    a_rev = a[::-1].copy()
    conv_g = _full_convolve(a_rev, w)  # length n + (2n-1) - 1 = 3n-2
    g = 2.0 * conv_g[n-1:n-1+n]  # extract the right portion
    
    # Full gradient (w.r.t. a, not log a)
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * lse / (S**3)) * np.ones(n)
    
    return score, grad, c, w

"""
FFT-accelerated optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations:
1. FFT-based gradient computation (O(n log n) per evaluation)
2. Aggressive logsumexp continuation with many iterations
3. Responsibility-based local search
4. Multi-scale exploration
5. Symmetry-aware optimization

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_from_z(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm(a):
    """Score for sum=1 sequence: 2n * max_conv."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_and_smoothed_grad(a, beta):
    """
    Compute smoothed score and gradient w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    c = _full_convolve(a, a)  # length 2n-1
    S = float(a.sum())
    
    # Smoothed max via logsumexp
    c_max = float(c.max())
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    s_e = float(e.sum())
    if s_e < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        lse = c_max
    else:
        w = e / s_e
        lse = c_max + beta * np.log(s_e)
    
    score = 2.0 * n * lse / (S * S)
    
    # Gradient: d(score)/d(a_i)
    # = (2n/S^2) * 2 * sum_k w_k * a_{k-i} * 1[0<=k-i<n]
    #   - (4n*lse/S^3) * 1
    # 
    # The first term: g_i = 2 * (a_rev * w)[n-1+i]
    a_rev = a[::-1].copy()
    conv_g = _full_convolve(a_rev, w)  # length n + (2n-1) - 1 = 3n-2
    g = 2.0 * conv_g[n-1:2*n-1]  # extract n elements starting at index n-1
    
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * lse / (S ** 3)) * np.ones(n)
    
    return score, grad, w

def _obj_and_grad_z(z, beta, n):
    """Objective and gradient in z-space (log parameterization)."""
    a = _softmax_from_z(z)
    score, grad_a, w = _score_and_smoothed_grad(a, beta)
    grad_z = a * grad_a  # chain rule: dF/dz_i = a_i * dF/da_i
    return score, grad_z

def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_bell_profile(n, width=0.3):
    t = np.linspace(0, 1, n)
    a = np.exp(-(t - 0.5)**2 / (2*width**2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _make_flat_profile(n):
    return np.ones(n) / n

def _make_bimodal(n, sep=0.4, width=0.15):
    t = np.linspace(0, 1, n)
    a = np.exp(-(t - 0.5 + sep/2)**2 / (2*width**2)) + np.exp(-(t - 0.5 - sep/2)**2 / (2*width**2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()

def _interp(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()

def _symmetrize(a):
    """Enforce symmetry: a_i = a_{n-1-i}."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 1e-12)
    return a_sym / a_sym.sum()

def _lbfgs_smoothed(a_init, beta, deadline, max_iter=5000):
    """Run L-BFGS on the smoothed objective at a fixed beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    def obj(z_):
        a_ = _softmax_from_z(z_)
        c_ = _full_convolve(a_, a_)
        c_max = float(c_.max())
        c_shifted = (c_ - c_max) / beta
        e_ = np.exp(c_shifted)
        s_ = float(e_.sum())
        if s_ < 1e-300:
            return 2.0 * n * c_max
        lse = c_max + beta * np.log(s_)
        S = float(a_.sum())
        return 2.0 * n * lse / (S * S)
    
    def grad(z_):
        a_ = _softmax_from_z(z_)
        sc, grad_z, _ = _obj_and_grad_z(z_, beta, n)
        return grad_z
    
    try:
        result = minimize(
            obj, z.copy(), jac=grad,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': max_iter * 20,
                     'maxcor': 20}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            z_new = result.x
            a_new = _softmax_from_z(z_new)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    except Exception:
        pass
    
    return best_a, best_score

def _continuation(a_init, deadline, beta_scale=1.0, n_levels=20, symmetrize=False):
    """Logsumexp continuation method."""
    n = len(a_init)
    a = a_init.copy()
    c0 = _full_convolve(a, a)
    max_c0 = float(c0.max())
    
    best_a = a.copy()
    best_score = _score_norm(a)
    
    # Beta schedule
    betas = max_c0 * beta_scale * np.array([1.0, 0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.013, 0.008, 0.005, 0.003, 0.002, 0.001, 0.0006, 0.0003, 0.0001])[:n_levels]
    
    per_level = max(5.0, (deadline - time.monotonic()) / max(1, n_levels))
    
    for level, beta in enumerate(betas):
        t_level_start = time.monotonic()
        if time.monotonic() > deadline - 5:
            break

        if symmetrize:
            a = _symmetrize(a)
            sc = _score_norm(a)
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  Cont level {level}: beta={beta:.6f}, score={sc:.10f}, time={time.monotonic()-t_level_start:.1f}s")
    
    return best_a, best_score

def _local_search(a_init, deadline, n_trials=50000, symmetrize=False):
    """Peak-targeted local search using gradient information."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1:
            break
        
        c = _full_convolve(a, a)
        max_lag = int(np.argmax(c))
        max_val = c[max_lag]
        
        # Compute gradient at the peak lag
        # g_i = 2 * a_{max_lag - i} (for valid indices)
        # This tells us which positions contribute to the peak
        
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        partners = max_lag - np.arange(i_min, i_max + 1)
        contributions = a[i_min:i_max+1] * a[partners]
        
        # Find the position with highest contribution to the peak
        peak_contrib_idx = i_min + int(np.argmax(contributions))
        
        # Find "safe" positions: those with low contribution to top few lags
        top_k = min(5, 2*n-1)
        top_lags = np.argsort(c)[-top_k:]
        
        contrib_sum = np.zeros(n)
        for kl in top_lags:
            kl = int(kl)
            lo = max(0, kl - (n-1))
            hi = min(n-1, kl)
            for j in range(lo, hi+1):
                p = kl - j
                if 0 <= p < n:
                    contrib_sum[j] += a[p]
        
        # Strategy: move mass from peak-contributing position to safe position
        if it % 2 == 0:
            safe_positions = np.argsort(contrib_sum)[:max(1, n//4)]
            target = int(safe_positions[np.random.randint(len(safe_positions))])
            frac = np.random.uniform(0.05, 0.3)
            delta = a[peak_contrib_idx] * frac
            
            a_trial = a.copy()
            a_trial[peak_contrib_idx] -= delta
            a_trial[peak_contrib_idx] = max(0, a_trial[peak_contrib_idx])
            a_trial[target] += delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                if symmetrize:
                    a_trial = _symmetrize(a_trial)
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        # Strategy 2: Random pair perturbation
        else:
            i1 = np.random.randint(n)
            i2 = np.random.randint(n)
            if i1 == i2:
                continue
            frac = np.random.uniform(0.05, 0.25)
            delta = frac * min(a[i1], a[i2])
            
            direction = np.random.choice([-1, 1])
            a_trial = a.copy()
            a_trial[i1] -= direction * delta
            a_trial[i2] += direction * delta
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                if symmetrize:
                    a_trial = _symmetrize(a_trial)
                sc = _score_norm(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
    
    return best_a, best_sc

def _adam_optimize(a_init, deadline, lr=0.01, n_steps=20000, symmetrize=False):
    """Adam optimizer in log-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for step in range(n_steps):
        if time.monotonic() > deadline - 1:
            break
        t += 1
        a = _softmax_from_z(z)
        grad_a = _score_and_smoothed_grad(a, 0.001)[1]  # Use very small beta (near max)
        grad_z = a * grad_a
        
        m = b1 * m + (1-b1) * grad_z
        v = b2 * v + (1-b2) * grad_z**2
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        if symmetrize:
            a_sym = _softmax_from_z(0.5*(z + z[::-1]))
            z = np.log(np.maximum(a_sym, 1e-15))
            z = z - z.mean()
        
        if step % 10 == 0:
            a_new = _softmax_from_z(z)
            sc = _score_norm(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        
        # Decay learning rate
        lr = max(0.0005, 0.01 * (0.999 ** t))
    
    return best_a, best_score
