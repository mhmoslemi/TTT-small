"""
Multi-phase optimizer for minimizing 2*n*max(convolve(a,a))/sum(a)^2 over non-negative sequences.
Uses adaptive FISTA with geometry-aware smoothing, Adam log-space fine tuning,
subgradient polishing on the non-smooth objective, and multi-scale exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _top_spread(c, frac=0.1):
    """Compute the spread of the top fraction of convolution values."""
    n = len(c)
    k = max(1, int(n * frac))
    sorted_c = np.sort(c)[::-1]
    top_k = sorted_c[:k]
    return float(top_k.max() - top_k.min())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Gradient of sum(w * conv(a,a)) with respect to a (middle part)."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

    c_init = _full_convolve(a_curr, a_curr)
    max_c_init = float(c_init.max())
    best_score = 2.0 * n * max_c_init
    best_a = a_curr.copy()

    # Start with moderate beta
    spread_init = _top_spread(c_init, 0.05)
    beta = max(0.02, spread_init * 0.5)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())

        # Adaptive beta: shrink when top is flat
        spread = _top_spread(c_y, 0.05)
        beta = max(beta * 0.995, 1e-8 * max_c_init)
        # If top is very flat, can use smaller beta
        if spread < beta * 0.3:
            beta = spread * 0.1

        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_conv(y, w)

        # Lipschitz estimate: L = 2 * max(|w_convolved_with_a_rev|) approx
        L = 4.0 * n / beta
        # Cap step size for stability
        step = 1.0 / min(L, 100.0 * n)

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, betas_schedule, lr_init, max_iter_total):
    """Adam optimization in log-space with softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in betas_schedule:
        if time.monotonic() > deadline:
            break
        remain = max_iter_total - t
        if remain <= 0:
            break
        it_phase = 0
        while it_phase < remain:
            if time.monotonic() > deadline:
                break
            it_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            # Gradient of smoothed objective w.r.t. z
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)

            if not np.all(np.isfinite(grad)):
                grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_phase % 3000 == 0:
                current_lr = max(0.002, current_lr * 0.97)

    a_final = _softmax_compute(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _subgradient_polish(a_init, deadline, max_iter=50000):
    """Direct subgradient method on the non-smooth max objective."""
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_curr.copy()
    best_score = _score_norm(a_curr)

    lr = 1.0 / (4.0 * n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        cm = float(c.max())

        # Find active set (within small tolerance of max)
        tol = 1e-6 * cm + 1e-12
        active = c >= (cm - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = active.astype(np.float64) / active.sum()

        grad = _compute_grad_conv(a_curr, w)

        # Polyak-like step size
        sc_curr = 2.0 * n * cm
        sc_best = 2.0 * n * float(_score_norm(best_a) / (2.0 * n))
        # Simple diminishing step
        step = lr / (1.0 + 0.01 * it)

        a_next = a_curr - step * grad
        a_next = np.maximum(a_next, 0.0)
        s_next = a_next.sum()
        if s_next > 1e-30:
            a_next = a_next / s_next
        else:
            a_next = best_a.copy()

        sc_next = _score_norm(a_next)
        if sc_next < best_score:
            best_score = sc_next
            best_a = a_next.copy()
            a_curr = a_next.copy()
        elif it % 100 == 0:
            # If no improvement for a while, restart from best
            a_curr = best_a.copy()

    return best_a, best_score


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "u_shape":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.cos(np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.3
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()

            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent.copy()
        sc0 = _score(a0)
        print(f"Parent: n={n_main}, score={sc0:.10f}")

        a0 = _make_profile(n_main, "uniform")
        sc0 = _score(a0)
        print(f"No parent, uniform n={n_main}, score={sc0:.10f}")
        best_seq = a0.copy()

    max_c0 = float(c0.max())
    print(f"max_conv={max_c0:.10f}, top_spread={_top_spread(c0, 0.05):.10f}")

    # === Phase 1: FISTA with adaptive beta ===
    print("Phase 1: FISTA adaptive")
    phase1_end = min(deadline, t_start + 200.0)
    a_curr = a0.copy()

    # Coarse to fine with adaptive beta
    n_phases = 5
    per_phase = (phase1_end - time.monotonic()) / n_phases
    for p in range(n_phases):
        if time.monotonic() > phase1_end - 2:
            break
        phase_deadline = min(phase1_end, time.monotonic() + per_phase * 1.2)
        a_curr, sc = _fista_adaptive(a_curr, phase_deadline, 50000)
        a_curr = a_curr / a_curr.sum()
        sc_full = _score(a_curr)
        if sc_full < best_score:
            best_score = sc_full
            best_seq = a_curr.copy()
            print(f"  FISTA phase {p}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    print(f"End Phase 1: best={best_score:.10f}")

    # === Phase 2: Adam log-space ===
    print("Phase 2: Adam log-space")
    phase2_end = min(deadline, t_start + 700.0)
    a_work = best_seq.copy()
    c_work = _full_convolve(a_work, a_work)
    max_cw = float(c_work.max())

    betas_fine = []
    beta = 0.005 * max_cw
    while beta > 1e-6 * max_c0 and len(betas_fine) < 12:
        betas_fine.append(beta)

        if sc_opt_full < best_score:

            elif i % 5 == 1:

                noise = rng.standard_normal(len(z)) + 1j * rng.standard_normal(len(z))
                noise[0] = 0
                z = z + 0.01 * noise
                a_pert = np.fft.irfft(z, len(a_work))
                a_pert = np.maximum(a_pert, 1e-10)

            elif i % 5 == 3:

                half = len(a_work) // 2
