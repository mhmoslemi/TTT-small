def _next_pow2(n):

    while k < n:

    n = len(a_init)

    beta = max_c0 * 0.05

        if time.monotonic() > deadline - 1.0:
            break
        
        # Nesterov lookahead

        a_curr = _softmax_fn(z_look)

        h = w_conv[n-1:2*n-1]

        grad = 4.0 * n * a_curr * (h - W)

                best_a = a_new.copy()

The coordinate descent is the key new element. Let me implement it properly.

**Coordinate Descent with Exact 1D Minimization:**

For each coordinate i:
- Fix all a_j for j != i
- Let S' = sum_{j != i} a_j (sum excluding coordinate i)
- Current value: a_i = x_old
- We want to find x >= 0 that minimizes 2n * max_k c_new(k) / (S' + x)^2

where c_new(k) = c_old(k) + delta_k(x)

The delta depends on k:
- For k where k-i is in [0, n-1] and k != 2i:
  delta_k(x) = 2*(x - x_old) * a[k-i]
- For k = 2i (if 2i is in [0, 2n-2]):
  delta_k(x) = x^2 - x_old^2
- For other k:
  delta_k(x) = 0

So the function to minimize over x is:
f(x) = 2n * max_k [c_old(k) + delta_k(x)] / (S' + x)^2

This is a piecewise function (the max switches which k is active). We can find the optimum by:
1. Identifying the "critical" points where the active k changes
2. Evaluating f(x) at each critical point and at x=0
3. Taking the minimum

Or more simply: use golden section search on f(x) over [0, x_upper] where x_upper is some reasonable bound.

For the upper bound: if x is too large, the convolution entry at k=2i becomes x^2, and the score becomes 2n*x^2/(S'+x)^2 which grows quadratically. So the optimum is bounded.

A reasonable upper bound: x_upper = max(a) * 2 or x_upper = S' (the sum of all other elements).

def _coord_descent_step(a, i, c_current, S):
    """
    Find optimal value for a[i] given all other elements fixed.
    Returns (optimal_x, new_c, new_S, new_score).
    """
    n = len(a)
    x_old = a[i]
    S_prime = S - x_old  # Sum excluding a[i]
    
    # Precompute the "influence" of a[i] on each convolution entry
    # c_new[k] = c_current[k] + (x - x_old) * infl[k] + quad[k] * x^2
    # where infl[k] = 2*a[k-i] (if k-i in range, and k != 2i)
    # and quad[k] = 1 if k == 2i, else 0
    
    infl = np.zeros(2*n-1)
    for k in range(2*n-1):
        j = k - i
        if 0 <= j < n:
            if k == 2*i:
                # Quadratic term: x^2 - x_old^2
                # c_new[k] = c_current[k] - x_old^2 + x^2
                # = c_current[k] + (x^2 - x_old^2)
                # We handle this separately
                infl[k] = -2 * x_old  # c_new[k] = c_current[k] - x_old^2 + x^2 = c_current[k] + x^2 - x_old^2
                # Actually, let's just handle k=2i specially
            else:
                infl[k] = 2.0 * a[j]
    
    # For k = 2i: c_new[2i] = c_current[2i] - x_old^2 + x^2
    # For other k where k-i in [0,n-1]: c_new[k] = c_current[k] + 2*(x-x_old)*a[k-i]
    # For other k: c_new[k] = c_current[k]
    
    # We want to minimize f(x) = 2n * max_k c_new(k) / (S_prime + x)^2
    # over x >= 0
    
    # Golden section search
    # Upper bound: x where the score starts increasing
    # At x = x_old, score = current score
    # For large x, c_new[2i] ≈ x^2, score ≈ 2n*x^2/(S_prime+x)^2 → 2n
    # We want to stay in the region where score < current_score
    x_upper = max(x_old * 3, S_prime * 0.1, 1e-6)
    
    # Golden section
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ≈ 0.618
    
    def f_of_x(x):
        # Compute max convolution
        c_new = c_current.copy()
        for k in range(2*n-1):
            j = k - i
            if 0 <= j < n:
                if k == 2*i:
                    c_new[k] = c_current[k] - x_old*x_old + x*x
                else:
                    c_new[k] = c_current[k] + 2.0*(x - x_old) * a[j]
        max_c = c_new.max()
        s = S_prime + x
        if s < 1e-15:
            return np.inf
        return 2.0 * n * max_c / (s * s)
    
    # Golden section search
    a_gs = 0.0
    b_gs = x_upper
    c_gs = b_gs - gr * (b_gs - a_gs)
    d_gs = a_gs + gr * (b_gs - a_gs)
    
    fc = f_of_x(c_gs)
    fd = f_of_x(d_gs)
    
    for _ in range(20):  # 20 iterations of golden section gives ~gr^20 ≈ 1e-5 relative precision
        if fc < fd:
            b_gs = d_gs
            d_gs = c_gs
            fd = fc
            c_gs = b_gs - gr * (b_gs - a_gs)
            fc = f_of_x(c_gs)
        else:
            a_gs = c_gs
            c_gs = d_gs
            fc = fd
            d_gs = a_gs + gr * (b_gs - a_gs)
            fd = f_of_x(d_gs)
    
    x_opt = (a_gs + b_gs) / 2
    
    return x_opt, f_of_x(x_opt)

def _eval_coord(a, i, x, c_current, S_prime, S_old):
    """Compute score when a[i] = x, given current c and sum."""
    n = len(a)
    x_old = a[i]
    
    # c_new[k] = c_current[k] + 2*(x-x_old)*a[k-i] for k-i in [0,n-1], k != 2i
    # c_new[2i] = c_current[2i] - x_old^2 + x^2
    # Other k: c_new[k] = c_current[k]
    
    # The maximum of c_new is either:
    # 1. c_current[k] for some k not affected by i (i.e., k-i not in [0,n-1])
    # 2. c_current[k] + 2*(x-x_old)*a[k-i] for affected k (k != 2i)
    # 3. c_current[2i] - x_old^2 + x^2 (if 2i in [0, 2n-2])
    
    # Category 1: max of c_current over unaffected k
    # Unaffected k: those where k-i not in [0, n-1]
    # i.e., k < i or k > i+n-1
    # These are k in [0, i-1] and k in [i+n, 2n-2]
    
    # For efficiency, precompute:
    # - c_unaffected_max = max(c_current[0:i] + c_current[i+n:2n-1])
    # - affected indices: k in [i, i+n-1] (clipped to [0, 2n-2])
    #   with k != 2i
    
    # Actually, let's just compute it directly with numpy for clarity.
    delta = 2.0 * (x - x_old)
    
    # Build c_new efficiently
    c_new = c_current.copy()  # This is O(n) which is fine
    lo = max(0, i)  # k >= i
    hi = min(2*n-2, i+n-1)  # k <= i+n-1
    if lo <= hi:
        # For k in [lo, hi], k-i is in [0, n-1]
        partners = np.arange(lo, hi+1) - i
        c_new[lo:hi+1] += delta * a[partners]
        # Fix k = 2i
        if 2*i >= lo and 2*i <= hi:
            # c_new[2i] should be c_current[2i] - x_old^2 + x^2
            # But we already added delta * a[i] = 2*(x-x_old)*a[i]
            # So c_new[2i] = c_current[2i] + 2*(x-x_old)*a[i]
            # We need: c_current[2i] - x_old^2 + x^2
            # Difference: x^2 - x_old^2 - 2*(x-x_old)*x_old = (x-x_old)^2
            # So we need to add (x-x_old)^2 to c_new[2i]
            c_new[2*i] += (x - x_old)**2
    
    max_c = c_new.max()
    s = S_prime + x
    if s < 1e-15:
        return np.inf
    return 2.0 * n * max_c / (s * s)

"""
Step function optimizer minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS continuation, Nesterov-accelerated gradient, coordinate descent
with exact 1D line search, and multi-scale exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):
    t = np.linspace(0.0, 1.0, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (t - 0.5) * alpha))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bimodal_profile(n, separation, width):
    t = np.linspace(0.0, 1.0, n)
    c1 = 0.5 - separation / 2.0
    c2 = 0.5 + separation / 2.0
    a = np.exp(-(t - c1) ** 2 / (2 * width ** 2)) + np.exp(-(t - c2) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_w_profile(n, freq, depth):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + depth * np.cos(2.0 * np.pi * freq * t)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_power_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    a = (1.0 - np.abs(2.0 * t - 1.0)) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bspline_profile(n, order=4):
    t = np.linspace(0.0, 1.0, n)
    u = np.clip(t * 0.95 + 0.025, 0.01, 0.99)
    a = u ** (order - 1) * (1.0 - u) ** (order - 1)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_linear_profile(n, slope):
    t = np.linspace(0.0, 1.0, n)
    a = 1.0 + slope * (t - 0.5)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _compute_logsumexp_grad(z, beta):
    """Compute gradient of 2n*logsumexp(conv(softmax(z),softmax(z))/beta) w.r.t. z."""
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s_e = e.sum()
    if s_e < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s_e
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n)
    return grad, a, c, cm


def _compute_logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    c_shifted = (c - cm) / beta
    e = np.exp(c_shifted)
    s_e = e.sum()
    if s_e < 1e-300:
        return 2.0 * n * cm
    return 2.0 * n * (cm + beta * np.log(s_e))


def _lbfgs_continuation(a_init, deadline, rng, max_wall_time=300.0, beta_scale=1.0):
    """L-BFGS with logsumexp continuation on softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    betas = [b * max_c0 * beta_scale for b in beta_starts]

    if max_wall_time > 0:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 2.0:
            break
        level_end = min(deadline - 1.0, time.monotonic() + per_level)

        def obj_fn(z_):
            return _compute_logsumexp_obj(z_, beta)

        def grad_fn(z_):
            g, _, _, _ = _compute_logsumexp_grad(z_, beta)
            return g

        try:
            result = minimize(
                obj_fn, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 20000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass
        if time.monotonic() > level_end:
            continue

    return best_a, best_score


def _nesterov_opt(a_init, deadline, rng, total_iters=30000):
    """Nesterov-accelerated gradient descent with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Beta schedule
    n_levels = 8
    per_level = max(10, total_iters // n_levels)
    momentum = 0.85
    lr = 0.008

    for level in range(n_levels):
        if time.monotonic() > deadline - 3.0:
            break
        beta = max_c0 * (0.05 * (0.4 ** level))

        velocity = np.zeros(n)
        z_local = z.copy()

        for it in range(per_level):
            if time.monotonic() > deadline - 2.0:
                break

            # Nesterov lookahead
            z_look = z_local + momentum * velocity

            grad, a_curr, c_curr, cm = _compute_logsumexp_grad(z_look, beta)

            # Update
            velocity = momentum * velocity - lr * grad
            z_local = z_local + velocity
            z_local = np.clip(z_local, -50, 50)
            z_local = z_local - z_local.mean()

            # Periodic evaluation
            if it % 10 == 0:
                a_new = _softmax_fn(z_local)
                sc = _score_norm_fn(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()

        z = z_local.copy()
        lr *= 0.85
        momentum = min(0.95, momentum + 0.02)

    return best_a, best_score


def _coord_descent(a_init, deadline, rng, n_sweeps=50):
    """Coordinate descent with golden section search per coordinate."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 3.0:
            break

        c_current = _full_convolve(a, a)
        S = float(a.sum())

        # Shuffle coordinates
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline - 2.0:
                break

            x_old = a[i]
            S_prime = S - x_old

            if S_prime < 1e-12:
                continue

            # Precompute influence
            # c_new[k] = c_current[k] + 2*(x-x_old)*a[k-i] for k-i in [0,n-1], k != 2i
            # c_new[2i] = c_current[2i] - x_old^2 + x^2 (if 2i in [0,2n-2])

            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)

            if lo_k > hi_k:
                continue

            partners = np.arange(lo_k, hi_k + 1) - i
            affected_vals = c_current[lo_k:hi_k + 1].copy()
            cond_val = 2.0 * a[partners]

            # Handle quadratic term at k=2i
            has_quad = (2 * i >= lo_k) and (2 * i <= hi_k)
            quad_idx = 2 * i - lo_k if has_quad else -1

            def eval_x(x):
                # Compute max of c_new
                new_affected = affected_vals + 2.0 * (x - x_old) * cond_val
                if has_quad:
                    # c_new[2i] = c_current[2i] - x_old^2 + x^2
                    new_affected[quad_idx] = c_current[2 * i] - x_old * x_old + x * x
                max_affected = new_affected.max()
                # Also need max over unaffected entries
                # Unaffected: k < lo_k or k > hi_k
                if lo_k > 0:
                    max_unaffected_lo = c_current[:lo_k].max()
                else:
                    max_unaffected_lo = 0.0
                if hi_k < 2 * n - 2:
                    max_unaffected_hi = c_current[hi_k + 1:].max()
                else:
                    max_unaffected_hi = 0.0
                max_unaffected = max(max_unaffected_lo, max_unaffected_hi, 0.0)
                max_c = max(max_affected, max_unaffected)
                s = S_prime + x
                if s < 1e-15:
                    return np.inf
                return 2.0 * n * max_c / (s * s)

            # Golden section search on [0, x_upper]
            x_upper = max(x_old * 3.0, S_prime * 0.05, 1e-8)
            # Don't go too far above current
            x_upper = min(x_upper, x_old * 5.0 + S_prime * 0.1)

            a_gs = 0.0
            b_gs = x_upper
            c_gs = b_gs - gr * (b_gs - a_gs)
            d_gs = a_gs + gr * (b_gs - a_gs)
            fc = eval_x(c_gs)
            fd = eval_x(d_gs)

            for _ in range(18):
                if fc < fd:
                    b_gs = d_gs
                    d_gs = c_gs
                    fd = fc
                    c_gs = b_gs - gr * (b_gs - a_gs)
                    fc = eval_x(c_gs)
                else:
                    a_gs = c_gs
                    c_gs = d_gs
                    fc = fd
                    d_gs = a_gs + gr * (b_gs - a_gs)
                    fd = eval_x(d_gs)

            x_opt = (a_gs + b_gs) / 2.0
            f_opt = eval_x(x_opt)
            f_old = eval_x(x_old)

            if f_opt < f_old * 1.0001:
                a[i] = x_opt
                S = S_prime + x_opt
                # Update c_current incrementally
                delta = 2.0 * (x_opt - x_old)
                c_current[lo_k:hi_k + 1] += delta * cond_val
                if has_quad:
                    c_current[2 * i] = c_current[2 * i] - x_old * x_old + x_opt * x_opt - delta * a[i]
                    # Actually let's just recompute the affected part
                    # c_current[2i] should be: original + 2*(x_opt-x_old)*a[i] + (x_opt-x_old)^2
                    # But we already added 2*(x_opt-x_old)*a[i], so add (x_opt-x_old)^2
                    # Wait, we already set c_current[2i] above. Let me just recompute.
                    pass
                sc = _score_norm_fn(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            else:
                # No improvement, revert (a[i] unchanged)
                pass

        # Recompute c_current after full sweep
        if sweep % 5 == 4:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def _coord_descent_fast(a_init, deadline, rng, n_sweeps=100):
    """Faster coordinate descent using approximate evaluation."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline - 3.0:
            break

        c_current = _full_convolve(a, a)
        S = float(a.sum())
        max_c = float(c_current.max())

        order = rng.permutation(n)
        improved = 0

        for i in order:
            if time.monotonic() > deadline - 2.0:
                break

            x_old = a[i]
            S_prime = S - x_old
            if S_prime < 1e-12:
                continue

            lo_k = max(0, i)
            hi_k = min(2 * n - 2, i + n - 1)
            if lo_k > hi_k:
                continue

            partners = np.arange(lo_k, hi_k + 1) - i
            cond_val = 2.0 * a[partners]
            has_quad = (2 * i >= lo_k) and (2 * i <= hi_k)
            quad_idx = 2 * i - lo_k if has_quad else -1

            # Try a few candidate values
            candidates_x = [x_old * 0.5, x_old * 0.75, x_old * 1.25, x_old * 1.5]
            # Also try values that would reduce the peak
            peak_k = int(np.argmax(c_current))
            if lo_k <= peak_k <= hi_k:
                j_peak = peak_k - i
                if j_peak >= 0 and j_peak < n:
                    # Reduce x to reduce c[peak_k]
                    target_x = x_old * 0.3
                    candidates_x.append(target_x)
                    candidates_x.append(x_old * 0.1)

            best_x = x_old
            best_f = 2.0 * n * max_c / (S * S)

            for x_cand in candidates_x:
                x_cand = max(0.0, x_cand)
                # Quick evaluation
                new_affected = c_current[lo_k:hi_k + 1] + 2.0 * (x_cand - x_old) * cond_val
                if has_quad:
                    new_affected[quad_idx] = c_current[2 * i] - x_old * x_old + x_cand * x_cand
                max_aff = new_affected.max()

                # Unaffected max (precompute once per sweep)
                if lo_k > 0:
                    ma_lo = c_current[:lo_k].max()
                else:
                    ma_lo = 0.0
                if hi_k < 2 * n - 2:
                    ma_hi = c_current[hi_k + 1:].max()
                else:
                    ma_hi = 0.0
                max_c_new = max(max_aff, ma_lo, ma_hi, 0.0)

                s_new = S_prime + x_cand
                if s_new < 1e-15:
                    continue
                f_cand = 2.0 * n * max_c_new / (s_new * s_new)

                if f_cand < best_f:
                    best_f = f_cand
                    best_x = x_cand

            if best_x != x_old:
                delta = best_x - x_old
                c_current[lo_k:hi_k + 1] += 2.0 * delta * cond_val
                if has_quad:
                    c_current[2 * i] = c_current[2 * i] - 2.0 * x_old * best_x + best_x * best_x
                    # More carefully: c_new[2i] = c_old[2i] - x_old^2 + x_new^2
                    # We added 2*(x_new - x_old)*a[i] to c_current[2i]
                    # c_current[2i] now = c_old[2i] + 2*(x_new-x_old)*a[i]
                    # We want: c_old[2i] - x_old^2 + x_new^2
                    # Difference: -x_old^2 + x_new^2 - 2*(x_new-x_old)*a[i]
                    # Note a[i] = x_old (the old value)
                    # = -x_old^2 + x_new^2 - 2*x_new*x_old + 2*x_old^2
                    # = x_new^2 - 2*x_new*x_old + x_old^2
                    # = (x_new - x_old)^2
                    # So we need to add (x_new-x_old)^2
                    c_current[2 * i] += delta * delta
                a[i] = best_x
                S = S_prime + best_x
                improved += 1

        if improved == 0 and sweep > 3:
            break

        if sweep % 3 == 2:
            sc = _score_norm_fn(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score
