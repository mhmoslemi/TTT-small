Continuation-based L-BFGS and mirror descent optimization for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Combines smooth optimization (L-BFGS) with non-smooth fine-tuning (mirror descent)
and LP-based polishing.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _log_sum_exp(x, axis=None):
    m = np.max(x, axis=axis, keepdims=True)
    return (m + np.log(np.sum(np.exp(x - m), axis=axis, keepdims=True))).squeeze(axis) if axis is not None else m + np.log(np.sum(np.exp(x - m)))
