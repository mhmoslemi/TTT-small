def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    if s < 1e-15:

    return float(2.0 * n * c.max() / (s * s))

    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return lse, a, c, cm

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses Nesterov-accelerated log-sum-exp, direct subgradient peak-shaving,
structured Fourier perturbations, and L-BFGS continuation.
"""
import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0, a, c, cm
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return lse, a, c, cm


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z), a, c, cm
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z), a, c, cm
    return grad, a, c, cm


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    a = a / a.sum()
    return a


def _param_init(n, etype='uniform'):
    t = np.linspace(0, 1, n)
    if etype == 'uniform':
        a = np.ones(n)
    elif etype == 'tri':
        a = 1.0 - np.abs(2.0 * t - 1.0)
    elif etype == 'bimodal':
        a = 0.1 + 0.9 * (t ** 2 + (1.0 - t) ** 2)
    elif etype == 'exp_asym':
        a = np.exp(4.0 * (t - 0.5))
    elif etype == 'sqrt':
        a = np.sqrt(np.maximum(0.01 + 0.99 * t, 1e-8))
    elif etype == 'plateau':
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.1)
    elif etype == 'chirp_abs':
        a = np.abs(np.cos(np.pi * t * 3.0)) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _nesterov_lse(a_init, deadline, beta_start_frac=0.05, beta_decay=0.97,
                   min_beta_frac=0.00005, lr=0.3, max_iter=200000):
    """Nesterov accelerated gradient on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = beta_start_frac * cm0
    min_beta = min_beta_frac * cm0

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    y = z.copy()
    z_prev = z.copy()
    gamma = 1.0
    t = 0
    stagnation = 0
    check_every = 10
    rng = np.random.default_rng(12345)

    while time.monotonic() < deadline:
        t += 1
        gamma_new = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0
        y = z + (gamma / gamma_new) * (z - z_prev)
        z_prev = z.copy()
        gamma = gamma_new

        grad_y, a_y, c_y, cm_y = _lse_grad(y, beta, n)

        if not np.all(np.isfinite(grad_y)):
            break

        gn = float(np.linalg.norm(grad_y))
        if gn > 1e-30:
            scale = max(cm_y, 1e-15)
            step = lr * scale / (gn + 1e-8)
        else:
            step = lr * max(cm_y, 1e-15)
        step = min(step, 2.0 * max(cm_y, 1e-15))

        z = y - step * grad_y
        z = np.clip(z, -100, 100)
        z = z - z.mean()

        if t % check_every == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                stagnation = 0
            else:
                stagnation += 1

            beta *= beta_decay
            if beta < min_beta:
                beta = min_beta

            if stagnation > 20:
                k_pert = rng.integers(1, max(2, n // 4))
                amp = 0.05 * cm_y / n * n
                phase = rng.uniform(0, 2 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * k_pert * np.arange(n) / n + phase)
                z = z - z.mean()
                stagnation = 0
                beta = max(beta, 0.02 * cm_y)
                if stagnation > 80:
                    break
                break
            if stagnation > 80:
                break

    return best_a, best_score


def _subgradient_peak_shave(a_init, deadline, lr_init=0.5, max_iter=100000):
    """Direct subgradient method on the non-smooth max-convolution problem."""
    n = len(a_init)
    p = a_init.copy()
    p = p / p.sum()
    best_a = p.copy()
    best_score = _score_norm(p)

    lr = lr_init
    t = 0
    stagnation = 0
    f_star_est = best_score / (2.0 * n)

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(p, p)
        cm = float(c.max())
        current_val = cm

        # Polyak step size
        if current_val > f_star_est + 1e-12:
            gap = current_val - f_star_est
        else:
            gap = 1e-6 * cm

        # Compute subgradient: convex combo of gradients at peak indices
        peak_idx = np.where(c >= cm * (1.0 - 1e-8))[0]
        K = len(peak_idx)
        if K == 0:
            peak_idx = np.array([int(np.argmax(c))])
            K = 1

        # Subgradient dsp[j] = sum_k (1/K) * 2*p[k-j] (with valid range)
        # Efficiently: dsp = (2/K) * (convolve(indicator_peaks_reversed, p))[n-1:2n-1]
        w = np.zeros(2 * n - 1)
        w[peak_idx] = 1.0 / K
        w_rev = w[::-1]
        conv_w = _full_convolve(w_rev, p)
        dsp = (2.0 / K) * conv_w[n - 1:2 * n - 1]

        # Fix the diagonal (where 2j=k, gradient is p[j] not 2*p[k-j])
        # For each peak k, if k is even, position j=k/2 has gradient p[j] not 2*p[j]
        for k in peak_idx:
            if k % 2 == 0:
                j = k // 2
                if 0 <= j < n:
                    dsp[j] = dsp[j] - p[j] / K + p[j] / K  # correction: 2p[j]/K -> p[j]/K
                    # Actually: contribution from this k to dsp[j] should be p[j]/K not 2*p[j]/K
                    # We already added 2*p[k-j]/K = 2*p[j]/K (since k-j=j)
                    # So we need to subtract p[j]/K
                    dsp[j] = dsp[j] - p[j] / K

        dsn = float(np.linalg.norm(dsp))
        if dsn < 1e-30:
            break

        # Step size: Polyak
        alpha = gap / (dsn * dsn + 1e-30)
        alpha = min(alpha, 1.0)

        p_new = p - alpha * dsp
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-15:
            break
        p_new = p_new / s

        sc_new = _score_norm(p_new)
        if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
            best_score = sc_new
            best_a = p_new.copy()
            f_star_est = sc_new / (2.0 * n)
            stagnation = 0
        else:
            stagnation += 1
            lr *= 0.95
            alpha = alpha * 0.5
            p_new = p - alpha * dsp
            p_new = np.maximum(p_new, 0.0)
            s = p_new.sum()
            if s < 1e-15:
                break
            p_new = p_new / s
            sc_new = _score_norm(p_new)
            if np.isfinite(sc_new) and sc_new < best_score - 1e-12:
                best_score = sc_new
                best_a = p_new.copy()
                f_star_est = sc_new / (2.0 * n)
                stagnation = 0

        p = p_new

        if stagnation > 200:
            break

    return best_a, best_score


def _lbfgs_lse(a_init, deadline, beta_list=None, maxiter=10000):
    """L-BFGS on log-sum-exp smoothed objective."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def obj(z_in):
            val, _, _, _ = _lse_obj(z_in, beta, n)
            return val

        def grad(z_in):
            g, _, _, _ = _lse_grad(z_in, beta, n)
            return g

        try:
            result = minimize(obj, z, method='L-BFGS-B', jac=grad,
                              options={'maxiter': maxiter, 'ftol': 1e-22,
                                       'gtol': 1e-14, 'maxls': 50})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _adam_lse(a_init, deadline, lr=0.05, max_iter=500000):
    """Adam optimizer with adaptive log-sum-exp."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    beta = 0.01 * cm0

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30
    rng = np.random.default_rng(999)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-15 * cm, cm0 * 0.01 * (0.98 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 40:
                k_pert = rng.integers(1, max(2, n // 8))
                amp = 0.08 * cm / n
                phase = rng.uniform(0, 2 * np.pi)
                z = z + amp * np.cos(2.0 * np.pi * k_pert * np.arange(n) / n + phase)
                z = z - z.mean()
                no_improve = 0
                if no_improve > 120:
                    break
                break
            if no_improve > 120:
                break

    return best_a, best_score


def _fourier_perturb_optimize(a_init, deadline, num_modes=20):
    """Optimize by perturbing along Fourier cosine modes."""
    n = len(a_init)
    p = a_init.copy()
    p = p / p.sum()
    best_a = p.copy()
    best_score = _score_norm(p)
    rng = np.random.default_rng(777)

    # Pre-compute Fourier modes
    modes = []
    for k in range(1, num_modes + 1):
        modes.append(np.cos(2.0 * np.pi * k * np.arange(n) / n))

    for it in range(500):
        if time.monotonic() > deadline:
            break

        # Pick a random mode and direction
        mi = rng.integers(0, num_modes)
        sign = 1.0 if rng.random() > 0.5 else -1.0
        # Try different step sizes
        best_local = best_score
        best_local_a = best_a.copy()
        for amp in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            trial = p + sign * amp * modes[mi]
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            sc = _score_norm(trial)
            if sc < best_local - 1e-12:
                best_local = sc
                best_local_a = trial.copy()

        if best_local < best_score - 1e-12:
            best_score = best_local
            best_a = best_local_a.copy()
            p = best_local_a.copy()
        else:
            # Try a random perturbation
            trial = p + 0.005 * (rng.random(n) - 0.5)
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s > 0:
                trial = trial / s
                sc = _score_norm(trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = trial.copy()
                    p = trial.copy()
                else:
                    if it > 100:
                        break

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: diverse initialization, Nesterov log-sum-exp,
    subgradient peak-shaving, L-BFGS continuation, Adam polish, and
    Fourier-mode local search. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        p = np.asarray(p, dtype=np.float64)
        sc = float(sc)
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = p.copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_p = _score_norm(parent)
        print(f"Parent: n={n}, norm_score={sc_p:.10f}")
        update_best(parent, sc_p, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm(parent), "greedy_default")

    print(f"Starting n={n}, initial best={best_score:.10f}")

    # === Phase 1: Multi-start with quick L-BFGS (200s) ===
    print("Phase 1: Multi-start initialization")
    p1_end = min(deadline, t_start + 200.0)

    inits_to_try = ['parent', 'greedy', 'tri', 'bimodal', 'exp_asym', 'sqrt', 'plateau', 'chirp_abs']
    init_labels = {}
    init_labels['parent'] = best_seq if best_seq is not None else _greedy_init(n)
    init_labels['greedy'] = _greedy_init(n)
    for et in ['tri', 'bimodal', 'exp_asym', 'sqrt', 'plateau', 'chirp_abs']:
        init_labels[et] = _param_init(n, et)

    for label, a_init in init_labels.items():
        if time.monotonic() > p1_end:
            break
        sc_init = _score_norm(a_init)
        if sc_init < best_score * 1.05:
            update_best(a_init, sc_init, f"init_{label}")
            # Quick L-BFGS
            quick_end = min(p1_end - 5, time.monotonic() + 25)
            if time.monotonic() < quick_end:
                a_q, sc_q = _lbfgs_lse(a_init, quick_end,
                                       beta_list=[0.1 * _full_convolve(a_init, a_init).max(),
                                                   0.05 * _full_convolve(a_init, a_init).max()],
                                       maxiter=3000)
                update_best(a_q, sc_q, f"lbfgs_quick_{label}")

    print(f"  Phase 1 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 2: Nesterov log-sum-exp (450s) ===
    print("Phase 2: Nesterov log-sum-exp")
    p2_end = min(deadline, t_start + 650.0)

    if time.monotonic() < p2_end and best_seq is not None:
        a_nes, sc_nes = _nesterov_lse(best_seq, p2_end, beta_start_frac=0.03,
                                      beta_decay=0.975, min_beta_frac=0.00002,
                                      lr=0.5)
        update_best(a_nes, sc_nes, "nesterov_main")

        # Second Nesterov run from a different init
        if time.monotonic() < p2_end - 60:
            a_g = _greedy_init(n)
            a_nes2, sc_nes2 = _nesterov_lse(a_g, min(p2_end - 10, time.monotonic() + 150),
                                            beta_start_frac=0.05, beta_decay=0.98,
                                            min_beta_frac=0.00005, lr=0.3)
            update_best(a_nes2, sc_nes2, "nesterov_greedy")

    print(f"  Phase 2 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 3: Subgradient peak-shaving (300s) ===
    print("Phase 3: Subgradient peak-shaving")
    p3_end = min(deadline, t_start + 950.0)

    if time.monotonic() < p3_end and best_seq is not None:
        a_sub, sc_sub = _subgradient_peak_shave(best_seq, p3_end, lr_init=0.3)
        update_best(a_sub, sc_sub, "subgrad_shave1")

        if time.monotonic() < p3_end - 30:
            a_sub2, sc_sub2 = _subgradient_peak_shave(best_seq, p3_end, lr_init=0.1)
            update_best(a_sub2, sc_sub2, "subgrad_shave2")

    print(f"  Phase 3 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 4: L-BFGS fine-tuning (300s) ===
    print("Phase 4: L-BFGS fine-tuning")
    p4_end = min(deadline, t_start + 1250.0)

    if time.monotonic() < p4_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * b for b in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
                                         0.00005, 0.00002, 0.00001]]
        a_f, sc_f = _lbfgs_lse(best_seq, p4_end, beta_list=tiny_betas, maxiter=15000)
        update_best(a_f, sc_f, "lbfgs_tiny")

    print(f"  Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 5: Adam polish + Fourier local search (250s) ===
    print("Phase 5: Adam + Fourier local search")
    p5_end = min(deadline, t_start + 1500.0)

    if time.monotonic() < p5_end and best_seq is not None:
        a_ad, sc_ad = _adam_lse(best_seq, min(p5_end - 60, time.monotonic() + 120), lr=0.03)
        update_best(a_ad, sc_ad, "adam_polish")

        if time.monotonic() < p5_end - 30:
            a_four, sc_four = _fourier_perturb_optimize(best_seq, p5_end, num_modes=30)
            update_best(a_four, sc_four, "fourier_local")

    print(f"  Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 6: Multi-resolution (remaining time) ===
    if time.monotonic() < deadline - 50 and best_seq is not None:
        # Try interpolating to a larger size
        n_new = min(2 * len(best_seq), 16384)
        if n_new > len(best_seq):
            print(f"Phase 6: Interpolating to n={n_new}")
            a_interp = _interp_profile(best_seq, n_new)
            sc_in = _score_norm(a_interp)
            update_best(a_interp, sc_in, f"interp_n{n_new}")
            if time.monotonic() < deadline - 20:
                a_in_b, sc_in_b = _lbfgs_lse(a_interp, deadline - 10,
                                             beta_list=[0.01 * _full_convolve(a_interp, a_interp).max(),
                                                         0.005 * _full_convolve(a_interp, a_interp).max()],
                                             maxiter=5000)
                update_best(a_in_b, sc_in_b, f"lbfgs_n{n_new}")

        # Also try a different n
        if time.monotonic() < deadline - 30:
            n_alt = 4096
            if n_alt != len(best_seq):
                a_alt = _interp_profile(best_seq, n_alt)
                sc_alt = _score_norm(a_alt)
                update_best(a_alt, sc_alt, f"interp_n{n_alt}")
                if time.monotonic() < deadline - 10:
                    a_alt_b, sc_alt_b = _lbfgs_lse(a_alt, deadline - 5,
                                                   beta_list=[0.01 * _full_convolve(a_alt, a_alt).max()],
                                                   maxiter=3000)
                    update_best(a_alt_b, sc_alt_b, f"lbfgs_n{n_alt}")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        # Validate with evaluator
        valid = True
        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned non-finite, using fallback")
                    valid = False
            except Exception as ex:
                print(f"Eval error: {ex}")
                valid = False

        if valid:
            return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    sc_fb = _score_fn(fb)
    print(f"FALLBACK: n=512, score={sc_fb:.10f}")
    return [float(x) for x in fb]
