    s = a_new.sum()
    if s > 0:

Key insights:

2. The gradient w.r.t. log-parameterization is `4n * a * (h - W)` where `h = conv(w, a_rev)[n-1:2n-1]` and `w` is the softmax-smoothed indicator of where the max autocorrelation occurs

"""
Multi-phase Adam optimization with periodic reheating, diverse restarts,
and multi-resolution strategy for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: faster convergence via tuned hyperparameters, simulated
annealing reheating to escape local minima, and exploration of n values.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _max_conv(a):
    c = _full_convolve(a, a)
    return float(c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_run(a_init, deadline, lr_init=0.015, b1=0.9, b2=0.999,
              beta_scale_init=0.04, beta_scale_min=0.00005,
              beta_decay=0.9998, reheat_every=8000, reheat_factor=2.5,
              max_update_norm=0.8, patience=5000):
    """Adam optimizer with adaptive beta (smoothing) and periodic reheating."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    a = _softmax(z)
    c0 = _full_convolve(a, a)
    max_c0 = float(c0.max())
    best_score = max_c0
    best_a = a.copy()

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0

    beta = beta_scale_init * max_c0
    beta_min = beta_scale_min * max_c0

    stalled = 0
    prev_best = best_score
    rng_local = np.random.default_rng(12345 + n)

    while time.monotonic() < deadline:
        t_step += 1

        # Periodic reheat to escape local minima
        if t_step % reheat_every == 0 and beta < 0.01 * max_c0:
            beta = min(0.02 * max_c0, beta * reheat_factor)

        # Decay beta
        beta = max(beta_min, beta * beta_decay)

        a = _softmax(z)
        c = _full_convolve(a, a)
        cm = float(c.max())

        # Smoothed weights (log-sum-exp)
        c_shifted = (c - cm) / beta
        c_shifted = np.clip(c_shifted, -500.0, 500.0)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s

        # Gradient: 4n * a * (h - W)
        a_rev = a[::-1]
        wc = _full_convolve(w, a_rev)
        h = wc[n - 1:2 * n - 1]
        W = float(np.dot(w, c))
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        # Adam update
        lr = lr_init / (1.0 + 0.00005 * t_step)
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = lr * mh / (np.sqrt(vh) + 1e-8)
        update_norm = np.linalg.norm(update)
        if update_norm > max_update_norm:
            update = update * (max_update_norm / update_norm)

        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()

        # Track best (check every 50 steps to save time)
        if t_step % 50 == 0:
            a_check = _softmax(z)
            sc = _max_conv(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                stalled = 0
            else:
                stalled += 50

        # If stuck, perturb to escape
        if stalled >= patience:
            z = z + 0.05 * rng_local.standard_normal(n)
            z = z - z.mean()
            stalled = 0

    # Final check
    a_final = _softmax(z)
    sc_final = _max_conv(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _interp_to_n(a, n_new):
    """Interpolate sequence to new length, maintaining sum=1."""
    n0 = len(a)
    if n0 == n_new:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    a_new = np.interp(x1, x0, a)
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    return a_new / s


def _perturb(a, sigma, rng):
    """Perturb in log-space and renormalize."""
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)
