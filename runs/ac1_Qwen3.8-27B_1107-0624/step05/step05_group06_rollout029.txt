import numpy as np
import time
import scipy.optimize as sco
from scipy.optimize import minimize
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _lp_iteration(f, n):
    """
    Given f (length n), solve the LP:
    maximize sum(b) subject to (f*b)_k <= M for all k, b >= 0
    where M = max(f*f).
    Returns the optimized b.
    """
    c = _full_convolve(f, f)
    M = float(c.max())
    
    if M < 1e-15:
        return f.copy()
    
    # Build constraint matrix A where A[k,j] = f[k-j] if 0 <= k-j < n else 0
    # We want A @ b <= M * ones(2n-1)
    # A is (2n-1) x n
    
    # More efficient: use the structure of the matrix
    # A[k,j] = f[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
    
    # Build A using toeplitz-like structure
    # Row k has non-zero entries at positions j where 0 <= k-j < n, i.e., k-(n-1) <= j <= k
    # clipped to [0, n-1]
    
    # For efficiency, build A as a sparse-like matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        # A[k, j] = f[k-j] for j in [j_start, j_end)
        indices = np.arange(j_start, j_end)
        A[k, indices] = f[k - indices]
    
    # Solve LP: max 1^T b s.t. A b <= M * ones, b >= 0
    # Or equivalently: min -1^T b s.t. A b <= M * ones, 0 <= b <= inf
    
    # Use scipy's linprog
    # c_obj = -ones (minimize negative sum)
    # A_ub = A, b_ub = M * ones
    # bounds = [(0, None)] * n
    
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = sco.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not res.success:
        return f.copy()
    
    b = res.x
    b_sum = float(b.sum())
    if b_sum < 1e-15:
        return f.copy()
    
    # Rescale to match sum of f
    f_sum = float(f.sum())
    b = b * (f_sum / b_sum)
    
    return b

sum_{k in S} w_k * a[(k-i) mod something] = lambda/2 for all i.

This is a linear system that determines a. If we know the active set S and the weights w, we can solve for a directly!

The active set S is typically a few positions (maybe 1-10 out of 2n-1). The weights w are determined by the requirement that (a*a)_k = M for all k in S.

So the algorithm would be:
1. Guess the active set S and weights w
2. Solve the linear system for a
3. Check if (a*a)_k <= M for k not in S (and = M for k in S)
4. Adjust S and w

This is the "exact" active set method. It's like the method used in convex optimization where you identify the active constraints and solve the reduced problem.

This is a system of n equations in n+1 unknowns (a_0, ..., a_{n-1}, lambda). We can fix lambda by the normalization sum(a) = 1 (or any other normalization).

Actually, we have n equations and n+1 unknowns. The normalization sum(a) = 1 gives one more equation, making it n+1 equations in n+1 unknowns. We can solve this!

For i=1: w_0 * a[-1] (=0) + w_1 * a[0] + w_2 * a[1] + ... = lambda/2

This is a system of the form B @ a = (lambda/2) * 1, where B is a banded matrix determined by S and w.

Solving B @ a = (lambda/2) * 1 with the constraint sum(a) = 1:
- From the first equation set: a = (lambda/2) * B^{-1} * 1
- From normalization: sum(a) = 1 → (lambda/2) * sum(B^{-1} * 1) = 1 → lambda = 2 / sum(B^{-1} * 1)

But B might not be invertible, or B^{-1} * 1 might have negative components (which would violate a >= 0).

If all components of B^{-1} * 1 are positive, we have our solution! And it automatically satisfies the KKT conditions for the given (S, w).

"""
Combined L-BFGS (log-sum-exp smoothing), Nesterov-accelerated subgradient,
and active-set linear system polishing for minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Adds accelerated convergence, exact active-set resolution, and multi-length exploration
beyond the previous state of the art.

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _smoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))

def _smograd(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _gradient_active(z, a, c, n, tol_frac=0.001):
    c_max = c.max()
    tol = tol_frac * c_max
    active = c >= (c_max - tol)
    if active.sum() == 0:
        active = np.zeros_like(c, dtype=bool)
        active[int(np.argmax(c))] = True
    w = np.zeros_like(c)
    w[active] = 1.0 / active.sum()
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)

def _active_set_polish(a_init, deadline, tol_frac=0.0005, max_iter=200):
    """Solve the linear system for a given active set to refine the solution."""
    n = len(a_init)
    best_a = a_init.copy()
    best_sc = _score(a_init)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        a = best_a
        c = _full_convolve(a, a)
        c_max = c.max()
        tol = tol_frac * c_max
        active = np.where(c >= c_max - tol)[0]
        if len(active) == 0:
            break

        # Build linear system: sum_{k in active} w_k * a[k-i] = lambda/2 for all i
        # We use uniform weights w_k = 1/|active|
        w = 1.0 / len(active)
        # B[i, j] = sum_{k in active} w * (1 if k-j == i or k-i == j... )
        # Actually: the equation for index i is:
        #   sum_{k in active} w_k * a[k-i] (where a[m]=0 if m not in [0,n-1]) = lambda/2
        # This is: sum_{k in active} w_k * a[k-i] for valid k-i
        # Let's define: for each i, the coefficient of a[j] in the LHS is:
        #   sum_{k in active} w_k * 1[k-i == j] = sum_{k in active} w_k * 1[k == i+j]
        # So B[i, j] = sum_{k in active} w_k * 1[k == i+j]
        # B[i, j] is nonzero only when i+j is in active

        # Build B as (n x n)
        B = np.zeros((n, n))
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    B[i, j] += w

        # Solve B @ a = (lambda/2) * ones, with sum(a) = 1, a >= 0
        try:
            # a = (lambda/2) * B^{-1} * ones
            # sum(a) = (lambda/2) * sum(B^{-1} * ones) = 1
            # lambda = 2 / sum(B^{-1} * ones)
            # a = B^{-1} * ones / sum(B^{-1} * ones)
            ones = np.ones(n)
            a_sol = np.linalg.solve(B, ones)
            s = a_sol.sum()
            if s <= 0 or not np.isfinite(s):
                break
            a_sol = a_sol / s
            if np.any(a_sol < -1e-12):
                # Clip and renormalize
                a_sol = np.maximum(a_sol, 0.0)
                s2 = a_sol.sum()
                if s2 < 1e-15:
                    break
                a_sol = a_sol / s2
            sc = _score(a_sol)
            if sc < best_sc:
                best_sc = sc
                best_a = a_sol.copy()
                if best_sc < _score(a_init) * (1 - 1e-6) and it > 5:
                    # Converged
                    c_check = _full_convolve(best_a, best_a)
                    new_max = c_check.max()
                    if new_max <= c_max * (1 + 1e-9):
                        break
            else:
                # Try with tighter active set
                tol_frac = max(1e-6, tol_frac * 0.5)
                if tol_frac < 1e-7:
                    break
        except np.linalg.LinAlgError:
            # B is singular, try with fewer active constraints
            active = np.where(c >= c_max - tol_frac * 0.1 * c_max)[0]
            if len(active) <= 1:
                break

    return best_a, best_sc

def _nesterov_subgrad(a_init, deadline, max_steps=10000, lr_init=0.5):
    """Nesterov-accelerated subgradient method in z-space."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_sc = _score(a_init)
    v = np.zeros(n)
    beta_mom = 0.9

    for i in range(max_steps):
        if time.monotonic() > deadline:
            break

        a = _softmax(z)
        c = _full_convolve(a, a)
        grad = _gradient_active(z, a, c, n, tol_frac=0.001)

        # Nesterov update
        beta_t = (1.0 + float(i)) / (3.0 + float(i))
        step = lr_init / (1.0 + 0.1 * float(i))
        v = beta_mom * v + grad
        z = z - step * v
        z = np.clip(z, -80.0, 80.0)
        z = z - z.mean()

        if i % 100 == 0 or i == max_steps - 1:
            a_check = _softmax(z)
            sc = _score(a_check)
            if sc < best_sc:
                best_sc = sc
                best_a = a_check.copy()

    a_final = _softmax(z)
    sc_final = _score(a_final)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_final.copy()

    return best_a, best_sc

def _lbfgs_phase(a_init, n, deadline, min_time=5.0):
    """Run L-BFGS beta continuation schedule."""
    c_cur = _full_convolve(a_init, a_init)
    max_c = float(c_cur.max())
    if max_c < 1e-15:
        return a_init

    betas = max_c * np.array([0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    current_a = a_init.copy()
    current_sc = _score(current_a)

    for i, beta in enumerate(betas):
        if time.monotonic() + min_time > deadline:
            break
        remaining = deadline - min_time - time.monotonic()
        if remaining < 3:
            break

        est_time_per_iter = 0.004 * (n / 2048.0)
        max_iter = min(5000, max(100, int(remaining / (est_time_per_iter * 10))))

        z0 = np.log(np.maximum(current_a, 1e-15))
        z0 = z0 - z0.mean()

        def fun(z):
            return _smoobj(z, beta, n)

        def jac(z):
            return _smograd(z, beta, n)

        try:
            result = minimize(
                fun, z0, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxcor': 40, 'maxls': 200, 'disp': False}
            )
            a_out = _softmax(result.x)
            sc = _score(a_out)
            if sc < current_sc - 1e-14:
                current_a = a_out
                current_sc = sc
        except Exception:
            pass

    return current_a

def _perturb(a, sigma, rng):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)

def _perturb_fourier(a, rng, n_modes=4):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = rng.integers(1, max(2, n // 2))
        amp = 0.02 * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        z = z + amp * np.cos(2.0 * np.pi * freq * t / n + phase)
        phase2 = rng.uniform(0, 2 * np.pi)
        z = z + amp * 0.5 * np.sin(2.0 * np.pi * freq * t / n + phase2)
    z = z - z.mean()
    return _softmax(z)

def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * (t - 0.5))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "sinc_like":
        t = np.linspace(-1, 1, n)
        a = np.sinc(t * 2.0) ** 2
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "gaussian":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        return a / a.sum()
    else:
        return np.ones(n) / n

        if best_seq is not None and len(best_seq) > 100:
            # Upsample by interpolation
            from numpy import interp
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n2)
            a2 = interp(x_new, x_old, best_seq)
            a2 = np.maximum(a2, 1e-10)
            a2 = a2 / a2.sum()
        else:
            a2 = _make_profile(n2, "raised_cosine")

        sc2 = _score(a2)
        print(f"  n=4096 init score: {sc2:.10f}")
        if sc2 < 2.0:
            a2_opt = _lbfgs_phase(a2, n2, deadline, min_time=30.0)
            sc2_opt = _score(a2_opt)
            update_best(a2_opt, sc2_opt, "n4096")
            if time.monotonic() < deadline - 30:
                a2_sg, sc2_sg = _nesterov_subgrad(a2_opt, deadline - 30, max_steps=3000, lr_init=0.2)
                update_best(a2_sg, sc2_sg, "n4096_nesterov")

    # --- Phase 8: Final polishing ---
    print(f"\nPhase 8: Final polish (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_fin = len(best_seq)
        # One more L-BFGS with very small beta
        c_fin = _full_convolve(best_seq, best_seq)
        mb = float(c_fin.max())
        beta_final = mb * 0.00001
        if time.monotonic() < deadline - 3:
            z0 = np.log(np.maximum(best_seq, 1e-15))
            z0 = z0 - z0.mean()
            try:
                result = minimize(
                    lambda z: _smoobj(z, beta_final, n_fin), z0,
                    jac=lambda z: _smograd(z, beta_final, n_fin),
                    method='L-BFGS-B',
                    options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-14,
                             'maxcor': 50, 'maxls': 300, 'disp': False}
                )
                a_fin = _softmax(result.x)
                update_best(a_fin, _score(a_fin), "final_lbfgs")
            except Exception:
                pass

        # Final active set
        if time.monotonic() < deadline - 2:
            a_fin2, sc_fin2 = _active_set_polish(best_seq, deadline - 2, tol_frac=0.0001, max_iter=50)
            update_best(a_fin2, sc_fin2, "final_as")

    # --- Final validation ---
    print(f"\n{'='*60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = _make_profile(256, "uniform")
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = _make_profile(2048, "uniform")
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
