"""
Hybrid optimizer combining dual multiplicative weights with Conjugate Gradient
inner solver, followed by Adam polishing with log-sum-exp smoothing, for
minimizing 2*n*max(conv(a,a))/sum(a)^2.
"""
import numpy as np
import time

def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _toeplitz_matvec(p, x, n):
    """Compute Q*p where Q is symmetric Toeplitz with first row p."""
    # C_0 = [p_0, p_1, ..., p_{n-1}, 0, p_{n-1}, p_{n-2}, ..., p_1]
    C0 = np.zeros(2 * n)
    C0[:n] = p
    C0[n+1:] = p[:n-1][::-1]
    # Note: C0[n] = 0 (the zero in the middle)
    x_ext = np.zeros(2 * n)
    x_ext[:n] = x
    return np.fft.ifft(np.fft.fft(C0) * np.fft.fft(x_ext)).real[:n]

def _cg_solve(Q_matvec_fn, b, n, max_iter=200, tol=1e-8, eps_reg=1e-10):
    """Solve (Q + eps*I) x = b using conjugate gradient."""
    x = np.zeros(n)
    r = b.copy()  # r = b - (Q+eps*I)x = b - Qx - eps*x
    # Actually, let's just use Q and add regularization via the mat-vec
    def matvec(v):
        return Q_matvec_fn(v) + eps_reg * v
    p = r.copy()
    rs = np.dot(r, r)
    for it in range(max_iter):
        Ap = matvec(p)
        pAp = np.dot(p, Ap)
        if abs(pAp) < 1e-300:
            break
        alpha = rs / pAp
        x += alpha * p
        r -= alpha * Ap
        rs_new = np.dot(r, r)
        if rs_new ** 0.5 < tol * (np.dot(b, b) ** 0.5 + 1e-300):
            break
        p = r + (rs_new / rs) * p
        rs = rs_new
    return x

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s

def _mw_cg_step(a, beta, n, max_cg_iter=100, eps_reg=1e-10):
    """One dual multiplicative weights step with CG inner solver."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    
    # Update dual variable p
    p = _softmax(c / beta)
    
    # Solve inner QP: min a^T Q(p) a s.t. a >= 0, sum(a) = 1
    # Q is symmetric Toeplitz with first row p
    b = np.ones(n)
    x = _cg_solve(lambda v: _toeplitz_matvec(p, v, n), b, n, 
                  max_iter=max_cg_iter, eps_reg=eps_reg)
    
    # Project to simplex (clip negatives, normalize)
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-300:
        return a  # No improvement
    a_new = x / s
    
    # Check if score improved
    sc_old = _score_norm(a)
    sc_new = _score_norm(a_new)
    
    if sc_new < sc_old:
        return a_new
    else:
        # Try interpolation
        for t in [0.5, 0.3, 0.1]:
            a_cand = (1-t) * a + t * a_new
            sc_cand = _score_norm(a_cand)
            if sc_cand < sc_old:
                return a_cand
        return a

def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=5000000, lr=0.01):
    """Adam optimization with log-sum-exp smoothing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score
    
    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 1.0:
            update = update * (1.0 / update_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        if t % 5000 == 0:
            if best_score < prev_best * 0.9995:
                current_lr = min(0.05, current_lr * 1.05)
            else:
                current_lr = max(0.001, current_lr * 0.97)
            prev_best = best_score
    
    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score
