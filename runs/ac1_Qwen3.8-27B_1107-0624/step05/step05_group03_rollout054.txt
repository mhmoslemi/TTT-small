    best_score = _score_norm(a_init)
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta_starts = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    betas = [b * max_c0 * beta_scale for b in beta_starts]
    per_level = max(3.0, max_wall_time / len(betas))

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_end = min(deadline, time.monotonic() + per_level)

        def obj_func(z_):
            a_ = _softmax_local(z_, n)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            e = np.exp(cs)
            s = e.sum()
            if s < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s))

        def grad_func(z_):
            a_ = _softmax_local(z_, n)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            cs = (c_ - cm_) / beta
            e = np.exp(cs)
            s = e.sum()
            if s < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e / s
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            t0 = time.monotonic()
            result = minimize(
                obj_func, z.copy(), jac=grad_func,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-10,
                         'maxls': 50, 'maxfun': 20000}
            )
            if time.monotonic() > level_end + 30:
                continue
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_local(z_new, n)
                s_new = _score_norm(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _softmax_local(z, n):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n) / n
    return e / s


def _peak_shave_search(a_init, deadline, rng, n_trials=50000):
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _full_convolve(a, a)
    f_curr = float(c.max())

    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break

        if it % 10 == 0:
            c = _full_convolve(a, a)
            f_curr = float(c.max())
            if f_curr * 2.0 * n < best_sc:
                best_sc = f_curr * 2.0 * n
                best_a = a.copy()

        k_star = int(np.argmax(c))
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)

        if hi - lo + 1 < 2:
            continue

        pairs = np.arange(lo, hi + 1)
        partners = k_star - pairs
        products = a[pairs] * a[partners]

        # Strategy 1: Reduce the highest-product pair member
        if it % 3 == 0:
            idx = pairs[int(np.argmax(products))]
            amt = a[idx] * rng.uniform(0.1, 0.5)
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[idx] -= amt
            # Find safe destination (low contribution to top peaks)
            top5 = np.argsort(c)[-5:]
            contrib_safe = np.zeros(n)
            for kl in top5:
                kl = int(kl)
                lo_s = max(0, kl - (n - 1))
                hi_s = min(n - 1, kl)
                for j in range(lo_s, hi_s + 1):
                    p = kl - j
                    if 0 <= p < n:
                        contrib_safe[j] += a[p]
            safe_mask = contrib_safe < np.percentile(contrib_safe, 30)
            safe_indices = np.where(safe_mask)[0]
            if len(safe_indices) == 0:
                safe_indices = np.where(contrib_safe == contrib_safe.min())[0]
            add_idx = safe_indices[rng.integers(len(safe_indices))]
            a_trial[add_idx] += amt
            a_trial = _project_simplex(a_trial)
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        # Strategy 2: Block multiplicative perturbation
        elif it % 3 == 1:
            bs = int(rng.integers(3, max(4, min(32, n // 16))))
            start = int(rng.integers(0, n - bs))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=bs)
            a_trial[start:start + bs] *= mult
            a_trial = _project_simplex(a_trial)
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        # Strategy 3: Coordinated swap
        else:
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            frac = rng.uniform(0.05, 0.3)
            a_trial = a.copy()
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] -= amt1
            a_trial[i2] -= amt2
            a_trial[i1] += amt2
            a_trial[i2] += amt1
            a_trial = _project_simplex(a_trial)
            sc = _score_norm(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_sc


def _mirror_descent(a_init, deadline, rng, max_iters=100000):
    n = len(a_init)
    a = _project_simplex(a_init.copy())
    best_a = a.copy()
    best_sc = _score_norm(a)
    c = _full_convolve(a, a)
    f_curr = float(c.max())
    f_lower = 1.0 / n
    eta = 1.0 / np.sqrt(f_curr * n)
    patience = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 1.0:
            break
        g, k_star = _topk_direction(a, 1, 0.0)
        g = g / (np.linalg.norm(g) + 1e-30)
        # Multiplicative update
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= eta * g
        a_new = np.exp(log_a - np.max(log_a))
        a_new /= a_new.sum()
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 0.99:
            a_new = _project_simplex(a_new)
        c_new = _full_convolve(a_new, a_new)
        f_new = float(c_new.max())
        if f_new < f_curr:
            a = a_new
            f_curr = f_new
            if f_curr * 2.0 * n < best_sc:
                best_sc = f_curr * 2.0 * n
                best_a = a.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
            eta *= 0.95
        if patience > 3000:
            break
        if (it + 1) % 5000 == 0:
            eta *= 0.9

    return best_a, best_sc


def _make_triangular(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_bell(n, center=0.5, width=0.25):
    t = np.linspace(0.0, 1.0, n)
    a = np.exp(-(t - center) ** 2 / (2 * width ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_flat_taper(n, taper_frac=0.1):
    a = np.ones(n)
    m = max(1, int(n * taper_frac))
    ramp = np.linspace(0.1, 1.0, m)
    a[:m] = ramp
    a[-m:] = ramp[::-1]
    return a / a.sum()


def _make_dpss_like(n, alpha=2.5):
    """Approximate prolate spheroidal sequence shape."""
    t = np.linspace(0.0, 1.0, n)
    u = t - 0.5
    a = (np.sin(np.pi * alpha * u) / (np.pi * alpha * u + 1e-12)) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_poly(n, coeffs):
    t = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for c in coeffs:
        a = a * (1.0 + c * (2.0 * t - 1.0))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 8192
        parent = _make_triangular(n_parent)
        sc_parent = _score_norm(parent)
        print(f"No parent, triangular n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    n_work = n_parent
    print(f"Working with n={n_work}")

    # --- Phase 0: Diverse initialization (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    profiles = [
        _make_triangular(n_work),
        _make_bell(n_work, 0.5, 0.2),
        _make_bell(n_work, 0.5, 0.3),
        _make_bell(n_work, 0.5, 0.15),
        _make_flat_taper(n_work, 0.1),
        _make_flat_taper(n_work, 0.2),
        _make_flat_taper(n_work, 0.05),
        _make_dpss_like(n_work, 1.5),
        _make_dpss_like(n_work, 2.5),
        _make_dpss_like(n_work, 4.0),
        _make_dpss_like(n_work, 6.0),
        _make_poly(n_work, [0.3]),
        _make_poly(n_work, [0.5]),
        _make_poly(n_work, [-0.3]),
        _make_poly(n_work, [0.2, 0.1]),
        _make_poly(n_work, [0.2, -0.1]),
        _make_bell(n_work, 0.35, 0.2) + _make_bell(n_work, 0.65, 0.2),
        _make_bell(n_work, 0.3, 0.15) + _make_bell(n_work, 0.7, 0.15),
    ]

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        p = _project_simplex(p)
        sc = _score_norm(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # Pick best start for main optimization
    if parent is not None:
        sc_p = _score_norm(parent)
        if sc_p < candidates[0][0]:
            start_a = parent.copy()
        else:
            start_a = candidates[0][1].copy()
    else:
        start_a = candidates[0][1].copy()

    # --- Phase 1: Subgradient method (350s) ---
    phase1_end = min(deadline, t_start + 390.0)
    print(f"Phase 1: Subgradient, start score={_score_norm(start_a):.10f}")
    a_sg, sc_sg = _subgradient_method(start_a, phase1_end, rng, max_iters=300000)
    update_best(a_sg, sc_sg, "subgrad")
    print(f"Phase 1 done: best={best_score:.10f}")

    # --- Phase 2: L-BFGS continuation (300s) ---
    phase2_end = min(deadline, t_start + 690.0)
    if best_seq is not None and time.monotonic() < phase2_end:
        print("Phase 2: L-BFGS continuation")
        dl = phase2_end - 10.0
        a_lb, sc_lb = _lbfgs_continuation(best_seq, dl, rng, max_wall_time=280.0, beta_scale=1.0)
        update_best(a_lb, sc_lb, "lbfgs_main")

        if time.monotonic() < phase2_end - 30:
            dl2 = phase2_end - 10.0
            a_lb2, sc_lb2 = _lbfgs_continuation(best_seq, dl2, rng, max_wall_time=60.0, beta_scale=0.3)
            update_best(a_lb2, sc_lb2, "lbfgs_fine")
    print(f"Phase 2 done: best={best_score:.10f}")

    # --- Phase 3: More subgradient from current best (200s) ---
    phase3_end = min(deadline, t_start + 890.0)
    if best_seq is not None and time.monotonic() < phase3_end:
        print("Phase 3: Subgradient round 2")
        a_sg2, sc_sg2 = _subgradient_method(best_seq, phase3_end, rng, max_iters=200000)
        update_best(a_sg2, sc_sg2, "subgrad2")
    print(f"Phase 3 done: best={best_score:.10f}")

    # --- Phase 4: Peak-shaving local search (200s) ---
    phase4_end = min(deadline, t_start + 1090.0)
    if best_seq is not None and time.monotonic() < phase4_end:
        print("Phase 4: Peak-shaving search")
        a_ps, sc_ps = _peak_shave_search(best_seq, phase4_end, rng, n_trials=40000)
        update_best(a_ps, sc_ps, "peak_shave")
    print(f"Phase 4 done: best={best_score:.10f}")

    # --- Phase 5: Mirror descent (150s) ---
    phase5_end = min(deadline, t_start + 1240.0)
    if best_seq is not None and time.monotonic() < phase5_end:
        print("Phase 5: Mirror descent")
        a_md, sc_md = _mirror_descent(best_seq, phase5_end, rng, max_iters=50000)
        update_best(a_md, sc_md, "mirror")
    print(f"Phase 5 done: best={best_score:.10f}")

    # --- Phase 6: Multi-restart (200s) ---
    phase6_end = min(deadline, t_start + 1440.0)
    if best_seq is not None and time.monotonic() < phase6_end:
        print("Phase 6: Multi-restart")
        n_restarts = 3
        per_restart = max(20.0, (phase6_end - time.monotonic()) / max(1, n_restarts))
        for i in range(n_restarts):
            if time.monotonic() > phase6_end - 10:
                break
            noise = 0.03 + 0.02 * i
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + noise * rng.normal(size=len(best_seq))
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            p_start = _softmax_local(z, len(best_seq))
            dl = min(phase6_end - 5, time.monotonic() + per_restart)
            a_r, sc_r = _lbfgs_continuation(p_start, dl, rng, max_wall_time=per_restart - 5, beta_scale=0.5)
            update_best(a_r, sc_r, f"restart{i}")
    print(f"Phase 6 done: best={best_score:.10f}")

    # --- Phase 7: Try different n (150s) ---
    phase7_end = min(deadline, t_start + 1590.0)
    if best_seq is not None and time.monotonic() < phase7_end:
        n_curr = len(best_seq)
        tried_ns = set()
        for target_n in [n_curr * 2, n_curr // 2, 4096, 8192, 16384]:
            if target_n < 1024 or target_n > 32768 or target_n in tried_ns or target_n == n_curr:
                continue
            if time.monotonic() > phase7_end - 30:
                break
            tried_ns.add(target_n)
            p_interp = _interp_profile(best_seq, target_n)
            sc_interp = _score_norm(p_interp)
            print(f"  Interp to n={target_n}: score={sc_interp:.10f}")
            update_best(p_interp, sc_interp, f"interp_n{target_n}")
            if time.monotonic() < phase7_end - 20:
                dl = min(phase7_end - 15, time.monotonic() + 40.0)
                a_b, sc_b = _lbfgs_continuation(p_interp, dl, rng, max_wall_time=35.0, beta_scale=0.5)
                update_best(a_b, sc_b, f"lbfgs_n{target_n}")
    print(f"Phase 7 done: best={best_score:.10f}")

    # --- Phase 8: Final polishing (remaining) ---
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic())
        print(f"Phase 8: Final polishing, remaining={remaining:.0f}s")

        if remaining > 80:
            dl = min(deadline - 30, time.monotonic() + remaining * 0.4)
            a_f, sc_f = _lbfgs_continuation(best_seq, dl, rng,
                                             max_wall_time=remaining * 0.4, beta_scale=0.2)
            update_best(a_f, sc_f, "final_lbfgs")

        if time.monotonic() < deadline - 30:
            dl = min(deadline - 25, time.monotonic() + remaining * 0.3)
            a_s, sc_s = _subgradient_method(best_seq, dl, rng, max_iters=100000)
            update_best(a_s, sc_s, "final_subgrad")

        if time.monotonic() < deadline - 20:
            dl = min(deadline - 15, time.monotonic() + remaining * 0.25)
            a_p, sc_p = _peak_shave_search(best_seq, dl, rng, n_trials=10000)
            update_best(a_p, sc_p, "final_shave")

    # --- Final validation and return ---
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular(8192)
            s = float(final_seq.sum())
        final_seq = final_seq / s
        sc_final = float(2.0 * len(final_seq) * _full_convolve(final_seq, final_seq).max() / (s * s))
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        if len(final_seq) <= 32768:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
                else:
                    print(f"Evaluator returned non-finite: {sc_eval}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular(8192)
    sc_fb = _score_norm(fallback)
    print(f"FALLBACK: n=8192, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
