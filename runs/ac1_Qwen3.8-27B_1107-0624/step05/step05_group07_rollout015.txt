"""
Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternating minimization, L-BFGS with adaptive log-sum-exp,

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A

def _lp_minimax(f, timeout=30):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, timeout=timeout)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            s = g_val.sum()
            if s > 1e-10:
                g_val = g_val / s
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None

def _lp_alternate(f, n_iters=20, deadline=None):
    """LP-based alternating minimization with adaptive step size."""
    n = len(f)
    best_a = f.copy()
    best_score = _score_norm_fn(f)
    a = f.copy()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g, t_val = _lp_minimax(a)
        if g is None:
            break
        
        # Adaptive step size: larger when improvement is high
        c_old = _full_convolve(a, a)
        progress = float(c_old.max() - t_val) / max(float(c_old.max()), 1e-10)
        step = min(0.5, max(0.1, 0.3 * progress + 0.1))
        
        a_new = (1 - step) * a + step * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        
        if it % 10 == 0:
            elapsed = time.monotonic() - _lp_start_time
            print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}, t*={t_val:.8f}")
    
    return best_a, best_score

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _lbfgs_anneal(a_init, deadline, aggressive=True):
    """L-BFGS with aggressive temperature annealing."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    if aggressive:
        # More aggressive: start with higher beta and anneal faster
        fracs = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001,
                 0.0006, 0.0004, 0.0002, 0.0001, 0.00005, 0.00003, 0.00002, 0.00001]
    else:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]
    
    beta_list = [b * max_c0 for b in fracs]
    n_betas = len(beta_list)
    
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        
        # More iterations for smaller beta
        if beta < 0.001 * max_c0:
            maxiter = 3000
        elif beta < 0.01 * max_c0:
            maxiter = 2000
        else:
            maxiter = 1000
        
        time_per_beta = max(5.0, (deadline - time.monotonic()) / max(1, n_betas - idx))
        maxiter = min(maxiter, max(100, int(time_per_beta * 50)))
        
        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-15,
                         'maxls': 100, 'maxfun': maxiter * 10}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score

def _peak_shave(a_init, deadline, n_rounds=100):
    """Directly reduce the peak of the convolution by adjusting the sequence."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    a = a_init.copy()
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        M = cm  # current peak value
        
        # Find peak locations
        peak_thresh = 0.95 * M
        peak_mask = c >= peak_thresh
        peak_indices = np.where(peak_mask)[0]
        
        if len(peak_indices) == 0:
            break
        
        # Compute subgradient at peak: for each peak k*, grad_j += a_{k*-j} (normalized)
        grad = np.zeros(n)
        for k in peak_indices:
            weight = (c[k] - 0.9 * M) / M  # weight by how close to peak
            for j in range(n):
                kj = k - j
                if 0 <= kj < n:
                    grad[j] += weight * a[kj]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Try different step sizes
        best_step = 0
        best_local = best_score
        for eta in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            a_trial = a - eta * grad
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = _score_norm_fn(a_trial)
            if sc < best_local:
                best_local = sc
                best_step = eta
        
        if best_step == 0:
            # Try smaller steps
            for eta in [0.0001, 0.0005, 0.001]:
                a_trial = a - eta * grad
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = _score_norm_fn(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_step = eta
        
        if best_step == 0:
            break
        
        a = a - best_step * grad
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        sc = _score_norm_fn(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def _coord_descent_fine(a_init, deadline, max_rounds=20):

    n = len(a_init)
    best_a = a_init.copy()

    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    # Focus on a subset of coordinates for efficiency
    # Use random subsets in each round
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        n_trial = min(n, 200)  # try at most 200 coordinates per round
        indices = np.random.choice(n, size=n_trial, replace=False)
        indices = np.sort(indices)
        
        improved = False
        for j in indices:
            if time.monotonic() > deadline:
                break
            
            base_score = _score_norm_fn(_softmax_fn(z))
            best_delta = 0.0
            for delta in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < base_score - 1e-12:
                    base_score = sc
                    best_delta = delta
            
            if best_delta != 0.0:
                z[j] += best_delta
                z = np.clip(z, -80, 80)

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    
    # Hann window
    a = np.sin(np.pi * t) ** 2
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Raised cosine (different parameters)
    for alpha in [0.3, 0.5, 0.7]:
        a = 0.5 * (1 + np.cos(2 * np.pi * (t - 0.5) / (1 - alpha))) * (t >= alpha/2) * (t <= 1-alpha/2)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Gaussian
    for sigma in [0.15, 0.2, 0.25, 0.3]:
        a = np.exp(-0.5 * (u / sigma) ** 2)
        profiles.append(a / a.sum())
    
    # Power-law taper
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = (1 - np.abs(u)) ** p
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    
    # Exponential taper
    for rate in [1.0, 2.0, 3.0, 4.0, 6.0]:
        a = np.exp(-rate * np.abs(u))
        profiles.append(a / a.sum())
    
    # Linear
    a = 1.0 - np.abs(u)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Uniform
    a = np.ones(n)
    profiles.append(a / n)
    
    # Cosine (full)
    a = np.cos(np.pi * u / 2)
    a = np.maximum(a, 1e-12)
    profiles.append(a / a.sum())
    
    # Asymmetric: exponential one side
    for rate in [2.0, 4.0]:
        a = np.exp(-rate * np.maximum(0, -u)) * np.exp(-0.5 * np.maximum(0, u))
        profiles.append(a / a.sum())
    
    return profiles

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

def _spectral_peak_reduce(a, n_steps=10, eta_init=0.01):
    """Reduce convolution peaks using frequency domain modifications."""
    n = len(a)
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    a = a.copy()
    
    N = _next_pow2(2 * n - 1)
    
    for step in range(n_steps):
        c = _full_convolve(a, a)
        cm = float(c.max())
        peak_idx = np.argmax(c)
        
        # In frequency domain: c has FFT = |A|^2 (approximately, for circular)
        # To reduce c[peak_idx], we want to reduce the contribution at that lag
        # Modify: A_new = A * exp(-eta * real(FFT(peak_pattern)))
        
        # Create a peak pattern in time domain
        # The contribution to c[k] from frequency ω is |A(ω)|^2 * exp(2πiωk/N) / N
        # To reduce c[peak_idx], reduce |A(ω)| where cos(2πω*peak_idx/N) > 0
        
        A = np.fft.rfft(a, N)
        freqs = np.fft.rfftfreq(N)
        phase = 2 * np.pi * freqs * peak_idx
        # Reduce where cosine is positive (contributes positively to c[peak_idx])
        reduction = np.maximum(0, np.cos(phase))
        
        eta = eta_init * (0.8 ** step)
        A_new = A * np.exp(-eta * reduction)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller eta
            eta = eta * 0.1
            A_new = A * np.exp(-eta * reduction)
            a_new = np.fft.irfft(A_new, N)[:n]
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
                sc2 = _score_norm_fn(a_new)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_new.copy()
                    a = a_new
    
    return best_a, best_score

Improved hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Combines LP alternating minimization, aggressive L-BFGS log-sum-exp annealing,
spectral peak-shaping, subgradient peak shaving, and ultra-fine coordinate descent.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _build_conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, timeout_s):

    A = _build_conv_matrix(f)
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, timeout=timeout_s)
        if g.value is not None:

            s = g_val.sum()
            if s > 1e-10:

    return None, None
