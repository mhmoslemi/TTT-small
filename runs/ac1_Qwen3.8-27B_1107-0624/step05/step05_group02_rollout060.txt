```python
def topk_subgradient(a, K=5):
    n = len(a)
    c = _full_convolve(a, a)
    
    # Find top K indices
    top_k_idx = np.argsort(c)[-K:]
    top_k_idx = top_k_idx[np.argsort(c[top_k_idx])]  # sort by value
    
    # Compute gradient as average of gradients at top-K points
    # grad_k[j] = a[k-j] (if valid) + a[j-k] (if valid)
    # This equals: (conv of delta_k with a reversed) + (conv of delta_k with a)
    # More efficiently: for each k in top_k_idx, the gradient is:
    #   shift of a by -k (zero-padded) + shift of a by +k (zero-padded)
    
    grad = np.zeros(n)
    for k in top_k_idx:
        # a[k-j] for j in [0, n-1]: valid when 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
        lo1 = max(0, int(k) - n + 1)
        hi1 = min(n-1, int(k))
        if lo1 <= hi1:
            grad[lo1:hi1+1] += a[int(k) - np.arange(lo1, hi1+1)]
        
        # a[j-k] for j in [0, n-1]: valid when 0 <= j-k <= n-1, i.e., k <= j <= k+n-1
        lo2 = max(0, int(k))
        hi2 = min(n-1, int(k) + n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2+1] += a[np.arange(lo2, hi2+1) - int(k)]
    
    return grad / K, c.max()
```

```python
def _next_pow2(n):

    n, m = len(a), len(b)

            if g.sum() > 1e-15:
```

```python
def _compute_gradient(a, K, use_topk=True):
    """Compute gradient using top-K or softmax smoothing."""
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())

    if use_topk:
        if K >= 2 * n - 1:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            top_idx = np.argsort(c)[-K:]
            w = np.zeros(2 * n - 1)
            w[top_idx] = 1.0 / K
    else:
        beta = 0.01 * cm
        c_shifted = (c - cm) / beta
        w = _softmax(c_shifted)

    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    return grad, cm, c


def _adam_topk_optimize(a_init, deadline, K_schedule, lr=0.01, max_it=50000):
    """Adam optimizer with top-K subgradient and adaptive K schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr

    iters = 0
    K_idx = 0
    K = K_schedule[0]
    K_switch_interval = max(1, max_it // len(K_schedule))

    while iters < max_it:
        if time.monotonic() > deadline:
            break
        iters += 1
        t_step += 1

        # Update K based on schedule
        new_K_idx = min(iters // K_switch_interval, len(K_schedule) - 1)
        if new_K_idx != K_idx:
            K_idx = new_K_idx
            K = K_schedule[K_idx]

        a = _softmax(z)
        grad, cm, c = _compute_gradient(a, K, use_topk=True)

        if not np.all(np.isfinite(grad)):
            break

        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)

        z = z - update
        z = np.clip(z, -40.0, 40.0)
        z = z - z.mean()

        if iters % 100 == 0:
            a_check = _softmax(z)
            sc = float(_full_convolve(a_check, a_check).max())
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_check.copy()

        if iters % 2000 == 0:
            current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score

def _interp_profile(p, n_new):

    if n0 == n_new:
        return p.copy()

def _peak_peel(a, num_peels=50, strength=0.1):
    """Directly reduce peak contributions by adjusting coordinates."""
    n = len(a)
    a = a.copy()
    for _ in range(num_peels):
        c = _full_convolve(a, a)
        k_peak = int(np.argmax(c))
        cm = c[k_peak]
        # Find j that maximizes contribution a_j * a_{k_peak - j}
        valid_j = np.arange(max(0, k_peak - n + 1), min(n - 1, k_peak) + 1)
        if len(valid_j) == 0:
            break
        contrib = a[valid_j] * a[k_peak - valid_j]
        j_max = valid_j[np.argmax(contrib)]
        # Reduce a_j_max slightly, redistribute to mean
        reduction = strength * a[j_max] * 0.01
        a[j_max] -= reduction
        a += reduction / n
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    return a
```

        if iters % 2000 == 0:
            current_lr = max(0.0005, current_lr * 0.95)
    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _adam_smoothed_optimize(a_init, deadline, beta_scales, lr=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    c_init = _full_convolve(a_init, a_init)
    best_score = float(c_init.max())
    max_c_init = best_score
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    for beta_scale in beta_scales:
        if time.monotonic() > deadline:
            break
        beta = beta_scale * max_c_init
        iters = 0
        while iters < 30000:
            if time.monotonic() > deadline:
                break
            iters += 1
            t_step += 1
            a = _softmax(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            w = _softmax(c_shifted)
            a_rev = a[::-1]
            wc = _full_convolve(w, a_rev)
            h = wc[n - 1:2 * n - 1]
            W = float(np.dot(w, c))
            grad = 4.0 * n * a * (h - W)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t_step)
            vh = v_buf / (1.0 - b2 ** t_step)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)
            z = z - update
            z = np.clip(z, -40.0, 40.0)
            z = z - z.mean()
            if iters % 100 == 0:
                sc = float(_full_convolve(_softmax(z), _softmax(z)).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = _softmax(z).copy()
            if iters % 5000 == 0:
                current_lr = max(0.0005, current_lr * 0.93)
    a_final = _softmax(z)
    sc_final = float(_full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _make_uniform(n):
    return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax(z)


def _peak_peel(a, num_peels=80, strength=0.3):
    n = len(a)
    a = a.copy()
    for _ in range(num_peels):
        c = _full_convolve(a, a)
        k_peak = int(np.argmax(c))
        valid_j = np.arange(max(0, k_peak - n + 1), min(n - 1, k_peak) + 1)
        if len(valid_j) == 0:
            break
        contrib = a[valid_j] * a[k_peak - valid_j]
        j_max = valid_j[np.argmax(contrib)]
        reduction = strength * a[j_max] * 0.005
        a[j_max] -= reduction
        a += reduction / n
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
    return a
