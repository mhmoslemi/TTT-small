    C_im = -np.sin(angles) / N  # (n, N)

    # Wait: A[k]*exp(i*th) = (A_re[k] + i*A_im[k])*(cos(th) + i*sin(th))

    # Re(a[j]) = (1/N) * sum_k (A_re[k]*cos(th[j,k]) - A_im[k]*sin(th[j,k]))
    
    C_re_mat = np.cos(angles) / N
    C_im_mat = -np.sin(angles) / N
    
    a_re = C_re_mat @ A_re + C_im_mat @ A_im  # (n,)

"""
Hybrid ILT (Iterative Fourier Transform) + LP + Adam optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2. Uses Gerchberg-Saxton spectral flattening for global
exploration, linear programming for principled local refinement, and Adam with
log-sum-exp smoothing for final polishing. Key innovation: ILT explores a different
region of solution space than pure LP, enabling escape from local optima.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _ilt_step(a, N, alpha):
    n = len(a)
    a_padded = np.zeros(N)
    a_padded[:n] = a
    A = np.fft.fft(a_padded)
    mag = np.abs(A)
    phase = np.angle(A)
    log_mag = np.log(mag + 1e-30)
    target_log = float(np.mean(log_mag))
    new_log_mag = (1.0 - alpha) * log_mag + alpha * target_log
    new_mag = np.exp(new_log_mag)
    A_new = new_mag * np.exp(1j * phase)
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _ilt_run(a_init, N, alpha_start, alpha_end, n_iters, deadline):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm(a)
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        frac = it / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        a = _ilt_step(a, N, alpha)
        if it % 50 == 0 or it == n_iters - 1:
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    return best_a, best_sc


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _solve_lp(f, M, rows, cols, time_limit=30):
    n = len(f)
    vals = f[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0:
            return None
        return np.maximum(res.x, 0.0)
    except Exception:
        return None


def _lp_step(f, rows, cols):
    c = _full_convolve(f, f)
    M = float(c.max())
    g_0 = _solve_lp(f, M, rows, cols)
    if g_0 is None or g_0.sum() < 1e-15:
        return f, M
    g = g_0 / g_0.sum()
    best_t = 0.0
    best_sc = _score_norm(f)
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        f_cand = (1.0 - t) * f + t * g
        sc = _score_norm(f_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.01, 0.02, 0.03, 0.05]:
            f_cand = (1.0 - t) * f + t * g
            sc = _score_norm(f_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        f = (1.0 - best_t) * f + best_t * g
    return f, M


def _lp_loop(a, rows, cols, deadline, max_stall=30, check_interval=5):
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    prev_sc = best_sc
    stall = 0
    count = 0
    while time.monotonic() < deadline:
        a, M = _lp_step(a, rows, cols)
        sc = _score_norm(a)
        count += 1
        if sc < prev_sc - 1e-10:
            stall = 0
        else:
            stall += 1
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        if count % check_interval == 0:
            full_sc = _score(a)
            if np.isfinite(full_sc) and full_sc < _score(best_a):
                best_a = a.copy()
        if stall > max_stall:
            break
        prev_sc = sc
    return best_a, count


def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=1000000, lr=0.01):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        w_rev = w[::-1]
        conv_aw = _full_convolve(a, w_rev)
        grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
        W = float(np.dot(a, grad_a))
        grad_z = a * (grad_a - W)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.5:
            update = update * (0.5 / update_norm)
        z = z - update
        z = np.clip(z, -50.0, 50.0)
        z = z - z.mean()
        if t % 200 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        if t % 5000 == 0:
            if best_score < prev_best * 0.999:
                current_lr = min(0.05, current_lr * 1.1)
            else:
                current_lr = max(0.001, current_lr * 0.95)
            prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

        a_alt, _ = _ilt_run(a_alt, N_alt, 0.3, 0.05, 1500, min(phase5_end - 30, time.monotonic() + 40))
        sc_alt_ilt = _score(a_alt)
        print(f"  n={n_alt} ILT: score={sc_alt_ilt:.10f}, t={time.monotonic()-t_start:.0f}s")

        # Quick LP
        if time.monotonic() < phase5_end - 20:
            rows_alt, cols_alt = _build_lp_indices(n_alt)
            a_alt, _ = _lp_loop(a_alt, rows_alt, cols_alt, phase5_end - 10, max_stall=20, check_interval=3)
            sc_alt_lp = _score(a_alt)
            print(f"  n={n_alt} LP: score={sc_alt_lp:.10f}, t={time.monotonic()-t_start:.0f}s")

        if sc_alt_ilt < best_score or (time.monotonic() - t_start < budget_s and sc_alt_lp < best_score):
            final_sc = _score(a_alt)
            if final_sc < best_score:
                best_score = final_sc
                best_seq = a_alt.copy()
                print(f"  NEW BEST at n={n_alt}: {best_score:.10f}")

    # === Phase 6: Final Adam (1600 - deadline) ===
    phase6_end = deadline - 5
    print(f"\n=== Phase 6: Final Adam (until t={phase6_end-t_start:.0f}s) ===")

    if time.monotonic() < phase6_end - 10:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_cf = float(c_final.max())
        a_final, _ = _adam_smooth(
            best_seq, phase6_end,
            beta_start=0.0005 * max_cf, beta_end=0.000005 * max_cf,
            max_iter=1000000, lr=0.005
        )
        full_sc = _score(a_final)
        print(f"  Final Adam: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if full_sc < best_score:
            best_score = full_sc
            best_seq = a_final.copy()

    # === Final validation ===
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.6f}")
    return [float(x) for x in fallback]
