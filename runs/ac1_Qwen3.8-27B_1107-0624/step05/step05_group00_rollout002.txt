import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_norm(a):
    """Score assuming sum(a) = 1, so score = 2*n*max(conv(a,a))"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _proj_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses power-approximation smoothing (Lp norm) instead of softmax, FISTA with
adaptive step size, Adam in log-space, and coordinate descent polishing.
Key innovation: Lp-norm smoothing gives a different optimization landscape
than softmax, and coordinate descent provides exact 1D subproblem solutions
for final polishing.
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    """Assumes sum(a)=1, returns 2*n*max(conv(a,a))."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_w(a, w):
    """Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a."""
    n = len(a)
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _lp_weights(c, p):
    """Compute weights for Lp approximation: w[k] = c[k]^(p-1) / (sum c^p)^((p-1)/p)
    Normalized so that sum_k w[k]*c[k] = (sum c^p)^(1/p)."""
    c_pos = np.maximum(c, 0)
    cp = c_pos ** p
    s = cp.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    w = c_pos ** (p - 1)
    # Normalize so that w^T c = (sum c^p)^(1/p)
    wc = float(w.dot(c_pos))
    target = s ** (1.0 / p)
    if wc > 1e-300:
        w = w * (target / wc)
    return w


def _fista_lp(a_init, p_schedule, deadline, max_iter_per_phase):
    """FISTA with Lp-approximation, p increasing through schedule."""
    n = len(a_init)
    a_curr = a_init.copy()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        a_prev = a_curr.copy()
        t_val = 1.0
        L = 4.0 * n  # Lipschitz constant estimate
        step = 1.0 / L

        for it in range(max_iter_per_phase):
            if time.monotonic() > deadline:
                break

            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            sy = y.sum()
            if sy > 1e-30:
                y = y / sy
            else:
                y = np.ones(n) / n

            c_y = _full_convolve(y, y)
            w = _lp_weights(c_y, p)
            grad = _grad_w(y, w)
            # Scale gradient: we're optimizing sum_k w[k]*c[k] which equals (sum c^p)^(1/p)
            # But the Lipschitz constant for this might differ
            a_next = _proj_simplex(y - step * grad)

            a_prev = a_curr.copy()
            a_curr = a_next
            t_val = t_next

        sc = _score_norm(a_curr)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, p_schedule, lr_init=0.01, max_iter=100000):
    """Adam optimizer in log-space with Lp-approximation weights."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for p in p_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = np.exp(z - z.max())
            a = a / a.sum()
            c = _full_convolve(a, a)
            w = _lp_weights(c, p)
            grad = _grad_w(a, w)

            # Convert to log-space gradient: d/dz = a * d/da
            grad_z = a * grad
            if not np.all(np.isfinite(grad_z)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad_z
            v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = np.exp(z - z.max())
                a_chk = a_chk / a_chk.sum()
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 10000 == 0:
                current_lr = max(0.001, current_lr * 0.95)

    a_final = np.exp(z - z.max())
    a_final = a_final / a_final.sum()
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descent_polish(a, deadline, max_sweeps=5):
    """Coordinate descent: for each coordinate, find optimal value via 1D convex minimization."""
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        # Random order
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() > deadline:
                break
            # Compute base convolution with a[i] = 0
            a_save = a[i]
            a[i] = 0.0
            c_base = _full_convolve(a, a)
            a[i] = a_save

            # Linear coefficient: how conv[k] changes when a[i] changes from 0 to x
            # conv[k] with a[i]=x: c_base[k] + x*(a[k-i] + a[2i-k]) [with bounds]
            # Special: if k=2i, there's also an x^2 term
            lin = np.zeros(2 * n - 1)
            # For each k, the linear term from a[i]*a[k-i]:
            if 0 <= k1 := (n - 1 - i):
                pass
            # a[i]*a[k-i] contributes to position k: valid when 0<=k-i<=n-1
            # So k ranges from i to i+n-1
            lo = i
            hi = i + n - 1
            if lo < 2 * n - 1:
                hi = min(hi, 2 * n - 2)
                if lo <= hi:
                    indices = np.arange(lo, hi + 1)
                    j_indices = indices - i  # 0 to n-1 (clipped)
                    lin[lo:hi + 1] += a[j_indices]
            # a[2i-k]*a[i] contributes to position k: valid when 0<=2i-k<=n-1
            # So k ranges from 2i-(n-1) to 2i
            lo2 = max(0, 2 * i - (n - 1))
            hi2 = min(2 * i, 2 * n - 2)
            if lo2 <= hi2:
                indices2 = np.arange(lo2, hi2 + 1)
                j_indices2 = 2 * i - indices2  # 0 to n-1 (clipped)
                lin[lo2:hi2 + 1] += a[j_indices2]

            # Quadratic term at k=2i (if valid)
            quad_k = 2 * i
            has_quad = (0 <= quad_k <= 2 * n - 2)

            # Now find x >= 0 minimizing max_k (c_base[k] + lin[k]*x + (x^2 if k==quad_k else 0))
            # subject to sum(a) + (x - a_save) = 1, i.e., x can be anything >= 0
            # (we'll re-normalize after)
            # Actually, since we're on the simplex, changing a[i] to x means we need to
            # re-normalize. Let's just optimize x >= 0 without the sum constraint,
            # then re-normalize.
            # The 1D problem: minimize max_k f_k(x) where f_k is linear or quadratic in x.
            # This is convex. Use bisection on subgradient.

            # The peak is at some k. The subgradient at x is:
            # For the active set A(x) = {k : f_k(x) = max_j f_j(x)}:
            # subgrad = mean of {lin[k] + (2x if k==quad_k else 0) for k in A(x)}
            # (or any convex combination)

            # Bisection: find x where subgradient changes sign
            lo_x = 0.0
            hi_x = a_save * 3.0 + 0.1
            for _ in range(40):
                mid_x = (lo_x + hi_x) / 2.0
                # Compute f_k at mid_x
                if has_quad:
                    vals = c_base + lin * mid_x
                    vals[quad_k] += mid_x * mid_x
                else:
                    vals = c_base + lin * mid_x
                peak_val = vals.max()
                # Active set
                active = np.where(np.isclose(vals, peak_val, atol=peak_val * 1e-10 + 1e-15))[0]
                # Subgradient at mid_x
                if has_quad:
                    grads_active = lin[active]
                    for ak in active:
                        if ak == quad_k:
                            grads_active = np.append(grads_active, 0)
                            break
                    # Actually: subgrad for f_k(x) = lin[k]*x + (x^2 if k==quad_k)
                    # is lin[k] + (2x if k==quad_k else 0)
                    sg = np.mean(lin[active] + (2 * mid_x if quad_k in active else 0))
                    # More carefully:
                    sg = 0.0
                    for ak in active:
                        g = lin[ak]
                        if ak == quad_k:
                            g += 2 * mid_x
                        sg += g
                    sg /= len(active)
                else:
                    sg = np.mean(lin[active])

                if sg < 0:
                    lo_x = mid_x
                else:
                    hi_x = mid_x

            x_opt = (lo_x + hi_x) / 2.0
            # Clamp to reasonable range
            x_opt = max(0.0, min(x_opt, a_save * 5.0 + 0.5))

            a[i] = x_opt
            # Re-normalize
            s = a.sum()
            if s > 1e-15:
                a = a / s

        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
        else:
            # No improvement, try smaller step
            break

    return a, _score_norm(a)


def _make_initialization(n, kind="uniform", rng=None):
    if rng is None:
        rng = np.random.default_rng(0)
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "gaussian":
        t = np.linspace(-3, 3, n)
        a = np.exp(-0.3 * t * t)
        return a / a.sum()
    elif kind == "raised_cosine":
        t = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "hann":
        t = np.arange(n)
        a = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1))) if n > 1 else np.ones(1)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "symmetric_rand":
        half = n // 2
        a_half = rng.random(half) + 0.1
        if n % 2 == 0:
            a = np.concatenate([a_half, a_half[::-1]])
        else:
            a = np.concatenate([a_half, a_half[-1], a_half[-2::-1]])
        return a / a.sum()
    elif kind == "sinusoidal":
        t = np.linspace(0, 2 * np.pi, n)
        a = 1.0 + 0.3 * np.cos(t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(-3, 3, n)
        a = np.exp(-0.5 * (t - 1.5) ** 2) + np.exp(-0.5 * (t + 1.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    e = np.exp(z - z.max())
    return e / e.sum()
