```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses top-K smoothed max gradient, FISTA + Adam in log-space, coordinate descent polishing,
and multi-length exploration (2048, 4096, 8192).

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_grad_conv(a, w):
    """Compute gradient of <w, conv(a,a)> wrt a.
    grad_i = 2 * sum_{k=0}^{n-1} w_{i+k} * a_k
    This is 2 * cross_correlation(a, w)[i]
    """
    n = len(a)
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # cross_corr(a,w)[i] = sum_k a[k]*w[k+i] = ifft(conj(fft(a)) * fft(w))
    h = np.fft.irfft(np.conj(fa) * fw, N)[:n]
    return 2.0 * h


def _topk_softmax_weights(c, beta, k):
    """Compute softmax weights over only the top-k elements of c."""
    n_c = len(c)
    if k >= n_c:
        k = n_c
    # Find top-k indices
    if k < n_c:
        topk_thresh = np.partition(c, -k)[-k]
        topk_mask = c >= topk_thresh
    else:
        topk_mask = np.ones(n_c, dtype=bool)
    
    topk_indices = np.where(topk_mask)[0]
    if len(topk_indices) == 0:
        topk_indices = np.array([int(np.argmax(c))])
    
    c_topk = c[topk_indices]
    c_max = float(c_topk.max())
    exp_vals = np.exp((c_topk - c_max) / beta)
    w_sum = float(exp_vals.sum())
    
    w = np.zeros(n_c)
    if w_sum < 1e-300 or not np.isfinite(w_sum):
        w[topk_indices[0]] = 1.0
    else:
        w[topk_indices] = exp_vals / w_sum
    return w


def _logspace_grad(a, c, w, n):
    """Compute gradient in log-space (z-space) for the normalized problem.
    a is on the simplex (sum=1).
    dF/dz_i = 2n * a_i * (grad_i - <grad, a>)
    where grad_i is the gradient of the smoothed max of conv(a,a) wrt a_i.
    """
    grad_a = _compute_grad_conv(a, w)
    mean_grad = float(np.dot(grad_a, a))
    return 2.0 * n * a * (grad_a - mean_grad)


def _softmax_normalize(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _fista_smoothed(a_init, deadline, beta, max_iter, top_k=100):
    """FISTA on simplex with top-K softened max."""
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0

    c_init = _full_convolve(a_curr, a_curr)
    max_c = float(c_init.max())
    L = 8.0 * n / max(1e-10, max_c)
    step = 1.0 / L

    best_score = _score_norm(a_curr)
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        w = _topk_softmax_weights(c_y, beta, top_k)
        grad = _compute_grad_conv(y, w)

        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = _score_norm(a_next)
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = _score_norm(a_curr)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()

    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, top_k, max_iter):
    """Adam optimizer in log-space with top-K smoothed max."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm(a_init)

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_normalize(z)
            c = _full_convolve(a, a)
            w = _topk_softmax_weights(c, beta, top_k)
            grad = _logspace_grad(a, c, w, n)

            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 1.0:
                update = update * (1.0 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_normalize(z)
                sc = _score_norm(a_chk)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 3000 == 0:
                current_lr = max(0.0005, current_lr * 0.97)

        # Reset momentum at phase boundaries
        m_buf *= 0.5
        v_buf *= 0.5

    a_final = _softmax_normalize(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_a, best_score


def _coord_descend_polish(a, n_passes, top_k_w, deadline):
    """Cyclic coordinate descent polishing.
    For each coordinate i, optimize a_i >= 0 (renormalizing the rest)
    using a 1D search that considers top constraints.
    """
    n = len(a)
    a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(12345)

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        order = rng.permutation(n)
        improved = 0
        for i in order:
            if time.monotonic() > deadline:
                break
            # Current state
            c = _full_convolve(a, a)
            # For coordinate i, (a*a)_j = c_j - a_i*a_{j-i} + t*a_{j-i} (if 0<=j-i<n)
            # = c_j + (t - a_i) * a_{j-i}
            # So d_j = a_{j-i} for valid j, c_j_new = c_j - a_i * a_{j-i}
            # S0 = sum(a) - a_i = 1 - a_i (assuming normalized)
            
            a_i_old = a[i]
            if a_i_old < 1e-15:
                # Try setting to small value
                a[i] = 1e-4
                a = a / a.sum()
                continue

            # Compute d_j = a_{j-i} for valid j
            # d_j = a[(j-i) % n] if 0 <= j-i < n else 0
            d = np.zeros(2 * n - 1)
            valid_j = np.arange(i, min(i + n, 2 * n - 1))
            d[valid_j] = a[valid_j - i]

            # c_j = (a*a)_j - a_i * d_j
            c_base = c - a_i_old * d

            # S0 = 1 - a_i_old (since sum a = 1)
            S0 = 1.0 - a_i_old

            # Find top-K j by c_base_j + d_j * t for t near a_i_old
            # Approximate: use c_base + d * a_i_old = c (current conv)
            # Find top-K positions
            idx_top = np.argsort(c)[-top_k_w:]
            
            # Evaluate F(t) = 2n * max_j (c_base[j] + d[j]*t) / (S0+t)^2 for candidate t values
            # Candidate t values: 0, a_i_old/2, a_i_old, 2*a_i_old, and values from 1D optimization
            t_candidates = [0.0, a_i_old * 0.5, a_i_old, a_i_old * 1.5, a_i_old * 2.0,
                           a_i_old * 0.7, a_i_old * 0.3, a_i_old * 1.2]
            
            # Also add critical points: for each active j, t* = S0 - 2*c_base[j]/d[j] (if d[j]!=0)
            for j in idx_top:
                if d[j] > 1e-15:
                    t_star = S0 - 2.0 * c_base[j] / d[j]
                    if 0 <= t_star <= 10.0 * max(a_i_old, 0.1):
                        t_candidates.append(t_star)
                        t_candidates.append(t_star * 0.8)
                        t_candidates.append(t_star * 1.2)

            t_candidates = [max(0.0, min(10.0, tc)) for tc in t_candidates]
            t_candidates = list(set([round(tc, 10) for tc in t_candidates]))
            if not t_candidates:
                t_candidates = [a_i_old]

            best_t = a_i_old
            best_f = np.inf
            for tc in t_candidates:
                c_trial = c_base + d * tc
                max_c = float(c_trial.max())
                f_val = 2.0 * n * max_c / ((S0 + tc) ** 2)
                if f_val < best_f:
                    best_f = f_val
                    best_t = tc

            if best_t != a_i_old:
                a[i] = best_t
                s_a = a.sum()
                if s_a > 1e-15:
                    a = a / s_a
                improved += 1

            if improved % 100 == 0:
                sc = _score_norm(a)
                if sc < best_score:
                    best_score = sc

    sc_final = _score_norm(a)
    if sc_final < best_score:
        best_score = sc_final
    return a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _structured_init(n, kind="cos_sum"):
    """Create structured initializations."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "cos_sum":
        # Sum of a few cosines with different frequencies
        a = np.ones(n)
        for freq in [1, 2, 3]:
            a += 0.3 * np.cos(2.0 * np.pi * freq * np.arange(n) / n)
        a = np.maximum(a, 0.01)
        return a / a.sum()
    elif kind == "tapered":
        t = np.linspace(0.01, 0.99, n)
        a = np.sin(np.pi * t) ** 0.7
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bimodal":
        t = np.linspace(0, 1, n)
        a = np.exp(-20 * (t - 0.3) ** 2) + np.exp(-20 * (t - 0.7) ** 2) + 0.1
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = float(parent_raw.sum())
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent
    else:
        n_parent = 4096
        parent = _structured_init(n_parent, "cos_sum")
        sc_parent = _score(parent)
        print(f"No parent. Structured init n={n_parent}, score={sc_parent:.10f}")
        best_seq = parent.copy()
        best_score = sc_parent

    # Try multiple lengths
    lengths_to_try = [4096, 8192, 2048]
    # If parent is 4096, try 8192 and 2048 as well
    if parent is not None and len(parent) == 4096:
        lengths_to_try = [4096, 8192, 2048]
    elif parent is not None and len(parent) == 8192:
        lengths_to_try = [8192, 4096, 2048]
    else:
        lengths_to_try = [4096, 2048, 8192]

    for n_try in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break

        elapsed = time.monotonic() - t_start
        remaining = deadline - time.monotonic()
        print(f"\n=== Trying n={n_try} (remaining: {remaining:.1f}s) ===")

        # Initialize for this length
        if n_try == n_parent:
            a0 = parent.copy()
        else:
            a0 = _interp_profile(parent, n_try)

        sc0 = _score(a0)
        print(f"  Initial: score={sc0:.10f}")
        if sc0 < best_score:
            best_score = sc0
            best_seq = a0.copy()

        # Phase A: FISTA with top-K softmax (coarse to fine)
        phase_a_budget = min(remaining * 0.35, 300.0)
        phase_a_end = time.monotonic() + phase_a_budget

        c_curr = _full_convolve(a0, a0)
        max_cc = float(c_curr.max())
        a_curr = a0.copy()

        # Beta schedule for FISTA
        betas_fista = []
        beta = 0.02 * max_cc
        while beta > 0.000005 * max_cc:
            betas_fista.append(beta)
            beta *= 0.55
        if len(betas_fista) > 8:
            betas_fista = betas_fista[:8]

        for bi, beta in enumerate(betas_fista):
            if time.monotonic() > phase_a_end - 3:
                break
            per_time = phase_a_end - time.monotonic() - 1
            max_it = max(2000, int(per_time * 3000))
            top_k = max(50, min(200, int(50 * (1 + bi * 0.5))))
            a_curr, sc = _fista_smoothed(a_curr, time.monotonic() + min(40, per_time),
                                         beta, max_it, top_k)
            a_curr = a_curr / a_curr.sum()
            sc_full = _score(a_curr)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = a_curr.copy()
                print(f"  FISTA beta={beta:.8f} k={top_k}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Phase B: Adam log-space (fine)
        phase_b_budget = min(remaining * 0.35, 350.0)
        phase_b_end = time.monotonic() + phase_b_budget

        c_curr = _full_convolve(a_curr, a_curr)
        max_cc = float(c_curr.max())
        betas_fine = []
        beta = 0.003 * max_cc
        while beta > 0.000001 * max_cc:
            betas_fine.append(beta)
            beta *= 0.5
        if len(betas_fine) > 12:
            betas_fine = betas_fine[:12]

        top_k_adam = 100
        a_opt, sc_opt = _adam_logspace(a_curr, phase_b_end, betas_fine,
                                       lr_init=0.01, top_k=top_k_adam,
                                       max_iter=200000)
        sc_opt_full = _score(a_opt)
        if sc_opt_full < best_score:
            best_score = sc_opt_full
            best_seq = a_opt.copy()
            print(f"  Adam fine: score={sc_opt_full:.10f} t={time.monotonic()-t_start:.1f}s")

        # Use best of the two
        a_work = a_opt if sc_opt < _score_norm(a_curr) else a_curr
        a_work = np.clip(a_work, 0, 1000)
        a_work = a_work / a_work.sum()

        # Phase C: Coordinate descent polishing
        phase_c_budget = min(remaining * 0.15, 120.0)
        phase_c_end = time.monotonic() + phase_c_budget
        if phase_c_budget > 10:
            a_cd, sc_cd = _coord_descend_polish(a_work, n_passes=3,
                                                top_k_w=50, deadline=phase_c_end)
            sc_cd_full = _score(a_cd)
            if sc_cd_full < best_score:
                best_score = sc_cd_full
                best_seq = a_cd.copy()
                print(f"  CoordDescend: score={sc_cd_full:.10f} t={time.monotonic()-t_start:.1f}s")
                a_work = a_cd

        # Phase D: Ultra-fine Adam polish with very small beta
        if time.monotonic() < deadline - 30:
            phase_d_end = min(deadline - 30, time.monotonic() + 60)
            c_work = _full_convolve(a_work, a_work)
            max_cw = float(c_work.max())
            betas_ult = [0.0005 * max_cw, 0.0002 * max_cw, 0.0001 * max_cw]
            a_ult, sc_ult = _adam_logspace(a_work, phase_d_end, betas_ult,
                                           lr_init=0.003, top_k=50,
                                           max_iter=100000)
            sc_ult_full = _score(a_ult)
            if sc_ult_full < best_score:
                best_score = sc_ult_full
                best_seq = a_ult.copy()
                print(f"  Ultra-fine: score={sc_ult_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Phase E: Diversity restarts from best
    if time.monotonic() < deadline - 30 and best_seq is not None:
        print(f"\n=== Diversity restarts (remaining: {deadline - time.monotonic():.1f}s) ===")
        n_best = len(best_seq)
        a_base = best_seq.copy()
        a_base = a_base / a_base.sum()

        n_restarts = 4
        per_restart = (deadline - time.monotonic() - 20) / n_restarts

        for ri in range(n_restarts):
            if time.monotonic() > deadline - 20:
                break
            if ri == 0:
                p_start = a_base + 0.02 * rng.standard_normal(n_best)
            elif ri == 1:
                # Shift perturbation
                shift = rng.integers(1, max(2, n_best // 1024))
                p_start = np.roll(a_base, shift).copy()
            elif ri == 2:
                # Frequency perturbation
                freq = rng.uniform(0.5, 3.0)
                pert = 0.03 * np.cos(2 * np.pi * freq * np.arange(n_best) / n_best)
                p_start = a_base + pert
            else:
                # Block perturbation
                block = n_best // 8
                p_start = a_base.copy()
                p_start[rng.integers(0, n_best) : rng.integers(0, n_best)] *= 1.1

            p_start = np.maximum(p_start, 1e-10)
            p_start = p_start / p_start.sum()

            c_s = _full_convolve(p_start, p_start)
            max_cs = float(c_s.max())
            betas_r = [0.002 * max_cs, 0.001 * max_cs, 0.0005 * max_cs, 0.0002 * max_cs]

            p_best, sc_best = _adam_logspace(p_start, time.monotonic() + max(5, per_restart),
                                             betas_r, lr_init=0.008, top_k=80,
                                             max_iter=50000)
            sc_full = _score(p_best)
            if sc_full < best_score:
                best_score = sc_full
                best_seq = p_best.copy()
                print(f"  Restart {ri}: score={sc_full:.10f} t={time.monotonic()-t_start:.1f}s")

    # Final: ensure valid output
    print(f"\n=== Final: best_score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0} ===")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) * (0.01 / 256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    # Fallback
    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
```
