"""
Optimized step function search using adaptive L-BFGS with log-sum-exp
smoothification, spectral initialization, and perturbation-based restarts.
Improves upon the previous 1.505966 bound by more aggressive beta continuation
and diverse initialization strategies.

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _topk_lse_obj(z, beta, n, K):
    """Log-sum-exp using only top-K lags for focused optimization."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    # Find top K lags
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    cm = float(c_top.max())
    vals = (c_top - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _topk_lse_grad(z, beta, n, K):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    top_idx = np.argpartition(c, -K)[-K:]
    c_top = c[top_idx]
    cm = float(c_top.max())
    vals = (c_top - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w_top = e / e.sum()
    # Map back to full w
    w = np.zeros(len(c))
    w[top_idx] = w_top
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

def _lbfgs_run(a_init, beta, n, maxiter=5000, ftol=1e-22, gtol=1e-15):
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    try:
        result = minimize(
            _lse_obj, z, args=(beta, n),
            method='L-BFGS-B', jac=_lse_grad,
            options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol,
                     'maxls': 50, 'maxfun': maxiter * 3}
        )
        if result.x is not None and np.all(np.isfinite(result.x)):
            a_new = _softmax_fn(result.x)
            return a_new
    except Exception:
        pass
    return a_init

def _adaptive_continuation(a_init, deadline, n_max_betas=30):
    """Adaptive beta continuation: only decrease beta when converged."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    # Start with a moderate beta
    beta = 0.1 * max_c0
    min_beta = 1e-6 * max_c0
    
    converged_count = 0
    total_iters = 0
    
    for round_num in range(n_max_betas):
        if time.monotonic() > deadline:
            break
        
        maxiter = 3000
        a_new = _lbfgs_run(best_a, beta, n, maxiter=maxiter)
        total_iters += maxiter
        
        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            converged_count = 0
        else:
            converged_count += 1
        
        # Check convergence: if no improvement for 2+ rounds at this beta, decrease
        if converged_count >= 2 or sc > best_score * 0.9999:
            beta *= 0.7
            converged_count = 0
        
        if beta < min_beta:
            beta = min_beta
        
        if round_num % 5 == 0:
            print(f"    cont round {round_num}: beta={beta:.2e}, score={best_score:.10f}")
    
    return best_a, best_score

def _spectral_init(n, n_iters=100, rng=None):
    """Gerchberg-Saxton style flat spectrum initialization."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    N = _next_pow2(2 * n - 1)
    a = rng.random(n)
    a = a / a.sum()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        # Target: flat magnitude spectrum
        mag = np.abs(A)
        target_mag = mag.mean() * np.ones_like(mag)
        # Blend: partial update
        phase = A / (mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new
    
    return a

def _perturb(a, rng, scale=0.01):
    """Add small random perturbation and renormalize."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    z = z + scale * rng.normal(size=n)
    z = z - z.mean()
    a_new = _softmax_fn(z)
    return a_new

def _make_profiles(n, rng=None):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Linear
    for slope in [0.5, 1.0, 2.0, -1.0, -2.0]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-12)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * (t - 0.5))
        profiles.append(a / a.sum())
    # Cosine power
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        profiles.append(a / a.sum())
    # Uniform
    a = np.ones(n)
    profiles.append(a / n)
    # Quadratic
    a = 1.0 + u**2
    profiles.append(a / a.sum())
    a = 1.0 + u**4
    profiles.append(a / a.sum())
    # Sigmoid
    for shift in [-0.3, -0.1, 0.1, 0.3]:
        a = 1.0 / (1.0 + np.exp(-10.0 * (u - shift)))
        profiles.append(a / a.sum())
    return profiles

"""
Hybrid SQP + adaptive L-BFGS optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses sequential quadratic programming (LP linearization) for fast local convergence,
combined with log-sum-exp smooth optimization for global progress.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    e = np.exp(vals)
    lse = cm + beta * float(np.log(np.sum(e)))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _topk_lse_obj(z, beta, n, K):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
        c_top = c[top_idx]
    else:
        c_top = c
    cm = float(c_top.max())
    vals = (c_top - cm) / beta
    e = np.exp(vals)
    lse = cm + beta * float(np.log(np.sum(e)))
    return 2.0 * n * lse


def _topk_lse_grad(z, beta, n, K):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    top_idx = np.argpartition(c, -K)[-K:]
    c_top = c[top_idx]
    cm = float(c_top.max())
    vals = (c_top - cm) / beta
    e = np.exp(vals)
    w_top = e / e.sum()
    w = np.zeros(len(c))
    w[top_idx] = w_top
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad


def _build_cross_matrix(a):
    n = len(a)
    n_constr = 2 * n - 1
    row_idx = np.concatenate([np.arange(j, j + n, dtype=np.int32) for j in range(n)])
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(a, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(n_constr, n))
    return A


def _sqp_step(a):
    n = len(a)
    c = _full_convolve(a, a)
    A = _build_cross_matrix(a)

    d_var = cp.Variable(n)
    M_var = cp.Variable()

    one_vec = np.ones(2 * n - 1)
    constraints = [
        2.0 * (A @ d_var) - M_var <= -c,
        d_var >= -a,
        cp.sum(d_var) == 0,
        M_var >= 0
    ]

    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
        if d_var.value is not None:
            d_val = np.asarray(d_var.value).ravel()
            a_new = a + d_val
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                return a, None
            a_new = a_new / s
            return a_new, float(M_var.value)
    except Exception:
        pass
    return a, None


def _sqp_iterate(a, n_iters=50, deadline=None):
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        a_new, M_val = _sqp_step(a)
        if a_new is None or np.array_equal(a_new, a):
            break
        sc = _score_norm_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if it % 10 == 0:
            print(f"    SQP iter {it}: score={sc:.10f}, best={best_score:.10f}, M={M_val:.8f}")

    return best_a, best_score


def _lbfgs_run(a_init, beta, n, maxiter=5000, use_topk=False, K=100):
    z = np.log(np.maximum(a_init, 1e-18))
    z = z - z.mean()
    if use_topk and K < 2 * n - 1:
        try:
            result = minimize(
                _topk_lse_obj, z, args=(beta, n, K),
                method='L-BFGS-B', jac=_topk_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 50, 'maxfun': maxiter * 3}
            )
        except Exception:
            result = None
    else:
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15,
                         'maxls': 50, 'maxfun': maxiter * 3}
            )
        except Exception:
            result = None
    if result is not None and result.x is not None and np.all(np.isfinite(result.x)):
        a_new = _softmax_fn(result.x)
        return a_new
    return a_init


def _adaptive_continuation(a_init, deadline, n_max_betas=40, use_topk=False, K=100):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    beta = 0.1 * max_c0
    min_beta = 1e-7 * max_c0
    no_improve_count = 0

    for rnd in range(n_max_betas):
        if time.monotonic() > deadline:
            break

        remaining_betas = n_max_betas - rnd
        time_per_beta = (deadline - time.monotonic()) / max(1, remaining_betas)
        time_per_beta = min(time_per_beta, 120.0)
        beta_dl = time.monotonic() + time_per_beta
        maxiter = max(500, int(time_per_beta * 20))

        a_new = _lbfgs_run(best_a, beta, n, maxiter=maxiter, use_topk=use_topk, K=K)
        sc = _score_norm_fn(a_new)

        if np.isfinite(sc) and sc < best_score * (1 - 1e-10):
            best_score = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        if no_improve_count >= 2:
            beta *= 0.6
            no_improve_count = 0

        if beta < min_beta:
            beta = min_beta
            if no_improve_count >= 3:
                break

        if rnd % 5 == 0:
            print(f"    cont {rnd}: beta={beta:.2e}, score={best_score:.10f}")

    return best_a, best_score


def _spectral_init(n, n_iters=150, rng=None):
    if rng is None:
        rng = np.random.default_rng(123)
    N = _next_pow2(2 * n - 1)
    a = rng.random(n)
    a = a / a.sum()

    for it in range(n_iters):
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        mag[mag < 1e-30] = 1e-30
        phase = A / mag
        target_mag = mag.mean() * np.ones_like(mag)
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        # Blend to avoid divergence
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

    return a


def _perturb(a, rng, scale=0.01):
    n = len(a)
    z = np.log(np.maximum(a, 1e-18))
    z = z + scale * rng.normal(size=n)
    z = z - z.mean()
    return _softmax_fn(z)


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _make_diverse_inits(n, rng):
    inits = []
    t = np.linspace(0, 1, n)
    u = 2 * t - 1

    # Cosine-based
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        inits.append(a / a.sum())

    # Sigmoid-like
    for shift in [-0.4, -0.2, 0.0, 0.2, 0.4]:
        a = 1.0 / (1.0 + np.exp(-8.0 * (u - shift)))
        inits.append(a / a.sum())

    # Power law
    for gamma in [0.3, 0.5, 0.7, 1.0, 1.5]:
        a = t ** gamma + 0.01
        inits.append(a / a.sum())
        a2 = (1 - t) ** gamma + 0.01
        inits.append(a2 / a2.sum())

    # Random
    for _ in range(3):
        a = rng.random(n)
        inits.append(a / a.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 40.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = _make_diverse_inits(n_init, rng)[4]
        update_best(parent, _score_norm_fn(parent), "default")

    n_parent = len(parent)
    print(f"Starting with n={n_parent}")

    # Phase 1: L-BFGS adaptive continuation on parent (500s)
    phase1_end = min(deadline, t_start + 500.0)
    print(f"\n=== Phase 1: L-BFGS continuation n={n_parent} (500s) ===")
    a_best, sc_best = _adaptive_continuation(parent, phase1_end, n_max_betas=50)
    update_best(a_best, sc_best, "pbfgs_p1")

    # Phase 2: SQP iterations (300s)
    phase2_end = min(deadline, t_start + 800.0)
    print(f"\n=== Phase 2: SQP iterations (300s) ===")
    if time.monotonic() < phase2_end:
        a_sqp, sc_sqp = _sqp_iterate(best_seq, n_iters=80, deadline=phase2_end)
        update_best(a_sqp, sc_sqp, "sqp")

    # Phase 3: Top-K L-BFGS refinement (200s)
    phase3_end = min(deadline, t_start + 1000.0)
    print(f"\n=== Phase 3: Top-K L-BFGS (200s) ===")
    if time.monotonic() < phase3_end and best_seq is not None:
        n_curr = len(best_seq)
        c_curr = _full_convolve(best_seq, best_seq)
        max_c_curr = float(c_curr.max())
        for K in [20, 50, 100]:
            if time.monotonic() > phase3_end - 20:
                break
            beta = 0.01 * max_c_curr
            per_dl = min(phase3_end - 10, time.monotonic() + (phase3_end - phase2_end) / 3)
            a_k, sc_k = _adaptive_continuation(best_seq, per_dl, n_max_betas=20, use_topk=True, K=K)
            update_best(a_k, sc_k, f"topk{K}")

    # Phase 4: Multi-restart with perturbations (300s)
    phase4_end = min(deadline, t_start + 1300.0)
    print(f"\n=== Phase 4: Multi-restart (300s) ===")
    if time.monotonic() < phase4_end:
        n_restarts = 6
        per_restart = (phase4_end - time.monotonic()) / max(1, n_restarts)
        for r in range(n_restarts):
            if time.monotonic() > phase4_end - 15:
                break
            scale = 0.005 + 0.005 * r
            a_pert = _perturb(best_seq, rng, scale=scale)
            per_dl = min(phase4_end - 5, time.monotonic() + per_restart)
            a_opt, sc_opt = _adaptive_continuation(a_pert, per_dl, n_max_betas=15)
            update_best(a_opt, sc_opt, f"restart{r}")

            # Try SQP on the restart result
            if time.monotonic() < phase4_end - 10 and sc_opt < best_score + 1e-8:
                per_dl2 = min(phase4_end - 5, time.monotonic() + 30)
                a_sqp2, sc_sqp2 = _sqp_iterate(a_opt, n_iters=20, deadline=per_dl2)
                update_best(a_sqp2, sc_sqp2, f"restart{r}_sqp")

    # Phase 5: Try larger n (250s)
    phase5_end = min(deadline, t_start + 1550.0)
    print(f"\n=== Phase 5: Larger n (250s) ===")
    if time.monotonic() < phase5_end and best_seq is not None:
        n_large = 8192
        if len(best_seq) < n_large:
            p_large = _interp_profile(best_seq, n_large)
            sc_large = _score_norm_fn(p_large)
            update_best(p_large, sc_large, f"interp_n{n_large}")

            if time.monotonic() < phase5_end - 60:
                per_dl = min(phase5_end - 10, time.monotonic() + 180.0)
                a_large, sc_large_opt = _adaptive_continuation(p_large, per_dl, n_max_betas=20)
                update_best(a_large, sc_large_opt, f"large_n_opt")

                if time.monotonic() < phase5_end - 15:
                    per_dl2 = min(phase5_end - 5, time.monotonic() + 50)
                    a_large_sqp, sc_ls = _sqp_iterate(a_large, n_iters=20, deadline=per_dl2)
                    update_best(a_large_sqp, sc_ls, "large_n_sqp")

    # Phase 6: Final ultra-fine optimization (remaining time)
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        # Ultra-小 beta L-BFGS
        for beta_frac in [5e-5, 2e-5, 1e-5, 5e-6, 1e-6, 5e-7]:
            if time.monotonic() > deadline - 30:
                break
            beta = beta_frac * max_c_final
            maxiter = 20000
            a_new = _lbfgs_run(best_seq, beta, n_final, maxiter=maxiter)
            sc_new = _score_norm_fn(a_new)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_seq = a_new.copy()
                print(f"    ultra-beta {beta_frac:.0e}: score={best_score:.10f}")
            else:
                break

        # Final SQP polish
        if time.monotonic() < deadline - 20:
            per_dl = min(deadline - 10, time.monotonic() + 60)
            a_final_sqp, sc_fs = _sqp_iterate(best_seq, n_iters=30, deadline=per_dl)
            update_best(a_final_sqp, sc_fs, "final_sqp")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    return [float(x) for x in fallback]
