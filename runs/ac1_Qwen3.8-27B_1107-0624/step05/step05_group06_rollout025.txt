    # (w ∘ a)[j] = sum_{l=0}^{n-1} w[j+l] * a[l]
    # This is a cross-correlation. Use FFT.
    N = _next_pow2(2 * n - 1)
    fw = np.fft.rfft(w, N)
    # We need: result[j] = sum_l w[j+l] a[l]
    # This is the cross-correlation of w and a, which is ifft(conj(fft(a)) * fft(w))
    # But with the specific indexing... let me think.
    # result[j] = sum_l w[j+l] a[l] = (w * flip(a))[j + n - 1] ... 
    # Actually, let's use: result = ifft(fft(w) * conj(fft(a))) 
    # Hmm, let me just use the direct formula via the convolution trick.
    # result[j] = sum_l w[j+l] a[l]
    # Let w_flip = w[::-1], length 2n-1
    # Then result[j] = sum_l w[j+l] a[l] = (w * a)[2n-2-j] (full convolution, reversed)
    # Actually: (w * a)[k] = sum_i w[i] a[k-i]
    # result[j] = sum_l w[j+l] a[l], let i = j+l, l = i-j
    # = sum_i w[i] a[i-j] = (w * flip(a))[j] ... no
    # Let me just use numpy correlate or direct FFT.
    # result[j] = sum_{l=0}^{n-1} w[j+l] a[l]
    # = full_conv(w, a)[j + 2n - 2 - (n-1)] ... this is getting confusing.
    # Let's use the correlation definition:
    # numpy.correlate(w, a, mode='full')[j + n - 1] = sum_k w[k] a[k - (j+n-1) + n - 1]
    # This is getting messy. Let me just use the FFT approach directly.
    
    # result[j] = sum_{l=0}^{n-1} w[j+l] * a[l]
    # Pad w to length N, pad a to length N (reversed)
    # Actually the cleanest: result = IFFT(FFT(w) * FFT(a_padded_reversed))
    # No, let's just compute it as: result[j] = sum_l w[j+l] a[l]
    # This is a "sliding dot product" which is a cross-correlation.
    # In FFT terms: result = ifft(fft(w) * conj(fft(a))) gives correlation
    # where (f*g)[n] = sum_k f[k] conj(g[k-n])
    # We want result[j] = sum_l w[j+l] a[l] = sum_l w[j+l] a[l]
    # Let's define: h[m] = sum_l w[l] a[l+m] ... no that's different.
    
    # Simplest: use the fact that result[j] = (w * a_rev)[2n-2-j]
    # where a_rev = a[::-1] and * is full convolution.
    # (w * a_rev)[k] = sum_i w[i] a_rev[k-i] = sum_i w[i] a[n-1-(k-i)] = sum_i w[i] a[i+k-n+1]
    # Let l = i+k-n+1, i = l-k+n-1:
    # = sum_l w[l-k+n-1] a[l] ... this is getting circular.
    
    # Let me just brute-force with FFT properly.
    # Define: result[j] = sum_{l=0}^{n-1} w[j+l] a[l]
    # This is the cross-correlation R_wa[j] = sum_l w[l+j] a[l]
    # In FFT: R = ifft(conj(fft(a, N)) * fft(w, N)) ... 
    # Actually numpy.correlate(w, a, 'full') gives c[k] = sum_n w[n] conj(a[n-k+N-1])
    # For real: c[k] = sum_n w[n] a[n-k+N-1]
    # We want result[j] = sum_l w[j+l] a[l]. Let n = j+l, l = n-j:
    # = sum_n w[n] a[n-j] (where n ranges over j to j+n-1, but w has length 2n-1)
    # numpy.correlate(w, a, 'full')[j] = sum_n w[n] a[n - j + (len(a)-1)] ... 
    # I'm overcomplicating this. Let me just use a direct but efficient method.
    
    # The cleanest FFT approach:
    # result[j] = sum_{l=0}^{n-1} w[j+l] a[l]
    # = <w[j:j+n], a>
    # This is the aperiodic cross-correlation. In FFT:
    # R = ifft(fft(w, N) * conj(fft(a, N))) ... no, that gives circular.
    # For aperiodic: R[j] = ifft(fft(w, N) * conj(fft(a, N)))[j] ... with N >= len(w) + len(a) - 1
    
    # Actually, the standard formula: 
    # cross_correlation(f, g)[k] = sum_n f[n+k] conj(g[n])
    # = ifft(conj(FFT(g)) * FFT(f)) with appropriate zero-padding for N >= max(len(f), len(g)) + min(len(f), len(g)) - 1
    
    # Here f = w (length 2n-1), g = a (length n)
    # We want: sum_l w[l+j] a[l] for j = 0, ..., n-1
    # This is cross_correlation(w, a)[j] for j = 0, ..., n-1
    # = ifft(conj(FFT(a, N)) * FFT(w, N))[j] where N >= (2n-1) + n - 1 = 3n - 2
    
    N = _next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    R = np.fft.irfft(np.conj(fa) * fw, N)
    # R[j] = sum_l w[l+j] a[l] for j = 0, ..., 2n-2 (valid range)
    return R[:n]


def _grad_smoothed(a, beta):
    """Gradient of 2n * softmax(c) / S^2 with respect to a."""
    n = len(a)
    c = _full_convolve(a, a)
    c_max = c.max()
    S = float(a.sum())
    if S < 1e-15:
        return np.zeros(n), 0.0
    
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        M_soft = c_max
    else:
        w = e / s
        M_soft = c_max + beta * np.log(s)
    
    # cross = (w ∘ a)[j] = sum_k w[k] a[k-j] for 0<=k-j<n
    cross = _cross_corr_w_a(w, a)
    
    # grad_j = (2n/S^2) * (2*cross[j] - 2*a[j]*M_soft/S)
    grad = (2.0 * n / (S * S)) * (2.0 * cross - 2.0 * a * M_soft / S)
    
    obj = 2.0 * n * M_soft / (S * S)
    return grad, obj


def _obj_smoothed(a, beta):
    """Smoothed objective value."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        M_soft = c_max
    else:
        M_soft = c_max + beta * np.log(s)
    return 2.0 * n * M_soft / (S * S)


def _lbfgs_direct(a_init, n, deadline, max_iter=5000):
    """L-BFGS-B directly on a >= 0 (no sum constraint, using scale invariance)."""
    current_a = a_init.copy()
    current_sc = _score(current_a)
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    S = float(current_a.sum())
    scale = max_c / (S * S) if S > 0 else max_c
    
    betas = np.array([0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003, 0.001, 0.0005, 0.0002]) * scale
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        
        def fun(x):
            return _obj_smoothed(x, beta)
        
        def jac(x):
            g, _ = _grad_smoothed(x, beta)
            return g
        
        try:
            result = minimize(
                fun, current_a, jac=jac, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxcor': 50, 'maxls': 200, 'disp': False}
            )
            a_new = np.maximum(result.x, 0)
            if a_new.sum() < 1e-15:
                continue
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
            else:
                # Keep current_a if no improvement
                pass
        except Exception as e:
            print(f"    L-BFGS direct error at beta={beta:.2e}: {e}")
    
    return current_a


def _subgradient_bb(a_init, deadline, n_max=5000):
    """Subgradient descent with Barzilai-Borwein step size and exact active set."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score(a)
    
    c = _full_convolve(a, a)
    c_max = float(c.max())
    S = float(a.sum())
    
    # BB parameters
    s_prev = None
    y_prev = None
    step = 1.0 / (S * S * max_c) if max_c > 0 else 0.01
    max_c = c_max
    
    for i in range(n_max):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        c_max = float(c.max())
        S = float(a.sum())
        if S < 1e-15:
            break
        
        # Active set: coefficients within 0.5% of max
        tol = 0.005 * c_max
        active = c >= (c_max - tol)
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[int(np.argmax(c))] = True
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        cross = _cross_corr_w_a(w, a)
        # grad of M/S^2: (2n/S^2)(2*cross - 2*a*M/S)
        grad = (2.0 * n / (S * S)) * (2.0 * cross - 2.0 * a * c_max / S)
        
        # BB step size
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-18:
            break
        
        if s_prev is not None and y_prev is not None:
            sy = float(np.dot(s_prev, y_prev))
            if sy > 1e-15:
                yy = float(np.dot(y_prev, y_prev))
                ss = float(np.dot(s_prev, s_prev))
                step_bb2 = ss / sy  # BB2
                step_bb1 = sy / yy  # BB1
                # Damped BB
                step = 0.5 * (step_bb1 + step_bb2)
                step = max(1e-10, min(step, 10.0 * step))
        
        # Step
        s_curr = -step * grad
        a_new = a + s_curr
        a_new = np.maximum(a_new, 0)
        
        if time.monotonic() > deadline:
            break
        
        c_new = _full_convolve(a_new, a_new)
        c_max_new = float(c_new.max())
        S_new = float(a_new.sum())
        
        if S_new > 1e-15:
            sc_new = 2.0 * n * c_max_new / (S_new * S_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
        
        y_curr = c_max_new - c_max  # change in objective (approximate)
        s_prev = s_curr
        y_prev = np.gradient(c_max_new - c_max)[:, None] * 0  # placeholder
        # Actually for BB we need y = g_new - g_old, but that's expensive.
        # Use simpler: step decay
        step = step * 0.999
        
        a = a_new
        c_max = c_max_new
    
    return best_a, best_sc


def _build_A_matrix(f, n):
    """Build (2n-1) x n matrix for LP: A[k,j] = f[k-j] if 0<=k-j<n."""
    A = lil_matrix((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A.tocsr()


def _lp_solve(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    c_obj = -np.ones(n)
    A = _build_A_matrix(f, n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if result.success:
            return np.maximum(result.x, 0)
    except Exception:
        pass
    return None


def _lp_phase(f, n, deadline, max_iters=5000):
    """LP-based fixed-point iteration."""
    M = float(_full_convolve(f, f).max())
    S = float(f.sum())
    t = 0.08  # mixing parameter
    best_a = f.copy()
    best_sc = _score(f)
    
    for i in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        g = _lp_solve(f, M, n)
        if g is None:
            t *= 0.8
            continue
        
        S_g = float(g.sum())
        if S_g < 1e-15:
            break
        
        # Normalize g to same sum as f
        g_norm = g / S_g * S
        f_new = (1 - t) * f + t * g_norm
        f_new = np.maximum(f_new, 0)
        
        M_new = float(_full_convolve(f_new, f_new).max())
        
        if i % 200 == 0 or i < 10:
            sc = _score(f_new)
            if sc < best_sc:
                best_sc = sc
                best_a = f_new.copy()
            if i % 500 == 0:
                print(f"    LP iter {i}: score={sc:.10f}, M={M_new:.6f}, "
                      f"t={time.monotonic() - t_start:.1f}s")
        
        # Adaptive step
        if M_new > M * 1.01:
            t = max(0.01, t * 0.7)
        elif i % 500 == 0 and t < 0.08:
            t = min(0.08, t * 1.1)
        
        f = f_new
        M = M_new
    
    return best_a, best_sc


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smoobj_z(z, beta, n):
    """Smoothed objective in z-space (softmax parameterization)."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _smograd_z(z, beta, n):
    """Gradient in z-space for softmax parameterization."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _lbfgs_softmax(a_init, n, deadline, max_iter=5000):
    """L-BFGS in softmax z-space."""
    current_a = a_init.copy()
    current_sc = _score(current_a)
    z = np.log(np.maximum(current_a, 1e-15))
    z = z - z.mean()
    
    c_cur = _full_convolve(current_a, current_a)
    max_c = float(c_cur.max())
    
    betas = max_c * np.array([0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001])
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        
        def fun(z_):
            return _smoobj_z(z_, beta, n)
        
        def jac(z_):
            return _smograd_z(z_, beta, n)
        
        try:
            result = minimize(
                fun, z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                         'maxcor': 40, 'disp': False}
            )
            a_new = _softmax(result.x)
            sc_new = _score(a_new)
            if sc_new < current_sc - 1e-15:
                current_a = a_new
                current_sc = sc_new
                z = result.x - result.x.mean()
        except Exception:
            pass
    
    return current_a
