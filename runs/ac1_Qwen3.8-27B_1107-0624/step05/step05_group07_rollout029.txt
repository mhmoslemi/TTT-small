def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

            if not np.all(np.isfinite(gy)):
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            z_new = y - lr * gy + ((t_k - 1.0) / t_new) * (y - x)
            z_new = np.clip(z_new, -80, 80)
            z_new = z_new - z_new.mean()
            x = y
            y = z_new
            t_k = t_new
            lr = min(lr * 1.001, 1.0)

            if it % 200 == 0:
                a_check = _softmax_fn(y)
                sc = _score_norm_fn(a_check)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
                    z = y.copy()

        a_final = _softmax_fn(z)
        sc_final = _score_norm_fn(a_final)
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            z = z.copy()

    return best_a, best_score


def _subgradient_projection(a_init, deadline, max_iters=50000):
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a_init, _score_norm_fn(a_init)
    a = a / s

    best_a = a.copy()
    best_score = _score_norm_fn(a)
    f_lower = 2.0 * n / (2.0 * n - 1.0)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        max_c = float(c.max())
        f_val = 2.0 * n * max_c

        v = np.zeros(n)
        idx = np.arange(n)
        valid = (k_star - idx >= 0) & (k_star - idx < n)
        v[valid] = a[k_star - idx[valid]]
        v[k_star - np.where(valid)[0]] = np.where(valid, v, v)
        # Duplicates when k_star - i == i
        half = (k_star % 2 == 0)
        if half:
            v[k_star // 2] = 2.0 * a[k_star // 2]
        else:
            pass

        # Actually, the gradient of conv(a,a)_k w.r.t. a_j is:
        # 2*a_{k-j} if 0 <= k-j < n, else 0
        # (with the factor of 2 accounting for both i=j and i=k-j contributing)
        # Let me just compute it properly:
        g = np.zeros(n)
        for j in range(min(n, k_star + 1)):
            if k_star - j < n and k_star - j >= 0:
                g[j] += 2.0 * a[k_star - j]
        # But this is O(n) per iteration which is fine for n=4096

        g = 4.0 * n * g
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break

        eta = (f_val - f_lower) / (g_norm * g_norm + 1e-15)
        eta = min(eta, 0.1)

        a_new = a - eta * g
        # Project onto simplex
        a_new = _project_simplex(a_new)

        sc = _score_norm_fn(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _active_set_lp(f, deadline, top_k=10, max_rounds=50):
    n = len(f)
    a = f.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(a, a)
        M = float(c.max())

        # Find top-k lags
        top_lags = np.argsort(c)[::-1][:top_k]

        # Build constraint matrix for these lags
        rows = []
        for k in top_lags:
            row = np.zeros(n)
            for i in range(n):
                if 0 <= k - i < n:
                    row[i] = a[k - i] * 2.0
            rows.append(row)
        A = 2.0 * np.array(rows)  # Gradient of each active q_k w.r.t. a

        # LP: maximize sum(a) subject to A @ a <= M, a >= 0
        # (We want to find a direction that reduces the active lags)
        # Instead, solve: min t subject to A @ a <= t, sum(a) = 1, a >= 0
        # This is the same as the LP in the previous approach but with fewer constraints

        try:
            import cvxpy as cp
            g = cp.Variable(n)
            t = cp.Variable()
            constraints = [
                A @ g <= t,
                cp.sum(g) == 1,
                g >= 0
            ]
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=cp.CLARABEL)
            if g.value is not None:
                g_val = np.maximum(g.value, 0)
                gs = g_val.sum()
                if gs > 1e-15:
                    g_val = g_val / gs
                    a_new = 0.5 * a + 0.5 * g_val
                    a_new = _project_simplex(a_new)
                    sc = _score_norm_fn(a_new)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    a = a_new
        except Exception:
            break

    return best_a, best_score


def _nesterov_lbfgs(a_init, deadline, beta_list=None):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    if beta_list is None:
        fracs = [0.1, 0.05, 0.03, 0.02, 0.015, 0.01, 0.008, 0.006,
                 0.005, 0.004, 0.003, 0.002, 0.0015, 0.001,
                 0.0008, 0.0006, 0.0005, 0.0004, 0.0003,
                 0.0002, 0.00015, 0.0001, 0.00008, 0.00005,
                 0.00003, 0.00002, 0.00001, 0.000005]
        beta_list = [b * max_c0 for b in fracs]

    n_betas = len(beta_list)
    for idx, beta in enumerate(beta_list):
        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_per = (deadline - time.monotonic()) / max(1, remaining)
        time_per = min(time_per, 80.0)
        level_dl = time.monotonic() + time_per
        maxiter = max(300, int(time_per * 40))

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-16,
                         'maxls': 60, 'maxfun': maxiter * 6}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _local_search(a_init, deadline, rng, max_iter=200000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    c = _full_convolve(a, a)
    max_c = float(c.max())
    temperature = 0.01 * max_c

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Random perturbation of a few coordinates
        n_perturb = rng.integers(1, min(10, n // 4) + 1)
        indices = rng.integers(0, n, size=n_perturb)
        a_perturbed = a.copy()
        for idx in indices:
            delta = rng.normal(0, 0.01)
            a_perturbed[idx] = max(0, a_perturbed[idx] + delta)
        s = a_perturbed.sum()
        if s < 1e-15:
            continue
        a_perturbed = a_perturbed / s

        sc = _score_norm_fn(a_perturbed)
        if np.isfinite(sc):
            if sc < best_score:
                best_score = sc
                best_a = a_perturbed.copy()
                a = a_perturbed.copy()
                c = _full_convolve(a, a)
                max_c = float(c.max())
            elif sc < _score_norm_fn(a) + temperature:
                # Accept with some probability (simulated annealing)
                if rng.random() < np.exp(-(sc - _score_norm_fn(a)) / (temperature + 1e-15)):
                    a = a_perturbed.copy()
        if it % 5000 == 0:
            temperature *= 0.95

    return best_a, best_score


def _coord_fine(a_init, deadline, max_rounds=20):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        cur_score = _score_norm_fn(a)
        improved = False
        # Focus on coordinates that contribute most to the peak
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        # Coordinates that matter for lag k_star
        contrib = np.zeros(n)
        for j in range(min(n, k_star + 1)):
            if 0 <= k_star - j < n:
                contrib[j] = abs(a[k_star - j])
        # Sort by contribution
        order = np.argsort(contrib)[::-1]

        for j in order[:min(200, n)]:
            if time.monotonic() > deadline:
                break
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                z_trial = z.copy()
                z_trial[j] += delta
                a_trial = _softmax_fn(z_trial)
                sc = _score_norm_fn(a_trial)
                if sc < cur_score:
                    z[j] += delta
                    z = np.clip(z, -80, 80)
                    z = z - z.mean()
                    a = _softmax_fn(z)
                    cur_score = _score_norm_fn(a)
                    improved = True
                    break

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        if not improved:
            # Try smaller steps
            for j in order[:min(50, n)]:
                if time.monotonic() > deadline:
                    break
                for delta in [0.0001, -0.0001, 0.0005, -0.0005]:
                    z_trial = z.copy()
                    z_trial[j] += delta
                    a_trial = _softmax_fn(z_trial)
                    sc = _score_norm_fn(a_trial)
                    if sc < cur_score:
                        z[j] += delta
                        z = np.clip(z, -80, 80)
                        z = z - z.mean()
                        a = _softmax_fn(z)
                        cur_score = _score_norm_fn(a)
                        improved = True
                        break
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            if not improved:
                break

    return best_a, best_score


def _symmetrize(a):
    return (a + a[::-1]) / 2.0


def _make_diverse_init(n, rng):
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    inits = []
    # Symmetric profiles
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-12) ** p
        inits.append(a / a.sum())
    # Edge-concentrated
    for sigma in [0.1, 0.2, 0.3]:
        a = np.exp(-((t - 0.5) ** 2) / (2 * sigma ** 2))
        a_edge = np.exp(-(t ** 2) / (2 * sigma ** 2)) + np.exp(-((t - 1) ** 2) / (2 * sigma ** 2))
        inits.append(a / a.sum())
        inits.append(a_edge / a_edge.sum())
    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 20, size=5)
        amps = rng.normal(0, 1, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        a = 1.0 + 0.3 * sum(amps[i] * np.cos(2 * np.pi * freqs[i] * t + phases[i]) for i in range(5))
        a = np.maximum(a, 1e-12)
        inits.append(a / a.sum())
    # Uniform
    inits.append(np.ones(n) / n)
    return inits


def _grad_based_refine(a_init, deadline, rng):
    """Nesterov-accelerated gradient descent in log-domain with adaptive beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c = _full_convolve(a_init, a_init)
    max_c = float(c.max())
    beta = 0.05 * max_c
    lr = 0.3
    momentum = 0.9

    v = np.zeros(n)
    x_prev = z.copy()

    for it in range(200000):
        if time.monotonic() > deadline:
            break
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        # Adaptive beta: smaller when we're close to the max
        n_active = int(np.sum(c > 0.9 * cm))
        beta = max(1e-6 * cm, 0.02 * cm / max(1, n_active))

        vals = np.clip((c - cm) / beta, -50, 50)
        e = np.exp(vals)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        # Nesterov step
        v = momentum * v - lr * grad
        z_new = z + v
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()

        a_new = _softmax_fn(z_new)
        sc = _score_norm_fn(a_new)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        if sc < _score_norm_fn(x_prev):
            z = z_new
        else:
            v = momentum * v * 0.5
            z = z + v

        x_prev = z.copy()

        # Adaptive learning rate
        if it % 1000 == 0:
            lr = max(0.01, lr * 0.97)
            momentum = min(0.999, momentum + 0.001)

        # Periodic noise to escape local minima
        if it % 10000 == 0 and it > 0:
            z = z + 0.01 * rng.normal(size=n)
            z = z - z.mean()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            elapsed = time.monotonic() - t_start
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] t={elapsed:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_init = 4096
        parent = np.ones(n_init) / n_init
        update_best(parent, _score_norm_fn(parent), "default")

    n = len(parent)
    print(f"Starting with n={n}, score={best_score:.10f}")

    # Phase 1: Nesterov gradient refinement (350s)
    phase1_end = min(deadline, t_start + 350.0)
    print(f"\nPhase 1: Nesterov gradient refinement (350s)")

    a_cur = parent.copy()
    p_best, sc_best = _grad_based_refine(a_cur, phase1_end, rng)
    update_best(p_best, sc_best, "nesterov_grad")

    # Also try symmetrized version
    a_sym = _symmetrize(parent)
    a_sym = a_sym / a_sym.sum()
    sc_sym = _score_norm_fn(a_sym)
    update_best(a_sym, sc_sym, "symmetrized")
    p_best_sym, sc_best_sym = _grad_based_refine(a_sym, min(phase1_end, time.monotonic() + 100), rng)
    update_best(p_best_sym, sc_best_sym, "nesterov_sym")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # Phase 2: L-BFGS continuation with fine beta (500s)
    phase2_end = min(deadline, t_start + 850.0)
    print(f"\nPhase 2: L-BFGS continuation (500s)")

    n_restarts = 4
    per_restart_time = (phase2_end - time.monotonic()) / max(1, n_restarts)

    for r in range(n_restarts):
        if time.monotonic() > phase2_end - 20:
            break

        if r == 0:
            a_cur = best_seq.copy()
        elif r == 1:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.03 * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)
        elif r == 2:
            # Try symmetrized best
            a_cur = _symmetrize(best_seq)
            a_cur = a_cur / a_cur.sum()
        else:
            # Larger perturbation
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z = z + 0.1 * rng.normal(size=len(z))
            z = np.clip(z, -60, 60)
            z = z - z.mean()
            a_cur = _softmax_fn(z)

        per_dl = min(phase2_end - 10, time.monotonic() + per_restart_time)
        p_best, sc_best = _nesterov_lbfgs(a_cur, per_dl)
        update_best(p_best, sc_best, f"lbfgs_r{r}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # Phase 3: Active-set LP refinement (300s)
    phase3_end = min(deadline, t_start + 1150.0)
    print(f"\nPhase 3: Active-set LP refinement (300s)")

    # Use smaller n for LP (too slow at 4096)
    n_lp = min(n, 1024)
    if n > n_lp:
        x0 = np.linspace(0, 1, n)
        x1 = np.linspace(0, 1, n_lp)
        a_lp = np.interp(x1, x0, best_seq)
        a_lp = np.maximum(a_lp, 1e-12)
        a_lp = a_lp / a_lp.sum()
    else:
        a_lp = best_seq.copy()

    sc_lp = _score_norm_fn(a_lp)
    update_best(a_lp, sc_lp, f"lp_init_n{n_lp}")

    p_best_lp, sc_best_lp = _active_set_lp(a_lp, phase3_end, top_k=8, max_rounds=30)
    update_best(p_best_lp, sc_best_lp, f"active_lp_n{n_lp}")

    # Interpolate back to full n
    if n > n_lp:
        x0 = np.linspace(0, 1, n_lp)
        x1 = np.linspace(0, 1, n)
        a_full = np.interp(x1, x0, p_best_lp)
        a_full = np.maximum(a_full, 1e-12)
        a_full = a_full / a_full.sum()
        sc_full = _score_norm_fn(a_full)
        update_best(a_full, sc_full, f"lp_interp_n{n}")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # Phase 4: Subgradient projection (200s)
    phase4_end = min(deadline, t_start + 1350.0)
    print(f"\nPhase 4: Subgradient projection (200s)")

    a_sg = best_seq.copy()
    p_best_sg, sc_best_sg = _subgradient_projection(a_sg, phase4_end, max_iters=30000)
    update_best(p_best_sg, sc_best_sg, "subgradient")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Phase 5: Local search + coordinate descent polish (remaining)
    phase5_end = deadline
    print(f"\nPhase 5: Local search and polish")

    # Local search
    local_dl = min(phase5_end - 120, time.monotonic() + 200)
    p_best_ls, sc_best_ls = _local_search(best_seq, local_dl, rng, max_iter=50000)
    update_best(p_best_ls, sc_best_ls, "local_search")

    # Coordinate descent fine
    coord_dl = min(phase5_end - 60, time.monotonic() + 100)
    p_best_cd, sc_best_cd = _coord_fine(best_seq, coord_dl, max_rounds=15)
    update_best(p_best_cd, sc_best_cd, "coord_fine")

    # Final ultra-fine L-BFGS
    if time.monotonic() < phase5_end - 30:
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())
        beta_ultra = 1e-6 * max_c_final
        final_dl = min(phase5_end - 10, time.monotonic() + 50)
        p_best_final, sc_best_final = _nesterov_lbfgs(best_seq, final_dl,
                                                       beta_list=[beta_ultra, beta_ultra * 0.5, beta_ultra * 0.1])
        update_best(p_best_final, sc_best_final, "ultra_fine")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
