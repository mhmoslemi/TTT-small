"""
Multi-phase optimizer combining FISTA, Adam in log-space, and LP-based peak shaving.
Key innovation: LP constraint generation to find optimal peak-reducing directions,
plus multi-length exploration (4096, 8192) and aggressive multi-restart.
"""
import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n_a, n_b = len(a), len(b)
    n_full = n_a + n_b - 1
    N = _next_pow2(n_full)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:n_full]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _softmax_compute(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    return 4.0 * n * a * (h - W)


def _compute_grad_simplex(a, c, w, n):
    a_rev = a[::-1]
    wc = _full_convolve(w, a_rev)
    h = wc[n - 1:2 * n - 1]
    return 2.0 * h


def _fista_smoothed(a_init, beta, deadline, max_iter):
    n = len(a_init)
    a_curr = a_init.copy()
    a_prev = a_curr.copy()
    t_val = 1.0
    L = 8.0 * n / beta
    step = 1.0 / L
    c_init = _full_convolve(a_curr, a_curr)
    best_score = float(2.0 * n * c_init.max())
    best_a = a_curr.copy()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y = a_curr + ((t_val - 1.0) / t_next) * (a_curr - a_prev)
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-30:
            y = y / s_y
        else:
            y = np.ones(n) / n

        c_y = _full_convolve(y, y)
        c_max = float(c_y.max())
        w = np.exp((c_y - c_max) / beta)
        w_sum = float(w.sum())
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros_like(c_y)
            w[int(np.argmax(c_y))] = 1.0
        else:
            w = w / w_sum

        grad = _compute_grad_simplex(y, c_y, w, n)
        a_next = _proj_simplex(y - step * grad)

        if it % 200 == 0 or it < 5:
            sc = float(2.0 * n * _full_convolve(a_next, a_next).max())
            if sc < best_score:
                best_score = sc
                best_a = a_next.copy()

        a_prev = a_curr.copy()
        a_curr = a_next
        t_val = t_next

    sc_final = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_curr.copy()
    return best_a, best_score


def _adam_logspace(a_init, deadline, beta_schedule, lr_init, max_iter_total):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = float(2.0 * n * _full_convolve(a_init, a_init).max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr_init

    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        it_in_phase = 0
        while it_in_phase < max_iter_total:
            if time.monotonic() > deadline:
                break
            it_in_phase += 1
            t += 1

            a = _softmax_compute(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s = float(e.sum())
            if s < 1e-300 or not np.isfinite(s):
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s

            grad = _compute_grad_z(a, c, w, n)
            if not np.all(np.isfinite(grad)):
                break

            m_buf = b1 * m_buf + (1.0 - b1) * grad
            v_buf = b2 * v_buf + (1.0 - b2) * (grad * grad)
            mh = m_buf / (1.0 - b1 ** t)
            vh = v_buf / (1.0 - b2 ** t)

            update = current_lr * mh / (np.sqrt(vh) + eps)
            update_norm = np.linalg.norm(update)
            if update_norm > 0.3:
                update = update * (0.3 / update_norm)

            z = z - update
            z = np.clip(z, -50.0, 50.0)
            z = z - z.mean()

            if t % 50 == 0:
                a_chk = _softmax_compute(z)
                sc = float(2.0 * n * _full_convolve(a_chk, a_chk).max())
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_chk.copy()

            if it_in_phase % 8000 == 0:
                current_lr = max(0.0005, current_lr * 0.95)

    a_final = _softmax_compute(z)
    sc_final = float(2.0 * n * _full_convolve(a_final, a_final).max())
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _lp_peak_direction(a, M, max_constraints=200, max_lp_iters=15, timeout=30.0):
    """Solve LP to find direction g that maximizes sum(g) subject to conv(a,g) <= M."""
    n = len(a)
    t_start = time.monotonic()

    c_aa = _full_convolve(a, a)
    sort_idx = np.argsort(-c_aa)
    active_set = sort_idx[:max(20, max_constraints // 5)].tolist()

    best_g = np.ones(n) / n
    best_sum = 1.0

    for lp_iter in range(max_lp_iters):
        if time.monotonic() - t_start > timeout:
            break

        # Build constraint matrix for active constraints
        rows = []
        cols = []
        vals = []
        for k in active_set:
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(float(a[j]))

        if len(rows) == 0:
            return best_g

        A_active = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        b_active = np.zeros(2 * n - 1)
        for k in active_set:
            b_active[k] = M

        c_obj = -np.ones(n)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A_active, b_ub=b_active,
                          bounds=bounds, method='highs',
                          options={'time_limit': timeout})
            if res.status == 0 and res.x is not None:
                g_candidate = np.maximum(res.x, 0.0)
                g_sum = float(g_candidate.sum())
                if g_sum > best_sum:
                    best_sum = g_sum
                    best_g = g_candidate.copy()
                else:
                    break

                # Check for violated constraints
                c_ag = _full_convolve(a, g_candidate)
                violations = np.where(c_ag > M * 1.001)[0]
                if len(violations) == 0:
                    break
                # Add violated constraints (up to a limit)
                new_constraints = violations[:10].tolist()
                for k in new_constraints:
                    if k not in active_set:
                        active_set.append(int(k))
            else:
                break
        except Exception:
            break

    return best_g


def _lp_shave_step(a, deadline, max_iterations=30):
    """Iteratively apply LP-based peak shaving."""
    n = len(a)
    a_curr = a.copy()
    a_curr = a_curr / a_curr.sum()
    best_a = a_curr.copy()
    best_score = float(2.0 * n * _full_convolve(a_curr, a_curr).max())
    no_improve_count = 0

    for it in range(max_iterations):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a_curr, a_curr)
        M = float(c.max())

        g = _lp_peak_direction(a_curr, M, max_constraints=300,
                               max_lp_iters=10, timeout=20.0)
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search for blend parameter
        best_t = 0.0
        best_t_score = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_trial = (1.0 - t) * a_curr + t * g
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            sc = float(2.0 * n * _full_convolve(a_trial, a_trial).max())
            if sc < best_t_score:
                best_t_score = sc
                best_t = t

        if best_t > 0:
            a_curr = (1.0 - best_t) * a_curr + best_t * g
            a_curr = np.maximum(a_curr, 0.0)
            a_curr = a_curr / a_curr.sum()
            if best_t_score < best_score:
                best_score = best_t_score
                best_a = a_curr.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1

        if no_improve_count >= 5:
            break

        print(f"    LP iter {it}: score={best_score:.10f}, t={best_t}")

    return best_a, best_score


def _interp_profile(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _perturb(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_compute(z)


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "sinusoidal":
        t = np.linspace(0, 1, n)
        a = 1.0 + 0.1 * np.cos(2.0 * np.pi * t)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    else:
        return np.ones(n) / n
