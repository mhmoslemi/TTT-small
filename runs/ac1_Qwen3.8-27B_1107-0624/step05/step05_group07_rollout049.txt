"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _conv_matrix(f):
    n = len(f)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _lp_solve(f, max_time=30):
    n = len(f)
    g = cp.Variable(n)
    t = cp.Variable()
    A = _conv_matrix(f)
    constraints = [A @ g <= t, cp.sum(g) == 1, g >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            sv = g_val.sum()
            if sv > 1e-15:
                g_val = g_val / sv
                return g_val, float(t.value)
    except Exception:
        pass
    return None, None


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -60, 60)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad

    n = len(a_init)

        w[peak_lags] = 1.0 / len(peak_lags)
        W = float(np.dot(w, c))
        a_rev = a[::-1].copy()
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]

        lr = 0.1 / (1 + 0.1 * iteration)
        z = z - lr * grad / a  # In log-space, this is multiplicative

def _make_diverse_profiles(n, rng):

    # Linear ramps

        a = np.maximum(a, 1e-10)

    # Exponential
    for rate in [1.0, 2.0, -1.0, -2.0, 3.0]:

        profiles.append(a / a.sum())

    # Cosine-based
    for p in [0.5, 1.0, 2.0, 4.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        profiles.append(a / a.sum())

    profiles.append(a / a.sum())
    a = np.exp(-20 * (u) ** 2) + np.exp(-20 * (u + 0.5) ** 2)
    profiles.append(a / a.sum())

    # Random smooth

        a = np.zeros(n)
        for i, c in enumerate(coeffs):
            a += c * np.cos(np.pi * i * t)
        a = np.abs(a) + 0.1
        profiles.append(a / a.sum())

    return profiles


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    n_target = 4096

    if parent is not None:
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
        a_main = _interp_profile(parent, n_target)
    else:
        profiles = _make_diverse_profiles(n_target, rng)
        a_main = profiles[0]  # uniform
        sc = _score_norm_fn(a_main)
        update_best(a_main, sc, "uniform_init")

    update_best(a_main, _score_norm_fn(a_main), f"main_init_n{n_target}")

    # Phase 1: LP warm start (60s)
    print(f"\nPhase 1: LP warm start (n={n_target})")
    phase1_end = min(deadline, t_start + 60.0)
    a_lp, _ = _lp_solve(a_main)
    if a_lp is not None:
        sc_lp = _score_norm_fn(a_lp)
        update_best(a_lp, sc_lp, "lp_single")
        # Blend
        a_blend = 0.5 * a_main + 0.5 * a_lp
        a_blend = np.maximum(a_blend, 0)
        a_blend /= a_blend.sum()
        sc_blend = _score_norm_fn(a_blend)
        update_best(a_blend, sc_blend, "lp_blend")
        a_main = best_seq.copy() if best_seq is not None else a_main

    # Phase 2: Main adaptive L-BFGS (900s)
    print(f"\nPhase 2: Adaptive L-BFGS (n={n_target})")
    phase2_end = min(deadline, t_start + 960.0)
    a_best, sc_best = _adaptive_lbfgs(best_seq, phase2_end, rng)
    update_best(a_best, sc_best, "adaptive_lbfgs")

    # Phase 3: LP polish on best (120s)
    print(f"\nPhase 3: LP polish")
    phase3_end = min(deadline, t_start + 1080.0)
    a_lp2, sc_lp2 = _lp_polish(best_seq, phase3_end, max_rounds=30)
    update_best(a_lp2, sc_lp2, "lp_polish")

    # Phase 4: Peak shaving (120s)
    print(f"\nPhase 4: Peak shaving")
    phase4_end = min(deadline, t_start + 1200.0)
    a_ps, sc_ps = _peak_shave(best_seq, phase4_end, rng)
    update_best(a_ps, sc_ps, "peak_shave")

    # Phase 5: Another L-BFGS round with tighter beta (180s)
    print(f"\nPhase 5: Fine L-BFGS")
    phase5_end = min(deadline, t_start + 1380.0)
    if time.monotonic() < phase5_end:
        a_fine, sc_fine = _adaptive_lbfgs(best_seq, phase5_end, rng)
        update_best(a_fine, sc_fine, "fine_lbfgs")

    # Phase 6: Try n=2048 (often better ratio) (120s)
    print(f"\nPhase 6: Try n=2048")
    phase6_end = min(deadline, t_start + 1500.0)
    if time.monotonic() < phase6_end:
        a_2048 = _interp_profile(best_seq, 2048)
        sc_2048 = _score_norm_fn(a_2048)
        update_best(a_2048, sc_2048, "interp_2048")
        if time.monotonic() < phase6_end - 30:
            a_2048_opt, sc_2048_opt = _adaptive_lbfgs(a_2048, phase6_end, rng)
            update_best(a_2048_opt, sc_2048_opt, "opt_2048")

    # Phase 7: Try n=8192 (120s)
    print(f"\nPhase 7: Try n=8192")
    phase7_end = min(deadline, t_start + 1620.0)
    if time.monotonic() < phase7_end:
        a_8192 = _interp_profile(best_seq, 8192)
        sc_8192 = _score_norm_fn(a_8192)
        update_best(a_8192, sc_8192, "interp_8192")

    # Phase 8: Final multi-strategy polish (remaining time)
    print(f"\nPhase 8: Final polish")
    if time.monotonic() < deadline - 20:
        # Quick LP
        g_final, _ = _lp_solve(best_seq)
        if g_final is not None:
            for mix_w in [0.1, 0.2, 0.3, 0.5]:
                a_mix = (1 - mix_w) * best_seq + mix_w * g_final
                a_mix = np.maximum(a_mix, 0)
                a_mix /= a_mix.sum()
                sc_mix = _score_norm_fn(a_mix)
                update_best(a_mix, sc_mix, f"final_mix_{mix_w}")

        # Very fine L-BFGS
        if time.monotonic() < deadline - 15:
            remaining_dl = deadline - 15
            z_final = np.log(np.maximum(best_seq, 1e-15))
            z_final = z_final - z_final.mean()
            n_F = len(best_seq)
            c_F = _full_convolve(best_seq, best_seq)
            beta_ult = 2e-6 * float(c_F.max())
            try:
                result = minimize(
                    _lse_obj, z_final, args=(beta_ult, n_F),
                    method='L-BFGS-B', jac=_lse_grad,
                    options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-16,
                             'maxls': 40}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_ult = _softmax_fn(result.x)
                    sc_ult = _score_norm_fn(a_ult)
                    update_best(a_ult, sc_ult, "ultra_fine")
            except Exception:
                pass

    # Final validation
    print(f"\n{'='*60}")
    print(f"FINAL: score={best_score:.10f}, n={len(best_seq)}")

    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"Final normalized score: {sc_final:.10f}")

        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirms: {sc_eval:.10f}")
            except Exception as ex:
                print(f"Evaluate failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 4096
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
