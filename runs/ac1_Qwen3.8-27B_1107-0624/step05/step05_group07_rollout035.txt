where `a` is a non-negative sequence of length `n`.

Key observations:
1. The objective is scale-invariant (multiplying all elements by a constant doesn't change the score), so we can normalize `sum(a) = 1` (or any constant) without loss of generality.
2. The convolution `convolve(a, a)` has `2n-1` elements. The maximum is what matters.
3. We want to minimize the peak of the autocorrelation relative to the energy.

This is related to the "minimum peak-to-sidelobe ratio" problem in signal processing, or equivalently, we want a sequence whose autocorrelation is as flat as possible (like a "flat" autocorrelation).

The theoretical lower bound: If the convolution were perfectly flat (all `2n-1` values equal), then `max(conv) = (sum(a))^2 / (2n-1)`, and the score would be `2n * (sum(a))^2/(2n-1) / (sum(a))^2 = 2n/(2n-1) ≈ 1` for large n.

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses mirror descent (subgradient), LP best-response, and coordinate descent.
"""
import numpy as np
import time

def _fft_convolve(a, b):
    """Compute full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())

def _score_fn(a):
    """Full score with sum normalization."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return _score_normalized(a_norm)

def _gradient_at_lag(a, k):
    """
    Compute gradient of c[k] = (conv(a,a))[k] w.r.t. a.
    Result: grad[i] = 2*a[k-i] if max(0,k-n+1) <= i <= min(k,n-1), else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        # grad[i] = 2 * a[k-i] for i in [lo, hi]
        # k-i ranges from k-lo to k-hi
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi+1] = 2.0 * a[idx]
    return grad

def _subgradient_max_conv(a):
    """
    Compute a subgradient of max_k conv(a,a)[k].
    Returns gradient vector and the max value.
    """
    n = len(a)
    c = _fft_convolve(a, a)
    max_val = float(c.max())
    # Find all lags where c[k] == max (or very close)
    active = np.where(c >= max_val - 1e-12)[0]
    # Average the gradients at active lags
    grad = np.zeros(n)
    for k in active:
        grad += _gradient_at_lag(a, k)
    grad /= len(active)
    return grad, max_val, c

def _smoothed_subgradient(a, beta, top_k=5):
    """
    Compute smoothed subgradient using softmax over top-k lags.
    """
    n = len(a)
    c = _fft_convolve(a, a)
    # Get top-k lags
    if top_k >= len(c):
        top_indices = np.argsort(c)[::-1]
    else:
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_indices = top_indices[np.argsort(c[top_indices])[::-1]]
    
    # Softmax weights over top-k
    vals = c[top_indices] / beta
    vals = vals - vals.max()
    w = np.exp(vals)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_indices)):
        k = top_indices[idx]
        grad += w[idx] * _gradient_at_lag(a, k)
    
    max_val = float(c.max())
    return grad, max_val, c

def _mirror_descent(a, deadline, init_lr=0.1, lr_decay=0.999, top_k=5, beta=0.01):
    """
    Mirror descent (exponential weights) for minimizing max(conv(a,a)).
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = _score_normalized(a)
    lr = init_lr
    it = 0
    
    while time.monotonic() < deadline:
        grad, max_val, c = _smoothed_subgradient(a, beta, top_k)
        # Mirror descent step (in log-space): a[i] *= exp(-lr * grad[i])
        # Equivalently: z[i] -= lr * grad[i], a = exp(z) / sum(exp(z))
        # For stability, work with log(a):
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= lr * grad
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15 or not np.isfinite(s):
            break
        a = a_new / s
        
        sc = _score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        it += 1
        lr = max(0.001, lr * lr_decay)
        # Adapt beta
        beta = max(1e-6, 0.005 * max_val)
        
        if it % 500 == 0:

def _lp_best_response(a, timeout=30):
    """
    Solve the LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0.
    Returns the optimal g.
    """
    try:
        import cvxpy as cp
        n = len(a)
        g = cp.Variable(n)
        t = cp.Variable()
        
        # Build convolution: (a*g)[k] = sum_j a[k-j]*g[j] for valid k,j
        # We express this as A @ g <= t where A[k,j] = a[k-j] if valid
        # A is (2n-1) x n
        
        # Instead of building the full matrix, use the structure
        # (a*g)[k] for k=0..2n-2
        # A[k,j] = a[k-j] if 0<=k-j<=n-1, else 0
        
        # For efficiency, use the fact that A is a "convolution matrix"
        # We can express constraints as: for each k, sum_j A[k,j]*g[j] <= t
        # This is the same as: the k-th element of conv(a,g) <= t
        
        # cvxpy doesn't have a direct conv, so we build the matrix
        # For n up to ~2000, this should be OK
        if n > 3000:
            # Downsample for LP
            n_lp = 1024
            a_lp = np.interp(np.linspace(0, 1, n_lp), np.linspace(0, 1, n), a)
            a_lp = a_lp / a_lp.sum()
        else:
            a_lp = a
            n_lp = n
        
        # Build A matrix (sparse)
        from scipy.sparse import csr_matrix
        rows = []
        cols = []
        data = []
        for k in range(2 * n_lp - 1):
            for j in range(n_lp):
                idx = k - j
                if 0 <= idx < n_lp:
                    rows.append(k)
                    cols.append(j)
                    data.append(a_lp[idx])
        
        A = csr_matrix((data, (rows, cols)), shape=(2 * n_lp - 1, n_lp))
        
        constraints = [
            A @ g <= t,
            cp.sum(g) == 1,
            g >= 0
        ]
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        t0 = time.monotonic()
        prob.solve(solver=cp.CLARABEL)
        elapsed = time.monotonic() - t0
        
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            g_val = g_val / g_val.sum()
            
            if n_lp < n:
                # Upsample
                g_full = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_lp), g_val)
                g_full = g_full / g_full.sum()
                return g_full
            return g_val
    except Exception as e:
        print(f"    LP error: {e}")
    return None

def _lp_iterate(a, n_iters=20, deadline=None, step_size=0.3):
    """Iterate: a <- (1-t)*a + t*LP_best_response(a)."""
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        g = _lp_best_response(a)
        if g is None:
            break
        a_new = (1 - step_size) * a + step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_normalized(a_new)

            best_score = sc
            best_a = a_new.copy()
        a = a_new
        if it % 5 == 0:

def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    # Power laws
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = np.maximum(1.0 - p * np.abs(u) * 0.5, 1e-10)
        profiles.append(a / a.sum())
    # Exponential
    for rate in [0.5, 1.0, 2.0, -0.5, -1.0]:
        a = np.exp(rate * u)
        profiles.append(a / a.sum())
    # Cosine
    for p in [0.5, 1.0, 2.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        profiles.append(a / a.sum())
    # Uniform
    profiles.append(np.ones(n) / n)
    # Bimodal
    a = np.exp(-20*(u-0.3)**2) + np.exp(-20*(u+0.3)**2)
    profiles.append(a / a.sum())
    # Triangular
    a = 1 - np.abs(u)
    profiles.append(a / a.sum())
    return profiles

def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time


def _fft_convolve(a, b):
    """Compute full linear convolution using FFT, with next-power-of-2 sizing."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _fft_convolve(a, a)
    return 2.0 * n * float(c.max())


def _score_fn(a):
    """Full score: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    return 2.0 * n * float(_fft_convolve(a, a).max()) / (s * s)


def _grad_lag(a, k):
    """Gradient of (conv(a,a))[k] w.r.t. a. Returns vector of length n."""
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if lo <= hi:
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad


def _subgradient_topk(a, top_k, beta_frac):
    """
    Compute smoothed subgradient using softmax over top-k lags.
    Returns (gradient_vector, max_value, full_convolution).
    """
    n = len(a)
    c = _fft_convolve(a, a)
    max_val = float(c.max())
    
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    
    beta = beta_frac * max_val
    if beta < 1e-30:
        beta = max_val / top_k
    vals = c[top_idx] / beta
    vals -= vals.max()
    w = np.exp(vals)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(top_idx)) / len(top_idx)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = top_idx[idx]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1)
            grad[i_range] += w[idx] * 2.0 * a[k - i_range]
    
    return grad, max_val, c


def _mirror_descent(a, deadline, lr_init=5.0, lr_decay=0.9995, top_k=5, beta_frac=0.01,

    it = 0
    last_improve = 0

        if gnorm > 1e-15:
            grad = grad / gnorm
        else:

        z -= lr * grad
        z = np.clip(z, -100, 100)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a_sum = a.sum()

            if a_sum < 1e-15:

        a = a / a_sum
        
        sc = _score_normalized(a)

            best_a = a.copy()

        it += 1

            last_improve = it
            lr = min(lr * 2, lr_init)
        
        if it % 2000 == 0:

def _momentum_subgradient(a, deadline, lr_init=3.0, momentum=0.9, top_k=5, beta_frac=0.005):
    """Subgradient descent with momentum (Nesterov-like)."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-300))
    z -= z.mean()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    v = np.zeros(n)
    lr = lr_init
    it = 0
    last_improve = 0
    
    while time.monotonic() < deadline:
        a = np.exp(z - z.max())
        a = a / a.sum()
        
        grad, max_val, c = _subgradient_topk(a, top_k, beta_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        # Momentum update
        v = momentum * v - lr * grad
        z += v
        z = np.clip(z, -100, 100)
        z -= z.mean()
        
        a = np.exp(z - z.max())
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum
        
        sc = _score_normalized(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            last_improve = it
        
        it += 1
        lr = max(0.1, lr * 0.9999)
        
        if it - last_improve > 3000:
            z = np.log(np.maximum(best_a, 1e-300))
            z -= z.mean()
            v = np.zeros(n)
            lr = min(lr * 3, lr_init)
            last_improve = it
        
        if it % 2000 == 0:
            print(f"    MS iter {it}: score={sc:.10f}, best={best_score:.10f}")
    
    return best_a, best_score


def _lp_best_response(a, timeout=60):
    """Solve LP: min t s.t. (a*g)[k] <= t, sum(g)=1, g>=0. Returns optimal g."""
    try:
        import cvxpy as cp
        from scipy.sparse import csr_matrix
        
        n = len(a)
        if n > 2000:
            n_lp = 1024
            a_lp = np.interp(np.linspace(0, 1, n_lp), np.linspace(0, 1, n), a)
            a_lp = a_lp / a_lp.sum()
        else:
            a_lp = a.copy()
            n_lp = n
        
        # Build sparse convolution matrix A (size (2n-1) x n)
        rows = []
        cols = []
        data = []
        for k in range(2 * n_lp - 1):
            k_arr = k - np.arange(n_lp)
            valid = (k_arr >= 0) & (k_arr < n_lp)
            if valid.any():
                rows.extend([k] * valid.sum())
                cols.extend(np.where(valid)[0].tolist())
                data.extend(a_lp[k_arr[valid]].tolist())
        
        A = csr_matrix((data, (rows, cols)), shape=(2 * n_lp - 1, n_lp))
        
        g = cp.Variable(n_lp)
        t = cp.Variable()
        constraints = [A @ g <= t, cp.sum(g) == 1, g >= 0]
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        t0 = time.monotonic()
        prob.solve(solver=cp.CLARABEL)
        
        if time.monotonic() - t0 > timeout * 2:
            return None
        if g.value is not None:
            g_val = np.maximum(g.value, 0)
            gs = g_val.sum()
            if gs < 1e-15:
                return None
            g_val = g_val / gs
            if n_lp < n:
                g_full = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, n_lp), g_val)
                g_full = np.maximum(g_full, 0)
                g_full = g_full / g_full.sum()
                return g_full
            return g_val
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def _lp_iterate(a, n_iters=20, deadline=None, step_size=0.3):
    """Iterate: a <- (1-t)*a + t*LP_best_response(a)."""
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        g = _lp_best_response(a)
        if g is None:
            break
        a_new = (1 - step_size) * a + step_size * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        sc = _score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        print(f"    LP iter {it}: score={sc:.10f}, best={best_score:.10f}")
    
    return best_a, best_score


def _coord_descent(a, deadline, max_rounds=10):
    """
    Coordinate descent: for each j, try several values of a[j] and keep the best.
    """
    n = len(a)
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            best_local_sc = best_score
            best_local_val = old_val
            scale = old_val
            for frac in [0.0, 0.3, 0.5, 0.7, 1.3, 1.5]:
                a[j] = frac * scale
                s = a.sum()
                if s < 1e-15:
                    a[j] = old_val
                    continue
                sc = _score_normalized(a / s)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local_val = frac * scale
            a[j] = best_local_val
            a = a / a.sum()
            if best_local_sc < best_score:
                best_score = best_local_sc
                best_a = a.copy()
                improved = True
        if not improved:
            break
        if rnd % 2 == 0:
            print(f"    CD round {rnd}: score={best_score:.10f}")
    
    return best_a, best_score


def _make_profiles(n):
    """Generate diverse initial profiles."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    profiles = []
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        a = np.maximum(1.0 - 0.4 * p * np.abs(u), 1e-10)
        profiles.append(a / a.sum())
    for rate in [0.5, 1.0, 2.0, 3.0, -1.0, -2.0]:
        a = np.exp(rate * u)
        profiles.append(a / a.sum())
    for p in [0.5, 1.0, 2.0, 3.0]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        profiles.append(a / a.sum())
    profiles.append(np.ones(n) / n)
    a = np.exp(-15 * (u - 0.3) ** 2) + np.exp(-15 * (u + 0.3) ** 2)
    profiles.append(a / a.sum())
    a = 1 - np.abs(u)
    profiles.append(a / a.sum())
    a = np.maximum(1 - u ** 2, 1e-10)
    profiles.append(a / a.sum())
    return profiles


def _interp_profile(p0, n_new):
    """Interpolate a profile to a new length."""
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

            sc = _score_normalized(p)
            update_best(p, sc, f"final_interp_n{n_chk}")
    
    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")
        
        if len(final_seq) <= 5000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator score: {sc_eval:.10f}")
            except Exception as ex:
                print(f"evaluate_sequence failed: {ex}")
        
        return [float(x) for x in final_seq]
    
    # Fallback
    n_fb = 2048
    fallback = _make_profiles(n_fb)[3]
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
