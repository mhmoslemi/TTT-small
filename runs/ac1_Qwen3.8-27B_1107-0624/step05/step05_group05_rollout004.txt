"""
Nesterov Accelerated Gradient + LP hybrid optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous: Nesterov momentum (O(1/k^2) rate), backtracking line search,
adaptive beta with plateau detection, linearized LP for direct peak reduction, and
multiple perturbation restarts to escape local minima.

import numpy as np

import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_norm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _compute_grad_z(a, w):
    """Compute gradient of smooth max w.r.t. log-parameters z.
    w: probability weights over convolution coefficients.
    Returns grad in z-space (natural gradient on simplex).
    """
    n = len(a)
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    grad_a = 2.0 * conv_aw[n - 1:2 * n - 1][::-1]
    # Convert to z-space gradient: grad_z = a * (grad_a - <a, grad_a>)
    W = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - W)
    return grad_z


def _smooth_obj_and_grad(a, beta):
    """Compute smooth objective and gradient in z-space."""
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / max(beta, 1e-10)
    e = np.exp(c_shifted)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return c_max, _compute_grad_z(a, w)
    w = e / s
    obj = c_max + beta * np.log(s) / max(beta, 1e-10)
    # Actually obj = logsumexp(beta*c)/beta = c_max + log(s)/beta
    obj = c_max + np.log(s) / max(beta, 1e-10)
    grad_z = _compute_grad_z(a, w)
    return float(obj), grad_z


def _nesterov_loop(a_init, deadline, beta_start, beta_end, max_iter=5000000,
                   lr_init=0.05, momentum=0.9, verbose=True):
    """Nesterov accelerated gradient descent in log-softmax parameterization."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    v = np.zeros(n)
    t_mom = momentum
    t = 0
    prev_best = best_score
    stall_count = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        # Adaptive beta
        frac = min(1.0, t / max_iter)
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** frac

        # Nesterov lookahead
        z_look = z + t_mom * v
        a_look = _softmax(z_look)

        # Compute gradient at lookahead point
        _, grad_z = _smooth_obj_and_grad(a_look, beta)

        if not np.all(np.isfinite(grad_z)):
            break

        # Clip gradient
        g_norm = np.linalg.norm(grad_z)
        if g_norm > 1.0:
            grad_z = grad_z / g_norm

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.1 * frac)

        # Backtracking line search (simplified)
        step = -lr * (t_mom * v + grad_z)
        step_norm = np.linalg.norm(step)
        if step_norm > 1.0:
            step = step * (1.0 / step_norm)

        z_new = z + step
        v = step

        # Update
        z = z_new
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()

        # Periodic score check
        if t % 500 == 0:
            a_curr = _softmax(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                stall_count = 0
            else:
                stall_count += 1

            if stall_count > 20:
                # Reduce momentum and lr
                t_mom = max(0.5, t_mom * 0.95)
                lr_init = max(0.005, lr_init * 0.9)
                stall_count = 0

            if verbose and t % 5000 == 0:
                if time.monotonic() < deadline:
                    print(f"    Nest[{t}] sc={best_score:.10f} beta={beta:.2e} lr={lr:.4f} mom={t_mom:.3f}")

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _nesterov_fista(a_init, deadline, beta_start, beta_end, max_iter=5000000,
                    lr_init=0.1, verbose=True):
    """FISTA (Nesterov) with proper t_k schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-300))
    z = z - z.mean()
    y = z.copy()
    t_k = 1.0
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    t = 0

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1

        frac = min(1.0, t / max_iter)
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** frac

        a_y = _softmax(y)
        _, grad_y = _smooth_obj_and_grad(a_y, beta)

        if not np.all(np.isfinite(grad_y)):
            break

        g_norm = np.linalg.norm(grad_y)
        if g_norm > 1.0:
            grad_y = grad_y / g_norm
        elif g_norm < 1e-10:
            pass

        lr = lr_init / (1.0 + 0.5 * frac)
        z_new = y - lr * grad_y
        z_new = np.clip(z_new, -60.0, 60.0)
        z_new = z_new - z_new.mean()

        # Nesterov t update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        theta = (t_k - 1.0) / t_new
        y = z_new + theta * (z_new - z)
        y = np.clip(y, -60.0, 60.0)
        y = y - y.mean()
        t_k = t_new

        z = z_new

        if t % 500 == 0:
            a_curr = _softmax(z)
            sc = _score_norm(a_curr)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_curr.copy()

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score


def _build_lp_indices(n):
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    return np.concatenate(rows_list), np.concatenate(cols_list)


def _lp_step(a, rows, cols):
    """LP step: maximize sum(b) s.t. a*b <= M, b >= 0. Then move toward b."""
    n = len(a)
    c = _full_convolve(a, a)
    M = float(c.max())
    vals = a[rows - cols]
    A = sp.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.status != 0:
            return a, M
        b = np.maximum(res.x, 0.0)
        if b.sum() < 1e-15:
            return a, M
        g = b / b.sum()
    except Exception:
        return a, M

    best_t = 0.0
    best_sc = _score_norm(a)
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        a_cand = (1.0 - t) * a + t * g
        sc = _score_norm(a_cand)
        if sc < best_sc:
            best_sc = sc
            best_t = t
    if best_t == 0.0:
        for t in [0.02, 0.05, 0.08]:
            a_cand = (1.0 - t) * a + t * g
            sc = _score_norm(a_cand)
            if sc < best_sc:
                best_sc = sc
                best_t = t
    if best_t > 0.0:
        a = (1.0 - best_t) * a + best_t * g
    return a, M

def _coordinate_descent(a, deadline, max_rounds=100):
    """Simple coordinate descent for fine polishing."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            # Try reducing a[i] and redistributing
            if a[i] < 1e-10:
                continue
            # Try moving a small amount from i to neighbors
            delta = min(a[i] * 0.1, 1e-4)
            for j in range(max(0, i - 3), min(n, i + 4)):
                if j == i:
                    continue
                a_try = a.copy()
                a_try[i] -= delta * 0.5
                a_try[j] += delta * 0.5

                sc = _score_norm(a_try)
                if sc < best_sc - 1e-12:
                    a = a_try
                    best_sc = sc
                    best_a = a.copy()
                    improved = True
                    break
    return best_a, best_sc

    # ===== Phase 6: Perturbation restarts (1450-1650s) =====
    phase6_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase6_end:
        for attempt in range(5):
            if time.monotonic() > phase6_end - 30:
                break
            sigma = 0.02 + 0.015 * attempt
            p_start = _perturb(best_seq, sigma, rng)
            # Quick FISTA from perturbed start
            fista_dl = min(time.monotonic() + 30, phase6_end - 10)
            c_p = _full_convolve(p_start, p_start)
            max_cp = float(c_p.max())
            p_start, sc_p = _nesterov_fista(
                p_start, fista_dl,
                beta_start=0.0003 * max_cp,
                beta_end=0.00001 * max_cp,
                max_iter=500000, lr_init=0.03
            )
            full_sc = _score(p_start)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_start.copy()

    # ===== Phase 7: Final ultra-fine FISTA (1650-deadline) =====
    if time.monotonic() < deadline - 10:
        a = best_seq.copy()
        c_a = _full_convolve(a, a)
        max_ca = float(c_a.max())
        a, sc = _nesterov_fista(
            a, deadline - 5,
            beta_start=0.0001 * max_ca,
            beta_end=0.000005 * max_ca,
            max_iter=2000000, lr_init=0.01
        )
        full_sc = _score(a)
        print(f"Phase 7 final FISTA: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a.copy()

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
