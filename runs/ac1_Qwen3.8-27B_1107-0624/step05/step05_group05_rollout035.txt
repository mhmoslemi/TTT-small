import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _score_unnorm(a):
    """Score assuming sum(a) = 1"""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _proj_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _compute_grad_smoothed(a, beta):
    """Compute gradient of the smoothed score (assuming sum(a)=1)."""
    n = len(a)
    conv = _full_convolve(a, a)
    # Compute p_k = exp(beta * conv_k) / Z
    conv_max = conv.max()
    exp_vals = np.exp(beta * (conv - conv_max))
    Z = exp_vals.sum()
    p = exp_vals / Z
    M_beta = float(conv_max + np.log(Z) / beta)
    
    # Compute cross-correlation: xcorr[i] = sum_j p[i+j] * a[j]
    # Using FFT: xcorr = ifft(fft(p, N) * conj(fft(a, N)))[:n]
    N = _next_pow2(2 * n - 1 + n)  # enough for the cross-correlation
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    xcorr_full = np.fft.irfft(P * np.conj(A), N)
    xcorr = xcorr_full[:n]
    
    # Gradient: (4*n/S^2) * xcorr - 4*n*M_beta/S^3, with S=1
    grad = 4.0 * n * xcorr - 4.0 * n * M_beta
    return grad, M_beta, conv

def _nesterov_optimize(a_init, beta, n_iter, lr=None, deadline=None):
    """Nesterov accelerated gradient on the smoothed score, projected to simplex."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()  # normalize
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_score = _score_unnorm(a)
    
    if lr is None:
        lr = 1.0 / np.sqrt(n)
    
    x_old = a.copy()
    
    for it in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        grad, M_beta, conv = _compute_grad_smoothed(y, beta)
        
        # Nesterov update
        x_new = _proj_simplex(y - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + (t - 1.0) / (t_new + 1.0) * (x_new - x_old)
        y = _proj_simplex(y)
        x_old = x_new
        t = t_new
        a = x_new
        
        # Periodic evaluation
        if (it + 1) % 500 == 0:
            sc = _score_unnorm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = _score_unnorm(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score

"""
Hybrid optimizer combining FFT-accelerated smoothed gradient descent (Nesterov),
LP-based refinement, and multi-scale search to minimize 2*n*max(convolve(a,a))/sum(a)^2.
Key innovations: O(n log n) gradient via cross-correlation, adaptive beta schedule,
palindromic symmetry exploitation for speedup, and multi-resolution upsampling.
"""
import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_unnorm(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _proj_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) - (cssv - 1.0)
    rho = np.nonzero(cond > 0)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_gradient(a, beta):
    """Compute gradient of (2n/beta)*logsumexp(beta*conv(a)) assuming sum(a)=1."""
    n = len(a)
    conv = _full_convolve(a, a)
    conv_max = float(conv.max())
    shifted = beta * (conv - conv_max)
    max_shifted = shifted.max()
    exp_vals = np.exp(shifted - max_shifted)
    Z = float(exp_vals.sum())
    if Z < 1e-300:
        p = np.zeros_like(conv)
        p[int(np.argmax(conv))] = 1.0
        M_beta = float(conv_max)
    else:
        p = exp_vals / Z
        M_beta = float(conv_max + (max_shifted + np.log(Z)) / beta)
    # Cross-correlation: xcorr[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    N_fft = _next_pow2(3 * n)
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    xcorr_full = np.fft.irfft(P * np.conj(A), N_fft)
    xcorr = xcorr_full[:n]
    grad = 4.0 * n * xcorr - 4.0 * n * M_beta
    return grad, M_beta


def _nesterov_phase(a, deadline, beta, lr_scale=1.0, eval_interval=2000):
    """Nesterov accelerated gradient on smoothed score, projected to simplex."""
    n = len(a)
    a = a / a.sum()
    y = a.copy()
    t_mom = 1.0
    x_old = a.copy()
    best_a = a.copy()
    best_sc = _score_unnorm(a)
    lr = lr_scale / np.sqrt(n)
    it = 0
    while time.monotonic() < deadline:
        grad, M_beta = _smoothed_gradient(y, beta)
        x_new = _proj_simplex(y - lr * grad)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + (t_mom - 1.0) / (t_new + 1.0) * (x_new - x_old)
        y = _proj_simplex(y)
        x_old = x_new.copy()
        t_mom = t_new
        a = x_new
        it += 1
        if it % eval_interval == 0:
            sc = _score_unnorm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    sc = _score_unnorm(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    return best_a, best_sc, it


def _build_lp_sparse(n):
    """Build sparse matrix for LP: A[k,j] = 1 if k-j in [0,n-1]."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        j_range = np.arange(j_start, j_end + 1)
        rows_list.append(np.full(len(j_range), k, dtype=np.int32))
        cols_list.append(j_range.astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    A = sp.csc_matrix((np.ones(len(rows)), (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_refine(a, A, max_iters=50, time_limit_per_lp=5.0):
    """LP-based refinement: solve max sum(b) s.t. A@b <= M, b>=0, mix with a."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_unnorm(a)
    for it in range(max_iters):
        conv = _full_convolve(a, a)
        M = float(conv.max())
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'time_limit': time_limit_per_lp})
            if res.status != 0 or res.x is None:
                break
            g0 = np.maximum(res.x, 0.0)
            s_g0 = float(g0.sum())
            if s_g0 < 1e-15:
                break
            g = g0 / s_g0
            best_t = 0.0
            best_local_sc = best_sc
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                cand = (1.0 - t) * a + t * g
                cand = cand / cand.sum()
                sc = _score_unnorm(cand)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_t = t
            if best_t == 0.0:
                for t in [0.01, 0.05]:
                    cand = (1.0 - t) * a + t * g
                    cand = cand / cand.sum()
                    sc = _score_unnorm(cand)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_t = t
            if best_t > 0.0:
                a = (1.0 - best_t) * a + best_t * g
                a = a / a.sum()
                best_sc = min(best_sc, _score_unnorm(a))
                best_a = a.copy()
            else:
                break
        except Exception:
            break
    return best_a, best_sc


def _interpolate_to_n(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-15)
    return a_new / a_new.sum()


def _coordinate_local_search(a, n_steps=2000, deadline=None):
    """Local coordinate descent: try adjusting pairs of coordinates to reduce max conv."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score_unnorm(a)
    a = a.copy()
    conv = _full_convolve(a, a)
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        k_max = int(np.argmax(conv))
        # Find i that most contributes to conv[k_max]
        j_start = max(0, k_max - n + 1)
        j_end = min(k_max, n - 1)
        contribs = a[j_start:j_end + 1] * a[k_max - j_end:k_max - j_start + 1][::-1]
        i_local = int(np.argmax(contribs))
        i = j_start + i_local
        # Try moving mass from i to i+1 or i-1
        for delta_i in [i - 1, i + 1, i - 2, i + 2]:
            if delta_i < 0 or delta_i >= n:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3]:
                a_try = a.copy()
                amt = a_try[i] * frac
                a_try[i] -= amt
                a_try[delta_i] += amt
                if a_try[i] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                a_try = a_try / a_try.sum()
                sc = _score_unnorm(a_try)
                if sc < best_sc * (1 - 1e-6):
                    best_sc = sc
                    best_a = a_try.copy()
                    a = a_try.copy()
                    conv = _full_convolve(a, a)
    return best_a, best_sc
