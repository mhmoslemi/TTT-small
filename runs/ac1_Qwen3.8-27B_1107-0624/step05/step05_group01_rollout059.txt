import numpy as np
from scipy.optimize import minimize, linprog
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j-1] * (2.0*j - 1.0) / (2.0*j)
    a = b[::-1].copy()
    return a / a.sum()

def _downsample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    p = np.interp(x_new, x_old, a)
    p = np.maximum(p, 1e-15)
    return p / p.sum()

def _multiplicative_peak_shave(a_init, deadline, n_iters=5000, K_peaks=5, eta0=0.1):
    """
    Novel multiplicative peak shaving:
    At each step, identify top K peaks in autocorrelation.
    For each element a[i], compute its "responsibility" for each peak.
    Multiply a[i] by exp(-eta * total_responsibility).
    Normalize.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    
    eta = eta0
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        # Get top K peaks
        sorted_idx = np.argsort(c)[::-1][:K_peaks]
        
        # Compute responsibility for each element
        resp = np.zeros(n)
        for k in sorted_idx:
            for i in range(n):

                # d c[k] / d a[i] = a[k-i] (if k-i is valid) + a[i-k] (if i-k is valid, i != k-i)

                #                    a[k-i] (if 2i == k, i.e., a[i]^2 term)
                val = 0.0
                if 0 <= k - i < n:
                    val += a[k - i]
                if 0 <= i - k < n and i != k - i:
                    val += a[i - k]
                resp[i] += val / max(cm, 1e-30)
        
        # Multiplicative update
        a = a * np.exp(-eta * resp)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.02, 1.0)  # slightly increase

            eta *= 0.95

                a *= np.exp(0.1 * np.random.randn(n))
                a = np.maximum(a, 1e-30)
                a = a / a.sum()
                no_improve = 0
                eta = eta0 * 0.5
        
        if it % 100 == 0:
            print(f"    MPS iter {it}: score={_score_norm_fn(a):.10f}, best={best_score:.10f}, eta={eta:.4f}")
    
    return best_a, best_score

"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Novel approaches: coarse-to-fine shape search, multiplicative peak shaving,
adaptive subgradient with soft peak selection, and frequency-domain inspired
amplitude optimization with time-domain projection.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype):

def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _peak_gradient(c, a, k, n):
    """Compute d(conv[k])/da for a single peak index k."""
    g = np.zeros(n)
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        seg = a[k - lo1: k - lo1 + (hi1 - lo1 + 1)]
        g[lo1:hi1 + 1] += seg[::-1]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += a[lo2 - k: hi2 - k + 1]
    return g


def _multiplicative_peak_shave(a_init, deadline, K_peaks=5, eta0=0.05, max_iter=20000):
    """
    Multiplicative peak shaving: iteratively reduce elements that contribute
    most to the top-K autocorrelation peaks.
    """
    n = len(a_init)
    a = a_init.copy() / a_init.sum()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    eta = eta0
    no_improve = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        cm = float(c.max())

        # Soft top-K peak selection
        sorted_c = np.sort(c)[::-1]
        # Use top K peaks with weights based on their values
        top_K = np.argsort(c)[::-1][:K_peaks]
        peak_vals = c[top_K]
        weights = peak_vals / max(peak_vals.sum(), 1e-30)

        # Compute total responsibility
        resp = np.zeros(n)
        for idx, k in enumerate(top_K):
            g = _peak_gradient(c, a, k, n)
            resp += weights[idx] * g / max(cm, 1e-30)

        # Multiplicative update
        a = a * np.exp(-eta * resp)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        # Check improvement
        sc = _score_norm_fn(a)
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            if it % 20 == 0:
                eta = min(eta * 1.01, 0.5)
        else:
            no_improve += 1
            eta *= 0.97
            if no_improve > 100:
                a = best_a * np.exp(0.05 * rng.standard_normal(n))
                a = np.maximum(a, 1e-30)
                a = a / a.sum()
                no_improve = 0
                eta = eta0 * 0.5

        if it % 500 == 0:
            print(f"    MPS it={it}: cur={sc:.10f} best={best_score:.10f} eta={eta:.5f}")

    return best_a, best_score


def _subgradient_peak_balancing(a_init, deadline, max_iter=50000):
    """
    Subgradient descent on max(conv) with soft peak selection and
    Polyak-style adaptive step size.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    best_lse = _score_norm_fn(a_init)

    # Estimate gap
    uniform_score = 2.0 * n  # for uniform seq, norm score = 2n * (1/n) = 2
    # Actually for uniform: max conv = 1/n (with sum=1), norm score = 2n*(1/n) = 2
    gap = best_score - 2.0
    best_gap = gold = 1e-6

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        s = float(a.sum())
        cur_score = float(2.0 * n * cm / (s * s))

        # Soft top-K with temperature
        temp = 0.005 * cm + 1e-10
        w = np.exp((c - cm) / temp)
        w = w / w.sum()

        # Gradient of the soft-max surrogate w.r.t. a
        # d/d a of sum_k w_k * c[k]
        grad_a = np.zeros(n)
        # Only use peaks with significant weight
        active = np.where(w > 1e-6 / n)[0]
        if len(active) > 200:
            active = np.argsort(w)[::-1][:200]
        for k in active:
            if w[k] > 1e-8:
                g = _peak_gradient(c, a, k, n)
                grad_a += w[k] * g

        # Map to z-space
        Wa = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - Wa)

        # Adaptive step size
        step = gold / (np.linalg.norm(grad_z) + 1e-10)
        step = min(step, 1.0)
        z = z - step * grad_z
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        # Check improvement
        a_new = _softmax_fn(z)
        sc_new = _score_norm_fn(a_new)

            best_score = sc_new

            new_gap = sc_new - 2.0

            gold = max(gold * 0.99, 1e-4)
        else:
            gold *= 0.995

        if it % 2000 == 0:
            print(f"    SubGrad it={it}: score={cur_score:.10f} best={best_score:.10f}")

def _lp_phase(a_init, deadline, K=30, max_rounds=20, slack=1.001):
    """LP-based peak balancing with warm restarts and multiple K values."""
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _full_convolve(best_a, best_a)
        M = float(c.max())
        S = float(best_a.sum())

        # Adaptive peak selection: top K + all peaks above threshold
        sorted_idx = np.argsort(c)[::-1][:K]
        threshold = 0.90 * M
        additional = np.where(c >= threshold)[0]
        active = np.unique(np.concatenate([sorted_idx, additional]))
        if len(active) > 300:
            active = active[:300]

        # Build constraint matrix
        K_act = len(active)
        A_mat = np.zeros((K_act, n))
        for row, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A_mat[row, lo:hi + 1] = best_a[k - lo: k - lo + (hi - lo + 1)]

        b_vec = np.full(K_act, M * slack)

        try:
            res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0, None)] * n, method='highs',
                          options={'time_limit': 30, 'dual_feasibility_tolerance': 1e-8})
            if res.status != 0:
                break
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b < 1e-15:
                break
            b_scaled = b_new * (S / sum_b)
        except Exception:
            break

        # Try blending with multiple t values
        improved = False
        current_score = float(2.0 * n * M / (S * S))
        for t in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            a_trial = (1.0 - t) * best_a + t * b_scaled
            a_trial = np.maximum(a_trial, 0.0)
            s_trial = a_trial.sum()
            if s_trial < 1e-15:
                continue
            c_trial = _full_convolve(a_trial, a_trial)
            M_trial = float(c_trial.max())
            sc_trial = float(2.0 * n * M_trial / (s_trial * s_trial))
            if sc_trial < current_score - 1e-12:
                best_a = a_trial.copy()
                best_score = _score_norm_fn(a_trial)
                current_score = sc_trial
                improved = True
                break

        if not improved:
            # Try with larger slack
            slack *= 1.01
            if slack > 1.5:
                break

    return best_a, best_score


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005,
                 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002, 0.0001,
                 0.00005, 0.00003, 0.00001, 0.000005, 0.000001]
        beta_list = [b * m0 for b in fracs]

    for beta in beta_list:
        if time.monotonic() > deadline:
            break

        def lbfgs_obj(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return 0.0
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
            return float(2.0 * n * lse)

        def lbfgs_grad(z_in):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            cm = float(c_in.max())
            if cm < 1e-300:
                return np.zeros_like(z_in)
            shifted = (c_in - cm) / beta
            max_s = float(shifted.max())
            e = np.exp(shifted - max_s)
            w = e / e.sum()
            a_rev = a_in[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1: 2 * n - 1]
            W = float(np.dot(w, c_in))
            grad = 4.0 * n * a_in * (h - W)
            if not np.all(np.isfinite(grad)):
                return np.zeros_like(z_in)
            return grad

        try:
            result = minimize(
                lbfgs_obj, z, method='L-BFGS-B', jac=lbfgs_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 100, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass

    return best_a, best_score


def _coarse_to_fine(parent, deadline):

    Coarse-to-fine: optimize at progressively larger n values.

    # Try several coarse resolutions
    coarse_ns = [64, 128, 256, 512, 1024]
    # Filter to those smaller than parent

    if not coarse_ns:
        return best_a, _score_norm_fn(best_a)

    current = parent
    for cn in coarse_ns:
        if time.monotonic() > deadline:

        # Downsample
        a_coarse = _interp_profile(current, cn)
        sc_coarse = _score_norm_fn(a_coarse)

def _adam_polish(a_init, deadline, lr=0.02, max_iter=200000):
    """Adam optimizer with adaptive beta log-sum-exp surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    check_interval = 30

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        beta = max(1e-12 * cm, cm0 * 0.01 * (0.97 ** (t // 30)))

        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve(w, a_rev)
        h = w_conv[n - 1: 2 * n - 1]
        grad = 4.0 * n * a * (h - W)

        if not np.all(np.isfinite(grad)):
            break

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -80, 80)
        z = z - z.mean()

        if t % check_interval == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.05 * np.random.randn(n)
                z = z - z.mean()
                no_improve = 0
                if no_improve > 200:
                    break

    return best_a, best_score

        if p is not None and np.all(np.isfinite(p)) and np.isfinite(sc) and sc < best_score:
            best_seq = np.maximum(p, 0.0).copy()
            best_score = sc
            print(f"  NEW BEST: {sc:.10f} (n={len(p)}) [{label}] t={time.monotonic()-t_start:.0f}s")

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n}, norm_score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n = 8192
        parent = _greedy_init(n)
        update_best(parent, _score_norm_fn(parent), "greedy_init")

    print(f"Starting: n={n}, best={best_score:.10f}, deadline in {deadline - t_start:.0f}s")

    # === Phase 1: Coarse-to-fine (200s) ===
    print("Phase 1: Coarse-to-fine shape search")
    p1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < p1_end:
        a_c2f, sc_c2f = _coarse_to_fine(best_seq, p1_end)
        update_best(a_c2f, sc_c2f, "coarse_to_fine")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Multiplicative peak shaving (250s) ===
    print("Phase 2: Multiplicative peak shaving")
    p2_end = min(deadline, t_start + 450.0)
    if time.monotonic() < p2_end and best_seq is not None:
        for K in [3, 5, 8]:
            if time.monotonic() > p2_end - 30:
                break
            a_mps, sc_mps = _multiplicative_peak_shave(
                best_seq, min(p2_end - 10, time.monotonic() + 60),
                K_peaks=K, eta0=0.03, max_iter=10000)
            update_best(a_mps, sc_mps, f"mps_K{K}")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Subgradient peak balancing (200s) ===
    print("Phase 3: Subgradient peak balancing")
    p3_end = min(deadline, t_start + 650.0)
    if time.monotonic() < p3_end and best_seq is not None:
        a_sg, sc_sg = _subgradient_peak_balancing(
            best_seq, min(p3_end - 10, time.monotonic() + 80), max_iter=30000)
        update_best(a_sg, sc_sg, "subgradient")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: LP peak balancing (300s) ===
    print("Phase 4: LP peak balancing")
    p4_end = min(deadline, t_start + 950.0)
    if time.monotonic() < p4_end and best_seq is not None:
        for K_lp in [20, 40, 60]:
            if time.monotonic() > p4_end - 30:
                break
            a_lp, sc_lp = _lp_phase(
                best_seq, min(p4_end - 10, time.monotonic() + 60),
                K=K_lp, max_rounds=10, slack=1.001)
            update_best(a_lp, sc_lp, f"lp_K{K_lp}")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # === Phase 5: L-BFGS with continuation (250s) ===
    print("Phase 5: L-BFGS continuation")
    p5_end = min(deadline, t_start + 1200.0)
    if time.monotonic() < p5_end and best_seq is not None:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        beta_list = [m_b * b for b in [0.01, 0.005, 0.002, 0.001, 0.0005,
                                         0.0002, 0.0001, 0.00005, 0.00001]]
        a_lb, sc_lb = _run_lbfgs(best_seq, p5_end, beta_list=beta_list,
                                  maxiter_per_beta=6000)
        update_best(a_lb, sc_lb, "lbfgs_continuation")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Adam polish (200s) ===
    print("Phase 6: Adam polish")
    p6_end = min(deadline, t_start + 1400.0)
    if time.monotonic() < p6_end and best_seq is not None:
        a_ad, sc_ad = _adam_polish(best_seq, min(p6_end - 10, time.monotonic() + 80),
                                   lr=0.015, max_iter=100000)
        update_best(a_ad, sc_ad, "adam_015")

        if time.monotonic() < p6_end - 30:
            a_ad2, sc_ad2 = _adam_polish(best_seq, min(p6_end - 5, time.monotonic() + 50),
                                         lr=0.005, max_iter=80000)
            update_best(a_ad2, sc_ad2, "adam_005")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Try multiple n (150s) ===
    print("Phase 7: Try multiple n")
    p7_end = min(deadline, t_start + 1550.0)
    if time.monotonic() < p7_end and best_seq is not None:
        best_seq, best_score = _try_multiple_n(best_seq, best_score, p7_end)

    print(f"  Phase 7 done: best={best_score:.10f}")

    # === Phase 8: Final L-BFGS tiny beta (remaining) ===
    print("Phase 8: Final polish")
    if time.monotonic() < deadline - 20 and best_seq is not None:
        c_f = _full_convolve(best_seq, best_seq)
        m_f = float(c_f.max())
        tiny_betas = [m_f * b for b in [0.0001, 0.00005, 0.00001, 0.000005, 0.000001]]
        a_fin, sc_fin = _run_lbfgs(best_seq, deadline - 10,
                                    beta_list=tiny_betas, maxiter_per_beta=10000)
        update_best(a_fin, sc_fin, "final_tiny")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(256)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 10000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
