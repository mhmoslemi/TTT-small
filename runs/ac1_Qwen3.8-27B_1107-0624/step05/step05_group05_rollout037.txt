Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS-B with smooth max (logsumexp) annealing, LP refinement, and multi-restart.
"""
import numpy as np
import time
from scipy.optimize import linprog, minimize
import scipy.sparse as sp
from scipy.signal import correlate


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _smooth_obj_grad(z, n, T):
    """Compute smooth objective and gradient for L-BFGS."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = float(c.max())
    c_shifted = (c - c_max) / T
    e = np.exp(c_shifted)
    S = float(e.sum())
    if S < 1e-300:
        S = 1.0
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / S
    score = 2.0 * n * (c_max + T * np.log(S))
    
    # Gradient wrt a: grad_a[j] = 4n * sum_k w[k] * a[k-j]
    corr = correlate(w, a, mode='valid')
    # This gives sum_{m=0}^{n-1} w[j+m] * a[m] for j=0,...,n-1
    # Which is sum_k w[k] * a[k-j] for k=j,...,j+n-1
    # = sum_k w[k] * a[k-j] (since w is defined on 0..2n-2 and a on 0..n-1)
    # This is correct!
    
    mean_wc = float(np.dot(w, c))
    grad_a = 4.0 * n * corr
    # Softmax gradient: grad_z[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    # sum_i a[i]*grad_a[i] = 4n * sum_i a[i] * corr[i]
    # = 4n * sum_i a[i] * sum_k w[k] * a[k-i]
    # = 4n * sum_k w[k] * sum_i a[i] * a[k-i]
    # = 4n * sum_k w[k] * c[k]
    # = 4n * mean_wc
    grad_z = a * (grad_a - 4.0 * n * mean_wc)
    
    return score, grad_z

    best_sc = _score(a_init)

        if time.monotonic() > deadline:
            break
            
        def obj_grad(z):
            return _smooth_obj_grad(z, n, T)

                             options={'maxiter': max_iter_per_T, 'maxls': 50,

            z = result.x

        if sc < best_sc:

            best_a = a.copy()
    
    return best_a, best_sc

def _adam_smooth(a_init, deadline, beta_start, beta_end, max_iter=500000, lr=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    current_lr = lr
    prev_best = best_score

    while t < max_iter:
        if time.monotonic() > deadline:
            break
        t += 1
        a = _softmax(z)
        c = _full_convolve(a, a)
        c_max = float(c.max())
        beta = beta_start * (max(beta_end, 1e-12) / beta_start) ** (t / max_iter)
        c_shifted = (c - c_max) / max(beta, 1e-10)
        e = np.exp(c_shifted)
        s = float(e.sum())
        if s < 1e-300 or not np.isfinite(s):
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        corr = correlate(w, a, mode='valid')
        mean_wc = float(np.dot(w, c))
        grad_a = 4.0 * n * corr
        grad_z = a * (grad_a - 4.0 * n * mean_wc)
        if not np.all(np.isfinite(grad_z)):
            break
        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        mh = m_buf / (1.0 - b1 ** t)
        vh = v_buf / (1.0 - b2 ** t)
        update = current_lr * mh / (np.sqrt(vh) + eps)
        update_norm = np.linalg.norm(update)
        if update_norm > 0.3:
            update = update * (0.3 / update_norm)
        z = z - update
        z = np.clip(z, -60.0, 60.0)
        z = z - z.mean()
        if t % 500 == 0:
            sc = _score_norm(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
        if t % 5000 == 0:
            if best_score < prev_best * 0.999:
                current_lr = min(0.03, current_lr * 1.05)
            else:
                current_lr = max(0.001, current_lr * 0.97)
            prev_best = best_score

    a_final = _softmax(z)
    sc_final = _score_norm(a_final)
    if np.isfinite(sc_final) and sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()
    return best_a, best_score

            print(f"  After LP n={n2}: score={sc_l:.10f}")
            if sc_l < best_score:
                best_score = sc_l
                best_seq = a_large.copy()

    print(f"Phase 4 done: best={best_score:.10f}, t={time.monotonic()-t_start:.0f}s")

    # === Phase 5: Multi-restart with perturbations (1400-1650s) ===
    phase5_end = min(deadline, t_start + 1650.0)
    if time.monotonic() < phase5_end:
        print(f"\n=== Phase 5: Multi-restart (t={time.monotonic()-t_start:.0f}s) ===")
        n_best = len(best_seq)
        rows_b, cols_b = _build_lp_sparse(n_best)
        for attempt in range(4):
            if time.monotonic() > phase5_end - 30:
                break
            sigma = 0.02 + 0.015 * attempt
            p_start = _perturb(best_seq, sigma, rng)
            # Quick LP
            prev_sc = _score_norm(p_start)
            stall = 0
            for it in range(15):
                if time.monotonic() > phase5_end - 20:
                    break
                p_start, _ = _lp_step(p_start, rows_b, cols_b, timeout=15)
                sc = _score_norm(p_start)
                if sc < prev_sc - 1e-11:
                    stall = 0
                else:
                    stall += 1
                if stall > 8:
                    break
                prev_sc = sc
            full_sc = _score(p_start)
            print(f"  Restart[{attempt}] sigma={sigma:.3f}: score={full_sc:.10f} t={time.monotonic()-t_start:.0f}s")
            if np.isfinite(full_sc) and full_sc < best_score:
                best_score = full_sc
                best_seq = p_start.copy()

    # === Phase 6: Final Adam polish (1650-deadline) ===
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 6: Final Adam polish (t={time.monotonic()-t_start:.0f}s) ===")
        c_fin = _full_convolve(best_seq, best_seq)
        max_cf = float(c_fin.max())
        deadline6 = deadline - 10
        a_fin, sc_fin = _adam_smooth(
            best_seq, deadline6,
            beta_start=0.0002 * max_cf, beta_end=0.00001 * max_cf,
            max_iter=1000000, lr=0.003
        )
        full_sc = _score(a_fin)
        print(f"  Adam polish: score={full_sc:.10f}, t={time.monotonic()-t_start:.0f}s")
        if np.isfinite(full_sc) and full_sc < best_score:
            best_score = full_sc
            best_seq = a_fin.copy()

    # Final validation and return
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256)
            s = 1.0
        # Rescale to have sum = sqrt(2n) for good conditioning (scale-invariant anyway)
        n_final = len(final)
        target_sum = np.sqrt(2.0 * n_final)
        final = final * (target_sum / s)
        sc_final = _score(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={n_final}, time={time.monotonic()-t_start:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256)
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
