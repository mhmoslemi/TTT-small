def _fft_conv(a, b):
    """Compute full linear convolution using FFT."""
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = 1
    while N < out_n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _grad_smooth_max(a, beta):
    """Compute gradient of 2n * (1/beta) * log(sum exp(beta * conv(a,a))).
    Returns (gradient, max_value, weights)."""
    n = len(a)
    c = _fft_conv(a, a)
    # Softmax weights
    c_max = c.max()
    e = np.exp(beta * (c - c_max))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    # Gradient: 4n * conv(w, a_rev)[n-1 : 2n-1]
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    grad = 4.0 * n * wc[n-1:2*n-1]
    smooth_max = (1.0/beta) * (np.log(s) + beta * c_max)
    return grad, smooth_max, w

    while k < n:

Adam-based optimization with adaptive smooth-max for minimizing
2*n*max(full_conv(a,a))/sum(a)^2 over non-negative sequences.
Uses log-space parameterization, multi-phase strategy with
coarse-to-fine beta, multi-restart, and multi-scale exploration.

import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _fft_conv(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _eval(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _grad_logspace(a, beta, n):
    """Compute gradient of smooth max in log-space coordinates."""
    c = _fft_conv(a, a)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    e = np.exp(shifted)
    s = float(e.sum())
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    a_rev = a[::-1]
    wc = _fft_conv(w, a_rev)
    grad_a = 4.0 * n * wc[n - 1:2 * n - 1]
    grad_a = grad_a - grad_a.mean()
    grad_z = grad_a * a
    return grad_z

    """Adam optimizer in log-space with adaptive beta schedule."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30) + 1e-30)
    z = z - z.mean()

    best_a = a_init / a_init.sum()
    c0 = _fft_conv(best_a, best_a)
    best_m = float(c0.max())

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t_step = 0

    it = 0
    t_start = time.monotonic()
    max_iters = 100_000_000
    stall_count = 0
    stall_threshold = n_check * 20

    while it < max_iters:
        if time.monotonic() > deadline:
            break
        it += 1
        t_step += 1

        frac = min(1.0, it / (max_iters * 0.6))
        beta = beta_start * (beta_end / beta_start) ** frac

        a = np.exp(z - z.max())
        a = a / a.sum()

        grad_z = _grad_logspace(a, beta, n)

        gn = np.linalg.norm(grad_z)
        if gn > 10.0:
            grad_z *= 10.0 / gn

        m_buf = b1 * m_buf + (1.0 - b1) * grad_z
        v_buf = b2 * v_buf + (1.0 - b2) * (grad_z * grad_z)
        m_hat = m_buf / (1.0 - b1 ** t_step)
        v_hat = v_buf / (1.0 - b2 ** t_step)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        sn = np.linalg.norm(step)
        if sn > 1.0:
            step *= 1.0 / sn

        z = z - step
        z = z - z.mean()
        z = np.clip(z, -60.0, 60.0)

        if it % n_check == 0:
            a_ch = np.exp(z - z.max())
            a_ch = a_ch / a_ch.sum()
            c_ch = _fft_conv(a_ch, a_ch)
            m_ch = float(c_ch.max())
            if m_ch < best_m - 1e-12:
                best_m = m_ch
                best_a = a_ch.copy()
                stall_count = 0
            else:
                stall_count += 1

            if stall_count > stall_threshold:
                # Reduce lr and increase beta aggression
                lr *= 0.5
                stall_count = 0
                if lr < 1e-6:
                    break

            if it % (n_check * 10) == 0:
                print(f"  {label} it={it} beta={beta:.1f} M={best_m:.10f} "
                      f"score={2.0 * n * best_m:.10f} t={time.monotonic() - t_start:.0f}s")

    a_f = np.exp(z - z.max())
    a_f = a_f / a_f.sum()
    c_f = _fft_conv(a_f, a_f)
    mf = float(c_f.max())
    if mf < best_m:
        best_m = mf
        best_a = a_f.copy()

    return best_a, best_m


def _perturb_best(a_best, sigma, rng):
    """Create a perturbed version of the best sequence."""
    n = len(a_best)
    z = np.log(np.maximum(a_best, 1e-30) + 1e-30)
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    a = np.exp(z - z.max())
    a = a / a.sum()
    return a


def _interp_to_n(a, target_n):
    """Interpolate sequence to target length."""
    n0 = len(a)
    if n0 == target_n:
        return a.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, target_n)
    p = np.interp(x1, x0, a)
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    return p


def propose_candidate(seed=42, budget_s=1800):
    t0 = time.monotonic()
    deadline = t0 + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = np.inf

    # Load parent sequence
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent = np.maximum(parent, 0.0)
        if parent.sum() > 0.01:
            parent = parent / parent.sum()
            print(f"Parent: n={len(parent)}, score={_eval(parent):.10f}")
            best_seq = parent.copy()
            best_score = _eval(parent)
        else:
            raise ValueError
    except Exception:
        parent = np.ones(8192) / 8192
        best_seq = parent.copy()
        best_score = _eval(parent)
        print(f"No parent, using uniform n=8192, score={best_score:.10f}")

    n_main = len(best_seq)

    # === Phase 1: Coarse Adam (0 - 250s) ===
    print(f"\n=== Phase 1: Coarse Adam (n={n_main}) ===")
    p1_end = min(deadline, t0 + 250.0)
    time_budget_1 = p1_end - time.monotonic()
    if time_budget_1 > 10:
        a1, _ = _adam_optimize(best_seq, p1_end, lr=0.05,
                               beta_start=1.0, beta_end=100.0,
                               n_check=2000, label="P1")
        sc = _eval(a1)
        if sc < best_score:
            best_score = sc
            best_seq = a1.copy()
            print(f"  P1 improved to {best_score:.10f}")
    print(f"  After P1: best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 2: Sharpen (250s - 550s) ===
    print(f"\n=== Phase 2: Sharpen ===")
    p2_end = min(deadline, t0 + 550.0)
    if time.monotonic() < p2_end - 10:
        a2, _ = _adam_optimize(best_seq, p2_end, lr=0.02,
                               beta_start=50.0, beta_end=2000.0,
                               n_check=3000, label="P2")
        sc = _eval(a2)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print(f"  P2 improved to {best_score:.10f}")
    print(f"  After P2: best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 3: Multi-restart (550s - 1100s) ===
    print(f"\n=== Phase 3: Multi-restart ===")
    p3_end = min(deadline, t0 + 1100.0)
    n_restarts = 8
    sigmas = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]
    per_restart = max(30.0, (p3_end - time.monotonic()) / n_restarts)

    for i in range(n_restarts):
        if time.monotonic() > p3_end - 15:
            break
        sigma = sigmas[i % len(sigmas)]
        a_r = _perturb_best(best_seq, sigma, rng)
        dl_r = min(time.monotonic() + per_restart, p3_end - 10)
        lr_r = 0.03 if sigma < 0.1 else 0.05
        b_start = 10.0 if sigma < 0.1 else 1.0
        b_end = 500.0 if sigma < 0.1 else 200.0
        a_r2, _ = _adam_optimize(a_r, dl_r, lr=lr_r,
                                 beta_start=b_start, beta_end=b_end,
                                 n_check=2000, label=f"P3-{i}(s={sigma})")
        sc = _eval(a_r2)
        if sc < best_score:
            best_score = sc
            best_seq = a_r2.copy()
            print(f"  P3-{i} sigma={sigma} improved to {best_score:.10f}")
        else:
            print(f"  P3-{i} sigma={sigma} no improvement (got {sc:.10f})")

    print(f"  After P3: best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 4: Second round of fine-tuning (1100s - 1400s) ===
    print(f"\n=== Phase 4: Fine-tune ===")
    p4_end = min(deadline, t0 + 1400.0)
    if time.monotonic() < p4_end - 10:
        a4, _ = _adam_optimize(best_seq, p4_end, lr=0.01,
                               beta_start=100.0, beta_end=10000.0,
                               n_check=5000, label="P4")
        sc = _eval(a4)
        if sc < best_score:
            best_score = sc
            best_seq = a4.copy()
            print(f"  P4 improved to {best_score:.10f}")
    print(f"  After P4: best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 5: Multi-scale exploration (1400s - 1650s) ===
    print(f"\n=== Phase 5: Multi-scale ===")
    p5_end = min(deadline, t0 + 1650.0)
    if best_seq is not None:
        current_n = len(best_seq)
        for target_n in sorted([current_n * 2, current_n * 4], reverse=False):
            if time.monotonic() > p5_end - 60:
                break
            if target_n > 131072:
                break
            p_interp = _interp_to_n(best_seq, target_n)
            sc_int = _eval(p_interp)
            print(f"  Interp to n={target_n}: score={sc_int:.10f}")
            if sc_int < best_score:
                best_score = sc_int
                best_seq = p_interp.copy()
                print(f"  Improved to {best_score:.10f}")
            remaining = p5_end - time.monotonic() - 20
            if remaining > 60:
                a5, _ = _adam_optimize(p_interp, time.monotonic() + remaining,
                                       lr=0.03, beta_start=5.0, beta_end=500.0,
                                       n_check=5000, label=f"P5-n{target_n}")
                sc5 = _eval(a5)
                if sc5 < best_score:
                    best_score = sc5
                    best_seq = a5.copy()
                    print(f"  P5 n={target_n} improved to {best_score:.10f}")

    print(f"  After P5: best={best_score:.10f}, t={time.monotonic()-t0:.0f}s")

    # === Phase 6: Final polish (1650s - deadline) ===
    print(f"\n=== Phase 6: Final polish ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a6, _ = _adam_optimize(best_seq, time.monotonic() + remaining,
                                   lr=0.003, beta_start=500.0, beta_end=50000.0,
                                   n_check=10000, label="P6")
            sc6 = _eval(a6)
            if sc6 < best_score:
                best_score = sc6
                best_seq = a6.copy()
                print(f"  P6 improved to {best_score:.10f}")

    # Final validation
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(2048) / 2048
        sc_final = _eval(final)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final)}, "
              f"total_time={time.monotonic()-t0:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check: {ex}")
        return [float(x) for x in final]

    fb = np.ones(2048) / 2048
    print(f"FALLBACK: score={_eval(fb):.10f}")
    return [float(x) for x in fb]
