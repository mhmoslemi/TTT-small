def next_pow2(n):

        k <<= 1
    return max(2, k)

def full_convolve(a):
    """Compute the full convolution of a using FFT for efficiency."""
    n = len(a)
    # Determine the minimum size needed for next_pow2
    out_len = 2 * n - 1
    N = next_pow2(out_len)
    
    # Pad the sequence with zeros if needed for FFT
    a_padded = np.fft.irfft(np.zeros(N, dtype=np.complex128), N) * 0.0
    a_padded[:n] = a
    
    # FFT for one-sided representation
    fa = np.fft.irfft(np.zeros(N, dtype=np.complex128), N)
    fa[:n] = a_padded[:n]
    
    # Actually compute via FFT multiplication
    fb = np.fft.irfft(np.zeros(N, dtype=np.complex128), N)
    fb[:n] = a_padded[:n]
    
    fc = np.fft.irfft(fa * fb, N)
    
    # Trim to actual output length (2n-1)
    full_conv = fc[:out_len]
    return full_conv.astype(np.float64)

def score_fn(a):
    """Compute the exact score for a sequence according to the evaluator."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
        
    conv = full_convolve(a)
    max_val = np.max(conv)
    return float(2.0 * n * max_val / (s ** 2))

def _greedy_init(n, mode='asym'):

    if mode == 'asym':

    elif mode == 'discrete':
        # Discrete ripple initialization for fine-grained exploration
        d = np.float64(max(5, len(str(n)))[:small(]^2

def evaluate_sequence(seq):
    """Full evaluation but ensures valid input and output."""
    if not isinstance(seq, (list, np.ndarray)):
        return np.inf
        
    seq = np.asarray(seq, dtype=np.float64)
    if not np.issubdtype(seq.dtype, np.floating):
        return np.inf
        
    # Defensive bounds to 0.0 and 1000.0 (as in evaluator)
    seq = np.minimum(seq, 1000.0)
    seq = np.maximum(seq, 0.0)
    total = np.sum(seq)
    if total < 0.01:
        return np.inf
        
    conv = full_convolve(seq)
    max_val = np.max(conv)
    return float(2.0 * len(seq) * max_val / (total ** 2))
