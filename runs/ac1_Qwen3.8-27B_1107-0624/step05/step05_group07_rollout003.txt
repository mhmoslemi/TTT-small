    """
    Solve: minimize t subject to conv(b, f)[k] <= t for all k,
           sum(b) = 1, b >= 0.
    Returns (b, t) or (None, None) on failure.
    """
    n = len(f)
    # Build convolution matrix: (A @ b)[k] = conv(f, b)[k] = sum_j f[k-j] * b[j]
    # A has shape (2n-1, n)
    n_rows = 2 * n - 1
    # Column j: A[k, j] = f[k-j] for k-j in [0, n-1]
    # i.e., for column j, rows j to j+n-1 have values f[0], f[1], ..., f[n-1]
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    row_idx = (np.arange(n, dtype=np.int32)[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    data = np.tile(f, n).astype(np.float64)
    A_conv = csr_matrix((data, (row_idx, col_idx)), shape=(n_rows, n))

    # Minimize t: c = [0,...,0, 1]
    # Equality: sum(b) = 1
    # Inequality: A_conv @ b - t <= 0

    c_obj[-1] = 1.0

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = np.clip((c - cm) / beta, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad = 4.0 * n * a * (h - W)
    return grad
