def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

        return 0.0

    max_s = float(shifted.max())
    lse = cm + beta * (max_s + float(np.log(np.sum(np.exp(shifted - max_s)))))
    return float(2.0 * n * lse)

    h = w_conv[n - 1:2 * n - 1]

Bilinear LP + L-BFGS + Adam annealing hybrid for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovation: alternating LP in both convolution directions with adaptive

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

            if s_trial < 1e-15:
                continue
            sc = _score_fn(a_trial)
            if np.isfinite(sc) and sc < current_score - 1e-10:
                best_a = a_trial.copy()
                current_score = sc
                improved = True
                break
        
        # Try blending with y (secondary direction)
        if y is not None:
            sum_y = float(y.sum())
            if sum_y > 1e-15:
                y_scaled = y * (best_a.sum() / sum_y)
                for t in [0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
                    a_trial = (1.0 - t) * best_a + t * y_scaled
                    a_trial = np.maximum(a_trial, 0.0)
                    s_trial = a_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    sc = _score_fn(a_trial)
                    if np.isfinite(sc) and sc < current_score - 1e-10:
                        best_a = a_trial.copy()
                        current_score = sc
                        improved = True
                        break
        
        if not improved:
            for t in [0.02, 0.01, 0.005]:
                a_trial = (1.0 - t) * best_a + t * x1_scaled
                a_trial = np.maximum(a_trial, 0.0)
                sc = _score_fn(a_trial)
                if np.isfinite(sc) and sc < current_score - 1e-10:
                    best_a = a_trial.copy()
                    current_score = sc
                    improved = True
                    break
            if not improved:
                break
        
        if rnd % 30 == 0:
            print(f"    Bilinear LP round {rnd}: score={current_score:.10f}")
    
    return best_a, best_score


def _run_lbfgs(a_init, deadline, beta_list=None, maxiter=5000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        fracs = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]
        beta_list = [b * m0 for b in fracs]
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': maxiter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 50, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _adam_anneal(a_init, deadline, lr=0.015, beta_start_frac=0.05, beta_end_frac=0.00005, seed=42):
    """Adam optimizer with geometric beta annealing and restarts."""
    n = len(a_init)
    rng = np.random.default_rng(seed)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    cm0 = float(c0.max())
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    no_improve = 0
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        t += 1
        
        progress = min(1.0, t / 30000.0)
        beta = cm0 * beta_start_frac * (beta_end_frac / beta_start_frac) ** progress
        beta = max(beta, 1e-12 * cm0)
        
        a = _softmax_fn(z)
        c = _full_convolve(a, a)
        cm = float(c.max())
        
        shifted = (c - cm) / beta
        max_s = float(shifted.max())
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        W = float(np.dot(w, c))
        a_rev = a[::-1]
        w_conv = _full_convolve_a_b(w, a_rev)
        h = w_conv[n - 1:2 * n - 1]
        grad = 4.0 * n * a * (h - W)
        
        if not np.all(np.isfinite(grad)):
            break
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -100, 100)
        z = z - z.mean()
        
        if t % 80 == 0:
            a_cur = _softmax_fn(z)
            sc = _score_norm_fn(a_cur)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 40:
                z = z + 0.04 * rng.standard_normal(n)
                z = z - z.mean()
                no_improve = 0
    
    return best_a, best_score


def _perturb_and_optimize(a_init, deadline, rng, max_iter=2000):
    """Add perturbation and re-optimize to escape local minima."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    # Add perturbation
    z = z + 0.02 * rng.standard_normal(n)
    z = z - z.mean()
    
    c0 = _full_convolve(_softmax_fn(z), _softmax_fn(z))
    m0 = float(c0.max())
    beta_list = [m0 * b for b in [0.02, 0.01, 0.005, 0.002, 0.001]]
    
    best_a = _softmax_fn(z)
    best_score = _score_norm_fn(best_a)
    
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _softmin_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_softmin_grad,
                options={'maxiter': max_iter, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 50, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _active_peak_reduction(a_init, deadline, K=5, max_rounds=50):
    """Directly minimize the top K peaks using L-BFGS on a targeted surrogate."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    for outer in range(max_rounds):
        if time.monotonic() > deadline:
            break
        a_cur = _softmax_fn(z)
        c = _full_convolve(a_cur, a_cur)
        peak_indices = np.argsort(c)[::-1][:K]
        
        def _peak_obj(z_in, pi=peak_indices, Kval=K):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[pi]
            pm = float(peaks.max())
            if pm < 1e-300:
                return 0.0
            beta = 1e-6 * pm
            sh = (peaks - pm) / beta
            lse = pm + beta * float(np.log(np.sum(np.exp(sh - sh.max()))))
            return float(2.0 * n * lse)
        
        def _peak_grad(z_in, pi=peak_indices, Kval=K):
            a_in = _softmax_fn(z_in)
            c_in = _full_convolve(a_in, a_in)
            peaks = c_in[pi]
            pm = float(peaks.max())
            if pm < 1e-300:
                return np.zeros_like(z_in)
            beta = 1e-6 * pm
            sh = (peaks - pm) / beta
            e = np.exp(sh - sh.max())
            w = e / e.sum()
            
            grad_a = np.zeros(n)
            for idx_k in range(len(pi)):
                k = pi[idx_k]
                wi = w[idx_k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    grad_a[i_lo:i_hi + 1] += wi * 2.0 * a_in[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
            
            Wa = float(np.dot(grad_a, a_in))
            grad_z = 2.0 * n * a_in * (grad_a - Wa)
            if not np.all(np.isfinite(grad_z)):
                return np.zeros_like(z_in)
            return grad_z
        
        try:
            result = minimize(_peak_obj, z, method='L-BFGS-B', jac=_peak_grad,
                              options={'maxiter': 2000, 'ftol': 1e-20,
                                       'gtol': 1e-14, 'maxls': 50, 'maxfun': 8000})
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score - 1e-12:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
                else:
                    break
        except Exception:
            break
    
    return best_a, best_score
