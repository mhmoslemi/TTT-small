    while k < n:

        phase = np.angle(A)

import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]

    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _mirror_descent(a, n_iters=10000, eta=None, deadline=None, noise=0.0):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Uses the subgradient of max_k (a*a)[k].
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = _score_fn(a)
    
    if eta is None:
        eta = 1.0 / np.sqrt(n)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute convolution and find peak
        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))
        
        # Subgradient: g[j] = 2*a[k_star - j] for valid j
        g = np.zeros(n)
        valid = np.arange(n)
        k_minus_j = k_star - valid
        mask = (k_minus_j >= 0) & (k_minus_j < n)
        g[mask] = 2.0 * a[k_minus_j[mask]]
        
        # Add noise for exploration
        if noise > 0:
            g = g + noise * np.random.randn(n)
        
        # Mirror descent update: a_new[j] = a[j] * exp(-eta * g[j]) / Z
        log_update = -eta * g
        log_update -= np.max(log_update)  # For numerical stability
        update = np.exp(log_update)
        a_new = a * update
        s_new = a_new.sum()
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Evaluate
        sc = _score_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        # Decay eta
        if it % 500 == 0 and it > 0:
            eta *= 0.9
        
        # Add periodic noise
        if noise > 0 and it % 1000 == 500:
            perturb = 0.01 * np.random.randn(n)
            a = a + perturb
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
    
    return best_a, best_score

def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse

def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    # Gradient computation
    # grad_a = 4*n * a * (conv(w, a_rev)[n-1:2n-1] - W)
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * n * (2.0 * a * (h - W))
    # Chain rule: grad_z = a * (grad_a - sum(a * grad_a))
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z

def _lbfgs_refine(a_init, deadline, beta_list=None):
    """L-BFGS with log-sum-exp smoothing."""
    n = len(a_init)

def _make_initializations(n, rng):
    """Generate diverse initializations."""
    t = np.linspace(0, 1, n)
    u = 2 * t - 1
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Linear ramps
    for slope in [1, 2, 3, -1, -2]:
        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Exponential
    for rate in [1, 2, 3, -1, -2, -3]:
        a = np.exp(rate * (t - 0.5))
        inits.append(a / a.sum())
    
    # Cosine
    for p in [0.5, 1, 2, 3, 4]:
        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        inits.append(a / a.sum())
    
    # Gaussian-like
    for sigma in [0.1, 0.2, 0.3, 0.5]:
        a = np.exp(-u**2 / (2 * sigma**2))
        inits.append(a / a.sum())
    
    # Random smooth
    for _ in range(5):
        coarse = rng.random(n // 16 + 1)
        a = np.interp(t, np.linspace(0, 1, len(coarse)), coarse)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    # Bump at random position
    for _ in range(3):
        center = rng.random()
        width = 0.1 + 0.3 * rng.random()
        a = np.exp(-((t - center) / width)**2)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
    
    return inits

def _poly_refine(a, deadline, n_rounds=50):

            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-10:
                return 1e10
            a_trial = a_trial / s
            return _score_fn(a_trial)
        
        # Use Nelder-Mead or L-BFGS on the coefficients
        try:
            result = minimize(obj, coeffs, method='Nelder-Mead',

            if result.x is not None:

import numpy as np

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

        return a, np.inf

    current_alpha = 1.0

def _mirror_descent(a, n_iters=50000, eta_init=None, deadline=None, noise=0.0, rng=None):
    """
    Mirror descent (multiplicative weights) on the simplex.
    Subgradient of max_k (a*a)[k] is 2*a[k*-j] at the active peak k*.
    """
    n = len(a)
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _score_fn(a)

    if eta_init is None:
        eta = 2.0 / np.sqrt(n)
    else:
        eta = eta_init

    if rng is None:
        rng = np.random.default_rng(0)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _full_convolve(a, a)
        k_star = int(np.argmax(c))

        # Subgradient: g[j] = 2*a[k_star - j] for valid j
        g = np.zeros(n)
        j_idx = np.arange(n)
        k_minus_j = k_star - j_idx
        mask = (k_minus_j >= 0) & (k_minus_j < n)
        g[mask] = 2.0 * a[k_minus_j[mask]]

        if noise > 0:
            g = g + noise * rng.standard_normal(n)

        # Multiplicative update
        log_updates = -eta * g
        log_updates -= np.max(log_updates)
        updates = np.exp(log_updates)
        a_new = a * updates
        s_new = a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        sc = _score_fn(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        # Schedule: decay eta, occasionally reset to best
        if it > 0 and it % 1000 == 0:
            eta *= 0.85
            if sc > best_score * 1.01:
                a = best_a.copy()

        if noise > 0 and it % 5000 == 2500:
            a = a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()

    return best_a, best_score


def _logsumexp_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    lse = cm + beta * float(np.log(np.sum(np.exp(vals))))
    return 2.0 * n * lse


def _logsumexp_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    vals = (c - cm) / beta
    vals = np.clip(vals, -50, 50)
    e = np.exp(vals)
    w = e / e.sum()
    W = float(np.dot(w, c))
    a_rev = a[::-1].copy()
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 4.0 * n * a * (h - W)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    return grad_z

def _make_initializations(n, rng):

        a = 1.0 + slope * (t - 0.5)
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    for rate in [1, 2, 3, -1, -2, -3, 0.5, -0.5]:
        a = np.exp(rate * (t - 0.5))
        inits.append(a / a.sum())

        a = np.maximum(np.cos(np.pi * u / 2), 1e-10) ** p
        inits.append(a / a.sum())

        a = np.exp(-u ** 2 / (2 * sigma ** 2))
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

        a = 1.0 + c * u ** 2
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())
        a = 1.0 - c * u ** 2
        a = np.maximum(a, 1e-10)
        inits.append(a / a.sum())

    # Random smooth
    for _ in range(8):
        n_coarse = max(4, n // 32)
        coarse = rng.random(n_coarse)
        a = np.interp(t, np.linspace(0, 1, n_coarse), coarse)

def _poly_optimize(a_init, deadline):
    """Optimize a polynomial parameterization of the sequence."""
    n = len(a_init)
    x = np.linspace(-1, 1, n)
    best_a = a_init.copy()
    best_score = _score_fn(a_init)

    for degree in [6, 8, 10, 12]:
        if time.monotonic() > deadline:
            break
        coeffs = np.polyfit(x, a_init, degree)

        def obj(cv):
            a_t = np.polyval(cv, x)
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                return 1e10
            a_t = a_t / s
            return _score_fn(a_t)

        def grad(cv):
            eps = 1e-7
            g = np.zeros_like(cv)
            f0 = obj(cv)
            for i in range(len(cv)):
                cv[i] += eps
                g[i] = (obj(cv) - f0) / eps
                cv[i] -= eps
            return g

        try:
            result = minimize(obj, coeffs, jac=grad, method='L-BFGS-B',
                             options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            if result.x is not None:
                a_t = np.polyval(result.x, x)
                a_t = np.maximum(a_t, 0)
                s = a_t.sum()
                if s > 1e-10:
                    a_t = a_t / s
                    sc = _score_fn(a_t)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_t.copy()
        except Exception:
            pass

    return best_a, best_score


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

        n_trials = min(4, len(inits))
        for tr in range(n_trials):
            if time.monotonic() > phase1_end - 10:
                break
            a_init = inits[tr]
            per_dl = min(phase1_end - 5, time.monotonic() + 200)

            # DC-concentrated spectral inversion
            a_best, sc_best = _spectral_inversion(a_init, n_iters=2000,
                                                   alpha=0.9, window='hann', deadline=per_dl)
            update_best(a_best, sc_best, f"spec_dc_n{n_val}_t{tr}")

            if time.monotonic() < phase1_end - 30:
                # Flat spectral inversion
                per_dl2 = min(phase1_end - 5, time.monotonic() + 100)
                a_b2, sc_b2 = _spectral_inversion_flat(a_best, n_iters=1000, deadline=per_dl2)
                update_best(a_b2, sc_b2, f"spec_flat_n{n_val}_t{tr}")

    print(f"  Phase 1 done: best={best_score:.10f}")

    # === Phase 2: Mirror descent (20% budget) ===
    phase2_end = t_start + 0.65 * (deadline - t_start)
    print(f"\nPhase 2: Mirror descent")

    if best_seq is not None:
        n_cur = len(best_seq)
        a_md = best_seq.copy()

        for eta_mult in [1.0, 2.0, 0.5]:
            if time.monotonic() > phase2_end - 20:
                break
            eta = eta_mult * 2.0 / np.sqrt(n_cur)
            per_dl = min(phase2_end - 5, time.monotonic() + (phase2_end - time.monotonic()) * 0.6)
            a_md_best, sc_md = _mirror_descent(a_md, n_iters=100000, eta_init=eta,
                                               deadline=per_dl, rng=rng)
            update_best(a_md_best, sc_md, f"mirror_eta{eta_mult}")
            a_md = a_md_best

        # Noisy mirror descent for exploration
        if time.monotonic() < phase2_end - 30:
            a_noisy = best_seq + 0.02 * rng.standard_normal(n_cur)
            a_noisy = np.maximum(a_noisy, 1e-10)
            a_noisy = a_noisy / a_noisy.sum()
            per_dl = min(phase2_end - 5, time.monotonic() + 120)
            a_nb, sc_nb = _mirror_descent(a_noisy, n_iters=50000,
                                          eta_init=2.0 / np.sqrt(n_cur),
                                          deadline=per_dl, noise=0.05, rng=rng)
            update_best(a_nb, sc_nb, "mirror_noisy")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: L-BFGS + Adaptive GD (20% budget) ===
    phase3_end = t_start + 0.85 * (deadline - t_start)
    print(f"\nPhase 3: L-BFGS + Adaptive GD")

    if best_seq is not None:
        # L-BFGS from best
        per_dl = min(phase3_end - 30, time.monotonic() + (phase3_end - time.monotonic()) * 0.5)
        a_lb, sc_lb = _lbfgs_refine(best_seq, per_dl)
        update_best(a_lb, sc_lb, "lbfgs")

        # Adaptive GD
        if time.monotonic() < phase3_end - 30:
            per_dl2 = min(phase3_end - 20, time.monotonic() + 120)
            a_gd, sc_gd = _adaptive_gd(a_lb, per_dl2, rng, lr_init=0.3)
            update_best(a_gd, sc_gd, "adaptive_gd")

        # L-BFGS from perturbed
        if time.monotonic() < phase3_end - 30:
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            z_p = z + 0.03 * rng.standard_normal(len(z))
            a_pert = _softmax_fn(z_p)
            per_dl3 = min(phase3_end - 5, time.monotonic() + 90)
            a_lp, sc_lp = _lbfgs_refine(a_pert, per_dl3)
            update_best(a_lp, sc_lp, "lbfgs_perturbed")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Polish (15% budget) ===
    phase4_end = deadline
    print(f"\nPhase 4: Polish")

    if best_seq is not None:
        # Polynomial optimization
        if time.monotonic() < phase4_end - 60:
            per_dl = min(phase4_end - 40, time.monotonic() + 60)
            a_poly, sc_poly = _poly_optimize(best_seq, per_dl)
            update_best(a_poly, sc_poly, "poly")

        # Final L-BFGS with tiny beta
        if time.monotonic() < phase4_end - 30:
            n_cur = len(best_seq)
            c_cur = _full_convolve(best_seq, best_seq)
            beta_tiny = 5e-5 * float(c_cur.max())
            z = np.log(np.maximum(best_seq, 1e-15))
            z = z - z.mean()
            per_dl = min(phase4_end - 10, time.monotonic() + (phase4_end - time.monotonic()) * 0.6)
            try:
                result = minimize(
                    _logsumexp_obj, z, args=(beta_tiny, n_cur),
                    method='L-BFGS-B', jac=_logsumexp_grad,
                    options={'maxiter': 10000, 'ftol': 1e-20, 'gtol': 1e-15,
                             'maxls': 50, 'maxfun': 30000}
                )
                if result.x is not None and np.all(np.isfinite(result.x)):
                    a_fin = _softmax_fn(result.x)
                    sc_fin = _score_fn(a_fin)
                    update_best(a_fin, sc_fin, "final_tiny_beta")
            except Exception:
                pass

        # Coordinate descent
        if time.monotonic() < phase4_end - 10:
            per_dl = min(phase4_end - 5, time.monotonic() + (phase4_end - time.monotonic()) * 0.7)
            a_cd, sc_cd = _coord_descent(best_seq, per_dl, max_rounds=10)
            update_best(a_cd, sc_cd, "coord_descent")

    print(f"  Phase 4 done: best={best_score:.10f}")

    # Final: try slightly different n by interpolation
    if best_seq is not None and time.monotonic() < deadline - 5:
        for n_try in [len(best_seq) + 512, len(best_seq) - 512]:
            if n_try < 128:
                continue
            p_interp = _interp_profile(best_seq, n_try)
            sc_interp = _score_fn(p_interp)
            update_best(p_interp, sc_interp, f"interp_n{n_try}")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.10f}, n={len(final_seq)}")

        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
            if np.isfinite(sc_eval):
                print(f"Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"evaluate_sequence failed: {ex}")

        return [float(x) for x in final_seq]

    # Fallback
    n_fb = 2048
    fallback = np.ones(n_fb) / n_fb
    print(f"FALLBACK: n={n_fb}, score={_score_fn(fallback):.10f}")
    return [float(x) for x in fallback]
