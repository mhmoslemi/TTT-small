"""
Optimized step-function sequence minimizer for 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

def _next_pow2(n):
    k = 2
    while k < n:
        k <<= 1
    return k


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _make_triangular_weight(n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_parabolic_profile(n, p):
    t = np.linspace(0.0, 1.0, n)
    u = np.abs(2.0 * t - 1.0)
    a = (1.0 - u) ** p
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n, alpha=1.0):

def _interp_profile(p0, n_new):

    n0 = len(p0)

        return p0 / p0.sum()

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(ind[cond])
    theta = cssv[rho - 1] / rho
    return np.maximum(v - theta, 0.0)


def _subgradient_polish(a_init, deadline, rng, max_iters=200000):
    """
    Projected subgradient method for minimizing max(conv(a,a)) on the simplex.
    Uses adaptive step size and multi-lag weighting.
    """
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    c = _full_convolve(a, a)
    c_max = float(c.max())

    eta = 0.1
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break

        # Compute weighted subgradient using top lags
        num_top = min(10, len(c))
        sorted_idx = np.argsort(c)[-num_top:]
        c_top = c[sorted_idx]
        c_max_local = c_top[-1]
        # Softmax weights over top lags (temperature ~ (c_max - c_min_top))
        temp = max(1e-10, (c_max_local - c_top[0]) * 0.5)
        w = np.exp((c_top - c_max_local) / temp)
        w = w / w.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for wi, ki in zip(w, sorted_idx):
            ki = int(ki)
            # sg[j] += wi * 2 * a[ki - j] for valid ki-j
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            for j in range(lo, hi + 1):
                sg[j] += wi * 2.0 * a[ki - j]

        # Adaptive step size
        sg_norm_sq = float(np.dot(sg, sg))
        if sg_norm_sq < 1e-30:
            break

        # Polyak-like step size
        step = eta * c_max / (sg_norm_sq * (it + 1))
        step = min(step, 0.5)

        a_new = a - step * sg
        a_new = _project_simplex(a_new)

        sc_new = _score_norm_fn(a_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        c = _full_convolve(a, a)
        c_max = float(c.max())

        # Decay eta
        if it % 500 == 0:
            eta = max(0.01, eta * 0.7)

    return best_a, best_sc


def _multilag_lbfgs(a_init, deadline, rng, max_wall_time=None):
    """
    L-BFGS with logsumexp continuation on the softmax parameterization.
    Very aggressive beta schedule for tight convergence.
    """
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())

    # Very fine beta schedule
    beta_fracs = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005,
                  0.0002, 0.0001, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    betas = [b * max_c0 for b in beta_fracs]

    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(betas))
    else:
        per_level = 30.0

    for beta in betas:
        if time.monotonic() > deadline - 3.0:
            break
        level_end = min(deadline, time.monotonic() + per_level)

        def _obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300:
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * float(np.log(s_)))

        def _grad(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_sh = (c_ - cm_) / beta
            e_ = np.exp(c_sh)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g

        try:
            result = minimize(
                _obj, z.copy(), jac=_grad,
                method='L-BFGS-B',
                options={'maxiter': 5000, 'ftol': 1e-25, 'gtol': 1e-14,
                         'maxls': 80, 'maxfun': 50000, 'maxcor': 30}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                z_new = result.x
                a_new = _softmax_fn(z_new)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = z_new.copy()
        except Exception:
            pass

    return best_a, best_score


def _lp_fw_step(f, M, timeout=120.0):
    """Solve LP: max sum(g) s.t. convolve(f, g) <= M, g >= 0."""
    n = len(f)
    try:
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
        from scipy.sparse import csr_matrix
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        g = cp.Variable(n, non_neg=True)
        constraints = [A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)

        for solver in [cp.CBC, cp.GLPK]:
            try:
                prob.solve(solver=solver, threads=1)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    if g.value is not None:
                        return np.maximum(g.value, 0.0)
            except Exception:
                continue
    except Exception:
        pass
    return None


def _lp_fw_method(f_init, deadline, rng, max_iters=20):
    """LP-based Frank-Wolfe (conditional gradient) method."""
    n = len(f_init)
    f = f_init.copy()
    c = _full_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * best_M

    for it in range(max_iters):
        if time.monotonic() > deadline - 30.0:
            break

        t_step = min(2.0 / (it + 2.0), 0.4)
        g = _lp_fw_step(f, M, timeout=90.0)
        if g is None:
            break

        s_f = f.sum()
        s_g = g.sum()
        if s_g < 1e-15:
            break
        g = g * (s_f / s_g)

        f_new = (1.0 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new = f_new / s * s_f

        c_new = _full_convolve(f_new, f_new)
        M_new = float(c_new.max())
        sc_new = 2.0 * n * M_new

        if M_new < best_M:
            best_M = M_new
            best_f = f_new.copy()
            best_score = sc_new
            print(f"  FW iter {it}: score={sc_new:.10f}, t={t_step:.3f}")

        f = f_new
        M = M_new

    return best_f / best_f.sum(), best_score


def _peak_local_search(a_init, deadline, rng, n_trials=50000):
    """Peak-targeted local search with multiple strategies."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)

    c = _full_convolve(a, a)
    for it in range(n_trials):
        if time.monotonic() > deadline - 2.0:
            break

        if it % 500 == 0:
            c = _full_convolve(a, a)

        K = min(5, len(c))
        top_lags = np.argsort(c)[-K:]
        max_lag = int(top_lags[-1])

        strategy = it % 4
        if strategy == 0:
            # Reduce at highest contributing position for max_lag
            i_min = max(0, max_lag - (n - 1))
            i_max = min(n - 1, max_lag)
            if i_min <= i_max:
                pairs = np.arange(i_min, i_max + 1)
                partners = max_lag - pairs
                products = a[pairs] * a[partners]
                idx = int(pairs[np.argmax(products)])

                a_trial = a.copy()
                frac = rng.uniform(0.3, 0.7)
                delta = a_trial[idx] * frac
                a_trial[idx] = max(0.0, a_trial[idx] - delta)

                # Add to safe position
                contrib = np.zeros(n)
                for kl in top_lags:
                    kl = int(kl)
                    lo = max(0, kl - (n - 1))
                    hi = min(n - 1, kl)
                    contrib[lo:hi + 1] += a[lo:hi + 1] * a[kl - lo:kl - hi + 1][::-1]
                safe = np.argsort(contrib)[:max(1, n // 10)]
                add_idx = int(safe[rng.integers(len(safe))])
                a_trial[add_idx] += delta
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm_fn(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()

        elif strategy == 1:
            # Block perturbation
            block_size = int(rng.integers(2, min(32, n // 8)))
            start = int(rng.integers(0, max(1, n - block_size)))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        elif strategy == 2:
            # Pair swap
            i1 = int(rng.integers(n))
            i2 = int(rng.integers(n))
            if i1 == i2:
                continue
            a_trial = a.copy()
            frac = rng.uniform(0.1, 0.5)
            amt1 = a_trial[i1] * frac
            amt2 = a_trial[i2] * frac
            a_trial[i1] = max(0.0, a_trial[i1] - amt1)
            a_trial[i2] = max(0.0, a_trial[i2] - amt2)
            a_trial[i1] += amt2
            a_trial[i2] += amt1
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

        else:
            # Multi-position redistribution
            num_move = int(rng.integers(2, 6))
            src_ids = rng.choice(n, size=num_move, replace=False)
            a_trial = a.copy()
            total = 0.0
            for si in src_ids:
                frac = rng.uniform(0.2, 0.5)
                amt = a_trial[si] * frac
                a_trial[si] -= amt
                total += amt
            dst_ids = rng.choice(n, size=num_move, replace=False)
            for di in dst_ids:
                a_trial[di] += total / num_move
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()

    return best_a, best_sc


def _adam_topk(a_init, deadline, rng, lr=0.015, k_top=5):
    """Adam optimizer with top-K lag gradient."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()

    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.8, 0.999, 1e-8

    max_iters = 100000
    for it in range(max_iters):
        if time.monotonic() > deadline - 2.0:
            break

        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        k_act = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_act:]

        grad = np.zeros(n)
        for k_val in top_idx:
            k_val = int(k_val)
            lo = max(0, k_val - (n - 1))
            hi = min(n - 1, k_val)
            grad[lo:hi + 1] += 2.0 * a_curr[k_val - hi:k_val - lo + 1][::-1]
        grad /= k_act
        grad *= 4.0 * n

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        z = z - lr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -60, 60)
        z = z - z.mean()

        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()

        if t % 5000 == 0:
            lr = max(0.001, lr * 0.9)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimizer: diverse init -> LP-FW -> L-BFGS continuation ->
    subgradient polish -> Adam -> local search -> final refine.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")

    # Get parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None

    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_triangular_weight(n_parent)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using triangular n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")

    print(f"Phase 0: Diverse initialization")
    phase0_end = min(deadline, t_start + 50.0)
    n_work = n_parent

    profiles = [
        _make_triangular_weight(n_work),
        _make_parabolic_profile(n_work, 1.0),
        _make_parabolic_profile(n_work, 1.5),
        _make_parabolic_profile(n_work, 2.0),
        _make_parabolic_profile(n_work, 3.0),
        _make_parabolic_profile(n_work, 4.0),
        _make_raised_cosine(n_work, 1.0),
        _make_raised_cosine(n_work, 2.0),
        _make_raised_cosine(n_work, 3.0),
        _make_sine_mod(n_work, 1, 0.3),
        _make_sine_mod(n_work, 1, 0.5),
        _make_sine_mod(n_work, 2, 0.3),
        _make_sine_mod(n_work, 2, 0.5),
        _make_sine_mod(n_work, 3, 0.3),
        _make_sine_mod(n_work, 3, 0.4),
        _make_sine_mod(n_work, 4, 0.3),
        _make_bimodal_profile(n_work, 0.3, 0.12),
        _make_bimodal_profile(n_work, 0.4, 0.15),
        _make_bimodal_profile(n_work, 0.5, 0.2),
        _make_bimodal_profile(n_work, 0.6, 0.1),
        _make_bspline_profile(n_work, 3),
        _make_bspline_profile(n_work, 4),
        _make_bspline_profile(n_work, 5),
        _make_bspline_profile(n_work, 6),
        _make_gaussian_profile(n_work, 0.15),
        _make_gaussian_profile(n_work, 0.2),
        _make_gaussian_profile(n_work, 0.25),
        _make_quadratic_profile(n_work, 0.5, 0.3),
        _make_quadratic_profile(n_work, 0.8, -0.2),
        _make_quadratic_profile(n_work, 1.0, 0.2),
        _make_quadratic_profile(n_work, 0.3, 0.5),
        _make_quadratic_profile(n_work, -0.3, 0.4),
    ]

    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")

    # Also try different n values from parent
    if parent is not None:
        for target_n in [n_work // 2, n_work * 2]:
            if target_n < 512 or target_n > 32768:
                continue
            if time.monotonic() > phase0_end:
                break
            p_interp = _interp_profile(parent, target_n)
            sc = _score_norm_fn(p_interp)
            candidates.append((sc, p_interp.copy(), f"parent_n{target_n}"))
            update_best(p_interp, sc, f"parent_n{target_n}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")

    # Pick best n and starting profile
    start_seq = candidates[0][1].copy() if candidates else parent.copy()
    n_curr = len(start_seq)

    # Phase 1: LP-based Frank-Wolfe (200s)
    phase1_end = min(deadline, t_start + 200.0)
    if time.monotonic() < phase1_end - 10:
        print(f"Phase 1: LP Frank-Wolfe (score={_score_norm_fn(start_seq):.10f})")
        a_fw, sc_fw = _lp_fw_method(start_seq, phase1_end, rng, max_iters=15)
        update_best(a_fw, sc_fw, "fw_lp")
        if sc_fw < _score_norm_fn(start_seq):
            start_seq = a_fw.copy()

    # Phase 2: L-BFGS continuation with aggressive beta (700s)
    phase2_end = min(deadline, t_start + 900.0)
    if best_seq is not None and time.monotonic() < phase2_end - 10:
        print(f"Phase 2: L-BFGS continuation (score={_score_norm_fn(best_seq):.10f})")
        dl = phase2_end - 5.0
        a_best, sc_best = _multilag_lbfgs(best_seq, dl, rng, max_wall_time=680.0)
        update_best(a_best, sc_best, "lbfgs_aggressive")

        # Second pass from the improved solution
        if time.monotonic() < phase2_end - 30:
            dl2 = phase2_end - 5.0
            a_best2, sc_best2 = _multilag_lbfgs(best_seq, dl2, rng, max_wall_time=120.0)
            update_best(a_best2, sc_best2, "lbfgs_pass2")

    print(f"After Phase 2: best={best_score:.10f}")

    # Phase 3: Subgradient polishing (300s)
    phase3_end = min(deadline, t_start + 1200.0)
    if best_seq is not None and time.monotonic() < phase3_end - 10:
        print(f"Phase 3: Subgradient polishing (score={_score_norm_fn(best_seq):.10f})")
        dl = phase3_end - 5.0
        a_best, sc_best = _subgradient_polish(best_seq, dl, rng, max_iters=300000)
        update_best(a_best, sc_best, "subgrad")

    print(f"After Phase 3: best={best_score:.10f}")

    # Phase 4: Adam top-K (200s)
    phase4_end = min(deadline, t_start + 1400.0)
    if best_seq is not None and time.monotonic() < phase4_end - 10:
        print(f"Phase 4: Adam top-K (score={_score_norm_fn(best_seq):.10f})")
        dl = phase4_end - 5.0
        a_best, sc_best = _adam_topk(best_seq, dl, rng, lr=0.01, k_top=3)
        update_best(a_best, sc_best, "adam_k3")

        if time.monotonic() < phase4_end - 30:
            dl2 = phase4_end - 5.0
            a_best2, sc_best2 = _adam_topk(best_seq, dl2, rng, lr=0.006, k_top=7)
            update_best(a_best2, sc_best2, "adam_k7")

    print(f"After Phase 4: best={best_score:.10f}")

    # Phase 5: Local search (200s)
    phase5_end = min(deadline, t_start + 1600.0)
    if best_seq is not None and time.monotonic() < phase5_end - 10:
        print(f"Phase 5: Local search (score={_score_norm_fn(best_seq):.10f})")
        dl = phase5_end - 5.0
        a_best, sc_best = _peak_local_search(best_seq, dl, rng, n_trials=60000)
        update_best(a_best, sc_best, "local_search")

    print(f"After Phase 5: best={best_score:.10f}")

    # Phase 6: Final L-BFGS polish with very low beta (remaining time)
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = max(20.0, deadline - time.monotonic() - 10)
        print(f"Phase 6: Final polish, remaining={remaining:.0f}s")

        # 6a: Fine L-BFGS
        if remaining > 40:
            dl = time.monotonic() + remaining * 0.5
            a_best, sc_best = _multilag_lbfgs(best_seq, dl, rng, max_wall_time=remaining * 0.5)
            update_best(a_best, sc_best, "final_lbfgs")

        # 6b: Quick local search
        if time.monotonic() < deadline - 20:
            dl2 = deadline - 15
            a_best2, sc_best2 = _peak_local_search(best_seq, dl2, rng, n_trials=20000)
            update_best(a_best2, sc_best2, "final_local")

        # 6c: Final subgradient
        if time.monotonic() < deadline - 15:
            dl3 = deadline - 10
            a_best3, sc_best3 = _subgradient_polish(best_seq, dl3, rng, max_iters=100000)
            update_best(a_best3, sc_best3, "final_subgrad")

    # Final validation
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s < 0.01:
            final_seq = _make_triangular_weight(4096)
            s = float(final_seq.sum())
        final_seq = final_seq / s

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}, sum={s:.6f}")

        return [float(x) for x in final_seq]

    # Fallback
    fallback = _make_triangular_weight(4096)
    sc_fb = _score_fn(fallback)
    print(f"FALLBACK: n=4096, score={sc_fb:.10f}")
    return [float(x) for x in fallback]
