from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r

def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _score_norm_fn(a):
    """Score with sum(a) assumed to be 1 (normalized)."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum

def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:

    e = np.exp(shifted)

    a_rev = a[::-1]

    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z

def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()

    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()

    betas = np.geomspace(beta_start, beta_end, n_betas)

        if time.monotonic() > deadline:
            break
        remaining = n_betas - idx
        time_budget = (deadline - time.monotonic()) / max(1, remaining)
        time_budget = min(time_budget, 120.0)
        level_deadline = time.monotonic() + time_budget
        
        maxiter = 300 if beta > 0.1 * max_c0 else (1000 if beta > 0.01 * max_c0 else 3000)
        maxfun = maxiter * 5
        
        try:

                _logsumexp_obj, z, args=(beta,),
                method='L-BFGS-B', jac=_logsumexp_grad,

                         'maxls': 50, 'maxfun': maxfun}

Phase 2: LP-based iterative peak reduction using constraint generation at n=4096

import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _logsumexp_obj(z, beta):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return 0.0
    shifted = (c - cm) / beta
    logsum = cm / beta + float(np.log(np.sum(np.exp(shifted))))
    return beta * logsum


def _logsumexp_grad(z, beta):
    a = _softmax_fn(z)
    n = len(a)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-30:
        return np.zeros(n)
    shifted = (c - cm) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(a_rev, w)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    offset = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - offset)
    return grad_z


def _upsample(a, target_n):
    n = len(a)
    if n == target_n:
        return a.copy()
    if n == 1:
        return np.ones(target_n) / target_n
    x0 = np.linspace(0.0, 1.0, n)
    x1 = np.linspace(0.0, 1.0, target_n)
    new_a = np.interp(x1, x0, a)
    new_a = np.maximum(new_a, 1e-15)
    return new_a / new_a.sum()


def _make_ramp(n, slope=1.5):
    t = np.linspace(0, 1, n)
    a = 1.0 + slope * t
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_profiles(n):

def _solve_lp_with_cg(f, M, deadline, max_rounds=20, initial_k=100, add_k=100):
    n = len(f)
    c_ff = _full_convolve(f, f)

    peak_idx = np.argsort(c_ff)[::-1][:initial_k]
    A = _build_lp_matrix(f, peak_idx, n)
    b = np.full(len(peak_idx), M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    g = None
    for rd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        try:
            result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds,
                             method='highs', options={'presolve': True})
            if not result.success:
                break
            g = result.x

            fg = _full_convolve(f, g)
            violations = np.where(fg > M * (1 + 1e-8))[0]
            if len(violations) == 0:

            viol_sorted = np.argsort(fg[violations])[::-1][:add_k]

            A = np.vstack([A, new_A_rows])

        except Exception:
            break

        if time.monotonic() > phase1_end - 10:
            break
        a_target = _upsample(a_cur, target_n)
        s_t = _score_norm_fn(a_target)
        alloc = time_alloc.get(target_n, 40)
        opt_dl = min(phase1_end - 5, time.monotonic() + alloc)
        a_opt, sc_opt = _run_lbfgs_continuation(a_target, opt_dl, n_betas=12)
        print(f"  n={target_n}: upsample={s_t:.12f}, optimized={sc_opt:.12f}")
        if sc_opt < sc_cur:
            a_cur = a_opt
            sc_cur = sc_opt

    update_best(a_cur, sc_cur, "phase1_multiscale")
    print(f"  Phase 1 complete: best={best_score:.12f}")

    # Ensure we work at n=4096 for Phase 2
    work_a = best_seq.copy()
    if len(work_a) != 4096:
        work_a = _upsample(work_a, 4096)
        work_a = work_a / work_a.sum()
    sc_work = _score_norm_fn(work_a)
    if sc_work < best_score:
        update_best(work_a, sc_work, "upsampled_to_4096")
    else:
        work_a = best_seq.copy()
        sc_work = best_score

    # Phase 2: LP-based refinement (40% of time)
    phase2_end = t_start + effective_budget * 0.80
    print(f"\n=== Phase 2: LP-based refinement (to t={phase2_end - t_start:.0f}s) ===")

    if time.monotonic() < phase2_end - 20:
        # First do a quick L-BFGS pass at n=4096 to polish before LP
        lbfgs_dl = min(phase2_end - 60, time.monotonic() + 60.0)
        a_p2, sc_p2 = _run_lbfgs_continuation(work_a, lbfgs_dl, n_betas=10)
        update_best(a_p2, sc_p2, "phase2_lbfgs")
        work_a = best_seq.copy()

        # Now LP refinement
        lp_dl = min(phase2_end - 10, deadline)
        a_lp, sc_lp = _lp_refine(work_a, lp_dl, top_k=200, max_outer=25)
        update_best(a_lp, sc_lp, "phase2_lp")
        work_a = best_seq.copy()

        # Another round of LP with different parameters
        if time.monotonic() < phase2_end - 30:
            a_lp2, sc_lp2 = _lp_refine(work_a, phase2_end - 10, top_k=100, max_outer=15)
            update_best(a_lp2, sc_lp2, "phase2_lp2")

    print(f"  Phase 2 complete: best={best_score:.12f}")

    # Phase 3: Adam + peak surgery (15% of time)
    phase3_end = t_start + effective_budget * 0.95
    print(f"\n=== Phase 3: Adam + Peak Surgery (to t={phase3_end - t_start:.0f}s) ===")

    if time.monotonic() < phase3_end - 10 and best_seq is not None:
        gd_dl = min(phase3_end - 30, time.monotonic() + 60.0)
        a_gd, sc_gd = _adam_refine(best_seq, gd_dl, lr_init=0.03)
        update_best(a_gd, sc_gd, "adam")

        if time.monotonic() < phase3_end - 15:
            ps_dl = min(phase3_end - 5, time.monotonic() + 40.0)
            a_ps, sc_ps = _peak_surgery(best_seq, ps_dl)
            update_best(a_ps, sc_ps, "peak_surgery")

        if time.monotonic() < phase3_end - 10:
            a_gd2, sc_gd2 = _adam_refine(best_seq, phase3_end - 5, lr_init=0.01)
            update_best(a_gd2, sc_gd2, "adam_fine")

    print(f"  Phase 3 complete: best={best_score:.12f}")

    # Phase 4: Final tiny-beta L-BFGS polish
    print(f"\n=== Phase 4: Final polish ===")
    if time.monotonic() < deadline - 10 and best_seq is not None:
        n_final = len(best_seq)
        c_final = _full_convolve(best_seq, best_seq)
        max_c_final = float(c_final.max())

        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()
        beta_tiny = 1e-6 * max_c_final

        try:
            result = minimize(
                _logsumexp_obj, z, args=(beta_tiny,),
                method='L-BFGS-B', jac=_logsumexp_grad,
                options={'maxiter': 10000, 'ftol': 1e-17, 'gtol': 1e-12,
                         'maxls': 30, 'maxfun': 30000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    update_best(a_new, sc_new, "final_lbfgs")
        except Exception as ex:
            print(f"  Final L-BFGS failed: {ex}")

        # One more LP round if time
        if time.monotonic() < deadline - 15 and best_seq is not None:
            a_final_lp, sc_final_lp = _lp_refine(best_seq, deadline - 5,
                                                 top_k=150, max_outer=5)
            update_best(a_final_lp, sc_final_lp, "final_lp")

    # Final validation and return
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _make_ramp(256, 1.5)

        sc_final = _score_fn(final_seq)
        print(f"\nFINAL: score={sc_final:.12f}, n={len(final_seq)}")
        return [float(x) for x in final_seq]

    fallback = _make_ramp(256, 1.5)
    print(f"FALLBACK: n=256, score={_score_fn(fallback):.12f}")
    return [float(x) for x in fallback]
