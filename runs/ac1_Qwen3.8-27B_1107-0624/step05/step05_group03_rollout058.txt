def mirror_descent(a, n_steps, deadline):
    # a is normalized to sum=1
    z = log(a) - mean(log(a))
    best = a
    best_score = score(a)
    for t in range(n_steps):
        a_curr = softmax(z)
        c = convolve(a_curr, a_curr)
        # Find active set (top-K peaks)
        K = 10
        top_idx = argsort(c)[-K:]
        # Compute subgradient
        g = zeros(n)
        for k in top_idx:
            lo = max(0, k-n+1)
            hi = min(n-1, k)
            g[lo:hi+1] += a_curr[k-hi:k-lo+1][::-1]
        g = 2.0 * g / len(top_idx) * 2*n  # scale for the objective
        # Step size (Polyak)
        eta = 2.0 / (t+1) * best_score  # approximate
        # Mirror step
        z = z - eta * g / max(abs(g).max(), 1e-10)
        z = z - z.mean()
        a_new = softmax(z)
        sc = score(a_new)
        if sc < best_score:
            best = a_new
            best_score = sc
    return best, best_score

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())

    s = float(a.sum())
    if s < 1e-15:

    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))

    x1 = np.linspace(0.0, 1.0, n_new)
    p = cs(x1)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _compute_subgradient(a, k_top=5):
    """Compute subgradient of max(conv(a,a)) w.r.t. a, using top-k peaks."""
    n = len(a)
    c = _full_convolve(a, a)
    k_actual = min(k_top, len(c))
    top_idx = np.argsort(c)[-k_actual:]
    g = np.zeros(n)
    for k_val in top_idx:
        kv = int(k_val)
        lo = max(0, kv - (n - 1))
        hi = min(n - 1, kv)
        if lo <= hi:
            g[lo:hi + 1] += a[kv - hi:kv - lo + 1][::-1]
    g *= 2.0 / k_actual
    return g, c, top_idx


def _mirror_descent(a_init, deadline, rng, k_top=5, lr_scale=1.0, use_nesterov=False):
    """Mirror descent with exact subgradient and Polyak averaging."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    # Running average (Polyak)
    avg_z = z.copy()
    
    t = 0
    max_iters = 50000
    while t < max_iters and time.monotonic() < deadline - 2.0:
        t += 1
        a_curr = _softmax_fn(z)
        g, c, _ = _compute_subgradient(a_curr, k_top)
        
        # Scale gradient for the objective 2n*max(c)/sum^2 (with sum=1, it's 2n*max(c))
        g = g * 2.0 * n
        
        # Adaptive step size
        eta = lr_scale / (np.sqrt(t + 1.0) * np.linalg.norm(g) + 1e-10)
        eta = min(eta, 0.1)
        
        if use_nesterov:
            # Nesterov: evaluate gradient at lookahead point
            # Simplified: just use momentum
            pass
        
        z = z - eta * g
        z = np.clip(z, -80, 80)
        z = z - z.mean()
        
        # Update running average
        avg_z = avg_z + (z - avg_z) / t
        
        # Check improvement every few iterations
        if t % 5 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            # Also check the Polyak average
            a_avg = _softmax_fn(avg_z)
            sc_avg = _score_norm_fn(a_avg)
            if np.isfinite(sc_avg) and sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
    
    return best_a, best_score


def _nesterov_mirror_descent(a_init, deadline, rng, k_top=5, lr_scale=1.0):
    """Nesterov-accelerated mirror descent."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    y = z.copy()  # Nesterov momentum variable
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    t = 0
    max_iters = 30000
    while t < max_iters and time.monotonic() < deadline - 2.0:
        t += 1
        # Nesterov step: y = z + (t-1)/(t+2) * (z - z_prev)
        # Simplified: use momentum
        a_y = _softmax_fn(y)
        g, c, _ = _compute_subgradient(a_y, k_top)
        g = g * 2.0 * n
        
        eta = lr_scale / (np.sqrt(t + 1.0) * (np.linalg.norm(g) + 1e-10))
        eta = min(eta, 0.08)
        
        # Gradient step on y
        y = y - eta * g
        y = np.clip(y, -80, 80)
        y = y - y.mean()
        
        # Update z with Nesterov momentum
        momentum = (t - 1.0) / (t + 2.0)
        z_new = y + momentum * (y - z)
        z_new = np.clip(z_new, -80, 80)
        z_new = z_new - z_new.mean()
        z = z_new
        
        if t % 10 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score


def _build_conv_matrix(n, f):
    """Build sparse matrix A where A @ g = convolve(f, g). Shape: (2n-1, n)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_fw_step(f, M, timeout=120.0):
    """Solve LP: max sum(g) s.t. convolve(f, g) <= M, g >= 0. Use PDLP first."""
    n = len(f)
    try:
        A = _build_conv_matrix(n, f)
        g = cp.Variable(n, non_neg=True)
        constraints = [A @ g <= M]
        objective = cp.Maximize(cp.sum(g))
        prob = cp.Problem(objective, constraints)
        
        t0 = time.monotonic()
        # Try PDLP first (fastest for large sparse LPs)
        solvers = [cp.PDLP, cp.ECOS, cp.GLPK, cp.CBC]
        for solver in solvers:
            if time.monotonic() - t0 > timeout:
                break
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    if g.value is not None:
                        return np.maximum(g.value, 0.0)
            except Exception:
                continue
        
        if g.value is not None:
            return np.maximum(g.value, 0.0)
    except Exception:
        pass
    return None


def _lp_fw_method(f_init, deadline, rng, max_iters=20):
    """LP-based Frank-Wolfe method with PDLP."""
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s * np.sqrt(2.0 * n)
    
    c = _full_convolve(f, f)
    M = float(c.max())
    best_f = f.copy()
    best_M = M
    best_score = 2.0 * n * best_M / (2.0 * n)
    
    print(f"  FW start: M={M:.10f}, score={best_score:.10f}, n={n}")
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 60.0:
            break
        
        # Adaptive step size (diminishing)
        t = 2.0 / (it + 2.0)
        t = min(t, 0.4)
        
        g = _lp_fw_step(f, M, timeout=90.0)
        if g is None:
            # Try smaller M (more relaxed)
            M_relaxed = M * 1.01
            g = _lp_fw_step(f, M_relaxed, timeout=60.0)
            if g is None:
                break
        
        s_g = g.sum()
        if s_g < 1e-15:
            break
        g = g * (s / s_g)
        
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_fnew = f_new.sum()
        if s_fnew < 1e-15:
            break
        f_new = f_new / s_fnew * s
        
        c_new = _full_convolve(f_new, f_new)
        M_new = float(c_new.max())
        score_new = 2.0 * n * M_new / (s * s)
        
        if M_new < best_M * 1.0005:
            best_M = M_new
            best_f = f_new.copy()
            best_score = score_new
            print(f"  FW iter {it}: M={M_new:.10f}, score={best_score:.10f}, t={t:.3f}")
        
        f = f_new.copy()
        M = M_new
    
    best_f_norm = best_f / best_f.sum()
    return best_f_norm, best_score


def _run_lbfgs_continuation(a_init, deadline, rng, beta_scale=1.0, max_wall_time=None, min_beta_frac=0.0001):
    """L-BFGS with logsumexp continuation - aggressive version."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    c0 = _full_convolve(a_init, a_init)
    max_c0 = float(c0.max())
    
    # More levels, more aggressive schedule
    beta_fracs = [0.1, 0.05, 0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001, min_beta_frac]
    
    if max_wall_time is not None:
        per_level = max(2.0, max_wall_time / len(beta_fracs))
    else:
        per_level = 30.0
    
    for i, beta_frac in enumerate(beta_fracs):
        if time.monotonic() > deadline - 3.0:
            break
        level_deadline = min(deadline, time.monotonic() + per_level)
        beta = beta_frac * max_c0 * beta_scale
        
        def obj(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300 or not np.isfinite(s_):
                return 2.0 * n * cm_
            return 2.0 * n * (cm_ + beta * np.log(s_))
        
        def grad_fn(z_):
            a_ = _softmax_fn(z_)
            c_ = _full_convolve(a_, a_)
            cm_ = float(c_.max())
            c_shifted = (c_ - cm_) / beta
            e_ = np.exp(c_shifted)
            s_ = e_.sum()
            if s_ < 1e-300:
                w = np.zeros_like(c_)
                w[int(np.argmax(c_))] = 1.0
            else:
                w = e_ / s_
            W = float(np.dot(w, c_))
            a_rev = a_[::-1]
            w_conv = _full_convolve(w, a_rev)
            h = w_conv[n - 1:2 * n - 1]
            g = 4.0 * n * a_ * (h - W)
            if not np.all(np.isfinite(g)):
                return np.zeros_like(z_)
            return g
        
        try:
            result = minimize(
                obj, z.copy(), jac=grad_fn,
                method='L-BFGS-B',
                options={'maxiter': 3000, 'ftol': 1e-22, 'gtol': 1e-14,
                         'maxls': 50, 'maxfun': 15000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                sc_new = _score_norm_fn(a_new)
                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    
    return best_a, best_score


def _peak_surgery(a_init, deadline, rng, n_trials=50000):
    """Advanced peak surgery: identify responsible elements and redistribute."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_sc = _score_norm_fn(a)
    
    for it in range(n_trials):
        if time.monotonic() > deadline - 1.0:
            break
        
        c = _full_convolve(a, a)
        max_c = float(c.max())
        max_lag = int(np.argmax(c))
        
        # Find all elements contributing to the top peak
        i_min = max(0, max_lag - (n - 1))
        i_max = min(n - 1, max_lag)
        
        if i_min > i_max:
            continue
        
        # Compute "responsibility" of each element for the top peak
        # Element i contributes a[i] * a[max_lag - i] to c[max_lag]
        partners = max_lag - np.arange(i_min, i_max + 1)
        contrib = a[i_min:i_max + 1] * a[partners]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            continue
        
        # Find the most "responsible" element (highest contribution)
        idx_rel = int(np.argmax(contrib))
        idx = i_min + idx_rel
        
        # Strategy A: Reduce the most responsible element
        if it % 4 == 0:
            a_trial = a.copy()
            delta = a_trial[idx] * rng.uniform(0.3, 0.7)
            a_trial[idx] -= delta
            a_trial[idx] = max(0.0, a_trial[idx])
            
            # Add to a "safe" position (low contribution to top peaks)
            # Compute safety score: how much would adding here increase the max?
            top_k_lags = np.argsort(c)[-min(10, len(c)):]
            safety = np.zeros(n)
            for kl in top_k_lags:
                kl = int(kl)
                lo = max(0, kl - (n - 1))
                hi = min(n - 1, kl)
                if lo <= hi:
                    safety[lo:hi + 1] += a[kl - hi:kl - lo + 1][::-1]
            
            # Pick from the safest positions (lowest safety score)
            safe_threshold = np.percentile(safety, 20)
            safe_mask = safety <= safe_threshold
            if safe_mask.sum() > 0:
                safe_indices = np.where(safe_mask)[0]
                add_idx = int(safe_indices[rng.integers(len(safe_indices))])
                a_trial[add_idx] += delta
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 1e-10:
                    a_trial /= s
                    sc = _score_norm_fn(a_trial)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
        
        # Strategy B: Spread reduction over top contributors
        elif it % 4 == 1:
            top_contrib_idx = np.argsort(contrib)[-min(5, len(contrib)):]
            a_trial = a.copy()
            for tc_idx in top_contrib_idx:
                tc_pos = i_min + int(tc_idx)
                delta = a_trial[tc_pos] * rng.uniform(0.1, 0.3)
                a_trial[tc_pos] -= delta
                a_trial[tc_pos] = max(0.0, a_trial[tc_pos])
                # Add to random position
                add_idx = int(rng.integers(n))
                a_trial[add_idx] += delta
            a_trial = np.maximum(a_trial, 1e-15)
            s = a_trial.sum()
            if s > 1e-10:
                a_trial /= s
                sc = _score_norm_fn(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
        
        # Strategy C: Block perturbation
        elif it % 4 == 2:
            block_size = int(rng.integers(3, min(32, n // 4)))
            start = int(rng.integers(0, n - block_size))
            a_trial = a.copy()
            mult = rng.uniform(0.7, 1.3, size=block_size)
            a_trial[start:start + block_size] *= mult
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
        
        # Strategy D: Sinusoidal perturbation (smooth global change)
        else:
            a_trial = a.copy()
            freq = rng.uniform(1.0, 8.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.1)
            t_arr = np.linspace(0, 1, n)
            pert = 1.0 + amp * np.cos(2 * np.pi * freq * t_arr + phase)
            a_trial *= pert
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial /= a_trial.sum()
            sc = _score_norm_fn(a_trial)
            if sc < best_sc:
                best_sc = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
    
    return best_a, best_sc


def _gradient_descent_smoothed(a_init, deadline, rng, lr=0.01, k_top=3):
    """Simple gradient descent on the smoothed objective with adaptive step."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    max_iters = 100000
    while t < max_iters and time.monotonic() < deadline - 2.0:
        t += 1
        a_curr = _softmax_fn(z)
        c = _full_convolve(a_curr, a_curr)
        k_actual = min(k_top, len(c))
        top_idx = np.argsort(c)[-k_actual:]
        
        # Compute gradient of max via top-k averaging
        g_raw = np.zeros(n)
        for kv in top_idx:
            kv = int(kv)
            lo = max(0, kv - (n - 1))
            hi = min(n - 1, kv)
            if lo <= hi:
                g_raw[lo:hi + 1] += a_curr[kv - hi:kv - lo + 1][::-1]
        g_raw *= 4.0 * n / k_actual
        
        # Adam update in log-space
        m = b1 * m + (1 - b1) * g_raw
        v = b2 * v + (1 - b2) * (g_raw * g_raw)
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        # Adaptive learning rate
        lr_curr = lr / (1.0 + 0.01 * t)
        z = z - lr_curr * mh / (np.sqrt(vh) + eps)
        z = np.clip(z, -70, 70)
        z = z - z.mean()
        
        if t % 20 == 0:
            a_new = _softmax_fn(z)
            sc = _score_norm_fn(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
    
    return best_a, best_score


def _upscale_and_refine(a_small, n_target, deadline, rng, refine_time=60.0):
    """Upscale a small sequence to target size and refine."""
    a_up = _cubic_interp_profile(a_small, n_target)
    sc_up = _score_norm_fn(a_up)
    print(f"  Upscaled to n={n_target}: score={sc_up:.10f}")
    
    if time.monotonic() < deadline - max(10, refine_time):
        dl = min(deadline - 10, time.monotonic() + refine_time)
        a_best, sc_best = _run_lbfgs_continuation(a_up, dl, rng,
                                                    max_wall_time=refine_time,
                                                    beta_scale=0.3,
                                                    min_beta_frac=0.00005)
        if sc_best < sc_up:
            print(f"  After refinement n={n_target}: score={sc_best:.10f}")
            return a_best, sc_best
    
    return a_up, sc_up


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-scale optimizer combining:
    - Diverse initialization
    - L-BFGS logsumexp continuation
    - Mirror descent with exact subgradients
    - LP Frank-Wolfe with PDLP
    - Multi-scale upscaling
    - Peak surgery
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))
    
    best_seq = None
    best_score = np.inf
    
    def update_best(p, sc, label=""):
        nonlocal best_seq, best_score
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_seq = np.asarray(p, dtype=np.float64).copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}]")
    
    # --- Get parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        parent = None
    
    if parent is not None:
        n_parent = len(parent)
        sc_parent = _score_norm_fn(parent)
        print(f"Parent: n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "parent")
    else:
        n_parent = 4096
        parent = _make_triangular_weight(n_parent)
        sc_parent = _score_norm_fn(parent)
        print(f"No parent, using triangular n={n_parent}, score={sc_parent:.10f}")
        update_best(parent, sc_parent, "init")
    
    # --- Phase 0: Diverse initialization at n=4096 (40s) ---
    phase0_end = min(deadline, t_start + 40.0)
    n_small = 4096
    
    profiles = [
        _make_triangular_weight(n_small),
        _make_linear_profile(n_small, 0.5),
        _make_linear_profile(n_small, 1.0),
        _make_linear_profile(n_small, 2.0),
        _make_quadratic_profile(n_small, 0.5, 0.3),
        _make_quadratic_profile(n_small, 1.0, -0.2),
        _make_bimodal_profile(n_small, 0.4, 0.15),
        _make_bimodal_profile(n_small, 0.5, 0.2),
        _make_w_profile(n_small, 2, 0.3),
        _make_w_profile(n_small, 2, 0.5),
        _make_w_profile(n_small, 3, 0.3),
        _make_bspline_profile(n_small, 3),
        _make_bspline_profile(n_small, 4),
        _make_bspline_profile(n_small, 5),
        _make_raised_cosine(n_small, 1.0),
        _make_raised_cosine(n_small, 2.0),
    ]
    
    candidates = []
    for idx, p in enumerate(profiles):
        if time.monotonic() > phase0_end:
            break
        sc = _score_norm_fn(p)
        candidates.append((sc, p.copy(), f"prof{idx}"))
        update_best(p, sc, f"prof{idx}")
    
    # Also add parent (resampled to n_small)
    if parent is not None and len(parent) != n_small:
        p_resample = _interp_profile(parent, n_small)
        sc = _score_norm_fn(p_resample)
        candidates.append((sc, p_resample, "parent_resample"))
        update_best(p_resample, sc, "parent_resample")
    elif parent is not None:
        candidates.append((_score_norm_fn(parent), parent.copy(), "parent"))
    
    candidates.sort(key=lambda x: x[0])
    print(f"Phase 0 done: best={best_score:.10f}")
    
    # --- Phase 1: L-BFGS at n=4096 (120s) ---
    phase1_end = min(deadline, t_start + 160.0)
    start_p = candidates[0][1].copy()
    print(f"Phase 1: L-BFGS at n={len(start_p)}, start score={candidates[0][0]:.10f}")
    a_p1, sc_p1 = _run_lbfgs_continuation(start_p, phase1_end, rng,
                                            max_wall_time=110.0, beta_scale=1.0)
    update_best(a_p1, sc_p1, "lbfgs_n4096")
    
    # --- Phase 2: Mirror descent at n=4096 (60s) ---
    phase2_end = min(deadline, t_start + 220.0)
    if best_seq is not None and len(best_seq) == n_small and time.monotonic() < phase2_end:
        print("Phase 2: Mirror descent at n=4096")
        a_p2, sc_p2 = _mirror_descent(best_seq, phase2_end, rng, k_top=5, lr_scale=0.5)
        update_best(a_p2, sc_p2, "mirror_n4096")
    
    # --- Phase 3: L-BFGS at n=8192 (120s) ---
    phase3_end = min(deadline, t_start + 340.0)
