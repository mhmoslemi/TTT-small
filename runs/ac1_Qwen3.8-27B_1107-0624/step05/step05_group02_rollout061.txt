import numpy as np
import time

def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)

def _convolve_fft(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]

def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    c = _convolve_fft(a, a)
    return float(2.0 * n * c.max() / (s * s))

def _compute_gradient(a, beta):
    """Compute gradient of 2*n*max(conv(a,a)) using softmax approximation.
    Returns gradient w.r.t. a (before normalization)."""
    n = len(a)
    c = _convolve_fft(a, a)
    # Softmax weights
    c_shifted = (c - c.max()) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * conv(w, a[::-1])[n-1 : 2n-1]
    N = _next_pow2(len(w) + n - 1)  # = next_pow2(3n-2)

    conv_result = np.fft.irfft(W * A_rev, N)
    g = 2.0 * conv_result[n-1 : n-1+n]
    return g * (2.0 * n)  # Multiply by 2n for the full objective (before /sum^2)

"""
FFT-based gradient optimization in log-space with adaptive softmax approximation
of max, multi-beta annealing, multi-restart, and multi-length exploration
to minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key innovations over previous approach:
- O(n log n) gradient computation via FFT (vs O(n^2) LP)
- Log-space parameterization for automatic non-negativity
- Adaptive beta annealing for smooth-to-sharp max approximation
- Frequency-domain insights for initialization and perturbation
"""
import numpy as np
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _convolve_fft(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    return np.fft.irfft(fa * fb, N)[:out_n]


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    c = _convolve_fft(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))"""
    n = len(a)
    c = _convolve_fft(a, a)
    return float(2.0 * n * c.max())


def _compute_grad_obj(a, beta):
    """
    Compute gradient of 2*n*max(conv(a,a)) w.r.t. a using softmax approximation.
    a should be normalized (sum=1). Returns gradient vector of same length as a.
    """
    n = len(a)
    c = _convolve_fft(a, a)
    c_max = c.max()
    c_shifted = (c - c_max) / beta
    e = np.exp(c_shifted)
    w = e / e.sum()
    # g[j] = 2 * conv(w, a[::-1])[n-1+j] for j=0..n-1
    # Full objective gradient: 2*n * g[j]
    N = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, N)
    A_rev = np.fft.rfft(a[::-1].copy(), N)
    conv_result = np.fft.irfft(W * A_rev, N)
    g = 2.0 * conv_result[n - 1: n - 1 + n]
    return 2.0 * n * g


def _softmax_safe(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -500, 500))
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _adam_logspace(z_init, betas, deadline, lr=0.05, max_update_norm=1.0,
                   check_every=20, patience=100000):
    """
    Adam optimization in log-space where a = softmax(z).
    Minimizes 2*n*max(conv(a,a)) with adaptive beta for softmax approximation.
    Returns (best_z, best_score).
    """
    n = len(z_init)
    z = z_init.copy().astype(np.float64)
    z = z - z.mean()

    best_score = float('inf')
    best_z = z.copy()
    no_improve = 0

    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    beta_idx = 0
    beta = betas[0]

    while time.monotonic() < deadline:
        t_step += 1
        a = _softmax_safe(z)

        # Periodic score check
        if t_step % check_every == 1:
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
                no_improve = 0
            else:
                no_improve += check_every

            # Announce improvements
            if int(sc * 1e8) % 10000 == 0 or t_step % (check_every * 50) == 1:
                print(f"    step={t_step}, score={sc:.10f}, beta={beta:.2e}, lr={lr:.4f}")

        # Compute gradient in a-space
        g_a = _compute_grad_obj(a, beta)

        if not np.all(np.isfinite(g_a)):
            break

        # Convert to z-space: g_z[j] = a[j] * (g_a[j] - dot(a, g_a))
        dot = float(np.dot(a, g_a))
        g_z = a * (g_a - dot)

        # Clamp gradient to prevent numerical issues
        g_z = np.clip(g_z, -1e4, 1e4)

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * g_z
        v_buf = b2 * v_buf + (1.0 - b2) * (g_z * g_z)
        mh = m_buf / (1.0 - b1 ** t_step)
        vh = v_buf / (1.0 - b2 ** t_step)

        update = lr * mh / (np.sqrt(vh) + eps)
        update_norm = float(np.linalg.norm(update))
        if update_norm > max_update_norm:
            update *= max_update_norm / update_norm

        z = z - update
        z = z - z.mean()
        z = np.clip(z, -100.0, 100.0)

        # Adaptive beta: decrease periodically
        if t_step % 5000 == 0 and beta_idx < len(betas) - 1:
            beta_idx += 1
            beta = betas[beta_idx]

        # Learning rate decay
        if t_step % 20000 == 0:
            lr = max(0.001, lr * 0.9)

        # Early stopping if stuck
        if no_improve > patience:
            break

    # Final score
    a_final = _softmax_safe(z)
    sc_final = _score_normalized(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_z = z.copy()

    return best_z, best_score


def _interp_to_length(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()


def _make_z_from_a(a):
    """Convert normalized sequence to log-space parameter."""
    z = np.log(np.maximum(a, 1e-15))
    z = z - z.mean()
    return z


def _make_betas(max_c, start_frac=1.0, end_frac=0.001, num=12):
    """Create geometric beta schedule from max_c * start_frac to max_c * end_frac."""
    fracs = np.geomspace(start_frac, end_frac, num)
    betas = [max_c * f for f in fracs]
    return [max(b, 1e-8) for b in betas]


def _perturb_z(z, sigma, rng):
    """Perturb log-space parameters."""
    z_p = z + sigma * rng.standard_normal(len(z))
    z_p = z_p - z_p.mean()
    return z_p

            else:
                # Random smooth mask
                mask = 1.0 + 0.5 * np.sin(2 * np.pi * (freqs * 3 + profile_idx))

            A_masked = A_fft * mask
            a_freq = np.fft.irfft(A_masked, N_fft)[:n_explore]
            a_freq = np.maximum(a_freq, 0)
            a_freq_sum = a_freq.sum()
            if a_freq_sum > 0.01:
                a_freq = a_freq / a_freq_sum
                best_score, best_seq = _try_update(best_score, best_seq, a_freq)
                z_freq = _make_z_from_a(a_freq)
                max_c_f = _convolve_fft(a_freq, a_freq).max()
                betas_f = _make_betas(max_c_f, 0.1, 0.001, 8)
                remaining = phase3_deadline - time.monotonic() - 20
                if remaining > 30:
                    z_rf, sc_rf = _adam_logspace(z_freq, betas_f,
                                                  time.monotonic() + remaining,
                                                  lr=0.05, max_update_norm=0.5)
                    a_rf = _softmax_safe(z_rf)
                    best_score, best_seq = _try_update(best_score, best_seq, a_rf)

    # === Phase 4: Larger n exploration ===
    print(f"\n=== Phase 4: Larger n exploration ===")
    phase4_deadline = min(deadline, _T_START + 1650)

    if best_seq is not None and time.monotonic() < phase4_deadline:
        for target_n in [4096, 8192, 16384]:
            if time.monotonic() > phase4_deadline - 60:
                break
            if len(best_seq) >= target_n:
                continue

            a_interp = _interp_to_length(best_seq, target_n)
            best_score, best_seq = _try_update(best_score, best_seq, a_interp)

            z_interp = _make_z_from_a(a_interp)
            max_c_i = _convolve_fft(a_interp, a_interp).max()
            betas_i = _make_betas(max_c_i, 0.05, 0.0005, 10)

            remaining = phase4_deadline - time.monotonic() - 30
            if remaining > 30:
                z_ri, sc_ri = _adam_logspace(z_interp, betas_i,
                                              time.monotonic() + remaining,
                                              lr=0.04, max_update_norm=0.5,
                                              check_every=20)
                a_ri = _softmax_safe(z_ri)
                best_score, best_seq = _try_update(best_score, best_seq, a_ri)

    # === Phase 5: Final fine-tuning ===
    print(f"\n=== Phase 5: Final fine-tuning ===")
    if best_seq is not None and time.monotonic() < deadline - 20:
        a_best = best_seq / best_seq.sum()
        z_best = _make_z_from_a(a_best)
        max_c_b = _convolve_fft(a_best, a_best).max()

        # Very fine-tuned run
        betas_final = _make_betas(max_c_b, 0.01, 0.0003, 8)
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            z_final, sc_final = _adam_logspace(z_best, betas_final,
                                                time.monotonic() + remaining,
                                                lr=0.005, max_update_norm=0.2,
                                                check_every=5)
            a_final = _softmax_safe(z_final)
            best_score, best_seq = _try_update(best_score, best_seq, a_final)

    # Final output
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        sc = _score(final)
        print(f"\nFINAL: score={sc:.10f}, n={len(final)}, time={time.monotonic()-_T_START:.1f}s")
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"Evaluator confirms: {sc_eval:.10f}")
        except Exception as ex:
            print(f"Eval check error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(4096) / 4096
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
