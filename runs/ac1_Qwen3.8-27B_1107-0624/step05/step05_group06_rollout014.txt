def _peak_shave(a, n_steps=10000, deadline=None):
    """
    Targeted peak-shaving: iteratively reduce the convolution peak
    by adjusting the most contributing coefficients.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(a)
    
    for step in range(n_steps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c, c_max, k_star = _conv_max_info(best_a)
        s = float(best_a.sum())
        
        # Compute gradient of c[k*] w.r.t. a
        # dc[k*]/da[i] = 2*a[k*-i] (when 0 <= k*-i < n)
        g = np.zeros(n)
        valid = np.arange(n)
        k_minus_i = k_star - valid
        mask = (k_minus_i >= 0) & (k_minus_i < n)
        g[mask] = 2.0 * best_a[k_minus_i[mask]]
        
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Adaptive step size
        # We want to reduce c[k*] by a small amount
        # c[k*] changes by approximately g @ delta
        # Let's use a fixed fraction of the current value
        lr = 0.01
        decay = 1.0 / (1.0 + step * 0.0005)
        
        # Update in log-space to maintain non-negativity
        z = np.log(np.maximum(best_a, 1e-15))
        z = z - lr * decay * (g / g_norm)
        z = np.clip(z, -50.0, 50.0)
        new_a = _softmax(z)
        
        # Ensure sum is preserved (softmax already does this)
        sc = _score(new_a)
        if sc < best_sc:
            best_a = new_a
            best_sc = sc
        else:
            # Reduce step size
            lr *= 0.7
            if lr < 1e-8:
                break
    
    return best_a, best_sc

    return _softmax(z)

Multi-stage optimizer combining variance-based smooth optimization,
L-BFGS with log-sum-exp smoothing, targeted peak-shaving, and
Fourier-domain restarts to minimize 2*n*max(conv(a,a))/sum(a)^2.

import numpy as np

from scipy.optimize import minimize


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _conv_and_max(a):
    c = _full_convolve(a, a)
    idx = int(np.argmax(c))
    return c, float(c[idx]), idx


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _lsmoobj(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return 2.0 * n * c_max
    return 2.0 * n * (c_max + beta * np.log(s))


def _lsmod_grad(z, beta, n):
    a = _softmax(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

def _var_grad(z, n):

    a = _softmax(z)
    c = _full_convolve(a, a)
    c_centered = c - c.mean()

    cross = _full_convolve(c_centered.astype(np.float64), a_rev)
    grad_a = (4.0 / (2 * n - 1)) * cross[n - 1:2 * n - 1]

def _peak_shave(a, n_steps, deadline, lr_init=0.02):
    """
    Targeted peak shaving using subgradient of the max convolution term.
    Uses backtracking for step size adaptation.
    """
    n = len(a)
    best_a = a.copy()
    best_sc = _score(best_a)
    lr = lr_init
    stall_count = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c, c_max, k_star = _conv_and_max(best_a)

        # Subgradient of max(conv) at k_star:
        # dc[ks]/da[i] = 2*a[ks-i] for valid indices
        g_a = np.zeros(n)
        k_mi = k_star - np.arange(n)
        mask = (k_mi >= 0) & (k_mi < n)
        g_a[mask] = 2.0 * best_a[k_mi[mask]]

        gn = np.linalg.norm(g_a)
        if gn < 1e-15:
            break
        g_a = g_a / gn

        # In log-space: z_new = z - lr * g_a (softmax Jacobian approximately identity for small lr)
        z = np.log(np.maximum(best_a, 1e-15))
        z_new = z - lr * g_a
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_a = a_new
            best_sc = sc_new
            stall_count = 0
            lr = min(lr * 1.05, 0.1)
        else:
            stall_count += 1
            lr *= 0.6
            if stall_count > 200:
                break

    return best_a, best_sc


def _multi_peak_shave(a, n_steps, deadline, n_peaks=5):
    """Shave multiple peaks simultaneously using top-K weighted subgradient."""
    n = len(a)
    best_a = a.copy()
    best_sc = _score(best_a)
    lr = 0.015
    stall_count = 0

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        c, c_max, _ = _conv_and_max(best_a)

        # Top K peaks
        top_k = np.argsort(c)[-n_peaks:]
        top_k_vals = c[top_k]
        # Weight by how close to max (exponential)
        weights = np.exp(top_k_vals / (c_max * 0.01) - np.max(top_k_vals) / (c_max * 0.01))
        weights /= weights.sum()

        # Combined gradient
        g_a = np.zeros(n)
        for k_pos, w in zip(top_k, weights):
            k_mi = int(k_pos) - np.arange(n)
            mask = (k_mi >= 0) & (k_mi < n)
            g_a[mask] += w * 2.0 * best_a[k_mi[mask]]

        gn = np.linalg.norm(g_a)
        if gn < 1e-15:
            break
        g_a = g_a / gn

        z = np.log(np.maximum(best_a, 1e-15))
        z_new = z - lr * g_a
        z_new = np.clip(z_new, -50.0, 50.0)
        z_new = z_new - z_new.mean()
        a_new = _softmax(z_new)

        sc_new = _score(a_new)
        if sc_new < best_sc - 1e-14:
            best_a = a_new
            best_sc = sc_new
            stall_count = 0
            lr = min(lr * 1.03, 0.08)
        else:
            stall_count += 1
            lr *= 0.65
            if stall_count > 300:
                break

    return best_a, best_sc


def _fourier_perturb(a, rng, n_modes=3, amp=0.05):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    t = np.arange(n)
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 2)))
        a_i = amp * rng.uniform(0.5, 2.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        z = z + a_i * np.cos(2.0 * np.pi * freq * t / n + phase)
    z = z - z.mean()
    return _softmax(z)


def _gauss_perturb(a, rng, sigma=0.04):
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(n)
    z = z - z.mean()
    return _softmax(z)


def _block_perturb(a, rng, n_blocks=4, block_frac=0.1, sigma=0.1):
    """Perturb random blocks of the sequence."""
    n = len(a)
    z = np.log(np.maximum(a, 1e-15))
    block_size = max(10, int(n * block_frac))
    for _ in range(n_blocks):
        start = int(rng.integers(0, n - block_size))
        end = start + block_size
        z[start:end] += sigma * rng.standard_normal(block_size)
    z = z - z.mean()
    return _softmax(z)


def _pipeline_v1(a_start, deadline, min_remaining=30.0):
    """Pipeline: variance optimization -> L-SMO -> peak shave."""
    n = len(a_start)

    # Step 1: Variance minimization (smooth, helps escape local min)
    if time.monotonic() < deadline - min_remaining - 60:
        a_var = _var_lbfgs(a_start, n, deadline, min_remaining + 60)
        if _score(a_var) < _score(a_start):
            a_start = a_var

    # Step 2: L-SMO with aggressive beta schedule
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    betas = max_c * np.array([0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005])
    a_lsmo = _lbfgs_lsmo(a_start, betas, n, deadline, min_remaining + 30)

    # Step 3: Peak shaving
    if time.monotonic() < deadline - min_remaining:
        a_shaved, _ = _peak_shave(a_lsmo, 15000, deadline - min_remaining)
        if _score(a_shaved) < _score(a_lsmo):
            a_lsmo = a_shaved

    return a_lsmo


def _pipeline_v2(a_start, deadline, min_remaining=30.0):
    """Pipeline: multi-peak shave -> L-SMO refine -> peak shave."""
    n = len(a_start)

    # Step 1: Multi-peak shave
    if time.monotonic() < deadline - min_remaining - 60:
        a_mp, _ = _multi_peak_shave(a_start, 10000, deadline - min_remaining - 60, n_peaks=3)
        if _score(a_mp) < _score(a_start):
            a_start = a_mp

    # Step 2: L-SMO fine tuning
    c_cur = _full_convolve(a_start, a_start)
    max_c = float(c_cur.max())
    betas = max_c * np.array([0.005, 0.002, 0.001, 0.0005, 0.0002])
    a_lsmo = _lbfgs_lsmo(a_start, betas, n, deadline, min_remaining + 20)

    # Step 3: Single peak shave
    if time.monotonic() < deadline - min_remaining:
        a_shaved, _ = _peak_shave(a_lsmo, 10000, deadline - min_remaining)
        if _score(a_shaved) < _score(a_lsmo):
            a_lsmo = a_shaved

    return a_lsmo


def _make_profile(n, kind="uniform"):
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "tapered":
        t = np.linspace(0, 1, n)
        a = np.sin(np.pi * t) ** 0.5
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "bell":
        t = np.linspace(0, 1, n)
        a = np.exp(-4.0 * (t - 0.5) ** 2)
        a = np.maximum(a, 1e-10)
        return a / a.sum()
    elif kind == "plateau":
        t = np.linspace(0, 1, n)
        a = np.ones(n)
        a[:n//10] *= 0.5
        a[-n//10:] *= 0.5
        return a / a.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-stage search: variance optimization, L-SMO refinement,
    peak shaving, and diversified restarts.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(int(seed) % (2**63 - 1))

    best_seq = None
    best_score = np.inf

    def update_best(p, sc=None, label=""):
        nonlocal best_seq, best_score
        if p is None:
            return
        p = np.asarray(p, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        s = float(p.sum())
        if s < 0.01:
            return
        if sc is None:
            sc = _score(p)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_seq = p.copy()
            best_score = float(sc)
            print(f"  NEW BEST: {best_score:.10f} (n={len(p)}) [{label}] "
                  f"delta={prev-best_score:.2e} t={time.monotonic()-t_start:.1f}s")

    # --- Load parent ---
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            if float(parent_raw.sum()) > 0.01:
                parent = parent_raw
    except Exception:
        parent = None

    if parent is not None:
        n_main = len(parent)
        a0 = parent / parent.sum()
        sc0 = _score(a0)
        print(f"Parent loaded: n={n_main}, score={sc0:.10f}")
        update_best(a0, sc0, "parent")
    else:
        n_main = 2048
        a0 = _make_profile(n_main, "uniform")
        print(f"No parent. Using uniform n={n_main}, score={_score(a0):.10f}")
        update_best(a0, None, "init")

    # --- Phase 1: Variance + L-SMO from parent ---
    print(f"\n[Phase 1] Variance + L-SMO main (t={time.monotonic()-t_start:.1f}s)")
    a1 = _pipeline_v1(a0, deadline, min_remaining=400.0)
    update_best(a1, None, "pipeline_v1")

    # --- Phase 2: Multi-peak shave + L-SMO refine ---
    print(f"[Phase 2] Multi-peak + L-SMO refine (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 300:
        a2 = _pipeline_v2(best_seq, deadline, min_remaining=300.0)
        update_best(a2, None, "pipeline_v2")

    # --- Phase 3: Extended peak shaving on best ---
    print(f"[Phase 3] Extended peak shave (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 200:
        ps_deadline = min(deadline - 200, time.monotonic() + 250)
        a3, _ = _peak_shave(best_seq.copy(), 30000, ps_deadline, lr_init=0.015)
        update_best(a3, None, "peak_shave_ext")

    # --- Phase 4: Multi-peak shave extended ---
    print(f"[Phase 4] Multi-peak shave (t={time.monotonic()-t_start:.1f}s)")
    if time.monotonic() < deadline - 150:
        mp_deadline = min(deadline - 150, time.monotonic() + 200)
        a4, _ = _multi_peak_shave(best_seq.copy(), 20000, mp_deadline, n_peaks=5)
        update_best(a4, None, "multi_peak")

    # --- Phase 5: Diversified restarts ---
    print(f"[Phase 5] Diversified restarts (t={time.monotonic()-t_start:.1f}s)")
    n_restarts = min(8, max(1, int((deadline - time.monotonic() - 40) / 100)))

    for i in range(n_restarts):
        if time.monotonic() > deadline - 40:
            break

        if i == 0:
            a_pert = _fourier_perturb(best_seq, rng, n_modes=2, amp=0.02)
        elif i == 1:
            a_pert = _gauss_perturb(best_seq, rng, sigma=0.02)
        elif i == 2:
            a_pert = _block_perturb(best_seq, rng, n_blocks=3, block_frac=0.05, sigma=0.08)
        elif i == 3:
            a_pert = _fourier_perturb(best_seq, rng, n_modes=4, amp=0.04)
        elif i == 4:
            a_pert = _block_perturb(best_seq, rng, n_blocks=5, block_frac=0.1, sigma=0.12)
        elif i == 5:
            a_pert = _gauss_perturb(best_seq, rng, sigma=0.05)
        elif i == 6:
            a_pert = _fourier_perturb(best_seq, rng, n_modes=1, amp=0.015)
        else:
            a_pert = _block_perturb(best_seq, rng, n_blocks=2, block_frac=0.03, sigma=0.06)

        # Quick pipeline on perturbed
        remaining_for_restart = deadline - 40 - time.monotonic()
        if remaining_for_restart < 20:
            break

        a_r = _pipeline_v2(a_pert, deadline, min_remaining=40.0)
        update_best(a_r, None, f"restart_{i}")

    # --- Phase 6: Final fine optimization ---
    print(f"[Phase 6] Final fine optimization (t={time.monotonic()-t_start:.1f}s)")
    if best_seq is not None and time.monotonic() < deadline - 5:
        n = len(best_seq)
        c_b = _full_convolve(best_seq, best_seq)
        mb = float(c_b.max())

        # Very fine L-SMO
        fine_betas = mb * np.array([0.0001, 0.00005, 0.00002])
        z = np.log(np.maximum(best_seq, 1e-15))
        z = z - z.mean()

        for fb in fine_betas:
            if time.monotonic() > deadline - 3:
                break
            max_iter = min(2000, max(100, int((deadline - 3 - time.monotonic()) / 0.01)))

            def obj_f(z):
                return _lsmoobj(z, fb, n)

            def jac_f(z):
                return _lsmod_grad(z, fb, n)

            z_new = _lbfgs_run(z, obj_f, jac_f, n, max_iter, 1e-16)
            a_f = _softmax(z_new)
            sc_f = _score(a_f)
            if sc_f < best_score:
                update_best(a_f, sc_f, "final_lsmo")
                z = z_new

        # Final peak shave
        if time.monotonic() < deadline - 2:
            a_fin, _ = _peak_shave(best_seq.copy(), 5000, deadline - 1, lr_init=0.008)
            update_best(a_fin, None, "final_shave")

    # --- Final validation ---
    print(f"\n{'=' * 60}")
    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256
        sc_final = _score(final)
        print(f"FINAL: score={sc_final:.10f}, n={len(final)}, time={time.monotonic()-t_start:.1f}s")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256
    print(f"FALLBACK: score={_score(fallback):.10f}")
    return [float(x) for x in fallback]
