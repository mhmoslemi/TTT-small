```python
def subgradient_fista(a_init, deadline, max_iter=100000):
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = _score_norm_fn(a)
    
    # Polyak step size: need upper bound on最优 value
    # f* >= 1/(2n-1) (flat lower bound)
    # f* <= f(a_0) (initial value)
    f_lower = 1.0 / (2*n - 1)
    f_upper = _score_norm_fn(a_init)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(y, y)
        k_star = c.argmax()
        M = c[k_star]
        
        # Subgradient of max(c) at y: 2*y[k_star - j] for j in [0, n-1]
        # i.e., grad[j] = 2*y[k_star - j] if k_star - j in [0, n-1]
        grad = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] = 2.0 * y[k_star - j_lo:k_star - j_lo + (j_hi - j_lo + 1)]
        
        # Polyak step size
        gap = M - f_lower
        if gap < 1e-15:
            break
        lr = gap / (grad @ grad + 1e-15)
        lr = min(lr, 1.0)  # cap step size
        
        # Gradient step
        z = y - lr * grad
        
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = _project_simplex(z)
        
        # Fista update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (t / t_new - 1.0) * (a_new - a)
        a = a_new
        t = t_new
        
        # Track best
        val = _score_norm_fn(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        
        # Also try y
        val_y = _score_norm_fn(_project_simplex(y))
```

```python
"""
Multi-phase optimization for minimum peak autocorrelation (non-negative sequences).
Uses p-norm continuation with L-BFGS, subgradient with simplex projection,
and coordinate descent for final polish. Targets score < 1.5027.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    """Return 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    """Full score: 2n * max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _pnorm_obj(z, p, n):
    """p-norm of convolution (smooth surrogate for max)."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-300:
        return 0.0
    u = c / M
    Sp = float(np.sum(u ** p))
    return float(2.0 * n * M * Sp ** (1.0 / p))


def _pnorm_grad(z, p, n):
    """Gradient of p-norm objective w.r.t. z."""
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-300:
        return np.zeros_like(z)
    u = c / M
    u_p = u ** p
    Sp = float(np.sum(u_p))
    if Sp < 1e-300:
        return np.zeros_like(z)
    # w[k] = u[k]^(p-1) / Sp^(1-1/p)
    w = (u ** (p - 1)) / (Sp ** (1.0 - 1.0 / p))
    # grad_a[j] = 2 * sum_k w[k] * a[k-j]
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    # Convert to z-space
    Wa = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _greedy_init(n):
    """Binomial-type initialization."""
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype='hann'):
    t = np.linspace(0, 1, n)
    if etype == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
    elif etype == 'hann_inv':
        a = 0.5 * (1 - np.cos(2 * np.pi * t))
    elif etype == 'raised_cos':
        a = np.cos(np.pi * (t - 0.5)) ** 2
    elif etype == 'parab':
        a = 1 - (2 * t - 1) ** 2
    elif etype == 'parab4':
        a = (1 - (2 * t - 1) ** 2) ** 2
    elif etype == 'flat_top':
        a = np.where(np.abs(t - 0.5) < 0.3, 1.0, 0.5 * (1 + np.cos(np.pi * (np.abs(t - 0.5) - 0.3) / 0.2)))
    elif etype == 'gauss':
        a = np.exp(-5 * (t - 0.5) ** 2)
    elif etype == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()

    """L-BFGS with p-norm continuation."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)

    if p_list is None:
        p_list = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384, 32768]

    for p in p_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _pnorm_obj, z, args=(p, n),
                method='L-BFGS-B', jac=_pnorm_grad,
                options={'maxiter': maxiter_per_p, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxls': 100, 'maxfun': 10000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score

    # Lower bound for max(c) when sum(a)=1: 1/(2n-1)
    f_lower = 1.0 / (2 * n - 1)
    f_upper = float(c.max()) if (c := _full_convolve(a, a)).max() > 0 else _score_norm_fn(a) / (2.0 * n)
    
    # Averaging
    avg_a = a.copy()
    count = 0
    
    no_improve = 0
    check_interval = 50
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        c = _full_convolve(y, y)
        k_star = int(c.argmax())
        M = float(c[k_star])
        
        # Subgradient: g[j] = 2 * y[k_star - j] (when k_star - j in [0, n-1])
        g = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] = 2.0 * y[k_star - j_lo:k_star - j_lo + (j_hi - j_lo + 1)]
        
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-30:
            break
        
        # Polyak step size
        gap = M - f_lower
        if gap < 1e-15:
            break
        lr = gap / g_norm_sq
        lr = min(lr, 0.5)  # cap
        
        # Step
        z = y - lr * g
        a_new = _project_simplex(z)
        
        # Fista-like momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_new + (t / t_new - 1.0) * (a_new - a)
        y = _project_simplex(y)
        a = a_new
        t = t_new
        
        # Running average
        count += 1
        avg_a = avg_a * (1 - 1.0/count) + a_new / count
        
        if it % check_interval == 0:
            sc = _score_norm_fn(a)
            sc_avg = _score_norm_fn(avg_a) if count > 10 else np.inf
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            elif sc_avg < best_score - 1e-14:
                best_score = sc_avg
                best_a = avg_a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 40:
                # Restart with perturbation
                y = a + 0.01 * np.random.randn(n)
                y = _project_simplex(y)
                t = 1.0
                no_improve = 0
    
    return best_a, best_score


def _coordinate_descent(a_init, deadline, max_rounds=10000):
    """Coordinate descent: adjust pairs of coordinates."""
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    rng = np.random.default_rng(12345)
    
    c = _full_convolve(a, a)
    M = float(c.max())
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        # Pick a random pair
        j = rng.integers(n)
        i = rng.integers(n)
        if i == j:
            continue
        
        # Try moving mass from j to i
        # a[j] -= eps, a[i] += eps
        # Change in c[k]:
        # c[k] changes by: -eps*a[k-j] + eps*a[k-i] - eps*a[j-k] + eps*a[i-k] (approximately)
        # For small eps, delta_c[k] = eps * (a[k-i+... ] - a[k-j+...])
        
        # Actually, let's just try a few epsilon values
        eps_candidates = [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]
        
        improved = False
        for eps in eps_candidates:
            if a[j] + eps < 0:
                continue
            a[j] += eps
            a[i] -= eps
            c_new = _full_convolve(a, a)
            M_new = float(c_new.max())
            if M_new < M - 1e-14:
                M = M_new
                c = c_new
                improved = True
                break
            else:
                a[j] -= eps
                a[i] += eps
        
        if not improved:
            # Try larger search
            pass
        
        if rnd % 500 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score


def _lp_direction(a, M, K=50):
    """Solve LP to find a direction that reduces peak convolution."""
    n = len(a)
    c = _full_convolve(a, a)
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:K]
    
    # Build constraint matrix
    A_mat = np.zeros((len(active), n))
    for row, k in enumerate(active):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            A_mat[row, i_lo:i_hi + 1] = a[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
    
    b_vec = np.full(len(active), M * 1.001)
    
    try:
        res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 20})
        if res.status == 0:
            b_new = res.x
            sum_b = b_new.sum()
            if sum_b > 1e-15:
                return b_new * (a.sum() / sum_b)
    except Exception:
        pass
    return None
```

```python
"""
Multi-phase optimization minimizing 2*n*max(conv(a,a))/sum(a)^2 for non-negative a.
Uses p-norm continuation with L-BFGS, subgradient with Polyak steps and simplex
projection, LP direction finding, and coordinate descent. Supports multiple
sequence lengths (n=8192, 16384) for improved attack surface.
"""
import numpy as np
from scipy.optimize import minimize, linprog
import time


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _score_norm_fn(a):
    n = len(a)
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max())


def _score_fn(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax_fn(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _pnorm_obj(z, p, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-300:
        return 0.0
    u = c / M
    Sp = float(np.sum(u ** p))
    if Sp < 1e-300:
        return 0.0
    return float(2.0 * n * M * Sp ** (1.0 / p))


def _pnorm_grad(z, p, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    M = float(c.max())
    if M < 1e-300:
        return np.zeros_like(z)
    u = c / M
    if p <= 2:
        w = u / float(np.sum(u))
    else:
        u_p = u ** p
        Sp = float(np.sum(u_p))
        if Sp < 1e-300:
            return np.zeros_like(z)
        w = (u ** (p - 1)) / (Sp ** (1.0 - 1.0 / p))
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    grad_a = 2.0 * h
    Wa = float(np.dot(a, grad_a))
    grad_z = a * (grad_a - Wa)
    if not np.all(np.isfinite(grad_z)):
        return np.zeros_like(z)
    return grad_z


def _lse_obj(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return 0.0
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    lse = cm + beta * (max_s + float(np.log(np.sum(e))))
    return float(2.0 * n * lse)


def _lse_grad(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    cm = float(c.max())
    if cm < 1e-300:
        return np.zeros_like(z)
    shifted = (c - cm) / beta
    max_s = float(shifted.max())
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    a_rev = a[::-1]
    w_conv = _full_convolve(w, a_rev)
    h = w_conv[n - 1:2 * n - 1]
    W = float(np.dot(w, c))
    grad = 4.0 * n * a * (h - W)
    if not np.all(np.isfinite(grad)):
        return np.zeros_like(z)
    return grad


def _greedy_init(n):
    b = np.zeros(n)
    b[0] = 1.0
    for j in range(1, n):
        b[j] = b[j - 1] * (2.0 * j - 1.0) / (2.0 * j)
    a = b[::-1].copy()
    return a / a.sum()


def _parametric_init(n, etype):
    t = np.linspace(0, 1, n)
    if etype == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    elif etype == 'raised_cos':
        a = np.cos(np.pi * (t - 0.5)) ** 2
    elif etype == 'parab':
        a = 1.0 - (2.0 * t - 1.0) ** 2
    elif etype == 'parab4':
        a = (1.0 - (2.0 * t - 1.0) ** 2) ** 2
    elif etype == 'gauss':
        a = np.exp(-4.0 * (t - 0.5) ** 2)
    elif etype == 'flat_top':
        x = np.abs(t - 0.5) * 2.0
        a = np.where(x < 0.5, 1.0, 0.5 * (1.0 + np.cos(2.0 * np.pi * (x - 0.5))))
    elif etype == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _interp_profile(p0, n_new):
    p0 = np.asarray(p0, dtype=np.float64).ravel()
    n0 = len(p0)
    if n0 == n_new:
        return p0 / p0.sum()
    x0 = np.linspace(0.0, 1.0, n0)
    x1 = np.linspace(0.0, 1.0, n_new)
    p = np.interp(x1, x0, p0)
    p = np.maximum(p, 1e-12)
    return p / p.sum()


def _run_pnorm_lbfgs(a_init, deadline, p_list=None, maxiter_per_p=2000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    if p_list is None:
        p_list = [2, 4, 8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192]
    for p in p_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _pnorm_obj, z, args=(p, n),
                method='L-BFGS-B', jac=_pnorm_grad,
                options={'maxiter': maxiter_per_p, 'ftol': 1e-20,
                         'gtol': 1e-12, 'maxls': 80, 'maxfun': 8000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _run_lse_lbfgs(a_init, deadline, beta_list=None, maxiter_per_beta=2000):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _score_norm_fn(a_init)
    if beta_list is None:
        c0 = _full_convolve(a_init, a_init)
        m0 = float(c0.max())
        frac_list = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002,
                     0.001, 0.0005, 0.0003, 0.0001, 0.00005, 0.00001]
        beta_list = [f * m0 for f in frac_list]
    for beta in beta_list:
        if time.monotonic() > deadline:
            break
        try:
            result = minimize(
                _lse_obj, z, args=(beta, n),
                method='L-BFGS-B', jac=_lse_grad,
                options={'maxiter': maxiter_per_beta, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxls': 80, 'maxfun': 8000}
            )
            if result.x is not None and np.all(np.isfinite(result.x)):
                a_new = _softmax_fn(result.x)
                s_new = _score_norm_fn(a_new)
                if np.isfinite(s_new) and s_new < best_score:
                    best_score = s_new
                    best_a = a_new.copy()
                    z = result.x.copy()
        except Exception:
            pass
    return best_a, best_score


def _subgradient_phase(a_init, deadline, max_iter=80000):
    n = len(a_init)
    a = a_init.copy()
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    f_lower = 1.0 / (2.0 * n - 1.0)
    avg_a = a.copy()
    count = 0
    no_improve = 0
    check_interval = 100
    rng = np.random.default_rng(777)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(y, y)
        k_star = int(c.argmax())
        M = float(c[k_star])
        g = np.zeros(n)
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] = 2.0 * y[k_star - j_lo:k_star - j_lo + (j_hi - j_lo + 1)]
        g_norm_sq = float(np.dot(g, g))
        if g_norm_sq < 1e-30:
            break
        gap = M - f_lower
        if gap < 1e-15:
            break
        lr = gap / g_norm_sq
        lr = min(lr, 0.3)
        z_step = y - lr * g
        a_new = _project_simplex(z_step)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + (t_mom / t_new - 1.0) * (a_new - a)
        y = _project_simplex(y)
        a = a_new
        t_mom = t_new
        count += 1
        avg_a = avg_a * (1.0 - 1.0 / count) + a_new / count
        if it % check_interval == 0:
            sc = _score_norm_fn(a)
            sc_avg = _score_norm_fn(avg_a) if count > 50 else np.inf
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            elif sc_avg < best_score - 1e-14:
                best_score = sc_avg
                best_a = avg_a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50:
                y = a + 0.005 * rng.standard_normal(n)
                y = _project_simplex(y)
                t_mom = 1.0
                no_improve = 0
    return best_a, best_score


def _lp_blend(a_init, deadline, K_list=None, max_rounds=5):
    n = len(a_init)
    if K_list is None:
        K_list = [30, 60]
    current = a_init.copy()
    best_score = _score_norm_fn(current)
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        c = _full_convolve(current, current)
        M = float(c.max())
        sorted_idx = np.argsort(c)[::-1]
        for K in K_list:
            if time.monotonic() > deadline - 10:
                break
            active = sorted_idx[:K]
            A_mat = np.zeros((len(active), n))
            for row, k in enumerate(active):
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_lo <= i_hi:
                    A_mat[row, i_lo:i_hi + 1] = current[k - i_lo:k - i_lo + (i_hi - i_lo + 1)]
            b_vec = np.full(len(active), M * 1.002)
            try:
                res = linprog(-np.ones(n), A_ub=A_mat, b_ub=b_vec,
                              bounds=[(0, None)] * n, method='highs',
                              options={'time_limit': 15})
                if res.status == 0 and res.x is not None:
                    b_new = res.x
                    sum_b = float(b_new.sum())
                    if sum_b > 1e-15:
                        b_scaled = b_new * (current.sum() / sum_b)
                        for t_blend in [0.5, 0.3, 0.7, 1.0, 0.2]:
                            a_trial = (1.0 - t_blend) * current + t_blend * b_scaled
                            a_trial = np.maximum(a_trial, 0.0)
                            s = a_trial.sum()
                            if s < 1e-15:
                                continue
                            a_trial = a_trial / s
                            sc_trial = _score_norm_fn(a_trial)
                            if sc_trial < best_score - 1e-14:
                                current = a_trial
                                best_score = sc_trial
                                M = float(c.max())
                                break
            except Exception:
                pass
    return current, best_score


def _coord_descent(a_init, deadline, max_rounds=30000):
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_score = _score_norm_fn(a)
    rng = np.random.default_rng(999)
    c_cur = _full_convolve(a, a)
    M_cur = float(c_cur.max())
    stagnation = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        j = int(rng.integers(n))
        i = j
        attempts = 0
        while i == j and attempts < 10:
            i = int(rng.integers(n))
            attempts += 1
        if i == j:
            continue
        eps_list = [0.002, 0.005, 0.01, 0.02, 0.001, -0.002, -0.005, -0.01]
        improved = False
        for eps in eps_list:
            if a[j] + eps < 0 or a[i] - eps < 0:
                continue
            a[j] += eps
            a[i] -= eps
            delta = np.zeros(2 * n - 1)
            for k in range(max(0, j - n + 1), min(n, j + n)):
                idx = k
                if idx >= 0 and idx < 2 * n - 1:
                    delta[idx] += eps * (a[k - j] + (a[j + n - 1 - (k - j)] if 0 <= k - j < n else 0.0))
                    delta[idx] -= eps * (a[k - i] + (a[i + n - 1 - (k - i)] if 0 <= k - i < n else 0.0))
            M_test = float(np.max(c_cur + 2.0 * delta))
            if M_test < M_cur - 1e-15:
                c_cur = c_cur + 2.0 * delta
                M_cur = M_test
                improved = True
                break
            else:
                a[j] -= eps
                a[i] += eps
        if improved:
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 500:
                a[j] += 0.003 * rng.standard_normal()
                a[i] -= 0.003 * rng.standard_normal()
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                c_cur = _full_convolve(a, a)
                M_cur = float(c_cur.max())
                stagnation = 0
        if rnd % 2000 == 0:
            sc = _score_norm_fn(a)
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization: p-norm continuation L-BFGS, subgradient with
    Polyak steps, LP blending, and coordinate descent. Targets score < 1.5027.
    """
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = np.inf

    # Load parent
    parent = None
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent_raw.size > 0:
            parent_raw = np.clip(parent_raw, 0.0, 1000.0)
            s = parent_raw.sum()
            if s > 0.01:
                parent = parent_raw / s
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        print(f"Loaded parent: n={n_parent}, score={_score_norm_fn(parent):.10f}")
        best_seq = parent.copy()
        best_score = _score_norm_fn(parent)
    else:
        n_parent = 4096
        best_seq = _greedy_init(n_parent)
        best_score = _score_norm_fn(best_seq)
        print(f"Default init: n={n_parent}, score={best_score:.10f}")

    # === Phase 1: Coarse p-norm L-BFGS on parent + interpolated 2n (0-350s) ===
    print("Phase 1: Coarse p-norm L-BFGS (multiple n)")
    p1_end = min(deadline, t_start + 350.0)

    n_list_to_try = [n_parent]
    if n_parent <= 8192:
        n_list_to_try.append(2 * n_parent)
    if 16384 not in n_list_to_try and n_parent < 16384:
        n_list_to_try.append(16384)

    time_per_n = (p1_end - t_start) / len(n_list_to_try)

    for n_try in n_list_to_try:
        if time.monotonic() > p1_end - 30:
            break
        if n_try == n_parent:
            a_init = best_seq.copy()
        else:
            a_init = _interp_profile(best_seq, n_try)
        t_budget = min(p1_end - 10, time.monotonic() + time_per_n)
        coarse_p = [2, 4, 8, 16, 32, 64, 128]
        a_opt, sc = _run_pnorm_lbfgs(a_init, t_budget, p_list=coarse_p, maxiter_per_p=1500)
        if sc < best_score:
            best_seq = a_opt.copy()
            best_score = sc
            print(f"  Phase1 n={n_try}: score={sc:.10f} [improved]")
        else:
            print(f"  Phase1 n={n_try}: score={sc:.10f} [no improve, best={best_score:.10f}]")

    print(f"  Phase 1 done: best={best_score:.10f}, n={len(best_seq)}")

    # === Phase 2: Fine p-norm L-BFGS (350s - 700s) ===
    print("Phase 2: Fine p-norm L-BFGS")
    p2_end = min(deadline, t_start + 700.0)
    if best_seq is not None and time.monotonic() < p2_end:
        fine_p = [128, 256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]
        a_fine, sc_fine = _run_pnorm_lbfgs(best_seq, p2_end, p_list=fine_p, maxiter_per_p=2000)
        if sc_fine < best_score:
            best_seq = a_fine.copy()
            best_score = sc_fine
            print(f"  Phase2: score={best_score:.10f} [improved]")
        else:
            print(f"  Phase2: score={sc_fine:.10f} [no improve]")

    print(f"  Phase 2 done: best={best_score:.10f}")

    # === Phase 3: Log-sum-exp L-BFGS with very small beta (700s - 950s) ===
    print("Phase 3: LSE L-BFGS tiny beta")
    p3_end = min(deadline, t_start + 950.0)
    if best_seq is not None and time.monotonic() < p3_end:
        c_b = _full_convolve(best_seq, best_seq)
        m_b = float(c_b.max())
        tiny_betas = [m_b * f for f in [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]]
        a_lse, sc_lse = _run_lse_lbfgs(best_seq, p3_end, beta_list=tiny_betas, maxiter_per_beta=2000)
        if sc_lse < best_score:
            best_seq = a_lse.copy()
            best_score = sc_lse
            print(f"  Phase3: score={best_score:.10f} [improved]")
        else:
            print(f"  Phase3: score={sc_lse:.10f} [no improve]")

    print(f"  Phase 3 done: best={best_score:.10f}")

    # === Phase 4: Subgradient with Polyak + averaging (950s - 1250s) ===
    print("Phase 4: Subgradient")
    p4_end = min(deadline, t_start + 1250.0)
    if best_seq is not None and time.monotonic() < p4_end:
        a_sg, sc_sg = _subgradient_phase(best_seq, p4_end, max_iter=100000)
        if sc_sg < best_score:
            best_seq = a_sg.copy()
            best_score = sc_sg

    # === Phase 5: LP blending + quick polish (1250s - 1500s) ===
    print("Phase 5: LP blending")
    p5_end = min(deadline, t_start + 1500.0)
    if best_seq is not None and time.monotonic() < p5_end:
        a_lp, sc_lp = _lp_blend(best_seq, min(p5_end - 60, time.monotonic() + 150),
                                K_list=[30, 60], max_rounds=4)
        if sc_lp < best_score:
            best_seq = a_lp.copy()
            best_score = sc_lp
            print(f"  Phase5 LP: score={best_score:.10f} [improved]")
        else:
            print(f"  Phase5 LP: score={sc_lp:.10f} [no improve]")
        # Quick p-norm polish
        if time.monotonic() < p5_end - 10:
            a_pol, sc_pol = _run_pnorm_lbfgs(best_seq, p5_end,
                                              p_list=[2048, 4096, 8192],
                                              maxiter_per_p=1500)
            if sc_pol < best_score:
                best_seq = a_pol.copy()
                best_score = sc_pol
                print(f"  Phase5 polish: score={best_score:.10f} [improved]")

    print(f"  Phase 5 done: best={best_score:.10f}")

    # === Phase 6: Coordinate descent + final subgradient (1500s - 1750s) ===
    print("Phase 6: Coord descent + final subgradient")
    p6_end = deadline - 20
    if best_seq is not None and time.monotonic() < p6_end:
        a_cd, sc_cd = _coord_descent(best_seq, min(p6_end - 80, time.monotonic() + 120),
                                     max_rounds=40000)
        if sc_cd < best_score:
            best_seq = a_cd.copy()
            best_score = sc_cd
            print(f"  Phase6 CD: score={best_score:.10f} [improved]")
        else:
            print(f"  Phase6 CD: score={sc_cd:.10f} [no improve]")

        if time.monotonic() < p6_end - 10:
            a_sg2, sc_sg2 = _subgradient_phase(best_seq, p6_end, max_iter=50000)
            if sc_sg2 < best_score:
                best_seq = a_sg2.copy()
                best_score = sc_sg2
                print(f"  Phase6 SG: score={best_score:.10f} [improved]")

    print(f"  Phase 6 done: best={best_score:.10f}")

    # === Phase 7: Final tiny LSE polish if time remains ===
    if best_seq is not None and time.monotonic() < deadline - 5:
        c_f = _full_convolve(best_seq, best_seq)
        m_f = float(c_f.max())
        final_betas = [m_f * f for f in [0.0001, 0.00005, 0.00002, 0.00001]]
        a_final, sc_final2 = _run_lse_lbfgs(best_seq, deadline - 2,
                                             beta_list=final_betas, maxiter_per_beta=1000)
        if sc_final2 < best_score:
            best_seq = a_final.copy()
            best_score = sc_final2
            print(f"  Phase7: score={best_score:.10f} [improved]")

    # === Final validation ===
    if best_seq is not None:
        final_seq = np.clip(best_seq, 0.0, 1000.0)
        s = float(final_seq.sum())
        if s > 0.01:
            final_seq = final_seq / s
        else:
            final_seq = _greedy_init(512)

        sc_final = _score_fn(final_seq)
        print(f"FINAL: score={sc_final:.10f}, n={len(final_seq)}")

        if len(final_seq) <= 20000:
            try:
                sc_eval = float(evaluate_sequence([float(x) for x in final_seq]))
                if np.isfinite(sc_eval):
                    print(f"Evaluator confirmed: {sc_eval:.10f}")
                else:
                    print("Evaluator returned inf, using fallback")
                    fb = _greedy_init(512)
                    return [float(x) for x in fb]
            except Exception as ex:
                print(f"Eval error: {ex}")

        return [float(x) for x in final_seq]

    fb = _greedy_init(512)
    print(f"FALLBACK: n=512, score={_score_fn(fb):.10f}")
    return [float(x) for x in fb]
```
