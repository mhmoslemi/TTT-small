    a = softmax(z)
    c = convolve(a, a)  # full convolution
    # Compute log-sum-exp max
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    w = e / e.sum()
    smoothed_max = c_max + beta * np.log(e.sum())
    F = 2 * n * smoothed_max
    
    # Gradient of F with respect to a
    # g[i] = 2 * sum_k w[k] * a[k-i] (with boundaries)

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing and beta annealing as the primary method,
combined with LP-based direction finding and multi-restart strategies.

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _true_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_w(c, beta):
    """Compute softmax weights w[k] = exp(c[k]/beta) / sum."""
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _smoothed_obj_and_grad(z, beta, n):
    """Compute smoothed objective and gradient with respect to z."""
    a = _softmax(z)
    c = _full_convolve(a, a)
    w = _compute_w(c, beta)
    smoothed_max = beta * np.log(np.sum(np.exp((c - c.max()) / beta)) + c.max() / beta)
    # Actually, let me use the stable form:
    c_max = c.max()
    log_sum_exp = c_max + beta * np.log(np.sum(np.exp((c - c_max) / beta)))
    F = 2.0 * n * log_sum_exp
    
    # Gradient: g[i] = 2 * sum_k w[k] * a[k-i]
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    g = 2.0 * conv_aw[n - 1: 2 * n - 1][::-1]
    
    # Gradient with respect to z
    za = float(np.dot(g, a))
    grad = a * (za - g)
    
    return F, grad


def _lbfgs_round(a_init, beta, max_iter=500, tol=1e-10):
    """One round of L-BFGS optimization at fixed beta."""
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()
    
    def fun_wrapped(z_):
        F, grad = _smoothed_obj_and_grad(z_, beta, n)
        return F, grad
    
    result = minimize(fun_wrapped, z, jac=True, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8,
                               'maxfun': max_iter * 3})
    
    if result.success and result.x is not None:
        return _softmax(result.x), result.fun
    else:
        return _softmax(z), _true_score(_softmax(z))

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses L-BFGS with log-sum-exp smoothing and beta annealing as primary method,
combined with reduced-constraint LP direction finding and multi-restart strategies.
Exploits FFT for O(n log n) convolution and gradient computation.
"""
import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def _next_pow2(n):
    k = 1
    while k < n:
        k <<= 1
    return max(2, k)


def _full_convolve(a, b):
    n, m = len(a), len(b)
    out_n = n + m - 1
    N = _next_pow2(out_n)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    r = np.fft.irfft(fa * fb, N)[:out_n]
    return r


def _softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return e / s


def _true_score(a):
    n = len(a)
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    c = _full_convolve(a, a)
    return float(2.0 * n * c.max() / (s * s))


def _compute_w(c, beta):
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return w


def _smoothed_obj_grad_z(z, beta, n):
    a = _softmax_fn(z)
    c = _full_convolve(a, a)
    c_max = c.max()
    shifted = (c - c_max) / beta
    e = np.exp(shifted)
    es = e.sum()
    if es < 1e-300:
        return np.inf, np.zeros(n)
    log_sum_exp = c_max + beta * np.log(es)
    F = 2.0 * n * log_sum_exp
    w = e / es
    w_rev = w[::-1]
    conv_aw = _full_convolve(a, w_rev)
    g = 2.0 * conv_aw[n - 1: 2 * n - 1][::-1]
    za = float(np.dot(g, a))
    grad = a * (za - g)
    if not np.all(np.isfinite(grad)):
        return np.inf, np.zeros(n)
    return F, grad


def _lbfgs_round(a_init, beta, n, max_iter):
    z = np.log(np.maximum(a_init, 1e-30))
    z = z - z.mean()

    def _fun_wrap(z_):
        F, grad = _smoothed_obj_grad_z(z_, beta, n)
        if not np.isfinite(F):
            return 1e10, np.zeros(n)
        return F, grad

    result = minimize(_fun_wrap, z, jac=True, method='L-BFGS-B',
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                               'maxfun': max_iter * 5, 'maxls': 50})
    x_final = result.x if result.x is not None else z
    return _softmax_fn(x_final)


def _annealing_optimize(a_init, deadline, t_start, beta_start=0.5, beta_end=0.001,
                        beta_decay=0.72, iters_per_round=400):
    n = len(a_init)
    best_a = a_init.copy()
    best_score = _true_score(a_init)
    beta = beta_start
    round_num = 0
    while time.monotonic() < deadline - 5 and beta > beta_end:
        round_num += 1
        a_new = _lbfgs_round(a_init, beta, n, max_iter=iters_per_round)
        a_init = a_new
        sc = _true_score(a_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        beta *= beta_decay
        if round_num % 3 == 0:
            print(f"    Anneal round {round_num}: beta={beta:.6f}, score={best_score:.10f}, "
                  f"elapsed={time.monotonic()-t_start:.1f}s")
    return best_a, best_score


def _build_lp_sparse(f, n, active_set):
    k_vals = np.array(active_set, dtype=int)
    j = np.arange(n)
    k_col = k_vals[:, np.newaxis]
    j_row = j[np.newaxis, :]
    diff = k_col - j_row
    mask = (diff >= 0) & (diff < n)
    rows = np.repeat(np.arange(len(k_vals)), n)[mask.ravel()]
    cols = np.tile(j, len(k_vals))[mask.ravel()]
    vals = f[diff[mask].ravel()]
    return csr_matrix((vals, (rows, cols)), shape=(len(k_vals), n))


def _lp_descent_step(f, deadline, t_start, n, top_k=50, time_per_lp=5.0):
    c = _full_convolve(f, f)
    M = float(c.max())
    sc = _true_score(f)
    print(f"  LP phase start: M={M:.12f}, score={sc:.10f}")
    for step in range(100):
        if time.monotonic() > deadline - 5:
            break
        c = _full_convolve(f, f)
        M = float(c.max())
        # Find top-k active set
        top_k_actual = min(top_k, 2 * n - 1)
        active_set = np.argsort(c)[::-1][:top_k_actual].tolist()
        # Build sparse matrix
        try:
            A = _build_lp_sparse(f, n, active_set)
        except Exception:
            break
        try:
            c_obj = -np.ones(n)
            b_ub = np.full(len(active_set), M)
            bounds_lp = [(0.0, None)] * n
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds_lp,
                          method='highs', options={'time_limit': time_per_lp,
                                                   'presolve': True})
            if res.status != 0 or res.x is None:
                break
            g = np.maximum(res.x, 0)
        except Exception:
            break
        if g.sum() < 1e-15:
            break
        g_norm = g / g.sum()
        # Backtracking line search
        t_try = 0.3
        improved = False
        for _ in range(8):
            f_new = (1 - t_try) * f + t_try * g_norm
            f_new = np.maximum(f_new, 0)
            s_new = f_new.sum()
            if s_new < 1e-15:
                t_try *= 0.5
                continue
            f_new = f_new / s_new
            c_new = _full_convolve(f_new, f_new)
            M_new = float(c_new.max())
            if M_new < M * (1 - 1e-10):
                f = f_new
                M = M_new
                t_try = min(1.0, t_try * 1.3)
                improved = True
                if step % 5 == 0:
                    sc_now = _true_score(f)
                    print(f"  LP step {step}: M={M:.12f}, score={sc_now:.10f}, "
                          f"elapsed={time.monotonic()-t_start:.1f}s")
                break
            else:
                t_try *= 0.5
        if not improved:
            t_try *= 0.5
            if t_try < 0.001:
                break
    return f


def _perturb_fn(a, sigma, rng):
    z = np.log(np.maximum(a, 1e-15))
    z = z + sigma * rng.standard_normal(len(a))
    z = z - z.mean()
    return _softmax_fn(z)


def _interp_profile_fn(p, n_new):
    n0 = len(p)
    if n0 == n_new:
        return p.copy()
    x0 = np.linspace(0, 1, n0)
    x1 = np.linspace(0, 1, n_new)
    p_new = np.interp(x1, x0, p)
    p_new = np.maximum(p_new, 1e-12)
    return p_new / p_new.sum()


def _make_gaussian(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    center = 0.5
    a = np.exp(-((x - center) ** 2) / (2 * (sigma_frac * 0.5) ** 2))
    return a / a.sum()


def _make_flat_top(n):
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x) ** 2
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def _adam_finetune(a_init, deadline, t_start, lr=0.005):
    n = len(a_init)
    z = np.log(np.maximum(a_init, 1e-15))
    z = z - z.mean()
    best_a = a_init.copy()
    best_score = _true_score(a_init)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    beta_scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    max_c_init = float(_full_convolve(a_init, a_init).max())
    current_lr = lr
    for beta_scale in beta_scales:
        if time.monotonic() > deadline - 5:
            break
        beta = beta_scale * max_c_init
        iters = 0
        while iters < 20000 and time.monotonic() < deadline - 5:
            iters += 1
            t_step += 1
            a = _softmax_fn(z)
            c = _full_convolve(a, a)
            cm = float(c.max())
            c_shifted = (c - cm) / beta
            e = np.exp(c_shifted)
            s_e = float(e.sum())
            if s_e < 1e-300:
                w = np.zeros_like(c)
                w[int(np.argmax(c))] = 1.0
            else:
                w = e / s_e
            w_rev = w[::-1]
            conv_aw = _full_convolve(a, w_rev)
            g = 2.0 * conv_aw[n - 1: 2 * n - 1][::-1]
            za = float(np.dot(g, a))
            grad = a * (za - g)
            if not np.all(np.isfinite(grad)):
                break
            m_buf = b1 * m_buf + (1 - b1) * grad
            v_buf = b2 * v_buf + (1 - b2) * (grad * grad)
            mh = m_buf / (1 - b1 ** t_step)
            vh = v_buf / (1 - b2 ** t_step)
            update = current_lr * mh / (np.sqrt(vh) + eps)
            un = np.linalg.norm(update)
            if un > 0.3:
                update = update * (0.3 / un)
            z = z - update
            z = np.clip(z, -50, 50)
            z = z - z.mean()
            if t_step % 200 == 0:
                sc = _true_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a.copy()
            if iters % 5000 == 0:
                current_lr = max(0.001, current_lr * 0.9)
    a_final = _softmax_fn(z)

                                              beta_start=0.3, beta_end=0.005,
                                              beta_decay=0.7, iters_per_round=200)
                _update_best(a_r)
        print(f"  Phase 5 done: best={best_score:.10f}, t={time.monotonic()-t_start:.1f}s")

    # === Phase 6: Final Adam polish (1720s - deadline) ===
    print(f"\n=== Phase 6: Final Adam polish ===")
    if best_seq is not None and time.monotonic() < deadline - 15:
        a_best = best_seq / best_seq.sum()
        remaining = deadline - time.monotonic() - 10
        if remaining > 20:
            a_final, _ = _adam_finetune(a_best, time.monotonic() + remaining, t_start, lr=0.003)
            _update_best(a_final)

    print(f"\n=== FINAL: score={best_score:.10f}, n={len(best_seq) if best_seq is not None else 0}, "
          f"t={time.monotonic()-t_start:.1f}s ===")

    if best_seq is not None:
        final = np.clip(best_seq, 0.0, 1000.0)
        s = float(final.sum())
        if s < 0.01:
            final = np.ones(256) / 256.0
        try:
            sc_eval = float(evaluate_sequence([float(x) for x in final]))
            print(f"  Evaluator score: {sc_eval:.10f}")
        except Exception as ex:
            print(f"  Eval check error: {ex}")
        return [float(x) for x in final]

    fallback = np.ones(256) / 256.0
    print(f"  FALLBACK: score={_true_score(fallback):.10f}")
    return [float(x) for x in fallback]
